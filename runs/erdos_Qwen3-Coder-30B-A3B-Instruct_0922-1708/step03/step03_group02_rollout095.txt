import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_init))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_init)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    else:
        h = np.ones(n_points) * 0.5
        population.append(h)
    
    # Generate diverse initial population
    for _ in range(pop_size - 1):
        h = np.random.uniform(0.2, 0.8, n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Selection (tournament)
        new_population = []
        elite_count = int(elite_rate * pop_size)
        elite_indices = np.argsort(fitness)[:elite_count]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent1_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            if np.random.random() < cross_rate:
                parent2_idx = tournament_indices[np.argmin([fitness[i] for i in tournament_indices if i != parent1_idx])]
                # Crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([population[parent1_idx][:crossover_point], 
                                       population[parent2_idx][crossover_point:]])
                child2 = np.concatenate([population[parent2_idx][:crossover_point], 
                                       population[parent1_idx][crossover_point:]])
                # Mutation
                if np.random.random() < mut_rate:
                    mut_idx = np.random.randint(n_points)
                    child1[mut_idx] += np.random.normal(0, 0.1)
                if np.random.random() < mut_rate:
                    mut_idx = np.random.randint(n_points)
                    child2[mut_idx] += np.random.normal(0, 0.1)
            else:
                child1 = population[parent1_idx].copy()
                if np.random.random() < mut_rate:
                    mut_idx = np.random.randint(n_points)
                    child1[mut_idx] += np.random.normal(0, 0.1)
                child2 = None
            
            # Ensure constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            new_population.append(child1)
            
            if child2 is not None:
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                new_population.append(child2)
        
        # Trim to exact population size
        population = new_population[:pop_size]
    
    # Final check
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
    else:
        final_c5 = best_c5
        if best_h is None:
            final_c5 = float('inf')
            # Reinitialize best to be safe
            best_h = population[np.argmin(fitness)].copy()
    
    if final_c5 > best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
