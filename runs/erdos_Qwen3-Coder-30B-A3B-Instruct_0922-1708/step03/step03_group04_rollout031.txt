import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    eps = 1e-5
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Helper function to compute gradient for a specific shift
    def compute_gradient_for_shift(h_array, shift):
        n = len(h_array)
        grad = np.zeros(n)
        # For shift k (as index), gradient is 1 - h[(i - k) % n] + h[(i + k) % n]
        # But with careful handling of circular indices
        for i in range(n):
            # When we compute the shift for i with offset k,
            # it's h[i] * (1 - h[(i - k) % n])
            # So the gradient w.r.t. h[i] is (1 - h[(i - k) % n])
            # But we need to be careful since we're using forward correlation
            # so gradient is h[(i+k) % n] - 1 for correct cross-correlation
            # Compute cross-correlation
            if shift >= 0:
                pos = (i - shift) % n
            else:
                pos = (i + shift) % n
            grad[i] = -(1 - h_array[pos])
        return grad
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute full correlation to find the max
        correlation = np.correlate(h, 1 - h, mode='full')
        max_shift_idx = np.argmax(correlation)
        # The full correlation has length 2*n - 1, so indices go from -n+1 to n-1
        shift = max_shift_idx - (n_points - 1)  # This gives us actual shift in [0, 2)
        
        # Compute true gradient for this specific maximum
        grad = compute_gradient_for_shift(h, shift)
        
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        if iteration % 200 == 0:
            # Occasionally random restart
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                if len(h0) != n_points:
                    x_orig = np.linspace(0, 2, len(h0))
                    x_new = np.linspace(0, 2, n_points)
                    h0 = np.interp(x_new, x_orig, h0)
                h0 = np.clip(h0, 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
                h = h0
            else:
                h = np.ones(n_points) * 0.5
                
    return (best_h.tolist(), best_c5, n_points)
