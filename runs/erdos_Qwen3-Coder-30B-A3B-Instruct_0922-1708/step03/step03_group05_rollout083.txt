import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h"""
    correlation = correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def project_h(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize swarm
    num_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    
    # Initialize positions and velocities
    if 'initial_h_values' in globals():
        particles = []
        # Seed 10% with initial construction
        for i in range(num_particles//10):
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h)
            particles.append(h)
        # Fill remaining with random
        for i in range(num_particles//10, num_particles):
            h = np.random.rand(n_points)
            h = project_h(h, n_points, dx)
            particles.append(h)
    else:
        particles = [project_h(np.random.rand(n_points), n_points, dx) for _ in range(num_particles)]
    
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(num_particles)]
    
    # Track personal bests
    p_best = [particles[i].copy() for i in range(num_particles)]
    p_best_scores = [evaluate_h(particles[i], dx) for i in range(num_particles)]
    
    # Global best
    g_best_idx = np.argmin(p_best_scores)
    g_best = p_best[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (p_best[i] - particles[i]) +
                c2 * r2 * (g_best - particles[i])
            )
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_h(particles[i], n_points, dx)
            
            # Evaluate
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best_scores[i] = score
                p_best[i] = particles[i].copy()
                
                # Update global best
                if score < g_best_score:
                    g_best_score = score
                    g_best = particles[i].copy()
    
    # Final validation and return
    final_score = evaluate_h(g_best, dx)
    return (g_best.tolist(), final_score, n_points)
