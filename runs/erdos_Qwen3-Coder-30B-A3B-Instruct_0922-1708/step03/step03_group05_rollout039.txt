import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points

        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Physics-inspired relaxation parameters
    lambda_reg = 0.1
    damping_factor = 0.99
    max_no_improvement = 500
    no_improvement_count = 0

    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Physics-inspired relaxed convex optimization
        # Build a convex optimization problem for approximate solution
        h_relaxed = cp.Variable(n_points)
        h_sum = cp.sum(h_relaxed)
        
        # Compute overlap as a quadratic form
        # For the overlap ∫ h(x)(1 - h(x+k)) dx, we approximate it with a quadratic penalty
        # Note: this approach approximates trying to minimize sum_{i,k} h[i] * (1 - h[i+k]) * dx
        
        # We'll do a more precise relaxation and use a quadratic penalty
        h_total = cp.sum(h_relaxed)
        objective = cp.Minimize(h_total)
        constraints = [h_total == n_points / 2]
        
        # Add regularization to smoother the solution
        reg_term = cp.sum_squares(h_relaxed - np.mean(h_relaxed))
        objective = cp.Minimize(h_total + lambda_reg * reg_term)
        
        # Solve the convex relaxation
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_new = h_relaxed.value
        except:
            h_new = h.copy()  # fallback copy
            
        # Apply physics-based perturbations
        # Add oscillatory perturbation
        dt = 0.01
        perturbation = 0.01 * np.sin(iteration + np.arange(n_points)) * dt
        h_new = h_new + perturbation
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)        
        
        # Update for next iteration
        h = h_new
        
        # Decay regularization strength
        if iteration % 100 == 0:
            lambda_reg *= damping_factor
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
