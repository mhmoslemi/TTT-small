import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    temperature = 0.1
    success_count = 0
    total_attempts = 0
    
    # Define resolver scales for multi-resolution search
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for resolver in resolvers:
        if time.time() - start_time > budget_s - 1:
            break
            
        num_iterations = 1000
        for i in range(num_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate perturbed candidate
            perturbation = np.random.normal(0, resolver, n_points)
            h_candidate = h + perturbation
            
            # Clip and rescale
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute overlap
            overlap = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(overlap) * dx
            
            # Accept or reject
            if c5_candidate < best_c5:
                h = h_candidate
                best_c5 = c5_candidate
                success_count += 1
                resolver = max(resolver * 1.1, 0.001)  # Adaptive increase
            else:
                # Accept with probability based on temperature
                delta = c5_candidate - best_c5
                if delta > 0 and np.random.rand() < np.exp(-delta / temperature):
                    h = h_candidate
                    # No update to best_c5
                    success_count += 1
                
            total_attempts += 1
            
            # Reduce temperature
            if i % 100 == 0:
                temperature *= 0.99
                temperature = max(temperature, 1e-6)
                
            # Adaptive perturbation adjustment
            if i % 50 == 0 and i > 0:
                if total_attempts > 0:
                    success_ratio = success_count / total_attempts
                    if success_ratio > 0.2:
                        resolver = min(resolver * 1.5, 1.0)
                    elif success_ratio < 0.1:
                        resolver = max(resolver * 0.5, 0.001)
                    success_count = 0
                    total_attempts = 0
        
        # Additional fine-tuning at the end of each resolver
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            perturbation = np.random.normal(0, resolver/10, n_points)
            h_candidate = h + perturbation
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            overlap = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(overlap) * dx
            
            if c5_candidate < best_c5:
                h = h_candidate
                best_c5 = c5_candidate
    
    return (best_h.tolist(), best_c5, n_points)
