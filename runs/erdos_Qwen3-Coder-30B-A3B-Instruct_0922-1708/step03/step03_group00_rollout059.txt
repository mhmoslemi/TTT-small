import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 200
    
    # Initialize h from initial_h_values or use uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to meet the constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Discretize shifts from 0 to 2 with 200 points (enough resolution)
    shifts = np.linspace(0, 2, 200)
    
    start_time = time.time()
    
    # Iterative constraint addition approach
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps to find the maximum
        max_overlap = -1.0
        violating_shift = 0
        for k in shifts:
            # Compute integral for this shift
            # h(x) * (1 - h(x + k)) -> convolve h with (1 - h) shifted
            # But since we're working with discrete samples, compute manually
            overlap = 0.0
            # Since h is defined over [0,2], and we're looking at h(x+k)
            # we must consider what happens when x+k > 2
            # This is handled by wrapping or truncating, but here we assume
            # in this case of discrete sampling we handle periodicity to a degree
            # Simple discrete version
            h_shifted = np.roll(h, int(k / dx))
            o = np.sum(h * (1 - h_shifted)) * dx
            if o > max_overlap:
                max_overlap = o
                violating_shift = k
        
        # Check if we found a better solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Add constraint corresponding to this worst shift
        # The constraint would be: \int h(x) * (1 - h(x + k)) dx <= C
        # Since we're using cvxpy, add a constraint to the problem
        
        # Re-solve as convex optimization problem with the constraint
        # We'll model this using cvxpy
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Objective
        objective = cp.Minimize(C)
        
        constraints = []
        
        # Bounds on h
        constraints.extend([h_vars[i] >= 0 for i in range(n_points)])
        constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
        
        # Mass constraint
        constraints.append(cp.sum(h_vars) * dx == 1)
        
        # Constraint based on current worst-case overlap
        # We create a constraint based on the current h and the violation
        h_shifted_new = np.roll(h, int(violating_shift / dx))
        # approximate integral as sum
        integral_approx = cp.sum(cp.multiply(h_vars, 1 - h_shifted_new)) * dx
        constraints.append(integral_approx <= C)
        
        # Objective function to minimize C, subject to the various constraints
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                # Update h using the optimized values
                h_new = h_vars.value
                if h_new is not None:
                    # Normalize if needed
                    h = h_new.copy()
                    h = h * (n_points / 2) / np.sum(h)
        except Exception:
            # If optimization fails, proceed with original h.
            pass
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
