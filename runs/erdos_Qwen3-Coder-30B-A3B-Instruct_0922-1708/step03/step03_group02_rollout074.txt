import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h"""
    correlation = correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def normalize_h(h, n_points):
    """Ensure h sums to n_points/2 and stays in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def mutate_individual(h, mut_rate, n_points):
    """Apply mutation to an individual"""
    h = h.copy()
    mask = np.random.random(n_points) < mut_rate
    h[mask] += np.random.normal(0, 0.1, np.sum(mask))
    return normalize_h(h, n_points)

def crossover_individuals(h1, h2, cross_rate):
    """Perform crossover between two individuals"""
    if np.random.random() > cross_rate:
        return h1.copy(), h2.copy()
    
    crossover_point = np.random.randint(1, len(h1))
    h1_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h2_new = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    
    return h1_new, h2_new

def tournament_selection(population, fitness, tournament_size=5):
    """Select an individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_index].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_size = int(0.2 * pop_size)
    mut_rate = 0.1
    cross_rate = 0.7
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = normalize_h(h_init, n_points)
    else:
        h_init = np.ones(n_points) * 0.5
        
    # Create initial population
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        h = h_init + np.random.normal(0, 0.1, n_points)
        population.append(normalize_h(h, n_points))
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Evaluate fitness of entire population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Find best individual in current population
        best_in_gen_idx = np.argmin(fitness)
        best_in_gen_c5 = fitness[best_in_gen_idx]
        
        if best_in_gen_c5 < best_c5:
            best_c5 = best_in_gen_c5
            best_h = population[best_in_gen_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate rest of population
        while len(new_population) < pop_size:
            # Tournament selection for parents
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            offspring1, offspring2 = crossover_individuals(parent1, parent2, cross_rate)
            
            # Mutation
            offspring1 = mutate_individual(offspring1, mut_rate, n_points)
            offspring2 = mutate_individual(offspring2, mut_rate, n_points)
            
            new_population.extend([offspring1, offspring2])
            
        # Trim to exact population size
        population = new_population[:pop_size]
        generation += 1
    
    # Final check with the best solution
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
