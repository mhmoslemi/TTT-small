import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    num_particles = 50
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.extend([h_init] * min(5, num_particles))
        velocities.extend([np.random.uniform(-0.1, 0.1, n_points) for _ in range(min(5, num_particles))])
    
    # Fill remaining particles with random feasible solutions
    remaining = num_particles - len(particles)
    for _ in range(remaining):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    
    # Track best solutions
    p_best = [p.copy() for p in particles]
    p_best_values = [float('inf')] * num_particles
    g_best = None
    g_best_value = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate all particles
        for i, particle in enumerate(particles):
            # Compute overlap integral
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < p_best_values[i]:
                p_best_values[i] = c5
                p_best[i] = particle.copy()
                
            if c5 < g_best_value:
                g_best_value = c5
                g_best = particle.copy()
        
        # Update particles
        for i in range(num_particles):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (g_best - particles[i]))
            
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Final evaluation
    correlation = np.correlate(g_best, 1 - g_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (g_best.tolist(), final_c5, n_points)
