import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency, skip 0 since it's always 0
    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1:
        # Try convex relaxation approach
        # Relax the problem:
        h_relaxed = cp.Variable(n_points)
        constraints = [h_relaxed <= 1, h_relaxed >= 0, cp.sum(h_relaxed) == n_points/2]
        
        # Compute all overlaps for the relaxed version
        overlaps = []
        for k in lags:
            overlap = cp.sum(cp.multiply(h_relaxed[:-k], 1 - h_relaxed[k:]))
            overlaps.append(overlap)
        
        # Minimize the maximum overlap
        objective = cp.Minimize(cp.max(cp.stack(overlaps)))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_new = h_relaxed.value
                # Round to binary and adjust to satisfy sum constraint
                h_rounded = (h_new >= 0.5).astype(float)
                current_sum = np.sum(h_rounded)
                diff = current_sum - n_points / 2
                
                if diff > 0:
                    # Need to turn some 1s into 0s
                    indices = np.argsort(h_new)
                    for i in range(int(diff)):
                        h_rounded[indices[i]] = 0
                elif diff < 0:
                    # Need to turn some 0s into 1s
                    indices = np.argsort(-h_new)
                    for i in range(int(-diff)):
                        h_rounded[indices[i]] = 1
                
                # Final adjustment to ensure exact sum
                current_sum = np.sum(h_rounded)
                diff = current_sum - n_points / 2
                if diff > 0:
                    to_zero = np.where(h_rounded == 1)[0]
                    indices = np.argsort([h_new[i] for i in to_zero])
                    for i in indices[:int(diff)]:
                        h_rounded[to_zero[i]] = 0
                elif diff < 0:
                    to_one = np.where(h_rounded == 0)[0]
                    indices = np.argsort([-h_new[i] for i in to_one])
                    for i in indices[:int(-diff)]:
                        h_rounded[to_one[i]] = 1
                
                h_new = np.clip(h_rounded, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate
                correlation = np.correlate(h_new, 1 - h_new, mode='full')
                c5 = np.max(correlation) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                
                # Use this as current solution (for next iteration)
                h = h_new
            else:
                # Fallback: use perturbation around current h
                h_new = h + np.random.normal(0, 0.01, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except:
            # Fallback to perturbation
            h_new = h + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
        
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final validation of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
