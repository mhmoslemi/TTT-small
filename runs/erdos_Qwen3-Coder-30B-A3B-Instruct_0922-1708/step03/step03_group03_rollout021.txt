import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # uniform distribution
        h[0] += 0.1 * np.random.random()  # add noise to break symmetry
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Multi-resolution approach with adaptive perturbations
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    temperature = 0.1
    perturbation_scale = 0.1
    num_iterations = 1000
    success_count = 0
    last_success_count = 0
    start_time = time.time()
    
    for resolver in resolvers:
        if time.time() - start_time > budget_s - 1:
            break
            
        perturbation_scale = resolver
        success_count = 0
        
        for i in range(num_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate candidate
            h_candidate = h + np.random.normal(0, perturbation_scale, n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute overlap integral
            correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(correlation) * dx
            
            # Accept or reject
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                success_count += 1
                perturbation_scale *= 0.99  # Adaptive refinement
            else:
                # Accept with probability based on temperature
                delta = c5_candidate - best_c5
                if np.random.random() < np.exp(-delta / temperature):
                    h = h_candidate.copy()
                    best_c5 = c5_candidate
                    success_count += 1
                    
            # Adjust perturbation scale based on recent success rate
            if i % 50 == 0 and i > 0:
                success_rate = success_count / 50.0
                if success_rate > 0.2:
                    perturbation_scale *= 1.5
                elif success_rate < 0.1:
                    perturbation_scale *= 0.5
                perturbation_scale = np.clip(perturbation_scale, 0.001, 1.0)
                
            # Cooling schedule
            if i % 100 == 0:
                temperature *= 0.99
                temperature = max(temperature, 1e-6)
        
        # Update last success count
        last_success_count = success_count
        
    # Final computation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
