import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # PSO Parameters
    n_particles = 20
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.1
    
    # Initialize particles
    particles = [h.copy() for _ in range(n_particles)]
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [float('inf')] * n_particles
    global_best = h.copy()
    global_best_fitness = float('inf')
    
    # Helper to evaluate fitness and handle constraints
    def eval_fitness(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5, h_vec
    
    # Initial evaluation
    for i in range(n_particles):
        fitness, particles[i] = eval_fitness(particles[i])
        if fitness < personal_best_fitness[i]:
            personal_best[i] = particles[i].copy()
            personal_best_fitness[i] = fitness
        if fitness < global_best_fitness:
            global_best = particles[i].copy()
            global_best_fitness = fitness
    
    # PSO Loop
    no_improvement_count = 0
    iteration = 0
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 50:
        iteration += 1
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update particle
            particles[i] += velocities[i]
            particles[i], _ = eval_fitness(particles[i])
            
            # Update personal best
            fitness, _ = eval_fitness(particles[i])
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
                    no_improvement_count = 0  # Reset counter
        no_improvement_count += 1
        
    # Refine with gradient descent around the best solution
    h = global_best.copy()
    best_h = h.copy()
    best_c5 = global_best_fitness
    
    # Gradient Descent Refinement
    max_grad_iter = 1000
    grad_lr = 0.01
    eps = 1e-4
    
    for _ in range(max_grad_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Estimate gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
        
        # Update
        h_new = h - grad_lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
        else:
            grad_lr *= 0.99  # Reduce learning rate if no improvement
            
        if np.linalg.norm(grad) < 1e-6:
            break  # Converged
    
    return (h.tolist(), best_c5, n_points)
