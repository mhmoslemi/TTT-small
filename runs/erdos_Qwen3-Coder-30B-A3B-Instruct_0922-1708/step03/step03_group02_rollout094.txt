import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing with cooling schedule
    T = 0.1
    alpha = 0.995
    max_steps = 2000
    
    # Nelder-Mead simplex optimization
    def nelder_mead_optimize(initial_h, max_iter=1000):
        # Initialize simplex with perturbed points
        simplex = [initial_h.copy()]
        for i in range(4):
            h_pert = initial_h.copy()
            # Perturb a few random points to create simplex
            indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
            for idx in indices:
                h_pert[idx] += (np.random.random() - 0.5) * 0.2
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            simplex.append(h_pert)
            
        iteration = 0
        while iteration < max_iter and (time.time() - start_time) < budget_s - 10:
            # Evaluate function at all simplex points
            values = []
            for point in simplex:
                correlation = np.correlate(point, 1 - point, mode='full')
                c5 = np.max(correlation) * dx
                values.append(c5)
            
            # Sort simplex
            indices = np.argsort(values)
            simplex = [simplex[i] for i in indices]
            values = [values[i] for i in indices]
            
            # Check if we've converged
            if values[4] - values[0] < 1e-8:
                break
                
            # Reflect
            centroid = np.mean(simplex[:-1], axis=0)
            x_r = 2 * centroid - simplex[4]
            x_r = np.clip(x_r, 0, 1)
            x_r = x_r * (n_points / 2) / np.sum(x_r)
            
            c_r = np.max(np.correlate(x_r, 1 - x_r, mode='full')) * dx
            
            if values[0] <= c_r < values[3]:
                simplex[4] = x_r
            elif c_r < values[0]:
                # Expand
                x_e = 2 * x_r - centroid
                x_e = np.clip(x_e, 0, 1)
                x_e = x_e * (n_points / 2) / np.sum(x_e)
                c_e = np.max(np.correlate(x_e, 1 - x_e, mode='full')) * dx
                if c_e < c_r:
                    simplex[4] = x_e
                else:
                    simplex[4] = x_r
            else:
                # Contract
                x_c = 0.5 * (simplex[0] + simplex[4])
                x_c = np.clip(x_c, 0, 1)
                x_c = x_c * (n_points / 2) / np.sum(x_c)
                c_c = np.max(np.correlate(x_c, 1 - x_c, mode='full')) * dx
                if c_c < values[4]:
                    simplex[4] = x_c
                else:
                    # Shrink
                    for i in range(1, 5):
                        simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                        simplex[i] = np.clip(simplex[i], 0, 1)
                        simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
            
            iteration += 1
        
        # Return best point from simplex
        final_values = [np.max(np.correlate(point, 1 - point, mode='full')) * dx for point in simplex]
        best_idx = np.argmin(final_values)
        return simplex[best_idx]
    
    # Main loop
    steps = 0
    while (time.time() - start_time) < budget_s - 10 and steps < max_steps:
        # Perturb current best solution
        h_new = best_h.copy()
        # Select a random subset to perturb
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        for idx in indices:
            h_new[idx] += (np.random.random() - 0.5) * 0.2
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluation
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept or reject based on simulated annealing criteria
        if c5_new < best_c5 or np.random.random() < np.exp((best_c5 - c5_new) / T):
            best_h = h_new
            best_c5 = c5_new
        
        # Cooling schedule
        T *= alpha
        steps += 1
    
    # Apply Nelder-Mead for further refinement
    refined_h = nelder_mead_optimize(best_h)
    
    # Final check
    correlation = np.correlate(refined_h, 1 - refined_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (refined_h.tolist(), final_c5, n_points)
