import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_generations = 120
    
    # Initialize population
    pop_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill rest of population with random valid individuals
    for i in range(pop_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    generation = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
        
        # Check for stagnation
        no_improvement_count += 1
        if no_improvement_count > 100:
            break
            
        # Selection - keep top 40%
        selected_indices = np.argsort(fitness)[:int(0.4 * pop_size)]
        selected = [population[i] for i in selected_indices]
        
        # Create new population
        new_population = selected.copy()
        
        # Generate offspring through crossover
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = selected[random.randint(0, len(selected) - 1)]
            parent2 = selected[random.randint(0, len(selected) - 1)]
            
            # Crossover: blend neighboring points
            child = np.zeros(n_points)
            crossover_point = random.randint(1, n_points - 1)
            child[:crossover_point] = parent1[:crossover_point]
            child[crossover_point:] = parent2[crossover_point:]
            
            # Mutation
            if random.random() < 0.2:  # 20% chance to mutate
                mutate_indices = random.sample(range(n_points), int(0.05 * n_points))
                for i in mutate_indices:
                    noise = random.gauss(0, 0.1)
                    child[i] += noise
                    child[i] = max(0, min(1, child[i]))
                    
                # Rescale to maintain sum constraint
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child.copy())
            
        population = new_population[:pop_size]
        
    return (best_h.tolist(), best_c5, n_points)
