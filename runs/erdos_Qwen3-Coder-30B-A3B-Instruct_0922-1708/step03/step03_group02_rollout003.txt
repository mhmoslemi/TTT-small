import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)  # Normalize
    else:
        # Create a structured initial guess: piecewise constant
        h = np.zeros(n_points)
        # Create step function with high values in middle regions
        mid_start = int(0.3 * n_points)
        mid_end = int(0.7 * n_points)
        h[mid_start:mid_end] = 1.0
        h = h * (n_points / 2) / np.sum(h)  # Normalize
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    T = 0.1  # Initial temperature
    alpha = 0.995  # Cooling factor
    lr = 0.01  # Learning rate
    
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing Step: Perturb 10% of values
        h_new = h.copy()
        n_perturb = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            delta = np.random.normal(0, 0.02)
            h_new[i] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept or reject with SA
        if c5_new < best_c5 or random.random() < np.exp(-(c5_new - best_c5) / T):
            h = h_new
            c5 = c5_new
        
        # Gradient Descent Step
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update h
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Update if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # Adapt learning rate
            lr = min(0.02, lr * 1.01)
        else:
            lr = max(0.001, lr * 0.95)
        
        # Update temperature
        T *= alpha
        
        h = h_new
        
    # Final verification
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
