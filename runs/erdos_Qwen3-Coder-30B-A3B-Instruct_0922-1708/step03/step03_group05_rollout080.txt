import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    num_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_vel = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        # Add initial solution to particles
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        for i in range(1, num_particles):
            h = np.random.rand(n_points) * 0.5
            h = h / np.sum(h) * (n_points / 2)
            particles.append(h)
            velocities.append(np.random.rand(n_points) * 2 * max_vel - max_vel)
    else:
        # Initialize random particles
        for i in range(num_particles):
            h = np.random.rand(n_points) * 0.5
            h = h / np.sum(h) * (n_points / 2)
            particles.append(h)
            velocities.append(np.random.rand(n_points) * 2 * max_vel - max_vel)
    
    # Convert to arrays for faster computation
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # Track personal bests and global best
    p_best = particles.copy()
    p_best_scores = np.full(num_particles, float('inf'))
    
    g_best_idx = 0
    g_best_score = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate all particles
        for i in range(num_particles):
            h = particles[i]
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            score = np.max(correlation) * dx
            
            if score < p_best_scores[i]:
                p_best_scores[i] = score
                p_best[i] = h.copy()
                
            if score < g_best_score:
                g_best_score = score
                g_best_idx = i
                
        # Update velocities and positions
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (p_best[g_best_idx] - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            particles[i] += velocities[i]
            
            # Enforce constraints: clip to [0,1] and re-scale
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
    
    # Return best solution
    best_h = p_best[g_best_idx]
    return (best_h.tolist(), g_best_score, n_points)
