    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0

    # Smoothing coefficient

        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Solve relaxed convex optimization
        # Minimize E(h) = C5 + lambda * sum((h[i] - h[j])^2)
        # Subject to sum(h) = n/2
        
        # Build the energy function components
        # We approximate the overlap term for the linearization
        
        # Constraint: sum(h) = n_points/2
        # For explicit optimization, we use a simple projection
        
        # Gradient estimation with small perturbations
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Apply gradient descent with momentum-like effect
        lr = 0.01 + 0.005 * np.sin(iteration / 50)
        h_new = h - lr * grad
        
        # Apply smoothing regularization
        if iteration > 100:
            # Add repulsion between values
            for i in range(n_points):
                for j in range(i+1, n_points):
                    diff = h_new[i] - h_new[j]
                    h_new[i] -= lambda_reg * diff * 0.001
                    h_new[j] += lambda_reg * diff * 0.001

    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
