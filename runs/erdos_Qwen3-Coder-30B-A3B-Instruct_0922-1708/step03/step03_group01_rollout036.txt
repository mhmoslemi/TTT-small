import numpy as np
import time
import hashlib
import random

def evaluate_erdos_solution(h):
    """Compute the C5 bound for given h"""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # TABU search parameters
    tabu_list = set()
    tabu_list_size = 50
    perturbation_count = max(1, int(0.2 * n_points))
    step_size = 0.1
    stagnation_counter = 0
    stagnation_threshold = 100
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Generate multiple candidate solutions through perturbation
        candidates = []
        for _ in range(50):
            # Create perturbed version
            candidate_h = h.copy()
            
            # Select random indices to perturb
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            
            # Apply perturbations
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                candidate_h[i] += delta
            
            # Project to [0, 1] and rescale
            candidate_h = np.clip(candidate_h, 0, 1)
            candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
            
            # Hash the candidate for TABU check
            candidate_hash = hashlib.md5(candidate_h.tobytes()).hexdigest()
            
            # Skip if in TABU list
            if candidate_hash in tabu_list:
                continue
            
            # Evaluate
            c5 = evaluate_erdos_solution(candidate_h)
            candidates.append((candidate_h.copy(), c5, candidate_hash))
        
        if not candidates:
            # If no valid candidates, try random perturbation of current h
            candidate_h = h.copy()
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                candidate_h[i] += delta
            candidate_h = np.clip(candidate_h, 0, 1)
            candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
            c5 = evaluate_erdos_solution(candidate_h)
            candidates.append((candidate_h.copy(), c5, hashlib.md5(candidate_h.tobytes()).hexdigest()))
        
        # Select best candidate (lowest C5)
        candidate_h, c5, candidate_hash = min(candidates, key=lambda x: x[1])
        
        # Check if we can add to TABU list
        if len(tabu_list) >= tabu_list_size:
            tabu_list.pop()
        
        # If improvement, update best
        if c5 < best_c5:
            best_h = candidate_h.copy()
            best_c5 = c5
            # Add previous h to TABU
            tabu_list.add(hashlib.md5(h.tobytes()).hexdigest())
            h = candidate_h.copy()
            stagnation_counter = 0
        else:
            # Try aspiration mechanism
            if random.random() < 0.1:
                # Accept worse solution occasionally to escape local minima
                tabu_list.add(hashlib.md5(h.tobytes()).hexdigest())
                h = candidate_h.copy()
                stagnation_counter = 0
            else:
                stagnation_counter += 1
                # Reinitialize if stagnation occurs
                if stagnation_counter > stagnation_threshold:
                    step_size *= 1.2
                    tabu_list.clear()
                    stagnation_counter = 0
        
        # Occasionally add current solution to TABU to prevent cycles in non-improving loops
        if len(tabu_list) < tabu_list_size:
            tabu_list.add(hashlib.md5(h.tobytes()).hexdigest())
    
    # Final re-evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
