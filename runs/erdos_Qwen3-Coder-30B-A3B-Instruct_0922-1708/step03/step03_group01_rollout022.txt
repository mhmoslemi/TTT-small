import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using Frank-Zadoff-like sequence for better starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        # Generate Frank-Zadoff-inspired sequence with varying frequencies
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = (np.sin(2 * np.pi * i * 0.37) + np.cos(2 * np.pi * i * 0.73)) * 0.25 + 0.5
        h = np.clip(h, 0, 1)
    
    # Normalize to exact sum
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Phase 1: Binary search on C5
    low, high = 0.38, 0.4
    while time.time() - start_time < budget_s * 0.4 and high - low > 1e-4:
        c_mid = (low + high) / 2
        # Check feasibility: solve QCQP via CVXPY
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0, 
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        for k in range(n_points):
            # Sum over all i: h[i] * (1 - h[(i+k) % n_points])
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            constraints.append(overlap_expr <= c_mid / dx)
            
        obj = cp.Minimize(0)  # Dummy objective for feasibility
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(verbose=False, solver=cp.MOSEK)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h_var.value
                if h_solution is not None:
                    h = np.clip(h_solution, 0, 1)
                    # Verify the current h
                    corr = np.correlate(h, 1 - h, mode='full')
                    c5_current = np.max(corr) * dx
                    if c5_current < best_c5:
                        best_h = h.copy()
                        best_c5 = c5_current
                    high = c_mid
                else:
                    low = c_mid
            else:
                low = c_mid
        except Exception:
            low = c_mid
    
    # Phase 2: Local refinement with adaptive step
    h = best_h.copy()
    max_iter = 1000
    alpha = 0.01
    beta = 0.995
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        iteration += 1
        
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(corr) * dx
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        
        # Find index of maximum
        max_idx = np.argmax(corr)
        k_best = max_idx - (len(corr) - 1) // 2  # Convert full index to shift
        if k_best < 0:
            k_best += len(h)
            
        # Compute analytical gradient for max shift
        grad = np.zeros(n_points)
        for idx in range(n_points):
            j = (idx + k_best) % n_points
            grad[idx] = (1 - h[j]) - h[idx - k_best] if idx - k_best >= 0 else (1 - h[j]) - h[idx - k_best + n_points]
        grad *= dx
        
        # Update step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept or reject based on improvement
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        if c5_new < c5_current:
            h = h_new
            alpha *= 1.02  # Increase step size when improving
        else:
            alpha *= beta  # Reduce step size when not improving
            
        if i % 100 == 0:
            # Periodic rescaling
            h = h * (n_points / 2) / np.sum(h)
    
    # Final validation and downsampling to 100 samples
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    h_final = best_h[:100] if len(best_h) > 100 else best_h.copy()
    h_final = h_final * (100 / 2) / np.sum(h_final)
    
    return (h_final.tolist(), final_c5, 100)
