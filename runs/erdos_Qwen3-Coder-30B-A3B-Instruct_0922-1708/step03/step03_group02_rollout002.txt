import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a structured initial guess
        h = np.zeros(n_points)
        # Place some mass in two intervals
        idx1 = np.arange(int(0.2*n_points), int(0.4*n_points))
        idx2 = np.arange(int(1.2*n_points), int(1.4*n_points))
        h[idx1] = 1.0
        h[idx2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Adaptive parameters
    temp = 0.1
    cooling_rate = 0.995
    lr = 0.01
    min_lr = 0.001
    max_lr = 0.02
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Simulated Annealing Step - random perturbation
        h_candidate = h.copy()
        # Perturb about 10% of points
        perturb_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in perturb_indices:
            # Add small random perturbation
            delta = np.random.normal(0, 0.05)
            h_candidate[i] += delta
        h_candidate = project_h(h_candidate)
        
        c5_candidate = compute_c5(h_candidate)
        
        # Accept with simulated annealing probability
        if c5_candidate < best_c5 or random.random() < np.exp(-(c5_candidate - best_c5) / temp):
            h = h_candidate
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
        
        # Gradient Descent Step
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            
            grad[i] = (c5_plus - compute_c5(h)) / eps
        
        # Update with gradient descent
        h_new = h - lr * grad
        
        # Project back to feasible region
        h_new = project_h(h_new)
        
        # Compute new value
        c5_new = compute_c5(h_new)
        
        # Adaptive learning rate
        if c5_new < best_c5:
            lr = min(max_lr, lr * 1.01)
        else:
            lr = max(min_lr, lr * 0.95)
        
        h = h_new
        
        # Update best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        
        # Cool temperature
        temp *= cooling_rate
        
        # Early stopping if improvement is too small
        if iteration > 100 and abs(c5_new - compute_c5(best_h)) < 1e-7:
            break
    
    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
