import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_generations = 120
    pop_size = 50
    crossover_rate = 0.5
    mutation_rate = 0.2
    mutation_intensity = 0.1

    def initialize_population():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_orig, h)
            h = np.clip(h, 0, 1)
        else:
            h = np.ones(n_points) * 0.5
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        return [h.copy() for _ in range(pop_size)]

    def evaluate_individual(h):
        correlation = correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5

    def mutate_individual(h):
        h_new = h.copy()
        num_muts = max(1, int(n_points * mutation_rate * 0.05))
        indices = np.random.choice(n_points, size=num_muts, replace=False)
        for i in indices:
            delta = np.random.normal(0, mutation_intensity)
            h_new[i] += delta
        h_new = np.clip(h_new, 0, 1)
        # Rescale to maintain sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    def crossover_individuals(h1, h2):
        if np.random.random() > crossover_rate:
            return h1.copy(), h2.copy()
        # Blend the two parents
        alpha = np.random.random()
        h_child1 = alpha * h1 + (1 - alpha) * h2
        h_child2 = (1 - alpha) * h1 + alpha * h2
        # Rescale
        h_child1 = h_child1 * (n_points / 2) / np.sum(h_child1)
        h_child2 = h_child2 * (n_points / 2) / np.sum(h_child2)
        return h_child1, h_child2

    def select_parents(population, fitnesses):
        sorted_indices = np.argsort(fitnesses)
        top_40 = sorted_indices[:int(0.4 * len(population))]
        return [population[i] for i in np.random.choice(top_40, size=len(population), replace=True)]

    population = initialize_population()
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    generation = 0
    no_improvement_count = 0

    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        fitnesses = [evaluate_individual(h) for h in population]
        if min(fitnesses) < best_c5:
            idx = np.argmin(fitnesses)
            best_c5 = fitnesses[idx]
            best_h = population[idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        if no_improvement_count > 100:
            break
        
        # Selection
        parents = select_parents(population, fitnesses)
        new_population = []
        for i in range(0, len(parents), 2):
            p1, p2 = parents[i], parents[i+1]
            child1, child2 = crossover_individuals(p1, p2)
            new_population.append(mutate_individual(child1))
            new_population.append(mutate_individual(child2))
        population = new_population[:pop_size]

    if best_h is None:
        # Fallback to initial solution
        best_h = population[0] if population else np.ones(n_points) * 0.5
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_individual(best_h)

    return (best_h.tolist(), best_c5, n_points)
