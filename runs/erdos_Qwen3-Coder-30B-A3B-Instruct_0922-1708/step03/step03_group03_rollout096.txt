import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to meet the sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Helper function to compute C5
    def compute_c5(h):
        # Direct computation of maximum overlap
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    
    # Adaptively adjust perturbation scale
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    temperature = 0.1
    success_count = 0
    total_count = 0
    perturbation_scale = 0.1
    
    for resolver in resolvers:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Set perturbation scale for this resolver
        perturbation_scale = resolver
        
        for iteration in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate candidate
            h_candidate = h + np.random.normal(0, perturbation_scale, n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Rescale to meet constraint
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute new C5
            c5_candidate = compute_c5(h_candidate)
            
            # Accept or reject
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                success_count += 1
                perturbation_scale *= 1.05  # Increase scale slightly
            else:
                # Metropolis acceptance
                try:
                    prob = np.exp(-(c5_candidate - best_c5) / temperature)
                    if np.random.rand() < prob:
                        h = h_candidate.copy()
                        success_count += 1
                except:
                    pass
            
            # Adjust temperature
            if iteration % 100 == 0:
                temperature *= 0.99
                if temperature < 1e-6:
                    temperature = 1e-6
            
            # Adjust perturbation scale based on success
            total_count += 1
            if total_count >= 50:
                success_ratio = success_count / total_count
                if success_ratio > 0.2:
                    perturbation_scale *= 1.5
                elif success_ratio < 0.1:
                    perturbation_scale *= 0.5
                
                success_count = 0
                total_count = 0
            
            # Clip perturbation scale
            perturbation_scale = max(0.001, min(1.0, perturbation_scale))
    
    # Final verification
    c5_final = compute_c5(best_h)
    
    return (best_h.tolist(), c5_final, n_points)
