import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute lags
    lags = list(range(1, n_points))
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 5000:
        iteration += 1
        
        # Solve convex relaxation
        h_relaxed = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points / 2]
        
        # Compute all overlaps
        overlaps = []
        for k in range(n_points):
            overlap = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_points-k]))
            overlaps.append(overlap)
            
        # Objective: minimize the maximum overlap
        objective = cp.Minimize(cp.max(cp.hstack(overlaps)))
        
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-6, reltol=1e-6, feastol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_relaxed_solution = h_relaxed.value
            else:
                h_relaxed_solution = h
        except Exception:
            h_relaxed_solution = h
            
        # Rounding heuristic
        h_rounded = (h_relaxed_solution >= 0.5).astype(float)
        current_sum = np.sum(h_rounded)
        
        # Adjust to meet integral sum constraint
        if current_sum > n_points / 2:
            # Remove some 1's
            diff = int(current_sum - n_points / 2)
            # Find indices with smallest relaxed values
            idxs = np.argsort(h_relaxed_solution)
            to_flip = idxs[:diff]
            h_rounded[to_flip] = 0
        elif current_sum < n_points / 2:
            # Add some 0's
            diff = int(n_points / 2 - current_sum)
            # Find indices with largest relaxed values
            idxs = np.argsort(-h_relaxed_solution)
            to_flip = idxs[:diff]
            h_rounded[to_flip] = 1
            
        # Ensure sum constraint exactly
        h_rounded = h_rounded * (n_points / 2) / np.sum(h_rounded)
        
        # Verify bounds
        h_rounded = np.clip(h_rounded, 0, 1)
        
        # Compute C5
        correlation = np.correlate(h_rounded, 1 - h_rounded, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_rounded.copy()
            
        # Smooth perturbation to avoid local optima
        h = h_rounded + 0.01 * (np.random.rand(n_points) - 0.5)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
