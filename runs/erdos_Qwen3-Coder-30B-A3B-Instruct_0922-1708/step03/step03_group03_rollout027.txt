import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a modified Barker sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a modified Barker-like initialization
        h = np.zeros(n_points)
        # Place pulses to minimize overlap
        pulse_width = n_points // 4
        h[:pulse_width] = 1.0
        h[pulse_width:2*pulse_width] = 0.0
        h[2*pulse_width:3*pulse_width] = 1.0
        h[3*pulse_width:] = 0.0
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def objective(h_vals):
        # Normalize
        h_norm = h_vals * (n_points / 2) / np.sum(h_vals)
        # Compute overlap
        correlation = np.correlate(h_norm, 1 - h_norm, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def gradient(h_vals):
        # Numerical gradient calculation
        eps = 1e-6
        grad = np.zeros_like(h_vals)
        base_obj = objective(h_vals)
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            grad[i] = (objective(h_plus) - base_obj) / eps
        return grad
    
    # Optimization using L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Ensure sum constraint is met by adjusting the last variable
    def constraint_sums(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sums}
    
    # Run optimization with multiple restarts
    restarts = 5
    for r in range(restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb the current solution slightly
        if r > 0:
            h_start = h.copy() + np.random.normal(0, 0.01, n_points)
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
        else:
            h_start = h.copy()
            
        try:
            result = minimize(objective, h_start, method='L-BFGS-B', 
                             bounds=bounds, constraints=cons,
                             jac=gradient, options={'maxiter': 500})
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
        # If we're running low on time or at the end, continue
        if time.time() - start_time > budget_s - 5:
            break

    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
