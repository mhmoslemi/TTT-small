import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search for C5
    low, high = 0.38, 0.4
    tol = 1e-4
    
    while time.time() - start_time < budget_s - 1 and high - low > tol:
        c_mid = (low + high) / 2
        # Formulate QCQP
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]

        # Add overlap constraints
        for k in range(n_points):
            # Shifted indices
            if k == 0:
                overlap = cp.sum(cp.multiply(h_var, 1 - h_var))
            else:
                shifted = np.roll(1 - h_var.value, k) if h_var.value is not None else 1 - h_var[k:] + np.concatenate([h_var[:k], [0]*k])
                shifted = np.roll(1 - h_var, k)
                overlap = cp.sum(cp.multiply(h_var, shifted))
            constraints.append(overlap <= c_mid / dx)
            
        objective = cp.Minimize(c_mid)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                c5_new = c_mid
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                high = c_mid
            else:
                low = c_mid
        except:
            low = c_mid

    # Final verification with exact computation
    if best_c5 == float('inf'):
        # Revert to initial solution
        best_h = h
        correlation = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(correlation) * dx

    return (best_h.tolist(), best_c5, n_points)
