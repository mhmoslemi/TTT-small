import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    num_particles = 50
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial construction for 10% of particles
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * (n_points / 2)
        for _ in range(num_particles // 10):
            particles.append(h0.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize remaining particles randomly
    for _ in range(num_particles - len(particles)):
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    
    # Track best solutions
    p_best = particles.copy()
    p_best_values = np.full(num_particles, np.inf)
    g_best_idx = 0
    g_best_value = np.inf
    
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 5000:
        iteration += 1
        
        # Compute C5 for all particles
        current_values = []
        for i in range(num_particles):
            h = particles[i]
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            current_values.append(c5)
            
            # Update personal best
            if c5 < p_best_values[i]:
                p_best_values[i] = c5
                p_best[i] = h.copy()
            
            # Update global best
            if c5 < g_best_value:
                g_best_value = c5
                g_best_idx = i
        
        current_values = np.array(current_values)
        
        # Update velocities and positions
        r1, r2 = np.random.rand(2)
        for i in range(num_particles):
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (p_best[g_best_idx] - particles[i]))
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
    
    # Return best solution
    best_h = p_best[g_best_idx]
    return (best_h.tolist(), g_best_value, n_points)
