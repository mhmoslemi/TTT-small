import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points

    # Initialize using the provided parent or create Frank-Zadoff-like sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a Frank-Zadoff-like sequence for initialization
        h = np.ones(n_points) * 0.5
        # Slight modification to achieve better distribution
        for i in range(n_points):
            h[i] = (1 + np.sin(2 * np.pi * i / n_points * 3)) / 2
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = float('inf')

    start_time = time.time()
    time_remaining = lambda: budget_s - (time.time() - start_time)

    # Phase 1: Binary search with QCQP feasibility checks
    low, high = 0.38, 0.4
    epsilon = 1e-4
    while high - low > epsilon and time_remaining() > 10:
        Cmid = (low + high) / 2
        # QCQP formulation
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # For each shift, add QCQP constraint
        for k in range(n_points):
            shift_expr = cp.multiply(h_var, cp.roll(1 - h_var, k))
            constraints.append(cp.sum(shift_expr) <= Cmid / dx)
            
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                high = Cmid
            else:
                low = Cmid
        except Exception:
            low = Cmid

    # Phase 2: Locally refine using accelerated gradient descent
    h = best_h.copy() if time_remaining() > 10 else h
    if time_remaining() > 10:
        alpha = 0.01
        max_iter = 1000
        for it in range(max_iter):
            if time_remaining() < 1:
                break

            # Compute the maximal overlap
            overlaps = []
            for k in range(n_points):
                shifted = np.roll(1 - h, k)
                overlap = np.sum(h * shifted)
                overlaps.append(overlap)
            overlaps = np.array(overlaps)
            max_k = np.argmax(overlaps)
            max_overlap = overlaps[max_k]

            # Compute analytical gradient w.r.t. h for the max overlap
            grad = np.zeros_like(h)
            # d/dh_i of h[i] * (1 - h[i + k_max]) 
            for i in range(n_points):
                grad[i] = dx * (1 - h[(i + max_k) % n_points]) - dx * h[(i - max_k) % n_points]

            # Update
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new

            # Compute current C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Adaptive learning rate
            if it % 50 == 0:
                alpha *= 0.95

    # Downsample to 100 points if needed
    final_h = best_h if 'best_h' in locals() else h
    if n_points != 100:
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, 100)
        final_h = np.interp(x_new, x_orig, final_h)
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * 50.0 / np.sum(final_h)
        n_points = 100

    # Final verification
    correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(correlation) * (2.0 / 100)
    
    return (final_h.tolist(), final_c5, n_points)
