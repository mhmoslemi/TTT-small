import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h with the parent construction or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search on C5 to find tightest upper bound
    low = 0.38
    high = 0.4
    
    # We'll use CVXPY to solve the QCQP
    while high - low > 1e-4 and time.time() - start_time < budget_s - 1:
        c_mid = (low + high) / 2.0
        
        # Formulate QCQP problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add overlap constraints for all possible shifts
        for k in range(n_points):
            # Compute the overlap for shift k
            # h[i] * (1 - h[i+k]) for all valid i
            # Create the shifted version
            shifted = cp.hstack([h_var[k:], h_var[:k]])
            overlap = cp.sum(cp.multiply(h_var, 1 - shifted))
            constraints.append(overlap <= c_mid / dx)
        
        # Objective is to minimize C5 (we'll pass c_mid as a parameter)
        # Since we're solving the feasibility of c_mid, we set objective to any valid objective
        # We'll use a dummy objective and only check feasibility
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                # Feasible solution found
                h_solution = h_var.value
                if h_solution is not None:
                    # Check if this feasible solution is better
                    # Compute the actual C5 from the solution
                    corr = np.correlate(h_solution, 1 - h_solution, mode='full')
                    c5_actual = np.max(corr) * dx
                    if c5_actual < best_c5:
                        best_c5 = c5_actual
                        best_h = h_solution.copy()
                    high = c_mid
                else:
                    low = c_mid
            else:
                # Infeasible: try a larger C5
                low = c_mid
        except:
            # If any solver error, try more conservative bounds
            low = c_mid
    
    # Final check with the best solution found
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
