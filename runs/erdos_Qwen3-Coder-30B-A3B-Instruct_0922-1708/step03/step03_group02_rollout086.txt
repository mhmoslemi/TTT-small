import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init.copy())
    
    # Add random particles
    while len(particles) < n_particles:
        h = np.random.uniform(0.4, 0.6, n_points)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
    
    # Initialize velocities
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    
    # Initialize personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * n_particles
    
    # Global best
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # PSO parameters
    w = 1.0  # Inertia weight
    c1 = c2 = 2.0  # Acceleration constants
    max_vel = 0.2  # Maximum velocity
    
    start_time = time.time()
    step = 0
    
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Update inertia weight linearly
        w = 1.0 - (step / (budget_s / 2)) * 0.2
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            score = np.max(corr) * dx
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = h.copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = h.copy()
            
            # Update particle
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - h) + 
                           c2 * r2 * (global_best - h))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = h + velocities[i]
            
            # Re-normalize sum to exact constraint
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
        
        # Periodic reset to maintain diversity
        if step % 100 == 0:
            for i in range(n_particles):
                if np.random.random() < 0.1:  # 10% chance to reset
                    particles[i] = np.random.uniform(0.4, 0.6, n_points)
                    particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
    
    # Final evaluation
    corr = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (global_best.tolist(), final_c5, n_points)
