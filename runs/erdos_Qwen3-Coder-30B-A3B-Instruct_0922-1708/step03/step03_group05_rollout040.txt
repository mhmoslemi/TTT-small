import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Convex relaxation approach
        h_relaxed = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_relaxed <= 1, h_relaxed >= 0, cp.sum(h_relaxed) == n_points/2]
        
        # Objective: minimize maximum overlap
        overlaps = []
        for k in range(n_points):
            overlap = 0
            for i in range(n_points - k):
                overlap += h_relaxed[i] * (1 - h_relaxed[i + k])
            overlaps.append(overlap)
        
        objective = cp.Minimize(cp.max(overlaps))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            
            if prob.status == cp.OPTIMAL:
                h_sol = h_relaxed.value
                h_sol = np.clip(h_sol, 0, 1)
                
                # Greedy rounding
                h_rounded = (h_sol >= 0.5).astype(float)
                current_sum = np.sum(h_rounded)
                
                # Adjust to meet sum constraint
                if current_sum > n_points / 2:
                    # Remove some 1s
                    diff = int(current_sum - n_points / 2)
                    indices = np.argsort(h_sol)
                    for i in range(diff):
                        if h_rounded[indices[i]] == 1:
                            h_rounded[indices[i]] = 0
                elif current_sum < n_points / 2:
                    # Add some 0s
                    diff = int(n_points / 2 - current_sum)
                    indices = np.argsort(-h_sol)
                    for i in range(diff):
                        if h_rounded[indices[i]] == 0:
                            h_rounded[indices[i]] = 1
                
                h = h_rounded
                
        except:
            # Fallback to simple gradient descent if optimization fails
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Small perturbation to avoid local minima
        noise = np.random.normal(0, 0.01, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
