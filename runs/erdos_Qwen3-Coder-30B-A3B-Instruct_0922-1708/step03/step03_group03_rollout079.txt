        correlation = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(correlation) * dx
        
        return max_overlap
    
    def optimize_step():
        # Use L-BFGS-B optimization for better convergence
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Add smoothing to the optimization problem to avoid sharp gradients
        def smooth_objective(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlation = np.correlate(h, 1 - h, mode='full')
            return np.max(correlation) * dx

            correlation = np.correlate(h, 1 - h, mode='full')
