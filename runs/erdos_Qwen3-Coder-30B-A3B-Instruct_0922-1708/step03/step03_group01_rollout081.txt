import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill population with random valid individuals
    for i in range(pop_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h.copy())
        
    best_h = population[0].copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        generation += 1
        
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            correlation = correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Selection: keep top 40%
        sorted_indices = np.argsort(fitness)
        elite_count = max(1, int(0.4 * pop_size))
        elite_indices = sorted_indices[:elite_count]
        elites = [population[i] for i in elite_indices]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - elite_count:
            # Tournament selection
            parent1_idx = np.random.choice(elite_indices)
            parent2_idx = np.random.choice(elite_indices)
            
            # Crossover (blend)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Blend 50% of points from each parent
            blend_mask = np.random.random(n_points) < 0.5
            child = np.where(blend_mask, parent1, parent2)
            
            # Mutation (20% of individuals)
            if np.random.random() < 0.2:
                # Mutate 5% of points
                mutate_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                for idx in mutate_indices:
                    noise = np.random.normal(0, 0.1)
                    child[idx] += noise
                # Clip to [0,1]
                child = np.clip(child, 0, 1)
                # Rescale to sum to n_points/2
                child = child * (n_points / 2) / np.sum(child)
                
            offspring.append(child)
            
        # Create new population
        population = elites + offspring
        
        # Check for stagnation
        if generation > 10:
            current_best = min(fitness)
            if abs(current_best - best_c5) < 1e-8:
                no_improvement_count += 1
            else:
                no_improvement_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
