import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement = 0
    max_no_improvement = 100
    
    # PSO parameters
    pop_size = 20
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Create initial swarm
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            # Use the initial solution as the first particle
            particle = h.copy()
        else:
            # Random initialization with constraints
            particle = np.random.rand(n_points)
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        corr = np.correlate(particle, 1 - particle, mode='full')
        pbest.append(particle.copy())
        pbest_values.append(np.max(corr) * dx)
    
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx]
    gbest_value = pbest_values[gbest_idx]
    
    # PSO main loop
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        iteration += 1
        
        for i in range(pop_size):
            particle = swarm[i]
            vel = velocities[i]
            particle_pbest = pbest[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (inertia * vel 
                   + c1 * r1 * (particle_pbest - particle) 
                   + c2 * r2 * (gbest - particle))
            
            # Clip velocity
            vel = np.clip(vel, -max_vel, max_vel)
            
            # Update position
            particle = particle + vel
            
            # Ensure feasibility
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Evaluate
            corr = np.correlate(particle, 1 - particle, mode='full')
            particle_value = np.max(corr) * dx
            
            # Update personal best
            if particle_value < pbest_values[i]:
                pbest[i] = particle.copy()
                pbest_values[i] = particle_value
                
                # Update global best
                if particle_value < gbest_value:
                    gbest = particle.copy()
                    gbest_value = particle_value
                    no_improvement = 0
            
            # Update swarm
            swarm[i] = particle
            velocities[i] = vel
        
        no_improvement += 1
        
        # Local refinement on the global best
        if iteration % 10 == 0:
            h_refined = gbest.copy()
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5_refined = np.max(corr) * dx
            
            # Gradient descent refinement
            for _ in range(50):
                grad = np.zeros_like(h_refined)
                eps = 1e-4
                
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5_refined) / eps
                
                # Adaptive learning rate
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h_refined - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr_new) * dx
                
                if c5_new < c5_refined:
                    h_refined = h_new
                    c5_refined = c5_new
                else:
                    break
            
            if c5_refined < gbest_value:
                gbest = h_refined
                gbest_value = c5_refined
                no_improvement = 0
        
        # Early termination check
        if time.time() - start_time > budget_s - 5:
            break
    
    # Final evaluation of the best solution
    best_h = gbest.copy()
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), best_c5, n_points)
