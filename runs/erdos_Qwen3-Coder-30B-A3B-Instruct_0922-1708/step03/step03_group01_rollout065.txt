import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    low = 0.38
    high = 0.4
    tolerance = 1e-4
    
    # Binary search on C5
    while (high - low > tolerance) and (time.time() - start_time < budget_s - 1):
        C_mid = (low + high) / 2.0
        
        # Define variables
        h_vars = cp.Variable(n_points)
        # Constraints 0 <= h[i] <= 1
        constraints = [h_vars >= 0, h_vars <= 1]
        # Constraint sum(h) = n_points/2
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Add overlap constraints for all shifts
        for k in range(n_points):
            # Compute overlap sum for shift k
            overlap_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_sum += h_vars[i] * (1 - h_vars[j])
            constraints.append(overlap_sum <= C_mid / dx)
        
        # Objective: minimize C5 (already encoded in constraints)
        problem = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False)  # ECOS is good for this QCQP
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h_vars.value
                if h_solution is not None:
                    h_solution = np.clip(h_solution, 0, 1)
                    h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
                    # Evaluate this h
                    correlation = np.correlate(h_solution, 1 - h_solution, mode='full')
                    c5 = np.max(correlation) * dx
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_solution.copy()
                    high = C_mid
                else:
                    low = C_mid
            else:
                low = C_mid
        except Exception:
            low = C_mid
            
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    return (best_h.tolist(), final_c5, n_points)
