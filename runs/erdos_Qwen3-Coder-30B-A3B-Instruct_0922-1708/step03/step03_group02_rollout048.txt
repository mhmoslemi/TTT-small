import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # If initial_h_values exists, use it as one of the individuals
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    population = [h.copy()]
    for _ in range(pop_size - 1):
        individual = h + np.random.normal(0, 0.1, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    best_h = h.copy()
    best_c5 = float('inf')
    no_improvement_count = 0
    max_no_improvement = 100
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Evaluate fitness of all individuals
        fitness = []
        for h_ind in population:
            corr = correlate(h_ind, 1 - h_ind, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ind.copy()
                no_improvement_count = 0
                
        no_improvement_count += 1
                
        # Sort population by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        elite_size = int(elite_rate * pop_size)
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite.copy()
        
        # Fill rest via tournament selection, crossover, and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            competitors = np.random.choice(pop_size, tournament_size)
            best_comp = competitors[np.argmin([fitness[i] for i in competitors])]
            parent1 = population[best_comp]
            
            if np.random.random() < cross_rate:
                competitors2 = np.random.choice(pop_size, tournament_size)
                best_comp2 = competitors2[np.argmin([fitness[i] for i in competitors2])]
                parent2 = population[best_comp2]
                
                # Single point crossover
                crossover_point = np.random.randint(1, n_points-1)
                offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                if np.random.random() < mut_rate:
                    mut_idx = np.random.randint(0, n_points)
                    offspring1[mut_idx] += np.random.normal(0, 0.1)
                if np.random.random() < mut_rate:
                    mut_idx = np.random.randint(0, n_points)
                    offspring2[mut_idx] += np.random.normal(0, 0.1)
                
                # Ensure feasibility
                offspring1 = np.clip(offspring1, 0, 1)
                offspring1 = offspring1 * (n_points / 2) / np.sum(offspring1)
                offspring2 = np.clip(offspring2, 0, 1)
                offspring2 = offspring2 * (n_points / 2) / np.sum(offspring2)
                
                new_population.extend([offspring1, offspring2])
            else:
                # Just mutation
                offspring = parent1.copy()
                for i in range(n_points):
                    if np.random.random() < mut_rate:
                        offspring[i] += np.random.normal(0, 0.1)
                offspring = np.clip(offspring, 0, 1)
                offspring = offspring * (n_points / 2) / np.sum(offspring)
                new_population.append(offspring)
        
        # Keep only pop_size individuals
        population = new_population[:pop_size]
    
    # Final verification
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
