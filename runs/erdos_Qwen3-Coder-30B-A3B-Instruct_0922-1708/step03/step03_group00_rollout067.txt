    best_c5 = float('inf')

    T = 0.1
    alpha = 0.995
    perturb_scale = 0.05
    min_temp = 1e-6

    sa_iterations = 0
    while time.time() - start_time < budget_s * 0.8 and sa_iterations < 1000:
        sa_iterations += 1
        # Compute current overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Generate neighbor
        h_new = h + np.random.normal(0, perturb_scale, n_points)
        h_new = np.clip(h_new, 0, 1)

        # Compute new overlap
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept or reject
        if c5_new < c5 or np.random.rand() < np.exp(-(c5_new - c5) / T):
            h = h_new
            
        # Cooling
        T = max(T * alpha, min_temp)
        
        # Occasionally try a small random perturbation to escape local optima
        if sa_iterations % 100 == 0:

    ref_iter = 0
    max_ref = 300

        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx

            best_c5 = c5
            best_h = h.copy()

            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adaptive learning rate
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)

    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
