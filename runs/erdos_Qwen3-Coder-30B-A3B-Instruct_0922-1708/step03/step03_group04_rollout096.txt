import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    eps = 1e-5
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Convex relaxation initialization
    def initialize_convex():
        # Variable
        h_var = cp.Variable(n_points)
        # Objective: minimize sum(h) - 2*sum(h^2) due to convexity
        obj = cp.Minimize(cp.sum(h_var) - 2 * cp.sum(cp.square(h_var)))
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        # Problem
        prob = cp.Problem(obj, constraints)
        # Solve
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_relaxed = np.array(h_var.value).flatten()
            # Clip and normalize
            h_relaxed = np.clip(h_relaxed, 0, 1)
            h_relaxed = h_relaxed * (n_points / 2) / np.sum(h_relaxed)
            return h_relaxed
        else:
            return h.copy()
    
    # Initialize with convex relaxation
    h = initialize_convex()
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps

        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart
        if iteration % 200 == 0:
            h = initialize_convex()
                
    return (best_h.tolist(), best_c5, n_points)
