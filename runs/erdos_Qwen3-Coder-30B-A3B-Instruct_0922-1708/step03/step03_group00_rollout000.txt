import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 20
    max_iter = 200
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(n_particles):
        if i == 0 and 'initial_h_values' in globals():
            particle = h.copy()
        else:
            particle = np.random.rand(n_points) * 0.5
            particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
        pbest.append(particle.copy())
        
        # Evaluate initial particle
        correlation = np.correlate(particle, 1 - particle, mode='full')
        c5 = np.max(correlation) * dx
        pbest_vals.append(c5)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = particle.copy()
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    
    # PSO loop
    for iter_idx in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        iteration += 1
            
        for i in range(n_particles):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure bounds and integral constraint
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate particle
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = c5
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = particles[i].copy()
        
        # Update global best
        gbest_idx = np.argmin(pbest_vals)
        gbest = pbest[gbest_idx].copy()
        
        # Occasionally make a small random perturbation
        if iter_idx % 50 == 0:
            perturb_idx = np.random.randint(0, n_particles)
            particles[perturb_idx] = np.clip(particles[perturb_idx] + 
                                           np.random.normal(0, 0.005, n_points), 0, 1)
            particles[perturb_idx] = particles[perturb_idx] * (n_points / 2) / np.sum(particles[perturb_idx])
    
    # Refine with gradient descent on best particle
    h = best_h.copy()
    max_refine_iter = 1000
    for refine_iter in range(max_refine_iter):
        if time.time() - start_time > budget_s - 1:
            break
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final evaluation
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
