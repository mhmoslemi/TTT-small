import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    population = [h.copy()]
    for _ in range(pop_size - 1):
        h_mut = h + np.random.normal(0, 0.1, n_points)
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        population.append(h_mut)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    gen = 0
    
    while time.time() - start_time < budget_s - 1:
        gen += 1
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Update best
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            
        # Selection and reproduction
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        elite_count = int(elite_rate * pop_size)
        elite_indices = sorted_indices[:elite_count]
        
        # Create new population
        new_population = [population[i].copy() for i in elite_indices]
        
        # Tournament selection
        while len(new_population) < pop_size:
            # Tournament
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Copy and mutate
            child = population[winner_idx].copy()
            
            # Crossover
            if np.random.rand() < cross_rate:
                other_idx = np.random.choice(pop_size, 1)[0]
                other = population[other_idx]
                cx_point = np.random.randint(1, n_points - 1)
                child[:cx_point] = population[winner_idx][:cx_point]
                child[cx_point:] = other[cx_point:]
            
            # Mutation
            if np.random.rand() < mut_rate:
                mut_indices = np.random.choice(n_points, int(mut_rate * n_points), replace=False)
                for i in mut_indices:
                    child[i] += np.random.normal(0, 0.05)
                child = np.clip(child, 0, 1)
            
            # Normalize
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
            
        population = new_population[:pop_size]
        
        # Occasionally add diversity through random solutions
        if gen % 10 == 0:
            for i in range(5):
                new_h = np.random.rand(n_points) * 0.5 + 0.25
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                population[np.random.randint(pop_size)] = new_h
    
    # Final evaluation of best solution
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
