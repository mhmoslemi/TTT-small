import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    num_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_vel = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Seed some particles with initial solution
    for i in range(num_particles):
        if i < num_particles // 10 and 'initial_h_values' in globals():
            h_init = h.copy()
        else:
            h_init = np.random.rand(n_points) * 0.5 + 0.25
            h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init)
        velocities.append(np.random.rand(n_points) * 2 * max_vel - max_vel)
    
    # Initialize best positions
    p_best = [p.copy() for p in particles]
    p_best_c5 = [float('inf')] * num_particles
    g_best = h.copy()
    g_best_c5 = float('inf')
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(num_particles):
            h = particles[i]
            
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < p_best_c5[i]:
                p_best_c5[i] = c5
                p_best[i] = h.copy()
                
            if c5 < g_best_c5:
                g_best_c5 = c5
                g_best = h.copy()
        
        # Update particles
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            
            # Update velocity
            v = w * velocities[i] + \
                c1 * r1 * (p_best[i] - particles[i]) + \
                c2 * r2 * (g_best - particles[i])
            
            # Clip velocity
            v = np.clip(v, -max_vel, max_vel)
            velocities[i] = v
            
            # Update position
            new_pos = particles[i] + v
            
            # Project to feasible region
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * (n_points / 2) / np.sum(new_pos)
            
            particles[i] = new_pos
            
        # Update best solution
        if g_best_c5 < best_c5:
            best_c5 = g_best_c5
            best_h = g_best.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
