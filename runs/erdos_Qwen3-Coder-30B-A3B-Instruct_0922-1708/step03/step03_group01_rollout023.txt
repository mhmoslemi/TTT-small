import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or Frank-Zadoff-like sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Generate a Frank-Zadoff-like sequence adapted for this problem
        h = np.ones(n_points) * 0.5
        # Use a sequence of 1s and 0s with minimal crossing correlation
        # Introduce a pattern that avoids sharp transitions
        h[:n_points//4] = 0.0
        h[n_points//4:2*n_points//4] = 1.0
        h[2*n_points//4:3*n_points//4] = 0.0
        h[3*n_points//4:] = 1.0
        # Normalize for sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    alpha = 0.01
    adaptive_alpha = alpha
    
    # Phase 1: Maintain a binary search in the feasible region for C5
    low, high = 0.38, 0.40
    while time.time() - start_time < budget_s * 0.7 and high - low > 1e-4:
        c_mid = (low + high) / 2
        # Use quadratic constraints to check if a feasible h exists
        # This is a simplified placeholder - larger budget could allow more sophisticated QCQP solving
        # We just sample to get approximate behavior
        if np.sum(h) > 0:
            c5_test = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if c5_test <= c_mid:
                high = c_mid
            else:
                low = c_mid

    # Phase 2: Local refinement with gradient descent to improve solution
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral for the current h
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient for the shift that results in max overlap
        k_max = np.argmax(correlation)
        # Get indices for k_max shift
        shift_indices = (np.arange(n_points) + k_max) % n_points
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Gradient for term h[i](1 - h[i + k_max])
            # Note: This is derivative of C5 wrt h[i], treating correlation max as fixed
            # Actually, we compute the gradient of the shift observed to be maximal,
            # considering partial derivatives wrt the function:
            # L = \sum_h[i](1 - h[i + k_max])
            # This is h[i + k_max] - (1 - h[i])
            # But given that total overlap is the max for a k, we use that index's constraint
            # So we relax to take derivative of current shift's constraint only
            if (i + k_max) % n_points < n_points:
                grad[i] = 1 - 2 * h[(i + k_max) % n_points]
            else:
                grad[i] = 1 - 2 * h[i]
                
        # Normalize gradient
        norm = np.linalg.norm(grad)
        if norm > 1e-8:
            grad = grad / norm
        
        h_new = h - adaptive_alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)

        h = h_new
        
        # Adjust learning rate
        if iteration % 50 == 0:
            adaptive_alpha *= 0.95
        if adaptive_alpha < 1e-5:
            adaptive_alpha = 1e-5
            
    # Final evaluation of best solution
    correlation_final = np.correlate(best_h, 1 - best_h, mode='full')
    c5_final = np.max(correlation_final) * (2.0 / 100)  # downsampled to 100 points for final result

    # Output with 100 points downsampled
    h_out = np.interp(np.linspace(0, 2, 100), np.linspace(0, 2, 200), best_h)
    h_out = np.clip(h_out, 0, 1)
    h_out = h_out * 50.0 / np.sum(h_out)  # normalize for sum = 50 = 100/2

    return (h_out.tolist(), c5_final, 100)
