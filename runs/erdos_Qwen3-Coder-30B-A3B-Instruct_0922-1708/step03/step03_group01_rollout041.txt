import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search for minimal C5
    low = 0.38
    high = 0.4
    
    # Prepare all shift indices
    shift_indices = []
    for k in range(n_points):
        shift_indices.append(np.arange(n_points) + k)
        # Handle wraparound
        shift_indices[-1] = np.mod(shift_indices[-1], n_points)
    
    while time.time() - start_time < budget_s - 1 and high - low > 1e-4:
        c_mid = (low + high) / 2.0
        
        # Construct the QCQP problem
        h_vars = cp.Variable(n_points)
        # Objective: minimize c5
        obj = cp.Minimize(1.0)
        
        # Constraints for each shift
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Add overlap constraints using built-in matrix operations for efficiency
        # Create a matrix of all shift operations
        for k in range(n_points):
            l_indices = shift_indices[k]
            # h[i] * (1 - h[i+k]) sum
            expr = cp.sum(cp.multiply(h_vars, 1 - h_vars[l_indices]))
            constraints.append(expr <= c_mid / dx)
            
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                if h_new is not None:
                    # Project to feasible region
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    # Compute actual C5
                    correlation = np.correlate(h_new, 1 - h_new, mode='full')
                    c5_new = np.max(correlation) * dx
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    high = c_mid
                else:
                    low = c_mid
            else:
                low = c_mid
        except:
            low = c_mid

    # Final check with official evaluator
    if best_c5 == float('inf'):
        # If binary search never succeeded, try a direct approach
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(correlation) * dx
        
    return (best_h.tolist(), best_c5, n_points)
