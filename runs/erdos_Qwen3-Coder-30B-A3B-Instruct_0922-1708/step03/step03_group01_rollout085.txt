import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Evaluate C₅ for given step function h"""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # TABU search parameters
    tabu_list = set()
    tabu_length = 50
    perturbation_count = max(1, int(0.2 * n_points))
    step_size = 0.1
    stagnation_counter = 0
    max_stagnation = 100
    
    start_time = time.time()
    iteration = 0
    
    def perturb_h(current_h, step):
        candidate_h = current_h.copy()
        indices = np.random.choice(n_points, size=perturbation_count, replace=False)
        for i in indices:
            delta = np.random.uniform(-step, step)
            candidate_h[i] += delta
        # Clip and normalize
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        return candidate_h
    
    def hash_h(h_vector):
        # Convert to hashable tuple with fixed precision for consistency
        return hashlib.md5(np.round(h_vector, decimals=6).tobytes()).hexdigest()
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Generate candidates by perturbing current solution
        candidates = []
        for _ in range(50):
            candidate_h = perturb_h(h, step_size)
            h_hash = hash_h(candidate_h)
            if h_hash not in tabu_list:
                c5 = evaluate_erdos_solution(candidate_h)
                candidates.append((candidate_h, c5))
        
        # Select best candidate not in TABU list
        if candidates:
            candidate_h, c5 = min(candidates, key=lambda x: x[1])
            
            # Accept if improvement or with aspiration probability
            if c5 < best_c5 or (np.random.random() < 0.1):
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = candidate_h.copy()
                
                # Update tabu list
                old_h_hash = hash_h(h)
                tabu_list.add(old_h_hash)
                if len(tabu_list) > tabu_length:
                    # Remove oldest entries
                    tabu_list.pop()
                
                h = candidate_h
                stagnation_counter = 0
            else:
                stagnation_counter += 1
        else:
            # No new candidates, just perturb anyway to explore
            h = perturb_h(h, step_size)
            stagnation_counter += 1
        
        # Update step size if stagnation
        if stagnation_counter >= max_stagnation:
            step_size *= 1.2
            tabu_list.clear()
            stagnation_counter = 0
        
        # Occasionally reset step size and tabu list
        if iteration % 500 == 0:
            step_size = 0.1
            tabu_list.clear()
    
    # Final validation of best solution
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
