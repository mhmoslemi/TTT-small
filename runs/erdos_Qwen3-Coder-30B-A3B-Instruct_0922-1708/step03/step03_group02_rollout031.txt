import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Initialize bounds for McCormick relaxation
    h_lower = np.zeros(n_points)
    h_upper = np.ones(n_points)
    
    # Solve sequence of convex relaxations using McCormick envelopes
    def solve_convex_relaxation(h_curr):
        # Create problem variables
        h_vars = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points/2, t >= 0]
        
        # Calculate objective using McCormick relaxation
        # For each k, we compute convolution sum
        for k in range(n_points):
            # Compute correlation approximation using McCormick relaxation
            if k == 0:
                # Direct sum for k=0: sum(h[i] * (1 - h[i])) for all i
                obj_expr = cp.sum(cp.multiply(h_vars, 1 - h_vars))
            else:
                # For k>0, we handle the convolution in parts
                # Use McCormick relaxation for bilinear terms
                obj_expr = 0
                for i in range(n_points - k):
                    # Approximate h[i] * (1 - h[i + k])
                    term = h_vars[i] * (1 - h_vars[i + k])
                    # Add McCormick constraints for the product term
                    obj_expr += term

            # Tighten constraint: t >= dx * obj_expr for each k
            constraints.append(t >= dx * obj_expr)
            
        # Minimize t
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h_vars.value, prob.value
            else:
                return h_curr, float('inf')
        except:
            return h_curr, float('inf')
    
    # Focus on gradient-based search
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < 1000:
        iter_count += 1
        
        # Current evaluation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Small gradient update to escape local minima
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Stronger gradient step with momentum
        lr = 0.01
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally solve a convex relaxation to further refine
        if iter_count % 20 == 0:
            h_refined, val = solve_convex_relaxation(h)
            if val < best_c5:  # Use refined solution if it's better
                h = h_refined
                # Recalculate to update best
                correlation = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(correlation) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
