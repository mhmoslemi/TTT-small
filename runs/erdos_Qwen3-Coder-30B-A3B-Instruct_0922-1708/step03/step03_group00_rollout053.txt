import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 20
    inertia_w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Start with initial solution
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    pbest.append(h.copy())
    pbest_vals.append(float('inf'))
    
    # Add random particles
    for _ in range(n_particles - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        particles.append(h_rand)
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
        pbest.append(h_rand)
        pbest_vals.append(float('inf'))
    
    # Global best tracking
    gbest = h.copy()
    gbest_val = float('inf')
    
    max_no_improvement = 50
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h_particle = particles[i]
            
            # Compute the overlap integral (C5)
            correlation = np.correlate(h_particle, 1 - h_particle, mode='full')
            c5_particle = np.max(correlation) * dx
            
            if c5_particle < pbest_vals[i]:
                pbest_vals[i] = c5_particle
                pbest[i] = h_particle.copy()
                
            if c5_particle < gbest_val:
                gbest_val = c5_particle
                gbest = h_particle.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Update global best if there's been no improvement
        if no_improvement_count > max_no_improvement:
            # Restart with a new random solution
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
            particles[0] = h_restart
            velocities[0] = np.random.uniform(-max_vel, max_vel, n_points)
            pbest[0] = h_restart
            pbest_vals[0] = float('inf')
            no_improvement_count = 0
            
        # Update particles based on PSO rules
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            vel = (inertia_w * velocities[i] + 
                   c1 * r1 * (pbest[i] - particles[i]) + 
                   c2 * r2 * (gbest - particles[i]))
            
            # Clip velocities
            vel = np.clip(vel, -max_vel, max_vel)
            
            # Update position 
            new_pos = particles[i] + vel
            
            # Apply constraints
            new_pos = np.clip(new_pos, 0, 1)
            
            # Rescale to satisfy integral constraint
            new_pos = new_pos * (n_points / 2) / np.sum(new_pos)
            
            particles[i] = new_pos
            velocities[i] = vel
            
            # Store current best
            if best_c5 > gbest_val:
                best_c5 = gbest_val
                best_h = gbest.copy()
        
        # Occasionally refine best solution with gradient descent
        if iteration % 10 == 0:
            h_refined = gbest.copy()
            initial_c5 = gbest_val
            
            # Simple GD refinement on the best solution
            for _ in range(20):
                grad = np.zeros_like(h_refined)
                eps = 1e-4
                
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - initial_c5) / eps
                
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h_refined - lr * grad
                
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_refined = h_new
                
            corr_refined = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5_refined = np.max(corr_refined) * dx
            
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
        
        if no_improvement_count > max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
