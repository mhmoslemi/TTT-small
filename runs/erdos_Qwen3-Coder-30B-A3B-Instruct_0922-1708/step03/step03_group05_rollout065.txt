import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Convex relaxation and solve
        h_relaxed = cp.Variable(n_points)
        constraints = [h_relaxed <= 1, h_relaxed >= 0, cp.sum(h_relaxed) == n_points/2]
        
        # Compute overlaps for all lags
        overlaps = []
        for k in range(n_points):
            overlap = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_points-k]))
            overlaps.append(overlap)
        objective = cp.Minimize(cp.max(cp.vstack(overlaps)))
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == 'optimal':
                h_relaxed_sol = h_relaxed.value
            else:
                h_relaxed_sol = h
        except:
            h_relaxed_sol = h
            
        # ROUNDING with adjustment
        h_rounded = (h_relaxed_sol >= 0.5).astype(float)
        current_sum = np.sum(h_rounded)
        
        if current_sum > n_points / 2:
            # Need to flip some 1s to 0s
            diff = int(current_sum - n_points / 2)
            indices = np.argsort(h_relaxed_sol[h_rounded == 1])
            for i in range(diff):
                if i < len(indices):
                    idx = np.where(h_rounded == 1)[0][indices[i]]
                    h_rounded[idx] = 0
        elif current_sum < n_points / 2:
            # Need to flip some 0s to 1s
            diff = int(n_points / 2 - current_sum)
            indices = np.argsort(-h_relaxed_sol[h_rounded == 0])
            for i in range(diff):
                if i < len(indices):
                    idx = np.where(h_rounded == 0)[0][indices[i]]
                    h_rounded[idx] = 1
                    
        # Ensure equal sum
        h_rounded = h_rounded * (n_points / 2) / np.sum(h_rounded)
        
        # Evaluate
        correlation = np.correlate(h_rounded, 1 - h_rounded, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_rounded.copy()
            
        # Perturb for exploration
        pert = np.random.normal(0, 0.01, n_points)
        h_perturbed = h_rounded + pert
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Check perturbed solution
        correlation_pert = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        c5_pert = np.max(correlation_pert) * dx
        
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_perturbed.copy()
            
        h = h_rounded.copy()
        
    # Final evaluation
    correlation_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
