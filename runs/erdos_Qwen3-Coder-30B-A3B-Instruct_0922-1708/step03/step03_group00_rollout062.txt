import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Optimization loop with spectral approach
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Spectral optimization approach: fast Fourier transform
        # We optimize in frequency domain then transform back
        fft_h = np.fft.fft(h)
        fft_1_minus_h = np.fft.fft(1 - h)
        
        # Compute overlap in frequency domain
        spectral_overlap = fft_h * np.conj(fft_1_minus_h)
        
        # Get the index of maximum spectral overlap
        max_idx = np.argmax(np.abs(spectral_overlap))
        
        # Compute gradient in frequency domain
        grad_fft = np.zeros_like(fft_h, dtype=complex)
        eps = 1e-4
        for i in range(len(fft_h)):
            # Perturb one frequency component
            fft_h_perturbed = fft_h.copy()
            fft_h_perturbed[i] += eps
            
            # Reconstruct h from perturbed fft
            h_perturbed = np.fft.ifft(fft_h_perturbed).real
            # Clip and rescale
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            # Compute correlation
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            c5_perturbed = np.max(corr_perturbed) * dx
            
            grad_fft[i] = (c5_perturbed - c5) / eps
            
        # Update fft coefficients
        lr = 0.001 * (1 + 0.01 * iteration)**(-0.5)  # Adaptive learning rate
        fft_h_new = fft_h - lr * grad_fft
        
        # Inverse transform to get new h
        h_new = np.fft.ifft(fft_h_new).real
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally try a small random perturbation to escape local optima
        if iteration % 100 == 0:
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
