import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill population
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    def evaluate_fitness(h):
        correlation = correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def crossover(parent1, parent2):
        # Simple blend crossover
        alpha = np.random.random()
        child = alpha * parent1 + (1 - alpha) * parent2
        return child
    
    def mutate(h, rate=0.2, intensity=0.1):
        # Randomly mutate points
        mutated = h.copy()
        n_mutate = int(rate * n_points)
        indices = np.random.choice(n_points, n_mutate, replace=False)
        for i in indices:
            mutated[i] += np.random.normal(0, intensity)
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    def select_parents(population, fitnesses):
        # Tournament selection
        selected = []
        for _ in range(len(population)):
            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            if fitnesses[idx1] < fitnesses[idx2]:
                selected.append(population[idx1].copy())
            else:
                selected.append(population[idx2].copy())
        return selected
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness of population
        fitnesses = []
        for h in population:
            c5 = evaluate_fitness(h)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
        
        no_improvement_count += 1
        if no_improvement_count > 100:
            break
            
        # Selection
        selected = select_parents(population, fitnesses)
        
        # Create new population
        new_population = [best_h.copy()]  # Elitism
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population[:pop_size]
        
    return (best_h.tolist(), best_c5, n_points)
