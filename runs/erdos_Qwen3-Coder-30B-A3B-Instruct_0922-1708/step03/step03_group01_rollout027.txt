import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 constant from the step function h."""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def compute_gradient(h, dx):
    """Compute gradient of C5 with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # Find which shift gives the maximum overlap
    correlation = np.correlate(h, 1 - h, mode='full')
    max_k = np.argmax(correlation)
    # Convert full correlation index to actual shift k
    k = max_k - (n - 1)
    
    # The gradient contribution is based on the effect of changing h[i]
    # on the overlap term h[i](1-h[i+k]) for the maximizing shift
    for i in range(n):
        j = i + k
        if 0 <= j < n:
            grad[i] += dx * (1 - 2 * h[j])
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    start_time = time.time()
    iteration = 0
    max_iter = 10000
    
    # Constraint for sum(h) = n_points/2
    def constraint_func(x):
        return np.sum(x) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Bounds for h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current value of objective and gradient
        c5 = compute_c5(h, dx)
        grad = compute_gradient(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Perform optimization step
        res = minimize(
            lambda x: compute_c5(x, dx),
            h,
            method='SLSQP',
            jac=compute_gradient,
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        if res.success:
            h = res.x
            # Re-project to feasible set
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Occasionally recompute for better direction
            if iteration % 10 == 0:
                h = h * (n_points / 2) / np.sum(h)
        else:
            # Fallback to gradient descent if optimization fails
            step_size = 0.001
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
    return (best_h.tolist(), best_c5, n_points)
