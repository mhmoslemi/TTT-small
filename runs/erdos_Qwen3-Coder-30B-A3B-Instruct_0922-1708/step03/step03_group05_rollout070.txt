import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    # Scale to correct integral
    h = h * (n_points / 2) / np.sum(h)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated annealing parameters
    temp = 0.1
    temp_decay = 0.99
    max_iter = 1000
    start_time = time.time()
    
    # Run optimization
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Evaluate current
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate neighbor
        h_new = h.copy()
        
        # Select random points to modify (5%)
        n_modify = max(1, n_points // 20)
        indices = np.random.choice(n_points, size=n_modify, replace=False)
        
        # Perturb values
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
            h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Re-normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
        else:
            # Accept with probability
            prob = np.exp((current_c5 - new_c5) / temp)
            if random.random() < prob:
                h = h_new
        
        # Anneal temperature
        temp *= temp_decay
    
    # Final evaluation
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
