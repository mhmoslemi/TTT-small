import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 20
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.1
    
    # Initialize particles
    particles = [h.copy()]
    for _ in range(n_particles - 1):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        particles.append(h_rand)
    
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [float('inf')] * n_particles
    
    # Global best tracking
    global_best = h.copy()
    global_best_fitness = float('inf')
    
    # PSO iterations
    pso_iterations = 200
    no_improvement_count = 0
    max_no_improvement = 50
    
    for pso_iter in range(pso_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            particle = particles[i]
            
            # Compute energy
            correlation = np.correlate(particle, 1 - particle, mode='full')
            fitness = np.max(correlation) * dx
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best[i] = particle.copy()
                
            # Update global best
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best = particle.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            # Check early termination
            if no_improvement_count >= max_no_improvement:
                break
                
        # Refine with gradient descent using global best
        if global_best_fitness < best_c5:
            best_c5 = global_best_fitness
            best_h = global_best.copy()
            
        # Update particles using PSO rules
        for i in range(n_particles):
            particle = particles[i]
            velocity = np.random.rand(n_points) * 2 - 1  # Random velocity in [-1, 1]
            velocity *= max_velocity
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (personal_best[i] - particle)
            social = c2 * r2 * (global_best - particle)
            velocity = inertia * velocity + cognitive + social
            
            # Update position
            particle_new = particle + velocity
            
            # Clip to bounds
            particle_new = np.clip(particle_new, 0, 1)
            
            # Rescale to satisfy constraint
            particle_new = particle_new * (n_points / 2) / np.sum(particle_new)
            
            particles[i] = particle_new
            
    # Final refinement by gradient descent on the best solution found
    h = best_h.copy()
    
    # Gradient descent refinement
    max_refinements = 1000
    lr = 0.01
    for i in range(max_refinements):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
        
        # Update with gradient
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update
        h = h_new
        
        # Update best if better
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
