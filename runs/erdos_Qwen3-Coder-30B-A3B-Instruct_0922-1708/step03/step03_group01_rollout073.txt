import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available, otherwise create Frank-Zadoff type sequence
    n_points = 200  # High resolution for global search
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a Frank-Zadoff-like sequence with minimal autocorrelation
        h = np.ones(n_points) * 0.5
        # Introduce some structure
        for i in range(n_points):
            h[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points)  # Basic sinusoidal
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Set up full overlap computation for gradient calculation
    def compute_overlap(h_vec):
        # Compute correlation for all shifts
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Get maximum (this is C5 * dx)
        max_corr = np.max(corr)
        return max_corr * dx

    # Gradient estimation for current maximum overlap shift
    def estimate_gradients(h_vec, eps=1e-6):
        grad = np.zeros_like(h_vec)
        max_overlap_idx = np.argmax(np.correlate(h_vec, 1 - h_vec, mode="full"))
        max_k = max_overlap_idx - (len(h_vec) - 1)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            
            corr_orig = np.correlate(h_vec, 1 - h_vec, mode="full")
            c5_orig = np.max(corr_orig) * dx
            
            grad[i] = (c5_plus - c5_orig) / eps
            
        return grad

    # Gradient descent based refinement
    alpha = 0.01
    while time.time() - start_time < budget_s - 1 and iteration < 3000:
        iteration += 1
        c5 = compute_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive decreasing step size
        if iteration % 100 == 0 and alpha > 1e-6:
            alpha *= 0.95
        
        # Compute gradient
        grad = estimate_gradients(h)
        
        # Update h with negative gradient
        h_new = h - alpha * grad
        
        # Project to feasible region and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally check for convergence
        if iteration % 200 == 0:
            c5_new = compute_overlap(h)
            if abs(c5 - c5_new) < 1e-8:
                break

    # Downsample to 100 for final output
    h_final = np.array(best_h[::2])  # choose every second sample
    h_final = h_final * (100 / 2) / np.sum(h_final)
    final_c5 = compute_overlap(h_final)  # re-calculate for correct n_points

    return (h_final.tolist(), final_c5, 100)
