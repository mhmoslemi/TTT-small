import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing with Nelder-Mead refinement
    T = 0.1
    alpha = 0.995
    steps = 0
    max_anneal_steps = 2000
    max_neldermead_steps = 1000
    
    # Initialize simplex for Nelder-Mead 
    simplex_size = min(10, n_points)
    simplex = [h.copy()]
    for _ in range(simplex_size - 1):
        perturbed = h.copy() + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        simplex.append(perturbed)
    
    # Annealing phase
    while steps < max_anneal_steps and time.time() - start_time < budget_s - 5:
        # Perturb a random subset of h
        perturb_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        h_new = h.copy()
        h_new[perturb_indices] += np.random.normal(0, 0.1, len(perturb_indices))
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept with simulated annealing acceptance probability
        if c5_new < best_c5 or np.random.rand() < math.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
        T *= alpha
        steps += 1
    
    # Nelder-Mead refinement phase
    nm_steps = 0
    while nm_steps < max_neldermead_steps and time.time() - start_time < budget_s - 1:
        # Evaluate all points in simplex
        values = []
        for point in simplex:
            corr = np.correlate(point, 1 - point, mode='full')
            c5_val = np.max(corr) * dx
            values.append(c5_val)
        
        # Sort simplex by function values
        sorted_indices = np.argsort(values)
        simplex = [simplex[i] for i in sorted_indices]
        values = [values[i] for i in sorted_indices]
        
        # Nelder-Mead operations
        # Reflection
        x_r = np.mean(simplex[:-1], axis=0) + 1.0 * (np.mean(simplex[:-1], axis=0) - simplex[-1])
        x_r = np.clip(x_r, 0, 1)
        x_r = x_r * (n_points / 2) / np.sum(x_r)
        
        corr_r = np.correlate(x_r, 1 - x_r, mode='full')
        c5_r = np.max(corr_r) * dx
        
        if values[0] <= c5_r < values[-2]:
            simplex[-1] = x_r
            values[-1] = c5_r
        else:
            # Expansion
            if c5_r < values[0]:
                x_e = np.mean(simplex[:-1], axis=0) + 2.0 * (np.mean(simplex[:-1], axis=0) - simplex[-1])
                x_e = np.clip(x_e, 0, 1)
                x_e = x_e * (n_points / 2) / np.sum(x_e)
                
                corr_e = np.correlate(x_e, 1 - x_e, mode='full')
                c5_e = np.max(corr_e) * dx
                
                if c5_e < c5_r:
                    simplex[-1] = x_e
                    values[-1] = c5_e
                else:
                    simplex[-1] = x_r
                    values[-1] = c5_r
            else:
                # Contraction
                if c5_r < values[-1]:
                    x_c = np.mean(simplex[:-1], axis=0) + 0.5 * (simplex[-1] - np.mean(simplex[:-1], axis=0))
                    x_c = np.clip(x_c, 0, 1)
                    x_c = x_c * (n_points / 2) / np.sum(x_c)
                    
                    corr_c = np.correlate(x_c, 1 - x_c, mode='full')
                    c5_c = np.max(corr_c) * dx
                    
                    if c5_c < values[-1]:
                        simplex[-1] = x_c
                        values[-1] = c5_c
                    else:
                        # Shrink
                        for i in range(1, len(simplex)):
                            simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                            simplex[i] = np.clip(simplex[i], 0, 1)
                            simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
                else:
                    # Shrink
                    for i in range(1, len(simplex)):
                        simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                        simplex[i] = np.clip(simplex[i], 0, 1)
                        simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
        
        # Update best if found
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
        
        nm_steps += 1
    
    # Final refinement with local optimization
    # Use scipy's minimize with bounds
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    # Convert to scipy format
    h_scipy = best_h.copy()
    
    # Bounds for each variable between 0 and 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Do a local optimization for final refinement
    # Use SLSQP method for constrained optimization
    result = minimize(objective, h_scipy, method='SLSQP',
                      bounds=bounds, 
                      options={'maxiter': 50},
                      tol=1e-6)
    
    if result.success:
        h_final = result.x.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
    else:
        h_final = best_h.copy()
    
    # Final evaluation
    correlation = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (h_final.tolist(), final_c5, n_points)
