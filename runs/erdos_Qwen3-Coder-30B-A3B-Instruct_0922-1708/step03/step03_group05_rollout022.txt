import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    T0 = 0.1
    alpha = 0.99
    perturb_fraction = 0.05
    perturb_magnitude = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Annealing schedule
    T = T0
    iteration = 0
    
    # Precompute indices for efficiency
    indices = np.arange(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate new candidate
        h_new = h.copy()
        num_perturb = max(1, int(n_points * perturb_fraction))
        perturb_indices = np.random.choice(indices, size=num_perturb, replace=False)
        
        for i in perturb_indices:
            # Random perturbation
            delta = np.random.uniform(-perturb_magnitude, perturb_magnitude)
            h_new[i] += delta
            
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept with probability based on temperature
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            h = h_new.copy()
        
        # Update temperature
        T *= alpha
    
    # Final check with better precision
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
