import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Initialize optimization parameters
    max_iter = 1000
    tol = 1e-8
    start_time = time.time()
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_fun(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_fun}
    
    def compute_c5_and_grad(h):
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        shifts = np.arange(len(h))
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        c5 = max_overlap * dx
        
        # Compute gradient w.r.t. h
        grad = np.zeros_like(h)
        # For shift k_max, compute gradient of the overlap term h[i] * (1 - h[i+k_max])
        for i in range(len(h)):
            if i + k_max < len(h):
                grad[i] += dx * (1 - h[i + k_max])
            if i - k_max >= 0:
                grad[i] += dx * (-h[i - k_max])
                
        return c5, grad
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Evaluate current solution
        c5, _ = compute_c5_and_grad(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Use SLSQP optimizer with gradient information
        res = minimize(
            lambda x: compute_c5_and_grad(x)[0],
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            jac=lambda x: compute_c5_and_grad(x)[1],
            options={'maxiter': 10, 'ftol': tol, 'gtol': tol},
            callback=lambda x: None
        )
        
        if res.success:
            h = res.x.copy()
            # Project to feasible region
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    if iteration > 0:
        c5_final, _ = compute_c5_and_grad(best_h)
        best_c5 = c5_final
    
    return (best_h.tolist(), best_c5, n_points)
