import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Start with a binary search approach for C5
    C_low = 0.0
    C_high = 0.38085  # Current record
    tolerance = 1e-5
    
    while time.time() - start_time < budget_s - 1 and C_high - C_low > tolerance:
        C_test = (C_low + C_high) / 2.0
        
        # Try to find feasible h for this C_test
        # Use a simple gradient-based approach to see if it's achievable
        h_trial = h.copy()
        
        # Perform gradient descent with adaptive learning rate
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current C5
            correlation = np.correlate(h_trial, 1 - h_trial, mode='full')
            current_c5 = np.max(correlation) * dx
            
            # If better than our target, try to reduce further
            if current_c5 <= C_test:
                # Try a simple modification to see if it improves
                # Perturb the function slightly to get closer to target
                pert = np.random.normal(0, 0.01, n_points)
                h_new = h_trial + pert
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                new_c5 = np.max(corr_new) * dx
                
                if new_c5 < current_c5:
                    h_trial = h_new
                else:
                    # Small perturbation not helping, accept current
                    break
            else:
                # Current is too large, we're trying to reduce it
                # Compute gradient of correlation with respect to h
                eps = 1e-4
                grad = np.zeros_like(h_trial)
                for i in range(n_points):
                    h_plus = h_trial.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[i] = (c5_plus - current_c5) / eps
                
                # Update with gradient descent
                lr = 0.02 if current_c5 > C_test + 0.01 else 0.005
                h_new = h_trial - lr * grad
                
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                if np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx < current_c5:
                    h_trial = h_new
                
        # Check if we found a feasible solution for C_test
        corr_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
        trial_c5 = np.max(corr_trial) * dx
        
        if trial_c5 <= C_test:
            C_high = C_test  # We found a feasible solution
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = h_trial.copy()
        else:
            C_low = C_test  # This value was too low
    
    # Final iteration to try to refine
    # Try to do a better optimization to find a lower bound
    if time.time() - start_time < budget_s - 1:
        # Try various initializations for better results
        for i in range(10):
            if time.time() - start_time > budget_s - 1:
                break

            for j in range(n_points):
                h_temp[j] = 0.5 + 0.5 * np.sin(2 * np.pi * j / n_points)
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            
            # Random perturbations
            for _ in range(50):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                idx = np.random.randint(n_points)
                h_temp[idx] += np.random.randn() * 0.02
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                
                corr = np.correlate(h_temp, 1 - h_temp, mode='full')
                c5 = np.max(corr) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_temp.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
