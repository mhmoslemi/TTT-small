import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # Gradient-based optimization with smarter search
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Alternative: try different perturbations to escape local minima
        if iteration % 5 == 0:
            # Try perturbing in different directions
            perturbations = [
                np.random.normal(0, 0.01, n_points),
                np.random.uniform(-0.02, 0.02, n_points),
                np.random.laplace(0, 0.01, n_points)
            ]
            for pert in perturbations:
                h_trial = h + pert
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
                trial_c5 = compute_c5(h_trial)
                if trial_c5 < current_c5:
                    h = h_trial
                    break
            else:
                # No improvement found, continue with gradient descent
                pass
        else:
            # Use a standard gradient descent approach
            eps = 1e-5
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c5_plus = compute_c5(h_plus)
                grad[i] = (c5_plus - current_c5) / eps
            
            # Adaptive learning rate
            lr = 0.002 * (1.0 + 0.5 * np.sin(iteration / 10))
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
    
    # Final check with exact computation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
