import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 500:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Physics-inspired update: relax and solve convex problem
        # Define the energy function with smoothing term
        # E(h) = C5 + lambda * sum((h[i] - h[j])**2)
        lambda_reg = 0.1 * np.exp(-0.002 * iteration)
        
        # Create relaxation using CVXPY
        h_var = cp.Variable(n_points)
        
        # Objective: minimize the overlap + regularization
        # We linearize the max over lags by considering one large expression
        # Overlap term
        overlap_term = 0
        for k in lags:
            # For large k, the overlap sums over valid indices
            if k < n_points:
                overlap_term += cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
        
        # Regularization term
        reg_term = cp.sum_squares(h_var - cp.mean(h_var)) * lambda_reg
        
        # Combine objective
        objective = cp.Minimize(overlap_term * dx + reg_term)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Solve relax problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array(h_var.value).reshape(-1)
            else:
                # Revert to gradient step if convex solve fails
                h_new = h.copy()
                lr = 0.002
                grad = np.zeros_like(h)
                eps = 1e-6
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5) / eps
                h_new = h - lr * grad
        except:
            # Fallback to simple gradient step
            h_new = h.copy()
            lr = 0.002
            grad = np.zeros_like(h)
            eps = 1e-6
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            h_new = h - lr * grad
        
        # Apply physics-inspired perturbations and projection
        # Add noise to allow escape from local optima
        noise = 0.001 * np.random.randn(n_points)
        h_new += noise
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
