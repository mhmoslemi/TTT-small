import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # PSO parameters
    num_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    inertia_weight = 0.7
    max_velocity = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Seed some particles with initial solution
    num_seeded = num_particles // 10
    for i in range(num_seeded):
        particles.append(h.copy())
        velocities.append(np.random.uniform(-max_velocity, max_velocity, n_points))
    
    # Initialize other particles with random feasible solutions
    for i in range(num_seeded, num_particles):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        particles.append(h_rand)
        velocities.append(np.random.uniform(-max_velocity, max_velocity, n_points))
    
    # Track best positions
    p_best = [p.copy() for p in particles]
    p_best_values = [float('inf')] * num_particles
    g_best = h.copy()
    g_best_value = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(num_particles):
            particle = particles[i]
            
            # Compute overlap integral
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < p_best_values[i]:
                p_best_values[i] = c5
                p_best[i] = particle.copy()
            
            if c5 < g_best_value:
                g_best_value = c5
                g_best = particle.copy()
        
        # Update particles using PSO update rules
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            v = (w * velocities[i] + 
                 c1 * r1 * (p_best[i] - particles[i]) + 
                 c2 * r2 * (g_best - particles[i]))
            v = np.clip(v, -max_velocity, max_velocity)
            particles[i] = particles[i] + v
            
            # Project to feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            velocities[i] = v
    
    # Final check of global best
    correlation = np.correlate(g_best, 1 - g_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (g_best.tolist(), final_c5, n_points)
