import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up optimization parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.99
    T = T0
    start_time = time.time()
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 500
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Compute current C₅
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Generate neighbor solution
        h_new = h.copy()
        
        # Perturb 5% of points
        indices_to_perturb = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
        for i in indices_to_perturb:
            # Add small random noise
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
            
        # Clamp and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C₅
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject based on simulated annealing
        if new_c5 < current_c5:
            h = h_new
        else:
            # Accept with probability based on temperature
            acceptance_prob = np.exp((current_c5 - new_c5) / T)
            if np.random.random() < acceptance_prob:
                h = h_new
                
        # Update temperature
        T *= alpha
        
        # Periodic rescaling to maintain solution quality
        if iteration % 200 == 0 and iteration > 0:
            h = h * (n_points / 2) / np.sum(h)
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
