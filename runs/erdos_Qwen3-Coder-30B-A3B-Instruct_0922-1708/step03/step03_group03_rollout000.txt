import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a Barker-like initialization
        h = np.zeros(n_points)
        # Place pulses to reduce overlap
        pulse_positions = [0, 25, 50, 75]
        for pos in pulse_positions:
            if pos < n_points:
                h[pos] = 1.0
        # Scale to meet constraint
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define objective and constraints
    def objective(h):
        # Return negative because we minimize
        correlation = np.correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx
    
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Try Optimization with different initializations
    for restart in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation of current best
        h_init = best_h.copy() + np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        try:
            result = minimize(
                objective,
                h_init,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-6},
                tol=1e-8
            )
            
            if result.success:
                h_opt = result.x
                correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
                c5_opt = np.max(correlation) * dx
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
            
        # Simple gradient descent step as fallback
        h = best_h.copy()
        for i in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient estimate
            grad = np.zeros_like(h)
            eps = 1e-6
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - best_c5) / eps
            
            # Update step
            lr = 0.01
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Evaluate
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
