import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            x_orig = np.linspace(0, 2, len(h0))
            x_target = np.linspace(0, 2, n_points)
            h0 = np.interp(x_target, x_orig, h0)
        h0 = h0 / np.sum(h0) * (n_points / 2)
        particles.append(h0)
    else:
        h0 = np.ones(n_points) * 0.5
        h0 = h0 * (n_points / 2) / np.sum(h0)
        particles.append(h0)
    
    # Generate other particles with small random perturbations
    for i in range(n_particles - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
    
    particles = np.array(particles)
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    # Personal best positions and their fitness
    personal_best = particles.copy()
    personal_best_fitness = np.full(n_particles, np.inf)
    
    # Global best
    global_best_idx = 0
    global_best_fitness = np.inf
    
    start_time = time.time()
    step = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            # Ensure sum constraint
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < personal_best_fitness[i]:
                personal_best_fitness[i] = c5
                personal_best[i] = h.copy()
                
            if c5 < global_best_fitness:
                global_best_fitness = c5
                global_best_idx = i
                stagnation_count = 0
                
        stagnation_count += 1
        
        # PSO update
        c1, c2 = 2.0, 2.0
        inertia_weight = 1.0
        
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            vel = inertia_weight * velocities[i] + \
                  c1 * r1 * (personal_best[i] - particles[i]) + \
                  c2 * r2 * (personal_best[global_best_idx] - particles[i])
            
            # Apply velocity bounds
            vel = np.clip(vel, -0.2, 0.2)
            velocities[i] = vel
            particles[i] += vel
            
            # Ensure bounds
            particles[i] = np.clip(particles[i], 0, 1)
            # Rescale to maintain sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
        # Occasionally reset some particles to avoid stagnation
        if step % 20 == 0 and stagnation_count > 10:
            for i in range(n_particles // 5):
                pi = np.random.randint(0, n_particles)
                h = np.random.rand(n_points) * 0.2 + 0.4
                h = h * (n_points / 2) / np.sum(h)
                particles[pi] = h
                
    # Return best solution
    best_h = personal_best[global_best_idx]
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
