import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # TABU search parameters
    tabu_list = set()
    tabu_length = 50
    max_perturbations = int(0.2 * n_points)
    step_size = 0.1
    stagnation_count = 0
    max_stagnation = 100
    aspiration_probability = 0.1
    
    start_time = time.time()
    iteration = 0
    
    def hash_solution(sol):
        # Create a hashable representation of the solution
        return hashlib.md5(sol.astype(np.float32).tobytes()).hexdigest()
    
    def perturb_solution(current_h, step):
        # Create a perturbed solution
        candidate = current_h.copy()
        indices = np.random.choice(n_points, size=max_perturbations, replace=False)
        for i in indices:
            delta = np.random.uniform(-step, step)
            candidate[i] += delta
        
        # Clip to [0,1]
        candidate = np.clip(candidate, 0, 1)
        
        # Rescale to maintain constraint sum(h) = n_points/2
        scale_factor = (n_points / 2) / np.sum(candidate)
        candidate = candidate * scale_factor
        
        return candidate
    
    current_h = h.copy()
    current_c5 = best_c5
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        stagnation_count += 1
        
        # Generate candidate solutions
        candidates = []
        for k in range(50):
            candidate_h = perturb_solution(current_h, step_size)
            candidate_hash = hash_solution(candidate_h)
            
            if candidate_hash not in tabu_list:
                candidate_c5 = evaluate_erdos_solution(candidate_h)
                candidates.append((candidate_h, candidate_c5, candidate_hash))
        
        # Find the best candidate not in TABU list
        if not candidates:
            # If all candidates are in TABU, force a move
            num_attempts = 0
            while num_attempts < 10:
                candidate_h = perturb_solution(current_h, step_size)
                candidate_hash = hash_solution(candidate_h)
                if candidate_hash not in tabu_list:
                    candidate_c5 = evaluate_erdos_solution(candidate_h)
                    candidates.append((candidate_h, candidate_c5, candidate_hash))
                    break
                num_attempts += 1
            if not candidates:
                # Fallback if no valid moves
                candidate_h = perturb_solution(current_h, step_size)
                candidate_c5 = evaluate_erdos_solution(candidate_h)
                candidates.append((candidate_h, candidate_c5, hash_solution(candidate_h)))
        
        best_candidate_h, best_candidate_c5, best_candidate_hash = min(candidates, key=lambda x: x[1])
        
        # Update best solution if found
        if best_candidate_c5 < best_c5:
            best_c5 = best_candidate_c5
            best_h = best_candidate_h.copy()
            stagnation_count = 0
        
        # Update TABU list
        if len(tabu_list) >= tabu_length:
            tabu_list.pop()
        tabu_list.add(hash_solution(current_h))
        
        # Update current solution
        if best_candidate_c5 < current_c5:
            current_h = best_candidate_h.copy()
            current_c5 = best_candidate_c5
        else:
            # With some probability, accept a worse solution to escape local minima
            if np.random.random() < aspiration_probability:
                current_h = best_candidate_h.copy()
                current_c5 = best_candidate_c5
        
        # Adapt step size if stagnation detected
        if stagnation_count >= max_stagnation:
            step_size *= 1.2
            stagnation_count = 0
            tabu_list.clear()
        
        # Safety exit condition
        if iteration > 5000:
            break
    
    # Final evaluation of the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
