import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing with adaptive cooling
    T = 0.1
    alpha = 0.99
    epsilon = 0.05
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 2:
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > max_stagnation:
            break
            
        # Generate neighbor
        h_new = h + np.random.normal(0, epsilon, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept or reject
        if c5_new < c5 or np.random.rand() < np.exp(-(c5_new - c5) / T):
            h = h_new.copy()
            
        # Cooling
        T *= alpha
        if T < 1e-6:
            T = 1e-6
        
    # Fine-tune with gradient descent
    max_iter = 100
    lr = 0.001
    decay_rate = 0.95
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 and gradient using finite differences
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        c5 = np.max(correlation) * dx

        grad = np.zeros_like(best_h)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            correlation_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(correlation_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps

        # Update
        best_h = best_h - lr * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        lr *= decay_rate
        
        # Check if better
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            
    return (best_h.tolist(), best_c5, n_points)
