import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.99
    epsilon = 0.05
    max_no_improvement = 100
    
    # Annealing loop
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 5 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Generate neighbor with perturbation
        h_new = h + np.random.normal(0, epsilon, n_points)
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy the constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlap
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept or reject the new solution
        if c5_new < c5 or np.random.rand() < np.exp(-(c5_new - c5) / T):
            h = h_new
            
        # Cool down the temperature
        T *= alpha
        
        # Occasionally try a small random perturbation to escape local optima
        if iteration % 500 == 0:
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Refinement with gradient descent
    max_refinement_iters = 500
    learning_rate = 0.001
    
    for i in range(max_refinement_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Estimate gradient with finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Update with gradient
        h_new = h - learning_rate * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adjust learning rate
        if i % 50 == 0 and i > 0:
            learning_rate *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)
