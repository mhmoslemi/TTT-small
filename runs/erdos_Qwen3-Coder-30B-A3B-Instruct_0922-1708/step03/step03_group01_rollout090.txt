import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search for minimal C5
    low = 0.38
    high = 0.4
    tolerance = 1e-4
    
    while high - low > tolerance and time.time() - start_time < budget_s - 1:
        c_mid = (low + high) / 2.0
        
        # Define the optimization problem
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # Compute the overlap sum for shift k:
            # sum_{i=0}^{n_points-1} h[i] * (1 - h[(i+k) % n_points])
            overlap = 0
            for i in range(n_points):
                overlap += h_vars[i] * (1 - h_vars[(i + k) % n_points])
            constraints.append(overlap <= c_mid / dx)
        
        # Objective
        objective = cp.Minimize(0)  # We're only checking feasibility
        
        # Problem setup
        prob = cp.Problem(objective, constraints)
        
        try:
            # Try to solve
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                # Check if solution is valid and update best
                h_sol = h_vars.value
                if h_sol is not None:
                    # Normalize and clip
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    
                    # Evaluate C5 for this solution
                    corr = np.correlate(h_sol, 1 - h_sol, mode='full')
                    eval_c5 = np.max(corr) * dx
                    
                    if eval_c5 < best_c5:
                        best_c5 = eval_c5
                        best_h = h_sol.copy()
                        
                    high = c_mid
                else:
                    low = c_mid
            else:
                low = c_mid
        except:
            low = c_mid
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
