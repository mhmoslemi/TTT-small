import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a structured pattern
        h = np.zeros(n_points)
        # Create a two-peak pattern
        start1, end1 = int(0.2/dx), int(0.4/dx)
        start2, end2 = int(1.2/dx), int(1.4/dx)
        h[start1:end1] = 1.0
        h[start2:end2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Parameters
    T = 0.1  # Initial temperature
    alpha = 0.995  # Cooling factor
    lr = 0.01  # Learning rate
    eps = 1e-5  # Gradient step size
    perturb_rate = 0.1  # Fraction of elements to perturb
    
    start_time = time.time()
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    def project(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    def sa_step(h_vec, T):
        h_new = h_vec.copy()
        # Perturb a random subset of elements
        indices = np.random.choice(n_points, int(perturb_rate * n_points), replace=False)
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
        h_new = project(h_new)
        return h_new
    
    # Optimizer loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Simulated Annealing Step
        h_candidate = sa_step(h, T)
        c5_candidate = compute_c5(h_candidate)
        
        # Accept or reject based on SA criteria
        if c5_candidate < compute_c5(h) or np.random.rand() < np.exp(-(c5_candidate - compute_c5(h)) / T):
            h = h_candidate
            
        # Gradient Descent Step
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - compute_c5(h)) / eps
            
        # Update with gradient descent
        h_new = h - lr * grad
        h_new = project(h_new)
        
        # Update parameters
        if compute_c5(h_new) < compute_c5(h):
            lr = min(lr * 1.01, 0.02)
        else:
            lr = max(lr * 0.95, 0.001)
            
        h = h_new
        T *= alpha
        
        # Update best solution
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final computation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
