import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    def create_individual():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_orig, h)
        else:
            h = np.random.random(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    pop_size = 50
    population = [create_individual() for _ in range(pop_size)]
    
    def evaluate(individual):
        correlation = np.correlate(individual, 1 - individual, mode='full')
        return np.max(correlation) * dx
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Track convergence
    no_improvement_count = 0
    prev_best = float('inf')
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        iteration += 1
        # Evaluate population
        fitnesses = [evaluate(ind) for ind in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)
        current_best = population[current_best_idx]
        current_c5 = fitnesses[current_best_idx]
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Selection (keep top 40%)
        selected_indices = np.argsort(fitnesses)[:int(pop_size * 0.4)]
        selected = [population[i] for i in selected_indices]
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Crossover
            if len(selected) >= 2:
                i1, i2 = np.random.choice(len(selected), 2, replace=False)
                parent1, parent2 = selected[i1], selected[i2]
                
                # Blend crossover
                alpha = np.random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
                # Mutation
                if np.random.random() < 0.2:  # 20% chance to mutate
                    # Change 5% of points
                    num_mutations = int(0.05 * n_points)
                    mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
                    for idx in mutation_indices:
                        noise = np.random.normal(0, 0.1)
                        child[idx] += noise
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                
                new_population.append(child)
        
        population = new_population[:pop_size]
    
    if best_h is None:
        # Fallback to single optimization step if no good solution found
        h = create_individual()
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_vec):
            h_arr = np.array(h_vec)
            correlation = np.correlate(h_arr, 1 - h_arr, mode='full')
            return -np.max(correlation) * dx  # Negative for minimization
        
        def constraint(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        res = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        
        if res.success:
            best_c5 = -res.fun
            best_h = np.array(res.x)
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
