import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO parameters
    n_particles = 20
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    v_max = 0.1  # maximum velocity
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Add the initial solution as one particle
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    
    # Add random particles
    for _ in range(n_particles - 1):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        particles.append(h_rand)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * len(particles)
    
    # Global best
    global_best = h.copy()
    global_best_score = float('inf')
    
    # Optimization loop
    iteration = 0
    no_improvement = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        iteration += 1
        
        # Update global best if needed
        if iteration == 1:
            # Update global best from initial solutions
            for i, particle in enumerate(particles):
                # Compute overlap
                correlation = np.correlate(particle, 1 - particle, mode='full')
                score = np.max(correlation) * dx
                if score < global_best_score:
                    global_best_score = score
                    global_best = particle.copy()
                
        # Update all particles
        for i in range(len(particles)):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocity = (w * velocity + 
                        c1 * r1 * (personal_best[i] - particle) + 
                        c2 * r2 * (global_best - particle))
            
            # Clip velocity
            velocity = np.clip(velocity, -v_max, v_max)
            
            # Update particle position
            particle = particle + velocity
            
            # Apply boundary constraints
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Evaluate the new particle
            correlation = np.correlate(particle, 1 - particle, mode='full')
            score = np.max(correlation) * dx
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = particle.copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best = particle.copy()
                    no_improvement = 0

            # Update particle
            particles[i] = particle
            velocities[i] = velocity
                
        if global_best_score < best_c5:
            best_c5 = global_best_score
            best_h = global_best.copy()
            
        # Occasionally try random perturbations to escape local minima
        if iteration % 10 == 0:
            for i in range(len(particles)):
                if np.random.rand() < 0.3:
                    particles[i] = np.clip(particles[i] + np.random.normal(0, 0.005, n_points), 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        
        # Early termination
        no_improvement += 1
        
        # If best solution is still promising (C5 < 0.38), try gradient descent refinement
        if global_best_score < 0.38 and iteration % 50 == 0:
            h_refined = global_best.copy()
            for gd_iter in range(30):  # Limited gradient descent steps
                correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
                c5_current = np.max(correlation) * dx
                
                grad = np.zeros_like(h_refined)
                eps = 1e-5
                
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5_current) / eps
                
                # Adaptive learning rate
                lr = 0.05 / (1 + np.linalg.norm(grad) * 1e-3)
                h_refined_new = h_refined - lr * grad
                
                # Clip and rescale
                h_refined_new = np.clip(h_refined_new, 0, 1)
                h_refined_new = h_refined_new * (n_points / 2) / np.sum(h_refined_new)
                
                h_refined = h_refined_new
                
            # Test refined version
            correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5_refined = np.max(correlation) * dx
            if c5_refined < global_best_score:
                global_best = h_refined.copy()
                global_best_score = c5_refined
                
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined.copy()
                no_improvement = 0

    return (best_h.tolist(), best_c5, n_points)
