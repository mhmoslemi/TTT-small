import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def compute_c5(h):
        # Compute correlation for all shifts
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def compute_gradient(h, k_max):
        # Approximate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        c5_base = compute_c5(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    # Optimization loop
    iteration = 0
    alpha = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute the maximum overlap shift and gradient
        correlation = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(correlation)
        
        # Approximate gradient
        grad = compute_gradient(h, k_max)
        
        # Update step
        h_new = h - alpha * grad
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Adaptive learning rate
        if iteration % 50 == 0 and alpha > 1e-6:
            alpha *= 0.9
            
        h = h_new
        
    # Final verification
    best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
