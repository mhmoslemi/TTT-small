from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Precompute indices for efficient correlation
    indices = np.arange(n_points)

    def compute_c5_and_grad(h):
        # Compute full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        c5 = corr[max_k] * dx
        
        # Compute gradient for the maximal overlap
        grad = np.zeros_like(h)
        # For each position, compute the gradient contribution
        max_k_idx = max_k - (n_points - 1)  # convert full correlation index to shift
        
        # The gradient of the overlap integral w.r.t. h[i]
        # is (1-h[i+k]) - h[i-k] (if k is the max overlap shift) plus boundary effects
        for i in range(n_points):
            # i+k (with appropriate boundary handling)
            k = max_k_idx
            j1 = (i + k) % n_points  # equivalent to rolling h by k
            # i-k 
            j2 = (i - k) % n_points
            
            # derivative: partial of (h[i] * (1 - h[i+k])) + (h[i+k] * (1-h[i])) w.r.t. h[i]
            # This is a simplification considering that we're computing the gradient for 
            # the specific correlator that was maximal
            grad[i] = (1 - h[j1]) + (h[j2] - 1) 
        grad *= dx
        return c5, grad

    def objective_and_grad(h):
        h = np.clip(h, 0, 1)
        h_scaled = h * (n_points / 2) / np.sum(h)
        c5, grad = compute_c5_and_grad(h_scaled)
        return c5, grad

    def constraint_func(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Optimization parameters
    max_iter = 1000
    tol = 1e-8
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Make sure the initial point is valid
    if not np.isclose(np.sum(h), n_points / 2):
        h = h * (n_points / 2) / np.sum(h)
        
    start_time = time.time()
    iteration = 0
    
    # Run optimization
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Use SLSQP with analytical gradient if possible
        try:
            # Evaluate current value and gradient for this iteration
            c5, grad = objective_and_grad(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Manually update with a fixed step method to avoid gradient-dependent issues
            step_size = 0.001
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new if np.sum(h_new) > 0.1 else h  # avoid degen case
            
        except Exception as e:
            # On fail, fallback to initial solution
            break
            
        if np.sum(h) < 0.01:
            # Very wrong, reset
            h = best_h.copy() if 'best_h' in locals() else np.ones(n_points) * 0.5
            h = h * (n_points / 2) / np.sum(h)
        # Occasionally re-proj to maintain constraints
        if iteration % 50 == 0:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation
    if np.sum(h) > 0:
        c5_final, _ = compute_c5_and_grad(h)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
