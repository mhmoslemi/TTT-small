import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # PSO parameters
    num_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_vel = 0.1
    
    # Initialize swarm
    particles = []
    velocities = []
    
    # Add some particles seeded from initial solution
    num_seeded = max(1, num_particles // 10)
    for i in range(num_seeded):
        particles.append(h.copy())
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
    
    # Fill remaining particles with random feasible solutions
    for i in range(num_seeded, num_particles):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        particles.append(h_rand)
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
        
    # Track personal bests and global best
    p_best = [p.copy() for p in particles]
    p_best_scores = [float('inf')] * num_particles
    
    g_best = best_h.copy()
    g_best_score = best_c5
    
    # PSO main loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i, particle in enumerate(particles):
            # Compute the overlap integral
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < p_best_scores[i]:
                p_best_scores[i] = c5
                p_best[i] = particle.copy()
                
            if c5 < g_best_score:
                g_best_score = c5
                g_best = particle.copy()
                
        # Update particles
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            particle = particles[i]
            vel = velocities[i]
            
            # Update velocity
            vel = (w * vel + 
                   c1 * r1 * (p_best[i] - particle) + 
                   c2 * r2 * (g_best - particle))
            
            # Limit velocity
            vel = np.clip(vel, -max_vel, max_vel)
            
            # Update position
            particle = particle + vel
            
            # Project to feasible region
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            particles[i] = particle
            velocities[i] = vel
            
        # Update global best
        if g_best_score < best_c5:
            best_c5 = g_best_score
            best_h = g_best.copy()
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
