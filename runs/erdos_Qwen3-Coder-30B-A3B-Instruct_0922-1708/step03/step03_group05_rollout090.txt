import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try a more sophisticated approach using convex relaxation
        # Set up the convex problem
        h_relaxed = cp.Variable(n_points)
        constraints = [h_relaxed <= 1, h_relaxed >= 0, cp.sum(h_relaxed) == n_points/2]
        
        # Note: computing the actual max of correlations is non-linear in the variables
        # We'll compute a projection approach instead
        
        # Try greedy rounding with Lagrange multipliers
        h_rounded = np.round(h).clip(0, 1)
        current_sum = np.sum(h_rounded)
        
        if current_sum != n_points / 2:
            if current_sum > n_points / 2:
                # Need to reduce 1s to 0s
                # Find indices where h[i] is closest to 0.5
                diff = np.abs(h - 0.5)
                indices_to_reduce = np.argsort(diff)[:int(current_sum - n_points/2)]
                h_rounded[indices_to_reduce] = 0
            else:
                # Need to increase 0s to 1s
                diff = np.abs(h - 0.5)
                indices_to_increase = np.argsort(diff)[:int(n_points/2 - current_sum)]
                h_rounded[indices_to_increase] = 1
                
        # Use h_rounded as the new guess
        h_new = h_rounded
        h = h_new
        
        # Small noise to escape local minima
        noise_amplitude = 0.001
        noise = np.random.normal(0, noise_amplitude, n_points)
        h = h + noise
        h = h.clip(0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
