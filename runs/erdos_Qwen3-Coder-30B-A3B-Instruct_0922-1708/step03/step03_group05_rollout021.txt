import numpy as np
import time
import random
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    T0 = 0.1
    alpha = 0.99
    perturb_fraction = 0.05
    perturb_magnitude = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Main optimization loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute temperature
        T = T0 * (alpha ** iteration)
        
        # Generate neighborhood
        h_new = h.copy()
        indices_to_perturb = np.random.choice(n_points, int(n_points * perturb_fraction), replace=False)
        
        for idx in indices_to_perturb:
            delta = np.random.uniform(-perturb_magnitude, perturb_magnitude)
            h_new[idx] += delta
            
        # Clamp values to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        c5_new = np.max(correlate(h_new, 1 - h_new, mode='full')) * dx
        
        # Accept or reject
        if c5_new < c5 or np.random.rand() < np.exp((c5 - c5_new) / T):
            h = h_new
            
    # Final computation for best solution
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode='full')) * dx
    
    return (best_h.tolist(), final_c5, n_points)
