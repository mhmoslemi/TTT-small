def run(seed=42, budget_s=600, **kwargs):

    if 'initial_h_values' in globals():

    best_c5 = float('inf')
    
    start_time = time.time()

    def compute_c5(h_values):
        correlation = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(correlation) * dx
    
    # Initialize with a simple discrete relaxation and then refine
    # Start with a piecewise constant function which is often good for such problems
    # We'll use grid-based search around a few promising configurations
    
    # For this implementation, we'll still use gradient-based method but with 
    # better initialization, more refined steps, and early stopping
    
    # Generate a better initial point using a common heuristic for such problems:
    # The idea is to build a "peak" that captures most of the mass, 
    # with some extra near the boundaries
    
    def make_initial_clustered():
        # Create a piecewise constant or clustered structure
        # Maybe two or three spikes with some distribution
        h_new = np.zeros(n_points)
        
        # Try a 3-peak construction as a structured initializer
        mid = n_points // 2
        width = n_points // 4
        
        # Distribute mass in a way to avoid trivial overlaps
        h_new[mid - width//2 : mid + width//2] = 1.0
        h_new[0 : width//2] = 0.5
        h_new[-width//2:] = 0.5
        
        # Apply normalization
        h_new = h_new * (n_points / 2) / np.sum(h_new)

    h = make_initial_clustered()

        # Fallback to final check
        final_c5 = compute_c5(h)
    else:
        final_c5 = best_c5
    
    return (best_h.tolist(), final_c5, n_points)
