import numpy as np
from scipy.optimize import minimize
import time
import random
from scipy.spatial.distance import cdist

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.995
    perturbation_magnitude = 0.1
    
    # Nelder-Mead approach
    def evaluate_c5(current_h):
        # Make sure h is properly normalized
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        # Make sure h is in [0,1]
        current_h = np.clip(current_h, 0, 1)
        # Compute C_5
        correlation = np.correlate(current_h, 1 - current_h, mode='full')
        c5 = np.max(correlation) * dx
        return c5, current_h
    
    # Run Simulated Annealing first to explore space
    steps = 0
    sa_steps = 2000
    while steps < sa_steps and time.time() - start_time < budget_s - 10:
        # Randomly perturb some elements
        h_candidate = h.copy()
        num_perturb = max(1, int(n_points * 0.1))
        perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in perturb_indices:
            h_candidate[i] += np.random.normal(0, perturbation_magnitude)
            
        # Clip and re-normalize
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate
        c5_candidate, _ = evaluate_c5(h_candidate)
        
        # Accept based on simulated annealing
        if c5_candidate < best_c5 or np.random.rand() < np.exp(-(c5_candidate - best_c5) / T):
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
        
        T *= alpha
        steps += 1
    
    # Then run Nelder-Mead on the best solution so far
    # Convert numpy array to list of 1D arrays for Nelder-Mead
    def obj_func(x):
        # Clip x to [0,1]
        x = np.clip(x, 0, 1)
        # Re-normalize
        x = x * (n_points / 2) / np.sum(x)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx
        
    # Generate initial simplex
    def generate_simplex(center, size=0.05, n_points_m1=4):
        simplex = [center.copy()]
        for i in range(n_points_m1):
            perturbation = np.random.normal(0, size, len(center))
            point = center + perturbation
            # Ensure point is valid (in [0,1] and normalized)
            point = np.clip(point, 0, 1)
            point = point * (n_points/2) / np.sum(point)
            simplex.append(point)
        return simplex
    
    # Initialize with best_h
    current_h = best_h.copy()
    simplex = generate_simplex(current_h, size=0.05, n_points_m1=4)
    
    # Nelder-Mead iteration
    iter_count = 0
    max_iter = 1000
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        # Evaluate all vertices
        values = [obj_func(v) for v in simplex]
        
        # Sort by value
        indices = np.argsort(values)
        simplex_sorted = [simplex[i] for i in indices]
        values_sorted = [values[i] for i in indices]
        
        # Centroid of best points excluding worst
        centroid = np.mean(simplex_sorted[:-1], axis=0)
        
        # Reflection
        r_point = centroid + centroid - simplex_sorted[-1]
        r_point = np.clip(r_point, 0, 1)
        r_point = r_point * (n_points / 2) / np.sum(r_point)
        r_value = obj_func(r_point)
        
        if values_sorted[0] <= r_value < values_sorted[-2]:
            # Accept reflection
            simplex_sorted[-1] = r_point
        elif r_value < values_sorted[0]:
            # Expansion
            e_point = centroid + 2 * (r_point - centroid)
            e_point = np.clip(e_point, 0, 1)
            e_point = e_point * (n_points / 2) / np.sum(e_point)
            e_value = obj_func(e_point)
            
            if e_value < r_value:
                simplex_sorted[-1] = e_point
            else:
                simplex_sorted[-1] = r_point
        else:
            # Contraction
            c_point = centroid + 0.5 * (simplex_sorted[-1] - centroid)
            c_point = np.clip(c_point, 0, 1)
            c_point = c_point * (n_points / 2) / np.sum(c_point)
            c_value = obj_func(c_point)
            
            if c_value < values_sorted[-1]:
                simplex_sorted[-1] = c_point
            else:
                # Shrink towards best point
                for i in range(1, len(simplex_sorted)):
                    simplex_sorted[i] = simplex_sorted[0] + 0.5 * (simplex_sorted[i] - simplex_sorted[0])
        
        simplex = simplex_sorted
        iter_count += 1

        # Update best if we improved
        best_simplex_value = min(values_sorted)
        if best_simplex_value < best_c5:
            best_c5 = best_simplex_value
            best_h = simplex_sorted[0].copy()

    # Final check on best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx

    return (best_h.tolist(), final_c5, n_points)
