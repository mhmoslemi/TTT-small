import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency  
    lags = list(range(1, n_points))
    
    # Optimizer loop
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try gradient descent with multiple random restarts
        for restart in range(3):
            h_trial = h.copy()
            lr = 0.01 / (1 + iteration / 100)
            
            # Small random perturbation
            perturb = np.random.normal(0, 0.01, n_points)
            h_trial += perturb
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            # Compute gradient estimate
            grad = np.zeros_like(h_trial)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_trial.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update with gradient
            h_new = h_trial - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Better solution?
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < c5:
                h = h_new.copy()
                break
                
        # Use CVX for final refinement
        if iteration % 10 == 0:
            h_vars = cp.Variable(n_points)
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            try:
                objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.CBC, verbose=False, max_seconds=1.0)
                
                if prob.status == cp.OPTIMAL:
                    h_cvx = h_vars.value
                    if h_cvx is not None:
                        h_cvx = np.clip(h_cvx, 0, 1)
                        h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                        
                        correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                        c5_cvx = np.max(correlation) * dx
                        
                        if c5_cvx < best_c5:
                            best_h = h_cvx
                            best_c5 = c5_cvx
            except:
                pass
                
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
