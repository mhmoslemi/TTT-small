import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Binary search over C5
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-5
    
    def compute_c5(h_vector):
        correlation = np.correlate(h_vector, 1 - h_vector, mode='full')
        return np.max(correlation) * dx
    
    def solve_feasibility(h_current, C_target):
        # Create a copy of the current h for use in constraints
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Identify the most problematic lags by checking the current overlap
        correlation_current = np.correlate(h_current, 1 - h_current, mode='full')
        lag_indices = np.argsort(correlation_current)[-max(10, n_points//10):]  # Top 10% or 10 lags
        
        # Linearize the overlap constraints around current h
        for k in lag_indices:
            # Get the relevant indices for this lag
            if k < n_points:  # valid for forward convolution
                lag_values = np.zeros(n_points)
                for i in range(n_points):
                    lag_values[i] = h_current[i] * (1 - h_current[(i + k) % n_points])
                # Linearization point for the constraint
                # Sum of h[i] * (1 - h[(i+k)%n_points]) <= C_target * dx * n_points / dx = C_target * n_points
                # This is still simplified - form a linear approximation
                # Not exactly correct but close enough for this convex solver
                h_sum = 0
                for i in range(n_points):
                    h_sum += h_vars[i] * (1 - h_current[(i + k) % n_points]) - C_target * dx
                constraints.append(h_sum <= 0)
            
        # For more accurate feasibility, use the optimized dual approach:
        objective = cp.Minimize(cp.sum(h_vars))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h_vars.value
        except:
            pass
        return None
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 10 and C_high - C_low > tolerance:
        C_target = (C_low + C_high) / 2
        h_feasible = solve_feasibility(h, C_target)
        
        if h_feasible is not None:
            # If found a better feasible solution
            c5_val = compute_c5(h_feasible)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_feasible.copy()
                if c5_val < 0.3807:
                    break  # Acceleration term
            C_high = C_target
        else:
            C_low = C_target
            
        # Perform some gradient descent refinement in case of early convergence or continue improving
        # Stability on Newton-style descent is kept, using good heuristics
        if time.time() - start_time > 100:
            # Perform local search
            def objective(h_local):
                correlation = np.correlate(h_local, 1 - h_local, mode='full')
                return np.max(correlation) * dx
                
            def gradient(h_local):
                grad = np.zeros_like(h_local)
                eps = 1e-6
                for i in range(n_points):
                    h_plus = h_local.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - objective(h_local)) / eps
                return grad

            # Stochastic gradient descent step
            lr = 0.001 * (0.95 ** (time.time() - start_time) // 60)
            for _ in range(50):
                grad = gradient(h)
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                if objective(h_new) < objective(h):
                    h = h_new
                    if compute_c5(h) < best_c5:
                        best_c5 = compute_c5(h)
                        best_h = h.copy()
                else:
                    # Line search to find a good step
                    l = 0.5
                    h_candidate = h - l * grad
                    h_candidate = np.clip(h_candidate, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    if compute_c5(h_candidate) < objective(h):
                        h = h_candidate
                        if compute_c5(h) < best_c5:
                            best_c5 = compute_c5(h)
                            best_h = h.copy()
    
    # Final evaluation
    if compute_c5(best_h) < best_c5:
        best_c5 = compute_c5(best_h)
    return (best_h.tolist(), best_c5, n_points)
