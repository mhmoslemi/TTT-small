import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated annealing parameters
    T = 0.1
    alpha = 0.99
    epsilon = 0.05
    max_iter = 500
    patience = 100
    no_improve_count = 0
    
    start_time = time.time()
    iteration = 0
    
    # Annealing loop
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Generate neighbor solution
        h_new = h + np.random.normal(0, epsilon, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlap integral
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept or reject the new solution
        if c5_new < c5 or np.random.random() < np.exp(-(c5_new - c5) / T):
            h = h_new
            
        # Update temperature
        T *= alpha
        
        # Early termination if no improvement for a while
        if no_improve_count > patience:
            break
    
    # Refinement using gradient-based method
    h_refined = best_h.copy()
    
    # Use a more sophisticated gradient estimation
    def evaluate(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    iteration = 0
    lr = 0.001
    while time.time() - start_time < budget_s - 1 and iteration < 200:
        iteration += 1
        current_c5 = evaluate(h_refined)
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h_refined)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update with momentum
        h_refined = h_refined - lr * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Decrease learning rate
        if iteration % 50 == 0:
            lr *= 0.95
            
        # Update best if improved
        current_c5 = evaluate(h_refined)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
    
    return (best_h.tolist(), best_c5, n_points)
