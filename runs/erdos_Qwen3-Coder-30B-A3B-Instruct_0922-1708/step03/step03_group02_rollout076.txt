import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a structured initialization
        h = np.zeros(n_points)
        # Simple two-peak construction
        start1 = int(0.2/dx)
        end1 = int(0.4/dx)
        start2 = int(1.2/dx)
        end2 = int(1.4/dx)
        h[start1:end1] = 1.0
        h[start2:end2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize tracking variables
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Annealing parameters
    T = 0.1
    alpha = 0.995
    lr = 0.01
    
    start_time = time.time()
    
    # Function to compute C5
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Function to project h to feasible region
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing step
        h_trial = h.copy()
        # Perturb about 10% of points
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices:
            h_trial[i] += np.random.normal(0, 0.05)
        h_trial = project_h(h_trial)
        
        current_c5 = compute_c5(h)
        trial_c5 = compute_c5(h_trial)
        
        # Accept or reject based on simulated annealing
        if trial_c5 < current_c5 or np.random.rand() < np.exp(-(trial_c5 - current_c5) / T):
            h = h_trial.copy()
        
        # Gradient descent step
        grad = np.zeros(n_points)
        eps = 1e-5
        current_c5 = compute_c5(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update with gradient descent
        h_new = h - lr * grad
        
        # Ensure boundary constraints
        h_new = project_h(h_new)
        h = h_new.copy()
        
        # Update learning rate based on progress
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            lr *= 1.01  # Increase learning rate
        else:
            lr *= 0.95  # Decrease learning rate
        lr = np.clip(lr, 0.001, 0.02)  # Clip learning rate
        
        # Update temperature
        T *= alpha
        
        # Early termination for stability
        if T < 1e-6:
            T = 1e-6
    
    # Final computation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
