import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search bounds for C5
    low = 0.38
    high = 0.4
    tolerance = 1e-4
    
    # Precompute all possible shifts
    shifts = np.arange(n_points)
    
    while (high - low > tolerance) and (time.time() - start_time < budget_s - 1):
        c_mid = (low + high) / 2.0
        
        # Define the optimization variables
        h_vars = cp.Variable(n_points)
        
        # Define the constraints
        constraints = []
        
        # Bound constraints
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Overlap constraints
        for k in range(n_points):
            # Compute correlation for shift k
            corr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr += h_vars[i] * (1 - h_vars[j])
            constraints.append(corr <= c_mid / dx)
        
        # Objective: minimize C5 (implicit since we're checking feasibility)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    
                    # Check the actual C5 value
                    correlation = np.correlate(h_new, 1 - h_new, mode='full')
                    c5_new = np.max(correlation) * dx
                    
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    
                    high = c_mid
                else:
                    low = c_mid
            else:
                low = c_mid
        except Exception:
            low = c_mid
    
    # Final check with the best solution found
    if best_c5 == float('inf'):
        # Fall back to the initial solution
        correlation = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(correlation) * dx
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
