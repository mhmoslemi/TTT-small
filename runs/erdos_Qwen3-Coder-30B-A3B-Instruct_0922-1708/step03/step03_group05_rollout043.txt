import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Convex relaxation
        h_relaxed = cp.Variable(n_points)
        
        # Build constraints
        constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points/2]
        
        # Formulate the max overlap as linear constraints
        max_overlap = cp.Variable()
        overlap_exprs = []
        
        for k in range(n_points):
            # Overlap at lag k: sum h[i] * (1 - h[i+k])
            if k == 0:
                overlap = cp.sum(h_relaxed * (1 - h_relaxed))
            else:
                overlap = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_points-k]))
            overlap_exprs.append(overlap)
        
        # Constraint that max_overlap should be >= each overlap
        for expr in overlap_exprs:
            constraints.append(max_overlap >= expr)
        
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, reltol=1e-6)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed_solution = h_relaxed.value
            else:
                # Fallback: local optimizer
                h_relaxed_solution = h.copy()
        except:
            # If convex solve fails, fallback to previous best or random
            h_relaxed_solution = h.copy()
        
        # Rounding and adjustment
        h_binary = (h_relaxed_solution >= 0.5).astype(float)
        current_sum = np.sum(h_binary)
        target_sum = n_points / 2
        
        if current_sum > target_sum:
            # Need to reduce sum
            diff = current_sum - target_sum
            # Find indices with smallest relaxed values
            indices = np.argsort(h_relaxed_solution)[:int(diff)]
            for i in indices:
                h_binary[i] = 0
        elif current_sum < target_sum:
            # Need to increase sum
            diff = target_sum - current_sum
            # Find indices with largest relaxed values
            indices = np.argsort(h_relaxed_solution)[-int(diff):]
            for i in indices:
                h_binary[i] = 1
                
        # Ensure exact sum
        current_sum = np.sum(h_binary)
        if current_sum != target_sum:
            diff = int(target_sum - current_sum)
            if diff > 0:
                # Flip 0s to 1s
                indices = np.argsort(h_relaxed_solution)[::-1][:diff]
                for i in indices:
                    if h_binary[i] == 0:
                        h_binary[i] = 1
            else:
                # Flip 1s to 0s
                indices = np.argsort(h_relaxed_solution)[:abs(diff)]
                for i in indices:
                    if h_binary[i] == 1:
                        h_binary[i] = 0
        
        # Compute C5
        # Use correlation to compute overlap
        correlation = np.correlate(h_binary, 1 - h_binary, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_binary.copy()
            
        h = h_binary.copy()
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
