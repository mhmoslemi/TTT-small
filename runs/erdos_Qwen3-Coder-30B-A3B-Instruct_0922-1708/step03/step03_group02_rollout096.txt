import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h0))
        x_target = np.linspace(0, 2, n_points)
        h0 = np.interp(x_target, x_orig, h0)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Initialize population
    population = []
    population.append(h0)
    for i in range(pop_size - 1):
        h = h0 + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Track the best solution
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate fitness
        fitnesses = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitnesses.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitnesses[0] < best_c5:
            best_c5 = sorted_fitnesses[0]
            best_h = sorted_population[0].copy()
        
        # Create new population
        new_population = []
        
        # Elitism: keep top individuals
        elite_size = int(elite_rate * pop_size)
        new_population.extend(sorted_population[:elite_size])
        
        # Generate offspring through crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_size = 5
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitnesses = [sorted_fitnesses[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
            parent1 = sorted_population[winner_index]
            
            if np.random.random() < cross_rate and len(new_population) + 1 < pop_size:
                # Second parent
                tournament_indices = np.random.choice(pop_size, tournament_size)
                tournament_fitnesses = [sorted_fitnesses[i] for i in tournament_indices]
                winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
                parent2 = sorted_population[winner_index]
                
                # Crossover: single point
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                if np.random.random() < mut_rate:
                    mutate_indices = np.random.choice(n_points, int(mut_rate * n_points), replace=False)
                    for i in mutate_indices:
                        child1[i] += np.random.normal(0, 0.1)
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
                
                if np.random.random() < mut_rate:
                    mutate_indices = np.random.choice(n_points, int(mut_rate * n_points), replace=False)
                    for i in mutate_indices:
                        child2[i] += np.random.normal(0, 0.1)
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
                new_population.append(child1)
                if len(new_population) < pop_size:
                    new_population.append(child2)
            else:
                # Mutation only
                child = parent1.copy()
                if np.random.random() < mut_rate:
                    mutate_indices = np.random.choice(n_points, int(mut_rate * n_points), replace=False)
                    for i in mutate_indices:
                        child[i] += np.random.normal(0, 0.1)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        population = new_population[:pop_size]
    
    # Final check
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
    else:
        # Resort to the best from final population
        fitnesses = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitnesses.append(c5)
        best_index = np.argmin(fitnesses)
        best_h = population[best_index]
        final_c5 = fitnesses[best_index]
    
    return (best_h.tolist(), final_c5, n_points)
