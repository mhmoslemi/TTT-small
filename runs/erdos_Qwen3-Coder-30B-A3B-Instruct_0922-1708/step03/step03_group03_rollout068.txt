import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a piecewise constant function with two segments
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 2.0
        h[mid:] = 0.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Gradient estimation via random perturbations
        grad = np.zeros_like(h)
        eps = 1e-4
        
        # Sample a few random perturbations for gradient estimation
        num_perturb = min(20, n_points)
        for _ in range(num_perturb):
            i = np.random.randint(0, n_points)
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update h using gradient with momentum
        lr = 0.01
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally start fresh from a better solution candidate
        if iteration % 300 == 0 and best_c5 < 0.4:
            h = best_h.copy()
            # Add some noise to avoid local minima trapping
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Consolidation optimization with CVXPY
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Evaluate current overlap to determine how to constrain
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        current_max_overlap = np.max(correlation) * dx
        
        # Constraint the maximum overlap as a simple approximation
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.MOSEK, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass
    
    # Final refinement with local search
    # Try piecewise constant functions with different segment configurations
    best_refined = best_h.copy()
    best_refined_c5 = best_c5
    
    for _ in range(50):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Perturb the solution
        h_pert = best_h.copy()
        for _ in range(5):
            i1, i2 = np.random.randint(0, n_points, 2)
            # Randomly adjust two points
            delta = np.random.normal(0, 0.05)
            h_pert[i1] = np.clip(h_pert[i1] + delta, 0, 1)
            h_pert[i2] = np.clip(h_pert[i2] - delta, 0, 1)
        
        # Rescale
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        
        correlation = np.correlate(h_pert, 1 - h_pert, mode='full')
        c5_pert = np.max(correlation) * dx
        
        if c5_pert < best_refined_c5:
            best_refined_c5 = c5_pert
            best_refined = h_pert.copy()
    
    # Ensure we return final best
    correlation = np.correlate(best_refined, 1 - best_refined, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_refined.tolist(), final_c5, n_points)
