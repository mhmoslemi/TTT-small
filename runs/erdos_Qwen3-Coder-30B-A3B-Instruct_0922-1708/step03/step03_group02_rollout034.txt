import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    solve_time = 0
    
    # To track number of iterations for stability
    iteration_count = 0
    
    while time.time() - start_time < budget_s - 2 and iteration_count < 200:
        iteration_count += 1
        
        # Compute C5 with current h
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Construct convex relaxation for optimizing h
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Sum constraint
        constraints = [cp.sum(h_var) == n_points/2]
        
        # Bound constraints
        constraints += [h_var[i] >= 0 for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        
        # Objective: minimize t such that t >= correlation_k for all k
        # We will represent max_k int h(x)(1-h(x+k)) dx <= t explicitly
        
        # Use a different approach:
        # Iterate through all possible lags in zero-padded correlation array
        corr_full = np.correlate(h, 1 - h, mode='full')
        # corr_full length is 2*n_points - 1
        for k in range(2*n_points - 1):
            # We model sum of h[i] * (1 - h[i + k]) over valid indices
            # Start of window process for correlation term
            valid_len = 2*n_points - 1 - k
            offset = k - (n_points - 1)
            if offset < 0:
                start_idx = 0
                end_idx = valid_len + offset
            else:
                start_idx = offset
                end_idx = valid_len
                
            # Create affine expression that approximates 
            # sum_{i=0}^{end_idx-start_idx-1} h[i + start_idx] * (1 - h[i + start_idx + k])
            ci = cp.sum(cp.multiply(h_var[start_idx: start_idx + end_idx], 
                                  1 - h_var[start_idx + k: start_idx + end_idx + k]))
            constraints.append(t >= ci * dx)
        
        # Solve convex program
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h = h_var.value
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h) 
                solve_time += prob.solver_stats.solve_time
            else:
                # If optimization fails, fallback to gradient update with stronger regularization
                h = best_h.copy()
                eps = 1e-4
                grad = np.zeros_like(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5) / eps
                lr = 0.001
                h = h - lr * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            # Fallback if anything goes wrong
            h = best_h.copy()
            
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
