import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    initial_temp = 0.1
    cooling_rate = 0.99
    perturbation_scale = 0.05
    lr = 0.001
    lr_decay = 0.95
    lr_decay_interval = 50
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    temp = initial_temp
    iteration = 0
    no_improvement_count = 0
    
    # Simulated Annealing loop
    while iteration < max_iterations and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        correlation = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Generate neighbor via perturbation
        h_new = h + np.random.normal(0, perturbation_scale, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_new
            
        # Update temperature
        temp *= cooling_rate
        
        # Occasionally reset temperature if stuck
        if no_improvement_count > 100:
            temp = initial_temp
            no_improvement_count = 0
            
        # Occasionally apply gradient descent
        if iteration % 50 == 0:
            # Estimate gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - current_c5) / eps
            
            # Update with gradient
            h = h - lr * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Decay learning rate
            if iteration % lr_decay_interval == 0:
                lr *= lr_decay
    
    # Final refinement with more gradient descent
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        correlation = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / eps
        
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update learning rate
        lr *= lr_decay
    
    # Final evaluation
    final_correlation = correlate(h, 1 - h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
