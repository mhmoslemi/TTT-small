import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a Barker-like initialization
        h = np.zeros(n_points)
        # Make a pattern that has blocks of 0s and 1s to reduce overlap
        split = n_points // 2
        h[:split] = 2.0  # Higher values initially
        h[split:] = 0.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def objective(h_vec):
        # Normalize
        h_norm = h_vec * (n_points / 2) / np.sum(h_vec)
        # Compute overlaps
        correlation = np.correlate(h_norm, 1 - h_norm, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def gradient(h_vec):
        # Simple finite difference gradient
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        base_obj = objective(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            grad[i] = (objective(h_plus) - base_obj) / eps
        return grad
    
    # Constrained optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B for the optimization
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Optimize with L-BFGS-B
        result = minimize(
            objective, h, method='L-BFGS-B', jac=gradient, bounds=bounds, 
            options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-6}
        )
        
        if result.success:
            h = result.x
            # Ensure normalization
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # If optimization failed, try a random perturbation
            h = best_h.copy()
            perturb_idx = np.random.randint(n_points)
            h[perturb_idx] += np.random.normal(0, 0.01)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Occasionally try a different initialization approach
        if iteration % 10 == 0:
            # Perturb slightly for escape from local minima
            h = best_h.copy()
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Random restart
        if iteration % 20 == 0:
            # Reinitialize with a new variant
            h = np.zeros(n_points)
            # Create a more balanced step pattern
            split = n_points // 2
            h[:split] = 1.5  # less extreme values
            h[split:] = 0.5
            # Normalize
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
