import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap constant for given h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute convolution h * (1 - h) with full mode
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Initial best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # TABU search parameters
    tabu_list = set()
    tabu_length = 50
    stagnation_count = 0
    max_stagnation = 100
    step_size = 0.1
    perturb_count = max(1, int(0.2 * n_points))
    
    start_time = time.time()
    
    # Hash a vector for TABU check
    def hash_h(h_vec):
        # Convert to bytes, hash it
        h_bytes = h_vec.tobytes()
        return hashlib.md5(h_bytes).hexdigest()
    
    # Perturb h by modifying some elements
    def perturb_h(current_h, step=0.1, count=10):
        new_h = current_h.copy()
        indices = np.random.choice(n_points, size=count, replace=False)
        for i in indices:
            delta = np.random.uniform(-step, step)
            new_h[i] += delta
        # Clip and normalize
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        return new_h
    
    # Main TABU search loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Generate multiple candidate solutions
        candidates = []
        for _ in range(50):
            # Create candidate by perturbing current solution
            candidate_h = perturb_h(best_h, step_size, perturb_count)
            candidate_c5 = evaluate_erdos_solution(candidate_h)
            candidate_hash = hash_h(candidate_h)
            
            if candidate_hash not in tabu_list:
                candidates.append((candidate_h, candidate_c5))
        
        # Select best candidate that is not TABU
        if candidates:
            candidate_h, candidate_c5 = min(candidates, key=lambda x: x[1])
            
            # Check if we improve
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = candidate_h.copy()
                stagnation_count = 0
                # Add old best to TABU list
                tabu_list.add(hash_h(best_h))
                if len(tabu_list) > tabu_length:
                    tabu_list.pop()  # Remove one item to enforce size limit
            else:
                # Aspiration criterion: occasionally accept worse for escape
                if np.random.random() < 0.1:
                    best_c5 = candidate_c5
                    best_h = candidate_h.copy()
                    stagnation_count = 0
                    tabu_list.add(hash_h(best_h))
                    if len(tabu_list) > tabu_length:
                        tabu_list.pop()
                else:
                    stagnation_count += 1
        else:
            # No new candidates, perturb a bit to continue exploration
            candidate_h = perturb_h(best_h, step_size, perturb_count)
            candidate_c5 = evaluate_erdos_solution(candidate_h)
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = candidate_h.copy()
                stagnation_count = 0
                tabu_list.add(hash_h(best_h))
                if len(tabu_list) > tabu_length:
                    tabu_list.pop()
            else:
                stagnation_count += 1
        
        # Adapt step size when stagnating
        if stagnation_count > max_stagnation:
            step_size *= 1.2
            stagnation_count = 0
            tabu_list.clear()  # Reset tabu list to free memory
            
        # Occasionally rescale to ensure feasibility
        if iteration % 100 == 0:
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
