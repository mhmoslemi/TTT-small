import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    max_iterations = 200
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iterations:
        iter_count += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Setup convex relaxation using McCormick envelopes
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints for h bounds
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Add McCormick envelopes for bilinear terms
        # We model h[i] * (1 - h[j]) via auxiliary variables z_ij
        # z_ij >= h[i] + h[j] - 1
        # z_ij <= h[i]
        # z_ij <= 1 - h[j]
        # z_ij >= 0
        
        # Approximate the contribution to the objective
        # max_k int h(x)(1 - h(x+k)) dx
        # For each lag k, we compute the integral and enforce t >= each such integral
        # This is easier to model as: minimize t subject to t >= sum(h[i]*(1-h[i+k]))
        # but we have to be careful about the indexing
        
        # Instead of modeling all terms precisely in convex form,
        # we do a simplified approach with tight approximation
        
        # Let's solve the dual problem with constraints only on the value t:
        # t >= sum_{k=0}^{n_points-1} h[i] * (1-h[i+k]) for all valid i,k
        # Equivalent to a single max over such sums
        # But we'll just run a simpler derivation: 
        # Find the maximum correlation term directly using the already used method
            
        # Simple improvement: use local search from the current solution
        # Compute gradient with finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        # Estimate gradients for all components at once
        h_plus = h.reshape(-1, 1) + eps * np.eye(n_points)
        h_plus = np.clip(h_plus, 0, 1)
        # Rescale each
        for i in range(n_points):
            h_plus[i] *= (n_points / 2) / np.sum(h_plus[i])
        
        c5_plus = []
        for i in range(n_points):
            corr_plus = np.correlate(h_plus[i], 1 - h_plus[i], mode='full')
            c5_plus.append(np.max(corr_plus) * dx)
        
        c5_plus = np.array(c5_plus)
        c5_orig = c5
        grad = (c5_plus - c5_orig) / eps
        
        # Update with gradient descent
        lr = 0.01
        h_new = h - lr * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        if np.linalg.norm(h - h_new) < 1e-6:
            break
            
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Final check using exact method
    corr = np.correlate(best_h, 1-best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
