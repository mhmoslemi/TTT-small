import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Binary search with QCQP feasibility checks
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a Frank-Zadoff inspired sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        # Frank-Zadoff inspired step function
        h = np.ones(n_points) * 0.5
        # Add some structure for better convergence
        for i in range(n_points):
            if i % 4 == 0:
                h[i] = 0.8
            elif i % 4 == 2:
                h[i] = 0.2
        
    h = h * (n_points / 2) / np.sum(h)
    
    # Binary search bounds
    low, high = 0.38, 0.4
    tolerance = 1e-4
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Phase 1: Binary search with iterative QCQP checks
    while time.time() - start_time < budget_s - 20 and high - low > tolerance:
        c_mid = (low + high) / 2
        # QCQP feasibility check
        # Try to find if there exists an h such that all overlaps <= c_mid/dx
        # For simplicity, we'll run local optimization to see if we can meet the bound
        
        # Create our target overlap
        target = c_mid / dx
        
        # Run a local optimization with trust region to try to reduce the overlap
        def objective(h_vec):
            h = h_vec.reshape(-1)
            # Evaluate overlap with current shift
            c5_vals = []
            for k in range(n_points):
                s = 0
                for i in range(n_points):
                    s += h[i] * (1 - h[(i + k) % n_points])
                c5_vals.append(s)
            return -np.max(c5_vals)  # we minimize -max to maximize max
        
        def constraint_eq(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        def constraint_ineq(h_vec):
            h = h_vec.reshape(-1)
            # Inequality constraint: all overlaps <= target
            constraints = []
            for k in range(n_points):
                s = 0
                for i in range(n_points):
                    s += h[i] * (1 - h[(i + k) % n_points])
                constraints.append(target - s)  # we want target - s >= 0
            return np.array(constraints)
        
        # Constraint bounds and Jacobian
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            # Use SLSQP for local search
            res = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=[
                    {'type': 'eq', 'fun': constraint_eq},
                    {'type': 'ineq', 'fun': lambda x: constraint_ineq(x)},
                ],
                options={'maxiter': 100, 'ftol': 1e-6}
            )
            if res.success:
                # Check if this is better
                final_c5 = -res.fun * dx
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = res.x.copy()
                if final_c5 <= c_mid:
                    high = c_mid
                else:
                    low = c_mid
                h = res.x.copy()
            else:
                low = c_mid  # Not feasible
        except:
            low = c_mid  # Not feasible
            
    # Phase 2: Local refinement around best solution
    # Now use gradient-based method for better solution
    n_points = 100  # Final resolution
    dx = 2.0 / n_points
    
    # Downsample best_h
    if len(best_h) != n_points:
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, best_h)
    else:
        h = best_h.copy()
        
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize arrays
    best_h = h.copy()
    best_c5 = float('inf')
    
    iteration = 0
    max_iter = 10000
    alpha = 0.01
    epsilon = 1e-6
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlaps
        overlap_vals = []
        for k in range(n_points):
            s = 0
            for i in range(n_points):
                s += h[i] * (1 - h[(i + k) % n_points])
            overlap_vals.append(s)
        
        c5 = np.max(overlap_vals) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient for current max overlap
        max_k = np.argmax(overlap_vals)
        grad = np.zeros_like(h)
        for i in range(n_points):
            # derivative wrt h[i] of sum over all k of h[i](1 - h[i+k])
            # derivative wrt h[i] of h[i] * (1 - h[i+k]) where k is the max
            # we are looking at the term h[j] * (1 - h[j + k])
            # for j = i, and j = i - k
            # there are two derivative terms
            grad[i] = (1 - h[(i + max_k) % n_points]) - h[(i - max_k) % n_points]
            
        # Update with gradient descent
        h_new = h - alpha * grad
        
        # Project h_new to [0,1] and rescale to sum to n_points / 2
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive step size
        if iteration % 100 == 0:
            alpha *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)
