import numpy as np
import time
from scipy.interpolate import interp1d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Genetic Algorithm parameters
    pop_size = 50
    generations = 120  # Roughly 600s budget
    crossover_rate = 0.5
    mutation_rate = 0.2
    elite_size = 10
    
    # Population initialization
    population = []
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            # Use the provided initial solution
            individual = h.copy()
        else:
            # Random valid initialization
            individual = np.random.rand(n_points) * 0.5
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # GA loop
    start_time = time.time()
    no_improvement_count = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        generation += 1
        
        # Evaluate fitness of population
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Create next generation
        new_population = []
        
        # Elitism: keep the best
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = sorted_population[np.random.randint(0, pop_size//2)]
            parent2 = sorted_population[np.random.randint(0, pop_size//2)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    # Mutate 5% of points
                    num_mutations = max(1, int(0.05 * n_points))
                    mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
                    for idx in mutation_indices:
                        # Add noise and clip
                        child[idx] += np.random.normal(0, 0.05)
                        child[idx] = np.clip(child[idx], 0, 1)
                    
                    # Rescale to meet constraint
                    child = child * (n_points / 2) / np.sum(child)
                
                new_population.append(child)
            
            if len(new_population) >= pop_size:
                break
        
        # Replace population
        population = new_population[:pop_size]
        
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Ensure bounds are respected
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
