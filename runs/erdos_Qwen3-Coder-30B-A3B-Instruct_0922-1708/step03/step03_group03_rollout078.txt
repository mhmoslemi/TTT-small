import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a custom construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Use a Barker-like initialization (modified for our purposes)
        h = np.zeros(n_points)
        # Create a pattern inspired by Barker codes: mixed 0s and 1s
        # We use 4 segments
        segments = 4
        segment_size = n_points // segments
        for i in range(segments):
            start_idx = i * segment_size
            end_idx = (i + 1) * segment_size if i < segments - 1 else n_points
            if i % 2 == 0:
                h[start_idx:end_idx] = 1.0
            else:
                h[start_idx:end_idx] = 0.0
        # Normalize to satisfy integral constraint
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    def objective_and_gradient(h_vec):
        # Normalize h_vec to satisfy sum constraint
        h_normalized = h_vec * (n_points / 2) / np.sum(h_vec)
        # Compute overlap integral
        correlation = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        max_overlap = np.max(correlation) * dx
        # Gradient approximation (finite differences)
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus_norm = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus_norm, 1 - h_plus_norm, mode='full')
            max_plus = np.max(corr_plus) * dx
            grad[i] = (max_plus - max_overlap) / eps
        return max_overlap, grad
    
    # Optimize using L-BFGS
    # We use a conservative approach to avoid gradient explosion    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # This is a single optimization pass - typically very efficient
        result = minimize(
            lambda x: objective_and_gradient(x)[0],
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=lambda x: objective_and_gradient(x)[1],
            options={'maxiter': 10, 'ftol': 1e-8}
        )
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
        else:
            # Perturb randomly
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)
