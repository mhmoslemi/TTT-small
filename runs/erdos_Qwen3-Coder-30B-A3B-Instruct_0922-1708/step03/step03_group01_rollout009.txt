import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 120
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = normalize(h)
        population.append(h.copy())
    
    while len(population) < pop_size:
        h = np.random.uniform(0, 1, n_points)
        h = normalize(h)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= 100:
            break
            
        # Selection: keep top 40%
        selected_indices = np.argsort(fitness)[:int(pop_size * 0.4)]
        selected_population = [population[i] for i in selected_indices]
        
        # Generate offspring
        new_population = selected_population[:]
        
        while len(new_population) < pop_size:
            parent1, parent2 = random.sample(selected_population, 2)
            
            # Crossover: blend adjacent points
            crossover_point = random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Apply constraints
            child1 = normalize(child1)
            child2 = normalize(child2)
            
            # Mutation: 5% of points with Gaussian noise
            for child in [child1, child2]:
                num_mutate = max(1, int(0.05 * n_points))
                mutate_indices = random.sample(range(n_points), num_mutate)
                for i in mutate_indices:
                    noise = random.gauss(0, 0.1)
                    child[i] += noise
                child = np.clip(child, 0, 1)
                child = normalize(child)
                new_population.append(child)
                
                if len(new_population) >= pop_size:
                    break
        
        population = new_population[:pop_size]
    
    return (best_h.tolist(), best_c5, n_points)
