import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define constraint functions
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    def constraint_bounds(x):
        return np.concatenate([x, 1 - x])  # x >= 0 and x <= 1
    
    # Define the objective function to minimize
    def objective(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Define the gradient function
    def gradient(h_vec):
        # Find the shift that maximizes overlap
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_shift = np.argmax(correlation)
        
        # Gradients for each component
        grad = np.zeros_like(h_vec)
        
        # For each shift k, compute gradient contribution
        for i in range(len(h_vec)):
            # The gradient for h[i] is approximately:
            # dx * (contribution from shift k_max)
            if i + max_shift < len(h_vec):
                grad[i] += -dx  # Since the term is -h[i] in the overlap
            if i - max_shift >= 0:
                grad[i] += -dx
                
        return grad
    
    # Set up constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': lambda x: x},  # x >= 0
        {'type': 'ineq', 'fun': lambda x: 1 - x}  # x <= 1
    ]
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Get current function value
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Use scipy.optimize.minimize with SLSQP
        try:
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if result.success:
                h = result.x
                # Re-normalize to ensure sum constraint
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
            else:
                # Fall back to a basic gradient descent step
                grad = gradient(h)
                h = h - 0.001 * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
        except Exception:
            # Fallback on simple gradient descent
            grad = gradient(h)
            h = h - 0.001 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
