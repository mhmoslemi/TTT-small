import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Discretized shifts for constraint generation
    shift_points = 200
    shifts = np.linspace(-2, 2, shift_points)
    
    # Start with an overestimate of the upper bound
    max_c5 = 0.38085
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Iterative constraint generation approach
    max_iterations = 200
    iteration = 0
    
    while iteration < max_iterations and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Identify the most violated constraint
        # Compute overlap integrals for all shifts
        max_overlap_value = -1
        max_shift_index = -1
        
        for i, k in enumerate(shifts):
            # Compute correlation for shift k
            if k >= 0:
                # h(x) * (1 - h(x + k))
                correlation = np.convolve(h, 1 - np.roll(h, int(k/dx)), mode='valid')
            else:
                # h(x) * (1 - h(x + k))
                correlation = np.convolve(h, 1 - np.roll(h, int(-k/dx)), mode='valid')
            
            # Adjust correlation edges due to convolution
            corr_len = len(correlation)
            if corr_len > 0:
                val = np.max(correlation) * dx
                if val > max_overlap_value:
                    max_overlap_value = val
                    max_shift_index = i
        
        # Check current solution
        if max_overlap_value < best_c5:
            best_c5 = max_overlap_value
            best_h = h.copy()
            
        # Add constraint based on most violated shift
        # Use the constraint of the form: sum over x of h(x) * (1 - h(x + k)) <= C
        # Which can be approximated as: sum_i h[i] * (1 - h[i + k_idx]) <= C * dx
        
        # Compute the constraint matrix now including the most violated
        # This involves creating the full Toeplitz-like constraint matrix
        # rather than just finite differences, to build a careful tight constraint
        
        # For efficiency, using finite-difference-like approach:
        # Maximize overlap under current constraints, and adjust h accordingly
        
        # Simple gradient-free local searches and gradient-based maintenance
        # Now move to optimize the function with CVXPY
        # This is a convex optimization; we minimize C subject to constraints
        
        # To avoid most overheads, do this very carefully:
        h_cvx = cp.Variable(n_points)
        C = cp.Variable()
        
        # Set up constraints
        constraints = [0 <= h_cvx[i] for i in range(n_points)]
        constraints += [h_cvx[i] <= 1 for i in range(n_points)]
        constraints += [cp.sum(h_cvx) == n_points / 2]
        constraints += [C >= 0]
        
        # Add constraints for the current max overlap shift
        # Evaluate shift constraint with finite differences
        k = shifts[max_shift_index]
        # Reverse index for marking the shift
        k_int = int(k / dx)
        
        # Compute overlap from current h
        if k_int >= 0:
            corr = np.convolve(h, 1 - np.roll(h, k_int), mode='valid')
        else:
            corr = np.convolve(h, 1 - np.roll(h, -k_int), mode='valid')
            
        # Raw integration simulation per shift
        if len(corr) >= 1:
            # Approximate as: sum over i, h[i] * (1 - h[i + k_int])
            # Interpreted as: convolve h with (1-instanced h shifted)
            # We can express it as a simple matrix constraint
            if k_int >= 0 and k_int < len(h):
                # Build the constraint h[i] * (1 - h[i + k_int]) <= C*dx
                # Generate valid start indices
                valid_indices = range(len(h) - k_int)
                # Formulate only when valid
                for i in valid_indices:
                    # Add constraint that term is <= C*dx
                    constraints.append(h_cvx[i] * (1 - h_cvx[i + k_int]) <= C * dx)
            else:
                # Out of range, just default ignore or extend
                pass
            
        # Objective: minimize C
        objective = cp.Minimize(C)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_cvx.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback optimization
                h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback to random perturb and uniform rescaling
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Occasionally, refactor solution with a bit of new randomness
        if iteration % 50 == 0:
            h = np.clip(h + np.random.normal(0, 0.0005, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
