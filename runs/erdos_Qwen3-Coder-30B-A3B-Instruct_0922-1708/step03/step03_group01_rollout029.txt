import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    def compute_c5(h_vec):
        # Compute correlation using fft for efficiency
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def compute_gradient(h_vec):
        # Find the index k that maximizes correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        k_max = np.argmax(corr)
        # Gradient for each element
        grad = np.zeros_like(h_vec)
        # For the specific k that gives max correlation
        # grad[j] = dx * ( (1 - h[j + k_max]) - h[j - k_max] )
        # Handle wraparound using modulo
        n = len(h_vec)
        for j in range(n):
            if k_max <= j:
                diff1 = 0
                diff2 = h_vec[j - k_max]
            else:
                diff1 = 1 - h_vec[j + n - k_max]
                diff2 = 0
            grad[j] = dx * (diff1 - diff2)
        return grad
    
    # Optimize using scipy
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint: sum(h) = n_points / 2
    def constraint_fn(h_vec):
        return np.sum(h_vec) - n_points / 2
    cons = {'type': 'eq', 'fun': constraint_fn}
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute current value
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # For more precise gradient, compute over a small perturbation
        def obj(h_vec):
            return compute_c5(h_vec)
        # Use minimize with SLSQP
        result = minimize(
            obj, h, method='SLSQP', bounds=bounds, constraints=cons, 
            options={'maxiter': 20, 'ftol': 1e-8, 'gtol': 1e-8},
            callback=lambda x: None  # we already eval everywhere
        )
        
        if result.success:
            # Accept the new solution
            h = result.x
            
            # Rescale to exact sum
            h = h * (n_points / 2) / np.sum(h)
            
            # Clip to bounds
            h = np.clip(h, 0, 1)
        else:
            # If optimization fails, just take a small gradient step
            grad = compute_gradient(h)
            # Small step
            step_size = 1e-4
            h = h - step_size * grad
            # Project to feasible set
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
