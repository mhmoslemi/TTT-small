import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Genetic algorithm parameters
    pop_size = 50
    generations = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            individual = h.copy()
        else:
            individual = np.random.rand(n_points)
            individual = np.clip(individual, 0, 1)
            individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        generations += 1
        
        # Evaluate fitness
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Selection: Keep top 40%
        selected_indices = np.argsort(fitness)[:int(0.4 * pop_size)]
        selected_population = [population[i] for i in selected_indices]
        
        # Create new population through crossover and mutation
        new_population = selected_population.copy()
        
        # Crossover: Blend parents
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected_population, 2, replace=False)
            # Blend: take average of consecutive segments
            crossover_point = np.random.randint(1, n_points-1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Rescale to maintain constraint
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
            
        # Mutation: Perturb 5% of points in 20% of individuals
        for i in range(int(0.2 * pop_size)):
            idx = np.random.randint(0, len(new_population))
            individual = new_population[idx].copy()
            
            # Perturb 5% of points
            num_to_mutate = max(1, int(0.05 * n_points))
            mutation_indices = np.random.choice(n_points, num_to_mutate, replace=False)
            
            for j in mutation_indices:
                # Add Gaussian noise
                noise = np.random.normal(0, 0.05)
                individual[j] += noise
                individual[j] = np.clip(individual[j], 0, 1)
                
            # Rescale
            individual = individual * (n_points / 2) / np.sum(individual)
            new_population[idx] = individual
            
        population = new_population[:pop_size]
        
    # Final evaluation of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    return (best_h.tolist(), final_c5, n_points)
