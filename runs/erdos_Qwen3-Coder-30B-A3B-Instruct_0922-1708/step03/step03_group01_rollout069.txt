import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search bounds for C5
    low = 0.38
    high = 0.4
    
    while high - low > 1e-4 and time.time() - start_time < budget_s - 1:
        c_mid = (low + high) / 2.0
        
        # Set up QCQP
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0, 
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints for all shifts
        for k in range(n_points):
            # For shift k: sum h[i] * (1 - h[i+k])
            # We need to handle periodic boundary conditions
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_var[i] * (1 - h_var[j])
            constraints.append(expr <= c_mid / dx)
        
        # Objective: minimize C5 (encoded in the constraints)
        # We're solving feasibility, so we just check if the system is solvable
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                # Found a feasible solution
                h_sol = h_var.value
                # Check if this solution is valid
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    # Compute actual C5
                    correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                    computed_c5 = np.max(correlation) * dx
                    if computed_c5 < best_c5:
                        best_c5 = computed_c5
                        best_h = h_sol.copy()
                    high = c_mid
                else:
                    low = c_mid
            else:
                low = c_mid
        except:
            low = c_mid
    
    # Final verification
    if best_c5 == float('inf'):
        # Fall back to the initial solution
        correlation = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(correlation) * dx
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
