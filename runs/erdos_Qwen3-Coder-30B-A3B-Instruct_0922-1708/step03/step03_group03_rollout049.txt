import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # Uniform distribution, scaled
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    temperature = 0.1
    success_count = 0
    total_count = 0
    
    # Perturbation scales
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Track statistics for adaptation
    last_50_success = []
    
    for resolver in resolvers:
        if time.time() - start_time > budget_s - 1:
            break
            
        num_iterations = 1000
        for i in range(num_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate perturbation
            perturb = np.random.normal(0, resolver, n_points)
            h_candidate = h + perturb
            
            # Clip and normalize
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute overlap
            correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(correlation) * dx
            
            # Accept or reject with simulated annealing
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                success_count += 1
                last_50_success.append(1)
            else:
                # Accept with probability based on temperature
                delta = c5_candidate - best_c5
                if np.random.random() < np.exp(-delta / temperature):
                    h = h_candidate.copy()
                    best_c5 = c5_candidate
                    success_count += 1
                    last_50_success.append(1)
                else:
                    last_50_success.append(0)
                    
            # Update temperature
            if i % 100 == 0 and i > 0:
                temperature *= 0.99
                temperature = max(1e-6, temperature)
                
            # Adaptive perturbation scaling
            if i % 50 == 0 and i > 0:
                if len(last_50_success) == 50:
                    success_ratio = sum(last_50_success) / 50.0
                    if success_ratio > 0.2:
                        resolver *= 1.1
                    elif success_ratio < 0.1:
                        resolver *= 0.9
                    resolver = np.clip(resolver, 0.001, 1.0)
                    last_50_success = []
    
    # Final computation of overlap to ensure accuracy
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Final validation
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - n_points / 2) < 1e-6
    
    return (best_h.tolist(), final_c5, n_points)
