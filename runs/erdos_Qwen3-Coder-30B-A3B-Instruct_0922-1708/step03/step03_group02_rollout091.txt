import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # For the joint optimization approach
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Annealing parameters
    T = 0.1
    alpha = 0.995
    steps = 0
    max_annealing_steps = 2000
    
    # Nelder-Mead parameters
    simplex_size = 5
    
    # Simulated Annealing loop
    while steps < max_annealing_steps and time.time() - start_time < budget_s - 1:
        # Slightly perturb the solution
        h_new = h.copy()
        perturbed_indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
        for i in perturbed_indices:
            h_new[i] += np.random.normal(0, 0.1)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept or reject
        if c5_new < best_c5 or np.random.random() < np.exp(-(c5_new - best_c5) / T):
            h = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
        # Decrease temperature
        T *= alpha
        steps += 1

    # Nelder-Mead refinement
    # Initialize simplex around current best solution
    simplex = [best_h.copy()]
    for i in range(simplex_size - 1):
        h_pert = best_h.copy()
        for j in range(n_points // 10):
            k = np.random.randint(0, n_points)
            h_pert[k] += np.random.normal(0, 0.05)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        simplex.append(h_pert)

    while time.time() - start_time < budget_s - 1:
        # Evaluate simplex
        values = []
        for h_simplex in simplex:
            correlation = np.correlate(h_simplex, 1 - h_simplex, mode='full')
            c5_val = np.max(correlation) * dx
            values.append(c5_val)
            
        # Sort by value
        indices = np.argsort(values)
        simplex = [simplex[i] for i in indices]
        values = [values[i] for i in indices]
        
        # Compute centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        reflected = centroid + (centroid - simplex[-1]) * 1.0
        reflected = np.clip(reflected, 0, 1)
        reflected = reflected * (n_points / 2) / np.sum(reflected)
        
        correlation = np.correlate(reflected, 1 - reflected, mode='full')
        c5_reflected = np.max(correlation) * dx
        
        if values[0] <= c5_reflected < values[-2]:
            simplex[-1] = reflected
            continue
        elif c5_reflected < values[0]:
            # Expansion
            expanded = centroid + (reflected - centroid) * 2.0
            expanded = np.clip(expanded, 0, 1)
            expanded = expanded * (n_points / 2) / np.sum(expanded)
            correlation = np.correlate(expanded, 1 - expanded, mode='full')
            c5_expanded = np.max(correlation) * dx
            if c5_expanded < c5_reflected:
                simplex[-1] = expanded
            else:
                simplex[-1] = reflected
        else:
            # Contraction
            if c5_reflected < values[-1]:
                # Outside contraction
                contracted = centroid + (reflected - centroid) * 0.5
            else:
                # Inside contraction
                contracted = centroid + (simplex[-1] - centroid) * 0.5
            contracted = np.clip(contracted, 0, 1)
            contracted = contracted * (n_points / 2) / np.sum(contracted)
            correlation = np.correlate(contracted, 1 - contracted, mode='full')
            c5_contracted = np.max(correlation) * dx
            
            if c5_contracted < values[-1]:
                simplex[-1] = contracted
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + (simplex[i] - simplex[0]) * 0.5

        # Update best if we improved
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
