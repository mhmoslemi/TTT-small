import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init.copy())
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
    
    # Generate other particles with random perturbations
    for i in range(n_particles - 1):
        h = h_init + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    
    # Initialize velocities
    for i in range(n_particles):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize best positions
    particles_best = [p.copy() for p in particles]
    particles_best_c5 = []
    for p in particles:
        corr = np.correlate(p, 1 - p, mode='full')
        c5 = np.max(corr) * dx
        particles_best_c5.append(c5)
    
    # Global best
    global_best_idx = np.argmin(particles_best_c5)
    global_best_h = particles_best[global_best_idx].copy()
    global_best_c5 = particles_best_c5[global_best_idx]
    
    # PSO parameters
    w = 1.0  # inertia weight
    c1 = c2 = 2.0  # acceleration coefficients
    
    start_time = time.time()
    step = 0
    reset_frequency = max(1, int(0.1 * budget_s * 50 / 600))  # adjust based on budget
    
    while time.time() - start_time < budget_s - 1:
        # Update each particle
        for i in range(n_particles):
            # Apply velocity
            velocities[i] = w * velocities[i] + \
                            c1 * np.random.random(n_points) * (particles_best[i] - particles[i]) + \
                            c2 * np.random.random(n_points) * (global_best_h - particles[i])
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Clip and rescale
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate C5
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(corr) * dx
            
            # Update particle best
            if c5 < particles_best_c5[i]:
                particles_best[i] = particles[i].copy()
                particles_best_c5[i] = c5
                
                # Update global best
                if c5 < global_best_c5:
                    global_best_h = particles[i].copy()
                    global_best_c5 = c5
        
        # Reset some particles randomly to maintain diversity
        if step % reset_frequency == 0 and step > 0:
            for i in range(n_particles // 10):
                idx = np.random.randint(0, n_particles)
                # Resample with some perturbation
                h = global_best_h + np.random.normal(0, 0.02, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                particles[idx] = h.copy()
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
        
        step += 1
    
    # Final check
    final_corr = np.correlate(global_best_h, 1 - global_best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (global_best_h.tolist(), final_c5, n_points)
