import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize swarm
    num_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Create initial particles
    particles = []
    velocities = []
    
    # Use initial construction for some particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        for i in range(num_particles // 5):
            particles.append(h_init.copy())
            velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Fill remaining particles randomly
    for i in range(num_particles - len(particles)):
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # Track personal best and global best
    personal_best = particles.copy()
    personal_best_scores = np.full(num_particles, np.inf)
    global_best_idx = 0
    global_best_score = np.inf
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(num_particles):
            h = particles[i]
            
            # Compute overlap integral
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_best[i] = h.copy()
                
            # Update global best
            if c5 < global_best_score:
                global_best_score = c5
                global_best_idx = i
                
        # Update velocities and positions
        r1, r2 = np.random.rand(2)
        for i in range(num_particles):
            v = w * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (personal_best[global_best_idx] - particles[i])
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            new_pos = particles[i] + v
            
            # Project back to feasible region
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * (n_points / 2) / np.sum(new_pos)
            
            particles[i] = new_pos
    
    # Return best solution
    best_h = personal_best[global_best_idx]
    return (best_h.tolist(), global_best_score, n_points)
