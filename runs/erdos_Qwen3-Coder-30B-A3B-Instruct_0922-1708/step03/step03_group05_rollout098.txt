def run(seed=42, budget_s=600, **kwargs):

        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')

    iteration = 0

    alpha = 0.99

    max_iter = 1000

    # Coarse optimization phase
    while time.time() - start_time < budget_s - 5 and iteration < max_iter:

        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Generate neighbor: perturb a few points
        h_new = h.copy()
        
        # Select 5% of the indices to perturb
        indices_to_perturb = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)

        for i in indices_to_perturb:
            h_new[i] += np.random.uniform(-0.05, 0.05)

        h_new = np.clip(h_new, 0, 1)

        h_new = h_new * (n_points / 2) / np.sum(h_new)

            T_refined *= alpha
            
        best_h = h_refined
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
