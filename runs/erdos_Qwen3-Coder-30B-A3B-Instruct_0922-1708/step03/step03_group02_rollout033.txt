import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    max_iter = 100
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Solve convex relaxation with McCormick envelopes
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # McCormick envelopes for z_ij = h[i] * (1 - h[j])
        constraints = [
            cp.norm(h_var, 1) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            t >= 0
        ]
        
        # Add McCormick constraints for bilinear terms
        # For each lag k, compute sum_{i=0}^{n_points-1-k} h[i] * (1 - h[i+k])
        for k in range(n_points):
            if k == 0:
                # For k = 0, it's just sum of h[i] * (1 - h[i]) 
                # which we can bound more easily
                expr = cp.sum(cp.multiply(h_var, 1 - h_var))
                constraints.append(t >= dx * expr)
            else:
                # For k > 0, compute sum of h[i] * (1 - h[i+k])
                if k < n_points:
                    expr = cp.sum(cp.multiply(h_var[:-k], 1 - h_var[k:]))
                    constraints.append(t >= dx * expr)
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iter=1000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = h_var.value
                if h_sol is not None and np.all(h_sol >= 0) and np.all(h_sol <= 1):
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                    c5 = np.max(correlation) * dx
                    
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_sol.copy()
            else:
                # Optimization failed, use current h
                pass
        except:
            pass
        
        # Update h using gradient descent on the full expression
        grad = np.zeros_like(h)
        eps = 1e-5
        h_plus = h.copy()
        for i in range(n_points):
            h_plus[i] = min(1, max(0, h[i] + eps))
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            correlation_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(correlation_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
            h_plus[i] = h[i]  # restore original
            
        # Update with gradient descent
        lr = 0.01
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        iter_count += 1
        
        # Periodic best update check
        if iter_count % 10 == 0:
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final best check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
