import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Evaluate the Erdős minimum overlap constant C₅ for given h."""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    # TABU search parameters
    tabu_list_size = 50
    tabu_list = set()
    tabu_threshold = 100
    stagnation_count = 0
    
    step_size = 0.1
    perturbation_count = max(1, int(0.2 * n_points))
    
    current_h = h.copy()
    current_c5 = best_c5
    
    def hash_h(h_vec):
        # Create hashable representation
        return hashlib.md5(h_vec.astype(np.float32).tobytes()).hexdigest()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Generate candidates
        candidates = []
        for k in range(50):
            h_new = current_h.copy()
            
            # Select random indices to perturb
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            
            # Apply perturbations
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                h_new[i] += delta
            
            # Clip and normalize
            h_new = np.clip(h_new, 0, 1)
            scale_factor = (n_points / 2) / np.sum(h_new)
            h_new = h_new * scale_factor
            
            # Check if in TABU list
            h_hash = hash_h(h_new)
            if h_hash not in tabu_list:
                c5_new = evaluate_erdos_solution(h_new)
                candidates.append((h_new.copy(), c5_new))
        
        # Select best candidate
        if candidates:
            best_candidate_h, best_candidate_c5 = min(candidates, key=lambda x: x[1])
        else:
            best_candidate_h = current_h.copy()
            best_candidate_c5 = current_c5
            
        # Accept best candidate
        if best_candidate_c5 < best_c5:
            best_h = best_candidate_h.copy()
            best_c5 = best_candidate_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Update current
        current_h = best_candidate_h.copy()
        current_c5 = best_candidate_c5
        
        # Add to TABU list
        current_hash = hash_h(current_h)
        tabu_list.add(current_hash)
        if len(tabu_list) > tabu_list_size:
            # Remove oldest entry
            tabu_list.pop()
            
        # Adapt step size if stagnating
        if stagnation_count > tabu_threshold:
            step_size *= 1.2
            stagnation_count = 0
            tabu_list.clear()  # Reset TABU list to avoid getting stuck
            
        # Occasionally rescale to maintain precision
        if iteration % 100 == 0:
            scale_factor = (n_points / 2) / np.sum(current_h)
            current_h = current_h * scale_factor
            
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
