import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Perturbation scales
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Temperature schedule
    temperature = 0.1
    temp_decay = 0.99
    temp_min = 1e-6
    
    # Success tracking for adaptive perturbation
    success_count = 0
    total_count = 0
    success_threshold_low = 0.1
    success_threshold_high = 0.2
    
    # Iterations per resolver
    iterations_per_resolver = 1000
    
    for resolver in resolvers:
        for iteration in range(iterations_per_resolver):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate candidate by perturbing h
            perturbation = np.random.normal(0, resolver, n_points)
            h_candidate = h + perturbation
            
            # Clip to valid range
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Rescale to correct sum
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute overlap (max of all shifts)
            correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(correlation) * dx
            
            # Accept or reject based on improvement or simulated annealing
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                success_count += 1
                total_count += 1
                
                # Adaptively increase resolution (reduce perturbation) if success
                if success_count > 10 and success_count / max(total_count, 1) > success_threshold_high:
                    resolver *= 0.95
                    success_count = 0
                    total_count = 0
            else:
                # Accept with probability based on temperature
                delta = c5_candidate - best_c5
                if delta > 0:
                    prob = np.exp(-delta / temperature)
                    if np.random.rand() < prob:
                        h = h_candidate.copy()
                        best_c5 = c5_candidate
                        success_count += 1
                total_count += 1
            
            # Update temperature
            temperature = max(temperature * temp_decay, temp_min)
            
            # Adaptive adjustment of resolver
            if iteration % 100 == 0 and total_count > 0:
                success_ratio = success_count / max(total_count, 1)
                if success_ratio > success_threshold_high:
                    resolver *= 1.5
                elif success_ratio < success_threshold_low:
                    resolver *= 0.5
                resolver = np.clip(resolver, 0.001, 1.0)
                success_count = 0
                total_count = 0
    
    # Final refinement with small perturbations around best solution
    for _ in range(200):
        if time.time() - start_time > budget_s - 1:
            break
            
        perturbation = np.random.normal(0, 0.001, n_points)
        h_candidate = best_h + perturbation
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(correlation) * dx
        
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
            
    # Final computation of C5
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Ensure feasibility
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) * dx - 1.0) < 1e-6
    
    return (best_h.tolist(), final_c5, n_points)
