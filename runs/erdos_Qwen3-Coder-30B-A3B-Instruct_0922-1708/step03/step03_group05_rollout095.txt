import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    max_iterations = 1000
    T0 = 0.1
    alpha = 0.99
    perturb_fraction = 0.05
    perturb_magnitude = 0.1
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Temperature schedule
    T = T0
    start_time = time.time()
    
    # Precompute correlation indices
    lags = list(range(1, n_points))
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        correlation = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Generate neighbor
        h_new = h.copy()
        indices = np.random.choice(n_points, int(n_points * perturb_fraction), replace=False)
        
        for i in indices:
            # Perturb
            perturb = np.random.uniform(-perturb_magnitude, perturb_magnitude)
            h_new[i] += perturb
            # Clamp
            h_new[i] = np.clip(h_new[i], 0, 1)
            
        # Rescale to meet constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject based on SA
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            h = h_new
            
        # Update temperature
        T *= alpha
        
    # Final verification
    correlation_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
