import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Helper to compute overlap
    def compute_c5(h_vec):
        # Compute overlaps for all shifts
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Only consider the valid overlaps (non-circular)
        valid_correlation = correlation[len(h_vec)-1:]
        return np.max(valid_correlation) * dx
    
    # Optimization parameters
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    num_iterations = 1000
    temperature = 0.1
    success_count = 0
    success_window = 50
    success_history = []
    
    start_time = time.time()
    
    for resolver in resolvers:
        if time.time() - start_time > budget_s - 1:
            break
            
        current_scale = resolver
        for i in range(num_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate candidate by perturbing
            h_candidate = h + np.random.normal(0, current_scale, n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            c5_candidate = compute_c5(h_candidate)
            
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                success_count += 1
                success_history.append(1)
                # Reduce scale slightly on improvement
                current_scale = max(0.001, current_scale * 0.9)
            else:
                # Accept with probability based on temperature
                prob = np.exp(-(c5_candidate - best_c5) / max(temperature, 1e-10))
                if np.random.random() < prob:
                    h = h_candidate.copy()
                    success_count += 1
                    success_history.append(1)
                else:
                    success_history.append(0)
            
            # Update temperature
            temperature *= 0.99
            if len(success_history) > success_window:
                success_history.pop(0)
            
            # Adapt scale based on recent success
            if len(success_history) == success_window:
                success_ratio = sum(success_history) / success_window
                if success_ratio > 0.2:
                    current_scale *= 1.5
                elif success_ratio < 0.1:
                    current_scale *= 0.5
                current_scale = np.clip(current_scale, 0.001, 1.0)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
