import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Initialize population
    population_size = 50
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h_candidate = h + np.random.normal(0, 0.01, n_points)
        else:
            h_candidate = np.random.uniform(0, 1, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        population.append(h_candidate)
    
    # If initial provided, add as first candidate
    if 'initial_h_values' in globals():
        population[0] = h.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    stagnant = 0
    stagnant_limit = 100
    
    while time.time() - start_time < budget_s - 1 and stagnant < stagnant_limit:
        generation += 1
        # Evaluate all individuals in population
        fitness = []
        for individual in population:
            corr = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
        # Update global best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            stagnant = 0
        else:
            stagnant += 1
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        elite_size = population_size // 5
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Tournament selection with replacement
        new_population = elite.copy()
        while len(new_population) < population_size:
            # Tournament
            tournament_size = population_size // 10
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent = population[winner_idx]
            
            # Crossover with another parent
            if np.random.rand() < 0.8:  # 80% crossover rate
                parent2_idx = np.random.choice(population_size)
                parent2 = population[parent2_idx]
                
                # Blend crossover
                alpha = np.random.rand()
                child = alpha * parent + (1 - alpha) * parent2
                # Ensure feasibility
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            else:
                child = parent.copy()
                
            # Mutation
            if np.random.rand() < 0.1:  # 10% mutation rate
                mutation_magnitude = 0.02
                for i in range(n_points):
                    if np.random.rand() < 0.1:  # 10% of genes mutated
                        child[i] += np.random.normal(0, mutation_magnitude)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
            new_population.append(child)
            
        population = new_population[:population_size]
        
    # Final refinement with gradient descent
    h_refined = best_h.copy()
    for _ in range(100):
        correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
        c5 = np.max(correlation) * dx
        
        grad = np.zeros_like(h_refined)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        lr = 0.001
        h_refined = h_refined - lr * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
    # Final evaluation
    correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
