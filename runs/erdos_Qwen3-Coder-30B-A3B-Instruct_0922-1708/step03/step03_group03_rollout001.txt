import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a custom construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a modified Barker-like construction
        h = np.zeros(n_points)
        # Create a pattern similar to Barker codes
        # We'll try to use a simple alternating pattern that balances
        # concentration and spread
        pattern_length = 20
        pattern = np.array([1.0] * (pattern_length // 2) + [0.0] * (pattern_length // 2))
        for i in range(n_points // pattern_length + 1):
            start_idx = i * pattern_length
            end_idx = min(start_idx + len(pattern), n_points)
            if end_idx > start_idx:
                h[start_idx:end_idx] = pattern[:end_idx - start_idx]
        
        # Normalize to integral constraint
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def objective_and_gradient(h_vec):
        # Ensure h is within bounds
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        
        # Compute overlaps
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(correlation) * dx
        
        # Compute gradient (approximate)
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - max_overlap) / eps
            
        return max_overlap, grad
    
    # Optimization with restarts
    restarts = 5
    for restart in range(restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use a fresh randomized initialization for this restart
        h_init = h.copy()
        # Add small noise
        h_init += np.random.normal(0, 0.01, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Optimize
        try:
            # Use L-BFGS-B with bounds
            bounds = [(0, 1) for _ in range(n_points)]
            result = minimize(
                lambda x: objective_and_gradient(x)[0],
                h_init,
                method='L-BFGS-B',
                bounds=bounds,
                jac=lambda x: objective_and_gradient(x)[1],
                options={'maxiter': 1000, 'ftol': 1e-9}
            )
            
            if result.success:
                h_opt = np.clip(result.x, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                corr = np.correlate(h_opt, 1 - h_opt, mode='full')
                c5_opt = np.max(corr) * dx
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
        
        # Early exit if we're close to target
        if best_c5 < 0.3807:
            break
                    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Ensure we return valid values
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
