import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    alpha = 0.01
    tol = 1e-6
    
    # Initialize using the provided parent or Frank-Zadoff-like sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Generate sequence that minimizes correlation (Frank-Zadoff-like)
        h = np.zeros(n_points)
        # Fill with a simple pattern that avoids high correlations
        for i in range(n_points):
            h[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        # Compute C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient
        grad = np.zeros_like(h)
        max_shift_idx = np.argmax(correlation)
        # Get the actual shift
        if max_shift_idx >= len(h):
            actual_shift = max_shift_idx - len(h) + 1
        else:
            actual_shift = max_shift_idx
            
        # Compute gradient for max overlap
        for i in range(len(h)):
            # Derivative of overlap integral w.r.t h[i] for the max shift
            j = (i + actual_shift) % len(h)
            grad[i] = dx * ((1 - h[j]) - h[i])
            
        # Update h
        h_new = h - alpha * grad
        
        # Project h to [0, 1] and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update h
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            alpha *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)
