import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Compute full correlation matrix for gradients
    def compute_correlation_matrix(h_array):
        # This computes correlations for all shifts
        corr_full = np.correlate(h_array, 1 - h_array, mode='full')
        return corr_full
    
    # Helper to compute gradient for a specific shift
    def compute_gradient_for_shift(h_array, shift_idx):
        n = len(h_array)
        grad = np.zeros(n)
        # For a shift k, the derivative of correlation at k with respect to h[i] 
        # is (1 - h[(i+k)%n]) for the term h[i] * (1-h[i+k]) in the convolution
        # But we have to be more careful about how we index.
        # The effective correlation is corr[k] = sum_{i} h[i] * (1 - h[i-k]) 
        # and we want grad_i = d/dh_i of this sum
        # If we shift by @k, then we're looking at h[i] * (1 - h[i-k]) for all valid i.
        # So, grad[i] = 1 - h[i-k] for k = shift_idx
        shift = shift_idx - (n - 1)  # because correlate provides shifts from -(n-1) to (n-1)
        for i in range(n):
            k = (i - shift) % n  
            if (i - shift) >= 0 and (i - shift) < n:
                grad[i] = 1 - h_array[k]
            # Otherwise it's 0 due to periodicity, i.e. no contribution
        return grad
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute correlation matrix to find worst shift
        full_corr = compute_correlation_matrix(h)
        # The shift index for the worst overlap (0 to 2*n_points - 2)
        max_shift_idx = np.argmax(full_corr)
        # Map back to shift k = max_shift_idx - (n_points - 1)
        k_max = max_shift_idx - (n_points - 1)
        if k_max > n_points - 1:
            k_max = k_max - 2*n_points  # normalize to wrap around

        # Compute exact gradient w.r.t the worst shift
        grad = compute_gradient_for_shift(h, max_shift_idx)
        
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart
        if iteration % 200 == 0:
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                if len(h0) != n_points:
                    x_orig = np.linspace(0, 2, len(h0))
                    x_new = np.linspace(0, 2, n_points)
                    h0 = np.interp(x_new, x_orig, h0)
                h0 = np.clip(h0, 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
                h = h0
            else:
                h = np.ones(n_points) * 0.5
                
    return (best_h.tolist(), best_c5, n_points)
