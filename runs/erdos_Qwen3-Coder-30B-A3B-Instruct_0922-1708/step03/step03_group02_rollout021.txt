import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_size = int(pop_size * 0.2)
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Create initial population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        population.append(h_base.copy())
    else:
        h_base = np.ones(n_points) * 0.5
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        population.append(h_base.copy())
    
    # Add diverse individuals
    for i in range(pop_size - 1):
        h = h_base + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Generate offspring through crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            def tournament_select():
                idxs = np.random.choice(pop_size, tournament_size)
                best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
                return population[best_idx].copy()
            
            # Crossover
            if np.random.random() < cross_rate:
                parent1 = tournament_select()
                parent2 = tournament_select()
                
                # Uniform crossover
                mask = np.random.random(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Select one parent
                parent = tournament_select()
                child = parent.copy()
            
            # Mutation
            if np.random.random() < mut_rate:
                # Add Gaussian noise
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                # Rescale to maintain constraint
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        # Trim to exact population size
        population = new_population[:pop_size]
        generation += 1
    
    # Final evaluation
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
    else:
        # Fallback evaluation
        best_h = population[0]
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
