import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill population with random feasible individuals
    for _ in range(pop_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h.copy())
    
    def evaluate_individual(h):
        # Compute correlation and get max overlap
        correlation = correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.random(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        # Rescale to maintain constraint
        child = child * (n_points / 2) / np.sum(child)
        child = np.clip(child, 0, 1)
        return child
    
    def mutate(h, mutation_rate=0.05, noise_std=0.1):
        # Create a copy
        h_mut = h.copy()
        # Determine indices to mutate
        n_mutate = max(1, int(n_points * mutation_rate))
        indices = np.random.choice(n_points, n_mutate, replace=False)
        # Add noise
        noise = np.random.normal(0, noise_std, n_mutate)
        h_mut[indices] += noise
        # Clip and rescale
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        return h_mut
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    gen = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        gen += 1
        
        # Evaluate the entire population
        fitness = [evaluate_individual(h) for h in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination
        if no_improvement_count >= 100:
            break
            
        # Selection: keep top 40% (elitism)
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:int(0.4 * pop_size)]
        elite = [population[i] for i in top_indices]
        
        # Generate new population
        new_population = elite.copy()
        
        # Add offspring through crossover
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(elite, 2, replace=False)
            child = crossover(parent1, parent2)
            new_population.append(child)
            
        # Mutate some individuals
        for i in range(len(new_population)):
            if np.random.random() < 0.2:  # 20% mutation rate
                new_population[i] = mutate(new_population[i])
                
        population = new_population[:pop_size]
        
    return (best_h.tolist(), best_c5, n_points)
