import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Genetic algorithm parameters
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(h.copy())
    
    for _ in range(pop_size - 1):
        h_mut = h + np.random.normal(0, 0.1, n_points)
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        population.append(h_mut)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    # Track improvement
    last_improvement = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        # Evaluate population
        fitness = []
        for ind in population:
            correlation = np.correlate(ind, 1 - ind, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
        # Track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            last_improvement = generation
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Selection with tournament
        new_population = []
        
        # Elitism
        elite_count = int(elite_rate * pop_size)
        elite_indices = np.argsort(fitness)[:elite_count]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = tournament_selection(population, fitness, tournament_size)
            parent2_idx = tournament_selection(population, fitness, tournament_size)
            
            # Crossover
            if random.random() < cross_rate:
                child1, child2 = crossover(population[parent1_idx], population[parent2_idx])
            else:
                child1, child2 = population[parent1_idx].copy(), population[parent2_idx].copy()
            
            # Mutation
            if random.random() < mut_rate:
                child1 = mutate(child1, 0.1)
            if random.random() < mut_rate:
                child2 = mutate(child2, 0.1)
                
            new_population.extend([child1, child2])
            
        # Trim to exact size
        population = new_population[:pop_size]
        generation += 1
        
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)

def tournament_selection(population, fitness, k):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return best_idx

def crossover(parent1, parent2):
    """Single-point crossover."""
    point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(individual, std_dev):
    """Add Gaussian noise and maintain constraints."""
    mutated = individual + np.random.normal(0, std_dev, len(individual))
    mutated = np.clip(mutated, 0, 1)
    # Rescale to maintain sum constraint
    mutated = mutated * (len(individual) / 2) / np.sum(mutated)
    return mutated
