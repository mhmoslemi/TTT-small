    strategies = []
    
    # Strategy 1: Smoothed initial
    h_smooth = np.ones(n_points) * 0.5

    h_smooth = h_smooth / np.sum(h_smooth) * (n_points / 2)
    strategies.append(h_smooth)

    h_piece = np.ones(n_points) * 0.25
    h_piece[30:70] = 0.75
    h_piece = h_piece / np.sum(h_piece) * (n_points / 2)
    strategies.append(h_piece)

    x_vals = np.linspace(0, 2, n_points)

    strategies.append(h_gauss)

    h_refined = h.copy()

            c5_ref = np.max(corr_ref) * dx

            best_h = h.copy()
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
