import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def dual_optimization_phase(initial_h, n_points, budget_s, dx):
    """Dual optimization to minimize the maximum overlap"""
    start_time = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Initialize dual variables
    lambda_k = np.ones(2 * n_points - 1)
    
    # Parameters
    learning_rate = 0.01
    max_iter = 2000
    patience = 0
    max_patience = 1000
    
    for iter_num in range(max_iter):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Update h by solving weighted least squares problem
        # Construct system matrix
        # For simplicity, we'll directly optimize using projected gradient
        h_prev = h.copy()
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update h
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        # Update best solution
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
            
        # Slowly decrease learning rate
        if iter_num % 500 == 0 and iter_num > 0:
            learning_rate *= 0.5

    return best_h, best_c5
