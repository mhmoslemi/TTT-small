import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_symmetric_h(n_points):
    """Initialize symmetric h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():

            # Make symmetric
            k = n_points // 2
            for i in range(k):
                h[i] = h[2*k - i] = (h[i] + h[2*k - i]) / 2
            return project_h(h, n_points)
    except:
        pass
    # Default symmetric initialization: two bumps
    x = np.linspace(0, 2, n_points)

    h = project_h(h, n_points)
    # Ensure symmetry
    k = n_points // 2
    for i in range(k):
        h[i] = h[2*k - i] = (h[i] + h[2*k - i]) / 2

def compute_exact_gradient(h, max_shift):
    """Compute exact gradient with respect to the maximum overlap term."""
    n = len(h)
    dx = 2.0 / n
    # Convert max_shift to offset in array indexing
    offset = max_shift + (n - 1)  # offset in full convolution indices
    
    grad = np.zeros(n)
    for i in range(n):
        # Special handling for symmetric case
        # When shift k is applied, gradient is 1-h_{i+k}
        j = (i + max_shift) % n
        grad[i] = (1 - h[j]) * dx
    return grad

def find_worst_shift(h):
    """Find shift that gives maximum overlap."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
    # Take max in absolute value correctly
    max_idx = np.argmax(conv)
    shift = max_idx - (n - 1)
    return shift

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using symmetric Gradient Descent."""
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 45  # Odd number for symmetry
    h = initialize_symmetric_h(n_points)
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Set up gradient descent
    learning_rate = 0.1
    max_no_improvement = 10000
    iter_no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        # Find shift that gives maximum overlap
        max_shift = find_worst_shift(best_h)
        
        # Compute exact gradient for the identified shift
        grad = compute_exact_gradient(best_h, max_shift)
        
        # Update with gradient step
        h_new = best_h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Enforce symmetry for new point
        k = n_points // 2
        for i in range(k):
            h_new[i] = h_new[2*k - i] = (h_new[i] + h_new[2*k - i]) / 2
