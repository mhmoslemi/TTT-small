from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Use initial construction if available
    try:
        h_start = initial_h_values.copy()
        n_points = len(h_start)

    # Ensure bounds and integral constraint
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)

    dx = 2.0 / n_points

    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def project_constraints(h):
        h = np.clip(h, 0, 1)
        if np.sum(h) == 0:
            return h
        h = h * (n_points / 2) / np.sum(h)
        return h

    def objective(h):
        return evaluate(h)

    best_h = h_start.copy()
    best_c5 = evaluate(best_h)

    start_time = time.time()
    iteration = 0
    max_iterations = 1000

    while time.time() - start_time < budget_s and iteration < max_iterations:
        result = minimize(objective, best_h, method='L-BFGS-B',
                         bounds=[(0,1) for _ in range(n_points)],
                         options={'maxiter': 10, 'disp': False})

        h_candidate = project_constraints(result.x)
        c5_candidate = evaluate(h_candidate)

        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate

        # Periodic perturbation and smoothing
        if iteration % 10 == 0:
            best_h = np.convolve(best_h, np.ones(3)/3, mode='same')
            best_h = project_constraints(best_h + 0.1 * np.random.randn(n_points))

        iteration += 1

    return (best_h, best_c5, n_points)
