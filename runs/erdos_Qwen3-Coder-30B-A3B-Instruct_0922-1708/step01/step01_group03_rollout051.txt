import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # If defined, return a copy
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values).copy()
            # Ensure symmetry
            if len(h) == n_points:
                return h
    except:
        pass
    # Default initialization: symmetric with two Gaussian bumps
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * (x - 0.5))) * (1 + np.sin(np.pi * (x - 1.5)))
    h = project_h(h, n_points)
    return h

def enforce_symmetry(h):
    """Enforce symmetry around midpoint of the array."""
    n = len(h)
    if n % 2 == 1:
        # Odd case, center is n//2
        k = n // 2
        for i in range(k):
            avg = (h[i] + h[n-1-i]) / 2
            h[i] = h[n-1-i] = avg
    else:
        # Even case, no true center, average pairs from outside in
        k = n // 2 - 1
        for i in range(k+1):
            avg = (h[i] + h[n-1-i]) / 2
            h[i] = h[n-1-i] = avg
    return h

def compute_exact_gradient(h, best_shift):
    """Compute exact gradient for the specific shift that maximizes overlap."""
    n = len(h)
    dx = 2.0 / n
    # idx = shift + n - 1 => shift = idx - n + 1
    g = np.zeros(n)
    if best_shift >= 0:
        # shift is in range [0, n-1] for both arrays
        # h[i] contributes to term (1-h[i+shift])
        for i in range(n):
            j = i + best_shift
            if j < n:
                g[i] = -dx  # derivative of (1-h[i+shift])
    else:
        # shift is negative: use wraparound
        # For -shift >= 0 and < n
        for i in range(n):
            j = i + best_shift
            if j >= 0:
                g[i] = -dx  # d(1-h[j])/dh[i] = -1 if j==i
    return g

def newton_method_symmetric(h, n_points, time_budget_s):
    """Run Newton method with symmetry enforcement."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.1
    iter_no_improvement = 0
    max_no_improvement = 1000
    iteration = 0
    
    while time.time() - start_time < time_budget_s:
        iteration += 1
        # Evaluate current C5 and find best shift
        conv = np.correlate(best_h, 1 - best_h, mode="full")
        best_shift_idx = np.argmax(conv)
        best_shift = best_shift_idx - (n_points - 1)
        
        # Compute exact gradient
        g = compute_exact_gradient(best_h, best_shift)
        
        # Update with gradient descent step
        h_new = best_h - learning_rate * g
        h_new = project_h(h_new, n_points)
        h_new = enforce_symmetry(h_new)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            # Reduce learning rate
            learning_rate *= 0.9
            if learning_rate < 1e-6:
                learning_rate = 1e-6
        else:
            iter_no_improvement += 1
            # Increase learning rate
            learning_rate *= 1.1
            if learning_rate > 1.0:
                learning_rate = 1.0
                
        if iter_no_improvement > max_no_improvement:
            break
        
        if iteration % 500 == 0:
            # Reduce learning rate if stagnated
            learning_rate *= 0.8
            if learning_rate < 1e-6:
                break
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    
    # Use 45 points for good balance of speed and accuracy
    n_points = 45
    h = initialize_h(n_points)
    h = enforce_symmetry(h)
    
    # Run the Newton method optimization
    h_opt, c5_opt = newton_method_symmetric(h, n_points, budget_s)
    
    # Final check and validation
    h_opt = project_h(h_opt, n_points)
    h_opt = enforce_symmetry(h_opt)
    c5_final = evaluate_c5(h_opt)
    
    return (h_opt, c5_final, n_points)
