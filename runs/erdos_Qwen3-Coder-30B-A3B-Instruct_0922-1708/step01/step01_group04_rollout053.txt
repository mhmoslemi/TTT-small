import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=3):
        # Apply smoothing via convolution
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth

    def project_constraint(h):
        # Ensure integral constraint ∫h dx = 1
        h_proj = h * (n_points / 2) / np.sum(h)
        # Ensure bounds
        h_proj = np.clip(h_proj, 0, 1)
        return h_proj

    # Optimization
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    # Set up optimizers (scipy) 
    def objective(h):
        val = evaluate(h)
        return val
    
    # Constraint to maintain ∫h dx = 1
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    no_improvement_count = 0
    num_iterations = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < 50:
        num_iterations += 1
        
        # Use L-BFGS-B with bounds
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                          constraints=cons, options={'maxiter': 10})
        
        if result.success:
            h_new = result.x
            h_new = project_constraint(h_new)
            
            # Smoothing & small perturbation
            h_new = smooth(h_new, window=3)
            h_new = project_constraint(h_new)
            
            c5_new = evaluate(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            # Fall back to perturbation-based approach
            h_trial = best_h.copy()
            # Add random noise
            noise = np.random.normal(0, 0.02, n_points)
            h_trial += noise
            h_trial = project_constraint(h_trial)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1

    # Final projection and validation
    best_h = project_constraint(best_h)
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
