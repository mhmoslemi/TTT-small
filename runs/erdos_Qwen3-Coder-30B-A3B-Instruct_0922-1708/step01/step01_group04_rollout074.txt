import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction with good initial properties
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=3):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def local_perturb(h, delta=0.03):
        # Small, localized perturbations
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        h_new[idx] += np.random.uniform(-delta, delta)
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    def sag_hill_climb(h0, max_iter=1000):
        # Simplified stochastic hill climbing
        h_current = h0.copy()
        c_current = evaluate(h_current)
        best_h = h_current.copy()
        best_c = c_current
        
        for _ in range(max_iter):
            h_trial = local_perturb(h_current)
            c_trial = evaluate(h_trial)
            if c_trial < best_c:
                best_c = c_trial
                best_h = h_trial.copy()
            if c_trial < c_current:
                h_current = h_trial
                c_current = c_trial
        return best_h, best_c

    def optimize_segments(h_start, n_segments=10):
        # Initialize segments
        segment_size = n_points // n_segments
        residuals = n_points % n_segments
        segment_lengths = [segment_size + (1 if i < residuals else 0) for i in range(n_segments)]
        
        # Get initial segment averages
        segments = []
        start_idx = 0
        for length in segment_lengths:
            seg = h_start[start_idx:start_idx+length]
            segments.append(np.mean(seg))
            start_idx += length
        
        segments = np.array(segments)
        # Normalize to integral = 1
        segments = segments * (n_points / 2) / np.sum(segments)
        
        # Project to feasible region
        segments = np.clip(segments, 0, 1)
        
        # Simple optimization: gradient-free
        best_segs = segments.copy()
        best_c = evaluate(reconstruct_h(best_segs, segment_lengths))
        
        for _ in range(10000):
            # Try small changes
            segs_trial = best_segs.copy()
            idx = np.random.randint(0, len(segs_trial))
            delta = np.random.uniform(-0.05, 0.05)
            segs_trial[idx] += delta
            segs_trial = np.clip(segs_trial, 0, 1)
            segs_trial = segs_trial * (n_points / 2) / np.sum(segs_trial)
            
            h_trial = reconstruct_h(segs_trial, segment_lengths)
            c_trial = evaluate(h_trial)
            if c_trial < best_c:
                best_c = c_trial
                best_segs = segs_trial.copy()
        
        return best_segs, segment_lengths
    
    def reconstruct_h(segments, segment_lengths):
        # Reconstruct h from segment averages
        h = np.zeros(n_points)
        start_idx = 0
        for avg, length in zip(segments, segment_lengths):
            h[start_idx:start_idx+length] = avg
            start_idx += length
        return h

    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0

    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        
        # Look for better segment structure
        best_segs, seg_lengths = optimize_segments(best_h)
        h_candidate = reconstruct_h(best_segs, seg_lengths)
        
        # Enhance with local search
        h_candidate, c_candidate = sag_hill_climb(h_candidate, max_iter=200)
        
        if c_candidate < best_c5:
            best_c5 = c_candidate
            best_h = h_candidate.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Occasionally do a thorough perturbation
        if np.random.rand() < 0.05:
            h_trial = local_perturb(best_h, delta=0.1)
            h_trial = smooth(h_trial, window=3)
            c_trial = evaluate(h_trial)
            if c_trial < best_c5:
                best_c5 = c_trial
                best_h = h_trial.copy()
                no_improvement_count = 0

    # Smooth final result
    best_h = smooth(best_h, window=3)
    # Final normalization
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
