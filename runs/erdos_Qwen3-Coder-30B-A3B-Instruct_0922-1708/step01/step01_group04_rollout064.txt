import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def segment_based_optimization(h, m=7):
        # Create segments
        segment_size = n_points // m
        if segment_size * m < n_points:
            segment_size += 1
            
        segment_means = np.array([np.mean(h[i*segment_size:(i+1)*segment_size]) 
                                 for i in range(m)])
        segment_means = np.clip(segment_means, 0, 1)
        
        def objective(c):
            # Reconstruct h from segment means
            h_reconstructed = np.zeros(n_points)
            for i in range(m):
                start_idx = i * segment_size
                end_idx = min((i+1) * segment_size, n_points)
                h_reconstructed[start_idx:end_idx] = c[i]
            return evaluate(h_reconstructed)
        
        # Optimize segment means
        constraints = [{'type': 'eq', 'fun': lambda c: np.sum(c) - m/2}]
        bounds = [(0, 1) for _ in range(m)]
        
        try:
            res = minimize(objective, segment_means, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 100})
            if res.success:
                c_opt = res.x
                h_final = np.zeros(n_points)
                for i in range(m):
                    start_idx = i * segment_size
                    end_idx = min((i+1) * segment_size, n_points)
                    h_final[start_idx:end_idx] = c_opt[i]
                return h_final
        except:
            pass
        return h
    
    def smooth(h, window=3):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
        
    # Optimization loop
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    n_segments = 7
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        
        # Optimize with segment-based method
        h_trial = segment_based_optimization(best_h, n_segments)
        
        # Apply smoothing and normalize
        h_trial = smooth(h_trial, window=3)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        c5_trial = evaluate(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Occasionally increase resolution
        if iterations % 100 == 0 and n_segments < 21:
            n_segments += 2
            # Interpolate from current solution
            h_interp = np.interp(np.linspace(0, n_points-1, n_points), 
                               np.linspace(0, n_points-1, len(best_h)), 
                               best_h)
            best_h = h_interp
            best_c5 = evaluate(best_h)
            
        # Perturb occasionally to escape local minima
        if iterations % 25 == 0:
            h_perturbed = best_h.copy()
            n_perturb = max(1, int(n_points * 0.1))
            indices = np.random.choice(n_points, n_perturb, replace=False)
            for i in indices:
                h_perturbed[i] += np.random.uniform(-0.1, 0.1)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            c5_perturbed = evaluate(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                no_improvement_count = 0
    
    return (best_h, best_c5, n_points)
