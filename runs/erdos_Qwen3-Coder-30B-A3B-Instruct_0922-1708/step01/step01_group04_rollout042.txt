import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    h_start = np.clip(h_start, 0, 1)

    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    best_c5 = evaluate_h(best_h, dx)
    
    # Optimization using different granularities
    resolutions = [7, 14, 21, 35, 50, 70]
    
    for m in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
        if m > n_points:
            m = n_points
        
        # Initialize segment averages
        c = np.full(m, 1.0 / m)
        
        # Define constraint and bounds
        def obj(c):
            # Reconstruct full h
            h_full = np.repeat(c, n_points // m)
            # Pad if necessary
            if len(h_full) < n_points:
                h_full = np.pad(h_full, (0, n_points - len(h_full)), 'constant', constant_values=h_full[-1])
            elif len(h_full) > n_points:
                h_full = h_full[:n_points]
            
            # Apply smoothing to make it more continuous
            kernel = np.exp(-np.arange(-3, 4)**2 / 2)
            kernel = kernel / np.sum(kernel)
            h_smooth = np.convolve(h_full, kernel, mode='same')
            h_smooth = np.clip(h_smooth, 0, 1)
            
            return evaluate_h(h_smooth, dx)
        
        # Bounds for c
        bounds = [(0, 1) for _ in range(m)]
        
        # Constraint: sum(c) = m/2, equivalently sum(c) - m/2 = 0
        cons = {'type': 'eq', 'fun': lambda c: np.sum(c) - m/2}
        
        # Optimize with SLSQP
        result = minimize(obj, c, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6})
        
        if not result.success:
            continue
        
        # Reconstruct h
        c_opt = result.x
        h_reconstructed = np.repeat(c_opt, n_points // m)
        if len(h_reconstructed) < n_points:
            h_reconstructed = np.pad(h_reconstructed, (0, n_points - len(h_reconstructed)), 'constant', constant_values=h_reconstructed[-1])
        elif len(h_reconstructed) > n_points:
            h_reconstructed = h_reconstructed[:n_points]

        kernel = kernel / np.sum(kernel)
        h_final = np.convolve(h_reconstructed, kernel, mode='same')
        h_final = np.clip(h_final, 0, 1)
        
        # Normalize
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        
        # Evaluate

            return evaluate_h(h, dx)
        
        # Optimize using L-BFGS-B to fine-tune
        bounds = [(0, 1) for _ in range(n_points)]

            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_h = h_final_fine.copy()
    
    return (best_h, best_c5, n_points)
