import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Simulated Annealing
    start_time = time.time()
    current_h = h.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    T = 1.0
    alpha = 0.999
    step_size = 0.1
    max_iter = 10000
    
    for iteration in range(max_iter):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Mutation
        h_prime = current_h + np.random.uniform(-step_size, step_size, n_points)
        h_prime = project_h(h_prime, n_points)
        
        # Evaluate
        c5_prime = evaluate_h(h_prime, dx)
        
        # Accept or reject
        delta = c5_prime - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = h_prime
            current_c5 = c5_prime
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        T *= alpha
    
    # Local refinement with gradient descent
    h = best_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s:
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with gradient
        h = h - learning_rate * grad
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (best_h, best_c5, n_points)
