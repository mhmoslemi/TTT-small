import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = population_size // 10
    
    # Initialize population
    population = [project_h(np.random.rand(n_points)) for _ in range(population_size)]
    
    # Add initial h if available
    try:
        population[0] = project_h(initial_h_values)
    except NameError:
        pass
    
    generation = 0
    while (time.time() - start_time) < budget_s - 1 and generation < num_generations:
        # Evaluate fitness
        fitness_scores = [compute_c5(indiv) for indiv in population]
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        current_best = population[0]
        current_best_c5 = fitness_scores[0]
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5
        
        # Selection (roulette wheel)
        # Invert fitness for selection (lower fitness -> higher selection probability)
        inverted_fitness = [1.0 / (x + 1e-15) for x in fitness_scores]
        total_fitness = sum(inverted_fitness)
        probabilities = [f / total_fitness for f in inverted_fitness]
        
        # Create new population
        new_population = []
        
        # Elitism: keep top individuals
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parents = np.random.choice(population_size, size=2, p=probabilities)
            parent1, parent2 = population[parents[0]], population[parents[1]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate / np.sqrt(generation + 1):
                    child1[i] += np.random.normal(0, 0.1)
                    child2[i] += np.random.normal(0, 0.1)
            
            # Project to feasible set
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
        
        generation += 1
    
    # Final quality check
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
