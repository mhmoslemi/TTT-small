import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
        # Interpolate to n_points resolution
        h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), h)
        h = h_interp
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure integral constraint is maintained
    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Parameters for optimization
    lr_h = 0.05
    lr_b = 0.005
    max_iter = 1000
    patience = 50
    
    # Breakpoint candidate - we'll optimize the positions of K breakpoints
    # For efficient computation, we'll work with a reduced number of active breakpoints
    # and synthesize the full h vector
    
    # Use 10 breakpoints + 11 intervals as a reasonable compromise
    K = 11
    b_k = np.linspace(0, 2, K+1)
    
    # Make the optimization more efficient by working with just key parameters
    # We'll use a combination approach: optimize interval values and positions
    def compute_c5(h_vals, breakpoints):
        # Create full h from intervals
        h_full = np.zeros(n_points)
        for i in range(K):
            start_idx = int(breakpoints[i] / 2 * n_points)
            end_idx = int(breakpoints[i+1] / 2 * n_points)
            h_full[start_idx:end_idx] = h_vals[i]
        h_full = np.clip(h_full, 0, 1)
        return np.max(np.correlate(h_full, 1-h_full, mode='full')) * dx
    
    def compute_gradient_h(h_vals, breakpoints):
        grad = np.zeros_like(h_vals)
        eps = 1e-6
        base_c5 = compute_c5(h_vals, breakpoints)
        
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            c5_plus = compute_c5(h_plus, breakpoints)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Absolute improvement threshold
    improvement_threshold = 1e-6
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current C5
        c5 = compute_c5(h, b_k)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
        
        # Optimize h_vals
        num_intervals = len(b_k) - 1
        h_vals = np.zeros(num_intervals)
        for i in range(num_intervals):
            start_idx = int(b_k[i] / 2 * n_points)
            end_idx = int(b_k[i+1] / 2 * n_points)
            h_vals[i] = np.mean(h[start_idx:end_idx])
        
        grad_h = compute_gradient_h(h_vals, b_k)
        h_vals_new = h_vals - lr_h * grad_h
        h_vals_new = np.clip(h_vals_new, 0, 1)
        
        # Update h based on new interval values
        h_new = np.zeros(n_points)
        for i in range(num_intervals):
            start_idx = int(b_k[i] / 2 * n_points)
            end_idx = int(b_k[i+1] / 2 * n_points)
            h_new[start_idx:end_idx] = h_vals_new[i]
        
        # Adjust h_new to maintain integral
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new.copy()
        
        # Update h_vals again for consistency
        for i in range(num_intervals):
            start_idx = int(b_k[i] / 2 * n_points)
            end_idx = int(b_k[i+1] / 2 * n_points)
            h_vals[i] = np.mean(h[start_idx:end_idx])
            
        # Update breakpoints using gradient descent
        grad_b = np.zeros_like(b_k[1:-1])  # Don't optimize endpoints
        
        # Simple heuristic feedback for vertebrae (breakpoint) positions
        for i in range(len(grad_b)):
            # Keep some consistency
            grad_b[i] = b_k[i+2] - b_k[i]
            
        # Adjust breakpoint positions with sign from flow
        for i in range(len(grad_b)):
            b_k[i+1] += lr_b * grad_b[i]
            # Keep order
            b_k[i+1] = max(b_k[i], min(b_k[i+2], b_k[i+1]))
            
        # Restore the last position
        b_k[-1] = 2.0
        
        # Early stopping
        if iteration - last_improvement > patience:
            break
        
    # Final validation
    h = h * (n_points / 2) / np.sum(h)
    c5_final = np.max(np.correlate(h, 1-h, mode='full')) * dx
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
