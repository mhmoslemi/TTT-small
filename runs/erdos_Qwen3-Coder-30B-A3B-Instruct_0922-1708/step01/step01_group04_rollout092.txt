import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        n_points = len(h_start)
    except NameError:
        n_points = 70

        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Normalize to integral = 1
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def reconstruct_from_segments(c, n_points, segment_size):
        # Reconstruct h from segment averages
        h = np.repeat(c, segment_size)
        if len(h) > n_points:
            h = h[:n_points]
        elif len(h) < n_points:
            h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=c[-1])
        # Apply Gaussian smoothing
        if len(h) > 3:
            from scipy.ndimage import gaussian_filter1d
            h = gaussian_filter1d(h, sigma=1.0)
        return np.clip(h, 0, 1)

    def optimize_segments(m, h_init):
        # Define segment averages c as variables
        c0 = np.ones(m) * (m/2) / m  # Start with uniform distribution
        
        # Constraint: sum(c) = m/2
        def constraint_sum(c):
            return np.sum(c) - m/2
        
        # Bounds: 0 <= c_i <= 1
        bounds = [(0, 1) for _ in range(m)]
        
        # Objective function
        def objective(c):
            # Reconstruct h
            segment_size = max(1, n_points // m)
            h = reconstruct_from_segments(c, n_points, segment_size)
            return evaluate(h)
        
        # Constraint dictionary
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimize
        result = minimize(objective, c0, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 500, 'ftol': 1e-6})
        if result.success:
            segment_size = max(1, n_points // m)
            h = reconstruct_from_segments(result.x, n_points, segment_size)
            return h
        else:
            # Return initial guess if optimization fails
            segment_size = max(1, n_points // m)
            return reconstruct_from_segments(c0, n_points, segment_size)

    start_time = time.time()
    
    # Start with coarse resolution
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    min_segments = 7
    max_segments = n_points
    current_segments = min_segments
    
    iteration = 0
    
    while time.time() - start_time < budget_s and current_segments <= max_segments:
        iteration += 1
        # Optimize with current number of segments
        h_new = optimize_segments(current_segments, best_h)
        
        # Check if we improved
        c5_new = evaluate(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Double the segment count
        current_segments *= 2
        if current_segments > n_points:
            break

    # Final optimization with the full resolution
    if time.time() - start_time < budget_s:
        def objective_full(h):
            return evaluate(h)
        
        # Inner optimization with simplex method
        c0 = best_h.copy()
        
        # Optimize with bounds
        bounds = [(0, 1) for _ in range(n_points)]
    
        # Add constraint for integral = 1
        def constraint_integral(h):
            return np.sum(h) * dx - 1.0
        
        cons = {'type': 'eq', 'fun': constraint_integral}
        
        # Optimize with Nelder-Mead
        result = minimize(objective_full, c0, method='Nelder-Mead', 
                          options={'maxiter': 1000, 'xtol': 1e-6, 'ftol': 1e-6})
        
        if result.success:
            h_final = np.clip(result.x, 0, 1)
            c5_final = evaluate(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final

    return (best_h, best_c5, n_points)
