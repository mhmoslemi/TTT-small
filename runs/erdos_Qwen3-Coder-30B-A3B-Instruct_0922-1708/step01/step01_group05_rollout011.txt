import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # Initialize from existing solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a reasonable initial guess
        h = np.ones(n_points) * 0.5
        # Make it piecewise constant with some variation
        segments = 5
        seg_size = n_points // segments
        for i in range(segments):
            start = i * seg_size
            end = start + seg_size if i < segments - 1 else n_points
            h[start:end] = 0.3 + 0.4 * (i % 2)
    
    # Normalize to meet sum constraint
    h = h * (target_sum / np.sum(h))
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(x):
        # Ensure sum constraint
        x = x * (target_sum / np.sum(x))
        return compute_C(x)
    
    # Use differential evolution for global optimization
    result = differential_evolution(
        objective,
        bounds,
        maxiter=1000,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        polish=True,
        disp=False
    )
    
    best_h = result.x
    best_h = best_h * (target_sum / np.sum(best_h))
    best_C = compute_C(best_h)
    
    # Refine with local optimization
    for _ in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Use gradient descent
        eps = 1e-5
        grad = np.zeros(n_points)
        base_C = compute_C(best_h)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - base_C) / eps
        
        # Update with gradient
        learning_rate = 0.01
        best_h = best_h - learning_rate * grad
        
        # Project to feasible region
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        
        # Check if we've improved
        new_C = compute_C(best_h)
        if new_C >= best_C:
            learning_rate *= 0.999
        else:
            best_C = new_C

    return (best_h.tolist(), best_C, n_points)
