import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    max_iter = 1000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize PSO particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Initialize particles around the starting solution
    for i in range(n_particles):
        particle = h + np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (target_sum / np.sum(particle))
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(particle.copy())
        pbest_vals.append(float('inf'))
    
    # Global best
    gbest = h.copy()
    gbest_val = float('inf')
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def evaluate_and_update():
        nonlocal gbest, gbest_val
        C = compute_C(h_vals)
        if C < gbest_val:
            gbest_val = C
            gbest = h_vals.copy()
        return C
    
    # Evaluate initial solution
    evaluate_and_update()
    
    # PSO loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s - 5:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update particle
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Project back to feasible region
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            
            # Evaluate
            C = compute_C(particles[i])
            
            # Update personal best
            if C < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = C
                
        # Update global best
        for i in range(n_particles):
            if pbest_vals[i] < gbest_val:
                gbest_val = pbest_vals[i]
                gbest = pbest[i].copy()
    
    # Refinement with gradient descent on best solution
    h_refined = gbest.copy()
    best_C = gbest_val
    lr = 0.01
    no_improvement = 0
    max_no_improvement = 100
    
    def gradient_step(h_vals):
        C = compute_C(h_vals)
        eps = 1e-5
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C) / eps
        return grad
    
    while time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        grad = gradient_step(h_refined)
        h_refined -= lr * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (target_sum / np.sum(h_refined))
        
        new_C = compute_C(h_refined)
        if new_C < best_C:
            best_C = new_C
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement % 20 == 0:
                lr *= 0.5
        if no_improvement >= max_no_improvement:
            break
    
    return (h_refined.tolist(), best_C, n_points)
