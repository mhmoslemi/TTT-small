import time

    dx = 2.0 / n_points

    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use a more sophisticated optimization approach
    # First compute current C5
    correlation = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(correlation) * dx
    best_c5 = current_c5
    
    # For the optimization, we do gradient descent on a subset of variables
    # We'll prioritize changes to the sample that cause largest reduction in error
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute gradients
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # Compute finite difference gradient
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update with learning rate adapted based on iteration
        lr = 0.01 / (1 + iteration / 100)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)

        h = h_new

        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5

    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
