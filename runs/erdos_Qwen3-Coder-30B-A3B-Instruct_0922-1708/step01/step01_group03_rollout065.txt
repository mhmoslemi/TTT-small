import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a symmetrized function
    x = np.linspace(0, 2, n_points)
    # Sine wave but with triangular pulse shape to keep in [0,1] and satisfy sum
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function with tailored projected gradient descent."""
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 45
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Initialize parameters
    eta = 0.1  # initial learning rate
    max_iter = 10000
    min_improvement = 1e-6
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Loop with time control
    for i in range(max_iter):
        # Break if time is up
        if time.time() - start_time > budget_s * 0.99:
            break
            
        # Find the shift that gives the maximum overlap
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        # Get the index of maximum correlation
        max_idx = np.argmax(conv)
        # The shift is max_idx - (n-1) relative to center
        shift = max_idx - (n - 1)
        
        # Compute gradient components for this shift
        # We are computing d/dh[i] of the overlap at given shift
        gradient = np.zeros(n)
        for j in range(n):
            # This shift is equivalent to checking h[j] vs (1 - h[j + offset]) where offset = shift
            # apply modular arithmetic
            k = (j + shift) % n
            gradient[j] = -dx * (1 - h[k])  # negative because we want to decrease overlap

        # Step size by line search (simple binary search on step size in [0, eta])
        # We'll use a simple form of line search to find optimal step
        step_sizes = np.linspace(0, eta, 50)
        min_c5 = best_c5
        best_step = 0
        
        for step in step_sizes:
            h_test = h - step * gradient
            h_test = project_h(h_test, n_points)
            c5_test = evaluate_c5(h_test)
            if c5_test < min_c5:
                min_c5 = c5_test
                best_step = step
        
        # Apply the update
        h = h - best_step * gradient
        h = project_h(h, n_points)
        
        # Update best
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            eta = 0.1  # reset learning rate on improvement
        else:
            eta *= 0.5  # reduce learning rate
            if eta < 1e-6:
                break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n_points)
