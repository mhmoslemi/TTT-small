import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)

    h = h * (target_sum / np.sum(h))
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # First, try a simple gradient descent approach
    best_h = h.copy()
    best_C = compute_C(h)
    
    # Run gradient descent for 300s
    current_h = h.copy()
    lr = 0.01
    patience = 0
    max_patience = 100
    patience_lr = 0.5
    prev_C = compute_C(current_h)
    
    while time.time() - start_time < budget_s - 100:
        h = current_h.copy()
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - prev_C) / eps
        
        # Update with gradient
        current_h -= lr * grad
        # Projection onto [0,1]
        current_h = np.clip(current_h, 0, 1)
        
        # Rescale to meet integral constraint
        current_h = current_h * (target_sum / np.sum(current_h))
        
        new_C = compute_C(current_h)
        if new_C < best_C:
            best_C = new_C
            best_h = current_h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                lr *= patience_lr
                patience = 0
                
        prev_C = new_C
    
    # Perform a more sophisticated global search using Nelder-Mead
    # We'll work with a lower-dimensional representation
    def eval_h_from_params(params, n_points):
        # Simple piecewise linear interpolation
        # Use fewer points for simplicity
        n_nodes = min(20, n_points)  # Use fewer nodes to make it more efficient
        if n_nodes == n_points:
            h = np.array(params)
        else:
            # Interpolate to full resolution
            x_nodes = np.linspace(0, 2, n_nodes)
            x_full = np.linspace(0, 2, n_points)
            h = np.interp(x_full, x_nodes, params)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return h
    
    def objective(params):
        h = eval_h_from_params(params, n_points)
        return compute_C(h)
        
    # The global search starts from a random but better initial guess
    h_cand = np.random.rand(n_nodes) * 0.5 + 0.25  # between 0.25 and 0.75
    h_cand[0] = 0.5  # Try to keep integrality
    h_cand[-1] = 0.5
    h_cand = np.clip(h_cand, 0, 1)
    h_cand = h_cand * (target_sum / np.sum(h_cand))
    
    # Nested optimization with Nelder-Mead to refine this
    res = minimize(objective, h_cand, method='Nelder-Mead', 
                   options={'maxiter': 1000, 'xtol': 1e-6, 'ftol': 1e-6})
    
    # Evaluate and see if we found anything better
    if not res.success:
        # fallback to current best
        pass
    else:
        h_eval = eval_h_from_params(res.x, n_points)
        C_eval = compute_C(h_eval)
        if C_eval < best_C:
            best_C = C_eval
            best_h = h_eval.copy()
    
    # Final refinement with gradient-based optimization of the best candidate
    # Just to be sure, try an additional pass at the best solution found so far
    current_h = best_h.copy()
    lr = 0.01
    # We do a few gradient steps
    for _ in range(100):
        h = current_h.copy()
        # compute gradient with finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - compute_C(h)) / eps
            
        current_h -= lr * grad
        # project to [0,1] and rescale
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (target_sum / np.sum(current_h))
        
    final_C = compute_C(current_h)
    if final_C < best_C:
        best_C = final_C
        best_h = current_h.copy()
        
    return (best_h.tolist(), best_C, n_points)
