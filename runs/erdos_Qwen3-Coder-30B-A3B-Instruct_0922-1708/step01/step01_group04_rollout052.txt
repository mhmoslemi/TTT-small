import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    def constraint_fun(h):
        return np.sum(h) * dx - 1.0
    
    # Prepare constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_fun}
    
    # Smoothing function
    def smooth(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    # Perturbation function
    def perturb(h, delta=0.1):
        h_new = h.copy()
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Optimization loop
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    
    # Initialize with initial point
    current_h = best_h.copy()
    current_c5 = best_c5
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        
        # Try optimization step
        try:
            res = minimize(
                objective,
                current_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 10}
            )
            if res.success:
                h_trial = res.x
                # Project back to valid range
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
                c5_trial = evaluate(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                    
                current_h = h_trial
                current_c5 = c5_trial
            else:
                raise Exception("Optimization failed")
        except:
            # Fall back to perturbation approach
            h_trial = perturb(current_h)
            h_trial = smooth(h_trial, window=3)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            current_h = h_trial
            current_c5 = c5_trial
        
        # Occasionally do a large perturbation
        if iterations % 20 == 0:
            current_h = perturb(current_h, delta=0.2)
            current_h = smooth(current_h, window=5)
            current_c5 = evaluate(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
    
    return (best_h, best_c5, n_points)
