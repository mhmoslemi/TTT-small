import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def evaluate_and_update(h_vals):
        C = compute_C(h_vals)
        return C
    
    # PSO parameters
    n_particles = 100
    n_dims = n_points
    max_iter = 5000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    for _ in range(n_particles):
        particle = np.random.rand(n_dims)
        particle = np.clip(particle, 0, 1)
        particle = particle * (target_sum / np.sum(particle))
        particles.append(particle)
        velocities.append(np.random.rand(n_dims) * 0.1)
    
    # Initialize global best
    g_best = h.copy()
    g_best_C = evaluate_and_update(g_best)
    
    # PSO main loop
    start_time = time.time()
    iter_count = 0
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            p = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (inertia * v + 
                 cognitive * r1 * (p - particles[i]) + 
                 social * r2 * (g_best - particles[i]))
            
            # Update position
            new_p = p + v
            new_p = np.clip(new_p, 0, 1)
            new_p = new_p * (target_sum / np.sum(new_p))
            
            # Evaluate
            C = evaluate_and_update(new_p)
            if C < g_best_C:
                g_best = new_p.copy()
                g_best_C = C
            
            particles[i] = new_p.copy()
            velocities[i] = v.copy()
            
        iter_count += 1
        
    # Gradient refinement of best particle
    best_h = g_best.copy()
    lr = 0.01
    best_refined_C = g_best_C
    no_improvement_count = 0
    
    def func_and_grad(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        max_k = np.argmax(np.abs(corr))
        max_val = np.abs(corr[max_k])
        
        # Approximate gradient
        grad = np.zeros_like(h_vals)
        eps = 1e-5
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            corr_plus = np.correlate(h_plus, 1-h_plus, mode="full")
            corr_plus *= dx
            max_k_plus = np.argmax(np.abs(corr_plus))
            max_val_plus = np.abs(corr_plus[max_k_plus])
            grad[i] = (max_val_plus - max_val) / eps
        
        return max_val, grad
    
    for step in range(200):
        if time.time() - start_time > budget_s - 1:
            break
            
        C, grad = func_and_grad(best_h)
        
        # Update
        best_h -= lr * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        
        new_C, _ = func_and_grad(best_h)
        if new_C < best_refined_C:
            best_refined_C = new_C
        else:
            no_improvement_count += 1
            if no_improvement_count > 5:
                lr *= 0.5
                no_improvement_count = 0
                
    return (best_h.tolist(), best_refined_C, n_points)
