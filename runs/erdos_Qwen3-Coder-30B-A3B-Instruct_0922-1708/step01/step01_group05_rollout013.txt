import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create from scratch
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    # Objective function
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # Simulated Annealing Parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    max_iter = int(50000)
    
    best_h = h.copy()
    best_C = compute_C(h)
    
    iter_count = 0
    while time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        # Generate neighbor by perturbing small random segments
        h_new = h.copy()
        # Perturb a random number of segments (at most 5)
        n_perturb = np.random.randint(1, 6)
        for _ in range(n_perturb):
            # Pick a random segment
            seg_start = np.random.randint(0, n_points)
            seg_end = min(seg_start + np.random.randint(1, 10), n_points)
            # Add small change to this segment
            delta = np.random.normal(0, 0.05, seg_end - seg_start)
            h_new[seg_start:seg_end] += delta
        # Clip to valid range and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        C_new = compute_C(h_new)
        
        # Accept or reject the new point
        if C_new < best_C or np.random.rand() < np.exp(-(C_new - best_C) / T):
            h = h_new.copy()
            if C_new < best_C:
                best_C = C_new
                best_h = h.copy()
        
        # Update temperature
        T = max(min_T, T * alpha)
    
    # Gradient descent refinement
    # Simple gradient ascent on C
    t = time()
    step = 0.01
    patience = 0
    patience_limit = 1000
    
    while time() - t < 200 and time() - start_time < budget_s - 1:
        C_current = compute_C(best_h)
        grad = np.zeros_like(best_h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = h_plus * (target_sum / np.sum(h_plus))  # Renormalize
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C_current) / eps
        
        h_new = best_h - step * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        C_new = compute_C(h_new)
        if C_new < best_C:
            best_C = C_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                step *= 0.5
                patience = 0
        if step < 1e-8:
            break
    
    return (best_h.tolist(), best_C, n_points)
