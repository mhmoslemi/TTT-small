import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=3):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def perturb(h, delta=0.1):
        # Create a perturbed version
        h_new = h.copy()
        # Perturb about 10% of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def project(h):
        # Ensure the integral is 1 and values are in [0,1]
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def objective(h):
        return evaluate(h)
    
    def constraint_fn(h):
        return np.sum(h) * dx - 1.0
    
    # Initialize
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        
        # Try multiple random perturbations
        for _ in range(50):
            h_trial = perturb(best_h, delta=0.1)
            h_trial = smooth(h_trial, window=3)
            c5_trial = evaluate(h_trial)
            
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                break
        else:
            # Try gradient-based optimization
            try:
                # Create bounds for optimization
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Apply optimization step
                result = minimize(objective, best_h, method='L-BFGS-B', 
                                bounds=bounds, options={'maxiter': 5})
                if result.success:
                    h_new = project(result.x)
                    c5_new = evaluate(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new
                        no_improvement_count = 0
            except:
                pass
            
            no_improvement_count += 1
        
        # Periodic perturbation to escape local optima
        if iterations % 10 == 0:
            best_h = perturb(best_h, delta=0.05)
            best_h = smooth(best_h, window=5)
            best_c5 = evaluate(best_h)
            no_improvement_count = 0
    
    return (best_h, best_c5, n_points)
