import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap

    def smooth(h, window_size=3):
        if window_size < 2:
            return h
        # Apply symmetric smoothing
        return np.convolve(h, np.ones(window_size)/window_size, mode='same')

    def project_to_feasible(h):
        # Ensure h in [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to make integral equal to 1
        h = h * (n_points / 2) / np.sum(h)
        return h

    def perturb(h, strength=0.1):
        h_new = h.copy()
        # Perturb a fraction of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-strength, strength)
        # Project to feasible set
        return project_to_feasible(h_new)

    def optimize_step(h):
        # Define constraint function
        def constraint_func(h):
            return np.sum(h) * dx - 1.0
        
        # Optimizer function for minimization thread
        def obj(h):
            return evaluate(h)
        
        # Optimize with bounds and constraint
        bounds = [(0, 1) for _ in range(n_points)]
        # Use scipy.optimize.minimize with L-BFGS-B
        result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, 
                         options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-6})
        
        return result.x

    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    improvement_threshold = 1e-6

    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        
        # Step 1: Optimization
        h_opt = optimize_step(best_h)
        
        # Step 2: Perturbation + Smoothing
        h_trial = perturb(h_opt, strength=0.05)
        h_trial = smooth(h_trial, window_size=3)
        h_trial = project_to_feasible(h_trial)
        
        c5_trial = evaluate(h_trial) 
        
        if c5_trial < best_c5 - improvement_threshold:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Occasionally do a different kind of search
        if iterations % 10 == 0:
            # Try a random perturbation with larger step
            h_random = perturb(best_h, strength=0.2)
            h_random = smooth(h_random, window_size=5)
            h_random = project_to_feasible(h_random)
            c5_random = evaluate(h_random)
            if c5_random < best_c5 - improvement_threshold:
                best_c5 = c5_random
                best_h = h_random.copy()

    return (best_h, best_c5, n_points)
