import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangular pulse
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h, k, n_points, dx):
    """Compute the gradient of C5 with respect to h for a specific shift k."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + k) % n_points
        grad[i] = (1 - h[j]) * dx
    return grad

def line_search(h, grad, n_points, dx, max_alpha=0.5):
    """Find optimal step size using line search."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Bracket the minimum
    try:
        res = minimize_scalar(objective, bounds=(0, max_alpha), method='bounded')
        return res.x
    except:
        return max_alpha * 0.5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using gradient descent with exact gradients."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    dx = 2.0 / n_points
    
    # Initialize learning rate and tracking
    learning_rate = 0.1
    max_no_improvement = 1000
    no_improvement_count = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    iteration = 0
    while time.time() - start_time < budget_s and no_improvement_count < max_no_improvement:
        # Find the shift that causes maximum overlap
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute gradient for this shift
        grad = compute_gradient(h, shift, n_points, dx)
        
        # Line search for optimal step size
        alpha = line_search(h, grad, n_points, dx, learning_rate)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
            # Increase learning rate slightly
            learning_rate = min(learning_rate * 1.2, 0.5)
        else:
            no_improvement_count += 1
            # Decrease learning rate
            learning_rate = max(learning_rate * 0.5, 1e-6)
        
        iteration += 1
        if iteration % 100 == 0:
            pass  # Progress update
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
