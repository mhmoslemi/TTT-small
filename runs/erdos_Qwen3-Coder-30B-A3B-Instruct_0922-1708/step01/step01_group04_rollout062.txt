import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def segment_based_optimization(h, n_segments, max_iter=200):
        # Divide h into segments
        segment_size = n_points // n_segments
        if segment_size == 0:
            segment_size = 1
        segments = np.array([np.mean(h[i*segment_size:(i+1)*segment_size]) 
                             for i in range(n_segments)])
        
        # Make segments to fit constraint
        segments = np.clip(segments, 0, 1)
        sum_segments = np.sum(segments)
        if sum_segments > 0:
            segments = segments * (n_segments / 2) / sum_segments
            
        # Optimize segments using scipy minimize
        def objective(segments):
            # Reconstruct h from segments
            h_reconstructed = np.repeat(segments, segment_size)
            if len(h_reconstructed) > n_points:
                h_reconstructed = h_reconstructed[:n_points]
            elif len(h_reconstructed) < n_points:
                h_reconstructed = np.pad(h_reconstructed, (0, n_points - len(h_reconstructed)))
            return evaluate(h_reconstructed)
            
        # Constraint: sum of segments should be n_segments / 2
        def constraint_fn(segments):
            return np.sum(segments) - n_segments / 2
            
        cons = {'type': 'eq', 'fun': constraint_fn}
        bounds = [(0, 1) for _ in range(n_segments)]
        
        res = minimize(objective, segments, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter})
        
        # Reconstruct h
        segments_opt = res.x
        h_reconstructed = np.repeat(segments_opt, segment_size)
        if len(h_reconstructed) > n_points:
            h_reconstructed = h_reconstructed[:n_points]
        elif len(h_reconstructed) < n_points:
            h_reconstructed = np.pad(h_reconstructed, (0, n_points - len(h_reconstructed)))
            
        return h_reconstructed
    
    def smooth(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def perturb(h, delta=0.05):
        h_new = h.copy()
        n_perturb = max(1, int(len(h) * 0.1))
        indices = np.random.choice(len(h), n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (len(h) / 2) / np.sum(h_new)
        return h_new
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    n_segments = 7
    
    while time.time() - start_time < budget_s and no_improvement_count < 50:
        iterations += 1
        
        # Run segment-based optimization
        h_trial = segment_based_optimization(best_h, n_segments)
        h_trial = smooth(h_trial, window=3)
        c5_trial = evaluate(h_trial)
        
        # If improved, update
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Perturb occasionally
        if no_improvement_count % 20 == 0:
            h_trial = perturb(best_h)
            h_trial = smooth(h_trial, window=3)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                
        # Increase segment count if no improvement for a while
        if no_improvement_count > 30 and n_segments < 21:
            n_segments += 2
            no_improvement_count = 0
            
    return (best_h, best_c5, n_points)
