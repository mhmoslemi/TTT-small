import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

        h = project_h(h, n_points)
    
    # Dual optimization with gradient refinement
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Dual variables for overlap constraints
    dual_vars = np.ones(2 * n_points - 1)
    learning_rate = 0.01
    patience = 0
    max_patience = 1000

            patience += 1

                h_plus = h.copy()

                h_plus = project_h(h_plus, n_points)

                h_minus[i] -= eps
