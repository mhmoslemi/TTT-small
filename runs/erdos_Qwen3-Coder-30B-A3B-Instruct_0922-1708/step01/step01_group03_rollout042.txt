import numpy as np
from scipy import optimize
import time

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use symmetric function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using gradient-based approach."""
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45  # Fixed size for speed
    h = initialize_h(n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Adaptive learning rate
    eta = 0.1
    min_eta = 1e-6
    patience = 0
    max_patience = 50
    convergence_threshold = 1e-6
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 10000:
        iteration += 1
        # Find the shift that creates the maximum overlap
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        k = max_idx - (n - 1)  # Actual shift value
        
        # Compute gradient: derivative of overlap w.r.t. h[i]
        # For shift k, the overlap is sum h[i](1 - h[i+k])
        # So gradient w.r.t. h[i] is (1 - h[i+k]) for the term h[i](1-h[i+k])
        # But we're looking at the shift that maximizes overlap, so we compute gradients
        # over all possible shifts, but we only use one.
        grad = np.zeros(n)
        for i in range(n):
            j = (i + k) % n
            grad[i] = -dx * (1 - h[j])
        
        # Line search for step size alpha
        def objective(alpha):
            h_new = h - alpha * grad
            h_new = project_h(h_new, n_points)
            return evaluate_c5(h_new)
        
        # Simple line search with backtracking
        alpha = eta
        c5_new = objective(alpha)
        max_alpha = 10 * eta
        while c5_new >= best_c5 and alpha > 1e-8:
            alpha /= 2
            c5_new = objective(alpha)
        
        if alpha > 1e-8:
            h = h - alpha * grad
            h = project_h(h, n_points)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                patience = 0
                eta = min(eta * 2, 1.0)  # Increase learning rate
            else:
                patience += 1
                if patience > max_patience and eta > min_eta:
                    patience = 0
                    eta /= 2
                    if eta < min_eta:
                        break
        else:
            # If no acceptable step size found, reduce learning rate
            eta /= 2
            if eta < min_eta:
                break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n_points)
