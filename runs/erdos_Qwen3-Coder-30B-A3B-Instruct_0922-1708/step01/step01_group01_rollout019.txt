import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Use a reduced representation with fewer parameters
    # Try segments approach with fewer degrees of freedom
    m = 5  # number of segments
    segment_values = [0.0, 0.5, 1.0]
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Generate all possible segment combinations
    combinations = list(product(segment_values, repeat=m))
    candidates = []
    log_interval = max(1, len(combinations) // 100)
    
    for i, segment_vals in enumerate(combinations):
        if i % log_interval == 0 and time.time() - start_time > 0.7 * budget_s:
            break
            
        # Construct h from segments
        h_seg = np.zeros(n_points)
        segment_length = n_points // m
        remainder = n_points % m
        
        # Distribute segment_length + (1 if i < remainder else 0) into segments
        current_idx = 0
        for j in range(m):
            length = segment_length + (1 if j < remainder else 0)
            h_seg[current_idx:current_idx + length] = segment_vals[j]
            current_idx += length
            
        # Project to satisfy constraints
        h_seg = project_h(h_seg, n_points)
        c5 = evaluate_h(h_seg, dx)
        candidates.append((h_seg.copy(), c5))
    
    # Sort and get top candidates
    candidates.sort(key=lambda x: x[1])
    top_candidates = candidates[:10]
    
    # Refine with gradient descent
    learning_rate = 0.01
    max_patience = 5000
    patience = 0
    
    for h_init, _ in top_candidates:
        h_refine = h_init.copy()
        c5_refine = evaluate_h(h_refine, dx)
        
        for _ in range(10000):  # limited iterations per candidate
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Compute gradient
            eps = 1e-5
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_refine.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h_refine -= learning_rate * grad
            h_refine = project_h(h_refine, n_points)
            
            # Evaluate
            new_c5 = evaluate_h(h_refine, dx)
            if new_c5 < c5_refine:
                c5_refine = new_c5
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    break
            if patience > 0 and patience % 1000 == 0:
                learning_rate *= 0.5
                
        if c5_refine < best_c5:
            best_c5 = c5_refine
            best_h = h_refine.copy()
    
    return (best_h, best_c5, n_points)
