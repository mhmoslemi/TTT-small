import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sample(n_points, n_samples):
    """Generate Latin Hypercube Sampling points"""
    samples = np.zeros((n_samples, n_points))
    for i in range(n_points):
        # Generate random permutation for each dimension
        perm = np.random.permutation(n_samples)
        samples[:, i] = (perm + np.random.random(n_samples)) / n_samples
    return samples

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Phase 1: Latin Hypercube Sampling for global exploration
    n_samples = 50
    n_elites = 20
    lhs_samples = lhs_sample(n_points, n_samples)
    candidates = []
    
    for i in range(n_samples):
        h_trial = lhs_samples[i] * 0.5  # Scale to [0, 0.5] range
        h_trial = project_h(h_trial, n_points)
        c5_val = evaluate_h(h_trial, dx)
        candidates.append((h_trial, c5_val))
    
    # Add initial solution if it's good
    candidates.append((h.copy(), evaluate_h(h, dx)))
    
    # Sort and select elites
    candidates.sort(key=lambda x: x[1])
    elites = candidates[:n_elites]
    
    # Phase 2: Evolution Strategy
    n_offspring = 5
    max_generations = 1000
    generation = 0
    
    while time.time() - start_time < 0.7 * budget_s and generation < max_generations:
        offspring_pool = []
        for h_elite, _ in elites:
            for _ in range(n_offspring):
                # Small random mutation
                h_mut = h_elite + np.random.normal(0, 0.05, n_points)
                h_mut = project_h(h_mut, n_points)
                c5_val = evaluate_h(h_mut, dx)
                offspring_pool.append((h_mut, c5_val))
        
        # Combine elites and offspring
        all_candidates = elites + offspring_pool
        all_candidates.sort(key=lambda x: x[1])
        elites = all_candidates[:n_elites]
        generation += 1
    
    # Phase 3: Multi-Start Gradient Descent refinement
    best_h, best_c5 = elites[0]
    
    # Try multiple starting points from elites
    for h_start, _ in elites[:5]:  # Test top 5
        h_refined = h_start.copy()
        learning_rate = 0.01
        patience = 0
        max_patience = 5000
        
        while time.time() - start_time < budget_s - 1 and patience < max_patience:
            # Compute gradient
            eps = 1e-5
            grad = np.zeros(n_points)
            base_cost = evaluate_h(h_refined, dx)
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_refined.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h_refined -= learning_rate * grad
            h_refined = project_h(h_refined, n_points)
            
            new_cost = evaluate_h(h_refined, dx)
            if new_cost < best_c5:
                best_c5 = new_cost
                best_h = h_refined.copy()
                patience = 0
            else:
                patience += 1
                
            if patience > 0 and patience % 1000 == 0:
                learning_rate *= 0.5
                patience = 0
    
    return (best_h, best_c5, n_points)
