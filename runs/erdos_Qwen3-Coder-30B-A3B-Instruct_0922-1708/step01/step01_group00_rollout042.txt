import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 10000
    elite_size = pop_size // 5
    mutation_rate = 0.1
    mutation_strength = 0.05
    min_improvement_interval = 300
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Scale to sum to n_points/2
        return np.clip(h, 0, 1)
    
    # If initial_h_values exists, use it
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
        population = [initial_h]
        population.extend([create_individual() for _ in range(pop_size - 1)])
    else:
        population = [create_individual() for _ in range(pop_size)]
    
    # Helper to evaluate fitness
    def evaluate(h):
        # Compute overlap integral using correlation
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Helper to repair individuals
    def repair(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Create initial population fitness
    fitness = [evaluate(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    last_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            last_improvement = generation
        
        # Early termination if no improvement
        if generation - last_improvement > min_improvement_interval:
            break
            
        # Create new population
        new_population = population[:elite_size].copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            idx1, idx2 = random.sample(range(elite_size), 2)
            parent1, parent2 = population[idx1], population[idx2]
            
            # Arithmetic crossover
            alpha = random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            child = repair(child)
            
            # Mutation
            if random.random() < mutation_rate:
                noise = np.random.normal(0, mutation_strength, n_points)
                child += noise
                child = repair(child)
                
            new_population.append(child)
            
        population = new_population[:pop_size]
        fitness = [evaluate(h) for h in population]
        
        generation += 1

    return (best_h.tolist(), best_c5, n_points)
