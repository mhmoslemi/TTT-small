import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Genetic Algorithm Parameters
    pop_size = 50
    num_parents = int(pop_size * 0.2)
    mutation_rate = 0.1
    mutation_strength = 0.05
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
            individual = individual / np.sum(individual) * (n_points / 2)
        else:
            individual = np.random.uniform(0, 1, n_points) 
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    last_improvement = 0
    stall_threshold = 300  # seconds
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate population
        fitness_scores = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness_scores.append(c5)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness_scores)
        if fitness_scores[min_fitness_idx] < best_c5:
            best_c5 = fitness_scores[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            last_improvement = time.time()
        
        # Check for stagnation
        if time.time() - last_improvement > stall_threshold:
            break
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        parents = [population[i] for i in sorted_indices[:num_parents]]
        
        # Create new population through crossover and mutation
        new_population = parents.copy()
        
        # Crossover: create offspring from pairs of parents
        while len(new_population) < pop_size:
            parent1 = parents[np.random.randint(0, len(parents))]
            parent2 = parents[np.random.randint(0, len(parents))]
            
            # Arithmetic crossover
            alpha = np.random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, mutation_strength, n_points)
                child = child + noise
                
            # Clip and rescale
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:pop_size]
        
        # Occasionally apply gradient-based refinement
        if iteration % 10 == 0:
            for i in range(len(population)):
                individual = population[i]
                correlation = np.correlate(individual, 1 - individual, mode='full')
                c5 = np.max(correlation) * dx
                
                # Simple gradient approximation
                grad = np.zeros_like(individual)
                eps = 1e-5
                
                for j in range(n_points):
                    h_plus = individual.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5) / eps
                
                # Update with small step
                lr = 0.001
                individual_new = individual - lr * grad
                
                # Clip and rescale
                individual_new = np.clip(individual_new, 0, 1)
                individual_new = individual_new * (n_points / 2) / np.sum(individual_new)
                
                population[i] = individual_new
    
    return (best_h.tolist(), best_c5, n_points)
