import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a triangular distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to higher resolution
        h_high_res = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(h)), h)
        h = h_high_res
    else:
        # Create a triangular-like function
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1)
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure h is in [0,1] and has correct integral
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute exact gradient
        # Find where the maximum correlation occurs
        max_idx = np.argmax(correlation)
        max_k = max_idx - (n_points - 1)  # convert to shift
        
        # Compute gradient of C5 with respect to h
        grad = np.zeros_like(h)
        
        # For any k, the derivative of max_{k} (h * (1 - h_k)) is not straightforward
        # We will use a deterministic approach by computing contributions for a few nearby points
        
        # Based on the maximum location, we compute the derivative via backward/forward differences
        # Use a stable gradient approximation
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Clip the gradient to prevent large steps
        grad = np.clip(grad, -100, 100)
        
        # Update h using gradient descent with learning rate scheduling
        h_new = h - lr * grad
        
        # Project h_new onto the feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adjust learning rate every 1000 iterations if no improvement
        if iteration % 1000 == 0 and iteration > 0:
            lr *= 0.5
            if lr < 1e-6:
                lr = 1e-6
                
    return (best_h.tolist(), best_c5, n_points)
