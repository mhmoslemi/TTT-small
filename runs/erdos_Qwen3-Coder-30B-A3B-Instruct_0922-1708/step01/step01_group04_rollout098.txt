import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap

    def perturb(h, delta=0.05):
        # Create a perturbed version
        h_new = h.copy()
        # Perturb about 10% of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def smooth(h, window=3):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def projected_descent(h_init, max_iter=100, step_size=0.01):
        h = h_init.copy()
        best_h = h.copy()
        best_c5 = evaluate(h)
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient approximation
            eps = 1e-6
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = evaluate(h_plus)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                c5_minus = evaluate(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update step
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
            else:
                step_size *= 0.5
                if step_size < 1e-8:
                    break
                    
        return best_h, best_c5
    
    # Initialization
    h_best = h_start.copy()
    c5_best = evaluate(h_best)
    
    # Main optimization loop
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        iter_count += 1
        
        # Try a gradient-based descent
        h_trial, c5_trial = projected_descent(h_best, max_iter=100)
        
        if c5_trial < c5_best - 1e-8:
            h_best = h_trial
            c5_best = c5_trial
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Occasionally perturb
        if np.random.random() < 0.1:
            h_perturb = perturb(h_best, delta=0.02)
            h_perturb = smooth(h_perturb, window=3)
            c5_perturb = evaluate(h_perturb)
            if c5_perturb < c5_best - 1e-8:
                h_best = h_perturb
                c5_best = c5_perturb
                no_improvement_count = 0
        else:
            # Regular refinement
            for _ in range(50):
                if time.time() - start_time > budget_s - 1:
                    break
                h_trial = perturb(h_best, delta=0.01)
                h_trial = smooth(h_trial, window=3)
                c5_trial = evaluate(h_trial)
                if c5_trial < c5_best - 1e-10:
                    h_best = h_trial
                    c5_best = c5_trial
                    no_improvement_count = 0
        
        # Early exit if we're doing badly
        if iter_count % 1000 == 0 and c5_best > 0.381:
            # Try random restarts
            h_random = np.random.rand(n_points)
            h_random = h_random * (n_points / 2) / np.sum(h_random)
            c5_random = evaluate(h_random)
            if c5_random < c5_best - 1e-8:
                h_best = h_random
                c5_best = c5_random
                no_improvement_count = 0
                
    # Final smoothing/optimization
    h_final = smooth(h_best, window=5)
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    c5_final = evaluate(h_final)
    
    return (h_final, c5_final, n_points)
