import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=5):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def perturb(h, delta=0.05):
        # Create a perturbed version
        h_new = h.copy()
        # Perturb about 10% of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Constraint function for L-BFGS
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Derivative of constraint
    def constraint_jac(h):
        return np.ones_like(h)
    
    # Local optimization with constraints
    def optimize_step(h):
        # Define bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Constraint
        cons = {'type': 'eq', 'fun': constraint_func, 'jac': constraint_jac}
        
        # Optimize
        res = minimize(evaluate, h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                       options={'maxiter': 200})
        if res.success:
            return res.x
        else:
            return h
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    perturbation_delta = 0.05
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        # Step 1: Optimize with L-BFGS-B
        h_opt = optimize_step(best_h)
        
        # Step 2: Apply smoothing
        h_smooth = smooth(h_opt, window=5)
        
        # Step 3: Evaluate new solution
        c5_smooth = evaluate(h_smooth)
        
        # Step 4: Update best if improved
        if c5_smooth < best_c5:
            best_c5 = c5_smooth
            best_h = h_smooth.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Step 5: Apply random perturbation every 10 iterations
        if iterations % 10 == 0:
            h_perturbed = perturb(best_h.copy(), delta=perturbation_delta)
            # Adaptively reduce perturbation if not helpful
            c5_perturbed = evaluate(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                no_improvement_count = 0
            else:
                # Reduce perturbation strength
                perturbation_delta = max(0.001, perturbation_delta * 0.95)
    
    return (best_h, best_c5, n_points)
