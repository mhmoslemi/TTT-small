import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Enhanced optimization: A hybrid approach with local search
    n_iter = 0
    max_iter = 5000
    
    # Define optimization variables for CVXPY
    h_var = cp.Variable(n_points)
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    
    # Augment with more structured search in the constraint space
    while n_iter < max_iter and (time.time() - start_time) < budget_s - 1:
        # Update probability of jumping to better areas
        if n_iter % 100 == 0:
            # Introduce some randomness in structure
            h = project_h(h + np.random.normal(0, 0.01, n_points))
        
        # Try gradient like steps with momentum 
        current_c5 = compute_c5(h)
        
        # Small random perturbation 
        h_new = h + np.random.normal(0, 0.005, n_points)
        h_new = project_h(h_new)
        new_c5 = compute_c5(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            
        # Local search step using optimization
        if n_iter % 20 == 0:
            # Create a new candidate with better structure
            h_candidate = h.copy()
            # Add small perturbations with negative impact on the overlap
            # for a targeted search in lower C5 region
            for i in range(n_points):
                h_trial = h_candidate.copy()
                # Randomly shrink or increase one value  
                if np.random.rand() > 0.5:
                    h_trial[i] = min(h_trial[i] + 0.03, 1.0) 
                else:
                    h_trial[i] = max(h_trial[i] - 0.03, 0.0)
                h_trial = project_h(h_trial)
                c5_trial = compute_c5(h_trial)
                if c5_trial < current_c5:
                    h = h_trial
                    current_c5 = c5_trial

        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        n_iter += 1
        
        if (time.time() - start_time) > budget_s - 1:
            break

    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
