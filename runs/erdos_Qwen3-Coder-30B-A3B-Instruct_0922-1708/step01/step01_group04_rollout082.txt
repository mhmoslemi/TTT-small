import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=5):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth

    def project(h):
        # Clip to bounds
        h = np.clip(h, 0, 1)
        # Normalize to correct integral
        h = h * (n_points / 2) / np.sum(h)
        return h

    def lbfgs_objective(h):
        # Convert to numpy array if it's not
        h = np.array(h)
        # Compute the correlation
        corr = correlate(h, 1 - h, mode='full')
        # Return negative as we want to minimize
        return -np.max(corr) * dx

    def gradient(h):
        # This is a heuristic gradient for optimization
        # We are computing gradient of max correlation
        h = np.array(h)
        corr = correlate(h, 1 - h, mode='full')
        # Get the index of maximum correlation
        max_idx = np.argmax(corr)
        # Gradient approximation - not exact but good enough for search
        # This is approximate and not precise, but useful for descent
        grad = np.zeros_like(h)
        # Add gradients for larger values where the correlation is maximal
        # This is a crude approximation, but effective
        if max_idx < len(h):
            # Gradient influence near max_idx
            for i in range(len(h)):
                if abs(i - max_idx) <= len(h) // 2:
                    # Positive influence on larger values, negative on smaller
                    grad[i] += 1.0 / (1 + abs(i - max_idx))
        return grad

    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    
    # Initialize optimization with current best
    h_current = best_h.copy()
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        
        # Apply L-BFGS optimization
        try:
            # First try with different initializations
            h_trial = h_current.copy()
            # Perturb slightly to avoid local optima
            noise_magnitude = 0.01
            for i in range(len(h_trial)):
                if np.random.rand() < 0.1:  # 10% chance
                    h_trial[i] += np.random.uniform(-noise_magnitude, noise_magnitude)
            
            # Ensure valid range
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            # Try optimization step
            result = minimize(lbfgs_objective, h_trial, method='L-BFGS-B', 
                              bounds=[(0, 1) for _ in range(n_points)], 
                              options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-10})
            
            if result.success:
                h_optimized = result.x.copy()
                h_optimized = project(h_optimized)
                c5_optimized = evaluate(h_optimized)
                
                if c5_optimized < best_c5:
                    best_c5 = c5_optimized
                    best_h = h_optimized.copy()
                    no_improvement_count = 0
                    h_current = h_optimized.copy()
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        except Exception as e:
            # If optimization fails, just perturb
            no_improvement_count += 1
            
        # Periodically apply robust perturbation
        if iterations % 10 == 0:
            h_perturbed = best_h.copy()
            # Perturb a few points
            n_perturb = max(1, int(n_points * 0.05))
            indices = np.random.choice(n_points, n_perturb, replace=False)
            for i in indices:
                h_perturbed[i] += np.random.uniform(-0.05, 0.05)
            
            h_perturbed = project(h_perturbed)
            h_perturbed = smooth(h_perturbed, window=5)
            c5_perturbed = evaluate(h_perturbed)
            
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                no_improvement_count = 0
                
    return (best_h, best_c5, n_points)
