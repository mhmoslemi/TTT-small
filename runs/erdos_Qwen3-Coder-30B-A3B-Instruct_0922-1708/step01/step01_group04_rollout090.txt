import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def reconstruct_h(c, m, n_points):
        # Reconstruct step function from segment averages c
        segment_size = n_points // m
        h = np.zeros(n_points)
        for i in range(m):
            start_idx = i * segment_size
            end_idx = (i + 1) * segment_size if i < m - 1 else n_points
            h[start_idx:end_idx] = c[i]
        # Apply smoothing for continuity
        if m > 1:
            h = np.convolve(h, np.exp(-np.linspace(-2, 2, 21)**2/2), mode='same')
            h = np.clip(h, 0, 1)
        return h
    
    def optimize_segments(m, initial_c):
        # Optimize segment averages c for given m
        def objective(c):
            h = reconstruct_h(c, m, n_points)
            return evaluate(h)
        
        # Constraints
        def constraint_sum(c):
            return np.sum(c) - m/2
        
        bounds = [(0, 1) for _ in range(m)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimize
        result = minimize(objective, initial_c, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        if result.success:
            return result.x
        else:
            return initial_c
    
    # Start with a coarse resolution
    min_m = 7
    max_m = n_points
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    # Variational optimization at multiple resolutions
    m_resolution = [min_m]
    while m_resolution[-1] * 2 <= max_m:
        m_resolution.append(m_resolution[-1] * 2)
    if m_resolution[-1] < max_m:
        m_resolution.append(max_m)
    
    for m in m_resolution:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize c from best solution found so far
        if m == min_m:
            # Initialize with a simple construction
            c = np.ones(m) * 0.5
            # Adjust to satisfy sum = m/2
            c = c * (m/2) / np.sum(c)
        else:
            # Refine from previous best c
            # First reduce to m segments
            if len(best_h) > m:
                # Downsample
                segment_size = len(best_h) // m
                c = np.array([np.mean(best_h[i*segment_size:(i+1)*segment_size]) 
                              for i in range(m)])
            else:
                # Upsample: repeat each element
                c = np.repeat(best_h[:m], len(best_h) // m)
                if len(c) < m:
                    c = np.pad(c, (0, m - len(c)), mode='constant', constant_values=0.5)
            c = np.clip(c, 0, 1)
        
        # Optimize at this resolution
        c_opt = optimize_segments(m, c)
        
        # Reconstruct h
        h_new = reconstruct_h(c_opt, m, n_points)
        
        # Evaluate
        c5_new = evaluate(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final fine-tuning with local search
    no_improvement_count = 0
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb
        h_trial = best_h.copy()
        indices = np.random.choice(n_points, max(1, n_points // 10), replace=False)
        for i in indices:
            h_trial[i] += np.random.normal(0, 0.01)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        c5_trial = evaluate(h_trial)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        if no_improvement_count >= 100:
            break
    
    return (best_h, best_c5, n_points)
