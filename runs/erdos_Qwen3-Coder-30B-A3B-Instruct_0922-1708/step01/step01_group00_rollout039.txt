import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use 100 points for better resolution
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to our resolution
        h = np.interp(np.linspace(0, 2, n_points), 
                     np.linspace(0, 2, len(h)), h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Precompute indices for correlation computation
    indices = np.arange(n_points)
    
    start_time = time.time()
    iteration = 0
    
    # Gradient descent with momentum for step sizes
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradients via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update with momentum
        velocity = momentum * velocity - 0.01 * grad
        h_new = h + velocity
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Early stopping based on improvement
        if iteration % 100 == 0 and iteration > 0:
            # Check if relative improvement is small
            if best_c5 < 1e-6 or iteration > 1000:
                break
                
    return (best_h.tolist(), best_c5, n_points)
