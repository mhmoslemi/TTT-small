import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 10000
    learning_rate = 0.01
    patience = 300  # seconds to wait for improvement
    
    # Initialize from parent or create a good starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create symmetric sparse function
        h = np.zeros(n_points)
        h[20:30] = 0.8
        h[70:80] = 0.6
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    last_improvement = start_time
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = time.time()
        
        # Early termination if no improvement
        if time.time() - last_improvement > patience:
            break
            
        # Compute gradient using finite difference
        grad_h = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad_h[i] = (c5_plus - c5) / eps
        
        # Update step with adaptive learning rate
        h_new = h - learning_rate * grad_h
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Periodic rescale to avoid numerical drift
        if iteration % 100 == 0:
            h = h * (n_points / 2) / np.sum(h)
    
    # Final computation to ensure accuracy
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
