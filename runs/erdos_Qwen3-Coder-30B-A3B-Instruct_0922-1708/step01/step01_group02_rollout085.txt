import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Genetic algorithm parameters
    pop_size = 100
    num_generations = 5000
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = pop_size // 10

    # Initialize population
    population = []
    for _ in range(pop_size):
        ph = np.random.rand(n_points)
        ph = project_h(ph)
        population.append(ph)
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        population[0] = h.copy()

    for generation in range(num_generations):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = [compute_c5(ind) for ind in population]
        
        # Sorting by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best if improved
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection: roulette wheel
        inverses = [1.0 / (f + 1e-10) for f in fitness]
        total_fitness = sum(inverses)
        probabilities = [f / total_fitness for f in inverses]
        
        new_population = population[:elite_size].copy()
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Select parents
            parents = np.random.choice(pop_size, size=2, p=probabilities)
            parent1, parent2 = population[parents[0]], population[parents[1]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:point], parent2[point:]])
                child2 = np.concatenate([parent2[:point], parent1[point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate / (1 + generation**0.5):
                    child1[i] += np.random.normal(0, 0.1)
                    child2[i] += np.random.normal(0, 0.1)
            
            # Project and add to new population
            child1 = project_h(child1)
            child2 = project_h(child2)
            new_population.extend([child1, child2])
            
        # Trim population to exact size
        population = new_population[:pop_size]
        
        # Adaptive mutation rate
        if generation % 100 == 0 and mutation_rate > 0.01:
            mutation_rate *= 0.98
    
    # Final quality check
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
