import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def dual_optimization_phase(initial_h, n_points, budget_s, dx):
    """Dual optimization approach to refine solution"""
    start_time = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Dual variables, one for each possible shift
    dual_vars = np.ones(2 * n_points - 1)
    learning_rate = 0.01
    max_iter = 10000
    
    for iter_num in range(max_iter):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Update h using a simplified gradient-like approach based on dual vars
        # Approximate gradient via finite differences in a limited way
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute finite difference gradient for h (simplified)
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update h
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        # Update dual variables based on constraint violations
        # This is a form of Lagrangian relaxation to approximate the dual update
        current_corr = correlate(h, 1 - h, mode="full")
        max_corr_idx = np.argmax(current_corr)
        max_val = current_corr[max_corr_idx]
        
        # Dual update (adjustive update based on current overlap)
        # Negative influence where overlap is high, positive where it's low
        if max_val > 0.5:  # heuristic to decide update magnitude
            # This adjusts dual variables to indirectly penalize overlap
            dual_vars -= 0.1 * np.ones_like(dual_vars) * (max_val - 0.5) / max_val
            
        # Clip dual vars (must be non-negative)
        dual_vars = np.maximum(dual_vars, 0)
        
        # Evaluate current solution
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
        # Adaptive learning rate
        if iter_num % 1000 == 0 and iter_num > 0:
            learning_rate *= 0.8
    
    return best_h, best_c5

def gradient_refinement_phase(initial_h, n_points, budget_s, dx):
    """Refine with gradient descent on promising regions"""
    start_time = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    # For this phase, we use a more sophisticated gradient estimator
    while time.time() - start_time < 0.2 * budget_s:
        # Compute gradient for h using finite difference
        eps = 1e-5
        grad = np.zeros(n_points)
        base_energy = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update h using a sign-based variant that maintains feasibility
        h -= learning_rate * np.sign(grad)
        
        # Project to constraints
        h = project_h(h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
                
        # Adaptive learning rate
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h

        # Project to ensure it satisfies sum condition
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Dual optimization phase
    best_h, best_c5 = dual_optimization_phase(h, n_points, budget_s, dx)
    
    # Gradient refinement phase
    final_h, final_c5 = gradient_refinement_phase(best_h, n_points, budget_s, dx)
    
    return (final_h, final_c5, n_points)
