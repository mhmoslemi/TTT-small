from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Try different optimization strategies
    strategies = [
        ("Nelder-Mead", "minimize", {"method": "Nelder-Mead"}),
        ("L-BFGS", "minimize", {"method": "L-BFGS-B"}),
        ("Powell", "minimize", {"method": "Powell"})
    ]
    
    # Global optimization with multiple strategies
    for strategy_name, func_name, options in strategies:
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Copy starting point
        h_current = h.copy()

        def fun(h_vec):
            return compute_c5(h_vec)

                h_plus = project_h(h_plus)
                grad[i] = (fun(h_plus) - c5_base) / eps

        try:

            if strategy_name == "L-BFGS":
                res = minimize(fun, h_current, method='L-BFGS-B', 
                             bounds=[(0,1) for _ in range(len(h_current))],
                             jac=jac, options={'maxiter': 100})
                h_candidate = project_h(res.x)

                    best_h = h_candidate.copy()

        best_h = h_star.copy()

        return best_h, best_c5
        
    final_h, final_c5 = local_search(best_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
