import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction with 70 points
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap

    def objective(h):
        return evaluate(h)

    # Initial solution
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    # Gradient-based optimization starting from the initial solution
    # Define bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a simple descent method with adaptive perturbations
    def perturb(h, delta=0.02):
        h_new = h.copy()
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        # Clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    # Optimization loop
    no_improvement_count = 0
    iterations = 0

    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        # Try multiple perturbations
        for _ in range(100):
            h_trial = perturb(best_h)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                break
        else:
            no_improvement_count += 1

    # Final smoothing
    best_h = gaussian_filter1d(best_h, sigma=0.5)

    # Renormalize
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final evaluation
    c5_bound = evaluate(best_h)
    
    return (best_h, c5_bound, n_points)
