import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def simulated_annealing_phase(initial_h, n_points, budget_s, dx):
    """Simulated Annealing for global exploration"""
    start_time = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    T = 1.0
    alpha = 0.999
    step_size = 0.1
    max_iter = int(1e5)
    
    for i in range(max_iter):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Create candidate solution
        h_new = h + np.random.uniform(-step_size, step_size, n_points)
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                
        # Cooling
        T *= alpha
        
    return best_h, best_c5

def gradient_descent_phase(initial_h, n_points, budget_s, dx):
    """Gradient descent for local refinement"""
    start_time = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < 0.2 * budget_s:
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        # This is actually approximating the gradient for the worst case
        
        # Use finite differences to compute gradient
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Simulated Annealing phase
    best_h, best_c5 = simulated_annealing_phase(h, n_points, budget_s, dx)
    
    # Gradient Descent phase
    final_h, final_c5 = gradient_descent_phase(best_h, n_points, budget_s, dx)
    
    return (final_h, final_c5, n_points)
