import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values).copy()
            return project_h(h, n_points)
    except:
        pass
    # Default initialization: symmetric betwen 0.5 and 1.5 with some variation
    x = np.linspace(0, 2, n_points)
    # Create a symmetric function with peaks around 0.5 and 1.5
    h = np.exp(-((x - 0.5)**2) / 0.1) + 0.5 * np.exp(-((x - 1.5)**2) / 0.1)
    h = project_h(h, n_points)
    return h

def pso_phase(h, n_points, time_budget_s):
    """Run Particle Swarm Optimization to find a good starting point."""
    start_time = time.time()
    swarm_size = 30
    max_iter = 10000
    cognitive_weight = 2.0
    social_weight = 2.0
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_fitness = []
    global_best = h.copy()
    global_best_fitness = evaluate_c5(h)
    
    for _ in range(swarm_size):
        particle = initialize_h(n_points)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        pb = particle.copy()
        pbf = evaluate_c5(pb)
        personal_best.append(pb)
        personal_best_fitness.append(pbf)
        if pbf < global_best_fitness:
            global_best = pb.copy()
            global_best_fitness = pbf
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.8 and iteration < max_iter:
        iteration += 1
        for i in range(swarm_size):
            # Update particle velocity
            r1, r2 = random.random(), random.random()
            inertia = inertia_weight * velocities[i]
            cognitive = cognitive_weight * r1 * (personal_best[i] - swarm[i])
            social = social_weight * r2 * (global_best - swarm[i])
            velocities[i] = inertia + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update particle position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to valid domain
            swarm[i] = project_h(swarm[i], n_points)
            
            # Evaluate fitness
            fitness = evaluate_c5(swarm[i])
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_fitness[i] = fitness
                if fitness < global_best_fitness:
                    global_best = swarm[i].copy()
                    global_best_fitness = fitness
                    
        # Update inertia weight
        inertia_weight = 0.9 - (0.5 * iteration / max_iter)
    
    return global_best, global_best_fitness

def gradient_descent_phase(h, n_points, time_budget_s):
    """Run gradient descent with projected updates."""
    start_time = time.time()
    max_iter = 50000
    learning_rate = 0.05
    tolerance = 1e-6
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Get the shift that corresponds to the maximum in the convolution
    def find_worst_shift(h):
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)
        return shift

    # Compute gradient for the maximum shift
    def compute_gradient(h, shift):
        n = len(h)
        dx = 2.0 / n
        h = np.array(h)  # Make sure we're working with a copy
        # For numerical stability, we compute gradient via finite difference
        eps = 1e-5
        grad = np.zeros(n)
        f0 = evaluate_c5(h)
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n)
            f_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n)
            f_minus = evaluate_c5(h_minus)
            
            grad[i] = (f_plus - f_minus) / (2 * eps)
            
        return grad

    iteration = 0
    prev_c5 = float('inf')
    no_improvement_count = 0
    while time.time() - start_time < time_budget_s * 0.2 and iteration < max_iter:
        iteration += 1
        
        # No need to check shift; compute gradient for best (for balancing)
        grad = compute_gradient(best_h, 0)
        
        # Gradient descent step
        h_new = best_h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Adapt learning rate based on improvement
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Check convergence
        if abs(prev_c5 - best_c5) < tolerance:
            break
        prev_c5 = best_c5
        
        if no_improvement_count > 1000:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45  # odd number for symmetry
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: Particle Swarm Optimization
    h_pso, c5_pso = pso_phase(h, n_points, budget_s)
    
    # Phase 2: Gradient Descent
    h_gd, c5_gd = gradient_descent_phase(h_pso, n_points, budget_s)
    
    # Return best result
    final_h = h_gd if c5_gd < c5_pso else h_pso
    final_c5 = min(c5_gd, c5_pso)
    
    # Ensure final result is valid
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
