import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create from segments
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a piecewise constant function with alternating ones and zeros
        h = np.zeros(n_points)
        segments = 5
        segment_size = n_points // segments
        for i in range(segments):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size if i < segments - 1 else n_points
            h[start_idx:end_idx] = 1.0 if i % 2 == 0 else 0.0
        h = h * (target_sum / np.sum(h))
    
    # Helper to compute C5
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # Annealing parameters
    T = 1.0
    alpha = 0.99
    max_iter = 50000
    min_T = 1e-6
    no_improvement_limit = 500
    no_improvement_count = 0
    
    # Track best solution
    best_h = h.copy()
    best_C = compute_C(best_h)
    
    # Simulated Annealing Loop
    iter_count = 0
    while iter_count < max_iter and time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Generate neighbor
        h_new = h.copy()
        # Randomly select one point to modify
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.1)
        h_new[idx] += delta
        
        # Enforce bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Compute C values
        C_old = compute_C(h)
        C_new = compute_C(h_new)
        
        # Accept or reject
        if C_new < C_old:
            h = h_new.copy()
            if C_new < best_C:
                best_h = h_new.copy()
                best_C = C_new
            no_improvement_count = 0
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(C_new - C_old) / T):
                h = h_new.copy()
                if C_new < best_C:
                    best_h = h_new.copy()
                    best_C = C_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Cooling schedule
        if no_improvement_count > no_improvement_limit:
            no_improvement_count = 0
            T = max(min_T, T * alpha)
    
    # Refinement with gradient descent
    # Simple gradient descent
    def gradient_descent_step(h_vals, lr=0.01, eps=1e-5):
        h = np.array(h_vals)
        C_old = compute_C(h)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = h_plus * (target_sum / np.sum(h_plus))  # Renormalize
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C_old) / eps
        return h - lr * grad

    # Refinement steps
    refinement_steps = 200
    for step in range(refinement_steps):
        if time() - start_time > budget_s - 1:
            break
        h = gradient_descent_step(best_h)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))  # Renormalize
        C = compute_C(h)
        if C < best_C:
            best_C = C
            best_h = h.copy()
    
    return (best_h.tolist(), best_C, n_points)
