import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function with reduced dimensional search and gradient refinement"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():

        h = project_h(h, n_points)
    else:

        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Reduced dimension approach: piecewise constant with 5 segments
    m = 5  # number of segments
    segment_values = [0.0, 0.5, 1.0]  # possible segment values
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Time tracking
    start_time = time.time()
    
    # Exhaustive search over segment combinations
    segment_combinations = list(product(segment_values, repeat=m))
    
    for combo in segment_combinations:
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Construct h from segments
        segment_lengths = np.ones(m) * (2.0 / m)
        h_candidate = np.zeros(n_points)
        
        for i in range(m):
            start_idx = int(i * n_points / m)
            end_idx = int((i + 1) * n_points / m)
            h_candidate[start_idx:end_idx] = combo[i]
            
        # Project to satisfy sum constraint
        h_candidate = project_h(h_candidate, n_points)
        
        # Evaluate C5
        c5_val = evaluate_h(h_candidate, dx)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_candidate.copy()
    
    # Gradient descent refinement of best candidates from exhaustive search
    candidate_count = min(10, len(segment_combinations))
    candidates = []
    
    # Select top candidates (based on lowest C5 from exhaustive search)
    # We'll compute C5 values for a subset of candidates
    candidate_c5_pairs = []
    
    for combo in segment_combinations[:20]:  # Limit search for efficiency
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Construct h from segments
        segment_lengths = np.ones(m) * (2.0 / m)
        h_candidate = np.zeros(n_points)
        
        for i in range(m):
            start_idx = int(i * n_points / m)
            end_idx = int((i + 1) * n_points / m)
            h_candidate[start_idx:end_idx] = combo[i]
        
        # Project to satisfy sum constraint
        h_candidate = project_h(h_candidate, n_points)
        
        # Evaluate C5
        c5_val = evaluate_h(h_candidate, dx)
        candidate_c5_pairs.append((h_candidate, c5_val))
    
    # Sort candidates by C5
    candidate_c5_pairs.sort(key=lambda x: x[1])
    top_candidates = candidate_c5_pairs[:min(5, len(candidate_c5_pairs))]
    
    # Gradient descent refinement
    for h_candidate, _ in top_candidates:
        if time.time() - start_time > budget_s:
            break
            
        h_refined = h_candidate.copy()
        current_c5 = evaluate_h(h_refined, dx)
        
        # Gradient descent with adaptive stepsize
        learning_rate = 0.01
        max_patience = 1000
        patience = 0
        for _ in range(10000):  # Max iterations per candidate
            if time.time() - start_time > budget_s:
                break
                
            # Compute gradient using finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_refined.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update step
            h_refined -= learning_rate * grad
            h_refined = project_h(h_refined, n_points)
            
            # Evaluate
            new_c5 = evaluate_h(h_refined, dx)
            if new_c5 < current_c5:
                current_c5 = new_c5
                patience = 0
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_refined.copy()
            else:
                patience += 1
                if patience > max_patience:
                    break
                if patience % 100 == 0:
                    learning_rate *= 0.5
    
    return (best_h, best_c5, n_points)
