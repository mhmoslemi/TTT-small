import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def dual_optimization_phase(initial_h, n_points, budget_s, dx):
    start_time = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Initialize dual variables
    lambda_vals = np.ones(2 * n_points - 1)
    step_size = 0.01
    max_iter = 10000
    patience = 0
    max_patience = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Solve primal problem with fixed dual variables
        # Approximate by gradient-like update using correlation
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - (n_points - 1)
        
        # Use a gradient descent step toward minimizing overlap
        # Compute gradient estimate
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update h
        h -= step_size * grad
        h = project_h(h, n_points)
        
        # Update dual variables based on current overlap
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            step_size *= 0.5
            patience = 0
    
    return best_h, best_c5
