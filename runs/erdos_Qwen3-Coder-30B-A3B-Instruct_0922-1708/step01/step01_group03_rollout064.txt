import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def line_search(h, i, shift_idx, max_shift_idx, n_points, dx):
    """Perform line search for optimal h[i] value."""
    # Need to evaluate how changing h[i] affects C5 at a specific shift
    # Start with small step to see the direction of change
    h_orig = h[i]
    c5_current = evaluate_c5(h)
    
    # Try small steps to determine direction
    step_sizes = np.array([-0.01, -0.005, 0.005, 0.01])
    step_direction = 0
    
    for step in step_sizes:
        h_test = h.copy()
        h_test[i] += step
        # Clamp to [0,1]
        h_test[i] = np.clip(h_test[i], 0, 1)
        # Re-scale to maintain sum constraint
        current_sum = np.sum(h_test)
        h_test = h_test * (n_points / 2) / current_sum
        c5_test = evaluate_c5(h_test)
        if c5_test < c5_current:
            step_direction = step
            break
    
    if step_direction == 0:
        return h_orig
    
    # Do actual line search using golden section search interval around current value
    def c5_func(alpha):
        h_test = h.copy()
        h_test[i] += alpha * step_direction
        # Clamp to [0,1]
        h_test[i] = np.clip(h_test[i], 0, 1)
        # Re-scale to maintain sum constraint
        current_sum = np.sum(h_test)
        h_test = h_test * (n_points / 2) / current_sum
        return evaluate_c5(h_test)
    
    # Golden ratio
    phi = (1 + np.sqrt(5)) / 2
    resphi = 2 - phi
    
    # Initial points
    a = -0.01
    b = 0.01
    if a > b:
        a, b = b, a
    
    # Evaluate at the golden ratio points
    x1 = a + resphi * (b - a)
    x2 = a + phi * (b - a)
    f1 = c5_func(x1)
    f2 = c5_func(x2)
    
    while abs(b - a) > 1e-6:
        if f1 < f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = a + resphi * (b - a)
            f1 = c5_func(x1)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + phi * (b - a)
            f2 = c5_func(x2)
    
    # Return the best point
    best_alpha = (a + b) / 2
    final_h = h.copy()
    final_h[i] += best_alpha * step_direction
    final_h[i] = np.clip(final_h[i], 0, 1)
    # Re-scale to maintain sum constraint
    current_sum = np.sum(final_h)
    final_h = final_h * (n_points / 2) / current_sum
    
    return final_h[i]

def coordinate_wise_update(h, n_points, dx, max_iter=5000):
    """Perform coordinate-wise optimization."""
    # Phase 1: coordinate-wise optimization
    h = h.copy()
    best_c5 = evaluate_c5(h)
    iter_no_improvement = 0
    
    for it in range(max_iter):
        if iter_no_improvement > 1000:
            break
        prev_c5 = best_c5
        improvement = False
        
        # Go through all coordinates
        for i in range(n_points):
            # Find a good step size to improve by adjusting only h[i]
            new_h_i = line_search(h, i, 0, 0, n_points, dx)
            old_h_i = h[i]
            h[i] = new_h_i
            new_c5 = evaluate_c5(h)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                improvement = True
            else:
                # restore
                h[i] = old_h_i
        
        if not improvement:
            iter_no_improvement += 1
        else:
            iter_no_improvement = 0
            
    return h, best_c5

def nonlinear_cg_update(h, n_points, dx, max_iter=5000):
    """Perform nonlinear conjugate gradient optimization."""
    h = h.copy()
    c5 = evaluate_c5(h)
    g = np.zeros(n_points)
    old_g = np.zeros(n_points)
    
    # Compute gradient (numerical)
    eps = 1e-5
    
    def compute_gradient(h):
        g = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            g[i] = (c5_plus - c5_minus) / (2 * eps)
        return g
    
    # initial direction
    direction = -compute_gradient(h)
    iter_no_improvement = 0
    for it in range(max_iter):
        if iter_no_improvement > 1000:
            break
        # line search for step size
        def obj(alpha):
            h_new = h + alpha * direction
            h_new = project_h(h_new, n_points)
            return evaluate_c5(h_new)
        
        # Simple backtracking line search
        alpha = 1.0
        min_alpha = 1e-7
        max_alpha = 1.0
        
        while alpha >= min_alpha:
            h_new = h + alpha * direction
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5:
                break
            alpha /= 2.0
        
        if alpha < min_alpha:
            iter_no_improvement += 1
        else:
            iter_no_improvement = 0
        
        if alpha < min_alpha:
            break
            
        old_h = h.copy()
        old_c5 = c5
        h = h + alpha * direction
        h = project_h(h, n_points)
        c5 = evaluate_c5(h)
        
        # Compute new gradient and update direction
        old_g = compute_gradient(old_h)
        g = compute_gradient(h)
        
        # Polak-Ribiere
        beta = max(0, np.dot(g, g - old_g) / np.dot(old_g, old_g))
        direction = -g + beta * direction
        
        if abs(old_c5 - c5) < 1e-6:
            iter_no_improvement += 1
        else:
            iter_no_improvement = 0
    
    return h, c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    time_budget = budget_s
    start_time = time.time()
    
    n_points = 45
    h = initialize_h(n_points)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Phase 1: Coordinate-wise optimization
    h_cw, c5_cw = coordinate_wise_update(h, n_points, dx, max_iter=5000)
    if c5_cw < best_c5:
        best_h = h_cw
        best_c5 = c5_cw
    
    # Phase 2: Nonlinear conjugate gradient
    if time.time() - start_time < time_budget * 0.9:
        h_cg, c5_cg = nonlinear_cg_update(h_cw, n_points, dx, max_iter=5000)
        if c5_cg < best_c5:
            best_h = h_cg
            best_c5 = c5_cg

    # Final projection
    best_h = project_h(best_h, n_points)
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
