    # Objective function to minimize
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    # Optimization loop (Gradient-based with adaptive learning rate)
    best_h = h.copy()
    best_C = float('inf')
    no_improvement_count = 0
    lr = 0.01
    
    def evaluate_and_update():
        nonlocal best_C, best_h
        C = compute_C(h)
        if C < best_C:
            best_C = C
            best_h = h.copy()
        return C
    
    # Evaluate initial solution
    previous_C = evaluate_and_update()
    
    iter_count = 0
    while time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute the current overlap values to find worst case shift
        h_vals = h.copy()
        C = compute_C(h_vals)
        
        # Use finite differences to compute gradient
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C) / eps
            
        # Update with gradient descent
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))  # Rescale to integral constraint

        new_C = evaluate_and_update()

            no_improvement_count += 1

                no_improvement_count = 0
        else:
            no_improvement_count = 0
        previous_C = new_C
