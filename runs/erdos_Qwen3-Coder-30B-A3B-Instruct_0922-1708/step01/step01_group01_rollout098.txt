import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():

        h = project_h(h, n_points)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Dual Optimization
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    lambda_k = np.ones(2 * n_points - 1)
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < 0.8 * budget_s:
        iteration += 1
        
        # Solve primal for fixed lambda
        # This approximates minimizing sum_k lambda_k * (1 - h[i] + h[i] * h[(i+k)%n_points])
        # But we are actually using a simpler heuristic approach for coding simplicity
        
        # Simple greedy refinement: perturb in direction of descent
        # We will directly try local search on small perturbations
        
        # Check the worst overlap
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Try small perturbations
        perturbed_h = h.copy()
        step = 0.01
        found_better = False
        
        for i in range(n_points):
            for sign in [-1, 1]:
                h_test = h.copy()
                h_test[i] += step * sign
                h_test = project_h(h_test, n_points)
                c_test = evaluate_h(h_test, dx)
                if c_test < best_c5:
                    found_better = True
                    perturbed_h = h_test.copy()
                    best_c5 = c_test
                    best_h = h_test.copy()
        
        if not found_better:
            # Random search component to escape local minima
            h_rand = np.random.rand(n_points)
            h_rand = project_h(h_rand, n_points)
            c_rand = evaluate_h(h_rand, dx)
            if c_rand < best_c5:
                best_c5 = c_rand
                best_h = h_rand.copy()
        
        # Use the best candidate for next iteration
        h = best_h.copy()
    
    # Final gradient-based refinement
    max_iter = 10000
    lr = 0.01
    patience = 0
    max_patience = 5000 
    for i in range(max_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        eps = 1e-5
        grad = np.zeros(n_points)
        c_base = evaluate_h(h, dx)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps 
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[j] = (c_plus - c_minus) / (2 * eps)
            
        h -= lr * grad

            best_h = h.copy()
            patience = 0

            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            lr *= 0.5
            patience = 0

    return (best_h, best_c5, n_points)
