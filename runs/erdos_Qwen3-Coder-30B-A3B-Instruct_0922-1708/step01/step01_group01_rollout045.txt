import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Dual optimization approach with local refinement
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    dual_vars = np.ones(2 * n_points - 1)  # For all possible shifts
    learning_rate = 0.01
    patience = 0
    max_patience = 500
    
    while time.time() - start_time < 0.8 * budget_s:
        # Update h by solving a modified optimization problem
        # in each iteration, compute current correlations and adjust multipliers
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Add a small perturbation to the dual variable associated with the worst overlap
        # This pushes the solution away from that configuration
        if 0 <= max_shift_idx < len(dual_vars):
            dual_vars[max_shift_idx] += 0.01
            
        # Solve a modified problem: adjust h so as to reduce the maximum
        # We'll use a simple gradient descent on the correlation matrix
        # Compute gradient exactly for a modified loss function
        h_new = h.copy()
        for i in range(n_points):
            eps = 1e-5
            # Perturb h[i] up and down
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)

            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)

            grad_i = (c_plus - c_minus) / (2 * eps)
            h_new[i] -= learning_rate * grad_i
            
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_h(h_new, dx)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
                
        if patience > 0 and patience % 100 == 0:
            learning_rate *= 0.95
            
    # Final refinement with simple gradient descent to clean up the solution
    # If there's still time left
    while time.time() - start_time < budget_s - 1:
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        h_new = h.copy()
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad_i = (c_plus - c_minus) / (2 * eps)
            h_new[i] -= 0.01 * grad_i
            
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_h(h_new, dx)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
        else:
            break
            
    return (h, best_c5, n_points)
