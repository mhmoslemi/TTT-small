import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO + Gradient Descent"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO Parameters
    pop_size = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    adapt_interval = 500
    min_swarm_size = 20
    stagnation_count = 0
    stagnation_limit = 500
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # 10% random initializations
    for _ in range(pop_size // 5):
        h_new = np.random.rand(n_points)
        h_new = project_h(h_new, n_points)
        swarm.append(h_new.copy())
        velocities.append(np.zeros(n_points))
    
    # 90% from perturbed initial solution
    for _ in range(4 * pop_size // 5):
        h_new = h + np.random.normal(0, 0.05, n_points)
        h_new = project_h(h_new, n_points)
        swarm.append(h_new.copy())
        velocities.append(np.zeros(n_points))
    
    # Evaluate initial swarm
    fitness = [evaluate_h(particle, dx) for particle in swarm]
    
    # Track best
    best_idx = np.argmin(fitness)
    gbest = swarm[best_idx].copy()
    gbest_fitness = fitness[best_idx]
    
    # Track personal bests
    pbest = [particle.copy() for particle in swarm]
    pbest_fitness = fitness.copy()
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < 0.7 * budget_s and iteration < max_iter:
        iteration += 1
        
        for i in range(len(swarm)):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] = project_h(swarm[i] + velocities[i], n_points)
            
            # Evaluate
            new_fitness = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < gbest_fitness:
                    gbest = swarm[i].copy()
                    gbest_fitness = new_fitness
                    stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Adapt swarm size if no improvement
        if stagnation_count >= adapt_interval and len(swarm) > min_swarm_size:
            num_removed = len(swarm) // 5
            removal_indices = np.argsort(pbest_fitness)[-num_removed:]
            for idx in sorted(removal_indices, reverse=True):
                del swarm[idx]
                del velocities[idx]
                del pbest[idx]
                del pbest_fitness[idx]
            stagnation_count = 0
            
            # Reinitialize some particles
            for _ in range(num_removed):
                h_new = np.random.rand(n_points)
                h_new = project_h(h_new, n_points)
                swarm.append(h_new)
                velocities.append(np.zeros(n_points))
                pbest.append(h_new)
                pbest_fitness.append(evaluate_h(h_new, dx))
    
    # Gradient descent phase on gbest
    h = gbest.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s - 1.0:
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute finite difference gradient for worst overlap shift
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h = h - learning_rate * np.sign(grad)
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > 1000:
                learning_rate *= 0.5
                patience = 0
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (best_h, best_c5, n_points)
