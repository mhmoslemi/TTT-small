import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def segment_averages(c, n_points):
        # Reconstruct h from segment averages
        segment_size = n_points // len(c)
        h = np.repeat(c, segment_size)
        if len(h) < n_points:
            h = np.pad(h, (0, n_points - len(h)), mode='constant', constant_values=c[-1])
        return h
    
    def smooth_h(h, sigma=1.0):
        # Apply Gaussian smoothing
        kernel = np.exp(-np.linspace(-3, 3, 21)**2 / (2 * sigma**2))
        kernel = kernel / np.sum(kernel)
        return np.convolve(h, kernel, mode='same')
    
    def optimize_segments(m):
        # Initialize segment averages
        c_init = np.ones(m) * 0.5
        c_init[0] = 0.2
        c_init[-1] = 0.8
        
        def objective(c):
            h = segment_averages(c, n_points)
            h = smooth_h(h)
            return evaluate(h)
            
        def constraint_eq(c):
            return np.sum(c) - m / 2
            
        bounds = [(0, 1) for _ in range(m)]
        cons = {'type': 'eq', 'fun': constraint_eq}
        
        try:
            # Minimize with bounds and constraint
            result = minimize(objective, c_init, method='SLSQP', bounds=bounds, constraints=cons, 
                            options={'maxiter': 500, 'ftol': 1e-6})
            if result.success:
                return result.x
        except:
            pass
        return c_init
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    # Hierarchical optimization with increasing resolution
    m_values = [7, 14, 21, 28, 35, 42, 49, 56, 63]
    
    for m in m_values:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize segment averages
        c_opt = optimize_segments(m)
        
        # Reconstruct high-resolution h
        h_reconstructed = segment_averages(c_opt, n_points)
        h_reconstructed = smooth_h(h_reconstructed)
        h_reconstructed = np.clip(h_reconstructed, 0, 1)
        h_reconstructed = h_reconstructed * (n_points / 2) / np.sum(h_reconstructed)
        
        # Evaluate
        c5_trial = evaluate(h_reconstructed)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_reconstructed.copy()
    
    # Refinement with local search
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s and no_improvement_count < max_no_improvement:
        # Perturb best solution
        h_trial = best_h.copy()
        indices = np.random.choice(n_points, max(1, n_points // 10), replace=False)
        for i in indices:
            h_trial[i] += np.random.normal(0, 0.02)
        
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        c5_trial = evaluate(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
    
    return (best_h, best_c5, n_points)
