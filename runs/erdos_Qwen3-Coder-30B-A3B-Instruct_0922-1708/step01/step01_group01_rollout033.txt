import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    stagnation_limit = 500
    min_swarm_size = 20
    adaptive_decrease = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    # Generate initial particles
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            # 10% random, 90% perturbed
            if np.random.rand() < 0.1:
                particle = np.random.rand(n_points)
            else:
                particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_h(particle, n_points)
            
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        score = evaluate_h(particle, dx)
        pbest.append(particle.copy())
        pbest_scores.append(score)
    
    # Global best tracking
    gb_idx = np.argmin(pbest_scores)
    gbest = pbest[gb_idx].copy()
    gbest_score = pbest_scores[gb_idx]
    
    start_time = time.time()
    it = 0
    last_improvement = 0
    
    while time.time() - start_time < 0.7 * budget_s and it < max_iter:
        it += 1
        
        # Update particles
        for i in range(len(swarm)):
            particle = swarm[i]
            vel = velocities[i]
            px = pbest[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (w * vel + 
                   c1 * r1 * (px - particle) + 
                   c2 * r2 * (gbest - particle))
            
            # Update position
            particle = particle + vel
            particle = project_h(particle, n_points)
            
            # Evaluate
            score = evaluate_h(particle, dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particle.copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = particle.copy()
                    gbest_score = score
                    last_improvement = it
            
            # Store for next iteration
            swarm[i] = particle
            velocities[i] = vel
        
        # Adaptive swarm size adjustment
        if it - last_improvement > stagnation_limit:
            n_particles = max(min_swarm_size, int(n_particles * (1 - adaptive_decrease)))
            # Reinitialize particles if needed
            if len(swarm) > n_particles:
                swarm = swarm[:n_particles]
                velocities = velocities[:n_particles]
                pbest = pbest[:n_particles]
                pbest_scores = pbest_scores[:n_particles]
            else:
                # Add new particles
                for i in range(n_particles - len(swarm)):
                    particle = np.random.rand(n_points) if np.random.rand() < 0.1 else h + np.random.normal(0, 0.05, n_points)
                    particle = project_h(particle, n_points)
                    swarm.append(particle)
                    velocities.append(np.zeros(n_points))
                    score = evaluate_h(particle, dx)
                    pbest.append(particle.copy())
                    pbest_scores.append(score)
                    if score < gbest_score:
                        gbest = particle.copy()
                        gbest_score = score
                        last_improvement = it
        
        # Periodic improvement check
        if it % 100 == 0:
            gb_idx = np.argmin(pbest_scores)
            current_gb_score = pbest_scores[gb_idx]
            if current_gb_score < gbest_score:
                gbest = pbest[gb_idx].copy()
                gbest_score = current_gb_score
                
    # Gradient descent refinement
    final_h = gbest.copy()
    learning_rate = 0.01
    max_gd_iter = 5000
    patience = 0
    max_patience = 1000
    
    start_time_gd = time.time()
    for i in range(max_gd_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Compute overlap for all shifts and find worst one
        corr = correlate(final_h, 1 - final_h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute finite difference gradient for each coordinate
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for j in range(n_points):
            h_plus = final_h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = final_h.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[j] = (c_plus - c_minus) / (2 * eps)
        
        # Update using gradient
        updated_h = final_h - learning_rate * grad
        final_h = project_h(updated_h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(final_h, dx)
        if new_c5 < gbest_score:
            gbest = final_h.copy()
            gbest_score = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (final_h, gbest_score, n_points)
