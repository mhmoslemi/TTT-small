import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sample(n_points, n_samples):
    """Generate Latin Hypercube Samples"""
    samples = []
    for _ in range(n_samples):
        h = np.random.rand(n_points)
        # Apply LHS-like perturbation to avoid uniform distribution
        for i in range(n_points):
            h[i] = (i + np.random.rand()) / n_points
        samples.append(h)
    return samples

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Phase 1: LHS sampling
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    lhs_samples = lhs_sample(n_points, 50)
    for s in lhs_samples:
        s = project_h(s, n_points)
        c5 = evaluate_h(s, dx)
        if c5 < best_c5:
            best_h = s.copy()
            best_c5 = c5
    
    # Phase 2: Evolutionary algorithm (simplified version of genetic algo)
    pop_size = 25
    generations = 0
    max_generations = 1000
    
    population_h = [best_h.copy()]
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = project_h(h_rand, n_points)
        population_h.append(h_rand)
    
    while time.time() - start_time < 0.7 * budget_s and generations < max_generations:
        generations += 1
        fitness = [evaluate_h(h, dx) for h in population_h]
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population_h[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
        
        # Selection and reproduction
        sorted_indices = np.argsort(fitness)
        elites = [population_h[i] for i in sorted_indices[:pop_size // 2]]
        
        offspring = []
        for _ in range(pop_size - len(elites)):
            parent = elites[np.random.randint(0, len(elites))]
            child = parent.copy()
            # Mutation
            mut_idx = np.random.randint(0, n_points)
            child[mut_idx] = np.random.rand()
            child = project_h(child, n_points)
            offspring.append(child)
        
        population_h = elites + offspring
    
    # Phase 3: Gradient-based refinement
    h = best_h.copy()
    current_c5 = best_c5
    learning_rate = 0.01
    patience = 0
    max_patience = 2000
    
    while time.time() - start_time < 0.95 * budget_s:
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Approximate gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        old_h = h.copy()
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        new_c5 = evaluate_h(h, dx)
        if new_c5 < current_c5:
            current_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 500 == 0:
            learning_rate *= 0.5
            patience = 0
    
    # Ensure final h is valid
    h = project_h(h, n_points)
    best_c5 = evaluate_h(h, dx)
    
    return (h, best_c5, n_points)
