import numpy as np
from scipy import optimize
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric function
    x = np.linspace(0, 2, n_points)
    # Create symmetric pulse centered at 1
    h = 1.0 * (x >= 0.5) * (x <= 1.5)
    h = project_h(h, n_points)
    return h

def pso_phase(h, n_points, time_budget_s):
    """Run Particle Swarm Optimization to find a good starting point."""
    start_time = time.time()
    
    # Parameters
    swarm_size = 30
    max_iterations = 10000
    cognitive_weight = 2.0
    social_weight = 2.0
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Initialize swarm
    positions = []
    velocities = []
    personal_best_positions = []
    personal_best_scores = []
    global_best_position = None
    global_best_score = float('inf')
    
    for _ in range(swarm_size):
        pos = h.copy()
        pos += np.random.normal(0, 0.05, n_points)
        pos = project_h(pos, n_points)
        positions.append(pos)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        score = evaluate_c5(pos)
        personal_best_positions.append(pos.copy())
        personal_best_scores.append(score)
        if score < global_best_score:
            global_best_score = score
            global_best_position = pos.copy()
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.6 and iteration < max_iterations:
        iteration += 1
        inertia_weight = 0.9 - (0.5 * iteration / max_iterations)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive_component = cognitive_weight * r1 * (personal_best_positions[i] - positions[i])
            social_component = social_weight * r2 * (global_best_position - positions[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive_component + social_component
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            positions[i] += velocities[i]
            positions[i] = project_h(positions[i], n_points)
            
            # Evaluate position
            score = evaluate_c5(positions[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = positions[i].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best_position = positions[i].copy()
    
    return global_best_position, global_best_score

def gradient_descent_phase(h, n_points, time_budget_s):
    """Run gradient descent with projected updates."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.05
    max_iterations = 10000
    iter_no_improvement = 0
    max_no_improvement = 1000
    
    # Compute which shift gives the maximum overlap
    def find_worst_shift(h):
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)
        return shift

    iteration = 0
    while time.time() - start_time < time_budget_s * 0.4 and iteration < max_iterations:
        # Find shift that causes max overlap
        shift = find_worst_shift(best_h)
        
        # Compute gradient analytically
        n = len(best_h)
        dx = 2.0 / n
        grad = np.zeros(n)
        for i in range(n):
            # Approximate gradient for shift handling
            # Manually compute gradient for C5 computed at shift
            # Direct computational approach
            g = 0
            for k in range(n):
                idx = (i + k) % n
                g += (1 - best_h[idx]) if k == -shift else 0
            grad[i] = g * dx
        
        # Update with gradient
        h_new = best_h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
        
        if iter_no_improvement > max_no_improvement:
            break
        iteration += 1
        
        # Adjust learning rate
        if iteration % 100 == 0:
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45  # Odd number for symmetry
    
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: Particle Swarm Optimization
    h_pso, c5_pso = pso_phase(h, n_points, budget_s)
    
    # Phase 2: Gradient Descent
    h_gd, c5_gd = gradient_descent_phase(h_pso, n_points, budget_s)
    
    # Return best result
    final_h = h_gd if c5_gd < c5_pso else h_pso
    final_c5 = min(c5_gd, c5_pso)
    
    # Ensure final result is valid
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
