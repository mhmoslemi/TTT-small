import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # For better optimization, we'll use a more sophisticated approach
    # Let's try coordinate descent with adaptive learning rates
    
    # Prepare for coordinate descent
    convergence_threshold = 1e-4
    last_improvement = 0
    patience = 0
    max_patience = 100
    
    # Use a more aggressive learning rate schedule
    learning_rates = [0.05, 0.02, 0.01, 0.005]
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = 0
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Adaptive restart with new random direction
                h = np.random.uniform(0.1, 0.9, n_points)
                h = h * (n_points / 2) / np.sum(h)
                patience = 0
                
        # Gradient approximation using finite differences per component
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # Sample several directions for gradient estimation
        sample_points = min(10, n_points)
        sample_indices = np.random.choice(n_points, sample_points, replace=False)
        
        for i in sample_indices:
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Apply gradient descent with adaptive learning rate
        lr_idx = (iteration // 100) % len(learning_rates)
        lr = learning_rates[lr_idx]
        
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Check for convergence
        last_improvement += 1
        if last_improvement > 1000 and iteration > 500:
            if abs(c5 - best_c5) / (abs(best_c5) + 1e-8) < convergence_threshold:
                break
                
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
