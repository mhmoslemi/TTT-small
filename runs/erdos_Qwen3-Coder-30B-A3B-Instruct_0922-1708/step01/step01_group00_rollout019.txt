import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        # We treat this as max of correlations
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Adaptive genetic algorithm
        if iteration % 5 == 0:
            # Create offspring via crossover
            pop_size = 50
            population = [h.copy()]
            
            # Generate diverse population
            for _ in range(pop_size - 1):
                # Add variation
                if np.random.random() < 0.5:
                    # Artificially mutate from current solution
                    h_mut = h.copy()
                    idx = np.random.randint(0, n_points)
                    h_mut[idx] = np.clip(h_mut[idx] + np.random.normal(0, 0.1), 0, 1)
                    h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
                    population.append(h_mut)
                else:
                    # Random uniform distribution
                    h_rand = np.random.uniform(0, 1, n_points)
                    h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
                    population.append(h_rand)
                
            # Evaluate fitness of population
            fitnesses = []
            for h_cand in population:
                corr = np.correlate(h_cand, 1 - h_cand, mode='full')
                fitness = np.max(corr) * dx
                fitnesses.append(fitness)
            
            # Select best individuals for next generation
            sorted_indices = np.argsort(fitnesses)
            selected = [population[i] for i in sorted_indices[:pop_size//2]]
            
            # Crossover
            new_population = selected.copy()
            for i in range(pop_size//2):
                parent1 = selected[i]
                parent2 = selected[np.random.randint(0, len(selected))]
                # Blend crossover
                alpha = np.random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
                # Add small mutation
                if np.random.random() < 0.1:
                    idx = np.random.randint(0, n_points)
                    child[idx] = np.clip(child[idx] + np.random.normal(0, 0.05), 0, 1)
                # Rescale
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            
            # Keep best solution and some variation
            h = new_population[0]
            
        else:
            # Regular gradient descent with regularization
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Regularization to maintain feasible region
            lr = 0.01 + 0.001 / (1 + iteration/100)
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
        
        # Check early termination for plateau
        if iteration - last_improvement > 300:
            break
    
    return (best_h.tolist(), best_c5, n_points)
