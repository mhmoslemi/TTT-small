import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    inertia_decrease = 0.99
    swarm_size_min = 20
    stagnation_limit = 500
    patience_limit = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # 10% random particles
    for _ in range(n_particles // 10):
        h_rand = np.random.rand(n_points)
        h_rand = project_h(h_rand, n_points)
        particles.append(h_rand.copy())
        velocities.append(np.zeros(n_points))
    
    # 90% from initial solution with small perturbations
    for _ in range(9 * n_particles // 10):
        h_pert = h + np.random.normal(0, 0.05, n_points)
        h_pert = project_h(h_pert, n_points)
        particles.append(h_pert.copy())
        velocities.append(np.zeros(n_points))
    
    # Evaluate all particles
    fitness = [evaluate_h(p, dx) for p in particles]
    
    # Find best particle
    best_idx = np.argmin(fitness)
    gbest = particles[best_idx].copy()
    gbest_fitness = fitness[best_idx]
    
    # Personal bests
    pbest = [p.copy() for p in particles]
    pbest_fitness = fitness.copy()
    
    # PSO iteration
    start_time = time.time()
    iter_count = 0
    stagnation = 0
    best_found = gbest_fitness
    
    while iter_count < max_iter and time.time() - start_time < 0.8 * budget_s:
        iter_count += 1
        if stagnation > stagnation_limit and len(particles) > swarm_size_min:
            # Reduce swarm size
            particles = particles[:len(particles) // 10]
            velocities = velocities[:len(velocities) // 10]
            pbest = pbest[:len(pbest) // 10]
            pbest_fitness = pbest_fitness[:len(pbest_fitness) // 10]
            stagnation = 0
            
        # Update particles
        for i in range(len(particles)):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) +
                             c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = project_h(particles[i] + velocities[i], n_points)
            
            # Evaluate
            new_fitness = evaluate_h(particles[i], dx)
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = new_fitness
                    stagnation = 0
                    if gbest_fitness < best_found - 1e-5:
                        best_found = gbest_fitness
                        stagnation = 0
                        
        stagnation += 1
        w *= inertia_decrease
    
    # Gradient descent refinement
    h = gbest.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < 0.9 * budget_s:
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient w.r.t. h using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= learning_rate * np.sign(grad)
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < gbest_fitness:
            gbest = h.copy()
            gbest_fitness = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (gbest, gbest_fitness, n_points)
