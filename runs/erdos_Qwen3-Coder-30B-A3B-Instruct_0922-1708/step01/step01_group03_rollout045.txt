import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangular function
    x = np.linspace(0, 2, n_points)
    h = np.maximum(0, 1 - np.abs(x - 1))
    h = project_h(h, n_points)
    return h

def compute_gradient(h, k, n_points):
    """Compute gradient for shift k."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        grad[i] = (1 - h[(i + k) % n_points])
    return grad

def line_search(h, grad, max_k, n_points, dx):
    """Find optimal step size using line search."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        # Recompute convolution for the same shift
        conv = np.correlate(h_new, 1 - h_new, mode="full")
        return np.max(conv) * dx
    
    # Bracketing
    a, b = 0.0, 1.0
    fa = objective(0.0)
    fb = objective(1.0)
    
    # If not a descent direction, try smaller steps
    if fb >= fa:
        # Try to find a smaller step size
        alpha = 0.1
        while alpha > 1e-10:
            fa = objective(alpha)
            if fa < fa:
                break
            alpha *= 0.5
        return alpha
    
    # Binary search or quadratic fit
    try:
        result = minimize_scalar(objective, bounds=(0, 1), method='bounded')
        return max(0, min(1, result.x))
    except:
        return 0.05

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Set parameters for optimization
    eta = 0.1
    eta_min = 1e-6
    eta_max = 0.5
    patience = 0
    patience_limit = 1000
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    iterations = 0
    while time.time() - start_time < budget_s and patience < patience_limit:
        # Compute correlations to find worst shift
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        k = max_idx - (n_points - 1)  # Convert from full convolution index to shift
        
        # Compute gradient at this shift
        grad = compute_gradient(h, k, n_points)
        
        # Line search to find optimal step size
        alpha = line_search(h, grad, k, n_points, 2.0/n_points)
        
        # Update
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            patience = 0
            eta = min(eta_max, eta * 1.2)
        else:
            patience += 1
            eta = max(eta_min, eta / 2)
            
        h = h_new
        iterations += 1
    
    return (best_h, best_c5, n_points)
