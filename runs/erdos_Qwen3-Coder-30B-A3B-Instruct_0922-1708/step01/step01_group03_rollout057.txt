import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric step function
    h = np.zeros(n_points)
    mid = n_points // 2
    # Create a more structured initial function with characteristic shape
    for i in range(n_points):
        x = i * 2.0 / n_points
        # Triangular region with peak at x=1
        if x <= 0.5:
            h[i] = 2 * x
        elif x <= 1.5:
            h[i] = 2 - 2 * x
        else:
            h[i] = 2 * (x - 2)
    h = project_h(h, n_points)
    return h

def pso_phase(h, n_points, time_budget_s):
    """Run particle swarm optimization."""
    start_time = time.time()
    swarm_size = 30
    cognitive_weight = 2.0
    social_weight = 2.0
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Initialize swarm
    particles = [h.copy() for _ in range(swarm_size)]
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_c5(p) for p in particles]
    
    # Global best
    best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[best_idx].copy()
    global_best_score = personal_best_scores[best_idx]
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.7:
        iteration += 1
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (
                inertia_weight * velocities[i] +
                cognitive_weight * r1 * (personal_best[i] - particles[i]) +
                social_weight * r2 * (global_best - particles[i])
            )
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update particle
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_h(particles[i], n_points)
            
            # Evaluate particle
            score = evaluate_c5(particles[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = particles[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best = particles[i].copy()
        
        # Decrease inertia weight over time
        inertia_weight = 0.9 - (0.5 * iteration / 1000)
        if inertia_weight < 0.4:
            inertia_weight = 0.4
    
    return global_best, global_best_score

def gradient_descent_phase(h, n_points, time_budget_s):
    """Run projected gradient descent."""
    start_time = time.time()
    
    # Function to compute gradients for C5
    def compute_gradient(h):
        n = len(h)
        dx = 2.0 / n
        # Find the shift that gives the maximum correlation
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)
        
        # Compute gradient with respect to given shift
        grad = np.zeros(n)
        for i in range(n):
            grad[i] = (1 - h[(i - shift) % n]) * dx
        return grad
    
    # Gradient descent
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.05
    iteration = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < time_budget_s * 0.3 and iteration < max_no_improvement:
        grad = compute_gradient(best_h)
        h_new = best_h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iteration = 0
        else:
            iteration += 1
            # Reduce learning rate if no improvement
            if iteration % 100 == 0:
                learning_rate *= 0.9
                if learning_rate < 1e-6:
                    break
                    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 45  # odd number to enable symmetric constraint
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: Particle Swarm Optimization
    h_pso, c5_pso = pso_phase(h, n_points, budget_s)
    
    # Phase 2: Gradient Descent
    h_gd, c5_gd = gradient_descent_phase(h_pso, n_points, budget_s)
    
    # Return best result
    final_h = h_gd if c5_gd < c5_pso else h_pso
    final_c5 = min(c5_gd, c5_pso)
    
    # Ensure final result is valid
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
