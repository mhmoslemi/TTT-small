    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Use simulated annealing with adaptive cooling
    def simulated_annealing():
        current_h = h.copy()
        current_c5 = compute_c5(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        
        temp = 0.1
        min_temp = 1e-6
        cooling_rate = 0.995
        max_iter = 1000

            if (time.time() - start_time) > budget_s - 1:
                break

            idx = np.random.randint(0, n_points)
            old_val = current_h[idx]

            delta = np.random.normal(0, 0.05)
            new_val = old_val + delta

            new_val = np.clip(new_val, 0, 1)
            current_h[idx] = new_val
            
            # Project to sum constraint
            current_h = project_h(current_h)
            
            new_c5 = compute_c5(current_h)
            
            # Metropolis criterion
            if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / temp):
                current_c5 = new_c5
            else:
                current_h[idx] = old_val
                
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()

        def objective(h_vec):
            return compute_c5(h_vec)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
        
        # Early termination check

                final_h,
                method='L-BFGS-B',

            )
            if result.success:

                final_c5 = compute_c5(final_h)
                if final_c5 < best_c5:
                    best_h = final_h.copy()
                    best_c5 = final_c5
        except:
            pass
    
    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
