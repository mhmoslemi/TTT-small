import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or create a simple one
    if 'initial_h_values' in globals():

        h = project_h(h, n_points)
    else:

        h = np.ones(n_points) * 0.5
        # Set some intervals to 0 and 1 to make it a step function
        h[:15] = 0
        h[15:30] = 1
        h[30:] = 0
        h = project_h(h, n_points)
    
    # Reduced dimension search approach
    # Enumerate possible piecewise constant functions with small number of segments
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Try different configurations with fewer segments - for example 3 segments
    start_time = time.time()
    segment_count = 3
    possible_values = [0.0, 0.5, 1.0]
    
    # Generate all combinations of segment values
    combinations = list(product(possible_values, repeat=segment_count))
    
    # Try several combinations
    for combo in combinations[:]:
        # Create step function with this combo
        segments = []
        segment_length = n_points // segment_count
        for i, value in enumerate(combo):
            # Distribute segments evenly
            start_idx = i * segment_length
            end_idx = (i+1) * segment_length if i < segment_count - 1 else n_points
            segments.extend([value] * (end_idx - start_idx))
        
        # Ensure proper length
        segments = segments[:n_points]
        if len(segments) < n_points:
            segments.extend([combo[-1]] * (n_points - len(segments)))
            
        h_test = np.array(segments)
        h_test = project_h(h_test, n_points)
        c5_test = evaluate_h(h_test, dx)
        
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
            
    # Use the best out of the combinatorial search for gradient refinement
    h = best_h.copy()
    
    # Gradient descent refinement - for a few iterations
    learning_rate = 0.01
    patience = 0
    max_patience = 1000
    refine_start = time.time()
    while time.time() - refine_start < 100 and time.time() - start_time < 0.8 * budget_s:
        # Compute overlaps and find worst shift
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Estimate gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:

        if patience > 0 and patience % 500 == 0:
            learning_rate *= 0.5
            
    return (best_h, best_c5, n_points)
