import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth_h(h, sigma=1.0):
        # Apply Gaussian smoothing
        if len(h) < 3:
            return h
        kernel = np.exp(-0.5 * np.arange(-3*sigma, 3*sigma+1)**2 / sigma**2)
        kernel = kernel / np.sum(kernel)
        return np.convolve(h, kernel, mode='same')
    
    def reconstruct_h(c, n_segments, n_points):
        # Reconstruct h from segment averages
        h = np.repeat(c, n_points // n_segments)
        if len(h) < n_points:
            h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=h[-1])
        elif len(h) > n_points:
            h = h[:n_points]
        return smooth_h(h)
    
    def optimize_segments(c, n_segments, n_points):
        # Objective function
        def obj(c):
            h = reconstruct_h(c, n_segments, n_points)
            return evaluate(h)
        
        # Constraint
        def constraint_sum(c):
            return np.sum(c) - n_segments / 2
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_segments)]
        
        # Minimize
        res = minimize(obj, c, method='SLSQP', bounds=bounds, 
                       constraints={'type': 'eq', 'fun': constraint_sum},
                       options={'ftol': 1e-6, 'disp': False})
        
        return res.x if res.success else c
    
    # Start optimization from coarse to fine
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    no_improvement_count = 0
    
    # Multi-resolution approach
    resolution_levels = [7, 14, 21, 28, 35, 42, 49, 56, 63, 70]
    
    # Keep track of best solution found
    best_solution = (best_h.copy(), best_c5, n_points)
    
    for n_segments in resolution_levels:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Initialize segment averages
        if n_segments <= len(h_start):
            c = np.mean(h_start.reshape(n_segments, -1), axis=1)
        else:
            # Interpolate
            c = np.linspace(0.2, 0.8, n_segments)
        
        # Optimize
        c_opt = optimize_segments(c, n_segments, n_points)
        
        # Reconstruct h
        h_opt = reconstruct_h(c_opt, n_segments, n_points)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        # Evaluate
        c5_new = evaluate(h_opt)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_opt.copy()
            best_solution = (best_h.copy(), best_c5, n_points)
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= 100:
            break
            
        # Increase resolution by refining
        if n_segments == 70:  # last level
            break
                
    return best_solution
