import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using hybrid Simulated Annealing and Gradient Descent"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h from provided construction or use sinusoidal
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    step = 0.1
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    start_time = time.time()
    
    # Simulated Annealing Phase
    for i in range(max_iter):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Mutation: perturb random indices
        h_new = h.copy()
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for idx in indices:
            h_new[idx] += np.random.uniform(-step, step)
        
        # Project to feasible region
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        # Cool down
        T *= alpha
    
    # Gradient Descent Phase
    h = best_h.copy()
    lr = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s - 1:
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[j] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= lr * np.sign(grad)
        h = project_h(h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(h, dx)
        
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
            if patience % 1000 == 0:
                lr *= 0.5
                
    return (best_h, best_c5, n_points)
