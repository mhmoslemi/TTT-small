import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():

    except:

    x = np.linspace(0, 2, n_points)
    # Create symmetric structure around center
    midpoint = n_points // 2
    h = np.zeros(n_points)
    # Add two bumps to make a symmetric function
    for i in range(n_points):
        x_val = x[i]
        # Create smooth symmetric shape
        h[i] = max(0, min(1, 0.5 * (1 + np.sin(np.pi*(x_val-0.5)) + np.sin(np.pi*(x_val-1.5)))))
    # Enforce symmetry and rescale
    h = project_h(h, n_points)
    return h

def enforce_symmetry(h):
    """Enforce symmetry h[i] = h[n-1-i] for the symmetric case."""
    n = len(h)
    if n % 2 == 0:
        # Ensure even length symmetry at midpoint
        mid = n // 2
        for i in range(mid):
            avg = (h[i] + h[n-1-i]) / 2
            h[i] = h[n-1-i] = avg
    else:
        # Odd length, center is fixed
        mid = n // 2
        for i in range(mid):
            avg = (h[i] + h[n-1-i]) / 2
            h[i] = h[n-1-i] = avg
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using symmetric structure and gradient descents."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45  # Odd for symmetric property
    
    # Initialize and enforce symmetry
    h = initialize_h(n_points)
    h = enforce_symmetry(h)
    h = project_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Hyperparameters
    learning_rate = 0.1
    max_no_improvement = 1000
    iter_no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Find shift that causes max overlap to compute gradient
        conv = np.correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute analytical gradient for shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # The contribution to gradient of h[i] is 1 - h[i+shift] 
            # For wrapping around, correct for indices outside range
            if i + shift >= 0 and i + shift < n_points:
                gradient[i] = (1 - best_h[i + shift]) * (2.0 / n_points)
            else:
                if i + shift < 0:
                    gradient[i] = (1 - best_h[i + shift + n_points]) * (2.0 / n_points)
                elif i + shift >= n_points:
                    gradient[i] = (1 - best_h[i + shift - n_points]) * (2.0 / n_points)

        # Update step
        h_new = best_h - learning_rate * gradient
        h_new = project_h(h_new, n_points)
        h_new = enforce_symmetry(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            # Reduce learning rate if we're improving
            if learning_rate > 1e-6:
                learning_rate *= 0.99
        else:
            iter_no_improvement += 1
            # Increase learning rate if no progress
            learning_rate *= 1.01
            if learning_rate > 1:
                learning_rate = 1
                
        if iter_no_improvement > max_no_improvement:
            break
            
        if iteration % 500 == 0:
            if time.time() - start_time > budget_s * 0.9:
                break
    
    # Final validation and return
    final_h = project_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
