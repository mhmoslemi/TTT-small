import time
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import math

def run(seed=42, budget_s=600, **kwargs):

        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Use differential evolution for global optimization
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        # Project and return C5
        h_proj = project_h(h_vec)
        return compute_c5(h_proj)
    
    # Run differential evolution
    try:
        result = differential_evolution(
            objective, 
            bounds, 
            maxiter=1000,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            callback=lambda x, convergence: None,
            polish=True
        )
        
        if not result.success:
            # If DE failed, fallback to initial
            h_final = h
        else:
            h_final = project_h(result.x)
        
    except Exception as e:
        # Fall back to initial guess
        h_final = h
    
    # Post-processing: local refinement using gradient-based method
    h_refined = h_final.copy()
    best_refined = compute_c5(h_refined)
    
    # Simple local optimization with projected gradient steps
    max_iter_local = 1000
    learning_rate = 0.1
    
    for i in range(max_iter_local):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Estimate gradient using finite differences
        grad = np.zeros_like(h_refined)
        eps = 1e-5
        
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h_refined.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update
        h_new = h_refined - learning_rate * grad
        h_new = project_h(h_new)
        
        # Accept the update if it improves the objective
        c5_new = compute_c5(h_new)
        if c5_new < best_refined:
            h_refined = h_new
            best_refined = c5_new
            
        if i % 100 == 0:
            learning_rate *= 0.95  # Anneal learning rate
    
    # Final validation
    if best_refined < best_c5:
        best_h = h_refined
        best_c5 = best_refined
    
    return (best_h.tolist(), best_c5, len(best_h))
