import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Use initial_h_values if available
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: triangular pulse
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def line_search(h, grad, max_step, n_points, dx):
    """Perform line search to find optimal step size."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Use scipy minimize_scalar for line search
    try:
        res = optimize.minimize_scalar(objective, bounds=(0, max_step), method='bounded')
        if res.success:
            return res.x
    except:
        pass
    return 0.0

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Adaptive learning rate
    eta = 0.1
    min_eta = 1e-6
    max_iter = 10000
    no_improvement_count = 0
    patience = 100
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    dx = 2.0 / n_points
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute all correlations to find the worst shift
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        worst_shift = max_idx - (n_points - 1)
        
        # Compute the gradient for the worst shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Gradient component for shift k
            j = (i - worst_shift) % n_points
            grad[i] = dx * (1 - h[j])
        
        # Line search for optimal step size
        alpha = line_search(h, grad, eta, n_points, dx)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate the new solution
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
            # Reset learning rate on improvement
            eta = 0.1
        else:
            no_improvement_count += 1
            if no_improvement_count > patience:
                # Reduce learning rate
                eta = max(eta / 2, min_eta)
                no_improvement_count = 0
                
    return (best_h, best_c5, n_points)
