import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangular pulse
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h, shift_index, n_points):
    """Compute gradient of C5 w.r.t. h for the shift that causes maximum overlap."""
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    # For shift k = shift_index - (n_points - 1),
    # the gradient element h[i] is dx * (1 - h[i + k])
    k = shift_index - (n_points - 1)
    for i in range(n_points):
        j = (i + k) % n_points
        grad[i] = dx * (1 - h[j])
    return grad

def line_search(h, grad, max_alpha=0.1, n_points=45):
    """Find optimal step size using line search."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    try:
        res = optimize.minimize_scalar(objective, bounds=(0, max_alpha), method='bounded')
        return max(0, res.x)
    except:
        return 0.0

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using gradient-based approach."""
    np.random.seed(seed)
    n_points = 45
    h = initialize_h(n_points)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    eta = 0.1  # Initial learning rate
    max_iter = 10000
    iter_no_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find shift that causes maximum overlap
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(conv)
        # Find shift k that caused this maximum (convert from full convolution index)
        k = max_shift_idx - (n - 1)
        
        # Compute gradient
        grad = compute_gradient(h, max_shift_idx, n_points)
        
        # Line search for optimal step
        alpha = line_search(h, grad, eta, n_points)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Check improvement
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            # Reset learning rate multiplier
        else:
            iter_no_improvement += 1
            # Reduce learning rate
            eta = max(eta / 2, 1e-6)
        
        if iter_no_improvement > max_no_improvement:
            break
        
        if best_c5 < 0.3807:
            break

    # Ensure final result is valid
    final_h = project_h(h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
