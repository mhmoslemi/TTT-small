import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w = 0.7
    c1 = c2 = 1.5
    inertia_decay = 0.99
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        elif i < swarm_size // 10:
            particle = np.random.rand(n_points)
            particle = project_h(particle, n_points)
        else:
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_h(particle, n_points)
            
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        fitness = evaluate_h(particle, dx)
        pbest.append(particle.copy())
        pbest_fitness.append(fitness)
    
    # Global best
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    max_stagnation = 500
    
    while time.time() - start_time < 0.7 * budget_s and iteration < max_iter:
        iteration += 1
        # Update inertia weight
        w *= inertia_decay
        
        # Update particles
        for i in range(swarm_size):
            particle = swarm[i]
            velocity = velocities[i]
            pos = particle
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocity = (w * velocity + 
                       c1 * r1 * (pbest[i] - pos) + 
                       c2 * r2 * (gbest - pos))
            
            # Update position
            pos = pos + velocity
            pos = project_h(pos, n_points)
            
            # Evaluate
            new_fitness = evaluate_h(pos, dx)
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = pos.copy()
                pbest_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < gbest_fitness:
                    gbest = pos.copy()
                    gbest_fitness = new_fitness
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            
            swarm[i] = pos
            velocities[i] = velocity
            
        # Adapt swarm size if stuck
        if stagnation_count > max_stagnation and swarm_size > 20:
            swarm_size = max(20, int(swarm_size * 0.8))
            # Reinitialize some particles
            for i in range(swarm_size, len(swarm)):
                swarm[i] = np.random.rand(n_points)
                swarm[i] = project_h(swarm[i], n_points)
                velocities[i] = np.zeros(n_points)
                fitness = evaluate_h(swarm[i], dx)
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fitness
            swarm = swarm[:swarm_size]
            velocities = velocities[:swarm_size]
            pbest = pbest[:swarm_size]
            pbest_fitness = pbest_fitness[:swarm_size]
            gbest_idx = np.argmin(pbest_fitness)
            gbest = pbest[gbest_idx].copy()
            gbest_fitness = pbest_fitness[gbest_idx]
            stagnation_count = 0
    
    # Gradient descent refinement
    h = gbest.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    min_improvement = 1e-5
    
    while time.time() - start_time < 0.95 * budget_s:
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute finite difference gradients
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update using sign of gradient
        old_h = h.copy()
        h = h - learning_rate * np.sign(grad)
        h = project_h(h, n_points)
        
        # Check improvement
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            
        # Adaptive learning rate
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
            
        # Check for convergence
        if abs(new_c5 - best_c5) < min_improvement:
            break
                
    return (best_h, best_c5, n_points)
