import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.999
    T = T0
    max_iter = 100000
    
    current_h = h.copy()
    current_c5 = best_c5
    
    iteration = 0
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        # Generate candidate solution by adding noise
        candidate_h = current_h + np.random.normal(0, T * 0.1, n_points)
        
        # Project candidate to feasible region
        candidate_h = project_h(candidate_h)
        
        # Compute candidate C5
        candidate_c5 = compute_c5(candidate_h)
        
        # Accept or reject based on Metropolis criterion
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            current_h = candidate_h
            current_c5 = candidate_c5
            
            # Update best if improved
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        T *= alpha
        iteration += 1
        
        # Early termination
        if (time.time() - start_time) > budget_s - 1:
            break
    
    # Final adjustment to ensure constraints
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
