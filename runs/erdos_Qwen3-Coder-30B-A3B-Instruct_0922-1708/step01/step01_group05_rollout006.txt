import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to satisfy integral constraint
    h = h * (target_sum / np.sum(h))
    
    # Particle Swarm Optimization with gradient refinement
    n_particles = 100
    n_iterations = 5000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_velocity = 0.1
    
    # Initialize particles (values only, positions fixed)
    particles = []
    for _ in range(n_particles):
        particle_h = h + np.random.normal(0, 0.05, n_points)
        particle_h = np.clip(particle_h, 0, 1)
        particle_h = particle_h * (target_sum / np.sum(particle_h))
        particles.append(particle_h.copy())
    
    # Particle velocities
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Best particle tracking
    best_h = h.copy()
    best_c = float('inf')
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        return np.max(np.abs(corr)) * dx
    
    # Evaluation loop
    t0 = time.time()
    for iteration in range(n_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            particle_h = particles[i]
            
            # Compute current C
            C = compute_C(particle_h)
            if C < best_c:
                best_c = C
                best_h = particle_h.copy()
            
            # PSO update for this particle
            r1, r2 = np.random.rand(2)
            # Update velocity
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (particle_h - particle_h) +
                            social * r2 * (best_h - particle_h))
            
            # Check bounds for velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            new_h = particle_h + velocities[i]
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (target_sum / np.sum(new_h))
            
            # Update particle
            particles[i] = new_h.copy()
        
        # Occasionally refine with gradient descent
        if iteration % 200 == 0:
            candidates = sorted(enumerate(particles), key=lambda x: compute_C(x[1]))
            top_candidates = candidates[:int(n_particles * 0.1)]
            
            for idx, particle_h in top_candidates:
                # Gradient descent step
                lr = 0.01
                c_prev = compute_C(particle_h)
                for _ in range(10):
                    if time.time() - start_time > budget_s - 1:
                        break
                    # Approximate gradient via finite difference
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = particle_h.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (target_sum / np.sum(h_plus))
                        c_plus = compute_C(h_plus)
                        
                        grad[j] = (c_plus - c_prev) / eps
                    
                    # Apply gradient
                    particle_h -= lr * grad
                    particle_h = np.clip(particle_h, 0, 1)
                    particle_h = particle_h * (target_sum / np.sum(particle_h))
                    
                    c_new = compute_C(particle_h)
                    if c_new >= c_prev:
                        lr *= 0.5  # Reduce learning rate if no improvement
                    else:
                        c_prev = c_new
                
                particles[idx] = particle_h.copy()
    
    # Final evaluation of best solution
    final_c = compute_C(best_h)
    
    return (best_h.tolist(), final_c, n_points)
