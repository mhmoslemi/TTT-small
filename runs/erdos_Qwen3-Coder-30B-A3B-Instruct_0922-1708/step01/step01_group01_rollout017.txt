import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Number of segments for piecewise constant representation
    m = 5
    segment_values = [0, 0.5, 1]
    
    # Initial guess from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with a simple construction
        h = np.ones(n_points) * 0.5
    
    # Mark start time
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Generate all possible segment combinations
    combinations = list(product(segment_values, repeat=m))
    num_combinations = len(combinations)
    
    # Process combinations in batches to respect time budget
    batch_size = 50
    for i in range(0, num_combinations, batch_size):
        if time.time() - start_time > 0.7 * budget_s:
            break
        batch = combinations[i:i+batch_size]
        
        # For each combination, create a piecewise constant h
        for combo in batch:
            if time.time() - start_time > 0.7 * budget_s:
                break
                
            # Create segment boundaries
            segment_boundaries = np.linspace(0, 2, m+1)
            h_piecewise = np.zeros(n_points)
            for j in range(m):
                start_idx = int(np.round(segment_boundaries[j] / 2 * n_points))
                end_idx = int(np.round(segment_boundaries[j+1] / 2 * n_points))
                h_piecewise[start_idx:end_idx] = combo[j]
            
            # Make sure last segment fills to end
            h_piecewise[int(np.round(segment_boundaries[-1] / 2 * n_points)):] = combo[-1]
            
            # Project to meet constraints
            h_projected = project_h(h_piecewise, n_points)
            
            # Evaluate
            c5_val = evaluate_h(h_projected, dx)
            if c5_val < best_c5:
                best_h = h_projected.copy()
                best_c5 = c5_val
    
    # Gradient descent refinement on top candidates
    if time.time() - start_time < 0.3 * budget_s:
        h_refined = best_h.copy()
        learning_rate = 0.01
        patience = 0
        max_patience = 5000
        
        while time.time() - start_time < budget_s and patience < max_patience:
            # Compute current gradient via finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_refined.copy()
                h_minus[j] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[j] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h_refined -= learning_rate * grad
            h_refined = project_h(h_refined, n_points)
            
            # Evaluate
            new_c5 = evaluate_h(h_refined, dx)
            if new_c5 < best_c5:
                best_h = h_refined.copy()
                best_c5 = new_c5
                patience = 0
            else:
                patience += 1
                if patience % 1000 == 0:
                    learning_rate *= 0.5
                    patience = 0
    
    return (best_h, best_c5, n_points)
