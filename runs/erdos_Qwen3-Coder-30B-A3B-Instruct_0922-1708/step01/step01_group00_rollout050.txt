import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a triangular distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to higher resolution
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, len(h)), h)
        h = h_interp
    else:
        # Triangular distribution favoring lower overlap
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1)  # Triangle centered at 1
        h = np.clip(h, 0.01, 0.99)
    
    # Scale to meet integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Adam optimizer state
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    
    start_time = time.time()
    iteration = 0
    
    def compute_c5_and_grad(h_vec):
        # Compute correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(corr) * dx
        max_idx = np.argmax(corr)
        
        # Compute gradient using exact formula
        grad = np.zeros_like(h_vec)
        # Gradients are complex: for each i, 
        # ∂C5/∂h[i] = dx * (2 * h[i] - 1) at max_correlation point
        # But we simplify to approximate gradient
        
        # Compute finite difference with small steps for better gradient
        eps = 1e-6
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Clip gradient for stability
        grad = np.clip(grad, -100, 100)
        return c5, grad

    # Optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        c5, grad = compute_c5_and_grad(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = 0.01 / (np.sqrt(v_hat) + eps) * m_hat
        
        h_new = h - lr
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
