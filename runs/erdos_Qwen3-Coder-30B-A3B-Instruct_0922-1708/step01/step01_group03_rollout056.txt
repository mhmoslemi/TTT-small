import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # If initial_h_values is accessible and not None, use it
        if 'initial_h_values' in globals() and initial_h_values is not None:
            h = np.array(initial_h_values).copy()
            return h
    except:
        pass
    # Default to symmetric construction
    x = np.linspace(0, 2, n_points)
    # Make it piecewise linear and symmetric around 1
    h = np.zeros(n_points)
    midpoint = n_points // 2
    for i in range(n_points):
        xi = x[i]
        if xi <= 1:
            h[i] = max(0, 1 - 2 * abs(xi - 0.5))
        else:
            h[i] = max(0, 1 - 2 * abs(xi - 1.5))
    return project_h(h, n_points)

def pso_phase(h, n_points, time_budget_s):
    """Run Particle Swarm Optimization to find good initial solution."""
    start_time = time.time()
    swarm_size = 30
    max_iter = 1000
    cognitive_weight = 2.0
    social_weight = 2.0
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(swarm_size):
        individual = h.copy()
        swarm.append(individual)
        velocities.append(np.zeros(n_points))
        pbest.append(individual.copy())
        pbest_c5.append(evaluate_c5(individual))
    
    # Global best
    best_idx = np.argmin(pbest_c5)
    gbest = pbest[best_idx].copy()
    gbest_c5 = pbest_c5[best_idx]
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.7 and iteration < max_iter:
        iteration += 1
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                           cognitive_weight * r1 * (pbest[i] - swarm[i]) +
                           social_weight * r2 * (gbest - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to valid domain
            swarm[i] = project_h(swarm[i], n_points)
            
            # Evaluate
            current_c5 = evaluate_c5(swarm[i])
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = swarm[i].copy()
                pbest_c5[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest = swarm[i].copy()
                    gbest_c5 = current_c5
        
        # Decrease inertia weight
        inertia_weight = max(0.4, inertia_weight * 0.99)
    
    return gbest, gbest_c5

def gradient_descent_phase(h, n_points, time_budget_s):
    """Run gradient descent with projected updates."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.05
    max_iter = 10000
    patience = 1000
    no_improvement = 0
    
    def compute_exact_gradient(h):
        """Compute exact gradient for the maximum shift."""
        N = len(h)
        dx = 2.0 / N
        conv = np.correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(conv)
        max_shift = max_shift_idx - (N - 1)
        
        # For shift k, gradients are:
        # dC5/dh_i = (1 - h[i+k]) * dx if i+k < N, 0 otherwise
        # This is simplified for one shift, but for full max we need
        # the elastic net derivative or approximate approach.
        # We compute gradient for the shift that gives maximum overlap.
        
        # For computational efficiency during this phase, we directly compute
        # approximate gradient: fd for a sampled subset of elements
        gradient = np.zeros(N)
        eps = 1e-4
        for i in range(N):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, N)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, N)
            c5_minus = evaluate_c5(h_minus)
            
            gradient[i] = (c5_plus - c5_minus) / (2 * eps)
        
        return gradient
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_budget_s * 0.3:
            break
            
        # Compute gradient
        grad = compute_exact_gradient(best_h)
        
        # Update with learning rate
        h_new = best_h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Check if improvement
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > patience:
            break
            
        # Adjust learning rate
        if iteration % 500 == 0:
            learning_rate *= 0.9
            if learning_rate < 1e-6:
                break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use n_points=45 to keep it reasonably efficient
    n_points = 45
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: PSO
    h_pso, c5_pso = pso_phase(h, n_points, budget_s)
    
    # Phase 2: Gradient Descent
    h_gd, c5_gd = gradient_descent_phase(h_pso, n_points, budget_s)
    
    # Return best result
    final_h = h_gd if c5_gd < c5_pso else h_pso
    final_c5 = min(c5_gd, c5_pso)
    
    # Final validation and recompute
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
