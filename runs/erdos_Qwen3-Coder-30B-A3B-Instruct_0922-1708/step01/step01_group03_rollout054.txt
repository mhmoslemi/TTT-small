import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Create symmetric starting point
    x = np.linspace(0, 2, n_points)
    # Gaussian bumps
    h = 0.5 * (1 + np.exp(-10*(x-0.5)**2) + np.exp(-10*(x-1.5)**2))
    h = project_h(h, n_points)
    return h

def symmetric_project(h):
    """Ensure h is symmetric and within bounds."""
    n = len(h)
    k = (n - 1) // 2
    # Make symmetric: h[i] = (h[i] + h[n-1-i]) / 2
    h_sym = h.copy()
    for i in range(k):
        h_sym[i] = (h[i] + h[n-1-i]) / 2
        h_sym[n-1-i] = h_sym[i]
    return project_h(h_sym, len(h_sym))

def compute_exact_gradient(h, max_shift):
    """Compute exact gradient w.r.t. maximum overlap shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # Compute the shift index in the convolution that gives the maximum
    conv_len = 2 * n - 1
    shift_idx = max_shift + (n - 1)
    # For convolution, the gradient w.r.t. h[i] is related to the elements of 1-h shifted by max_shift
    # Specifically: for index i in h, the contribution to the maximum correlation is
    # from (1-h[i+shift]) or (1-h[i-shift]) etc.
    # But we're computing the overall maximum shift.
    # And the gradient is computed corresp. to convolution indices
    for i in range(n):
        conv_pos = i + max_shift + (n - 1)
        if 0 <= conv_pos < conv_len:
            grad[i] = -dx
        else:
            grad[i] = 0
    return grad

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using gradient descent with symmetry."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    h = initialize_h(n_points)
    h = symmetric_project(h)
    
    max_no_improvement = 1000
    iter_no_improvement = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.1
    
    # Time budget for gradient descent
    while time.time() - start_time < budget_s * 0.95:
        # Find the shift that produces maximum overlap
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        max_shift = max_idx - (n_points - 1)
        
        # Compute exact gradient
        gradient = compute_exact_gradient(h, max_shift)
        
        # Update with gradient
        h_new = h - learning_rate * gradient
        h_new = symmetric_project(h_new)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            iter_no_improvement = 0
            # Reduce learning rate slowly
            if learning_rate > 1e-6:
                learning_rate *= 0.99
        else:
            iter_no_improvement += 1
            # Increase learning rate if nothing worked
            learning_rate *= 1.1
            if learning_rate > 1:
                learning_rate = 1
            
        h = h_new
        if iter_no_improvement >= max_no_improvement:
            break
            
    # Final validation
    final_h = symmetric_project(best_h)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
