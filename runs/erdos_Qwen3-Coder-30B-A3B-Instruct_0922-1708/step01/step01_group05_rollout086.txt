            start_idx = i * segment_size
            end_idx = start_idx + segment_size if i < segments - 1 else n_points
            # Alternate high and low values
            if i % 2 == 0:
                h[start_idx:end_idx] = 1.0
            else:
                h[start_idx:end_idx] = 0.0
        # Rescale to match sum constraint
        h = h * (target_sum / np.sum(h))

        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def project_h(h_vals):
        # Ensure h is in [0,1] and sum to target_sum
        h_vals = np.clip(h_vals, 0, 1)
        scale = target_sum / np.sum(h_vals)
        return h_vals * scale
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    max_iter = 50000
    iter_count = 0
    
    best_h = h.copy()
    best_C = compute_C(h)
    
    # Annealing loop
    while iter_count < max_iter and time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Generate neighbor by making small perturbations
        h_new = h.copy()
        # Mutate one random segment
        seg_idx = np.random.randint(0, n_points)

        h_new = project_h(h_new)

        if C_new < best_C:

            best_C = C_new

            # Accept with probability based on temperature
            delta = C_new - best_C
            if np.random.rand() < np.exp(-delta / T):

        # Cooling schedule
        T *= alpha
        
    # Gradient descent refinement
    lr = 0.01
    max_grad_iter = 200
    no_improvement = 0
    patience = 20

            h_plus = project_h(h_plus)
            C_plus = compute_C(h_plus)

            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > patience:
                lr *= 0.5
                no_improvement = 0
    
    return (best_h.tolist(), best_C, n_points)
