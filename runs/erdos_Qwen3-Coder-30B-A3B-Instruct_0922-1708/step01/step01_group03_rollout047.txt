import numpy as np
from scipy import optimize
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():

    except:

    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def find_worst_shift(h):
    """Find shift that gives maximum overlap."""
    n = len(h)
    conv = np.correlate(h, 1 - h, mode="full")
    # Take max in absolute value correctly (shift between -n+1 and n-1)
    max_idx = np.argmax(conv)
    shift = max_idx - (n - 1)
    return shift

def compute_gradient(h, shift, n_points):
    """Compute gradient for a specific shift."""
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + shift) % n_points
        grad[i] = (1 - h[j]) * dx
    return grad

def line_search(h, grad, shift, n_points, eta):
    """Perform line search to find optimal step size."""
    def c5_at_alpha(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Try different alphas
    alphas = [eta * (0.5 ** i) for i in range(10)]
    best_alpha = 0
    best_c5 = evaluate_c5(h)
    
    for alpha in alphas:
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_alpha = alpha
    
    return best_alpha, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # Use smaller number for faster runtime, still effective
    h = initialize_h(n_points)
    
    # Set up learning rate
    eta = 0.1
    eta_min = 1e-6
    eta_max = 0.5
    iter_no_improvement = 0
    max_no_improvement = 1000
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s and iter_no_improvement < max_no_improvement:
        # Find shift that gives maximum overlap
        shift = find_worst_shift(best_h)
        
        # Compute gradient for this shift
        grad = compute_gradient(best_h, shift, n_points)
        
        # Line search to find optimal step size
        alpha, new_c5 = line_search(best_h, grad, shift, n_points, eta)
        
        # Update h
        h_new = best_h - alpha * grad
        h_new = project_h(h_new, n_points)
        
        # Check improvement
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            # Increase learning rate slightly if beneficial
            eta = min(eta * 1.2, eta_max)
        else:
            iter_no_improvement += 1
            # Decrease learning rate
            eta = max(eta / 2, eta_min)
        
        iteration += 1
        
        if iteration % 50 == 0:
            pass  # Progress update
    
    # Final validation
    final_h = project_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
