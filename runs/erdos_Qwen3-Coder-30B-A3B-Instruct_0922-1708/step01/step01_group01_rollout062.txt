import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sample(n_points, n_samples):
    """Generate Latin Hypercube Samples"""
    samples = np.zeros((n_samples, n_points))
    for i in range(n_points):
        # Generate samples for each dimension
        sample = np.random.rand(n_samples)
        # Shuffle to create LHS
        np.random.shuffle(sample)
        samples[:, i] = sample
    return samples

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h with either parent solution or sinusoid
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Phase 1: Latin Hypercube Sampling for diversity
    n_samples = 50
    n_elites = 20
    lhs_samples = lhs_sample(n_points, n_samples)
    candidates = []
    
    for i in range(n_samples):
        # Perturb the initial point with random LHS
        h_trial = h + 0.1 * (lhs_samples[i] - 0.5)
        h_trial = project_h(h_trial, n_points)
        c5 = evaluate_h(h_trial, dx)
        candidates.append((h_trial, c5))
    
    # Sort by performance
    candidates.sort(key=lambda x: x[1])
    elites = candidates[:n_elites]
    
    # Phase 2: Evolution Strategy
    n_offspring = 5
    max_generations = 1000
    generation = 0
    
    while time.time() - start_time < 0.7 * budget_s and generation < max_generations:
        new_candidates = []
        for h_elite, _ in elites:
            for _ in range(n_offspring):
                # Mutate the elite
                h_offspring = h_elite + 0.05 * (np.random.rand(n_points) - 0.5)
                h_offspring = project_h(h_offspring, n_points)
                c5 = evaluate_h(h_offspring, dx)
                new_candidates.append((h_offspring, c5))
        
        # Combine elites and offspring, keep top performers
        all_candidates = elites + new_candidates
        all_candidates.sort(key=lambda x: x[1])
        elites = all_candidates[:n_elites]
        generation += 1
    
    # Best candidate from evolution
    best_h = elites[0][0]
    best_c5 = elites[0][1]

    # Phase 3: Gradient Descent on best solution
    h = best_h.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 500
    
    def finite_diff_grad(h, dx, epsilon=1e-5):
        grad = np.zeros_like(h)
        base_c5 = evaluate_h(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * epsilon)
        return grad
    
    while time.time() - start_time < 0.95 * budget_s:
        grad = finite_diff_grad(h, dx)
        h -= learning_rate * np.sign(grad)
        h = project_h(h, n_points)
        
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 100 == 0:
            learning_rate *= 0.9
    
    return (best_h, best_c5, n_points)
