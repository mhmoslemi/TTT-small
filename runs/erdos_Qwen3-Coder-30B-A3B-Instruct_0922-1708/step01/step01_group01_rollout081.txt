import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO + gradient descent"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    swarm_size = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    stagnation_count = 0
    max_stagnation = 500
    min_swarm_size = 20
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Generate initial swarm
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            # Small random perturbations of initial solution
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_h(particle, n_points)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_values.append(evaluate_h(particle, dx))
    
    # Global best
    gb_idx = np.argmin(pbest_values)
    gbest = pbest[gb_idx].copy()
    gbest_value = pbest_values[gb_idx]
    
    start_time = time.time()
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] = project_h(swarm[i] + velocities[i], n_points)
            
            # Evaluate
            current_value = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = swarm[i].copy()
                pbest_values[i] = current_value
                if current_value < gbest_value:
                    gbest = swarm[i].copy()
                    gbest_value = current_value
                    stagnation_count = 0
        
        # Adapt swarm size if needed
        stagnation_count += 1
        if stagnation_count >= max_stagnation and swarm_size > min_swarm_size:
            swarm_size = max(min_swarm_size, int(swarm_size * 0.8))
            # Re-initialize swarm
            new_swarm = []
            new_velocities = []
            new_pbest = []
            new_pbest_values = []
            
            # Keep best particles
            best_indices = np.argsort(pbest_values)[:swarm_size]
            for idx in best_indices:
                new_swarm.append(pbest[idx].copy())
                new_velocities.append(np.zeros(n_points))
                new_pbest.append(pbest[idx].copy())
                new_pbest_values.append(pbest_values[idx])
            
            # Add new random particles to fill swarm
            for i in range(swarm_size - len(best_indices)):
                particle = h + np.random.normal(0, 0.1, n_points)
                particle = project_h(particle, n_points)
                new_swarm.append(particle)
                new_velocities.append(np.zeros(n_points))
                new_pbest.append(particle.copy())
                new_pbest_values.append(evaluate_h(particle, dx))
            
            swarm = new_swarm
            velocities = new_velocities
            pbest = new_pbest
            pbest_values = new_pbest_values
            
            gb_idx = np.argmin(pbest_values)
            gbest = pbest[gb_idx].copy()
            gbest_value = pbest_values[gb_idx]
            stagnation_count = 0
    
    # Gradient descent refinement
    h_refine = gbest.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s - 5 and patience < max_patience:
        corr = correlate(h_refine, 1 - h_refine, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h_refine.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_refine.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with gradient descent
        h_refine -= learning_rate * grad
        h_refine = project_h(h_refine, n_points)
        
        new_c5 = evaluate_h(h_refine, dx)
        
        if new_c5 < gbest_value:
            gbest_value = new_c5
            gbest = h_refine.copy()
            patience = 0
        else:
            patience += 1
            if patience > 500:
                learning_rate *= 0.5
                patience = 0
    
    return (gbest, gbest_value, n_points)
