import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Discretization
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def segment_based_optimization(h_init, m=7, max_iter=200):
        # Create segment boundaries
        segment_width = n_points // m
        if segment_width * m < n_points:
            segment_width += 1
            
        # Initialize segment values
        c = np.ones(m) * 0.5
        
        # Reconstruction function
        def reconstruct_h(c):
            h_recon = np.zeros(n_points)
            for i in range(m):
                start_idx = i * segment_width
                end_idx = min((i + 1) * segment_width, n_points)
                h_recon[start_idx:end_idx] = c[i]
            return h_recon
            
        # Normalize segment average to match integral constraint
        def normalize_c(c):
            current_sum = np.sum(c)
            target_sum = m / 2
            if current_sum != 0:
                c = c * target_sum / current_sum
            else:
                c = np.ones_like(c) * (target_sum / m)
            return c
        
        # Objective function for optimization
        def objective(c):
            h = reconstruct_h(c)
            return evaluate(h)
            
        # Optimization process
        best_c = c.copy()
        best_obj = objective(best_c)
        no_improvement_count = 0
        
        for iteration in range(max_iter):
            # Normalize before optimization
            best_c = normalize_c(best_c)
            
            # Optimization step using SLSQP
            try:
                res = minimize(objective, best_c, method='SLSQP', 
                             bounds=[(0, 1) for _ in range(m)], 
                             options={'maxiter': 50})
                if res.success:
                    new_c = res.x
                    # Ensure all constraints are respected
                    new_c = np.clip(new_c, 0, 1)
                    new_obj = objective(new_c)
                    if new_obj < best_obj:
                        best_c = new_c.copy()
                        best_obj = new_obj
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
                else:
                    no_improvement_count += 1
                    
            except:
                no_improvement_count += 1
                
            # Periodic perturbations
            if iteration % 25 == 0 and iteration > 0:
                idx_to_perturb = np.random.choice(m, size=max(1, m//10), replace=False)
                for i in idx_to_perturb:
                    perturb_val = np.random.uniform(-0.1, 0.1)
                    best_c[i] += perturb_val
                best_c = np.clip(best_c, 0, 1)
                
            # Stop if no improvement
            if no_improvement_count > 20:
                break
                
        # Final reconstruction and final evaluation
        final_h = reconstruct_h(best_c)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_obj = evaluate(final_h)
        
        return final_h, final_obj, best_c
    
    # Run initial phase
    start_time = time.time()
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    # Try multiple refinement strategies
    for segments in [5, 7, 9]:
        if time.time() - start_time > budget_s - 5:
            break
        h_trial, c5_trial, _ = segment_based_optimization(best_h, segments, max_iter=200)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            
        # Early exit if good enough or time nearly up
        if best_c5 <= 0.3807:
            break
            
    # Apply final smoothing to the best solution
    if time.time() - start_time < budget_s - 2:
        # Use a bit more smoothing to analyze
        smoothed_h = np.convolve(best_h, np.ones(3)/3, mode='same')
        smoothed_h = np.clip(smoothed_h, 0, 1)
        smoothed_h = smoothed_h * (n_points / 2) / np.sum(smoothed_h)
        smooth_c5 = evaluate(smoothed_h)
        if smooth_c5 < best_c5:
            best_h = smoothed_h
            best_c5 = smooth_c5
    
    # Final evaluation to be sure
    final_c5 = evaluate(best_h)
    
    return (best_h, final_c5, n_points)
