import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def project_h(h):
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def smooth(h, window=5):
        if window < 2:
            return h
        # Use convolution with uniform kernel for smoothing
        kernel = np.ones(window) / window
        h_smooth = np.convolve(h, kernel, mode='same')
        return h_smooth
    
    def perturb(h, delta=0.05):
        h_new = h.copy()
        # Perturb about 10% of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        return project_h(h_new)
    
    def objective(h):
        # We optimize the negative of the maximum overlap
        return -evaluate(h)
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Create constraints for scipy.optimize
    cons = {'type': 'eq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial point and best solution
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    current_delta = 0.05
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        # Optimize using L-BFGS-B
        try:
            result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons,
                              options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-5})
            if result.success:
                h_opt = project_h(result.x)
                c5_opt = evaluate(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                # Fall back to perturbation if L-BFGS fails
                h_trial = perturb(best_h, current_delta)
                h_trial = smooth(h_trial, window=5)
                c5_trial = evaluate(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
        except:
            # Fallback to perturbation in case of optimization error
            h_trial = perturb(best_h, current_delta)
            h_trial = smooth(h_trial, window=5)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Adapt delta for next iteration if no improvement for a while
        if no_improvement_count > 20:
            current_delta = min(0.1, current_delta * 1.2)
        
        iterations += 1
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    
    return (best_h, final_c5, n_points)
