import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def optimize_segments(c_vals, m, n_points, dx):
        # Reconstruct h from segment values
        h = np.zeros(n_points)
        segment_size = n_points // m
        for i in range(m):
            start_idx = i * segment_size
            end_idx = (i+1) * segment_size if i < m-1 else n_points
            h[start_idx:end_idx] = c_vals[i]
        return h
    
    def segment_optimization(c_vals, m, n_points, dx, maxiter=1000):
        # Optimize the segment values with constraints
        def objective(c_vals):
            h = optimize_segments(c_vals, m, n_points, dx)
            return evaluate(h)
        
        # Initial guess
        c0 = c_vals.copy()
        
        # Constraints
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - m/2}]
        
        # Bounds
        bounds = [(0, 1) for _ in range(m)]
        
        # Optimization
        res = minimize(objective, c0, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': maxiter, 'ftol': 1e-8})
        return res.x if res.success else c0
    
    def perturb_segment_values(c_vals, perturb_fraction=0.1, delta=0.1):
        c_new = c_vals.copy()
        n_perturb = max(1, int(len(c_vals) * perturb_fraction))
        indices = np.random.choice(len(c_vals), n_perturb, replace=False)
        for i in indices:
            c_new[i] += np.random.uniform(-delta, delta)
        # Clip to [0, 1]
        c_new = np.clip(c_new, 0, 1)
        # Renormalize to maintain sum constraint
        c_new = c_new * (len(c_new)/2) / np.sum(c_new)
        return c_new
    
    def smooth_h(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth

    # Initialize optimization parameters
    m = 7  # initial number of segments
    c_vals = np.full(m, 0.5)
    best_c_vals = c_vals.copy()
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    # Using initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
        # Convert to segments
        segment_size = n_points // m
        c_vals = np.array([np.mean(h_start[i*segment_size:(i+1)*segment_size]) for i in range(m)])
        c_vals = np.clip(c_vals, 0, 1)
        # Normalize
        c_vals = c_vals * (m/2) / np.sum(c_vals)
    except NameError:
        pass

    while time.time() - start_time < budget_s and no_improvement_count < 50:
        iteration += 1
        
        # Optimize segment values
        try:
            c_vals_opt = segment_optimization(c_vals, m, n_points, dx, maxiter=500)
        except Exception:
            c_vals_opt = c_vals.copy()
        
        # Perturb after optimization
        c_vals_perturb = perturb_segment_values(c_vals_opt, 0.1, 0.1)
        
        # Reconstruct h and evaluate
        h_trial = optimize_segments(c_vals_perturb, m, n_points, dx)
        h_trial = smooth_h(h_trial, window=3)
        
        # Ensure bounds and normalization
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        c5_trial = evaluate(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_c_vals = c_vals_perturb.copy()
            best_h = h_trial.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Occasionally increase resolution
        if iteration % 100 == 0 and m < 21 and time.time() - start_time < budget_s * 0.9:
            m += 2
            # Interpolate c_vals
            old_c_vals = best_c_vals.copy()
            c_vals = np.interp(np.linspace(0, m-1, m), np.linspace(0, len(old_c_vals)-1, len(old_c_vals)), old_c_vals)
            c_vals = np.clip(c_vals, 0, 1)
            # Renormalize
            c_vals = c_vals * (m/2) / np.sum(c_vals)
    
    # Final evaluation with best solution
    h_final = optimize_segments(best_c_vals, m, n_points, dx)
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    c5_final = evaluate(h_final)
    
    return (h_final, c5_final, n_points)
