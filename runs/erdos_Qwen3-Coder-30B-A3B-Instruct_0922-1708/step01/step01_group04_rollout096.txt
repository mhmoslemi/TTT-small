        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Try optimization using a continuation approach
    best_h = h_start.copy()

    def build_h_from_segments(segments, segment_lengths, n_total):
        h = np.zeros(n_total)
        idx = 0
        for i, (seg_val, length) in enumerate(zip(segments, segment_lengths)):
            end_idx = idx + int(length)
            h[idx:end_idx] = seg_val
            idx = end_idx
        return h
    
    # Optimization loop
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 200
    
    # Adaptive learning rate and trust region
    lr = 0.1
    tr_radius = 0.1

    if best_h is None or len(best_h) != n_points:

        best_c5 = evaluate(best_h)
