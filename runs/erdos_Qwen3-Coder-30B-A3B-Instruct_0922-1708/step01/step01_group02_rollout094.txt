import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Use differential evolution with custom objective
    def objective(h_vec):
        h_vec = project_h(h_vec)
        return compute_c5(h_vec)
    
    # Bounds for variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    max_iter = 1000
    popsize = 15
    mutation = (0.5, 1)
    recombination = 0.7
    
    de_result = differential_evolution(
        objective, 
        bounds, 
        maxiter=max_iter, 
        popsize=popsize, 
        mutation=mutation, 
        recombination=recombination,
        seed=seed,
        disp=False,
        polish=True
    )
    
    # Final refinement
    final_h = project_h(de_result.x)
    final_c5 = compute_c5(final_h)
    
    # Ensure we return the best solution found within time limit
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, len(best_h))
