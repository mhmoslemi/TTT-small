import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate(h):
    # Compute correlation with 1-h
    corr = correlate(h, 1 - h, mode='full')
    # The maximum overlap
    max_overlap = np.max(corr) * (2.0 / len(h))
    return max_overlap

    h_start = np.clip(h_start, 0, 1)

    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    # Try various resolutions and improve
    # Begin with a coarse grid to explore solution space
    segment_counts = [7, 14, 21, 35, 50, 70]
    
    def reconstruct_h(segment_values, n_points):
        # Reconstruct step function from segment averages
        m = len(segment_values)
        segment_width = n_points / m
        h_reconstructed = np.zeros(n_points)
        for i in range(m):
            start_idx = int(i * segment_width)
            end_idx = int((i+1) * segment_width)
            h_reconstructed[start_idx:end_idx] = segment_values[i]
        return h_reconstructed
    
    def smooth_h(h, sigma=1.0):
        # Apply Gaussian smoothing to make it continuous
        kernel = np.exp(-0.5 * np.linspace(-3*sigma, 3*sigma, int(6*sigma+1))**2 / sigma**2)
        kernel = kernel / np.sum(kernel)
        h_smooth = np.convolve(h, kernel, mode='same')
        h_smooth = np.clip(h_smooth, 0, 1)
        return h_smooth
    
    last_improvement = 0
    for m in segment_counts:
        if time.time() - start_time > budget_s - 1:
            break
            
        # If we're at the finest resolution and we're near our target, do a high-res optimization
        if m == segment_counts[-1]:

            def objective(h_values):
                # Reconstruct full function
                h = reconstruct_h(h_values, n_points)
                # Smooth for better gradient estimate
                h_smooth = smooth_h(h)
                return evaluate(h_smooth)
            
            def constraint_eq(x):
                return np.sum(x) - m/2
            
            # Bounds: ensure segment values are between 0 and 1
            bounds = [(0, 1) for _ in range(m)]

            best_segment_h = None

                last_improvement += 1
