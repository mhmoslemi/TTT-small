import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Genetic Algorithm Parameters
    pop_size = 100
    num_generations = 10000
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = pop_size // 10
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        candidate = np.random.rand(n_points)
        candidate = project_h(candidate)
        population.append(candidate)
    
    # Include initial solution if available
    if 'initial_h_values' in globals():
        population[0] = project_h(np.array(initial_h_values))
        
    # GA Loop
    gen = 0
    while (time.time() - start_time) < budget_s - 1 and gen < num_generations:
        # Evaluate fitness for entire population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        current_best = population[0]
        current_c5 = fitness[0]
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
            
        # Selection - Roulette wheel
        inverted_fitness = [1.0 / (f + 1e-10) for f in fitness]
        total_fitness = sum(inverted_fitness)
        probabilities = [f / total_fitness for f in inverted_fitness]
        
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        while len(new_population) < pop_size:
            # Select parents
            parents = []
            for _ in range(2):
                r = np.random.rand()
                cumulative = 0
                for i, p in enumerate(probabilities):
                    cumulative += p
                    if r <= cumulative:
                        parents.append(population[i])
                        break
            
            # Crossover
            if np.random.rand() < crossover_rate and len(parents) >= 2:
                parent1, parent2 = parents[0], parents[1]
                point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:point], parent2[point:]])
                child2 = np.concatenate([parent2[:point], parent1[point:]])
            else:
                child1, child2 = parents[0].copy(), parents[1].copy()
                
            # Mutation
            mutation_strength = mutation_rate / np.sqrt(gen + 1) + 1e-6
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:  # 10% chance of mutation per gene
                        child1[i] += np.random.normal(0, 0.1 * mutation_strength)
                child1 = project_h(child1)
            
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child2[i] += np.random.normal(0, 0.1 * mutation_strength)
                child2 = project_h(child2)
                
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
                
        population = new_population
        gen += 1
        
    # Final evaluation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
