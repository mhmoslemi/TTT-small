import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    mutation_rate = 0.1
    elite_size = pop_size // 5
    max_no_improvement = 300
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        
    population = []
    for _ in range(pop_size):
        h = h_init + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            break
            
        # Selection
        elite = population[:elite_size]
        
        # Crossover and mutation
        new_population = elite.copy()
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = elite[np.random.randint(0, elite_size)]
            parent2 = elite[np.random.randint(0, elite_size)]
            
            # Arithmetic crossover
            alpha = np.random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child = child + noise
                
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population
        
    return (best_h.tolist(), best_c5, n_points)
