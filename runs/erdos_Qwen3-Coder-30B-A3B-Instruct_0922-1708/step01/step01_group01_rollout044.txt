import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Reduced search approach: represent with few segments
    m = 5  # number of segments
    segment_values = [0, 0.5, 1]
    segment_count = len(segment_values)
    
    # Generate all combinations for segments
    combinations = list(product(segment_values, repeat=m))
    best_c5 = float('inf')
    best_h = None
    
    for i, seg_vals in enumerate(combinations):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Create piecewise constant h from segments
        h_seg = np.zeros(n_points)
        segment_length = n_points // m
        remainder = n_points % m
        
        idx = 0
        for j, val in enumerate(seg_vals):
            length = segment_length + (1 if j < remainder else 0)
            h_seg[idx:idx+length] = val
            idx += length
            
        h_seg = project_h(h_seg, n_points)
        c5 = evaluate_h(h_seg, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_seg.copy()
    
    # Refine with gradient descent
    if best_h is not None:
        h = best_h.copy()
        best_h_refined = h.copy()
        best_c5_refined = evaluate_h(h, dx)
        learning_rate = 0.01
        patience = 0
        max_patience = 1000
        
        while time.time() - start_time < budget_s - 5:
            # Compute gradient by finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update step
            h -= learning_rate * grad
            h = project_h(h, n_points)
            
            # Evaluate
            new_c5 = evaluate_h(h, dx)
            if new_c5 < best_c5_refined:
                best_h_refined = h.copy()
                best_c5_refined = new_c5
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    break
                if patience % 1000 == 0:
                    learning_rate *= 0.5
            
        best_h = best_h_refined
        best_c5 = best_c5_refined
    
    return (best_h, best_c5, n_points)
