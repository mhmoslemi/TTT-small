import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sample(n_points, n_samples):
    """Generate Latin Hypercube Samples"""
    samples = []
    for _ in range(n_samples):
        # Generate uniform random samples
        u = np.random.rand(n_points)
        # Create LHS by sorting and scaling
        idx = np.argsort(u)
        sample = (idx + np.random.rand(n_points)) / n_points
        samples.append(sample)
    return np.array(samples)

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal pattern
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1.0 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
        
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Phase 1: Latin Hypercube Sampling
    n_samples = 50
    n_elites = 20
    samples = lhs_sample(n_points, n_samples)
    sample_c5 = []
    
    for s in samples:
        s = project_h(s, n_points)
        c5 = evaluate_h(s, dx)
        sample_c5.append(c5)
    
    # Select best samples
    indices = np.argsort(sample_c5)[:n_elites]
    elites = [project_h(samples[i], n_points) for i in indices]
    elite_c5 = [sample_c5[i] for i in indices]
    
    # Phase 2: Evolution Strategy
    n_offspring = 5
    max_generations = 5000
    generation = 0
    
    while time.time() - start_time < 0.7 * budget_s and generation < max_generations:
        offspring = []
        for elite in elites:
            for _ in range(n_offspring):
                # Add small random noise
                mutated = elite + np.random.normal(0, 0.05, n_points)
                mutated = project_h(mutated, n_points)
                offspring.append(mutated)
        
        # Evaluate offspring
        offspring_c5 = [evaluate_h(o, dx) for o in offspring]
        combined = elites + offspring
        combined_c5 = elite_c5 + offspring_c5
        
        # Keep best
        indices = np.argsort(combined_c5)[:n_elites]
        elites = [combined[i] for i in indices]
        elite_c5 = [combined_c5[i] for i in indices]
        
        # Update best
        if elite_c5[0] < best_c5:
            best_h = elites[0].copy()
            best_c5 = elite_c5[0]
        
        generation += 1
    
    # Phase 3: Gradient Descent
    h = best_h.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s - 10:
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with sign of gradient
        h -= learning_rate * np.sign(grad)
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (best_h, best_c5, n_points)
