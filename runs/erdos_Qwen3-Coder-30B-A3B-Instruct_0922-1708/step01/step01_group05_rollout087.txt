import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h from initial_h_values or create from segments
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create piecewise constant h with alternating segments to start
        h = np.zeros(n_points)
        segments = 5
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = (i + 1) * segment_size if i < segments - 1 else n_points
            h[start:end] = 1 if i % 2 == 0 else 0
        # Adjust to meet sum constraint
        h = h * (target_sum / np.sum(h))
    
    # Helper functions
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def project_h(h_vals):
        # Clamp to [0, 1] and rescale to target sum
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals
    
    # Simulated Annealing Phase
    best_h = h.copy()
    best_C = compute_C(best_h)
    T = 1.0
    alpha = 0.99
    T_min = 1e-6
    iteration = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.66 and T > T_min:
        # Generate neighbor
        h_new = best_h.copy()
        # Perturb some part of h
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        h_new[idx] += delta
        h_new = project_h(h_new)
        
        # Compute acceptance probability
        C_new = compute_C(h_new)
        if C_new < best_C:
            best_h = h_new.copy()
            best_C = C_new
            no_improvement_count = 0
        else:
            if np.random.rand() < np.exp(-(C_new - best_C) / T):
                best_h = h_new.copy()
                best_C = C_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        # Update temperature
        T *= alpha
        if no_improvement_count > 100:
            T *= 0.95  # Cool faster if stagnated
        iteration += 1

    # Gradient Descent Phase (Local Refinement)
    h = best_h.copy()
    lr = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    
    def estimate_gradient(h_vals, eps=1e-4):
        grad = np.zeros_like(h_vals)
        base_C = compute_C(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - base_C) / eps
        return grad
    
    for i in range(200):  # Max 200 steps
        if time() - start_time > budget_s - 1:
            break
        grad = estimate_gradient(h)
        h -= lr * grad
        h = project_h(h)
        C = compute_C(h)
        if C < best_C:
            best_C = C
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                lr *= 0.5
                no_improvement_count = 0

    return (best_h.tolist(), best_C, n_points)
