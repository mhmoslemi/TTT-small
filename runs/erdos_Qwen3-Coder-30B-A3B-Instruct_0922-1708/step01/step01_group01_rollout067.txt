import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Create a simple smooth initial function
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Phase 1: Exhaustive search on reduced representation
    # Use segment-based approach for efficient exploration
    m = 5  # number of segments
    segment_values = [0, 0.5, 1]
    segment_combinations = list(product(segment_values, repeat=m))
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Evaluate a subset of combinations to save time
    subset_size = min(50, len(segment_combinations))
    sampled_combinations = np.random.choice(len(segment_combinations), subset_size, replace=False)
    
    for idx in sampled_combinations:
        # Create h from segment values
        combination = segment_combinations[idx]
        h_temp = np.zeros(n_points)
        segment_len = n_points // m
        remainder = n_points % m
        
        pos = 0
        for i in range(m):
            # Adjust segment length to distribute remainder
            actual_len = segment_len + (1 if i < remainder else 0)
            h_temp[pos:pos+actual_len] = combination[i]
            pos += actual_len
        
        # Project and evaluate
        h_temp = project_h(h_temp, n_points)
        c5 = evaluate_h(h_temp, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_temp.copy()
    
    # Phase 2: Gradient descent refinement of top candidates
    # If we still have time, refine our best solution
    if time.time() - start_time < 0.9 * budget_s:
        max_refinements = 10
        remaining_time = 0.1 * budget_s
        
        # Refine the best_h found
        h = best_h.copy()
        learning_rate = 0.01
        patience = 0
        max_patience = 1000
        last_improvement = 0
        
        for step in range(max_patience):
            if time.time() - start_time > budget_s:
                break
                
            # Compute gradient
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h -= learning_rate * grad
            h = project_h(h, n_points)
            
            # Evaluate
            new_c5 = evaluate_h(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                last_improvement = step
            else:
                patience += 1
                if patience > 100:
                    learning_rate *= 0.5
                    patience = 0
                    if learning_rate < 1e-6:
                        break
    
    return (best_h, best_c5, n_points)
