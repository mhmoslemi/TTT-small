import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Try to use the initial construction
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create an initial guess
        n_points = 70
        h_start = np.full(n_points, 0.5)
        # Make a tailored initial construction for better C5
        # High values in the middle to decrease overlaps
        h_start[n_points//2 - 5:n_points//2 + 5] = 0.8
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def project_h(h):
        """Ensure h is in [0,1] and integral = 1"""
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def smooth_h(h, sigma=2.0):
        """Apply Gaussian smoothing to make it piecewise continuous"""
        h_smooth = np.zeros_like(h)
        for i in range(len(h)):
            for j in range(len(h)):
                dist = abs(i - j)
                weight = np.exp(-0.5 * (dist / sigma)**2)
                h_smooth[i] += h[j] * weight
        return h_smooth / np.sum(h_smooth) * (n_points / 2)
    
    # Optimizer approach: simulated annealing combined with local search
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    # Parameters for the optimization
    temperature = 0.1
    cooling_rate = 0.999
    min_temp = 1e-6
    no_improvement_limit = 50
    
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < no_improvement_limit:
        iteration += 1
        
        # Create neighbor solution with some randomness
        h_trial = best_h.copy()
        
        # Perturb a portion of the values
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for i in indices:
            # Gaussian perturbation with adaptive step size
            h_trial[i] += np.random.normal(0, 0.05)
        
        # Apply smoothing to reduce oscillations
        h_trial = smooth_h(h_trial, sigma=2.0)
        
        # Project to feasible region
        h_trial = project_h(h_trial)
        
        # Evaluate
        c5_trial = evaluate(h_trial)
        
        # Accept or reject
        if c5_trial < best_c5 or np.random.rand() < np.exp((best_c5 - c5_trial) / temperature):
            best_h = h_trial.copy()
            best_c5 = c5_trial
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Reduce temperature
        temperature = max(min_temp, temperature * cooling_rate)
    
    # Final validity check
    final_h = project_h(best_h)
    final_c5 = evaluate(final_h)
    
    # Ensure no more than 1000 points
    if n_points > 1000:
        n_points = 1000
        final_h = final_h[:n_points]
        final_c5 = evaluate(final_h)
    
    return (final_h, final_c5, n_points)
