import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    
    # Default initialization: symmetric construction based on Gaussian-like bumps
    x = np.linspace(0, 2, n_points)
    # Two bumps to create a symmetric and smooth function
    h = 0.5 * (1 + np.sin(np.pi * (x - 0.5))) * (1 + np.sin(np.pi * (x - 1.5)))
    h = project_h(h, n_points)
    return h

def enforce_symmetry(h, n_points):
    """Make h symmetric around x=1."""
    if n_points % 2 == 1:
        k = (n_points - 1) // 2
        # Preserve center point
        for i in range(k):
            avg = (h[i] + h[2*k - i]) / 2
            h[i] = avg
            h[2*k - i] = avg
    else:
        k = n_points // 2
        # For even n_points, we need to carefully handle the symmetric pairs
        for i in range(k):
            avg = (h[i] + h[2*k - 1 - i]) / 2
            h[i] = avg
            h[2*k - 1 - i] = avg
    return h

def compute_exact_gradient(h, max_shift):
    """Compute exact gradient at the shift that achieves max overlap."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # Compute gradient contribution from all positions to the max shift
    # For shift k, dx * (1 - h[i + k]) for all i
    # We need to handle the wrap-around via indexing
    
    # The full convolution conv[i + j] = sum_{k} h[i+k] * (1 - h[j+k])
    # i is the position in the first array ([:]), j is position in second (shifted)
    # The shift is fixed to max_shift, so it's effectively conv[i + max_shift]
    # Gradient w.r.t. h[i] for shift max_shift is (1 - h[i + max_shift])
    # Careful with wraparound
    for i in range(n):
        shifted_index = i + max_shift
        if shifted_index >= n:
            shifted_index -= n  # wrap around for positive shift
        if shifted_index < 0:
            shifted_index += n  # wrap around for negative shift
        grad[i] = (1 - h[shifted_index]) * dx
    
    return grad

def find_worst_shift(h):
    """Find shift that causes maximum overlap."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(conv)
    shift = max_idx - (n - 1)
    return shift

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function with symmetric initialization and gradient descent."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # odd for symmetry
    h = initialize_h(n_points)
    h = enforce_symmetry(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Gradient descent with symmetric updates
    step_size = 0.1
    max_no_improvement = 10000
    no_improvement = 0
    
    while time.time() - start_time < budget_s * 0.95:
        # Find shift that maximizes overlap
        shift = find_worst_shift(best_h)
        grad = compute_exact_gradient(best_h, shift)
        
        # Perform update
        h_new = best_h - step_size * grad
        h_new = project_h(h_new, n_points)
        h_new = enforce_symmetry(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improvement = 0
            # Reduce step size if improved
            step_size *= 0.9
        else:
            no_improvement += 1
            step_size *= 1.1
            if step_size > 1.0:
                step_size = 1.0
                
            if no_improvement > max_no_improvement:
                break
    
    # Final validation
    best_h = enforce_symmetry(best_h, n_points)
    final_c5 = evaluate_c5(best_h)
    
    # Use the best found solution
    return (best_h, final_c5, n_points)
