import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def make_symmetric(h, n_points):
    """Make h symmetric around x=1."""
    k = (n_points - 1) // 2  # Midpoint index
    h_sym = h.copy()
    for i in range(k):
        avg = (h[i] + h[n_points - 1 - i]) / 2
        h_sym[i] = avg
        h_sym[n_points - 1 - i] = avg
    return h_sym

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # This is odd to allow symmetry
    k = (n_points - 1) // 2  # Midpoint index
    
    # Initialize h to a symmetric shape
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    # Use two Gaussian bumps for initial structure
    h += 0.5 * (1 + np.sin(np.pi * (x - 0.5)))  # Centered at 0.5
    h += 0.5 * (1 + np.sin(np.pi * (x - 1.5)))  # Centered at 1.5
    h = project_h(h, n_points)
    h = make_symmetric(h, n_points)
    
    # Ensure the sum constraint is met exactly
    h = project_h(h, n_points)  # May alter symmetry slightly, re-apply symmetry
    h = make_symmetric(h, n_points)
    
    # Optimization loop
    learning_rate = 0.1
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    patience = 0
    max_patience = 1000
    min_lr = 1e-6
    
    while time.time() - start_time < budget_s * 0.95:
        # Evaluate current state
        c5 = evaluate_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
            patience = 0
        else:
            patience += 1
            if patience >= max_patience:
                break
                
        # Find the shift that produces the maximum correlation
        conv = np.correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute gradient for the current max shift
        grad = np.zeros(n_points)
        if shift >= 0:
            for i in range(n_points - shift):
                grad[i] = (1 - best_h[i + shift])
        else:
            for i in range(-shift, n_points):
                grad[i] = (1 - best_h[i + shift])
        
        # Update step
        h_new = best_h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        h_new = make_symmetric(h_new, n_points)
        
        # Check if update helps
        c5_new = evaluate_c5(h_new)
        if c5_new < c5:
            best_h = h_new
            # Reduce learning rate
            learning_rate *= 0.95
            if learning_rate < min_lr:
                learning_rate = min_lr
        else:
            # Increase learning rate for next time
            learning_rate *= 1.05
            if learning_rate > 1.0:
                learning_rate = 1.0
        if patience > 100:
            patience = 0
    
    # Final validation
    final_h = make_symmetric(best_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
