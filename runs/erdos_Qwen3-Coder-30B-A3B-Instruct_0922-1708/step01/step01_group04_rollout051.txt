import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def objective(h):
        return evaluate(h)
    
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    def project_h(h):
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def smooth(h, window=3):
        if window < 2:
            return h
        # Apply moving average smoothing
        weights = np.ones(window) / window
        h_smooth = np.convolve(h, weights, mode='same')
        return h_smooth
    
    def perturb(h, strength=0.1):
        h_new = h.copy()
        # Perturb a fraction of points
        n_perturb = max(1, int(n_points * 0.15))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            delta = np.random.uniform(-strength, strength)
            h_new[i] += delta
        return project_h(h_new)
    
    # Optimization setup
    constraints = {'type': 'eq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s and no_improvement_count < max_no_improvement:
        iterations += 1
        
        # Try gradient-based optimization
        try:
            result = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 20, 'disp': False},
                callback=lambda x: None  # No callback needed
            )
            if result.success:
                h_opt = project_h(result.x)
                c5_opt = evaluate(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                # If optimization fails, fallback to perturbation
                h_trial = perturb(best_h, strength=0.05)
                h_trial = smooth(h_trial, window=3)
                c5_trial = evaluate(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
        except Exception:
            # Fallback to simple perturbation if optimization fails
            h_trial = perturb(best_h, strength=0.05)
            h_trial = smooth(h_trial, window=3)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Periodic global perturbation
        if iterations % 10 == 0:
            h_trial = perturb(best_h, strength=0.1)
            h_trial = smooth(h_trial, window=5)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
    
    return (best_h, best_c5, n_points)
