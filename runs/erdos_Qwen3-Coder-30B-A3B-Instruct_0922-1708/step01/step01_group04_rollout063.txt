from scipy.optimize import minimize
import time

        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

        # Create segments
        segment_width = n_points // n_segments
        if segment_width * n_segments < n_points:
            segment_width += 1
        
        segments = []
        for i in range(n_segments):
            start_idx = i * segment_width
            end_idx = min((i+1) * segment_width, n_points)
            seg_vals = h[start_idx:end_idx]
            if len(seg_vals) > 0:
                segment_avg = np.mean(seg_vals)
                segments.append(segment_avg)
            else:
                segments.append(0.5)
        
        # Normalize to target sum
        target_sum = n_segments / 2.0
        actual_sum = np.sum(segments)
        if actual_sum > 0:
            segments = np.array(segments) * target_sum / actual_sum
        
        # Reconstruct signal from segments
        h_reconstructed = np.zeros(n_points)
        for i, seg_value in enumerate(segments):
            start_idx = i * segment_width
            end_idx = min((i+1) * segment_width, n_points)
            h_reconstructed[start_idx:end_idx] = seg_value
        
        return h_reconstructed, segments

    def smooth_h(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth

    def perturb_segments(segments, perturb_fraction=0.1, delta=0.1):
        segments_new = segments.copy()
        n_to_perturb = max(1, int(len(segments) * perturb_fraction))
        indices = np.random.choice(len(segments), n_to_perturb, replace=False)
        for i in indices:
            segments_new[i] += np.random.uniform(-delta, delta)
            segments_new[i] = np.clip(segments_new[i], 0, 1)
        
        return segments_new

    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    # Set up optimization parameters

    resolution_increase_iter = 0

            resolution_increase_iter = 0

    # Final evaluation
    final_c5 = evaluate(best_h)
    
    return (best_h, final_c5, n_points)
