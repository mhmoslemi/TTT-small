import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a triangular distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to higher resolution
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(h)), h)
        h = h_interp
    else:
        # Create a triangular-like distribution
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1) * 2
        h = np.clip(h, 0, 1)
    
    # Normalize to integrate to 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Parameters for Adam
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    start_time = time.time()
    iteration = 0
    
    def compute_c5_and_grad(h):
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_k = np.argmax(corr)
        
        # Compute gradients for the maximum correlation
        grad = np.zeros_like(h)
        # Calculate contribution to the gradient from the maximum lag
        for i in range(n_points):
            # This is a simplified gradient computation at the peak
            # A full gradient would require more careful handling
            # Simple finite differencing with step size to estimate grad
            eps = 1e-5
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        return c5, grad
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5 and gradient
        c5, grad = compute_c5_and_grad(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Update using Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Early termination check every 10 iterations
        if iteration % 10 == 0:
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final computation for best solution
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
