import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.asarray(h)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=5):
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Segment-based optimization approach
    n_segments = 10
    n_segment_points = n_points // n_segments
    if n_segment_points < 1:
        n_segment_points = 1
        n_segments = n_points
        
    h_best = h_start.copy()
    c5_best = evaluate(h_best)
    
    # Initial segment averages with variable spacing
    c = np.ones(n_segments) * 0.5
    c[0] = 0.2
    c[-1] = 0.8
    
    # Normalize segment averages to match integral constraint
    c = c * (n_points / 2) / (n_segments * n_segment_points / 2)
    
    # Trust-region optimization
    trust_radius = 0.1
    max_no_improvement = 100
    no_improvement = 0
    start_time = time.time()
    
    # Simulated annealing parameters
    T_start = 0.1
    T = T_start
    cooling_rate = 0.995
    
    while time.time() - start_time < budget_s and no_improvement < max_no_improvement:
        # Propose new segment averages
        c_new = c.copy()
        # Perturb some segments
        perturb_indices = np.random.choice(n_segments, max(1, n_segments // 4), replace=False)
        for i in perturb_indices:
            delta = np.random.uniform(-trust_radius, trust_radius)
            c_new[i] += delta
        
        # Project to feasible region
        c_new = np.clip(c_new, 0, 1)
        # Ensure correct integral
        c_new = c_new * (n_points / 2) / np.sum(c_new)
        
        # Expand to full function
        h_new = np.array([c_new[i // n_segment_points] for i in range(n_points)])
        
        # Apply smoothing
        h_new = smooth(h_new, window=5)
        h_new = project_h(h_new)
        
        # Evaluate
        c5_new = evaluate(h_new)
        
        # Acceptance criterion with simulated annealing
        if c5_new < c5_best or np.random.rand() < np.exp((c5_best - c5_new) / T):
            c = c_new
            h_best = h_new
            c5_best = c5_new
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Update trust region size
        if no_improvement <= 20:
            trust_radius = min(0.5, trust_radius * 1.05)
        else:
            trust_radius = max(0.001, trust_radius * 0.95)
            
        # Annealing
        T = max(0.001, T * cooling_rate)
    
    # Final formatting
    h_out = h_best.copy()
    c5_out = c5_best
    
    # Apply final smoothing
    h_out = smooth(h_out, window=5)
    h_out = project_h(h_out)
    c5_out = evaluate(h_out)
    
    return (h_out, c5_out, n_points)
