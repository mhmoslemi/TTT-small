def evaluate_c5(h_vec, dx):
    """Evaluate the C5 bound for a given h vector."""
    conv = correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(conv) * dx

def project_onto_feasible(h_vec, target_sum):
    """Project h onto the feasible set with sum = target_sum."""
    h_vec = np.clip(h_vec, 0, 1)
    current_sum = np.sum(h_vec)
    if current_sum > 0:
        h_vec = h_vec * target_sum / current_sum
    return h_vec
