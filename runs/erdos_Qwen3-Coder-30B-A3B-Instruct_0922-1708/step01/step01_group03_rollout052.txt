import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a combination of bump functions
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    # Add multiple gaussian bumps to create more complex pattern
    for center in [0.3, 0.7, 1.3, 1.7]:
        h += np.exp(-50 * (x - center)**2)
    h = project_h(h, n_points)
    return h

def compute_exact_gradient(h, max_shift):
    """Compute exact gradient for a given shift configuration."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # The gradient of C5 w.r.t. h_i depends on the correlation structure
    # For shift k, the derivatives involve terms where h_i appears: 
    # C5 = sum_i h_i * (1 - h_{i+k}) for all valid i
    # dC5/dh_i = 1 - h_{i+k} when i+k is within bounds
    
    # Since we are using mode="full", we get shifts from -n+1 to n-1
    # The actual shift that maximizes C5 is at max_shift - (n-1)
    actual_shift = max_shift - (n - 1) 
    
    # We now compute gradient for the shift that gives max C5
    for i in range(n):
        j = i + actual_shift
        if 0 <= j < n:
            grad[i] = -dx * (1 - h[j])
        # Else the term doesn't contribute to gradient
    return grad

def symmetric_project(h):
    """Project h to be symmetric around midpoint."""
    n = len(h)
    if n % 2 == 0:
        k = n // 2 - 1  # midpoint indices
        for i in range(k + 1):
            avg = (h[i] + h[n-1-i]) / 2
            h[i] = avg
            h[n-1-i] = avg
    else:
        k = n // 2  # midpoint index
        for i in range(k):
            avg = (h[i] + h[n-1-i]) / 2
            h[i] = avg
            h[n-1-i] = avg
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function with symmetric gradient descent."""
    np.random.seed(seed)
    n_points = 45  # odd number allows for symmetric structure
    start_time = time.time()
    
    # Phase 1: Initialize with symmetric structure
    h = initialize_h(n_points)
    h = symmetric_project(h)
    h = project_h(h, n_points)  # ensure sum constraint
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Main loop parameters
    learning_rate = 0.1
    max_no_improvement = 10000
    iter_no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:  # reserve some time for final evaluation
        # Find the current shift that gives maximum overlap
        conv = np.correlate(best_h, 1 - best_h, mode="full")
        max_shift = np.argmax(conv)
        shift_from_full = max_shift - (n_points - 1)  # actual shift value
        
        # Compute exact gradient
        grad = compute_exact_gradient(best_h, max_shift)
        
        # Update with GD step
        h_new = best_h - learning_rate * grad
        h_new = symmetric_project(h_new)
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        
        # Update if better or accept with decay
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            # Anneal learning rate if improvement
            if learning_rate > 1e-6:
                learning_rate *= 0.9995
        else:
            iter_no_improvement += 1
            # Increase learning rate if no improvement to explore more
            if learning_rate < 1:
                learning_rate *= 1.0005
        
        # Early stopping
        if iter_no_improvement > max_no_improvement:
            break
            
        iteration += 1
        
        if iteration % 500 == 0:
            # Periodic validation
            test_h = symmetric_project(best_h.copy())
            test_h = project_h(test_h, n_points)
            test_c5 = evaluate_c5(test_h)
            if test_c5 < best_c5:
                best_h = test_h
                best_c5 = test_c5
    
    # Final validation
    final_h = symmetric_project(best_h.copy())
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    # Return best available result
    return (final_h, final_c5, n_points)
