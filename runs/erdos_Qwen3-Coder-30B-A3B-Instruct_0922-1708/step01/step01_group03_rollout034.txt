```python
import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Access initial_h_values directly
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default symmetric initialization: triangular pulse
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    # Create symmetric triangle
    for i in range(n_points):
        x_i = x[i]
        if x_i <= 1:
            h[i] = x_i
        else:
            h[i] = 2 - x_i
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45  # odd so we have a central point for symmetry
    h = initialize_h(n_points)
    
    start_time = time.time()
    
    # PSO parameters
    swarm_size = 30
    max_iter = 1000
    inertia_weight = 0.9
    inertia_decay = 0.99
    cognitive_weight = 2.0
    social_weight = 2.0
    velocity_max = 0.2
    
    # Initialize swarm
    particles = []
    velocities = []
    pbests = []
    pbest_values = []
    
    for _ in range(swarm_size):
        particle = h.copy()
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        pbests.append(particle.copy())
        pbest_values.append(evaluate_c5(particle))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbests[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO main loop
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s * 0.8 and iteration < max_iter:
        iteration += 1
        # Adjust inertia weight
        inertia_weight *= inertia_decay
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            velocities[i] = (
                inertia_weight * velocities[i] +
                cognitive_weight * r1 * (pbests[i] - particles[i]) +
                social_weight * r2 * (gbest - particles[i])
            )
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to valid domain
            particles[i] = project_h(particles[i], n_points)
            
            # Evaluate
            current_value = evaluate_c5(particles[i])
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbests[i] = particles[i].copy()
                pbest_values[i] = current_value
                # Update global best if needed
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
        
        if no_improvement_count > max_no_improvement:
            break
    
    # Gradient descent refinement
    best_h = gbest.copy()
    best_c5 = gbest_value
    
    # Gradient descent for the best PSO solution
    max_gd_iter = 1000
    learning_rate = 0.05
    
    def compute_gradient(h, shift):
        """Compute gradient w.r.t. h for a given shift."""
        n = len(h)
        dx = 2.0 / n
        # correlation at shift
        # overlap contribution from h[i] to h[i+shift] is h[i] * (1-h[i+shift])
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= (i + shift) < n:
                grad[i] += -dx  # derivative of h[i] * (1 - h[i+shift]) w.r.t. h[i] is -dx
        return grad
    
    def find_worst_shift(h):
        """Find shift that maximizes the correlation."""
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (len(h) - 1)
        return shift
    
    iter_no_improvement = 0
    for _ in range(max_gd_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        shift = find_worst_shift(best_h)
        grad = compute_gradient(best_h, shift)
        new_h = best_h - learning_rate * grad
        new_h = project_h(new_h, n_points)
        new_c5 = evaluate_c5(new_h)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            if iter_no_improvement > 100:
                break
    
    return (best_h, best_c5, n_points)
```
