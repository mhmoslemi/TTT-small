import scipy.optimize as so
from time import time
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h from initial_h_values or create from scratch
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a piecewise constant function with alternating 0s and 1s
        h = np.zeros(n_points)
        segments = 5
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = start + segment_size if i < segments - 1 else n_points
            h[start:end] = 1.0 if i % 2 == 0 else 0.0
        
        # Add some smoothing
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
    
    # Normalize
    h = h * (target_sum / np.sum(h))

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    T_min = 1e-6
    max_iter_sa = 50000
    max_iter_gd = 200

    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))

    def neighbor(h_vals, T):
        h_new = h_vals.copy()
        # Randomly perturb one element
        idx = np.random.randint(0, len(h_vals))
        delta = np.random.normal(0, T * 0.5)
        h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
        
        # Adjust other values to maintain sum
        diff = np.sum(h_new) - target_sum
        if abs(diff) > 1e-6:
            # Redistribute to other elements
            for i in range(len(h_vals)):
                if i != idx:
                    delta_i = -diff * 0.1
                    h_new[i] = np.clip(h_new[i] + delta_i, 0, 1)
                    diff += delta_i
                    if abs(diff) < 1e-6:
                        break
            h_new[idx] = np.clip(h_new[idx] - diff, 0, 1)
        return h_new

    # Optimization loop
    best_C = float('inf')
    best_h = h.copy()

    # Run Simulated Annealing
    sa_iter = 0
    while sa_iter < max_iter_sa and (time() - start_time) < budget_s - 1:
        h_new = neighbor(h, T)
        C_new = compute_C(h_new)
        C_old = compute_C(h)
        
        if C_new < C_old or np.random.rand() < np.exp(-(C_new - C_old) / T):
            h = h_new
            if C_new < best_C:
                best_C = C_new
                best_h = h.copy()
        
        T = max(T_min, T * alpha)
        sa_iter += 1

    # Gradient Descent Refinement
    lr = 0.01
    patience = 0
    best_C_gd = best_C
    h_refined = best_h.copy()

    def gradient_descent_step(h_vals, lr):
        h_vals = h_vals.copy()
        C_base = compute_C(h_vals)
        eps = 1e-5
        grad = np.zeros_like(h_vals)
        
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            # Ensure the sum constraint is maintained
            diff = np.sum(h_plus) - target_sum
            if abs(diff) > 1e-6:
                # Adjust others to maintain sum
                for j in range(len(h_vals)):
                    if j != i:
                        h_plus[j] = np.clip(h_plus[j] - diff * 0.1, 0, 1)
                        diff += diff * 0.1
                        if abs(diff) < 1e-6:
                            break
                h_plus[i] = np.clip(h_plus[i] - diff, 0, 1)
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C_base) / eps
        
        h_new = h_vals - lr * grad
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        return h_new

    for i in range(max_iter_gd):
        if (time() - start_time) >= budget_s - 1:
            break
        h_new = gradient_descent_step(h_refined, lr)
        C_new = compute_C(h_new)
        if C_new < best_C_gd:
            best_C_gd = C_new
            h_refined = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > 10:
                lr *= 0.5
                patience = 0
        if C_new < best_C:
            best_C = C_new
            best_h = h_new.copy()

    return (best_h.tolist(), best_C, n_points)
