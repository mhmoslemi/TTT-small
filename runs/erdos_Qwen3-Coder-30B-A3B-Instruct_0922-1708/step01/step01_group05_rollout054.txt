import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    max_iter = 1000
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter

    # Initialize particles and velocities in the space of h values
    particles = np.tile(h, (n_particles, 1))
    velocities = np.random.randn(n_particles, n_points) * 0.01
    personal_best = particles.copy()
    personal_best_scores = np.full(n_particles, float('inf'))
    
    # Global best
    global_best = h.copy()
    global_best_score = float('inf')
    
    # Function to compute C_5
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    # Function to ensure constraints
    def enforce_constraints(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals

    def optimize_once(h_vals):
        """Use scipy.optimize with bounds for local improvement"""
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}]
        
        result = so.minimize(
            lambda x: compute_C(enforce_constraints(x)),
            h_vals,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50}
        )
        return enforce_constraints(result.x)

    # PSO iteration
    iter_count = 0
    for iter in range(max_iter):
        if time() - start_time > budget_s - 2:
            break
            
        # Evaluate all particles and update personal bests
        for i in range(n_particles):
            h_particle = particles[i]
            score = compute_C(h_particle)
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = h_particle.copy()
            
            if score < global_best_score:
                global_best_score = score
                global_best = h_particle.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best[i] - particles[i]) +
                c2 * r2 * (global_best - particles[i])
            )
            particles[i] = particles[i] + velocities[i]
            particles[i] = enforce_constraints(particles[i])

        iter_count += 1

    # Local refinement
    if global_best_score < float('inf'):
        refined_best = global_best.copy()
        for _ in range(10):
            if time() - start_time > budget_s - 1:
                break
            refined = optimize_once(refined_best)
            if compute_C(refined) < compute_C(refined_best):
                refined_best = refined.copy()
            else:
                break
        global_best = refined_best

    return (global_best.tolist(), global_best_score, n_points)
