import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    # GA parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 5
    
    # Generate initial population
    population = []
    for _ in range(pop_size):
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_init = np.clip(h_init, 0, 1)
        population.append(h_init)
    
    # Add the initial solution if available
    if 'initial_h_values' in globals():
        population[0] = h.copy()
    
    while time.time() - start_time < budget_s - 1:
        if iteration % 100 == 0:
            # Check if we're stuck
            if iteration - last_improvement > 300:
                break

        iteration += 1
        
        # Evaluate population
        fitness = []
        for ind in population:
            correlation = np.correlate(ind, 1 - ind, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]

        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            last_improvement = iteration
            
        # Create new population
        new_population = population[:elite_size]  # Keep elite
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Selection (tournament)
            tournament_size = 5
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Crossover with random partner
            if np.random.rand() < crossover_rate:
                partner_idx = np.random.choice(pop_size, p=np.arange(pop_size, 0, -1) / np.sum(np.arange(pop_size, 0, -1)))
                parent1 = population[winner_idx]
                parent2 = population[partner_idx]
                
                # Arithmetic crossover
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = population[winner_idx].copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, 0.05)
            
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population
    
    return (best_h.tolist(), best_c5, n_points)
