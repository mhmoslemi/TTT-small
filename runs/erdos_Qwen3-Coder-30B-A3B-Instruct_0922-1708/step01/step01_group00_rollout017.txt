import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Simple gradient descent with momentum and adaptive learning rate
        eps = 1e-5
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate based on previous improvement
        lr = 0.05
        if iteration - last_improvement > 50:
            lr *= 0.9
            
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Add some random noise for diversity
        if iteration % 10 == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h_new = h_new + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
        h = h_new
        
        # Occasionally reset to the best solution to avoid local minima
        if iteration % 100 == 0 and iteration > 200:
            h = best_h.copy() + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        if iteration - last_improvement > 300:
            break
    
    return (best_h.tolist(), best_c5, n_points)
