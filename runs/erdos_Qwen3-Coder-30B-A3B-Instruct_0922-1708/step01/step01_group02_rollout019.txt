import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h)

        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Use differential evolution for global optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(x):
        # Project to feasibility
        x = project_h(x)
        return compute_c5(x)
    
    # Differential evolution search
    de_result = differential_evolution(
        objective, 
        bounds, 
        maxiter=200, 
        popsize=15, 
        recombination=0.7, 
        mutation=(0.5, 1.0),
        seed=seed,
        disp=False,
        polish=True
    )
    
    # Project final result
    final_h = project_h(de_result.x)
    final_c5 = compute_c5(final_h)
    
    # Refine with local optimization if time allows
    if (time.time() - start_time) < budget_s - 5:
        # Local refinement using L-BFGS-B
        from scipy.optimize import minimize
        
        # Prepare constraints for scipy
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        local_result = minimize(
            objective,
            final_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100}
        )
        
        refined_h = project_h(local_result.x)
        refined_c5 = compute_c5(refined_h)
        
        if refined_c5 < final_c5:
            final_h = refined_h
            final_c5 = refined_c5
    
    # Final validation
    final_h = project_h(final_h)  # Ensure correctness
    final_c5 = compute_c5(final_h)

    if (time.time() - start_time) >= budget_s - 1:
        # Try a simpler approach with a few more iterations to get better result
        # Binary search for better bound
        current_bound = best_c5

                    break

                c5_trial = compute_c5(h_trial)
                if c5_trial < current_bound:
                    current_bound = c5_trial
                    final_h = h_trial.copy()
        except:
            pass
    
    return (final_h.tolist(), final_c5, len(final_h))
