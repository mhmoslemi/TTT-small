    n_points = 100
    dx = 2.0 / n_points

        h = h / np.sum(h) * (n_points / 2)

        h[20:30] = 0.8
        h[70:80] = 0.8
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    last_improvement = time.time()
    
    # Gradient descent with adaptive step size
    learning_rate = 0.02
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = time.time()
        
        # Check for stagnation
        if time.time() - last_improvement > 300:
            break
            
        # Compute approximate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update with gradient descent
        h_new = h - learning_rate * grad
        
        # Clip values and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally adjust learning rate
        if iteration % 500 == 0 and iteration > 0:
            learning_rate *= 0.9
