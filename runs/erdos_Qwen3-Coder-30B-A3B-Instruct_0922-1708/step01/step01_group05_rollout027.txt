import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO Parameters
    n_particles = 100
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_iters = 10000
    
    # Initialize PSO particles with random perturbations
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest = None
    gbest_val = float('inf')
    
    # Randomly initialize particles around the starting h
    for _ in range(n_particles):
        h_pert = h + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (target_sum / np.sum(h_pert))
        particles.append(h_pert.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        val = compute_C(h_pert)
        pbest.append(h_pert.copy())
        pbest_vals.append(val)
        if val < gbest_val:
            gbest = h_pert.copy()
            gbest_val = val
    
    # PSO main loop
    iter_count = 0
    while iter_count < max_iters and time() - start_time < budget_s - 1:
        iter_count += 1
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (pbest[i] - particles[i]) +
                social * r2 * (gbest - particles[i])
            )
            # Update position
            particles[i] += velocities[i]
            # Keep in bounds
            particles[i] = np.clip(particles[i], 0, 1)
            # Rescale to meet constraint
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            
            # Evaluate fitness
            current_val = compute_C(particles[i])
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
            
            # Prevent stagnation
            if iter_count % 500 == 0:
                for j in range(n_particles):
                    velocities[j] = np.random.uniform(-0.1, 0.1, n_points)
    
    # Refinement stage: gradient descent from the best PSO solution
    best_h = gbest.copy()
    lr = 0.01
    no_improvement_count = 0
    best_C = gbest_val
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))

    def compute_C_grad(h_vals):
        # Analytical gradient approximation
        eps = 1e-5
        grad = np.zeros_like(h_vals)
        C_base = compute_C(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C_base) / eps
        return grad

    # Gradient descent refinement
    max_grad_iter = 500
    for i in range(max_grad_iter):
        if time() - start_time >= budget_s - 1:
            break
            
        grad = compute_C_grad(best_h)
        best_h -= lr * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        
        new_C = compute_C(best_h)
        if new_C < best_C:
            best_C = new_C
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                lr *= 0.5
                no_improvement_count = 0

    # Final evaluation
    final_C = compute_C(best_h)
    
    return (best_h.tolist(), final_C, n_points)
