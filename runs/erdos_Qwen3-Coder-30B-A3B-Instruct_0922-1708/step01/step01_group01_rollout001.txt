import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Simulated Annealing with Gradient Descent
    start_time = time.time()
    
    # Annealing parameters
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    step_size = 0.1
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Annealing loop
    for iteration in range(max_iter):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Mutate
        h_new = h + np.random.uniform(-step_size, step_size, n_points)
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            h = h_new.copy()
            best_h = h_new.copy()
            best_c5 = c5_new
        else:
            # Accept with probability
            delta = c5_new - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new.copy()
        
        # Cool
        T *= alpha
        
        # Occasionally apply gradient descent
        if iteration % 50 == 0 and iteration > 0:

            # Approximate gradient via finite differences
            eps = 1e-5

                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update with gradient (adaptive learning rate)
            eta = 0.01
            h_new = h - eta * np.sign(grad) * step_size
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < best_c5:
                h = h_new.copy()
                best_h = h.copy()
                best_c5 = c5_new
    
    # Final refinement with gradient descent
    if time.time() - start_time < 0.1 * budget_s:

            if time.time() - start_time > budget_s:

            for i in range(n_points):

                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
