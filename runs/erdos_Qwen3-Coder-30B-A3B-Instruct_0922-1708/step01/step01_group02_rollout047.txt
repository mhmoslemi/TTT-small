import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available

        n_points = len(h)
    except NameError:
        # Create a symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Apply a central plateau
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Particle Swarm Optimization
    n_particles = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(n_particles):
        pos = np.random.rand(n_points) * 0.5 + 0.25  # Initial random in [0.25, 0.75]
        pos = project_h(pos)
        particles.append(pos.copy())
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
        pbest_positions.append(pos.copy())
        pbest_values.append(compute_c5(pos))
    
    # Global best
    gb_index = np.argmin(pbest_values)
    gbest_position = pbest_positions[gb_index].copy()
    gbest_value = pbest_values[gb_index]
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    inertia = w_start
    
    iter_no_improve = 0
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 1:
            break

        # Update each particle
        for i in range(n_particles):
            pos = particles[i]
            vel = velocities[i]
            pbest_pos = pbest_positions[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            inertia = w_start - (w_start - w_end) * iteration / max_iter
            vel = (inertia * vel 
                   + c1 * r1 * (pbest_pos - pos)
                   + c2 * r2 * (gbest_position - pos))
            
            # Update position
            pos = pos + vel
            pos = project_h(pos)
            
            # Evaluate
            current_c5 = compute_c5(pos)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_positions[i] = pos.copy()
                pbest_values[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_value:
                    gbest_position = pos.copy()
                    gbest_value = current_c5
                    iter_no_improve = 0
                else:
                    iter_no_improve += 1
            
            # Save new values
            particles[i] = pos
            velocities[i] = vel
        
        # Early stopping
        if iter_no_improve > 1000:
            break
            
    # Return final best
    return (gbest_position.tolist(), gbest_value, n_points)
