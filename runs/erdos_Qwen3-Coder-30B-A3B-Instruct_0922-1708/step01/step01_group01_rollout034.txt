import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using hybrid PSO and gradient descent"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    swarm_size = n_particles
    stagnation_count = 0
    stagnation_limit = 500
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            # 10% random init, 90% perturb existing
            if np.random.rand() < 0.1:
                particle = np.random.rand(n_points)
            else:
                particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_h(particle, n_points)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        score = evaluate_h(particle, dx)
        pbest.append(particle.copy())
        pbest_scores.append(score)
    
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    start_time = time.time()
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < 0.7 * budget_s:
        iteration += 1
        
        # Update particles
        for i in range(swarm_size):
            particle = particles[i]
            velocity = velocities[i]
            pbest_pos = pbest[i]
            rand1 = np.random.rand(n_points)
            rand2 = np.random.rand(n_points)
            
            # Update velocity
            velocity = (w * velocity + 
                       c1 * rand1 * (pbest_pos - particle) + 
                       c2 * rand2 * (gbest - particle))
            
            # Update position
            particle = particle + velocity
            particle = project_h(particle, n_points)
            
            # Evaluate
            new_score = evaluate_h(particle, dx)
            
            # Update personal best
            if new_score < pbest_scores[i]:
                pbest[i] = particle.copy()
                pbest_scores[i] = new_score
                
                # Update global best
                if new_score < gbest_score:
                    gbest = particle.copy()
                    gbest_score = new_score
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        
        # Adapt swarm size if necessary
        if stagnation_count > stagnation_limit and swarm_size > 20:
            swarm_size = max(20, int(swarm_size * 0.8))
            # Reinitialize the swarm
            particles = []
            velocities = []
            pbest = []
            pbest_scores = []
            for i in range(swarm_size):
                if i == 0:
                    particle = h.copy()
                else:
                    if np.random.rand() < 0.1:
                        particle = np.random.rand(n_points)
                    else:
                        particle = h + np.random.normal(0, 0.05, n_points)
                    particle = project_h(particle, n_points)
                particles.append(particle)
                velocities.append(np.zeros(n_points))
                score = evaluate_h(particle, dx)
                pbest.append(particle.copy())
                pbest_scores.append(score)
            gbest_idx = np.argmin(pbest_scores)
            gbest = pbest[gbest_idx].copy()
            gbest_score = pbest_scores[gbest_idx]
            stagnation_count = 0
        
        # Early stopping
        if stagnation_count > stagnation_limit * 2:
            break
    
    # Refine with gradient descent
    h_refined = gbest.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s - 1 and patience < max_patience:
        # Find worst shift
        corr = correlate(h_refined, 1 - h_refined, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)

            h_minus = h_refined.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with gradient descent
        h_prev = h_refined.copy()
        h_refined -= learning_rate * grad
        h_refined = project_h(h_refined, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h_refined, dx)
        if new_c5 < gbest_score:
            gbest_score = new_c5
            gbest = h_refined.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
    
    return (gbest, gbest_score, n_points)
