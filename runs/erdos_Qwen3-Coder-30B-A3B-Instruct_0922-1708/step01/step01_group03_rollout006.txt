import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h):
    """Compute gradient of C5 with respect to h values."""
    # For gradient, we'll estimate using finite differences
    eps = 1e-5
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # Get current C5 value
    current_c5 = evaluate_c5(h)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n)
        c5_plus = evaluate_c5(h_plus)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_h(h_minus, n)
        c5_minus = evaluate_c5(h_minus)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45  # odd to support symmetry
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: Particle Swarm Optimization
    # Setup for PSO
    swarm_size = 30
    inertia_weight = 0.9
    cognitive_weight = 2.0
    social_weight = 2.0
    velocity_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_values = []
    
    for _ in range(swarm_size):
        particle = h.copy()
        particle = project_h(particle, n_points)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        personal_best.append(particle.copy())
        personal_best_values.append(evaluate_c5(particle))
    
    # Global best
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    max_iter = 2500
    iter_no_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.75:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            inertia = inertia_weight * velocities[i]
            cognitive = cognitive_weight * r1 * (personal_best[i] - particles[i])
            social = social_weight * r2 * (global_best - particles[i])
            
            velocities[i] = inertia + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to valid range and rescale
            particles[i] = project_h(particles[i], n_points)
            
            # Evaluate fitness
            fitness = evaluate_c5(particles[i])
            
            # Update personal best
            if fitness < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = fitness
                
                # Update global best if needed
                if fitness < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = fitness
                    iter_no_improvement = 0
                else:
                    iter_no_improvement += 1
        
        # Adjust inertia weight
        inertia_weight = 0.9 - (0.5 * iteration / max_iter)
        
        # Early termination check
        if iter_no_improvement > max_no_improvement:
            break
    
    # Phase 2: Gradient Descent
    # Use the best particle from PSO as starting point
    h = global_best.copy()
    best_h = h.copy()
    best_c5 = global_best_value
    learning_rate = 0.05
    iter_no_improvement = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s:
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update using gradient descent
        h_new = h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new
            best_h = h.copy()
            best_c5 = new_c5
            iter_no_improvement = 0
            if learning_rate > 1e-6:
                learning_rate *= 0.9
        else:
            iter_no_improvement += 1
            if iter_no_improvement > max_no_improvement:
                break
    
    # Ensure final result is valid
    final_h = best_h
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
