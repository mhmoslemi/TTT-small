import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        n_points = len(h_start)
        # Convert to segment-based representation
        m = 7  # Initial number of segments
        segment_size = n_points // m
        c = np.array([np.mean(h_start[i*segment_size:(i+1)*segment_size]) 
                     for i in range(m)])
        # Ensure it satisfies constraint sum(c) = m/2
        c = c * (m/2) / np.sum(c)
    except NameError:
        # Create a basic construction
        n_points = 70
        m = 7
        c = np.ones(m) * 0.5
        c[0] = 0.2
        c[-1] = 0.8
    
    dx = 2.0 / n_points
    
    def reconstruct_h(c_vals):
        # Reconstruct h from segment values
        segment_size = n_points // len(c_vals)
        h = np.zeros(n_points)
        for i, c_val in enumerate(c_vals):
            start_idx = i * segment_size
            end_idx = (i+1) * segment_size
            h[start_idx:end_idx] = c_val
        # Handle remainder
        if len(c_vals) * segment_size < n_points:
            h[len(c_vals) * segment_size:] = c_vals[-1]
        return h
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def objective(c_vals):
        h = reconstruct_h(c_vals)
        return evaluate(h)
    
    def constraint_sum(c_vals):
        return np.sum(c_vals) - len(c_vals)/2
    
    def constraint_bounds(c_vals):
        return np.concatenate([c_vals, 1 - c_vals])
    
    # Optimization parameters
    bounds = [(0, 1) for _ in range(m)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    best_c = c.copy()
    best_h = reconstruct_h(best_c)
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_segments = 21
    
    while time.time() - start_time < budget_s and no_improvement_count < 50:
        iteration += 1
        
        # Optimize segment values
        try:
            result = minimize(objective, best_c, method='SLSQP', 
                            bounds=bounds, constraints=cons, 
                            options={'maxiter': 100, 'ftol': 1e-8})
            if not result.success:
                # Fallback to simple gradient ascent
                delta = 0.01
                for _ in range(1000):
                    grad = np.zeros_like(best_c)
                    for i in range(len(best_c)):
                        c_plus = best_c.copy()
                        c_plus[i] += delta
                        if np.sum(c_plus) > 0:
                            c_plus = c_plus * (len(c_plus)/2) / np.sum(c_plus)
                        h_plus = reconstruct_h(c_plus)
                        val_plus = evaluate(h_plus)
                        
                        c_minus = best_c.copy()
                        c_minus[i] -= delta
                        if np.sum(c_minus) > 0:
                            c_minus = c_minus * (len(c_minus)/2) / np.sum(c_minus)
                        h_minus = reconstruct_h(c_minus)
                        val_minus = evaluate(h_minus)

                    # Update

                    best_c = best_c * (len(best_c)/2) / np.sum(best_c)
                    
                    h_trial = reconstruct_h(best_c)

                        best_c5 = c5_trial

                        no_improvement_count = 0

                best_c = best_c * (len(best_c)/2) / np.sum(best_c)
                h_trial = reconstruct_h(best_c)
                c5_trial = evaluate(h_trial)

                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1

            no_improvement_count += 1

        if iteration % 25 == 0:
            idxs = np.random.choice(len(best_c), size=max(1, int(len(best_c)*0.1)), replace=False)

            best_c = np.clip(best_c, 0, 1)
            best_c = best_c * (len(best_c)/2) / np.sum(best_c)
            h_trial = reconstruct_h(best_c)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                
        # Occasionally increase resolution
        if iteration % 100 == 0 and m < max_segments:
            m += 2
            c_new = np.interp(np.linspace(0, len(best_c)-1, m), 
                              np.arange(len(best_c)), best_c)
            best_c = c_new
            # Ensure constraint
            best_c = best_c * (len(best_c)/2) / np.sum(best_c)
            
        # Even if failed to improve, check final accuracy
        if iteration % 10 == 0:
            h_trial = reconstruct_h(best_c)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:

                best_h = h_trial.copy()
                
    # Final reconstruction after optimization
    h_final = reconstruct_h(best_c)
    # Normalize to ensure integral constraint exactly
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Apply soft smoothing
    if len(h_final) > 10:
        h_smoothed = np.convolve(h_final, np.ones(5)/5, mode='same')
        c5_final = evaluate(h_smoothed)
        if c5_final < best_c5:
            best_c5 = c5_final
            h_final = h_smoothed
    
    # Make sure final values meet constraints
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    return (list(h_final), best_c5, len(h_final))
