import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    mutation_rate = 0.1
    elite_size = pop_size // 5
    max_plateau = 300
    
    # Initialize population
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    population = [h0.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = h0.copy()
    best_c5 = float('inf')
    last_improvement = 0
    start_time = time.time()
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Evaluate fitness
        fitness = []
        for h in population:
            c = correlate(h, 1 - h, 'full') * dx
            f = np.max(c)
            fitness.append(f)
        
        # Track best solution
        for i, f in enumerate(fitness):
            if f < best_c5:
                best_c5 = f
                best_h = population[i].copy()
                last_improvement = iter_count
        
        # Early termination
        if iter_count - last_improvement > max_plateau:
            break
            
        # Selection: choose top 20%
        selected_indices = np.argsort(fitness)[:elite_size]
        selected_pop = [population[i] for i in selected_indices]
        
        # Crossover and mutation
        new_pop = selected_pop[:]
        while len(new_pop) < pop_size:
            # Crossover: average two random parents
            p1, p2 = np.random.choice(selected_pop, 2, replace=False)
            child = (p1 + p2) / 2.0
            
            # Mutation: add Gaussian noise
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
            
            # Repair individual to satisfy constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_pop.append(child)
            
        population = new_pop[:pop_size]
        
        # Occasionally increase diversity by adding random individuals
        if iter_count % 50 == 0:
            for _ in range(pop_size // 10):
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                # Find replacement for worst
                worst_idx = np.argmax(fitness)
                population[worst_idx] = h
    
    return (best_h.tolist(), best_c5, n_points)
