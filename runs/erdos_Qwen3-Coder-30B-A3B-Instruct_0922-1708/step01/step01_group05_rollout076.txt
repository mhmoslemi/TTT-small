import scipy.optimize as so
from time import time

    start_time = time()

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    else:

    # Particle Swarm Optimization approach
    n_particles = 100
    n_dims = n_points
    
    # Initialize particles and velocities
    particles = np.random.rand(n_particles, n_dims)
    velocities = np.random.rand(n_particles, n_dims) * 0.1
    
    # Clip to [0,1]
    particles = np.clip(particles, 0, 1)
    
    # Ensure sum constraint
    for i in range(n_particles):
        particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
    
    # Initialize personal best positions
    personal_best = particles.copy()
    personal_best_scores = np.array([float('inf')] * n_particles)
    
    # Initialize global best position
    global_best_idx = 0
    global_best_score = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iter = 5000
    iter_count = 0
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def apply_constraints(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals
    
    # Evaluate initial particles
    for i in range(n_particles):
        particle = particles[i]
        score = compute_C(particle)
        if score < personal_best_scores[i]:
            personal_best_scores[i] = score
            personal_best[i] = particle.copy()
        if score < global_best_score:
            global_best_score = score
            global_best_idx = i
            
    # PSO main loop
    while time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) +
                           c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = apply_constraints(particles[i])
            
            # Evaluate new position
            score = compute_C(particles[i])
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = particles[i].copy()
                if score < global_best_score:
                    global_best_score = score
                    global_best_idx = i
                    
    # Refinement stage with gradient descent on best PSO solution
    h_best = personal_best[global_best_idx].copy()
    lr = 0.01
    no_improvement = 0
    max_no_improvement = 100
    
    def eval_grad(h_vals):
        # Numerical gradient
        eps = 1e-6
        grad = np.zeros_like(h_vals)
        base_score = compute_C(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = apply_constraints(h_plus)
            score_plus = compute_C(h_plus)
            grad[i] = (score_plus - base_score) / eps
        return grad
    
    for i in range(2000):
        if time() - start_time > budget_s - 1:
            break
        grad = eval_grad(h_best)
        h_best = h_best - lr * grad
        h_best = apply_constraints(h_best)
        new_score = compute_C(h_best)
        if new_score < global_best_score:
            global_best_score = new_score
            h_best = apply_constraints(h_best)
        else:
            no_improvement += 1
            if no_improvement > max_no_improvement:
                lr *= 0.5
                no_improvement = 0

    return (h_best.tolist(), global_best_score, n_points)
