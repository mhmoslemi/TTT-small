import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    stagnation_limit = 500
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    # Generate initial swarm
    for i in range(n_particles):
        if i == 0:
            # Use the initial solution for the first particle
            particle = h.copy()
        else:
            # Random perturbation of initial solution
            particle = h + np.random.normal(0, 0.1, n_points)
            particle = project_h(particle, n_points)
        
        swarm.append(particle)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_h(particle, dx)
        pbest.append(particle.copy())
        pbest_scores.append(score)
    
    # Find global best
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    start_time = time.time()
    iter_count = 0
    stagnation_count = 0
    last_improvement = 0
    best_solution = gbest.copy()
    best_score = gbest_score
    
    while time.time() - start_time < 0.7 * budget_s and iter_count < max_iter:
        iter_count += 1
        # Adapt swarm size if needed
        if stagnation_count > stagnation_limit and n_particles > 20:
            n_particles = max(20, int(n_particles * 0.8))
            # Reinitialize swarm with smaller size
            swarm = [project_h(h + np.random.normal(0, 0.1, n_points), n_points) 
                     for _ in range(n_particles)]
            velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
            pbest = [swarm[i].copy() for i in range(n_particles)]
            pbest_scores = [evaluate_h(swarm[i], dx) for i in range(n_particles)]
            gbest_idx = np.argmin(pbest_scores)
            gbest = pbest[gbest_idx].copy()
            gbest_score = pbest_scores[gbest_idx]
            stagnation_count = 0
            last_improvement = 0
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - swarm[i]) + 
                            c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] = project_h(swarm[i] + velocities[i], n_points)
            
            # Evaluate
            score = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = swarm[i].copy()
                pbest_scores[i] = score
                
                # Update global best if needed
                if score < gbest_score:
                    gbest = swarm[i].copy()
                    gbest_score = score
                    stagnation_count = 0
                    last_improvement = iter_count
                    
                    if score < best_score:
                        best_solution = gbest.copy()
                        best_score = score
            else:
                stagnation_count += 1
        
        # Early stopping if no improvement recently
        if iter_count - last_improvement > 1000:
            # Reduce swarm size
            n_particles = max(20, int(n_particles * 0.9))
            swarm = swarm[:n_particles]
            velocities = velocities[:n_particles]
            pbest = pbest[:n_particles]
            pbest_scores = pbest_scores[:n_particles]
            if len(pbest) > 0:
                gbest_idx = np.argmin(pbest_scores)
                gbest = pbest[gbest_idx].copy()
                gbest_score = pbest_scores[gbest_idx]
            stagnation_count = 0
            last_improvement = iter_count
    
    # Gradient descent refinement
    h = best_solution.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    old_score = evaluate_h(h, dx)
    
    while time.time() - start_time < 0.9 * budget_s and patience < max_patience:
        # Find worst shift 
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute finite difference gradient for penalty weights (worst overlap)
        eps = 1e-5
        grad_approx = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad_approx[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with sign of gradient to reduce worst overlap
        h -= learning_rate * np.sign(grad_approx)
        h = project_h(h, n_points)
        
        new_score = evaluate_h(h, dx)
        if new_score < old_score:
            old_score = new_score
            patience = 0
        else:
            patience += 1
            
            # Reduce learning rate if stuck
            if patience % 1000 == 0:
                learning_rate *= 0.5
    
    final_score = evaluate_h(h, dx)
    
    return (h, final_score, n_points)
