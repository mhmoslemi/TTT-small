        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')

        return max_overlap

        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth

        # Clip to [0,1]

        h = h * (n_points / 2) / np.sum(h)
        return h
