import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Check if initial_h_values is defined and use it
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: create a smooth version of a known good construction
    x = np.linspace(0, 2, n_points)
    # Create a more oscillatory function to help in optimization
    h = 0.5 * (1 + np.sin(np.pi * x) + 0.3 * np.sin(3 * np.pi * x))
    h = project_h(h, n_points)
    return h

def line_search(h, i, n_points, dx, c5_current):
    """Perform line search along the i-th coordinate."""
    # Start with the current value
    h_i = h[i]
    # Evaluate C5 at current point
    c5_prev = c5_current
    
    # Try to find the best value of h[i] that minimizes C5
    # Keep the constraint that sum of h should be n_points/2
    # Try 100 different values of h[i] in [0,1]
    best_h_i = h_i
    best_c5 = c5_prev
    best_h = h.copy()
    
    # For small values around current, to find better h[i]
    step = 0.02
    # To avoid numerical issues, only optimize if h[i] is not at boundary
    min_val = 0.01
    max_val = 0.99
    
    if h[i] < min_val:
        h[i] = min_val
        c5_val = evaluate_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h_i = h[i]
    elif h[i] > max_val:
        h[i] = max_val
        c5_val = evaluate_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h_i = h[i]
    else:
        # Try various values around current
        test_vals = h[i] + np.linspace(-0.1, 0.1, 20)
        test_vals = np.clip(test_vals, 0, 1)
        for val in test_vals:
            h[i] = val
            # Need to adjust sum constraint by changing other h[i]
            # Weights to distribute change to other values
            # This is not exact but simple compromise
            current_sum = np.sum(h)
            if abs(current_sum) > 1e-10:
                diff = (n_points / 2) - current_sum
                # Use proportional distribution to adjust
                adjust = diff / (n_points - 1)  # approximate
                for j in range(n_points):
                    if j != i:
                        h[j] += adjust
                h[i] = val
                h = project_h(h, n_points)
                c5_val = evaluate_c5(h)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h_i = val
                # Undo changes for next iteration
                h = h.copy()  # make a copy for a clean start
    
    # Restore h[i] to best value and set others to maintain balance
    h[i] = best_h_i
    
    # Rebalance other points to make sure sum = n_points/2
    # This is approximately correct
    h = project_h(h, n_points)
    
    return h, best_c5

def optimize_coordinate_wise(h, n_points, dx, time_budget_s):
    """Perform coordinate-wise optimization."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    max_iter = 10000
    no_improvement = 0
    max_no_improvement = 5000
    
    for i in range(n_points):
        if time.time() - start_time > time_budget_s * 0.7:
            break
        
        # Try optimizations, proceed as needed
        h_test = best_h.copy()
        h_test, c5_test = line_search(h_test, i, n_points, dx, best_c5)
        
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > max_no_improvement:
            break
    
    return best_h, best_c5

    start_time = time.time()
    best_h = h.copy()
    old_c5 = evaluate_c5(best_h)
    max_iter = 10000
    no_improvement = 0
    max_no_improvement = 3000
    
    # Precompute conv result for efficiency
    def compute_conv_components(h):
        conv = np.correlate(h, 1 - h, mode="full")
        return conv

    def compute_gradient(h):
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_h(h_plus, n_points)
            h_minus = project_h(h_minus, n_points)
            c5_plus = evaluate_c5(h_plus)
            c5_minus = evaluate_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad

    direction = -compute_gradient(best_h)
    alpha = 0.01  # learning rate
    
    for iteration in range(max_iter):

            break
            
        h_new = best_h + alpha * direction
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < old_c5:
            old_c5 = new_c5
            best_h = h_new
            no_improvement = 0
        else:
            no_improvement += 1
            # Iterate but decrease learning rate
            alpha *= 0.8
            if alpha < 1e-8:
                break

        grad = compute_gradient(best_h)
        # Polak-Ribière step

        # Use beta only if it is not negative
        beta = max(0, beta)
        direction = -grad + beta * direction
        
        # For numerical stability
        if np.linalg.norm(direction) < 1e-12:

        grad_old = grad.copy()
        
        if no_improvement > max_no_improvement:

    return best_h, old_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    
    n_points = 45
    h = initialize_h(n_points)
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Phase 1: Coordinate-wise optimization
    h_phase1, c5_phase1 = optimize_coordinate_wise(h, n_points, dx, budget_s)
    
    # Phase 2: Nonlinear CG optimization
    h_phase2, c5_phase2 = nonlinear_cg_optimize(h_phase1, n_points, dx, budget_s)
    
    # Final evaluation
    final_h = h_phase2 if c5_phase2 < c5_phase1 else h_phase1
    final_c5 = min(c5_phase2, c5_phase1)
    
    # Ensure final result is valid
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
