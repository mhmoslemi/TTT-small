import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    max_iter = 1000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize PSO particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Initialize each particle with random configuration
    for _ in range(n_particles):
        particle = h.copy() + np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (target_sum / np.sum(particle))
        particles.append(particle.copy())
        velocities.append(np.random.normal(0, 0.01, n_points))
        pbest.append(particle.copy())
        pbest_values.append(float('inf'))
    
    # Global best
    gbest = h.copy()
    gbest_value = float('inf')
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def evaluate_particle(particle):
        # Ensure particle satisfies constraints
        particle = np.clip(particle, 0, 1)
        particle = particle * (target_sum / np.sum(particle))
        return compute_C(particle)
    
    # PSO main loop
    for iter_count in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Evaluate particle
            particle_value = evaluate_particle(particles[i])
            
            # Update personal best
            if particle_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = particle_value
            
            # Update global best
            if particle_value < gbest_value:
                gbest = particles[i].copy()
                gbest_value = particle_value

            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (pbest[i] - particles[i]) +
                            social * r2 * (gbest - particles[i]))
            
            particles[i] += velocities[i]
            
            # Keep in bounds
            particles[i] = np.clip(particles[i], 0, 1)
            # Rescale to meet integral constraint
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
    
    # Refinement phase: gradient descent on best solution
    best_h = gbest.copy()
    best_C = gbest_value
    
    # Use gradient descent to refine
    lr = 0.01
    no_improvement_count = 0
    previous_C = best_C
    
    def gradient_descent_step(h_vals):
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h_vals)
        h_base = h_vals.copy()
        base_C = compute_C(h_base)
        
        for i in range(len(h_vals)):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - base_C) / eps
            
        return grad
    
    # Refine for remaining budget
    while time() - start_time < budget_s - 1:
        # Compute gradient
        grad = gradient_descent_step(best_h)
        
        # Update h
        best_h = best_h - lr * grad
        
        # Ensure constraints
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        
        # Evaluate
        new_C = compute_C(best_h)
        
        # Adaptive learning rate
        if new_C >= previous_C:
            no_improvement_count += 1
            if no_improvement_count > 10:
                lr *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
            
        if new_C < best_C:
            best_C = new_C
        previous_C = new_C
    
    return (best_h.tolist(), best_C, n_points)
