import numpy as np

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

    h = project_h(h, n_points)

def coordinate_wise_optimization(h, n_points, max_iter=1000):
    """Perform coordinate-wise optimization on h."""
    h = h.copy()
    for iteration in range(max_iter):
        improvement = False
        # Loop through each coordinate
        for i in range(n_points):
            orig = h[i]
            # Find best h[i] for the current shift maximizing overlap
            # We can approximate by just looking at closest "important" shift
            best_h = line_search(h, i, 0, n_points, n_points)
            h[i] = best_h
            h = project_h(h, n_points)
            if abs(h[i] - orig) > 1e-6:
                improvement = True
            if not improvement:
                break
        if not improvement:
            break
    return h
