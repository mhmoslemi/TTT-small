import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution to better capture function features
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    # Genetic algorithm parameters
    pop_size = 50
    num_keep = 10  # Top 20% to keep
    mutation_rate = 0.1
    mutation_strength = 0.05
    
    # Create initial population
    population = []
    for i in range(pop_size):
        if i == 0:
            # Start with best solution so far or initial
            individual = h.copy()
        else:
            # Random initialization with scaling
            individual = np.random.rand(n_points) * 0.8 + 0.1  # Keep within [0.1, 0.9]
            individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # GA loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate population
        fitness = []
        for individual in population:
            # Compute overlap integral
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            last_improvement = iteration
            
        # Early termination if no improvement in a while
        if iteration - last_improvement > 1000:
            break
            
        # Select top individuals
        selected = population[:num_keep]
        
        # Create new population through crossover and mutation
        new_population = selected.copy()
        
        # Crossover: average pairs
        while len(new_population) < pop_size:
            idx1, idx2 = np.random.choice(num_keep, 2, replace=False)
            parent1 = selected[idx1]
            parent2 = selected[idx2]
            
            # Arithmetic crossover
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_strength, n_points)
                child = child + noise
                
            # Repair: clip and rescale
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population[:]
        
    return (best_h.tolist(), best_c5, n_points)
