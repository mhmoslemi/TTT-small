import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure constraint
    h = h * (target_sum / np.sum(h))
    
    # Objective function
    def compute_C(h_vals):
        h_vals = np.array(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # Optimization
    best_h = h.copy()
    best_C = compute_C(h)
    
    # Use scipy.optimize with constraints
    def objective(h_vals):
        return compute_C(h_vals)
        
    # Bounds (0, 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum should be target_sum
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Initial guess
    x0 = h.copy()
    
    # Run optimization
    result = so.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-6},
        callback=lambda x: None  # We don't need callbacks
    )
    
    # Extract results
    if result.success:
        final_h = result.x
    else:
        final_h = h  # fallback
        
    # Ensure final constraints
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (target_sum / np.sum(final_h))
    
    final_C = compute_C(final_h)
    
    # Refine with gradient descent loop
    h = final_h.copy()
    lr = 0.01
    
    # Objective function for gradient
    def objective_grad(h_vals):
        h_vals = np.array(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # Gradient computation (finite difference)
    def compute_grad(h_vals, eps=1e-5):
        grad = np.zeros_like(h_vals)
        base_val = objective_grad(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            val_plus = objective_grad(h_plus)
            grad[i] = (val_plus - base_val) / eps
        return grad
    
    # Optimization loop
    max_iter = int((budget_s - (time() - start_time)) * 5)  # Give some margin for finalization  
    if max_iter < 1:
        max_iter = 1
        
    last_improvement = 0
    for i in range(max_iter):
        if time() - start_time >= budget_s - 1:
            break
            
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        C = objective_grad(h)
        if C < best_C:
            best_C = C
            best_h = h.copy()
            
        grad = compute_grad(h)
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Adaptive step size
        if i > 0 and i % 100 == 0:
            if abs(C - last_improvement) < 1e-7:
                lr *= 0.5
        last_improvement = C
        
        # Check for improvement
        if i % 50 == 0:
            C_current = objective_grad(h)
            if C_current >= last_improvement:
                if abs(C_current - last_improvement) < 1e-6:
                    lr *= 0.8  # Reduce learning rate if no improvement

    # Ensure final result is valid
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (target_sum / np.sum(final_h))
    final_C = compute_C(final_h)

    return (final_h.tolist(), final_C, n_points)
