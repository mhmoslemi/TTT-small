import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Try different optimization strategies
    for restart in range(5):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Create a new random initialization
        h_init = np.random.rand(n_points) * 0.5 + 0.25
        h_init = project_h(h_init)
        
        # Simulated Annealing with adaptive step size
        current_h = h_init.copy()
        current_c5 = compute_c5(current_h)
        temp = 1.0
        cooling_rate = 0.95
        step_size = 0.1
        
        # Inner optimization loop for this initialization
        inner_iterations = 0
        max_inner = 2000
        
        while inner_iterations < max_inner and (time.time() - start_time) < budget_s - 1:
            # Random neighbor perturbation
            neighbor = current_h.copy()
            idx = np.random.randint(0, n_points)
            perturb = np.random.normal(0, step_size)
            neighbor[idx] += perturb
            neighbor = project_h(neighbor)
            
            neighbor_c5 = compute_c5(neighbor)
            
            # Accept if better
            if neighbor_c5 < current_c5:
                current_h = neighbor
                current_c5 = neighbor_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            else:
                # Accept with probability based on temperature
                if np.random.rand() < np.exp(-(neighbor_c5 - current_c5) / temp):
                    current_h = neighbor
                    current_c5 = neighbor_c5
            
            inner_iterations += 1
            temp *= cooling_rate
            if temp < 0.001:
                temp = 0.001
                
    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
