import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: peak in middle with symmetric shape
    x = np.linspace(0, 2, n_points)
    # Create a simple symmetric pulse for better chance of convergence
    h = np.zeros(n_points)
    center = n_points // 2
    width = n_points // 4
    for i in range(max(0, center - width), min(n_points, center + width)):
        h[i] = 1.0 - abs(i - center) / width
    h = project_h(h, n_points)
    return h

def pso_phase(h, n_points, time_budget_s):
    """Run particle swarm optimization to find a good starting point."""
    start_time = time.time()
    
    # PSO parameters
    swarm_size = 30
    max_iter = 10000
    cognitive_weight = 2.0
    social_weight = 2.0
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    for _ in range(swarm_size):
        particle_h = h.copy()
        particles.append(particle_h)
        velocities.append(np.zeros(n_points))
        pbest.append(particle_h.copy())
        pbest_scores.append(evaluate_c5(particle_h))
    
    # Global best
    best_index = np.argmin(pbest_scores)
    gbest = pbest[best_index].copy()
    gbest_score = pbest_scores[best_index]
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.7 and iteration < max_iter:
        iteration += 1
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                           cognitive_weight * r1 * (pbest[i] - particles[i]) + 
                           social_weight * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to constraints
            particles[i] = project_h(particles[i], n_points)
            
            # Evaluate
            score = evaluate_c5(particles[i])
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
            
            # Update global best
            if score < gbest_score:
                gbest = particles[i].copy()
                gbest_score = score
        
        # Reduce inertia weight
        inertia_weight = 0.9 - (0.5 * iteration / max_iter)
    
    return gbest, gbest_score

def gradient_descent_phase(h, n_points, time_budget_s):
    """Run gradient descent with projected updates."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.05
    
    # Use a simple gradient estimation by finite differences
    eps = 1e-4
    max_iter = 10000
    iter_no_improvement = 0
    max_no_improvement = 1000
    
    # Find the shift that maximizes overlap (for gradient computation)
    def find_worst_shift(h):
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)
        return shift
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_budget_s * 0.3:
            break
            
        # Compute approximate gradient
        shift = find_worst_shift(best_h)
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update
        h_new = best_h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Accept update
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            
        if iter_no_improvement > max_no_improvement:
            break
            
        # Adjust learning rate
        if iteration % 100 == 0 and learning_rate > 1e-6:
            learning_rate *= 0.9
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45  # Odd number for symmetric
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: PSO for global search
    h_pso, c5_pso = pso_phase(h, n_points, budget_s)
    
    # Phase 2: Gradient Descent for local refinement
    h_gd, c5_gd = gradient_descent_phase(h_pso, n_points, budget_s)
    
    # Return best result
    final_h = h_gd if c5_gd < c5_pso else h_pso
    final_c5 = min(c5_gd, c5_pso)
    
    # Ensure final result is valid
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
