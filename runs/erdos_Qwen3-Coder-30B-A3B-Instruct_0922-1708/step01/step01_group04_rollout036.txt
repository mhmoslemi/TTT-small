import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialization using initial_h_values if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    def evaluate(h):
        dx = 2.0 / len(h)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def reconstruct_from_segments(c_vals, n_points, n_segments):
        # Create h from segment values
        segment_width = n_points // n_segments
        h = np.zeros(n_points)
        for i in range(n_segments):
            start = i * segment_width
            end = start + segment_width
            if i == n_segments - 1:
                end = n_points
            h[start:end] = c_vals[i]
        return h
    
    def optimize_segments(c_vals, n_points, n_segments):
        def objective(c_vals):
            h = reconstruct_from_segments(c_vals, n_points, n_segments)
            return evaluate(h)
            
        def constraint(c_vals):
            return np.sum(c_vals) - n_segments / 2
            
        cons = {'type': 'eq', 'fun': constraint}
        bounds = [(0, 1) for _ in range(n_segments)]
        
        try:
            res = minimize(objective, c_vals, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                return res.x
        except:
            pass
        return c_vals
    
    def perturb_segments(c_vals, rate=0.1, magnitude=0.1):
        n = len(c_vals)
        indices = np.random.choice(n, max(1, int(n * rate)), replace=False)
        new_vals = c_vals.copy()
        for i in indices:
            new_vals[i] += np.random.uniform(-magnitude, magnitude)
        new_vals = np.clip(new_vals, 0, 1)
        # Re-normalize
        new_vals = new_vals * (n / 2) / np.sum(new_vals)
        return new_vals
    
    def smooth_h(h, window=3):
        if window < 2:
            return h
        return np.convolve(h, np.ones(window)/window, mode='same')
    
    # Start with a coarser representation
    n_segments = 7
    if len(h_start) < 70:
        n_segments = min(7, len(h_start) // 10)
        
    c_best = np.ones(n_segments) * 0.5
    h_best = reconstruct_from_segments(c_best, n_points, n_segments)
    c_best = optimize_segments(c_best, n_points, n_segments)
    h_best = reconstruct_from_segments(c_best, n_points, n_segments)
    best_c5 = evaluate(h_best)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < 50:
        iterations += 1
        
        # Optimization phase
        new_c = optimize_segments(c_best, n_points, n_segments)
        h_trial = reconstruct_from_segments(new_c, n_points, n_segments)
        h_trial = smooth_h(h_trial, window=3)
        c5_trial = evaluate(h_trial)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            c_best = new_c
            h_best = h_trial
            no_improvement_count = 0
            last_improvement = iterations
        else:
            no_improvement_count += 1
            
        # Perturbation phase
        if iterations % 25 == 0 or (iterations - last_improvement) > 10:
            c_trial = perturb_segments(c_best, rate=0.15, magnitude=0.1)
            h_trial = reconstruct_from_segments(c_trial, n_points, n_segments)
            h_trial = smooth_h(h_trial, window=3)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                c_best = c_trial
                h_best = h_trial
                no_improvement_count = 0
                last_improvement = iterations
        
        # Resolution increase
        if no_improvement_count > 20 and n_segments < 21:
            n_segments += 2
            if n_segments > 21:
                n_segments = 21
            # Interpolate
            c_new = np.interp(np.linspace(0, n_segments-1, n_points//10), 
                              np.arange(n_segments), c_best)
            c_best = c_new
            no_improvement_count = 0
    
    # Final reconstruction 
    h_final = reconstruct_from_segments(c_best, n_points, n_segments)
    h_final = smooth_h(h_final, window=3)
    c5_final = evaluate(h_final)
    
    # Ensure final integral
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    c5_final = evaluate(h_final)
    
    # Return the best solution found
    return (h_final, c5_final, n_points)
