import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO with gradient refinement"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w = 0.7
    c1 = c2 = 1.5
    inertia = w
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    # Generate initial swarm
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            # 90% from perturbed initial solution, 10% random
            if np.random.rand() < 0.1:
                particle = np.random.rand(n_points)
            else:
                particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_h(particle, n_points)
        
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        pbest_positions.append(particle.copy())
        pbest_values.append(evaluate_h(particle, dx))
    
    # Initialize global best
    best_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    start_time = time.time()
    iter_count = 0
    stagnation_count = 0
    best_improvement = float('inf')
    
    while time.time() - start_time < 0.7 * budget_s and iter_count < max_iter:
        iter_count += 1
        
        for i in range(swarm_size):
            particle = swarm[i]
            vel = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (inertia * vel + 
                   c1 * r1 * (pbest_positions[i] - particle) + 
                   c2 * r2 * (gbest_position - particle))
            
            # Update position
            new_position = particle + vel
            new_position = project_h(new_position, n_points)
            
            # Evaluate
            new_value = evaluate_h(new_position, dx)
            
            # Update personal best
            if new_value < pbest_values[i]:
                pbest_positions[i] = new_position.copy()
                pbest_values[i] = new_value
                
                # Update global best
                if new_value < gbest_value:
                    gbest_position = new_position.copy()
                    gbest_value = new_value
                    stagnation_count = 0
                    best_improvement = float('inf')
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
                
            # Update swarm data
            swarm[i] = new_position
            velocities[i] = vel
            
        # Adapt inertia and swarm size based on stagnation
        if stagnation_count > 500:
            inertia = max(0.5, inertia * 0.95)
            if swarm_size > 20:
                swarm_size = max(20, int(swarm_size * 0.8))
                # Reinitialize smaller swarm
                new_swarm = []
                for j in range(swarm_size):
                    if j == 0:
                        new_swarm.append(gbest_position.copy())
                    else:
                        # Perturb from best
                        perturbed = gbest_position + np.random.normal(0, 0.05, n_points)
                        new_swarm.append(project_h(perturbed, n_points))
                swarm = new_swarm
                velocities = [np.zeros(n_points) for _ in range(swarm_size)]
                pbest_positions = [swarm[j].copy() for j in range(swarm_size)]
                pbest_values = [evaluate_h(swarm[j], dx) for j in range(swarm_size)]
                stagnation_count = 0
                inertia = 0.7
                
    # Gradient descent refinement
    h = gbest_position.copy()
    best_h = h.copy()
    best_c5 = gbest_value
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s - 10 and patience <= max_patience:
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient w.r.t. h using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with sign of gradient
        h -= learning_rate * np.sign(grad)
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience % 1000 == 0:
                learning_rate *= 0.5
                patience = 0
    
    return (best_h, best_c5, n_points)
