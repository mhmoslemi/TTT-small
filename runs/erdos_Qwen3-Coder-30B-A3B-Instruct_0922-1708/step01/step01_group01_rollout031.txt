import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    adapt_interval = 500
    min_swarm_size = 20
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    # Generate initial particles
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    pbest.append(h.copy())
    pbest_scores.append(evaluate_h(h, dx))
    
    # 9 remaining particles: small random perturbations
    for _ in range(n_particles - 1):
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = h + perturb
        new_h = project_h(new_h, n_points)
        particles.append(new_h)
        velocities.append(np.zeros(n_points))
        pbest.append(new_h.copy())
        pbest_scores.append(evaluate_h(new_h, dx))
    
    # Global best tracking
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    start_time = time.time()
    iter_count = 0
    no_improve_count = 0
    
    while time.time() - start_time < 0.9 * budget_s and iter_count < max_iter:
        iter_count += 1
        
        # PSO update
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = project_h(particles[i] + velocities[i], n_points)
            
            # Evaluate
            new_score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if new_score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = new_score
            
            # Update global best
            if new_score < gbest_score:
                gbest = particles[i].copy()
                gbest_score = new_score
                no_improve_count = 0
            else:
                no_improve_count += 1
        
        # Adapt swarm size if needed
        if no_improve_count > adapt_interval and n_particles > min_swarm_size:
            n_particles = max(min_swarm_size, n_particles - int(n_particles * 0.2))
            # Rebuild swarm
            new_particles = [gbest.copy()]  # keep global best
            for _ in range(n_particles - 1):
                perturb = np.random.normal(0, 0.05, n_points)
                new_h = gbest + perturb
                new_h = project_h(new_h, n_points)
                new_particles.append(new_h)
            particles = new_particles
            velocities = [np.zeros(n_points) for _ in range(n_particles)]
            pbest = [p.copy() for p in particles]
            pbest_scores = [evaluate_h(p, dx) for p in particles]
            gbest_idx = np.argmin(pbest_scores)
            gbest = pbest[gbest_idx].copy()
            gbest_score = pbest_scores[gbest_idx]
            no_improve_count = 0

    # Gradient descent refinement
    h = gbest.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < 0.95 * budget_s and patience < max_patience:
        # Find worst overlap
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with sign of gradient
        update = -learning_rate * np.sign(grad)
        h = project_h(h + update, n_points)
        
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience % 1000 == 0:
                learning_rate *= 0.5
                patience = 0
    
    return (best_h, best_c5, n_points)
