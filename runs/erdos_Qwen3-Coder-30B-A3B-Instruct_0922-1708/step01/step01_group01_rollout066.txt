```python
import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
        best_h = h.copy()
        best_c5 = evaluate_h(best_h, dx)
    else:
        # Initialize with sin wave
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
        best_h = h.copy()
        best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    
    # Segment-based approach
    # Use 5 segments over [0, 2], each segment gets one of {0, 0.5, 1}
    m = 5  # number of segments 
    segment_values = [0.0, 0.5, 1.0]
    segment_combinations = list(product(segment_values, repeat=m))
    
    best_c5 = float('inf')
    best_h = None
    
    for combo in segment_combinations:
        # Create piecewise constant h from segments
        h_seg = np.zeros(n_points)
        segment_length = n_points // m
        remainder = n_points % m
        for i, value in enumerate(combo):
            start_idx = i * segment_length + min(i, remainder)
            end_idx = start_idx + segment_length + (1 if i < remainder else 0)
            h_seg[start_idx:end_idx] = value
        
        # Project to satisfy sum constraint
        h_seg = project_h(h_seg, n_points)
        
        # Evaluate
        c5 = evaluate_h(h_seg, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_seg.copy()
            
        # Check if we need to stop
        if time.time() - start_time > 0.8 * budget_s:
            break
    
    # Refine top candidates with gradient descent
    top_candidates = []
    
    # Search for best 10 candidates from the segments space
    candidate_scores = []
    for combo in segment_combinations:
        h_seg = np.zeros(n_points)
        segment_length = n_points // m
        remainder = n_points % m
        for i, value in enumerate(combo):
            start_idx = i * segment_length + min(i, remainder)
            end_idx = start_idx + segment_length + (1 if i < remainder else 0)
            h_seg[start_idx:end_idx] = value
        h_seg = project_h(h_seg, n_points)
        c5 = evaluate_h(h_seg, dx)
        candidate_scores.append((c5, h_seg))
    
    # Keep top 10
    candidate_scores.sort(key=lambda x: x[0])
    top_candidates = [h for _, h in candidate_scores[:10]]
    
    # Gradient descent refinement on top candidates
    for h_refine in top_candidates:
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        h = h_refine.copy()
        current_c5 = evaluate_h(h, dx)
        
        # Gradient descent
        max_iter = 2000
        learning_rate = 0.01
        patience = 0
        max_patience = 200
        prev_c5 = current_c5
        
        for _ in range(max_iter):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Approximate gradient
            eps = 1e-5
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h -= learning_rate * grad
            h = project_h(h, n_points)
            
            # Evaluate
            new_c5 = evaluate_h(h, dx)
            if new_c5 < current_c5:
                current_c5 = new_c5
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    break
                if patience % 100 == 0:
                    learning_rate *= 0.5
            
            if abs(prev_c5 - current_c5) < 1e-8:
                patience += 1
                if patience > max_patience * 2:
                    break
            prev_c5 = current_c5
            
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h, best_c5, n_points)
```
