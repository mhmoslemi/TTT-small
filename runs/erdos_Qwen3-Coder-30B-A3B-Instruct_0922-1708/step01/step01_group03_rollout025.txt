import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h_symmetric(n_points):
    """Initialize h symmetrically."""
    try:
        # Use the provided initial construction
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values).copy()
            # Ensure it's the right length
            if len(h) != n_points:
                h = h[:n_points] if len(h) > n_points else np.pad(h, (0, n_points - len(h)), 'constant', constant_values=0)
            # Ensure symmetry
            k = n_points // 2
            for i in range(k):
                h[i] = (h[i] + h[n_points - 1 - i]) / 2
                h[n_points - 1 - i] = h[i]
            return project_h(h, n_points)
    except:
        pass
    
    # Default initialization: symmetric Gaussian-like bumps
    x = np.linspace(0, 2, n_points)
    # Create symmetric function
    h = np.zeros(n_points)
    # Add two Gaussian-like bumps
    for mu in [0.5, 1.5]:
        h += np.exp(-4 * (x - mu)**2)
    # Normalize and project
    h = project_h(h, n_points)
    
    # Enforce symmetry by averaging with mirror
    k = n_points // 2
    for i in range(k):
        h[i] = (h[i] + h[n_points - 1 - i]) / 2
        h[n_points - 1 - i] = h[i]
        
    return h

def symmetric_gradient(h, max_shift):
    """Compute exact gradient for symmetric h with respect to max_shift."""
    n = len(h)
    # Assume max_shift is in the range [-n+1, n-1]
    # Convolution starts at idx 0 for shift -n+1, ends at idx 2n-2 for shift n-1
    # Actual shift for index i = i - (n-1) 
    # So for shift k, correct index = k + (n-1)
    grad = np.zeros(n)
    # The contribution: for each h[i], the term (1 - h[i+k]) 
    # This gives a simple relationship:
    # Gradient at i = - (1 - h[i+shift]) = h[i+shift] - 1
    # But boundaries must be handled - let's use the safe way: 
    # Assume any direct sparse gradient or do the accounting properly via forward/backward diffs.
    # For now, let's calculate the index for the shift array where convolution is maximized
    # and do center finite difference for midpoint at that index
    # This is: 
    safe_shift = max_shift + (n - 1)  # from shift to index
    if safe_shift < 0 or safe_shift >= 2*n - 1:
        # Code error if reached
        return np.zeros(n)
    
    # Manually compute for max_indices respecting the constraint
    # Since convolution output len is 2n-1, it has indices 0 to 2n-2 for shifts -n+1 to n-1
    shift_from_idx = safe_shift - (n - 1)  # The actual shift
    # The gradient for each h[i]
    # C5 = sum over i of h[i] * (1-h[i+shift])
    # so dC5/dh[i] = (1 - h[i+shift]) if we cross out correlation terms that don't involve h[i]
    # At clamped boundary, we assume account for wraparound or just stay within valid range
    # Expectation is max_shift is known from previous computation
    for i in range(n):
        j = i + shift_from_idx
        if j >= 0 and j < n:
            grad[i] = 1 - h[j]
        else:
            # Handle out-of-bounds: can either ignore or do wrap-around effects
            # For simplicity and small leftover: ignore out-of-bounds
            grad[i] = 0
    return grad

def optimize_symmetric_phase(h, n_points, time_budget_s):
    """Optimize with symmetric constraints and exact gradients."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    # Start learning rate
    learn_rate = 0.1
    max_no_improvement = 2000
    iter_no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < time_budget_s and iter_no_improvement < max_no_improvement:
        iteration += 1
        # Evaluate C5 to find max shift
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        max_shift = max_idx - (n - 1)  # convert back to shift value
        
        # Compute exact gradient for current maximum correlation shift
        grad = symmetric_gradient(h, max_shift)
        if np.allclose(grad, 0):
            break
        
        # Update
        h_new = h - learn_rate * grad
        # Ensure symmetry manually at update
        k = n_points // 2
        for i in range(k):
            h_new[i] = (h_new[i] + h_new[n_points - 1 - i]) / 2
            h_new[n_points - 1 - i] = h_new[i]
        h_new = project_h(h_new, n_points)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            iter_no_improvement = 0
            # Reduce learning rate for more fine tuning
            learn_rate *= 0.95
        else:
            iter_no_improvement += 1
            # Increase learning rate for aggressive escape
            learn_rate *= 1.05
            if learn_rate > 1.0:
                learn_rate = 0.1  # reset if too large

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    
    n_points = 45  # odd for symmetric handling
    h = initialize_h_symmetric(n_points)
    
    start_time = time.time()
    
    # Phase 1: Symmetric optimization
    h_opt, c5_opt = optimize_symmetric_phase(h, n_points, budget_s * 0.9)
    
    # Small final fine-tune using standard evaluation
    h_final = project_h(h_opt, n_points)
    c5_final = evaluate_c5(h_final)
    
    if c5_final >= c5_opt:
        h_final = h_opt
        c5_final = c5_opt
    
    # Since it's symmetric, we should ensure symmetry holds to finish
    k = n_points // 2
    for i in range(k):
        h_final[i] = (h_final[i] + h_final[n_points - 1 - i]) / 2
        h_final[n_points - 1 - i] = h_final[i]
    
    h_final = project_h(h_final, n_points)
    c5_final = evaluate_c5(h_final)
    
    # Final validation
    assert np.isclose(np.sum(h_final), n_points / 2, rtol=1e-4), "Sum constraint violated"
    assert np.all(h_final >= 0) and np.all(h_final <= 1), "Bounds violated"
    
    return (h_final, c5_final, n_points)
