import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_symmetric_h(n_points):
    """Initialize symmetric h using Gaussian bumps."""
    # Use initial if available
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    
    # Otherwise, create symmetric starting point
    x = np.linspace(0, 2, n_points)
    # Symmetric construction around x=1: two bumps
    h = 0.5 * (1 + np.sin(np.pi * (x - 0.5))) + 0.5 * (1 + np.sin(np.pi * (x - 1.5)))
    h = project_h(h, n_points)
    return h

def enforce_symmetry(h, n_points):
    """Enforce symmetry h[i] = h[n_points - 1 - i] for i < n_points//2."""
    if n_points % 2 == 0:
        # Even length: average with symmetric point
        for i in range(n_points // 2):
            h[i] = h[n_points - 1 - i] = (h[i] + h[n_points - 1 - i]) / 2
    else:
        # Odd length: average with symmetric point, keep middle unchanged
        mid = n_points // 2
        for i in range(mid):
            h[i] = h[n_points - 1 - i] = (h[i] + h[n_points - 1 - i]) / 2
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using symmetric gradient descent."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45  # Must be odd for symmetric construction
    h = initialize_symmetric_h(n_points)
    h = enforce_symmetry(h, n_points)
    
    # Main optimization loop
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Gradient descent parameters
    learning_rate = 0.1
    max_no_improvement = 1000
    iter_no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 5:  # reserve some time for final check
        # Find the shift that gives the maximum overlap
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)
        
        # Convert to relevant shift indices for computation
        # For shift k, we get h[i] * (1 - h[i+k]) for valid k
        # The correlation at shift k is sum_i h[i] * (1 - h[i+k])
        # Gradient w.r.t. h[i] = (1 - h[i+k]) for k > 0
        # Or (1 - h[i-k]) for k < 0
        # We compute this for shift that maximizes overlap
        
        # Let's construct gradient at that particular shift
        gradient = np.zeros(n)
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                gradient[i] = (1 - h[j]) * dx
            elif j >= n:
                j_wrapped = j - n
                gradient[i] = (1 - h[j_wrapped]) * dx
            elif j < 0:
                j_wrapped = n + j
                gradient[i] = (1 - h[j_wrapped]) * dx
        
        # Update
        h_new = h - learning_rate * gradient
        h_new = project_h(h_new, n_points)
        h_new = enforce_symmetry(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            # Adaptive learning rate
            if learning_rate > 1e-6:
                learning_rate *= 0.97
        else:
            iter_no_improvement += 1
            # Adaptive learning rate
            learning_rate *= 1.05
            if learning_rate > 1:
                learning_rate = 1.0
        iteration += 1
        
        if iter_no_improvement > max_no_improvement:
            break
            
    # Final validation
    final_h = h.copy()
    final_c5 = evaluate_c5(final_h)
    
    # Ensure the final solution is symmetric
    final_h = enforce_symmetry(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
