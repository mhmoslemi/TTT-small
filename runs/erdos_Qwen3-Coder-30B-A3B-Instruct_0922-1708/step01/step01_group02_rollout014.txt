import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    n_points = 47
    population_size = 100
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = population_size // 10
    
    # Function to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(conv)
        dx = 2.0 / n_points
        return max_val * dx
    
    # Function to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2.0) / current_sum
        return h_vec
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        population.append(h_init)
    else:
        # If no initial construction, use another construction
        h_init = np.ones(n_points) * 0.5
        # Add some variation
        h_init[10:15] = 0.75
        h_init[30:35] = 0.25
        population.append(h_init)

    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
        
    # Best individual tracking
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    while (time.time() - start_time) < budget_s - 1:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Create next generation
        new_population = population[:elite_size]  # Elitism
        
        while len(new_population) < population_size:
            # Selection (roulette wheel)
            inv_fitness = [1.0 / (f + 1e-10) for f in fitness]
            total_fitness = sum(inv_fitness)
            if total_fitness > 0:
                probabilities = [f / total_fitness for f in inv_fitness]
            else:
                probabilities = [1.0 / len(inv_fitness)] * len(inv_fitness)
            
            parents_indices = np.random.choice(len(population), size=2, p=probabilities)
            parent1, parent2 = population[parents_indices[0]], population[parents_indices[1]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            mutation_prob = mutation_rate / (1.0 + generation**0.5)
            for i in range(n_points):
                if np.random.rand() < mutation_prob:
                    child1[i] += np.random.normal(0, 0.05)
                    child2[i] += np.random.normal(0, 0.05)
            
            # Project onto feasible set
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
        generation += 1
    
    # Final check and return
    if best_h is None:
        # If no valid solutions, return last evaluated one
        final_h = population[0]
        final_c5 = compute_c5(final_h)
    else:
        final_h = project_h(best_h)
        final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
