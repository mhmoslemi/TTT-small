        return np.max(np.abs(corr))

    best_C = compute_C(h)
    best_h = h.copy()
    
    # Gradient-based optimization with hand-coded finite differences
    def gradient_descent_optimization():
        nonlocal best_h, best_C
        
        # Use a simpler approach: optimize directly on raw h values but

        iter_count = 0
        no_improvement_count = 0

            # Compute gradient with finite differences
            eps = 1e-5
            grad = np.zeros_like(best_h)

                h_plus = best_h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)

                C_plus = compute_C(h_plus)
                grad[i] = (C_plus - C_current) / eps
            
            # Update with gradient descent
            updated_h = best_h - lr * grad

            updated_h = np.clip(updated_h, 0, 1)

    # Run gradient descent
    final_C, final_h = gradient_descent_optimization()
