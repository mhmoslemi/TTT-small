import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def latin_hypercube_sampling(n_points, n_samples):
    """Generate Latin Hypercube Samples"""
    samples = []
    for _ in range(n_samples):
        # Generate a random permutation for each dimension
        sample = np.random.random(n_points)
        # Scale to [0, 1) and add jitter to get Latin hypercube
        sample = (np.arange(n_points) + sample) / n_points
        samples.append(sample)
    return np.array(samples)

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # LHS sampling phase
    n_samples = 50
    n_elites = 20
    lhs_samples = latin_hypercube_sampling(n_points, n_samples)
    
    # Create candidates from LHS samples
    candidates = [h.copy()]  # Include the parent solution
    for sample in lhs_samples:
        # Interpolate the sample to create a new candidate
        x_new = np.linspace(0, 2, n_points)
        x_old = np.linspace(0, 2, n_points)
        h_new = np.interp(x_new, x_old, sample)
        # Perturb it slightly to avoid exact copies
        h_new += np.random.normal(0, 0.01, n_points)
        h_new = project_h(h_new, n_points)
        candidates.append(h_new)
    
    # Evaluate candidates
    fitness = [evaluate_h(h, dx) for h in candidates]
    sorted_indices = np.argsort(fitness)
    
    # Keep elites
    elites = [candidates[i] for i in sorted_indices[:n_elites]]
    best_c5 = fitness[sorted_indices[0]]
    best_h = elites[0].copy()
    
    # Evolutionary phase: Evolution Strategy
    n_offspring = 5
    max_generations = 1000
    generation = 0
    
    while time.time() - start_time < 0.7 * budget_s and generation < max_generations:
        generation += 1
        
        # For each elite, generate offspring
        offspring = []
        for elite in elites:
            for _ in range(n_offspring):
                # Small random perturbation
                child = elite + np.random.normal(0, 0.05, n_points)
                child = project_h(child, n_points)
                offspring.append(child)
        
        # Evaluate offspring and elites
        all_candidates = elites + offspring
        all_fitness = [evaluate_h(h, dx) for h in all_candidates]
        
        # Keep top n_elites
        sorted_indices = np.argsort(all_fitness)
        elites = [all_candidates[i] for i in sorted_indices[:n_elites]]
        
        # Update best
        if all_fitness[sorted_indices[0]] < best_c5:
            best_c5 = all_fitness[sorted_indices[0]]
            best_h = elites[0].copy()
    
    # Gradient descent phase
    h = best_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    learning_rate = 0.01
    patience = 0
    max_patience = 1000
    prev_c5 = float('inf')
    
    # Stop if time is almost up
    if time.time() - start_time > 0.9 * budget_s:
        return (best_h, best_c5, n_points)
    
    while time.time() - start_time < 0.95 * budget_s and patience < max_patience:
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= learning_rate * np.sign(grad)
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience % 100 == 0:
                learning_rate *= 0.5
                if learning_rate < 1e-6:
                    break
        if abs(prev_c5 - new_c5) < 1e-6:
            patience += 1
        prev_c5 = new_c5
    
    return (best_h, best_c5, n_points)
