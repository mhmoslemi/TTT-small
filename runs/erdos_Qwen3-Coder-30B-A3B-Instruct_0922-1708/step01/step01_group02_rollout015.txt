import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create symmetric construction
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx

    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec

    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Optimization with potentially better starting points
    def optimize_with_restarts():
        nonlocal h, best_h, best_c5
        
        # Try different starting configurations
        configs = []
        
        # Central plateau
        h_plateau = np.ones(n_points) * 0.5
        mid = n_points // 2
        h_plateau[mid-4:mid+4] = 0.8
        configs.append(("plateau", h_plateau))
        
        # Two peaks
        h_peaks = np.ones(n_points) * 0.3
        h_peaks[n_points//3] = 0.8
        h_peaks[2*n_points//3] = 0.8
        configs.append(("peaks", h_peaks))
        
        # Gaussian-like function
        x = np.linspace(0, 2, n_points)
        h_gauss = np.exp(-((x - 1)**2) / 0.2)
        h_gauss = h_gauss / np.sum(h_gauss) * target_sum
        configs.append(("gauss", h_gauss))
        
        # Random with constraint
        for _ in range(10):
            h_rand = np.random.rand(n_points)
            h_rand = project_h(h_rand)
            configs.append(("random", h_rand))
        
        for name, init_h in configs:
            if (time.time() - start_time) > budget_s - 5:
                break
                
            h_current = init_h.copy()
            h_current = project_h(h_current)
            
            # Local optimization using scipy.optimize.minimize
            def objective(h_vec):
                return compute_c5(h_vec)
            
            def constraint_func(h_vec):
                return np.sum(h_vec) - target_sum
            
            cons = {'type': 'eq', 'fun': constraint_func}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                result = minimize(objective, h_current, method='SLSQP', bounds=bounds, constraints=cons, 
                                options={'maxiter': 500, 'ftol': 1e-6, 'gtol': 1e-6})
                if result.success:
                    h_opt = result.x
                    h_opt = project_h(h_opt)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
            except:
                continue
                
            if (time.time() - start_time) > budget_s - 5:
                break

    # Run optimization
    optimize_with_restarts()

    # Final evaluation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
