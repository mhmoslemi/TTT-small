import numpy as np
from scipy import optimize
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a symmetric function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h, shift):
    """Compute exact gradient for a given shift (defined by max overlap)"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        j = i + shift
        if j >= 0 and j < n:
            grad[i] = -dx
        elif j >= n:
            j_wrap = j - n
            grad[i] = -dx
    return grad

def symmetric_project(h, n_points):
    """Project h to be symmetric around the middle and satisfy sum constraint"""
    # Ensure symmetry about middle point
    mid = (n_points - 1) // 2
    # for i in range(mid+1):
    #     avg = (h[i] + h[n_points - 1 - i]) / 2
    #     h[i] = avg
    #     h[n_points - 1 - i] = avg
    
    if n_points % 2 == 1:  # odd
        h[mid] = h[mid]  # preserve middle value
        for i in range(mid):
            avg_val = (h[i] + h[n_points-1-i]) / 2
            h[i] = avg_val
            h[n_points-1-i] = avg_val
    else:
        # even case
        # average symmetry around middle
        for i in range(mid):
            avg_val = (h[i] + h[n_points-1-i]) / 2
            h[i] = avg_val
            h[n_points-1-i] = avg_val
    # then rescale to match constraint
    h = project_h(h, n_points)
    return h

def optimize_symmetric_h(initial_h, n_points, time_budget_s):
    """Optimize with symmetric constraints and exact gradients"""
    start_time = time.time()
    
    # Initialize
    h = initial_h.copy()
    h = symmetric_project(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Parameters
    learning_rate = 0.1
    max_no_improvement = 1000
    iter_no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < time_budget_s * 0.95:
        iteration += 1
        
        # Find current worst shift for max overlap
        conv = np.correlate(h, 1-h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute exact gradient for this shift (max overlap shift)
        grad = compute_gradient(h, shift)
        
        # Update with learning rate
        h_new = h - learning_rate * grad
        h_new = symmetric_project(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Accept or reject update
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            iter_no_improvement = 0
            # Reduce learning rate
            learning_rate *= 0.999
        else:
            iter_no_improvement += 1
            # Increase learning rate if too small
            if learning_rate < 1e-6:
                learning_rate = 1e-6
            else:
                learning_rate *= 1.01
        
        if iter_no_improvement > max_no_improvement:
            break
        
        h = h_new.copy()
        
        # Occasionally check if we should adjust learning rate more aggressively
        if iteration % 100 == 0:
            if learning_rate < 1e-6:
                break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function following symmetric gradient approach"""
    np.random.seed(seed)
    n_points = 45  # Use odd number to allow symmetric projection
    
    # Initialize
    h_initial = initialize_h(n_points)
    h_initial = symmetric_project(h_initial, n_points)
    
    # Run optimization (no GA phase, just direct gradient-based)
    h_opt, c5_opt = optimize_symmetric_h(h_initial, n_points, budget_s)
    
    # Final verification
    h_opt = symmetric_project(h_opt, n_points)
    final_c5 = evaluate_c5(h_opt)
    
    return (h_opt, final_c5, n_points)
