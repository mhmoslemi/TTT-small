import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sample(n_points, n_samples):
    """Generate Latin Hypercube Samples"""
    samples = []
    for _ in range(n_samples):
        sample = np.random.rand(n_points)
        # Permute each dimension
        for i in range(n_points):
            sample[i] = (sample[i] + i) / n_points
        samples.append(sample)
    return samples

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    start_time = time.time()
    
    # Phase 1: Latin Hypercube Sampling
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    n_samples = 50
    sample_candidates = lhs_sample(n_points, n_samples)
    
    for i, sample in enumerate(sample_candidates):
        sample = project_h(sample, n_points)
        c5 = evaluate_h(sample, dx)
        if c5 < best_c5:
            best_h = sample.copy()
            best_c5 = c5
    
    # Phase 2: Evolution Strategy
    n_elites = 20
    n_offspring = 5
    elites = [best_h.copy()]
    
    # Generate elites from the best samples
    for i in range(n_elites - 1):
        sample = np.random.rand(n_points)
        # Add small random noise
        sample = sample * 0.1 + np.random.rand() * 0.9
        sample = project_h(sample, n_points)
        c5 = evaluate_h(sample, dx)
        if c5 < best_c5:
            best_h = sample.copy()
            best_c5 = c5
        elites.append(sample)
    
    # Evolution loop
    while time.time() - start_time < 0.7 * budget_s:
        offspring_pool = []
        for elite in elites:
            for _ in range(n_offspring):
                # Mutation
                child = elite + np.random.normal(0, 0.05, n_points)
                child = project_h(child, n_points)
                offspring_pool.append(child)
        
        # Evaluate offspring and combine with elites
        all_candidates = elites + offspring_pool
        fitness = [evaluate_h(h, dx) for h in all_candidates]
        
        # Select top elites
        sorted_indices = np.argsort(fitness)
        elites = [all_candidates[i] for i in sorted_indices[:n_elites]]
        
        # Update best
        new_c5 = fitness[sorted_indices[0]]
        if new_c5 < best_c5:
            best_h = elites[0].copy()
            best_c5 = new_c5

    # Phase 3: Gradient Descent Local Search
    h = best_h.copy()
    current_c5 = best_c5
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s - 2:
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient w.r.t. h
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < current_c5:
            current_c5 = new_c5
            patience = 0
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (best_h, best_c5, n_points)
