import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Setup
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Fourier basis and optimization
    m = 10  # number of cosine modes
    a = np.random.rand(m) * 0.5
    a[0] = 0.5
    a[1::2] = 0  # enforce even symmetry for better integral behavior
    
    # Optimization loop
    best_h = h.copy()
    best_C = float('inf')
    no_improvement_count = 0
    lr = 0.01
    
    def compute_h_from_a(a_coeffs):
        x = np.linspace(0, 2, n_points)
        h = a_coeffs[0] * np.ones_like(x)
        for k in range(1, len(a_coeffs)):
            # Use only cosine modes that preserve symmetry
            h += a_coeffs[k] * np.cos(np.pi * k * x / 2)
        return np.clip(h, 0, 1)
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def evaluate_and_update():
        nonlocal best_C, best_h
        h_vals = compute_h_from_a(a)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        C = compute_C(h_vals)
        if C < best_C:
            best_C = C
            best_h = h_vals.copy()
        return C

    # Initial evaluation
    previous_C = evaluate_and_update()
    
    iter_count = 0
    while time() - start_time < budget_s - 1:
        iter_count += 1
        # Generate h from current parameters
        h_vals = compute_h_from_a(a)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        
        # Compute current C5
        C = compute_C(h_vals)
        
        # Compute gradient with finite differences
        eps = 1e-5
        grad = np.zeros_like(a)
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus = compute_h_from_a(a_plus)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C) / eps
        
        # Update parameters using gradient descent
        a -= lr * grad
        a = np.clip(a, 0, 1)
        a[0] = 0.5  # preserve integral constraint
        
        # Adaptive learning rate
        new_C = evaluate_and_update()
        if new_C >= previous_C:
            no_improvement_count += 1
            if no_improvement_count > 100:
                lr *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
        previous_C = new_C

    return (best_h.tolist(), best_C, n_points)
