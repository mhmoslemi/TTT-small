import numpy as np
from scipy import optimize
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h symmetrically using a smooth function if no initial_h_values."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    
    # Symmetric initialization: combine two bumps
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * (x - 0.5))) + 0.5 * (1 + np.sin(np.pi * (x - 1.5)))
    h = project_h(h, n_points)
    return h

def enforce_symmetry(h, n_points):
    """Enforce symmetry h[i] = h[n_points-1-i] for i < n_points/2."""
    # For odd n_points, i ranges from 0 to n_points-1
    # Middle element stays unchanged
    for i in range(n_points // 2):
        mean_val = (h[i] + h[n_points - 1 - i]) / 2.0
        h[i] = h[n_points - 1 - i] = mean_val
    return h

def compute_gradient(h, max_shift, n_points):
    """Compute exact gradient for the max shift."""
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    
    # For convolution h * (1-h), the derivative of correlation for shift k is:
    # d/dh_i [h_i * (1 - h_{i+k})] = 1 - h_{i+k} for valid shifts
    # Assuming max_shift is 0-based offset in correlation output
    # We want the gradient to the part where maximum overlap occurs
    
    # Shift is placed at the very middle of the convolution which spans from -n+1 to n-1
    # Convolution index 0 corresponds to shift -n+1, index n-1 to shift 0, index 2n-2 to shift n-1
    shift = max_shift - (n_points - 1)  # actual shift in range [-n+1, n-1]
    # Now calculate gradient for all elements
    # The gradient of overlap with respect to h[i] depends on shifts that include h[i+k]
    for i in range(n_points):
        j = i + shift
        if 0 <= j < n_points:
            grad[i] = -dx * (1 - h[j]) 
    return grad

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function with symmetric gradient descent."""
    np.random.seed(seed)
    n_points = 45  # odd number to support symmetric structure
    h = initialize_h(n_points)
    h = enforce_symmetry(h, n_points)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Symmetric gradient descent with adaptive learning rate
    learning_rate = 0.1
    max_no_improvement = 1000
    iter_no_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s:
        # Evaluate C5 and find the shift that gives maximum overlap
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        max_shift = max_idx - (n - 1)  # Convert full convolution index to shift
        
        # Compute exact gradient
        grad = compute_gradient(h, max_idx, n_points)
        
        # Update step
        h_new = h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        h_new = enforce_symmetry(h_new, n_points)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            learning_rate *= 0.9  # Decrease learning rate slowly
        else:
            iter_no_improvement += 1
            learning_rate *= 1.2  # Increase learning rate
        
        # Reset learning rate if it goes too high or too low
        if learning_rate > 1.0:
            learning_rate = 0.1
        if learning_rate < 1e-6:
            learning_rate = 1e-6
            
        if iter_no_improvement > max_no_improvement:
            break
            
        iteration += 1
    
    best_h = project_h(best_h, n_points)
    best_h = enforce_symmetry(best_h, n_points)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
