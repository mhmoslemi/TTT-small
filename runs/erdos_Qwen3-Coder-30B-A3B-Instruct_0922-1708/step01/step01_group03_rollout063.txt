import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric function
    x = np.linspace(0, 2, n_points)
    # Create a piecewise linear function
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def coordinate_wise_optimize(h, n_points, max_iter=5000):
    """Optimize each coordinate individually to reduce C5."""
    h = h.copy()
    old_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = old_c5
    last_improvement = 0
    
    for iteration in range(max_iter):
        if iteration - last_improvement > 1000:
            break
        improved = False
        # Coordinate-wise optimization
        for i in range(n_points):
            # Store the old value
            old_val = h[i]
            # Look at all possible shifts with respect to the current value
            def compute_c5_at_val(val):
                # Temporarily set new value
                h[i] = val
                # Project to feasible space
                h = project_h(h, n_points)
                return evaluate_c5(h)
            
            # Find reasonable range
            bounds = [0., 1.]
            # Try a coarse search first
            best_val = old_val
            best_c5_val = old_c5
            test_vals = np.linspace(bounds[0], bounds[1], 20)
            for val in test_vals:
                if val != old_val:
                    h[i] = val
                    h = project_h(h, n_points)
                    c5_val = evaluate_c5(h)
                    if c5_val < best_c5_val:
                        best_c5_val = c5_val
                        best_val = val
            # Set new value
            h[i] = best_val
            h = project_h(h, n_points)
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                improved = True
                last_improvement = iteration
        if not improved:
            break
    return best_h, best_c5

def nonlinear_cg_optimize(h, n_points, max_iter=5000, time_limit_s=600):
    """Perform nonlinear conjugate gradient optimization."""
    start_time = time.time()
    h = h.copy()
    old_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = old_c5
    last_improvement = 0

    # For simplicity, we'll use a basic gradient approximation method
    def finite_diff_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad

    def line_search(h, grad, max_alpha=1.0, eps=1e-6):
        """Simple backtracking line search."""
        alpha = max_alpha
        current_c5 = evaluate_c5(h)
        while alpha > eps:
            new_h = h - alpha * grad
            new_h = project_h(new_h, n_points)
            new_c5 = evaluate_c5(new_h)
            if new_c5 < current_c5:
                return alpha, new_h
            alpha *= 0.5
        return alpha, h  # fallback

    # Initial direction
    grad = finite_diff_gradient(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_limit_s * 0.8:
            break
        # Find step size
        alpha, new_h = line_search(h, grad)
        if alpha == 0:
            break
        # Update
        h = new_h
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
        else:
            if iteration - last_improvement > 1000:
                break
                
        # Compute gradient
        old_grad = grad
        grad = finite_diff_gradient(h)
        
        # Polak-Ribiere formula for conjugate directions
        beta = max(0, np.dot(grad, grad - old_grad) / np.dot(old_grad, old_grad))
        if np.isnan(beta):
            beta = 0
        grad = grad + beta * old_grad

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45  # Optimal balance of speed and quality
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: Coordinate-wise optimization
    h_coord, c5_coord = coordinate_wise_optimize(h, n_points, max_iter=3000)
    
    if time.time() - start_time > 0.7 * budget_s:
        return (h_coord, c5_coord, n_points)
        
    # Phase 2: Nonlinear Conjugate Gradient
    h_cg, c5_cg = nonlinear_cg_optimize(h_coord, n_points, max_iter=2000, time_limit_s=budget_s)
    
    # Final validation
    final_h = h_cg if c5_cg < c5_coord else h_coord
    final_c5 = min(c5_cg, c5_coord)
    
    return (final_h, final_c5, n_points)
