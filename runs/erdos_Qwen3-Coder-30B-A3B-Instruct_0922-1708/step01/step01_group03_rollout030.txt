import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: triangular pulse
    x = np.linspace(0, 2, n_points)
    h = np.maximum(0, 1 - np.abs(x - 1))
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO + Gradient Descent."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45  # odd for symmetry
    
    # Initialize symmetric particles
    def create_symmetric_h():
        h = initialize_h(n_points)
        # enforce symmetry h[i] = h[n_points-1-i]
        mid = n_points // 2
        for i in range(mid):
            h[n_points-1-i] = h[i]
        return project_h(h, n_points)
    
    # PSO parameters
    swarm_size = 30
    iterations = 0
    max_iterations = 1000
    cognitive_weight = 2.0
    social_weight = 2.0
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Initialize swarm
    particles = [create_symmetric_h() for _ in range(swarm_size)]
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    pbests = [p.copy() for p in particles]
    pbest_c5 = [evaluate_c5(p) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbests[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # Timing
    start_time = time.time()
    
    # PSO loop
    while iterations < max_iterations and time.time() - start_time < budget_s * 0.7:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                           cognitive_weight * r1 * (pbests[i] - particles[i]) + 
                           social_weight * r2 * (gbest - particles[i]))
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to valid range and rescale
            particles[i] = project_h(particles[i], n_points)
            
            # Evaluate
            c5 = evaluate_c5(particles[i])
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbests[i] = particles[i].copy()
                pbest_c5[i] = c5
                
                # Update global best
                if c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = c5
        
        # Update inertia weight
        inertia_weight = 0.9 - (0.5 * iterations / max_iterations)
        iterations += 1
    
    # Gradient descent refinement
    # Simple projected gradient descent
    h = gbest
    lr = 0.05
    max_iter = 5000
    no_improvement = 0
    min_improvement = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient (simplified approx)
        # For small deviation, approximate gradient
        eps = 1e-4
        
        # Compute large overlap in C5 (best shift)
        # This is for performance let's sample a few key shifts manually
        conv_vals = []
        conv = np.correlate(h, 1 - h, mode="full")
        shift = np.argmax(conv) - (n_points - 1)
        
        # Apply gradient descent update on h
        # Compute gradient numerically for dominant shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update step
        h_new = h - lr * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < gbest_c5:
            h = h_new
            gbest_c5 = new_c5
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 1000:
                break
            
        # Decrease learning rate if required
        if lr > 1e-5 and iteration % 100 == 0:
            lr *= 0.95

    return (h, gbest_c5, n_points)
