import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create flat start
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5  # Flat start
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Use PSO for global exploration
    n_particles = 100
    max_iter = 1000
    w = 0.7   # Inertia weight
    c1 = 1.5  # Cognitive parameter
    c2 = 1.5  # Social parameter
    
    # Particle positions and velocities (as nodal values)
    positions = np.tile(h, (n_particles, 1))
    velocities = np.random.rand(n_particles, n_points) * 0.1
    
    # Introduce some randomness for diversity
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            positions[i] = initial_h_values
        else:
            positions[i] = np.random.rand(n_points) * 0.5 + 0.25
    
    pbest_positions = positions.copy()
    pbest_values = [float('inf')] * n_particles
    
    # Global best
    gbest_value = float('inf')
    gbest_position = h.copy()
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def evaluate_individual(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return compute_C(h_vals)
    
    # Main PSO loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s - 2:
            break
            
        for i in range(n_particles):
            # Evaluate particle
            current_cost = evaluate_individual(positions[i])
            
            # Update personal best
            if current_cost < pbest_values[i]:
                pbest_values[i] = current_cost
                pbest_positions[i] = positions[i].copy()
                
            # Update global best
            if current_cost < gbest_value:
                gbest_value = current_cost
                gbest_position = positions[i].copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - positions[i]) +
                            c2 * r2 * (gbest_position - positions[i]))
            
            positions[i] += velocities[i]
    
    # Refinement with gradient descent on best solution
    best_h = gbest_position.copy()
    lr = 0.01
    best_C = gbest_value
    no_improvement_count = 0
    
    def penalized_gradient(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        
        C = compute_C(h_vals)
        
        # Gradient approximation
        eps = 1e-5
        grad = np.zeros_like(h_vals)
        for j in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[j] = (C_plus - C) / eps
            
        return grad
    
    # Gradient descent refinement
    for i in range(5000):
        if time() - start_time > budget_s - 1:
            break
            
        grad = penalized_gradient(best_h)
        best_h -= lr * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        
        new_C = compute_C(best_h)
        if new_C < best_C:
            best_C = new_C
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                lr *= 0.7
                no_improvement_count = 0
                if lr < 1e-6:
                    break
    
    return (best_h.tolist(), best_C, n_points)
