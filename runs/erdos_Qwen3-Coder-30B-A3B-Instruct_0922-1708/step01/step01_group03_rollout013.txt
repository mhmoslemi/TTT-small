import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Use provided initial construction
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use smooth function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def line_search(h, i, delta, n_points):
    """Perform line search along dimension i."""
    # Compute the maximum value of the max convolution with current h
    # For a small change in h[i], find the best h[i] that reduces overlap
    orig_h = h[i]
    min_c5 = np.inf
    best_h_i = orig_h
    step_size = delta

    # Sample several values around original
    test_vals = np.linspace(max(0, orig_h - 2*step_size), min(1, orig_h + 2*step_size), 20)
    test_vals = np.clip(test_vals, 0, 1)
    
    # Test different values for h[i] while adjusting others to maintain sum
    for h_i in test_vals:
        h_test = h.copy()
        h_test[i] = h_i
        
        h_test = project_h(h_test, n_points)
        c5 = evaluate_c5(h_test)
        if c5 < min_c5:
            min_c5 = c5
            best_h_i = h_i

    return best_h_i, min_c5

def coordinate_wise_optimization(h, n_points, max_iter=5000, time_budget_s=600):
    """Optimize coordinate by coordinate with line search."""
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()
    iter_count = 0
    last_improvement = 0
    
    # Adaptive step size
    step_size = 0.05
    
    for i in range(n_points):
        if time.time() - start_time > time_budget_s * 0.8:
            break
            
        if iter_count % 100 == 0:
            pass  # Periodic performance update
            
        h_new_i, new_c5 = line_search(best_h, i, step_size, n_points)
        best_h[i] = h_new_i
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            last_improvement = iter_count
        else:
            # sometimes increase step size when no improvement
            step_size = min(0.1, step_size * 1.05)
            
        iter_count += 1
        
        if iter_count - last_improvement > 500:
            step_size *= 0.8
            last_improvement = iter_count
            
    return best_h, best_c5

def nonlinear_cg_optimization(h, n_points, time_budget_s=600):
    """Use nonlinear conjugate gradient to refine solution."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Initial direction
    g = np.zeros(n_points)
    eps = 1e-5
    for i in range(n_points):
        h_plus = best_h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus)
        h_minus = best_h.copy()
        h_minus[i] -= eps
        h_minus = project_h(h_minus, n_points)
        c5_minus = evaluate_c5(h_minus)
        g[i] = (c5_plus - c5_minus) / (2 * eps)
    
    # Initial search direction (negative gradient)
    d = -g
    g_prev = g.copy()
    
    # Polak-Ribière parameter
    beta = 0.0
    
    iter_count = 0
    max_iter = 5000
    
    # Line search parameters
    alpha = 0.01
    max_line_search_iter = 20
    
    while time.time() - start_time < time_budget_s * 0.2 and iter_count < max_iter:
        iter_count += 1
        if iter_count % 100 == 0:
            pass
            
        # Find step size along current direction
        # Simple backtracking line search
        step_size = 1.0
        for _ in range(max_line_search_iter):
            h_new = best_h + step_size * d
            h_new = project_h(h_new, n_points)
            
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5 - 0.0001 * step_size * np.dot(g, d):
                break
            step_size *= 0.5
            
        # Update h
        best_h = best_h + step_size * d
        best_h = project_h(best_h, n_points)
        new_c5 = evaluate_c5(best_h)
        
        # Update gradient
        old_c5 = best_c5
        best_c5 = new_c5
        g = np.zeros(n_points)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            g[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Polak-Ribière formula
        if np.linalg.norm(g) > 1e-12:
            beta = max(0, np.dot(g - g_prev, g) / (np.linalg.norm(g_prev)**2 + 1e-12))
        else:
            beta = 0.0
            
        # Update direction
        g_prev = g.copy()
        d = -g + beta * d
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45  # Use number that works well and is fast
    h = initialize_h(n_points)
    
    # Phase 1: Coordinate-wise optimization
    h_coord, c5_coord = coordinate_wise_optimization(h, n_points, time_budget_s=budget_s)
    
    # Phase 2: Nonlinear conjugate gradient
    h_cg, c5_cg = nonlinear_cg_optimization(h_coord, n_points, time_budget_s=budget_s)
    
    # Return best result
    final_h = h_cg if c5_cg < c5_coord else h_coord
    final_c5 = min(c5_cg, c5_coord)
    
    # Project to ensure constraints
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
