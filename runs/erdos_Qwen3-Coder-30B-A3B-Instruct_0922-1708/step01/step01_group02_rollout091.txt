import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx

    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Use a hybrid approach - alternate between gradient descent and local search
    
    # Gradient based optimization
    learning_rate = 0.1
    iterations = 0
    max_iterations = 5000
    
    while iterations < max_iterations and (time.time() - start_time) < budget_s - 1:
        h_current = h.copy()
        c5_current = compute_c5(h_current)
        
        # Estimate gradient with finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_current) / eps
        
        # Update using gradient
        h = h_current - learning_rate * grad
        h = project_h(h)
        
        # Update best if improved
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Anneal learning rate
        if iterations % 1000 == 0 and iterations > 0:
            learning_rate *= 0.5
            
        iterations += 1
        
        # Early termination check
        if (time.time() - start_time) > budget_s - 1:
            break
            
    # Additionally perform a local search on best solution
    local_iter = 0
    max_local_iter = 1000
    while local_iter < max_local_iter and (time.time() - start_time) < budget_s - 1:
        # Random perturbations with small step size
        h_trial = best_h.copy()
        # Perturb few random elements
        num_pert = max(1, n_points // 20)
        indices = np.random.choice(n_points, size=num_pert, replace=False)
        for i in indices:
            step = np.random.normal(0, 0.01)
            h_trial[i] += step
        h_trial = project_h(h_trial)
        
        cand_c5 = compute_c5(h_trial)
        if cand_c5 < best_c5:
            best_h = h_trial.copy()
            best_c5 = cand_c5
            
        local_iter += 1
        
    # Final projection and evaluation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
