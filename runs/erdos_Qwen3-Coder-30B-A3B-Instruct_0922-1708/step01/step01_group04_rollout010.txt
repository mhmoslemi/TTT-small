import time

        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap

    def optimize_segments(c_values, m, n_points, dx):
        # Reconstruct h from segment values
        segment_width = n_points // m
        if segment_width * m < n_points:
            segment_width += 1
        h = np.zeros(n_points)
        for i in range(m):
            start_idx = i * segment_width
            end_idx = min((i + 1) * segment_width, n_points)
            h[start_idx:end_idx] = c_values[i]
        
        # Truncate if necessary
        h = h[:n_points]
        
        # Ensure normalization constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        
        return h

    def objective(c_values, m, n_points, dx):
        h = optimize_segments(c_values, m, n_points, dx)
        return evaluate(h)

    def constraint_sum(c_values, m):
        return np.sum(c_values) - m / 2.0

        bounds = [(0, 1) for _ in range(m)]

        # Optimize using SLSQP
        result = minimize(objective, c_vals, args=(m, n_points, dx),
