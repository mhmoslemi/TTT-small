import numpy as np
import scipy.optimize as so
from time import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create a smart initial guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # A reasonable starting construction
        h = np.zeros(n_points)
        # Fill with some structure to start with good overlap
        segments = 5
        seg_size = n_points // segments
        for i in range(segments):
            start = i * seg_size
            end = start + seg_size if i < segments - 1 else n_points
            h[start:end] = 0.5 + 0.3 * np.sin(np.linspace(0, np.pi, end-start))
        h = np.clip(h, 0, 1)
        
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    
    # Function to compute C5
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # Simulated Annealing with adaptive cooling
    best_h = h.copy()
    best_C = compute_C(h)
    
    # Parameters
    T = 1.0
    alpha = 0.995
    min_T = 1e-6
    max_iter = 50000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    # Annealing loop
    while iter_count < max_iter and time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Create neighbor solution
        h_new = h.copy()
        # Randomly modify a few segments
        num_changes = np.random.randint(1, 5)
        for _ in range(num_changes):
            idx = np.random.randint(0, n_points)
            # Perturb the value
            perturbation = np.random.normal(0, 0.1)
            h_new[idx] = np.clip(h_new[idx] + perturbation, 0, 1)
        
        # Rescale to maintain sum constraint
        h_new = h_new * (target_sum / np.sum(h_new))

        # Compute C5 for new configuration
        C_new = compute_C(h_new)
        
        # Accept or reject
        delta = C_new - best_C
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if C_new < best_C:
                best_C = C_new
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Adaptive cooling
        if no_improvement_count > max_no_improvement:
            T = max(min_T, T * alpha)
            no_improvement_count = 0
            
        # Check early termination
        if T < min_T or no_improvement_count > 2000:
            break

    # Gradient descent refinement
    def gradient_step(h_vals, step_size=0.01):
        h = h_vals.copy()
        current_C = compute_C(h)
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = h_plus * (target_sum / np.sum(h_plus))  # Renormalize
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - current_C) / eps
        h_new = h - step_size * grad
        # Clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        return h_new

    # Refinement steps
    step_size = 0.01
    for i in range(200):
        if time() - start_time > budget_s - 1:
            break
        h = gradient_step(best_h, step_size)
        new_C = compute_C(h)
        if new_C < best_C:
            best_C = new_C
            best_h = h.copy()
        else:
            step_size *= 0.99
            if step_size < 1e-6:
                step_size = 1e-6
    
    return (best_h.tolist(), best_C, n_points)
