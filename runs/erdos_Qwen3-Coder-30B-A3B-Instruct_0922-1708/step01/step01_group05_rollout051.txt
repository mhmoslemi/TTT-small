import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO Parameters
    n_particles = 100
    w = 0.7  # Inertia weight
    c1 = 1.5  # Cognitive parameter
    c2 = 1.5  # Social parameter
    max_iter = 500
    
    # Initialize particles and velocities
    particles = np.random.rand(n_particles, n_points)
    particles = np.clip(particles, 0, 1)
    velocities = np.random.rand(n_particles, n_points) * 0.1
    personal_best = particles.copy()
    personal_best_values = np.full(n_particles, float('inf'))
    
    # Global best
    global_best_idx = 0
    global_best_value = float('inf')
    
    # PSO iterations
    iter_count = 0
    while time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        # Evaluate particles
        for i in range(n_particles):
            particle = particles[i]
            # Ensure particle satisfies constraint
            particle = particle * (target_sum / np.sum(particle))
            
            # Compute C_5 for this particle
            corr = np.correlate(particle, 1-particle, mode="full")
            corr *= dx
            C_val = np.max(np.abs(corr))
            
            # Update personal best
            if C_val < personal_best_values[i]:
                personal_best_values[i] = C_val
                personal_best[i] = particle.copy()
                
            # Update global best
            if C_val < global_best_value:
                global_best_value = C_val
                global_best_idx = i
                
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            # Rescale for integral constraint
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
    
    # Refinement with gradient descent on best particle
    best_h = personal_best[global_best_idx].copy()
    best_C = global_best_value
    
    # Refinement using scipy.optimize
    def objective(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def constraint_func(h_vals):
        return np.sum(h_vals) - target_sum
    
    # Use L-BFGS-B to minimize C
    # Initial guess
    x0 = best_h.copy()
    # Define constraint as a dict
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Optimize
    result = so.minimize(objective, x0, method='L-BFGS-B', 
                         bounds=[(0, 1) for _ in range(n_points)], 
                         constraints=cons, options={'maxiter': 100})
    
    if result.success:
        refined_h = result.x
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = refined_h * (target_sum / np.sum(refined_h))
        corr = np.correlate(refined_h, 1-refined_h, mode="full")
        corr *= dx
        refined_C = np.max(np.abs(corr))
        if refined_C < best_C:
            best_h = refined_h
            best_C = refined_C
    
    return (best_h.tolist(), best_C, n_points)
