import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    # Optimizer settings
    max_iter = 5000
    lr = 0.01
    patience = 100
    no_improvement_count = 0
    
    # Cosine basis functions
    def compute_h_from_coeffs(a_coeffs, phi_coeffs):
        x = np.linspace(0, 2, n_points)
        h = a_coeffs[0] * np.ones_like(x)
        for k in range(1, len(a_coeffs)):
            h += a_coeffs[k] * np.cos(np.pi * k * x / 2 + phi_coeffs[k-1])
        return np.clip(h, 0, 1)
    
    def compute_C(h_vals):
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def rescale_h(h_vals):
        return h_vals * (target_sum / np.sum(h_vals))
    
    # Track best solution
    best_h = h.copy()
    best_C = compute_C(h)
    
    # Initialize cosine coefficients
    m = 10
    a = np.random.rand(m) * 0.5
    a[0] = 0.5  # Integral constraint
    
    phi = np.random.rand(m-1) * 2 * np.pi
    
    iter_count = 0
    while time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Compute h(x)
        h_current = compute_h_from_coeffs(a, phi)
        h_current = rescale_h(h_current)
        
        # Compute C5
        C_current = compute_C(h_current)
        
        # Store best
        if C_current < best_C:
            best_C = C_current
            best_h = h_current.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > patience:
            lr *= 0.9
        
        # Finite difference gradients
        eps = 1e-5
        grad_a = np.zeros_like(a)
        grad_phi = np.zeros(len(phi))
        
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus = compute_h_from_coeffs(a_plus, phi)
            h_plus = rescale_h(h_plus)
            C_plus = compute_C(h_plus)
            grad_a[i] = (C_plus - C_current) / eps
            
        for i in range(len(phi)):
            phi_plus = phi.copy()
            phi_plus[i] += eps
            h_plus = compute_h_from_coeffs(a, phi_plus)
            h_plus = rescale_h(h_plus)
            C_plus = compute_C(h_plus)
            grad_phi[i] = (C_plus - C_current) / eps
            
        # Update coefficients
        a -= lr * grad_a
        a = np.clip(a, 0, 1)
        phi -= lr * grad_phi
        phi = phi % (2 * np.pi)
        
    # Final refinement via binary search
    h_refined = best_h.copy()
    
    def binary_refine(h_vals, max_attempts=1000):
        for _ in range(max_attempts):
            # Find threshold that yields target sum
            h_sorted = np.sort(h_vals)
            idx = int(target_sum)
            if idx >= len(h_sorted):
                t = h_sorted[-1] + 1e-5
            else:
                t = h_sorted[idx]
                
            # Perform binary step function
            h_step = (h_vals > t).astype(float)
            h_step = h_step * (target_sum / np.sum(h_step))  # Rescale
            
            # Compute new C5
            C_new = compute_C(h_step)
            
            if C_new < best_C:
                best_C = C_new
                h_refined = h_step.copy()
            else:
                break
                
        return h_refined, best_C
    
    # Do binary refinement
    h_final, C_final = binary_refine(h_refined)
    
    return (h_final.tolist(), C_final, n_points)
