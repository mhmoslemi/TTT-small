import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Initialize breakpoints
    breakpoints = np.linspace(0, 2, 11)  # 10 intervals
    interval_values = np.ones(10) * 0.5
    
    def compute_overlap(h_vals):
        # Compute overlap integral
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
        
    def compute_h_from_intervals(breakpoints, interval_values):
        # Construct h from breakpoints and interval_values
        h = np.zeros(n_points)
        for i in range(len(breakpoints)-1):
            start_idx = int(np.round(breakpoints[i] / dx))
            end_idx = int(np.round(breakpoints[i+1] / dx))
            end_idx = min(end_idx, n_points)
            if start_idx < end_idx:
                h[start_idx:end_idx] = interval_values[i]
        return h
    
    def objective_and_gradient(breakpoints_and_values):
        # Split into breakpoints and interval values
        breakpoints = breakpoints_and_values[:10]
        interval_values = breakpoints_and_values[10:]
        
        # Ensure valid breakpoints
        breakpoints = np.clip(breakpoints, 0, 2)
        breakpoints = np.sort(breakpoints)
        breakpoints = np.concatenate([[0], breakpoints, [2]])
        
        # Ensure valid interval values
        interval_values = np.clip(interval_values, 0, 1)
        
        # Reconstruct h
        h = compute_h_from_intervals(breakpoints, interval_values)
        
        # Rescale to meet integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute objective
        c5 = compute_overlap(h)
        
        # Compute gradient (simplified finite difference)
        grad = np.zeros_like(breakpoints_and_values)
        eps = 1e-5
        
        for i in range(len(breakpoints_and_values)):
            perturbed = breakpoints_and_values.copy()
            perturbed[i] += eps
            c5_plus = objective_and_gradient(perturbed)[0]
            grad[i] = (c5_plus - c5) / eps
            
        return c5, grad
    
    # Optimization loop
    start_time = time.time()
    best_c5 = float('inf')
    best_breakpoints = breakpoints.copy()
    best_interval_values = interval_values.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Construct h from current parameters
        h = compute_h_from_intervals(breakpoints, interval_values)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute C5
        current_c5 = compute_overlap(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_breakpoints = breakpoints.copy()
            best_interval_values = interval_values.copy()
            
        # Optimizer with momentum and learning rate adjustments
        # Sample gradient descent update
        lr_bp = 0.01
        lr_val = 0.1
        
        # Update breakpoints
        new_breakpoints = breakpoints.copy()
        for i in range(len(breakpoints)):
            if i == 0 or i == len(breakpoints)-1:
                continue
            new_bp = breakpoints[i] - lr_bp * (1.0 if i < len(breakpoints)-1 else -1.0)
            new_breakpoints[i] = np.clip(new_bp, breakpoints[i-1], breakpoints[i+1])
        breakpoints = new_breakpoints
        
        # Update interval values
        new_interval_values = interval_values.copy()
        for i in range(len(interval_values)):
            # Compute gradient direction (simplified)
            h_test = compute_h_from_intervals(breakpoints, new_interval_values)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            current_c5_base = compute_overlap(h_test)
            
            h_plus = compute_h_from_intervals(breakpoints, new_interval_values)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            # Add epsilon to one element
            h_plus[i] += 1e-5
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_overlap(h_plus)
            
            grad_val = (c5_plus - current_c5_base) / 1e-5
            new_interval_values[i] = np.clip(interval_values[i] - lr_val * grad_val, 0, 1)
        interval_values = new_interval_values
    
    # Ensure final solution
    final_h = compute_h_from_intervals(best_breakpoints, best_interval_values)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = compute_overlap(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
