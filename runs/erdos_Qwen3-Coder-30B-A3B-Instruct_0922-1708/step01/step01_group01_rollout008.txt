import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO with gradient refinement"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO Parameters
    pop_size = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    stagnation_limit = 500
    min_pop_size = 20
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    # Create initial swarm
    for i in range(pop_size):
        if i == 0:
            particle = h.copy()
        else:
            # Start with small perturbations of the initial solution
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_h(particle, n_points)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_scores.append(evaluate_h(particle, dx))
    
    # Global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    # PSO main loop
    start_time = time.time()
    iter_count = 0
    stagnation_count = 0
    
    while time.time() - start_time < 0.7 * budget_s and iter_count < max_iter:
        iter_count += 1
        
        for i in range(len(swarm)):
            particle = swarm[i]
            vel = velocities[i]
            current_score = pbest_scores[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (w * vel + 
                   c1 * r1 * (pbest[i] - particle) + 
                   c2 * r2 * (gbest - particle))
            
            # Update position
            particle = particle + vel
            particle = project_h(particle, n_points)
            
            # Evaluate
            new_score = evaluate_h(particle, dx)
            
            # Update personal best
            if new_score < current_score:
                pbest[i] = particle.copy()
                pbest_scores[i] = new_score
                
                # Update global best
                if new_score < gbest_score:
                    gbest = particle.copy()
                    gbest_score = new_score
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            
            # Store updated particle and velocity
            swarm[i] = particle
            velocities[i] = vel
            
        # Adapt swarm size if stagnating
        if stagnation_count > stagnation_limit and len(swarm) > min_pop_size:
            # Reduce swarm by removing worst performers
            scores = [evaluate_h(p, dx) for p in swarm]
            worst_indices = np.argsort(scores)[-10:]  # Remove top 10%
            new_swarm = [swarm[i] for i in range(len(swarm)) if i not in worst_indices]
            new_velocities = [velocities[i] for i in range(len(velocities)) if i not in worst_indices]
            new_pbest = [pbest[i] for i in range(len(pbest)) if i not in worst_indices]
            new_pbest_scores = [pbest_scores[i] for i in range(len(pbest_scores)) if i not in worst_indices]
            
            swarm = new_swarm
            velocities = new_velocities
            pbest = new_pbest
            pbest_scores = new_pbest_scores
            stagnation_count = 0
    
    # Gradient descent refinement
    h_refined = gbest.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < 0.9 * budget_s:
        # Find the maximum overlap
        corr = correlate(h_refined, 1 - h_refined, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute finite difference gradient
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_refined.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with gradient
        update = learning_rate * np.sign(grad)
        h_refined = project_h(h_refined - update, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h_refined, dx)
        if new_c5 < gbest_score:
            gbest = h_refined.copy()
            gbest_score = new_c5
            patience = 0
        else:
            patience += 1
            if patience > 0 and patience % 1000 == 0:
                learning_rate *= 0.5
                patience = 0
        if patience > max_patience:
            break
    
    return (gbest, gbest_score, n_points)
