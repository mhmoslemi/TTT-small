import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # This will be available when run as a submission
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    
    # Default initialization: piecewise linear
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    # Create a two-bump structure
    for i in range(n_points):
        xi = x[i]
        h[i] = max(0, min(1, 2*(xi - 0.5) + 1)) if xi <= 1 else max(0, min(1, 2*(1.5 - xi) + 1))
    # Ensure sum constraint
    h = project_h(h, n_points)
    return h

def enforce_symmetry(h, n_points):
    """Enforce symmetric structure h[i] = h[n_points-1-i]"""
    if n_points % 2 == 1:
        mid = n_points // 2
        for i in range(mid):
            avg = (h[i] + h[n_points - 1 - i]) / 2.0
            h[i] = h[n_points - 1 - i] = avg
    else:
        mid = n_points // 2
        for i in range(mid):
            avg = (h[i] + h[n_points - 1 - i]) / 2.0
            h[i] = h[n_points - 1 - i] = avg
    return h

def compute_exact_gradient(h, max_shift):
    """Compute exact gradient of C5 w.r.t. h for the max shift."""
    n = len(h)
    dx = 2.0 / n
    G = np.zeros(n)
    # The derivative w.r.t. h[i] is (1 - h[i + max_shift]) * dx for valid indices
    # and (1 - h[i + max_shift - n]) * dx for wraps
    if max_shift >= 0:
        for i in range(n):
            j = i + max_shift
            if j < n:
                G[i] = (1 - h[j]) * dx
            else:
                j_wrap = j - n
                G[i] = (1 - h[j_wrap]) * dx
    else:
        for i in range(n):
            j = i + max_shift
            if j >= 0:
                G[i] = (1 - h[j]) * dx
            else:
                j_wrap = j + n
                G[i] = (1 - h[j_wrap]) * dx
    return G

def optimized_gradient_descent(h, n_points, time_budget_s):
    """Run symmetric gradient descent with adaptive steps."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.1
    iter_no_improvement = 0
    max_no_improvement = 10000
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.95:
        # Find the shift with maximum overlap
        current_c5 = evaluate_c5(best_h)
        # Get indices from convolution for reference
        conv = np.correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(conv)
        # Compute shift in [-n+1, n-1]
        shift = max_idx - (n_points - 1)
        
        # Compute gradient
        g = compute_exact_gradient(best_h, shift)
        
        # Update
        h_new = best_h - learning_rate * g
        h_new = project_h(h_new, n_points)
        h_new = enforce_symmetry(h_new, n_points)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            # Adaptive learning rate
            learning_rate = max(1e-6, learning_rate * 0.98)
        else:
            iter_no_improvement += 1
            learning_rate = min(1.0, learning_rate * 1.05)
            if learning_rate < 1e-6:
                break
                
        if iter_no_improvement > max_no_improvement:
            break
            
        iteration += 1
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45  # Odd for symmetry
    h = initialize_h(n_points)
    h = enforce_symmetry(h, n_points)
    start_time = time.time()
    
    # Phase: Optimized Gradient Descent
    h_final, c5_final = optimized_gradient_descent(h, n_points, budget_s)
    
    # Final projection to ensure feasibility
    h_final = project_h(h_final, n_points)
    h_final = enforce_symmetry(h_final, n_points)
    c5_final = evaluate_c5(h_final)
    
    return (h_final, c5_final, n_points)
