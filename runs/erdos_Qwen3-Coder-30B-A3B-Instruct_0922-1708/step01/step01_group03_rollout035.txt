import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a symmetric step function with triangular shape
    x = np.linspace(0, 2, n_points)
    # Create symmetric piecewise linear function
    h = np.zeros(n_points)
    for i in range(n_points):
        xi = x[i]
        if 0 <= xi <= 1:
            h[i] = 2 * xi
        elif 1 < xi <= 2:
            h[i] = 2 * (2 - xi)
        else:
            h[i] = 0
    h = project_h(h, n_points)
    return h

def line_search_update(h, i, conv_vals, n_points):
    """Perform line search for updating h[i] along the direction of steepest descent."""
    # Get current value and compute its contribution to C5
    h_current = h[i]
    # Evaluate at current position
    c5_current = evaluate_c5(h)
    
    # Try values in a grid around current value to find good update
    step_size = 0.05  # Initial step size
    best_h = h.copy()
    best_c5 = c5_current
    
    # Try symmetric neighbor perturbations
    for delta in np.arange(-0.2, 0.201, 0.05):
        h_test = h.copy()
        h_test[i] += delta
        h_test = project_h(h_test, n_points)
        c5_test = evaluate_c5(h_test)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
    
    return best_h, best_c5

def coordinate_descent_phase(h, n_points, time_budget_s):
    """Coordinate-wise optimization with line search."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    iteration = 0
    no_improvement = 0
    max_no_improvement = 5000
    
    while time.time() - start_time < time_budget_s * 0.7 and no_improvement < max_no_improvement:
        improved = False
        for i in range(n_points):
            h_new, c5_new = line_search_update(best_h, i, None, n_points)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                improved = True
                no_improvement = 0
            else:
                no_improvement += 1
                
        if not improved:
            no_improvement += 1
        iteration += 1
        
    return best_h, best_c5

def conjugate_gradient_phase(h, n_points, time_budget_s):
    """Perform conjugate gradient optimization."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    max_iter = 1000
    iter_no_improvement = 0
    max_no_improvement = 5000
    
    # Compute initial gradient
    # Approximate gradient using finite differences
    eps = 1e-5
    prev_h = best_h.copy()
    prev_c5 = best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_budget_s * 0.3:
            break
            
        # Compute gradient approximated by finite differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # CG update using negative gradient as starting point
        # In this simplified version, update h directly
        learning_rate = 0.05
        new_h = best_h - learning_rate * grad
        new_h = project_h(new_h, n_points)
        new_c5 = evaluate_c5(new_h)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            
        if iter_no_improvement > max_no_improvement:
            break
            
        # Reduce learning rate once in a while
        if iteration % 100 == 0:
            learning_rate *= 0.9
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45  # Keeping smaller value for higher speed
    
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: Coordinate Descent
    h_cd, c5_cd = coordinate_descent_phase(h, n_points, budget_s)
    
    # Phase 2: Conjugate Gradient
    h_cg, c5_cg = conjugate_gradient_phase(h_cd, n_points, budget_s)
    
    # Return best result
    final_h = h_cg if c5_cg < c5_cd else h_cd
    final_c5 = min(c5_cg, c5_cd)
    
    # Final validation and voting
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
