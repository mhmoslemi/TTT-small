import numpy as np
from scipy import optimize
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: simple symmetric function
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    # Create two Gaussian-like bumps centered at 0.5 and 1.5
    for i in range(n_points):
        xi = x[i]
        h[i] = max(0, min(1, 0.5 * (1 + np.exp(-10*(xi - 0.5)**2)) + 0.5 * (1 + np.exp(-10*(xi - 1.5)**2))))
    # Apply projection
    h = project_h(h, n_points)
    return h

def compute_exact_gradient(h, max_shift):
    """Compute the exact gradient for the specified shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # For the mode='full' correlation, the gradient is known at the maximum shift
    # Specifically: ∂C5/∂h_i = (1 - h_{i+k}) * dx where k is the shift
    # Indices involved are: i+k (i.k.a. i + shift)
    if max_shift >= 0:
        # For positive shifts
        for i in range(n):
            j = i + max_shift
            if j < n:
                grad[i] = (1 - h[j]) * dx
    else:
        # For negative shifts
        for i in range(n):
            j = i + max_shift
            if j >= 0:
                grad[i] = (1 - h[j]) * dx
    return grad

def find_max_shift(h):
    """Find which shift gives maximum overlap."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(conv)
    shift = max_idx - (n - 1)
    return shift

def symmetric_project(h):
    """Project h to be symmetric around center."""
    n = len(h)
    if n % 2 == 1:
        k = n // 2  # Middle index
        # Keep middle unchanged, average symmetric elements
        for i in range(k):
            avg = (h[i] + h[n-1-i]) / 2.0
            h[i] = h[n-1-i] = avg
    else:
        k = n // 2
        # Average pairs
        for i in range(k):
            avg = (h[i] + h[n-1-i]) / 2.0
            h[i] = h[n-1-i] = avg
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function with symmetric gradient descent."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # odd for symmetry
    h = initialize_h(n_points)
    # Ensure initial h is symmetric
    h = symmetric_project(h)
    h = project_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    learning_rate = 0.1
    max_no_improvement = 1000
    iter_no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        # Find max shift
        max_shift = find_max_shift(best_h)
        
        # Compute exact gradient
        grad = compute_exact_gradient(best_h, max_shift)
        
        # Update with learning rate
        h_new = best_h - learning_rate * grad
        h_new = symmetric_project(h_new)
        h_new = project_h(h_new, n_points)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            if learning_rate > 1e-6:
                learning_rate *= 0.95  # Decrease learning rate very slightly
        else:
            iter_no_improvement += 1
            learning_rate *= 2.0  # Increase learning rate
            if learning_rate > 1.0:
                learning_rate = 1.0
                
        iteration += 1
        if iter_no_improvement > max_no_improvement:
            break
            
        if iteration % 100 == 0:
            pass  # Periodic progress notice
    
    # Final synchronization check
    best_h = symmetric_project(best_h)
    best_h = project_h(best_h, n_points)
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
