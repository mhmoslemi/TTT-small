import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 5000
    mutation_rate_init = 0.2
    crossover_rate = 0.7
    elite_size = population_size // 10
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h_candidate = np.random.rand(n_points)
        h_candidate = project_h(h_candidate)
        population.append(h_candidate)
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        population[0] = project_h(np.array(initial_h_values))
    
    for gen in range(generations):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection (roulette wheel)
            inv_fitness = [1.0 / (f + 1e-10) for f in fitness[:elite_size]]  # Avoid division by zero
            total_fitness = sum(inv_fitness)
            if total_fitness > 0:
                probs = [f / total_fitness for f in inv_fitness]
                selected_indices = np.random.choice(elite_size, size=2, p=probs)
            else:
                selected_indices = np.random.choice(elite_size, size=2)
            
            parent1 = population[selected_indices[0]]
            parent2 = population[selected_indices[1]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            mutation_rate = max(0.01, mutation_rate_init / np.sqrt(gen + 1))
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.1)
                    child2[i] += np.random.normal(0, 0.1)
            
            # Project back to constraints
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    # Final check on best solution
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
