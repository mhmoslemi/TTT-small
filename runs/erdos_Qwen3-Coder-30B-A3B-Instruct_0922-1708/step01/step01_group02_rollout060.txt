import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Initialize population
    population_size = 100
    population = []
    
    # Always include initial solution if available
    if 'initial_h_values' in globals():
        population.append(project_h(h))
    
    # Fill rest with random individuals
    while len(population) < population_size:
        h_rand = np.random.random(n_points)
        h_rand = project_h(h_rand)
        population.append(h_rand)
    
    # GA parameters
    crossover_rate = 0.7
    mutation_rate_init = 0.2
    generations = 0
    max_generations = 10000
    elitism_count = max(1, population_size // 10)
    
    best_h = project_h(h)
    best_c5 = compute_c5(best_h)
    
    # Track time
    while generations < max_generations and (time.time() - start_time) < budget_s - 1:
        # Evaluate population
        fitness = np.array([compute_c5(ind) for ind in population])
        
        # Update the best solution
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_h = population[min_idx].copy()
            best_c5 = fitness[min_idx]
        
        # Selection using roulette wheel
        # Invert fitness values to make lower better
        inverted_fitness = 1 / (fitness + 1e-10)
        total_fitness = np.sum(inverted_fitness)
        
        if total_fitness == 0:
            # If all fitness values are same and very small, use uniform selection
            selected_indices = np.random.choice(population_size, size=population_size, replace=True)
        else:
            probabilities = inverted_fitness / total_fitness
            selected_indices = np.random.choice(population_size, size=population_size, p=probabilities, replace=True)
            
        selected_population = [population[i] for i in selected_indices]
        
        # Create new population through crossover and mutation
        new_population = []
        
        # Elitism: retain best individuals
        elite_idx = np.argsort(fitness)[:elitism_count]
        for i in elite_idx:
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected_population, size=2, replace=False)
            
            if np.random.random() < crossover_rate:
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutation
            mutation_rate = max(0.01, mutation_rate_init / np.sqrt(generations + 1))
            
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    noise = np.random.normal(0, 0.1)
                    child1[i] += noise
                    child2[i] += noise
                    
            # Project to feasible set
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        # Keep population the same size
        population = new_population[:population_size]
        generations += 1
        
        if (time.time() - start_time) > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
