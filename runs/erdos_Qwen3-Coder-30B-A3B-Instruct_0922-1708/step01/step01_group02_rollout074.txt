import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Initialize PSO parameters
    n_particles = 50
    max_iter = 10000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    inertia_weight = w_start
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h_init = h.copy()
        else:
            h_init = np.random.rand(n_points)
        h_init = project_h(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.uniform(-1, 1, n_points))
        pbest.append(h_init.copy())
        pbest_vals.append(compute_c5(h_init))
    
    # Global best
    best_idx = np.argmin(pbest_vals)
    gbest = pbest[best_idx].copy()
    gbest_val = pbest_vals[best_idx]
    
    # PSO iteration
    iter_no_improve = 0
    for iteration in range(max_iter):
        # Check if we've run out of time
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Update inertia weight
        inertia_weight = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = project_h(particles[i])
            
            # Evaluate current particle
            current_c5 = compute_c5(particles[i])
            
            # Update particle best
            if current_c5 < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_c5
                    iter_no_improve = 0
                else:
                    iter_no_improve += 1
            
        # Early stopping
        if iter_no_improve > 1000:
            break
    
    # Return final result
    result_h = project_h(gbest)
    result_c5 = compute_c5(result_h)
    
    return (result_h.tolist(), result_c5, len(result_h))
