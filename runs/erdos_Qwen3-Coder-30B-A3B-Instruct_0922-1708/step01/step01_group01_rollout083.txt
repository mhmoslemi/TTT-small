import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO + Gradient Descent"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    swarm_size = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    stagnation_limit = 500
    min_swarm_size = 20
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_h(particle, n_points)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_vals.append(evaluate_h(particle, dx))
    
    # Global best tracking
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    
    while time.time() - start_time < 0.7 * budget_s and iteration < max_iter:
        iteration += 1
        improve = False
        
        # PSO iteration
        for i in range(len(swarm)):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - swarm[i]) + 
                            c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] = project_h(swarm[i] + velocities[i], n_points)
            
            # Evaluate
            current_val = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = current_val
                if current_val < gbest_val:
                    gbest = swarm[i].copy()
                    gbest_val = current_val
                    improve = True
        
        if not improve:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        # Adapt swarm size periodically or if stagnant
        if stagnation_count > 100 and len(swarm) > min_swarm_size:
            # Reduce swarm size by 20% randomly
            remove_count = max(1, int(len(swarm) * 0.2))
            indices_to_remove = np.random.choice(len(swarm), remove_count, replace=False)
            new_swarm = [swarm[i] for i in range(len(swarm)) if i not in indices_to_remove]
            new_velocities = [velocities[i] for i in range(len(velocities)) if i not in indices_to_remove]
            new_pbest = [pbest[i] for i in range(len(pbest)) if i not in indices_to_remove]
            new_pbest_vals = [pbest_vals[i] for i in range(len(pbest_vals)) if i not in indices_to_remove]
            
            swarm = new_swarm
            velocities = new_velocities
            pbest = new_pbest
            pbest_vals = new_pbest_vals
            stagnation_count = 0
    
    # Gradient descent refinement
    refinement_h = gbest.copy()
    refinement_val = gbest_val
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s and patience < max_patience:
        # Compute overlap for all shifts and find worst one
        corr = correlate(refinement_h, 1 - refinement_h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient using finite differences for worst overlap
        eps = 1e-5
        grad = np.zeros(n_points)
        # Only compute gradient for worst shift
        for i in range(n_points):
            # Forward difference
            h_plus = refinement_h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            # Backward difference
            h_minus = refinement_h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update using gradient
        new_h = refinement_h - learning_rate * np.sign(grad)
        new_h = project_h(new_h, n_points)
        
        # Evaluate new value
        new_val = evaluate_h(new_h, dx)
        
        if new_val < refinement_val:
            refinement_h = new_h
            refinement_val = new_val
            patience = 0
        else:
            patience += 1
            if patience % 1000 == 0:
                learning_rate *= 0.5
        
        if patience > 0 and patience % 100 == 0:
            # Occasionally recompute gradient for stability
            corr = correlate(refinement_h, 1 - refinement_h, mode="full")
            max_shift_idx = np.argmax(corr)
            worst_shift = max_shift_idx - (n_points - 1)
    
    return (refinement_h, refinement_val, n_points)
