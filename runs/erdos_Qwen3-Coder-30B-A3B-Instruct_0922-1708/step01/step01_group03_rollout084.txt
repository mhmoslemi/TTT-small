import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: smooth function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO + gradient descent."""
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 45
    start_time = time.time()
    
    # Initialization with symmetry
    h0 = initialize_h(n_points)
    # Ensure symmetry for PSO optimization
    h0 = (h0 + np.concatenate([h0[::-1], h0[-2::-1]])) / 2
    h0 = project_h(h0, n_points)
    
    # PSO parameters
    swarm_size = 30
    max_iter = 1000
    cognitive_weight = 2
    social_weight = 2
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    
    for _ in range(swarm_size):
        particle = h0 + np.random.normal(0, 0.01, n_points)
        particle = project_h(particle, n_points)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        score = evaluate_c5(particle)
        personal_best.append(particle.copy())
        personal_best_scores.append(score)
    
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iteration
    iter_no_improvement = 0
    max_no_improvement = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Update inertia weight
        inertia_weight = 0.9 - (0.5 * iteration / max_iter)
        
        for i in range(swarm_size):
            r1, r2 = np.random.random(2)
            # Update velocity
            velocities[i] = (inertia_weight * velocities[i] +
                            cognitive_weight * r1 * (personal_best[i] - particles[i]) +
                            social_weight * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_h(particles[i], n_points)
            
            # Evaluate new position
            score = evaluate_c5(particles[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                
                # Update global best
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
                    iter_no_improvement = 0
                else:
                    iter_no_improvement += 1
        
        if iter_no_improvement > max_no_improvement:
            break
    
    # Gradient descent refinement (starting from global best)
    h_refined = global_best.copy()
    best_c5 = global_best_score
    learning_rate = 0.05
    max_no_improvement = 1000
    iter_no_improvement = 0
    
    def compute_gradient(h):
        # Compute gradient based on the maximum shift contributing to C5
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(conv)
        max_shift = max_shift_idx - (n - 1)
        
        # Compute gradient at the max shift using finite difference
        grad = np.zeros(n)
        eps = 1e-6
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        return grad
    
    while time.time() - start_time < budget_s * 0.95:
        grad = compute_gradient(h_refined)
        h_new = h_refined - learning_rate * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h_refined = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            # Reduce learning rate if no improvement
            if iter_no_improvement > 100:
                learning_rate *= 0.9
                if learning_rate < 1e-6:
                    break
        
        if iter_no_improvement > max_no_improvement:
            break
    
    final_h = h_refined
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
