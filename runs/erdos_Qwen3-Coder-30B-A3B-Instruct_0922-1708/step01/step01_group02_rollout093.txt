import time

def run(seed=42, budget_s=600, **kwargs):

        h[mid-5:mid+5] = 0.75

    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set  
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Use simulated annealing with adaptive cooling
    def simulated_annealing():
        current_h = h.copy()
        current_c5 = compute_c5(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        
        temp = 0.1
        min_temp = 1e-6
        cooling_rate = 0.995
        max_iter = 10000
        
        for i in range(max_iter):
            if (time.time() - start_time) > budget_s - 1:
                break
                
            # Make a small random perturbation
            new_h = current_h.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.01)
            new_h[idx] += delta
            
            # Project
            new_h = project_h(new_h)
            new_c5 = compute_c5(new_h)
            
            # Accept or reject
            if new_c5 < current_c5 or np.random.random() < np.exp((current_c5 - new_c5) / temp):
                current_h = new_h
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            
            temp *= cooling_rate
            if temp < min_temp:
                temp = min_temp
                
        return best_h, best_c5
    
    # Try several restarts of simulated annealing
    for restart in range(5):
        if (time.time() - start_time) > budget_s - 1:

        if c5_restart < best_c5:
            best_h = h_restart
            best_c5 = c5_restart
    
    # Refine with local search on best solution
    def local_search(h_vec, max_iter=1000):
        current_h = h_vec.copy()
        current_c5 = compute_c5(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5

            idx = np.random.randint(0, n_points)

            new_h = project_h(new_h)
            new_c5 = compute_c5(new_h)
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        
        return best_h, best_c5
    
    if (time.time() - start_time) < budget_s - 1:
        best_h, best_c5 = local_search(best_h)
    
    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
