import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization with symmetric triangular pulse
    x = np.linspace(0, 2, n_points)
    # Create symmetric function
    h = np.zeros(n_points)
    # Triangle centered around 0.5
    for i in range(n_points):
        xi = x[i]
        if xi <= 0.5:
            h[i] = 2 * xi
        elif xi <= 1.5:
            h[i] = 2 - 2 * xi
        else:
            h[i] = 2 * (xi - 2)
    h = np.clip(h, 0, 1)
    h = project_h(h, n_points)
    return h

def pso_phase(h, n_points, time_budget_s):
    """Run Particle Swarm Optimization to find a good starting point."""
    start_time = time.time()
    
    # PSO Parameters
    swarm_size = 30
    max_iter = 1000
    inertia_weight = 0.9
    cognitive_weight = 2.0
    social_weight = 2.0
    velocity_max = 0.2
    
    # Initialize swarm
    particles = [h.copy() for _ in range(swarm_size)]
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_c5(p) for p in particles]
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.7 and iteration < max_iter:
        iteration += 1
        
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = random.random(), random.random()
            cognitive = cognitive_weight * r1 * (personal_best[i] - particles[i])
            social = social_weight * r2 * (global_best - particles[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update particle
            particles[i] += velocities[i]
            
            # Project to valid domain
            particles[i] = project_h(particles[i], n_points)
            
            # Evaluate fitness
            fitness = evaluate_c5(particles[i])
            
            # Update personal best
            if fitness < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = fitness
                
                # Update global best
                if fitness < global_best_score:
                    global_best = personal_best[i].copy()
                    global_best_score = fitness
        
        # Decrease inertia weight
        inertia_weight = max(0.4, inertia_weight * 0.99)
        
    return global_best, global_best_score

def gradient_descent_phase(h, n_points, time_budget_s):
    """Run projected gradient descent from PSO solution."""
    start_time = time.time()
    
    # Project to valid domain
    h = project_h(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Parameters for gradient descent
    learning_rate = 0.05
    max_iter = 10000
    patience = 1000
    
    def compute_gradient(h):
        """Compute gradient of C5 with respect to h."""
        # Find shift that produces maximum overlap
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(conv)
        # We compute gradient such that we minimize the maximum overlap
        # This is a complex problem in general, use finite difference instead.
        # But for optimization purposes, we can compute the actual gradient
        # for the case where the maximum is at a particular shift.
        # First, compute the max overlap shift
        shift = max_shift_idx - (n - 1)  # from -n+1 to n-1
        # The gradient on h[i] for a fixed shift index:
        # Each position contributes to convolution
        # dC5/dh[i] = sum_j (dC5/dh[i] for contribution at shift k) * dx
        # For the particular shift k: 
        # C5 = h[i] * (1 - h[i+k]) + h[i+1] * (1 - h[i+k+1]) + ...
        # But sum is over all terms of the convolution
        # So we can simplify:
        # For particles in h, we need to compute contributions to 1-h's elements.
        # But we're actually optimizing the max term, so 
        # We'll do a numerical approximation to see if we can gradient-descent
        # towards decreasing the largest overlap in a specific direction.
        # Centered finite difference:
        eps = 1e-6
        grad = np.zeros(n)
        # Approximate analytical gradient for one specific shift (but the max shift, which we identify)
        # The maximum of the full convolution tells us the shift of max overlap.
        # We compute gradient for that one shift.
        # To avoid inversion, we compute directional derivative using both directions
        # Even for zero testing.
        # Let's do separate finite differences
        for i in range(n):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n)
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n)
            c5_plus = evaluate_c5(h_plus)
            c5_minus = evaluate_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    iter_no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < time_budget_s * 0.3 and iteration < max_iter:
        iteration += 1
        
        # CG method, but using finite differences due to complicated correlation structure
        grad = compute_gradient(best_h)
        
        # Gradient descent step
        h_new = best_h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            
        # Reduce learning rate if no improvement for a while
        if iter_no_improvement > 100:
            learning_rate *= 0.8
            if learning_rate < 1e-6:
                break
        if iter_no_improvement > patience:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 45  # odd number for symmetric property
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: PSO
    h_pso, c5_pso = pso_phase(h, n_points, budget_s)
    
    # Phase 2: Gradient Descent
    h_gd, c5_gd = gradient_descent_phase(h_pso, n_points, budget_s)
    
    # Return best result
    final_h = h_gd if c5_gd < c5_pso else h_pso
    final_c5 = min(c5_gd, c5_pso)
    
    # Ensure final result is valid
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
