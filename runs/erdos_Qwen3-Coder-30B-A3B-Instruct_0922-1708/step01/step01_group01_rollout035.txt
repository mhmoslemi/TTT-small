import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def latin_hypercube_sampling(n_points, n_samples):
    """Generate Latin Hypercube Samples for initial population"""
    samples = []
    for _ in range(n_samples):
        h = np.random.rand(n_points)
        samples.append(h)
    return samples

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    start_time = time.time()
    
    # Phase 1: Latin Hypercube Sampling + Evaluation
    n_samples = 50
    samples = latin_hypercube_sampling(n_points, n_samples)
    samples = [project_h(s, n_points) for s in samples]
    
    # Evaluate all samples
    fitness = [evaluate_h(s, dx) for s in samples]
    sorted_indices = np.argsort(fitness)
    
    # Keep top 20 elites
    elites = [samples[i] for i in sorted_indices[:20]]
    elite_fitness = [fitness[i] for i in sorted_indices[:20]]
    
    # Phase 2: Evolution Strategy
    n_elites = 20
    n_offspring = 5
    max_generations = 1000
    
    for gen in range(max_generations):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        new_elites = []
        
        for elite in elites:
            # Create offspring by slight mutation
            for _ in range(n_offspring):
                mutated = elite + np.random.normal(0, 0.05, n_points)
                mutated = project_h(mutated, n_points)
                new_elites.append(mutated)
        
        # Evaluate offspring
        all_candidates = elites + new_elites
        fitness_all = [evaluate_h(s, dx) for s in all_candidates]
        sorted_indices = np.argsort(fitness_all)
        
        elites = [all_candidates[i] for i in sorted_indices[:n_elites]]
        elite_fitness = [fitness_all[i] for i in sorted_indices[:n_elites]]
    
    # Phase 3: Local Refinement with Gradient Descent
    best_h = elites[0].copy()
    best_c5 = elite_fitness[0]
    
    # Multi-start gradient descent from top candidates
    for i in range(min(5, len(elites))):
        h_start = elites[i].copy()
        h = h_start.copy()
        current_c5 = elite_fitness[i]
        
        # Fine tune with gradient descent
        learning_rate = 0.01
        patience = 0
        max_patience = 5000
        
        for step in range(max_patience):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Compute gradient via finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] = min(1.0, h_plus[j] + eps)
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[j] = max(0.0, h_minus[j] - eps)
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[j] = (c_plus - c_minus) / (2 * eps)
            
            # Update step
            h = h - learning_rate * grad
            h = project_h(h, n_points)
            
            new_c5 = evaluate_h(h, dx)
            
            if new_c5 < current_c5:
                current_c5 = new_c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
                if patience > 100:
                    break
            if patience > 0 and patience % 100 == 0:
                learning_rate *= 0.5
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h, best_c5, n_points)
