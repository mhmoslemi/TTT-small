import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    start_time = time.time()
    
    # Use reduced search space approach 
    # Represent h as piecewise constant with few segments
    m = 5  # Number of segments
    segment_values = [0.0, 0.5, 1.0]
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Enumerate all valid combinations of segment values
    segment_combinations = list(product(segment_values, repeat=m))
    
    # For each combination, create a piecewise constant function
    for seg_vals in segment_combinations:
        # Create piecewise constant h
        h_piecewise = np.zeros(n_points)
        segment_length = n_points // m
        remainder = n_points % m
        
        # Distribute points among segments
        actual_segment_lengths = [segment_length + (1 if i < remainder else 0) for i in range(m)]
        
        pos = 0
        for i, seg_val in enumerate(seg_vals):
            length = actual_segment_lengths[i]
            h_piecewise[pos:pos+length] = seg_val
            pos += length
        
        # Ensure it sums correctly
        h_piecewise = project_h(h_piecewise, n_points)
        
        # Evaluate
        c5 = evaluate_h(h_piecewise, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_piecewise.copy()
    
    # Refine with gradient descent on best candidates
    top_candidates = []
    
    # Get top 10 candidates from the exhaustive search
    segment_combinations = list(product(segment_values, repeat=m))
    candidate_list = []
    
    for seg_vals in segment_combinations:
        h_piecewise = np.zeros(n_points)
        segment_length = n_points // m
        remainder = n_points % m
        
        actual_segment_lengths = [segment_length + (1 if i < remainder else 0) for i in range(m)]
        
        pos = 0
        for i, seg_val in enumerate(seg_vals):
            length = actual_segment_lengths[i]
            h_piecewise[pos:pos+length] = seg_val
            pos += length
            
        h_piecewise = project_h(h_piecewise, n_points)
        c5 = evaluate_h(h_piecewise, dx)
        candidate_list.append((c5, h_piecewise))
    
    # Sort and take top 10
    candidate_list.sort(key=lambda x: x[0])
    top_candidates = [h for _, h in candidate_list[:10]]
    
    # Gradient descent refinement
    for h_init in top_candidates:
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        h_refined = h_init.copy()
        current_c5 = evaluate_h(h_refined, dx)
        best_refined = h_refined.copy()
        best_c5_refined = current_c5
        
        # Simple gradient descent with adaptive learning rate
        for iteration in range(2000):
            if time.time() - start_time > 0.9 * budget_s:
                break
                
            # Approximate gradient using finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_refined.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            lr = 0.01 * (0.9 ** (iteration // 500))
            h_refined = h_refined - lr * grad
            h_refined = project_h(h_refined, n_points)
            
            current_c5 = evaluate_h(h_refined, dx)
            if current_c5 < best_c5_refined:
                best_c5_refined = current_c5
                best_refined = h_refined.copy()
        
        if best_c5_refined < best_c5:
            best_c5 = best_c5_refined
            best_h = best_refined.copy()
    
    return (best_h, best_c5, n_points)
