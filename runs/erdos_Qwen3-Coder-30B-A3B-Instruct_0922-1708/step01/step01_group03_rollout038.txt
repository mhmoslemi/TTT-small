import numpy as np
from scipy import optimize
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric with triangular pulses
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def line_search(h, i, max_shift, dx, alpha=0.1):
    """Perform line search for optimal value of h[i] while maintaining sum constraint."""
    old_h_i = h[i]
    current_c5 = evaluate_c5(h)
    
    # Calculate how much the sum changes when we change h[i]
    # New sum = old_sum - h[i] + new_h[i]
    # To keep sum = n_points / 2, we need new_h[i] = h[i] + (n_points/2 - old_sum)
    # But we also want to retain positivity of others
    old_sum = np.sum(h)
    
    def objective(new_h_i):
        # simulate changing only h[i] and adjusting others to maintain sum
        new_h = h.copy()
        new_h[i] = new_h_i
        # Ensure that adjusting other components keeps the new_sum = n_points/2
        diff = (n_points / 2) - np.sum(new_h)
        # If diff != 0, adjust the others by proportionally reducing their values
        # For now, just change the value and project
        new_h = project_h(new_h, n_points)
        return evaluate_c5(new_h)
    
    # Use golden section search in [0, 1] to find optimal h[i]
    # But set a first estimate
    def bracket_minimum(func, a, b, c):
        if func(b) > func(c):
            a, b, c = b, c, b + 2 * (c - b)
        else:
            a, b, c = b, a, b + 2 * (a - b) 
        return a, b, c

    def golden_section_search(func, a, b, tolerance=1e-6):
        inv_phi = (math.sqrt(5) - 1) / 2
        inv_phi2 = (3 - math.sqrt(5)) / 2
        x1 = a + inv_phi2 * (b - a)
        x2 = a + inv_phi * (b - a)
        f1 = func(x1)
        f2 = func(x2)
        while abs(b - a) > tolerance:
            if f1 > f2:
                a = x1
                x1 = x2
                f1 = f2
                x2 = a + inv_phi * (b - a)
                f2 = func(x2)
            else:
                b = x2
                x2 = x1
                f2 = f1
                x1 = a + inv_phi2 * (b - a)
                f1 = func(x1)
        return (a + b) / 2
    
    # Try small value test from 0 to 1, and maybe do line search
    best_h_i = old_h_i
    best_c5 = current_c5
    
    # Test 0.0
    h[i] = 0.0
    c5_to_test = evaluate_c5(project_h(h, n_points))
    if c5_to_test < best_c5:
        best_c5 = c5_to_test
        best_h_i = 0.0
    h[i] = old_h_i
    
    # Test 1.0
    h[i] = 1.0
    c5_to_test = evaluate_c5(project_h(h, n_points))
    if c5_to_test < best_c5:
        best_c5 = c5_to_test
        best_h_i = 1.0
    h[i] = old_h_i
    
    # Golden section search between 0 and 1, if both limits geq, then fine.
    # Single variable optimal search (again within feasible region)
    try:
        h[i] = golden_section_search(objective, 0, 1)
        h[i] = project_h(h, n_points)[i]
        if evaluate_c5(h) < best_c5:
            best_c5 = evaluate_c5(h)
            best_h_i = h[i]
    except:
        pass
    
    h[i] = best_h_i
    return best_c5

def optimize_coordinates(h, n_points, num_iters):
    """Optimize each coordinate using line search."""
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    for _ in range(num_iters):
        for i in range(n_points):
            new_c5 = line_search(best_h, i, 2*n_points - 1, 2.0/n_points) 
            # Update best only if improved
        current_c5 = evaluate_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
    return best_h, best_c5

def nonlinear_conjugate_gradient(h, n_points, max_iter=500):
    """Nonlinear conjugate gradient optimization."""
    h = h.copy()
    func = lambda x: evaluate_c5(x)
    grad = np.zeros(n_points)
    alpha = 0.1
    
    # Compute initial gradient numerically (requires some perturbation)
    eps = 1e-5
    g = np.zeros(n_points)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n_points)
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_h(h_minus, n_points)
        g[i] = (func(h_plus) - func(h_minus)) / (2*eps)
    
    # Initialize CG
    p = -g
    old_g = g.copy()
    old_p = p.copy()
    best_h = h.copy()
    best_c5 = func(best_h)
    
    iter_count = 0
    for _ in range(max_iter):
        # Directional line search along p
        try:
            alpha = 0.05
            h_new = h + alpha * p
            h_new = project_h(h_new, n_points)
            c5_new = func(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            h = h_new
        except:
            pass
        iter_count += 1
        if iter_count % 100 == 0:
            # Recalculate gradient (not perfectly accurate but feasible)
            g_new = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                g_new[i] = (func(h_plus) - func(h_minus)) / (2*eps)
            # Compute beta using Polak-Ribiere
            beta = max(0, (np.dot(g_new, g_new - old_g) / np.dot(old_g, old_g)))
            p = -g_new + beta * old_p
            old_g = g_new.copy()
            old_p = p.copy()
            if evaluate_c5(h) < best_c5:
                best_c5 = evaluate_c5(h)
                best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    time_budget = budget_s
    start_time = time.time()
    
    n_points = 45
    h = initialize_h(n_points)
    
    # Phase 1: Coordinate-wise optimization
    h, c5 = optimize_coordinates(h, n_points, 500)
    
    # Phase 2: Nonlinear conjugate gradient
    h_final, c5_final = nonlinear_conjugate_gradient(h, n_points, 500)
    
    final_c5 = evaluate_c5(h_final)
    return (h_final, final_c5, n_points)
