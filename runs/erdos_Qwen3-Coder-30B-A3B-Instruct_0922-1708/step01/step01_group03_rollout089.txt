import numpy as np
from scipy import optimize
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric function
    x = np.linspace(0, 2, n_points)
    # Create a piecewise linear function centered at 1, with peaks near 0.5 and 1.5
    h = np.zeros(n_points)
    for i in range(n_points):
        xi = x[i]
        # Create two triangular pulses centered at 0.5 and 1.5
        h[i] = max(0, 1 - abs(xi - 0.5)) + max(0, 1 - abs(xi - 1.5))
    h = project_h(h, n_points)
    return h

def compute_gradient_approx(h, n_points, eps=1e-5):
    """Compute approximate gradient of C5 with respect to h."""
    current_c5 = evaluate_c5(h)
    grad = np.zeros(n_points)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_h(h_minus, n_points)
        c5_minus = evaluate_c5(h_minus)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def line_search(h, grad, n_points, max_alpha=1.0):
    """Perform line search along gradient direction."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Find alpha that minimizes the objective
    try:
        res = optimize.minimize_scalar(objective, bounds=(0, max_alpha), method='bounded')
        return res.x
    except:
        # fallback if optimization fails
        best_alpha = 0.0
        best_val = objective(0.0)
        for alpha in np.linspace(0, max_alpha, 100):
            val = objective(alpha)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        return best_alpha

def coordinate_wise_optimization(h, n_points, max_iter=5000, tol=1e-6):
    """Perform coordinate-wise optimization with line search."""
    h = project_h(h, n_points)
    best_c5 = evaluate_c5(h)
    
    for iteration in range(max_iter):
        if iteration % 100 == 0:
            current_c5 = evaluate_c5(h)
            if abs(current_c5 - best_c5) < tol:
                break
            best_c5 = current_c5
                
        for i in range(n_points):
            # Approximate gradient component at h[i]
            grad = compute_gradient_approx(h, n_points)
            # Line search to find optimal step along this component
            alpha = line_search(h, grad, n_points)
            # Update h
            h[i] -= alpha * grad[i]
            
        h = project_h(h, n_points)
        
        # Check convergence
        current_c5 = evaluate_c5(h)
        if abs(current_c5 - best_c5) < tol:
            break
        best_c5 = current_c5
        
    return h, best_c5

def nonlinear_conjugate_gradient(h, n_points, max_iter=5000, tol=1e-6):
    """Apply nonlinear conjugate gradient with Polak-Ribière formula."""
    h = project_h(h, n_points)
    best_c5 = evaluate_c5(h)
    old_grad = compute_gradient_approx(h, n_points)
    d = -old_grad
    iteration = 0
    
    while iteration < max_iter:
        if iteration % 500 == 0:
            current_c5 = evaluate_c5(h)
            if abs(current_c5 - best_c5) < tol:
                break
            best_c5 = current_c5

        # Line search along direction d
        alpha = line_search(h, d, n_points, max_alpha=0.5)
        h_new = h + alpha * d
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            h = h_new
        else:
            # Try a small step if we aren't improving
            h_new = h + 0.01 * d
            h_new = project_h(h_new, n_points)
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                h = h_new
            else:
                d = -compute_gradient_approx(h, n_points)
            alpha = 0.01
        # Compute new gradient
        new_grad = compute_gradient_approx(h, n_points)
        # Polak-Ribière formula
        beta = max(0, np.dot(new_grad - old_grad, new_grad) / np.dot(old_grad, old_grad))
        d = -new_grad + beta * d
        old_grad = new_grad
        iteration += 1
        
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()

    # Small size for faster computation
    n_points = 45
    h = initialize_h(n_points)

    # Phase 1: Coordinate-wise optimization
    h_co, c5_co = coordinate_wise_optimization(h, n_points, max_iter=2500)
    if time.time() - start_time > 0.8 * budget_s:
        return (h_co, c5_co, n_points)

    # Phase 2: Nonlinear conjugate gradient 
    h_cg, c5_cg = nonlinear_conjugate_gradient(h_co, n_points, max_iter=2500)
    
    # Return best result
    final_h = h_cg if c5_cg < c5_co else h_co
    final_c5 = min(c5_cg, c5_co)
    
    # Final projection and evaluation
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
