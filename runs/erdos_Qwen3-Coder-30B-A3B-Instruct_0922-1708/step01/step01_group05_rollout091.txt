import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Fourier basis setup
    m = 10  # Number of cosine modes (even symmetry)
    a = np.random.rand(m) * 0.5  # Coefficients for modes 1 to m-1
    a[0] = 0.5  # Ensure integral constraint
    a[1::2] = 0  # Only even modes for symmetry
    
    # Optimization loop (Gradient-based with L-BFGS)
    best_h = h.copy()
    best_C = float('inf')
    no_improvement_count = 0
    lr = 0.001
    
    def compute_h_from_a(a_coeffs):
        # Build h(x) from cosine coefficients
        x = np.linspace(0, 2, n_points)
        h = a_coeffs[0] * np.ones_like(x)
        for k in range(1, len(a_coeffs)):
            h += a_coeffs[k] * np.cos(np.pi * k * x / 2)
        return np.clip(h, 0, 1)
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def evaluate_and_update():
        nonlocal best_C, best_h
        h_vals = compute_h_from_a(a)
        h_vals = h_vals * (target_sum / np.sum(h_vals))  # Rescale
        C = compute_C(h_vals)
        if C < best_C:
            best_C = C
            best_h = h_vals.copy()
        return C
    
    # Evaluate original starting point
    previous_C = evaluate_and_update()
    iter_count = 0
    
    # Convert to combination of sinusoidal basis
    def obj_func(params):
        # Setup coefficients from flattened params
        a_new = params[:m].copy()
        a_new[0] = 0.5  # Keep integral constraint
        h_vals = compute_h_from_a(a_new)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        C = compute_C(h_vals)
        # Penalties for violating constraints or going too far from 0.5
        penalty = 0
        if np.sum(h_vals) < 0.9 * target_sum or np.sum(h_vals) > 1.1 * target_sum:
            penalty += 10000
        return C + penalty
    
    # Optimize using L-BFGS
    try:
        res = so.minimize(obj_func, a, method='L-BFGS-B', options={'maxiter': 2000})
        a = res.x.copy()
        a[0] = 0.5  # Maintain integral constraint
    except:
        # Fall back to gradient approach
        while time() - start_time < budget_s - 1:
            iter_count += 1
            h_vals = compute_h_from_a(a)
            h_vals = h_vals * (target_sum / np.sum(h_vals))  # Rescale to sum constraint

            C = compute_C(h_vals)
            # Find worst shift k
            k_max = np.argmax(np.abs(np.correlate(h_vals, 1-h_vals, mode="full")))
            # Finite diff to compute gradient approx
            eps = 1e-4
            grad = np.zeros_like(a)
            for i in range(len(a)):
                a_plus = a.copy()
                a_plus[i] += eps
                h_plus = compute_h_from_a(a_plus)
                h_plus = h_plus * (target_sum / np.sum(h_plus))
                C_plus = compute_C(h_plus)
                grad[i] = (C_plus - C) / eps
            
            # Gradient update
            a -= lr * grad
            a = np.clip(a, 0, 1)
            
            # Rescale to preserve integral condition
            if np.sum(h_vals) > 0:
                a[0] = 0.5  # keep integral constraint
                
            # Adaptive learning rate
            new_C = evaluate_and_update()
            if new_C >= previous_C:
                no_improvement_count += 1
                if no_improvement_count > 50:
                    lr *= 0.95
                    no_improvement_count = 0
            else:
                no_improvement_count = 0
            previous_C = new_C
        
    # Binary refinement step
    h_vals = compute_h_from_a(a)
    h_vals = h_vals * (target_sum / np.sum(h_vals))
    # Compute optimal threshold to meet constraint
    sorted_h = np.sort(h_vals)
    cumulative_sum = np.cumsum(sorted_h)
    threshold_idx = np.searchsorted(cumulative_sum, target_sum)
    if threshold_idx < len(h_vals):
        t = sorted_h[threshold_idx]
    else:
        t = 0.5
    # Threshold and rescale
    h_final = np.array(h_vals >= t, dtype=float)
    h_final = h_final * (target_sum / np.sum(h_final))
    
    # Further optimization by local search
    old_C = compute_C(h_final)
    best_h = h_final.copy()
    best_C = old_C
    
    for _ in range(2000):
        if time() - start_time > budget_s - 1:
            break
        i = np.random.randint(0, n_points)
        h_current = h_final.copy()
        h_current[i] = 1 - h_current[i]
        C = compute_C(h_current)
        if C < best_C:
            best_C = C
            best_h = h_current.copy()
            if best_C < old_C:
                old_C = best_C
            else:
                h_final = h_current.copy()
        else:
            # Accept with low probability
            if np.random.rand() < 0.02:
                h_final = h_current.copy()
    
    return (best_h.tolist(), best_C, n_points)
