import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sample(n_points, n_samples):
    """Generate Latin Hypercube Sampling"""
    samples = []
    for _ in range(n_samples):
        # Generate random samples in [0,1] for each dimension
        sample = np.random.rand(n_points)
        # Shuffle each dimension separately
        for i in range(n_points):
            sample[i] = (sample[i] + i) / n_points
        samples.append(sample)
    return samples

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)

    start_time = time.time()

    # Phase 1: Latin Hypercube Sampling
    n_samples = 50
    candidates = []
    initial_pop = [h]
    candidates.extend(initial_pop)
    lhs_samples = lhs_sample(n_points, n_samples - len(initial_pop))
    for sample in lhs_samples:
        sample = project_h(sample, n_points)
        candidates.append(sample)
    
    # Evaluate all candidates and find top performers
    fitness = [evaluate_h(c, dx) for c in candidates]
    top_indices = np.argsort(fitness)[:20]
    elites = [candidates[i] for i in top_indices]

    # Phase 2: Evolution Strategy
    n_offspring = 5
    elite_count = 20
    while time.time() - start_time < 0.7 * budget_s:
        offspring_pool = []
        for elite in elites:
            for _ in range(n_offspring):
                # Small random perturbation
                perturbed = elite + np.random.normal(0, 0.05, n_points)
                perturbed = project_h(perturbed, n_points)
                offspring_pool.append(perturbed)
        
        # Evaluate offspring and combine with elites
        all_candidates = elites + offspring_pool
        all_fitness = [evaluate_h(c, dx) for c in all_candidates]
        
        # Keep best elites
        best_indices = np.argsort(all_fitness)[:elite_count]
        elites = [all_candidates[i] for i in best_indices]

    # Phase 3: Multi-start Gradient Descent
    best_candidate = elites[0]
    best_c5 = evaluate_h(best_candidate, dx)
    
    # Multi-start from top elites
    for elite in elites[:5]:
        h = elite.copy()
        learning_rate = 0.01
        patience = 0
        max_patience = 1000
        prev_c5 = float('inf')
        
        while time.time() - start_time < budget_s and patience < max_patience:
            # Compute gradient via finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h -= learning_rate * grad
            h = project_h(h, n_points)
            
            new_c5 = evaluate_h(h, dx)
            if new_c5 < best_c5:
                best_candidate = h.copy()
                best_c5 = new_c5
                patience = 0
            else:
                patience += 1
                
            if patience > 0 and patience % 1000 == 0:
                learning_rate *= 0.5
                patience = 0
                
    return (best_candidate, best_c5, n_points)
