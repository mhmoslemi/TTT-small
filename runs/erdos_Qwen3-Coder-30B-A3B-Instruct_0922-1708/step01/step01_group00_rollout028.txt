import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to higher resolution
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(h)), h)
        h = h_interp
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Triangular distribution favoring lower overlap regions
        h = np.linspace(0, 1, n_points)
        h = np.minimum(h, 1-h)
        h = h / np.sum(h) * (n_points / 2)
        
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    
    # Adam moment estimates
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute exact gradient
        grad = np.zeros(n_points)
        max_lag = np.argmax(correlation)
        
        # Calculate gradient contribution from the maximum lag
        # We compute partial derivatives of correlation at max_lag
        for i in range(n_points):
            # For lag k, the contribution to correlation is h[i] * (1 - h[i+k])
            # Here we are looking at derivatives at the position where max correlation occurs
            # Approximate gradient from finite difference with small steps
            eps_val = 1e-5
            h_plus = h.copy()
            h_plus[i] += eps_val
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps_val
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        
        # Compute step update
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = h - step
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Every 100 iterations reduce learning rate
        if iteration % 100 == 0:
            lr *= 0.5

    return (best_h.tolist(), best_c5, n_points)
