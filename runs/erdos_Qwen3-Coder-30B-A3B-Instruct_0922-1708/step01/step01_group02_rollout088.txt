import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Handle initial construction
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 2000
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = population_size // 10
    
    # Initialize population
    population = []
    for _ in range(population_size):
        individual = np.random.rand(n_points)
        individual = project_h(individual)
        population.append(individual)
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        population[0] = project_h(h)
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    
    # Main GA loop
    for generation in range(generations):
        if (time.time() - start_time) > budget_s - 1:
            break
        
        # Evaluate fitness for all individuals
        fitness = []
        for individual in population:
            c5 = compute_c5(individual)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        current_best = population[0]
        current_c5 = fitness[0]
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection - roulette wheel
            fitness_rev = [1.0 / (f + 1e-10) for f in fitness]
            total_fitness = sum(fitness_rev)
            probabilities = [f / total_fitness for f in fitness_rev]
            
            parents_indices = np.random.choice(
                len(population), size=2, p=probabilities, replace=False
            )
            parent1, parent2 = population[parents_indices[0]], population[parents_indices[1]]
            
            # Crossover
            if np.random.rand() < crossover_rate and n_points > 1:
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            mutation_step = mutation_rate / np.sqrt(generation + 1)
            if mutation_step < 0.01:
                mutation_step = 0.01
            
            for i in range(n_points):
                if np.random.rand() < mutation_step:
                    child1[i] += np.random.normal(0, 0.1)
                if np.random.rand() < mutation_step:
                    child2[i] += np.random.normal(0, 0.1)
            
            # Project back to feasible set
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population[:population_size]
    
    # Final quality check
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
