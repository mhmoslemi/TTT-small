import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalization
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=5):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def perturb(h, delta=0.05):
        # Create a perturbed version
        h_new = h.copy()
        # Perturb a larger percentage of points
        n_perturb = max(1, int(n_points * 0.15))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    def project_h(h):
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Normalize to sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        return h

    def objective(h):
        return evaluate(h)
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2

    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_func}

    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    
    # Use L-BFGS-B with constraints
    while time.time() - start_time < budget_s and no_improvement_count < 200:
        iterations += 1
        
        # Try multiple gradient-based optimizations
        for _ in range(20):
            # Random initialization near current point for better exploration
            h_random = best_h + np.random.normal(0, 0.01, n_points)
            h_random = project_h(h_random)
            
            try:
                result = minimize(
                    objective,
                    h_random,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 50}
                )
                if result.success:
                    h_trial = result.x
                    h_trial = project_h(h_trial)
                    c5_trial = evaluate(h_trial)
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        best_h = h_trial.copy()
                        no_improvement_count = 0
                        break
            except:
                pass
        
        # Occasionally perturb the solution
        if iterations % 10 == 0:
            h_trial = perturb(best_h, delta=0.02)
            h_trial = smooth(h_trial, window=5)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
        
        no_improvement_count += 1

    return (best_h, best_c5, n_points)
