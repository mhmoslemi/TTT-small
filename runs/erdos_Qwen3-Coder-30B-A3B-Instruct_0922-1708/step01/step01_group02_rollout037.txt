import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx

    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec

    # Initialize population
    population_size = 100
    population = []
    for _ in range(population_size):
        h_candidate = np.random.rand(n_points) * 0.7 + 0.15  # Random values between 0.15 and 0.85
        h_candidate = project_h(h_candidate)
        population.append(h_candidate)
    
    # Add initial solution if available
    try:
        h_init = np.array(initial_h_values)
        h_init = project_h(h_init)
        population[0] = h_init
    except NameError:
        pass

    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    
    # GA parameters
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = population_size // 10
    
    generation = 0
    while (time.time() - start_time) < budget_s - 1:
        # Compute fitness for all individuals
        fitness = []
        for ind in population:
            c5 = compute_c5(ind)
            fitness.append(1.0 / (1.0 + c5))  # Inverse fitness for minimization
        
        # Sort population by fitness (descending)
        sorted_indices = np.argsort(fitness)[::-1]
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution if improved
        current_best = sorted_population[0]
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Create new population
        new_population = sorted_population[:elite_size]  # Elitism
        
        # Selection and reproduction
        while len(new_population) < population_size:
            # Roulette wheel selection
            selected_indices = []
            total_fitness = sum(sorted_fitness)
            if total_fitness > 0:
                for _ in range(2):
                    r = np.random.uniform(0, total_fitness)
                    cumulative = 0
                    for i, fit in enumerate(sorted_fitness):
                        cumulative += fit
                        if cumulative >= r:
                            selected_indices.append(i)
                            break
                parent1, parent2 = sorted_population[selected_indices[0]], sorted_population[selected_indices[1]]
            else:
                # If all fitnesses are equal, random selection
                idx1, idx2 = np.random.randint(0, population_size, 2)
                parent1, parent2 = sorted_population[idx1], sorted_population[idx2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate / (1 + generation**0.5):
                        child[i] += np.random.normal(0, 0.1)
                child = project_h(child)
                if len(new_population) < population_size:
                    new_population.append(child)
                else:
                    break
            
            if len(new_population) >= population_size:
                break
        
        population = new_population
        generation += 1
        if generation > 1000:
            break
    
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
