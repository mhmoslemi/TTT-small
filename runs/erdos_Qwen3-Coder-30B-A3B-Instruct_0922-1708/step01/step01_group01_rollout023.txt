import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Dual Optimization Phase
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Initialize dual variables
    lambda_k = np.ones(2 * n_points - 1)
    
    # Iterative refinement
    for iteration in range(1000):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Fix lambda, update h to minimize the dual objective
        # This is an approximation - we compute overlaps and adjust h
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Approximate gradient via finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)

            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)

        # Update h
        learning_rate = 0.001
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        # Evaluate and remember best
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5

        # Update dual variables (simplified update)
        lambda_k = np.maximum(0, lambda_k + 0.01 * (1 - corr))
    
    # Final local refinement with gradient descent
    h = best_h.copy()
    learning_rate = 0.005
    patience = 0
    max_patience = 1000
    
    for _ in range(5000):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient w.r.t. h using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)

            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)

        # Update
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 100 == 0:
            learning_rate *= 0.8
            
    return (best_h, best_c5, n_points)
