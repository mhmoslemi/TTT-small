import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        # Adjust to have sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    last_improvement = time.time()
    
    # Parameters for optimization
    learning_rate = 0.01
    patience = 300  # seconds without improvement before stopping
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = time.time()
        
        # Check for early termination
        if time.time() - last_improvement > patience:
            break
            
        # Gradient approximation using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update h with gradient descent
        h_new = h - learning_rate * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adjust learning rate occasionally
        if iteration % 1000 == 0 and iteration > 0:
            learning_rate *= 0.5
            
    return (best_h.tolist(), best_c5, n_points)
