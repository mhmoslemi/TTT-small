import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:

        # If defined, return a copy
        if 'initial_h_values' in globals():

    except:

    # Default initialization: triangular pulses
    x = np.linspace(0, 2, n_points)
    h = np.maximum(0, 1 - np.abs(x - 1))
    h = project_h(h, n_points)
    return h

def line_search(h, i, dx, n_points):
    """Optimize h[i] to minimize C5 for fixed h[except i]"""
    # Save the original value
    orig_h_i = h[i] 
    orig_c5 = evaluate_c5(h)
    
    # Compute the gradient approximation: dC5/dh[i]
    eps = 1e-6
    h_plus = h.copy()
    h_plus[i] += eps
    h_plus = project_h(h_plus, n_points)
    c5_plus = evaluate_c5(h_plus)
    
    h_minus = h.copy()
    h_minus[i] -= eps
    h_minus = project_h(h_minus, n_points)
    c5_minus = evaluate_c5(h_minus)
    
    grad = (c5_plus - c5_minus) / (2 * eps)
    
    # Find optimal step size along gradient direction
    # Proposed update: h[i] <- h[i] - alpha * grad
    # But we also need to ensure h[i] remains in [0,1] and sum constraint is maintained
    # We use a simple backtracking line search
    
    alpha = 0.05
    step_size = 1.0
    best_h_i = orig_h_i
    best_c5 = orig_c5
    
    # Check multiple step sizes
    for _ in range(10):
        new_h_i = orig_h_i - alpha * grad
        # Keep in [0,1] and maintain sum constraint
        new_h_i = np.clip(new_h_i, 0, 1)
        # Update h and project
        temp_h = h.copy()
        temp_h[i] = new_h_i
        temp_h = project_h(temp_h, n_points)
        new_c5 = evaluate_c5(temp_h)
        
        if new_c5 < best_c5:
            best_h_i = new_h_i
            best_c5 = new_c5
            step_size = alpha
        alpha *= 0.5
        
    return best_h_i, best_c5

def coordinate_wise_optimization(h, n_points, max_iter=5000, tolerance=1e-6):
    """Optimize h by updating each coordinate iteratively"""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    prev_c5 = best_c5 + tolerance + 1.0
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time < 500.0) and abs(best_c5 - prev_c5) > tolerance:
        prev_c5 = best_c5
        for i in range(n_points):
            if time.time() - start_time > 500.0:
                break
            # Perform coordinate-wise optimization
            new_h_i, new_c5 = line_search(best_h, i, 2.0/n_points, n_points)
            if new_c5 < best_c5:
                best_h[i] = new_h_i
                best_c5 = new_c5
                
        iteration += 1
        
        if iteration % 100 == 0:
            # Re-project every 100 steps to maintain feasibility
            best_h = project_h(best_h, n_points)
            best_c5 = evaluate_c5(best_h)
            
    return best_h, best_c5

def nonlinear_cg_optimization(h, n_points, max_iter=5000):

            # Estimate gradient using finite differences
            h_plus = x.copy()

            h_minus = x.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Use Polak-Ribière formula for conjugate direction
        if iter_count == 0:
            p = -grad
        else:
            beta = max(0, np.dot(grad, grad - grad_old) / delta_old)
            p = -grad + beta * p
            
        # Line search to find best step size
        alpha_max = 1.0

        alpha = step_size
        
        while alpha > 1e-8:
            trial_x = x + alpha * p

            trial_c5 = evaluate_c5(trial_x)
            if trial_c5 < current_c5:
                x = trial_x

            alpha *= 0.5

            best_h = x.copy()
            best_c5 = current_c5
            
        grad_old = grad.copy()
        delta_old = np.dot(grad, grad)

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)

    h_opt2, c5_opt2 = nonlinear_cg_optimization(h_opt1, n_points, max_iter=5000)
    
    # Return best result
    final_h = h_opt2 if c5_opt2 < c5_opt1 else h_opt1

    # Ensure final result is valid
    final_h = project_h(final_h, n_points)
