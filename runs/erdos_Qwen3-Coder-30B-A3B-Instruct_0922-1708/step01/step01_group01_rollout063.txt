import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sample(n_points, n_samples):
    """Generate Latin Hypercube Samples"""
    samples = np.zeros((n_samples, n_points))
    for i in range(n_points):
        # Generate uniform samples for each dimension
        u = np.random.uniform(i / n_points, (i + 1) / n_points, n_samples)
        # Shuffle them
        np.random.shuffle(u)
        samples[:, i] = u
    return samples

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Phase 1: LHS sampling for global exploration
    n_samples = 50
    n_elites = 20
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Generate LHS samples
    lhs_samples = lhs_sample(n_points, n_samples)
    candidate_h = []
    for i in range(n_samples):
        h_tmp = lhs_samples[i] * 2  # Scale to [0,2]
        h_tmp = project_h(h_tmp, n_points)
        candidate_h.append(h_tmp)
    
    # Add original if available (already projected)
    candidate_h.append(h)
    candidate_h = candidate_h[:n_samples+1]  # Ensure it's not more than we intended
    
    # Evaluate LHS candidates
    fitness_scores = []
    for h_cand in candidate_h:
        c5 = evaluate_h(h_cand, dx)
        fitness_scores.append(c5)
    
    # Sort and keep top elites
    sorted_indices = np.argsort(fitness_scores)
    elites = [candidate_h[i] for i in sorted_indices[:n_elites]]
    elite_scores = [fitness_scores[i] for i in sorted_indices[:n_elites]]
    
    # Update best found so far
    min_idx = np.argmin(elite_scores)
    if elite_scores[min_idx] < best_c5:
        best_h = elites[min_idx].copy()
        best_c5 = elite_scores[min_idx]
    
    # Phase 2: Evolution strategy
    n_offspring = 5
    max_generations = 200
    gen = 0
    patience = 0
    max_patience = 50
    
    while time.time() - start_time < 0.7 * budget_s and gen < max_generations:
        gen += 1
        offspring = []
        for elite in elites:
            for _ in range(n_offspring):
                # Mutation via small random perturbation
                mutation = np.random.normal(0, 0.05, n_points)
                child = elite + mutation
                child = project_h(child, n_points)
                offspring.append(child)
        
        # Evaluate offspring
        offspring_fitness = [evaluate_h(h_off, dx) for h_off in offspring]
        
        # Combine elites and offspring
        all_candidates = elites + offspring
        all_fitness = elite_scores + offspring_fitness
        combined_sorted_indices = np.argsort(all_fitness)
        
        # Keep top elites
        elites = [all_candidates[i] for i in combined_sorted_indices[:n_elites]]
        elite_scores = [all_fitness[i] for i in combined_sorted_indices[:n_elites]]
        
        # Update best if needed
        min_idx = np.argmin(elite_scores)
        if elite_scores[min_idx] < best_c5:
            best_h = elites[min_idx].copy()
            best_c5 = elite_scores[min_idx]
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

    # Phase 3: Gradient descent refinement
    h_refine = best_h.copy()
    learning_rate = 0.01
    max_iter = 2000
    patience_refine = 0
    max_patience_refine = 500
    
    for _ in range(max_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break

        # Compute gradients via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(h_refine, dx)
        
        for i in range(n_points):
            h_plus = h_refine.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_refine.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h_refine -= learning_rate * grad
        h_refine = project_h(h_refine, n_points)
        
        new_c5 = evaluate_h(h_refine, dx)
        if new_c5 < best_c5:
            best_h = h_refine.copy()
            best_c5 = new_c5
            patience_refine = 0
        else:
            patience_refine += 1
            if patience_refine > max_patience_refine:
                break
        if patience_refine > 0 and patience_refine % 500 == 0:
            learning_rate *= 0.5

    return (best_h, best_c5, n_points)
