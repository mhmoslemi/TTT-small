import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create flat
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Optimization loop (Gradient-based with careful handling)
    best_h = h.copy()
    best_C = float('inf')
    no_improvement_count = 0
    lr = 0.01
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def evaluate_and_update():
        nonlocal best_C, best_h
        C = compute_C(h_vals)
        if C < best_C:
            best_C = C
            best_h = h_vals.copy()
        return C
    
    # Evaluate starting point
    previous_C = evaluate_and_update()
    
    iter_count = 0
    while time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute gradient with finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        current_C = compute_C(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - current_C) / eps
        
        # Gradient update
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))  # Rescale to sum constraint
        
        # Adaptive learning rate
        new_C = evaluate_and_update()
        if new_C >= previous_C:
            no_improvement_count += 1
            if no_improvement_count > 50:
                lr *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
        previous_C = new_C
        
        # Occasionally restart with slightly perturbed solution
        if iter_count % 1000 == 0:
            h = best_h + np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
    
    return (best_h.tolist(), best_C, n_points)
