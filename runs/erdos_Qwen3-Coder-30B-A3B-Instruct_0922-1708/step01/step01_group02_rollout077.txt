import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Project to constraint sum = target_sum
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx

    h = project_h(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Simulated Annealing parameters
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    max_iter = int(budget_s * 1000)
    
    for i in range(max_iter):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Generate candidate by perturbing current solution
        h_candidate = h + np.random.normal(0, temp * 0.1, n_points)
        h_candidate = project_h(h_candidate)
        
        # Evaluate candidate
        c5_candidate = compute_c5(h_candidate)
        c5_current = compute_c5(h)
        
        # Accept or reject based on Metropolis criterion
        if c5_candidate < c5_current or np.random.random() < np.exp(-(c5_candidate - c5_current) / temp):
            h = h_candidate
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate
                
        # Cool the temperature
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
    
    # Final check
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
