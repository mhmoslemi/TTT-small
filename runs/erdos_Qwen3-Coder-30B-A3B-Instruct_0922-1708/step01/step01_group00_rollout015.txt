import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    max_iter = 5000
    mutation_rate = 0.1
    elite_size = pop_size // 5
    best_h = None
    best_c5 = float('inf')
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_init = np.clip(h_init, 0, 1)
        population.append(h_init)
    
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
            last_improvement = iteration
        
        # Early termination if no improvement
        if iteration - last_improvement > 300:
            break
        
        # Selection
        elite = sorted_population[:elite_size]
        
        # Create new population
        new_population = elite.copy()
        
        # Crossover
        while len(new_population) < pop_size:
            # Select two parents randomly
            idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
            h1, h2 = elite[idx1], elite[idx2]
            
            # Arithmetic crossover
            alpha = np.random.random()
            child = alpha * h1 + (1 - alpha) * h2
            new_population.append(child)
        
        # Mutation
        for i in range(elite_size, len(new_population)):
            h = new_population[i]
            mask = np.random.random(n_points) < mutation_rate
            noise = np.random.normal(0, 0.05, n_points)
            h = h + noise * mask
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            new_population[i] = h
        
        population = new_population
        
    return (best_h.tolist(), best_c5, n_points)
