        segment_size = n_points // 5
        for i in range(5):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size if i < 4 else n_points
            h[start_idx:end_idx] = 0.5 if i % 2 == 0 else 0.0
        # Adjust to meet sum constraint
        h = h * (target_sum / np.sum(h))
    
    # Objective function to minimize
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # Gradient-based optimization helper
    def finite_diff_grad(h_vals, eps=1e-4):
        grad = np.zeros_like(h_vals)
        base_c = compute_C(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            c_plus = compute_C(h_plus)
            grad[i] = (c_plus - base_c) / eps
        return grad
    
    # Simulated Annealing with local refinement
    best_h = h.copy()
    best_C = compute_C(h)
    T = 1.0
    alpha = 0.999
    max_iter = int(50000)
    iter_count = 0
    
    while time() - start_time < budget_s - 2:
        iter_count += 1
        
        # Generate neighbor by modifying a few segments
        h_new = h.copy()
        num_modifications = np.random.randint(1, 5)
        for _ in range(num_modifications):
            i = np.random.randint(0, n_points)
            # Random perturbation
            h_new[i] += np.random.normal(0, 0.03)
            h_new[i] = np.clip(h_new[i], 0, 1)

        h_new = h_new * (target_sum / np.sum(h_new))

        c_new = compute_C(h_new)
        c_old = compute_C(h)

        if c_new < c_old or np.random.rand() < np.exp(-(c_new - c_old) / T):
            h = h_new
            if c_new < best_C:
                best_C = c_new
                best_h = h_new.copy()
        
        T *= alpha

        # Adaptive learning rate
