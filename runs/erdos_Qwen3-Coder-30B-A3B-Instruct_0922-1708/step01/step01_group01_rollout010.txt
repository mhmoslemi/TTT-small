import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sampling(n_points, n_samples):
    """Generate Latin Hypercube samples"""
    samples = []
    for _ in range(n_samples):
        # Generate uniform random samples
        u = np.random.rand(n_points)
        # Generate latin hypercube samples
        samples.append(np.sort(u))
    return samples

    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)

    n_samples = 50
    candidates = []
    
    # Create initial samples
    lhs_samples = lhs_sampling(n_points, n_samples)
    
    for i in range(n_samples):
        # Use different initialization strategies
        if i == 0:
            # Original initialization
            h_candidate = h.copy()
        elif i == 1:
            # Uniform random
            h_candidate = np.random.rand(n_points)
        elif i == 2:
            # Sinusoidal
            x = np.linspace(0, 2, n_points)
            h_candidate = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        else:
            # LHS
            h_candidate = lhs_samples[i]
        
        h_candidate = project_h(h_candidate, n_points)
        candidates.append(h_candidate)

    candidate_scores = []
    for i, h_candidate in enumerate(candidates):
        score = evaluate_h(h_candidate, dx)
        candidate_scores.append((score, i))

    candidate_scores.sort()
    elites = [candidates[i] for _, i in candidate_scores[:20]]

        for elite in elites:

                # Perturb the elite
                h_offspring = elite + np.random.normal(0, 0.05, n_points)
                h_offspring = project_h(h_offspring, n_points)
                offspring.append(h_offspring)
        
        # Combine elites and offspring
        all_candidates = elites + offspring
        all_scores = []
        
        for i, h_candidate in enumerate(all_candidates):
            score = evaluate_h(h_candidate, dx)
            all_scores.append((score, i))
        
        # Keep top n_elites
