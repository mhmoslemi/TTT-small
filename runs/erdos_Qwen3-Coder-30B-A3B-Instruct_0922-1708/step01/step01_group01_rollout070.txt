import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Dual Optimization Phase
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Initialize dual variables
    dual_vars = np.ones(2 * n_points - 1)
    
    # Iteration parameters
    max_iter = 1000
    dual_step = 0.1
    adaptive_lr = 0.01
    patience = 0
    max_patience = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Solve primal problem with current dual variables (approximate)
        # In each iteration, perform gradient ascent on dual variables
        # and update h accordingly
        
        # Compute overlaps using current h
        corr_h = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr_h)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Perturb h to increase overlap in worst shift direction
        # Create gradient to reduce worst overlap
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Finite difference gradient estimation
            eps = 1e-4
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            
            c_plus = evaluate_h(h_plus, dx)
            c_minus = evaluate_h(h_minus, dx)
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update h
        h = h - adaptive_lr * grad
        h = project_h(h, n_points)
        
        # Update dual variables
        current_overlap = correlate(h, 1 - h, mode="full")
        dual_vars += dual_step * (1 - current_overlap) 
        dual_vars = np.maximum(dual_vars, 0)
        
        # Evaluate and update best
        current_c5 = evaluate_h(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
            
        if patience % 100 == 0:
            adaptive_lr *= 0.95
            if adaptive_lr < 1e-6:
                adaptive_lr = 1e-6
    
    # Gradient Descent Refinement on best solution
    refinement_time = time.time() - start_time
    remaining_time = 0.1 * budget_s
    
    h_refined = best_h.copy()
    lr = 0.01
    patience = 0
    max_patience = 500
    
    while time.time() - start_time < budget_s - 1 and patience < max_patience:
        # Compute gradient
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            
            h_minus = h_refined.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            
            c_plus = evaluate_h(h_plus, dx)
            c_minus = evaluate_h(h_minus, dx)
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h_refined = h_refined - lr * grad
        h_refined = project_h(h_refined, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h_refined, dx)
        if new_c5 < best_c5:
            best_h = h_refined.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience % 100 == 0:
                lr *= 0.5
                
        if patience > 0 and patience % 50 == 0:
            lr *= 0.9
        
    return (best_h, best_c5, n_points)
