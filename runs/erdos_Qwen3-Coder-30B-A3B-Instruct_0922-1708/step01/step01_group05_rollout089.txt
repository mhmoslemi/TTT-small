import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create from alternating segments
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create alternating segments of 1 and 0
        h = np.zeros(n_points)
        segments = 6
        seg_size = n_points // segments
        for i in range(segments):
            start = i * seg_size
            end = start + seg_size if i < segments - 1 else n_points
            h[start:end] = 1.0 if i % 2 == 0 else 0.0
        # Scale to target sum
        h = h * (target_sum / np.sum(h))
    
    # Simulated Annealing Parameters
    T = 1.0
    alpha = 0.99
    max_iter = int(50000)
    min_T = 1e-6
    
    # Gradient Descent Parameters
    gd_iter = 200
    lr = 0.01
    
    # Helper functions
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def project_h(h_vals):
        # Clip to [0, 1] and scale to correct sum
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals
    
    def neighbor(h_vals):
        # Slightly perturb a random segment
        h_new = h_vals.copy()
        seg_size = max(2, n_points // 10)
        seg_start = np.random.randint(0, n_points - seg_size)
        seg_end = seg_start + seg_size
        perturbation = np.random.normal(0, 0.05, seg_size)
        h_new[seg_start:seg_end] += perturbation
        return project_h(h_new)
    
    # Initialize best solution
    best_h = h.copy()
    best_C = compute_C(best_h)
    
    # Simulated Annealing
    h_current = h.copy()
    current_C = compute_C(h_current)
    
    iter_count = 0
    while iter_count < max_iter and time() - start_time < budget_s - 1:
        # Generate neighbor
        h_new = neighbor(h_current)
        new_C = compute_C(h_new)
        
        # Accept or reject
        delta = new_C - current_C
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new
            current_C = new_C
            
        # Update best solution
        if current_C < best_C:
            best_h = h_current.copy()
            best_C = current_C
            
        # Update temperature
        T = max(min_T, T * alpha)
        iter_count += 1
        
    # Gradient Descent refinement
    h_refined = best_h.copy()
    patience = 0
    patience_limit = 50
    
    for i in range(gd_iter):
        if time() - start_time >= budget_s - 1:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h_refined)
        eps = 1e-5
        base_C = compute_C(h_refined)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            C_plus = compute_C(project_h(h_plus))
            grad[j] = (C_plus - base_C) / eps
            
        # Update step
        h_refined = h_refined - lr * grad
        h_refined = project_h(h_refined)
        
        # Check improvement
        new_C = compute_C(h_refined)
        if new_C < best_C:
            best_C = new_C
            best_h = h_refined.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                lr *= 0.5
                patience = 0
                
    return (best_h.tolist(), best_C, n_points)
