import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    inertia = 0.7
    swarm_size = n_particles
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    gbest_idx = 0
    gbest_value = float('inf')
    
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            particle = h + np.random.normal(0, 0.05, n_points)
            if np.random.rand() < 0.1:  # 10% random particles
                particle = np.random.rand(n_points)
        particle = project_h(particle, n_points)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        value = evaluate_h(particle, dx)
        pbest.append(particle.copy())
        pbest_values.append(value)
        if value < gbest_value:
            gbest_value = value
            gbest_idx = i
    
    # PSO loop
    start_time = time.time()
    iter_count = 0
    stagnation_count = 0
    best_c5 = gbest_value
    best_h = particles[gbest_idx].copy()
    
    while time.time() - start_time < 0.7 * budget_s and iter_count < max_iter:
        iter_count += 1
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (particles[gbest_idx] - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_h(particles[i], n_points)
            
            # Evaluate
            value = evaluate_h(particles[i], dx)
            
            # Update personal best
            if value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = value
                if value < gbest_value:
                    gbest_value = value
                    gbest_idx = i
                    stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Adapt swarm size based on stagnation
        if stagnation_count > 500 and swarm_size > 20:
            swarm_size = max(20, int(swarm_size * 0.8))
            particles = particles[:swarm_size]
            velocities = velocities[:swarm_size]
            pbest = pbest[:swarm_size]
            pbest_values = pbest_values[:swarm_size]
            stagnation_count = 0
        
        # Update best solution found
        if gbest_value < best_c5:
            best_c5 = gbest_value
            best_h = particles[gbest_idx].copy()
    
    # Gradient descent refinement
    h_refined = best_h.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 1000
    
    # Generate a sequence for gradient descent
    while time.time() - start_time < 0.9 * budget_s:
        if patience > max_patience:
            break
            
        # Find worst overlap
        corr = correlate(h_refined, 1 - h_refined, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute finite difference gradient
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_refined.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update using gradient
        h_refined = h_refined - learning_rate * grad
        h_refined = project_h(h_refined, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h_refined, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_refined.copy()
            patience = 0
        else:
            patience += 1
            if patience % 1000 == 0:
                learning_rate *= 0.5
                patience = 0

    return (best_h, best_c5, n_points)
