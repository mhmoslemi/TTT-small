import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def make_symmetric(h):
    """Make h symmetric around center."""
    n = len(h)
    if n % 2 == 0:
        # Even length: use symmetry condition h[i] = h[n-1-i]
        for i in range(n // 2):
            h[i] = h[n - 1 - i] = (h[i] + h[n - 1 - i]) / 2
    else:
        # Odd length: center point stays the same
        center = n // 2
        for i in range(center):
            h[i] = h[n - 1 - i] = (h[i] + h[n - 1 - i]) / 2
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # If initial_h_values is accessible in this scope, use it
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values).copy()
            h = project_h(h, n_points)
            return make_symmetric(h)
    except:
        pass
    
    # Default initialization: symmetric with two bumps
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * (x - 0.5))) + 0.5 * (1 + np.sin(np.pi * (x - 1.5)))
    h = project_h(h, n_points)
    h = make_symmetric(h)
    return h

def compute_exact_gradient(h, shift):
    """Compute exact gradient for a given shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        j = i + shift
        if 0 <= j < n:
            grad[i] = -dx
        elif j >= n:
            j = j - n
            if 0 <= j < n:
                grad[i] = -dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using symmetric gradient descent."""
    np.random.seed(seed)
    n_points = 45  # Odd to allow for symmetric structure
    
    # Initialize symmetric h
    h = initialize_h(n_points)
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.1
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s * 0.95:
        # Find shift that gives maximum overlap
        conv = np.correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute exact gradient
        grad = compute_exact_gradient(best_h, shift)
        
        # Update h
        h_new = best_h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        h_new = make_symmetric(h_new)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
            if learning_rate > 1e-6:
                learning_rate *= 0.9
        else:
            no_improvement_count += 1
            learning_rate *= 2.0
            if learning_rate > 1.0:
                learning_rate = 1.0
                
        if no_improvement_count > max_no_improvement:
            break
    
    # Final evaluation
    final_h = project_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
