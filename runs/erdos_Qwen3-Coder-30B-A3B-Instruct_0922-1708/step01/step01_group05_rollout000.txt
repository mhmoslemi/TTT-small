import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    max_iter = 500
    w = 0.7  # inertia
    c1 = 1.5  # cognitive
    c2 = 1.5  # social
    v_max = 0.1
    
    # Particle positions and velocities (as n_points values)
    particles = np.random.rand(n_particles, n_points)
    particles = np.clip(particles, 0, 1)
    velocities = np.random.rand(n_particles, n_points) * 0.1 - 0.05
    
    # Evaluate initial particles
    p_best = particles.copy()
    p_best_vals = np.full(n_particles, float('inf'))
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    for i in range(n_particles):
        if np.sum(particles[i]) > 0:
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
        C = compute_C(particles[i])
        p_best_vals[i] = C
    
    # Global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    # Adaptive parameters for gradient descent
    lr = 0.01
    lr_decay = 0.99
    patience = 0
    patience_limit = 50
    
    # PSO iterations
    for iter in range(max_iter):
        if time() - start_time > budget_s - 5:
            break
            
        for p in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[p] = (w * velocities[p] + 
                             c1 * r1 * (p_best[p] - particles[p]) + 
                             c2 * r2 * (g_best - particles[p]))
            velocities[p] = np.clip(velocities[p], -v_max, v_max)
            
            # Update position
            particles[p] += velocities[p]
            particles[p] = np.clip(particles[p], 0, 1)
            
            # Rescale to ensure integral constraint
            if np.sum(particles[p]) > 0:
                particles[p] = particles[p] * (target_sum / np.sum(particles[p]))
            
            # Evaluate
            C = compute_C(particles[p])
            if C < p_best_vals[p]:
                p_best[p] = particles[p].copy()
                p_best_vals[p] = C
                if C < g_best_val:
                    g_best = particles[p].copy()
                    g_best_val = C
        
        # Early stopping based on improvement
        if iter > 100 and np.allclose(g_best_val, p_best_vals, rtol=1e-4):
            break
    
    # Gradient descent refinement on the best solution
    h = g_best.copy()
    best_C = g_best_val
    
    # Define optimization variable
    def objective_and_grad(h_vars):
        h_vals = np.array(h_vars)
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale
        if np.sum(h_vals) > 0:
            h_vals = h_vals * (target_sum / np.sum(h_vals))
        C = compute_C(h_vals)
        # Gradient: use finite difference
        eps = 1e-5
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            if np.sum(h_plus) > 0:
                h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C) / eps
        return C, grad
    
    # Bound constraints for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS with bounded constraints for refinement
    try:
        result = so.minimize(
            fun=lambda h_vals: objective_and_grad(h_vals)[0],
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            jac=lambda h_vals: objective_and_grad(h_vals)[1],
            options={'ftol': 1e-8, 'gtol': 1e-6}
        )
        if result.success:
            h_refined = np.clip(result.x, 0, 1)
            if np.sum(h_refined) > 0:
                h_refined = h_refined * (target_sum / np.sum(h_refined))
            h_C = compute_C(h_refined)
            if h_C < best_C:
                best_C = h_C
                h = h_refined
    except:
        pass
    
    return (h.tolist(), best_C, n_points)
