import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    stagnation_limit = 500
    min_swarm_size = 20
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Create initial particles
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            # Perturb initial solution
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_h(particle, n_points)
        
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_vals.append(evaluate_h(particle, dx))
    
    # Track global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    start_time = time.time()
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < 0.7 * budget_s and iter_count < max_iter:
        iter_count += 1
        improved = False
        
        # Update swarm size if stagnated
        if iter_count - last_improvement > stagnation_limit:
            n_particles = max(min_swarm_size, int(n_particles * 0.8))
            # Reinitialize some particles
            for i in range(n_particles, len(swarm)):
                swarm[i] = h + np.random.normal(0, 0.05, n_points)
                swarm[i] = project_h(swarm[i], n_points)
                velocities[i] = np.zeros(n_points)
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = evaluate_h(swarm[i], dx)
            last_improvement = iter_count
        
        # PSO update
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (pbest[i] - swarm[i])
                           + c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] = project_h(swarm[i] + velocities[i], n_points)
            
            # Evaluate
            current_val = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = current_val
                improved = True
                if current_val < gbest_val:
                    gbest = swarm[i].copy()
                    gbest_val = current_val
                    last_improvement = iter_count
        
        # Adjust parameters based on progress
        if iter_count % 100 == 0:
            w = max(0.4, w * 0.99)
            c1 = c2 = max(1.0, c1 * 0.99)
    
    # Gradient descent phase
    h = gbest.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < 0.95 * budget_s:
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient w.r.t. h
        # Approximate gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0

    return (best_h, best_c5, n_points)
