import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Discretization parameters
    dx = 2.0 / n_points
    max_iterations = 10000

    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def smooth_h(h, sigma=1.0):
        # Apply Gaussian smoothing for continuity
        if len(h) < 3:
            return h
        smoothed = np.convolve(h, np.exp(-np.linspace(-2, 2, 11)**2 / (2*sigma**2)), mode='same')
        return smoothed

    def segments_to_h(c, n_points):
        # c is the segment averages
        m = len(c)
        # Segment width
        segment_width = n_points / m
        # Extend c to match n_points
        h = np.repeat(c, int(np.ceil(segment_width)))
        # Trim to n_points
        h = h[:n_points]
        # Pad with last value if needed
        if len(h) < n_points:
            h = np.pad(h, (0, n_points - len(h)), constant_values=h[-1])
        return h

    def optimize_segments(m, h_initial):
        # Segmented approach for coarse optimization
        c_initial = np.linspace(0.1, 0.9, m)
        # Normalize so sum equals m/2
        c_initial = c_initial * (m / 2) / np.sum(c_initial)
        c_initial = np.clip(c_initial, 0, 1)
        
        def objective(c):
            # Reconstruct h from segments
            h = segments_to_h(c, n_points)
            return evaluate(h)

        def constraint_sum(c):
            return np.sum(c) - m / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(m)]
        
        result = minimize(objective, c_initial, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 200})
        
        if result.success:
            h_final = segments_to_h(result.x, n_points)
            h_final = smooth_h(h_final, sigma=1.0)
            return h_final
        return h_initial

    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    current_best_h = best_h.copy()
    
    iteration_count = 0
    no_improvement_count = 0

    # Hierarchical optimization: start from coarse, refine gradually
    segment_sizes = [7, 14, 21, 28, 35, 50, 60, 70]

    for m in segment_sizes:
        if time.time() - start_time > budget_s:
            break
        if m > n_points:
            continue
            
        # Reinitialize based on current best
        h_temp = optimize_segments(m, current_best_h)
        h_temp = np.clip(h_temp, 0, 1)
        # Normalize to integral = 1
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        h_temp = smooth_h(h_temp, sigma=1.0)
        c5_temp = evaluate(h_temp)

        if c5_temp < best_c5:
            best_c5 = c5_temp
            best_h = h_temp.copy()
            current_best_h = h_temp.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        iteration_count += 1
        if no_improvement_count >= 100:
            break

    # Final fine-tune with full resolution
    if time.time() - start_time < budget_s:
        # Some final random local search
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            # Small random perturbations
            h_trial = best_h + np.random.normal(0, 0.005, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            h_trial = smooth_h(h_trial, sigma=0.5)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()

    return (best_h, best_c5, n_points)
