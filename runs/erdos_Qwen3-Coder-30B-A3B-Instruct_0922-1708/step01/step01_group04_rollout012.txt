import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def get_segmented_h(c_values, n_points, n_segments):
        # Reconstruct h from segment values
        segment_length = n_points // n_segments
        remainder = n_points % n_segments
        h = np.zeros(n_points)
        start_idx = 0
        for i in range(n_segments):
            length = segment_length + (1 if i < remainder else 0)
            h[start_idx:start_idx+length] = c_values[i]
            start_idx += length
        return h
    
    def segment_optimization(c_values, n_points, n_segments):
        # Optimize the c_values to minimize C5
        def objective(c):
            h = get_segmented_h(c, n_points, n_segments)
            return evaluate(h)
        
        # Constraints
        def constraint_sum(c):
            return np.sum(c) - n_segments / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Bounds (0, 1) for each c_i
        bounds = [(0, 1) for _ in range(n_segments)]
        
        # Optimize
        result = minimize(objective, c_values, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-8})
        if result.success:
            c_opt = result.x
        else:
            c_opt = c_values
        
        return c_opt
    
    # Start with a small number of segments for coarse optimization
    n_segments = 7
    
    # Initialize segments
    segment_values = np.ones(n_segments) * 0.5
    
    best_c5 = float('inf')
    best_h = None
    best_segment_values = None
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < 50:
        iteration += 1
        
        # Optimize segment values
        segment_values = segment_optimization(segment_values, n_points, n_segments)
        
        # Reconstruct h from segments
        h = get_segmented_h(segment_values, n_points, n_segments)
        
        # Apply smoothing
        if n_segments > 3:
            h = np.convolve(h, np.ones(3)/3, mode='same')
        
        # Ensure bounds
        h = np.clip(h, 0, 1)
        
        # Re-normalize to match integral
        h = h * (n_points / 2) / np.sum(h)
        
        c5 = evaluate(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_segment_values = segment_values.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Occasionally perturb values
        if iteration % 25 == 0:
            n_perturb = max(1, n_segments // 10)
            indices = np.random.choice(n_segments, n_perturb, replace=False)
            for i in indices:
                delta = np.random.uniform(-0.1, 0.1)
                segment_values[i] = np.clip(segment_values[i] + delta, 0, 1)
            # Re-normalize segment sum
            current_sum = np.sum(segment_values)
            segment_values = segment_values * (n_segments / 2) / current_sum
        
        # Increase resolution if no improvement for a while
        if no_improvement_count > 20 and n_segments < 21:
            n_segments += 2
            # Interpolate old segment values
            old_segments = best_segment_values
            new_segment_values = np.interp(np.linspace(0, n_segments-1, n_segments),
                                           np.linspace(0, n_segments-3, n_segments-2),
                                           old_segments[:-1])
            segment_values = new_segment_values
            
            # Make sure new values are within bounds
            segment_values = np.clip(segment_values, 0, 1)
            # Renormalize
            current_sum = np.sum(segment_values)
            segment_values = segment_values * (n_segments / 2) / current_sum
            no_improvement_count = 0
    
    # Ensure we return a final version
    if best_h is None:
        best_h = h.copy()
    
    return (best_h, best_c5, n_points)
