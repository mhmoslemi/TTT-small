    h = h * (target_sum / np.sum(h))
    
    # Rescale to target sum
    h *= target_sum / np.sum(h)

        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def evaluate_and_update():
        nonlocal best_C, best_h
        C = compute_C(h)
        if C < best_C:
            best_C = C
            best_h = h.copy()
        return C
    
    # Track best solution
    best_h = h.copy()
    best_C = float('inf')
    
    # Initialize

    # Try a few different optimization strategies
    iter_count = 0
    max_iter = 10000
    lr = 0.02
    patience = 0
    patience_limit = 500
    
    # Store history for adaptive learning rate
    past_C = []
    window_size = 20
    
    while time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        C_prev = compute_C(h)
        
        # Compute gradient via finite differences for each component
        grad = np.zeros_like(h)

            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C_prev) / eps

        h -= lr * grad

        evaluate_and_update()

                    patience += 1
                else:
