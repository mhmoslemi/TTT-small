import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to the desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    # Define the objective function for optimization
    def objective(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Define the constraint that the integral equals 1
    def constraint(h):
        return np.sum(h) * dx - 1.0
    
    # Create bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5 value
        c5 = objective(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Try to optimize using scipy minimize
        try:
            # Use L-BFGS-B and add bounds and constraints
            result = minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 50}
            )
            
            if result.success:
                h_new = result.x
                # Adjust to satisfy integral constraint
                h_new = h_new / np.sum(h_new) * (n_points / 2)
                h = h_new
            else:
                # Fallback: gradient descent step
                grad = np.zeros_like(h)
                eps = 1e-5
                
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    c5_plus = objective(h_plus)
                    grad[i] = (c5_plus - c5) / eps
                
                lr = 0.005
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
        except:
            # Fallback to simple gradient step
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c5_plus = objective(h_plus)
                grad[i] = (c5_plus - c5) / eps
            
            lr = 0.003
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
        # Periodically check if we need to terminate
        if iteration - last_improvement > 500 and iteration > 2000:
            # Early termination if no improvement
            break
            
    return (best_h.tolist(), best_c5, n_points)
