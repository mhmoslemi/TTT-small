import numpy as np
from scipy import optimize
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals() and initial_h_values is not None:
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def coordinate_wise_optimization(h, n_points, max_iter=1000):
    """Optimize h by cyclically updating each coordinate."""
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    iter_no_improvement = 0
    max_no_improvement = 500
    
    # Store the full convolution for efficient reuse
    old_conv = np.correlate(best_h, 1 - best_h, mode="full")
    old_max_idx = np.argmax(old_conv)
    
    for iteration in range(max_iter):
        improved = False
        # Iterate over each coordinate in order
        for i in range(n_points):
            # Keep other coordinates fixed, optimize h[i] only
            old_h_i = best_h[i]
            old_contribution = old_conv[old_max_idx]
            
            # Try a simple line search approach for optimal h[i]
            # We know the new values change linearly
            # Use a fixed set of values around current h[i]
            try_points = np.linspace(max(0, old_h_i - 0.1), min(1, old_h_i + 0.1), 30)
            best_new_h_i = old_h_i
            best_new_c5 = best_c5
            
            # Compute C5 for variations around current value
            temp_h = best_h.copy()
            for new_h_i in try_points:
                temp_h[i] = new_h_i
                temp_h = project_h(temp_h, n_points)
                new_c5 = evaluate_c5(temp_h)
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_new_h_i = new_h_i
            
            if best_new_c5 < best_c5:
                best_h[i] = best_new_h_i
                best_h = project_h(best_h, n_points)
                best_c5 = best_new_c5
                improved = True
                iter_no_improvement = 0
                break  # Restart the iteration to pick up new max_idx
            else:
                # If not improving, maybe still try nearby values
                # Adjust the step size if it seems stuck
                pass
        
        if not improved:
            iter_no_improvement += 1
            if iter_no_improvement > max_no_improvement:
                break
                
    return best_h, best_c5

def nonlinear_conjugate_gradient(h, n_points, max_iter=2000):
    """Use nonlinear conjugate gradient to accelerate convergence."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    iter_no_improvement = 0
    max_no_improvement = 500
    max_iter = max_iter
    
    # Initialize CG variables
    x = h.copy()
    g = np.zeros(n_points)
    d = np.zeros(n_points)
    old_g = np.zeros(n_points)
    old_x = x.copy()
    
    old_c5 = best_c5
    beta = 0.0  # for Polak-Ribiere
    
    direction = "FR"
    momentum = 0.9
    
    def compute_gradient(x, pos=None):
        # Numerical gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(x)
        # For every element: compute difference, compute C5 difference, divide by eps
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += eps
            x_plus = project_h(x_plus, n_points)
            c5_plus = evaluate_c5(x_plus)
            x_minus = x.copy()
            x_minus[i] -= eps
            x_minus = project_h(x_minus, n_points)
            c5_minus = evaluate_c5(x_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    for iteration in range(max_iter):
        g = compute_gradient(x)
        if iteration == 0:
            d = -g
        else:
            # Polak-Ribiere CG
            beta = max(0, np.dot(g, g - old_g) / np.dot(old_g, old_g))
            d = -g + beta * d
        
        # Find stepsize via line search to minimize along direction d
        def objective(t):
            new_x = x + t * d
            new_x = project_h(new_x, n_points)
            return evaluate_c5(new_x)
        
        # Bracket
        a, b = 0, 1.0
        try:
            res = optimize.minimize_scalar(objective, bounds=(a, b), method='bounded')
            t_opt = res.x
        except:
            t_opt = 0.5
        
        new_x = x + t_opt * d
        new_x = project_h(new_x, n_points)
        new_c5 = evaluate_c5(new_x)
        
        if new_c5 < best_c5:
            x = new_x
            best_h = x.copy()
            best_c5 = new_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            
        if iter_no_improvement > max_no_improvement:
            break
            
        old_g = g.copy()
        old_x = x.copy()
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    time_budget_s = budget_s
    start_time = time.time()
    
    n_points = 45
    
    # Initialize
    h = initialize_h(n_points)
    
    # Phase 1: Coordinate-wise optimization
    h_cw, c5_cw = coordinate_wise_optimization(h, n_points, max_iter=1000)
    
    # Phase 2: Nonlinear conjugate gradient
    h_cg, c5_cg = nonlinear_conjugate_gradient(h_cw, n_points, max_iter=2000)
    
    # Return best result
    final_h = h_cg if c5_cg < c5_cw else h_cw
    final_c5 = min(c5_cg, c5_cw)
    
    # Final validation
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
