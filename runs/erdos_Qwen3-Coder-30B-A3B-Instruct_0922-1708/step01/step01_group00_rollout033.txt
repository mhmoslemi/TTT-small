import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Make it more structured to help optimization
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    def compute_c5(h_vals):
        # Compute the overlap integral
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # Gradient-based optimization
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        h_base = h.copy()
        
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps
            
        # Update using gradient descent
        lr = 0.005
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Periodic check for early stopping
        if iteration % 100 == 0:
            if best_c5 < 0.381:
                break
    
    # Post-processing - ensure it's a valid step function with integer values
    # of 0 or 1, by thresholding (since it might be close, we consider practical areas)
    # To maintain closeness to the best feasible solution, we're being careful:
    # Post-process with smoothing to be close to step function
    h_smooth = np.zeros_like(best_h)
    for i in range(n_points):
        # For smoothing we average the values + neighbors
        left_idx = max(0, i - 1)
        right_idx = min(n_points - 1, i + 1) 
        h_smooth[i] = np.mean(best_h[left_idx:right_idx+1])
    h = h_smooth

    # Final refinement of the shape to be binary values
    for i in range(n_points):
        h[i] = 1.0 if h[i] > 0.5 else 0.0

    # Normalize to area = 1
    h = h * (n_points / 2) / np.sum(h)

    # Final validation
    final_c5 = compute_c5(h)

    return (h.tolist(), final_c5, n_points)
