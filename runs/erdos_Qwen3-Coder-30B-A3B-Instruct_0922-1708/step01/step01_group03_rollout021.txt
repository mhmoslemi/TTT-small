import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use symmetric triangular pulses
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def find_worst_shift(h):
    """Find shift that gives maximum overlap."""
    n = len(h)
    conv = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(conv)
    shift = max_idx - (n - 1)
    return shift

def compute_exact_gradient(h, shift, n_points):
    """Compute exact gradient with respect to shift."""
    grad = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(n_points):
        # For shift k, derivative at position i is (1 - h[(i+k) mod n])
        k = shift
        j = (i + k) % n_points
        grad[i] = (1 - h[j]) * dx
    return grad

def line_search(h, grad, max_step, n_points):
    """Perform line search to find best step size."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    try:
        # Bracket the minimum
        result = minimize_scalar(objective, bounds=(0, max_step), method='bounded')
        return result.x
    except:
        return 0.0

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45
    h = initialize_h(n_points)
    
    total_time = 0.0
    start_time = time.time()
    
    # Set up learning rate schedules and bounds
    eta = 0.1
    eta_min = 1e-6
    eta_max = 0.5
    max_no_improvement = 1000
    no_improvement_count = 0
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    while total_time < budget_s:
        # Find the shift causing maximum overlap
        worst_shift = find_worst_shift(best_h)
        
        # Compute exact gradient at that shift
        grad = compute_exact_gradient(best_h, worst_shift, n_points)
        grad = grad / (np.linalg.norm(grad) + 1e-12)  # Normalize
        
        # Line search to determine optimal step size
        step_size = line_search(best_h, grad, eta, n_points)
        if step_size <= 0.0:
            break
            
        # Update h with new step
        h_new = best_h - step_size * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
            # Increase learning rate if we are improving
            eta = min(eta * 1.2, eta_max)
        else:
            no_improvement_count += 1
            # Reduce learning rate if we are not improving
            eta = max(eta / 2.0, eta_min)
        
        if no_improvement_count > max_no_improvement:
            break
            
        # Update time tracking
        total_time = time.time() - start_time
        
        if total_time >= budget_s:
            break
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
