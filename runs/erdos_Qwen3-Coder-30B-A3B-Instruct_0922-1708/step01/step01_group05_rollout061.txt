import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Better alternative initialization
        h = np.zeros(n_points)
        segments = 8
        seg_size = n_points // segments
        for i in range(segments):
            start = i * seg_size
            end = start + seg_size if i < segments - 1 else n_points
            if i % 2 == 0:
                h[start:end] = 1.0
            else:
                h[start:end] = 0.0
        # Adjust to correct sum
        h = h * (target_sum / np.sum(h))
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def perturb_h(h_vals, perturb_factor=0.1):
        # Slightly modify h in a feasible way
        h_new = h_vals.copy()
        # Select a random segment to perturb
        seg_size = max(2, n_points // 10)
        start = np.random.randint(0, n_points - seg_size)
        end = start + seg_size
        # Make a smooth local perturbation
        for i in range(start, end):
            delta = np.random.uniform(-perturb_factor, perturb_factor)
            h_new[i] += delta
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        return h_new
    
    def simulated_annealing():
        current_h = h.copy()
        current_C = compute_C(current_h)
        best_h = current_h.copy()
        best_C = current_C
        
        # Simulated Annealing parameters
        T = 1.0
        alpha = 0.995
        min_T = 1e-6
        max_iter = 50000
        iter_count = 0
        
        while iter_count < max_iter and time() - start_time < budget_s - 1:
            iter_count += 1
            # Perturb current solution
            new_h = perturb_h(current_h)
            new_C = compute_C(new_h)
            
            # Accept or reject the new solution
            if new_C < current_C or np.random.rand() < np.exp((current_C - new_C) / T):
                current_h = new_h
                current_C = new_C
                
                if new_C < best_C:
                    best_h = new_h.copy()
                    best_C = new_C
                    
            # Decrease temperature
            T = max(min_T, T * alpha)
            
        return best_h, best_C

    # Run simulated annealing to get a good starting point
    best_h, best_C = simulated_annealing()
    
    # Apply gradient descent refinement using scipy.optimize
    def obj_func(h_vals):
        # Return negative of C_5 because we want to minimize
        return -compute_C(h_vals)
    
    def constraint_func(h_vals):
        return np.sum(h_vals) - target_sum
    
    # Refine using scipy optimization
    constraints = {'type': 'eq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Create a local optimization starting with best_h
    h_refined = best_h.copy()
    
    # Use L-BFGS-B for better gradient-based refinement
    result = so.minimize(
        fun=obj_func,
        x0=h_refined,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    
    if result.success:
        h_final = result.x
    else:
        h_final = h_refined
    
    # Final refinement to ensure correct sum
    h_final = h_final * (target_sum / np.sum(h_final))
    final_C = compute_C(h_final)
    
    if final_C < best_C:
        best_C = final_C
        best_h = h_final
    
    return (best_h.tolist(), best_C, n_points)
