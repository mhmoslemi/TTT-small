import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Symmetric construction
        h = np.ones(n_points) * 0.5
        # Create a sparse, symmetric pattern
        mid = n_points // 2
        h[mid-10:mid+10] = 1.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and (time.time() - start_time - last_improvement) < 300:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = time.time() - start_time
            
        # Simple gradient estimation using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update h using gradient descent
        lr = 0.01
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally add some randomness to escape local minima
        if iteration % 50 == 0:
            noise_scale = 0.001
            h = h + np.random.normal(0, noise_scale, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
    return (best_h.tolist(), best_c5, n_points)
