import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: smooth triangular pulse
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Make it [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h, n_points):
    """Compute gradient of C5 with respect to h for the worst shift."""
    dx = 2.0 / n_points
    # Compute full convolution to find worst shift
    conv = np.correlate(h, 1 - h, mode="full")
    # Find the maximum overlap shift
    max_idx = np.argmax(conv)
    # This gives the shift in range [-n+1, n-1]
    shift = max_idx - (n_points - 1)
    
    # Compute gradient at this shift
    grad = np.zeros(n_points)
    for i in range(n_points):
        # At shift k, we look at h[i] * (1 - h[i+k]) 
        # For the gradient w.r.t. h[i], this term contributes (1 - h[i+k])
        j = (i + shift) % n_points
        grad[i] = -dx * (1 - h[j])
    return grad

def line_search(h, grad, n_points, max_alpha=1.0):
    """Perform line search to find optimal step size."""
    def c5_at_alpha(alpha):
        h_new = h - alpha * grad
        return evaluate_c5(project_h(h_new, n_points))
    
    # Try multiple alpha values
    alphas = np.linspace(0, max_alpha, 20)
    c5_values = [c5_at_alpha(alpha) for alpha in alphas]
    best_idx = np.argmin(c5_values)
    return alphas[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45  # Smaller for efficiency
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Parameters
    learning_rate = 0.1
    min_learning_rate = 1e-6
    patience = 0
    max_patience = 100
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    while time.time() - start_time < budget_s * 0.95:
        # Compute gradient
        grad = compute_gradient(h, n_points)
        
        # Line search for optimal step size
        alpha = line_search(h, grad, n_points, learning_rate)
        
        # Update h
        h_new = h - alpha * grad
        h = project_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h)
        
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
            # Reset learning rate on improvement
            learning_rate = 0.1
        else:
            patience += 1
            # Reduce learning rate if no improvement
            learning_rate *= 0.5
            if learning_rate < min_learning_rate:
                break
                
        if patience > max_patience:
            break
    
    # Final evaluation
    final_h = best_h
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
