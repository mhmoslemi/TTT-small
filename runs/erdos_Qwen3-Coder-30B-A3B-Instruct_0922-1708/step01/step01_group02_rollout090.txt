import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Start with a genetic algorithm approach
    # Population size
    pop_size = 20
    num_generations = 1000
    mutation_rate = 0.1
    elite_size = 4
    
    # Create initial population
    population = []
    for _ in range(pop_size):
        individual = np.random.rand(n_points) * 0.7 + 0.15  # Keep in [0.15, 0.85]
        individual = project_h(individual)
        population.append(individual)
    
    # Main evolutionary loop
    for gen in range(num_generations):
        if (time.time() - start_time) > budget_s - 1:
            break
        
        # Evaluate fitness (inverted C5)
        fitness = []
        for ind in population:
            c5 = compute_c5(ind)
            fitness.append(-c5)
            
        # Rank individuals
        ranked_indices = np.argsort(fitness)[::-1]
        ranked_population = [population[i] for i in ranked_indices]
        
        # Update best solution
        current_best = ranked_population[0]
        current_c5 = -fitness[ranked_indices[0]]
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
            
        # Create new population
        new_population = []
        # Elitism
        for i in range(elite_size):
            new_population.append(ranked_population[i].copy())
        
        # Crossover and mutation for rest
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.choice(elite_size)
            parent2_idx = np.random.choice(elite_size)
            parent1 = ranked_population[parent1_idx]
            parent2 = ranked_population[parent2_idx]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mut_idx = np.random.randint(0, n_points)
                child[mut_idx] = np.random.rand() * 0.7 + 0.15  # Keep in [0.15, 0.85]
            
            child = project_h(child)
            new_population.append(child)
            
        population = new_population[:pop_size]
    
    # Final quality check
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
