import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO and gradient descent."""
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 45  # Using odd number for symmetric structure
    start_time = time.time()
    
    # Symmetric initialization
    h_init = initialize_h(n_points)
    if n_points % 2 == 1:
        center = n_points // 2
        # Make it symmetric around center
        for i in range(center):
            h_init[i] = h_init[n_points - 1 - i]
    
    # Phase 1: Particle Swarm Optimization with symmetry
    swarm_size = 30
    max_iterations = 500
    inertia_weight = 0.9
    cognitive_weight = 2.0
    social_weight = 2.0
    velocity_max = 0.2
    
    # Initialize particles
    particles = []
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    for _ in range(swarm_size):
        h = h_init.copy()
        # Add some noise
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = project_h(h, n_points)
        if n_points % 2 == 1:
            center = n_points // 2
            for i in range(center):
                h[i] = h[n_points - 1 - i]
        particles.append(h)
    
    # Main PSO loop
    for iter_num in range(max_iterations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        for i, h in enumerate(particles):
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            
            # Update particles
            for j in range(n_points):
                # Skip symmetric pairs when odd number
                if n_points % 2 == 1 and j > n_points // 2:
                    continue
                    
                # Velocity update (simplified: no personal best tracking for time efficiency)
                r1, r2 = np.random.random(2)
                velocity = (inertia_weight * (np.random.random() - 0.5) * velocity_max + 
                            cognitive_weight * r1 * (best_h[j] - h[j]) + 
                            social_weight * r2 * (best_h[j] - h[j]))
                velocity = np.clip(velocity, -velocity_max, velocity_max)
                
                h[j] += velocity
                
                # Apply symmetry constraint
                if n_points % 2 == 1 and j < n_points // 2:
                    h[n_points - 1 - j] = h[j]
        
        # Adaptive inertia weight
        inertia_weight = 0.9 - (0.5 * iter_num / max_iterations)
        
        # Rescale solutions to maintain constraints
        for i in range(swarm_size):
            h = particles[i]
            h = project_h(h, n_points)
            if n_points % 2 == 1:
                center = n_points // 2
                for j in range(center):
                    h[j] = h[n_points - 1 - j]
            particles[i] = h
            
    # Phase 2: Refinement with gradient-based method
    h_opt = best_h.copy()
    
    # Simple gradient decent refinemnt
    def obj_func(h):
        return evaluate_c5(h)
        
    def grad_func(h):
        # Using finite differences to approximate gradient
        eps = 1e-5
        grad = np.zeros_like(h)
        base_c5 = obj_func(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = obj_func(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Optimization loop
    learning_rate = 0.05
    for inner_iter in range(5000):
        if time.time() - start_time > budget_s:
            break
            
        gradient = grad_func(h_opt)
        h_new = h_opt - learning_rate * gradient
        h_new = project_h(h_new, n_points)
        
        # Apply symmetry constraint
        if n_points % 2 == 1:
            center = n_points // 2
            for i in range(center):
                h_new[i] = h_new[n_points - 1 - i]
        
        new_c5 = obj_func(h_new)
        if new_c5 < evaluate_c5(h_opt):
            h_opt = h_new
            learning_rate *= 0.9995
        else:
            learning_rate *= 0.5
            
        if learning_rate < 1e-8:
            break
            
    # Final validation
    final_h = project_h(h_opt, n_points)
    final_c5 = evaluate_c5(final_h)
    
    # Ensure symmetry
    if n_points % 2 == 1:
        center = n_points // 2
        for i in range(center):
            final_h[i] = final_h[n_points - 1 - i]
        final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
