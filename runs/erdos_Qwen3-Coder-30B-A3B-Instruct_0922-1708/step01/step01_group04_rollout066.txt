import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def build_h_from_segments(c, n_points):
        # Reconstruct h from segment averages c
        m = len(c)
        segment_size = n_points // m
        if segment_size * m < n_points:
            segment_size += 1
        h_reconstructed = np.zeros(n_points)
        for i in range(m):
            start_idx = i * segment_size
            end_idx = min((i + 1) * segment_size, n_points)
            h_reconstructed[start_idx:end_idx] = c[i]
        # Apply Gaussian smoothing for continuity
        smoothed = np.convolve(h_reconstructed, np.exp(-np.linspace(-2, 2, 9)**2/2), mode='same')
        return np.clip(smoothed, 0, 1)
    
    def optimize_segments(c, n_points, target_m):
        # Function to minimize
        def objective(c):
            h = build_h_from_segments(c, n_points)
            return evaluate(h)
        
        # Constraints
        def constraint_sum(c):
            return np.sum(c) - target_m / 2
        
        # Bounds
        bounds = [(0, 1) for _ in range(len(c))]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimization
        result = minimize(objective, c, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 500})
        return result.x
    
    # Initialize best solution
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    # Refinement strategy: start with small m, increase gradually
    start_m = 7
    max_n_points = n_points
    factor = 2  # double segment count at each step
    
    start_time = time.time()
    last_improvement = 0
    
    m = start_m
    while m <= max_n_points and time.time() - start_time < budget_s:
        # Create initial c
        if m <= len(h_start):
            # Build segments from existing h_start
            segment_size = n_points // m
            c = np.zeros(m)
            for i in range(m):
                start_idx = i * segment_size
                end_idx = min((i + 1) * segment_size, n_points)
                c[i] = np.mean(h_start[start_idx:end_idx])
        else:
            # Create a new uniform distribution
            c = np.ones(m) * 0.5
            c[0] = 0.2
            c[-1] = 0.8
            # Normalize
            c = c * (m / 2) / np.sum(c)
        
        # Optimize
        c_opt = optimize_segments(c, n_points, m)
        h_opt = build_h_from_segments(c_opt, n_points)
        c5_opt = evaluate(h_opt)
        
        # Update best solution
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Increase m
        new_m = m * factor
        if new_m > max_n_points:
            break
        m = new_m
            
        if last_improvement > 100:
            break
    
    return (best_h, best_c5, n_points)
