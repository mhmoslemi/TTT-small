import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h symmetrically around center."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    
    # Symmetric initialization
    x = np.linspace(0, 2, n_points)
    center1 = 0.5
    center2 = 1.5
    h = np.zeros(n_points)
    for i in range(n_points):
        xi = x[i]
        h[i] = max(0, min(1, np.exp(-10 * (xi - center1)**2) + np.exp(-10 * (xi - center2)**2)))
    
    h = project_h(h, n_points)
    return h

def enforce_symmetry(h):
    """Enforce symmetry h[i] = h[n-1-i] around center."""
    n = len(h)
    if n % 2 == 0:
        # Even case
        mid = n // 2
        for i in range(mid):
            avg_val = (h[i] + h[n-1-i]) / 2
            h[i] = avg_val
            h[n-1-i] = avg_val
    else:
        # Odd case
        mid = n // 2
        for i in range(mid):
            avg_val = (h[i] + h[n-1-i]) / 2
            h[i] = avg_val
            h[n-1-i] = avg_val
    return h

def compute_exact_gradient(h, max_shift_idx):
    """Compute exact gradient for maximum overlap shift."""
    n = len(h)
    dx = 2.0 / n
    dC5_dh = np.zeros(n)
    
    # The shift index from convolution is:
    # max_shift_idx = shift + (n - 1)
    # where shift is from - (n - 1) to n - 1
    shift = max_shift_idx - (n - 1)
    
    if shift >= 0:
        # Right shift case h[i] affects h[i+shift]
        for i in range(n - shift):
            dC5_dh[i] = -dx
    else:
        # Left shift case h[i] affects h[i-shift]
        for i in range(-shift, n):
            dC5_dh[i] = -dx
    
    # Apply the correct signs based on the specific convolution index
    # We compute the gradient for the full convolution, then only the max one contributes
    # Let us recompute more carefully:
    conv = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(conv)
    shift_from_conv = max_idx - (n - 1)  # - (n-1) to n-1 range
    
    dC5_dh = np.zeros(n)
    # In the full convolution of h with (1-h):
    # For i in range(n), the convolution at shift k includes h[i] * (1-h[i+k])
    # Format of convolution: len = 2*n-1
    # We care about these correlations that contribute to the maximum
    # Actually, let's directly compute one gradient for correct shift
    
    # For the specific shift that maximizes the convolution:
    if shift_from_conv >= 0:
        # This shift implies more of h's terms contribute
        # Let i run from 0 to n - shift_from_conv - 1, h[i+shift_from_conv] will be used
        for i in range(n - shift_from_conv):
            dC5_dh[i] = -dx
    else:
        # shift_from_conv < 0
        # Let i run from -shift_from_conv to n-1 such that i+shift_from_conv >=0
        for i in range(-shift_from_conv, n):
            dC5_dh[i] = -dx
    
    return dC5_dh

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function based on symmetric gradient descent."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # odd for symmetry
    h = initialize_h(n_points)
    h = enforce_symmetry(h)
    
    # Use gradient descent on the symmetric variable
    learning_rate = 0.1
    max_no_improvement = 1000
    iter_no_improvement = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    while time.time() - start_time < budget_s * 0.9:
        # Evaluate current C5 and find the index of max overlap
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute gradient
        grad = compute_exact_gradient(h, max_idx)
        
        # Update step
        h_new = h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        h_new = enforce_symmetry(h_new)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            learning_rate *= 0.99  # Decrease learning rate with time
        else:
            iter_no_improvement += 1
            learning_rate *= 2.0  # Increase learning rate if no progress
            if learning_rate > 1:
                learning_rate = 1
            if iter_no_improvement > max_no_improvement:
                break
        
        # Periodic check for good performance
        if iter_no_improvement % 100 == 0:
            if time.time() - start_time > budget_s * 0.95:
                break
    
    # Final check and validation
    final_h = enforce_symmetry(project_h(h, n_points))
    final_c5 = evaluate_c5(final_h)
    return (final_h, final_c5, n_points)
