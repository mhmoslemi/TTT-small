import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Use a simple sinusoidal initialization
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Segment-based search with low dimensionality
    m = 5  # number of segments
    possible_values = [0.0, 0.5, 1.0]
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Generate all possible segment combinations
    segment_combinations = list(product(possible_values, repeat=m))
    n_combinations = len(segment_combinations)
    
    start_time = time.time()
    
    # Process each combination
    for i, combination in enumerate(segment_combinations):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Construct piecewise constant function with m segments
        h_candidate = np.zeros(n_points)
        segment_length = n_points // m
        remainder = n_points % m
        
        idx = 0
        for j, val in enumerate(combination):
            # Distribute remainder across first 'remainder' segments
            actual_length = segment_length + (1 if j < remainder else 0)
            h_candidate[idx:idx + actual_length] = val
            idx += actual_length
            
        h_candidate = project_h(h_candidate, n_points)
        c5 = evaluate_h(h_candidate, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
    
    # Gradient refinement on top candidates
    top_candidates = []
    top_scores = []
    
    # Check nearby candidates to top-performing
    for _ in range(10):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        h_refined = best_h.copy()
        # Add small random perturbations
        for j in range(n_points):
            if np.random.rand() < 0.1:  # 10% chance to perturb
                h_refined[j] += np.random.normal(0, 0.05)
        h_refined = project_h(h_refined, n_points)
        
        c5 = evaluate_h(h_refined, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_refined.copy()
    
    # Final gradient descent
    h_final = best_h.copy()
    c5_final = best_c5
    learning_rate = 0.01
    patience = 0
    max_patience = 1000
    
    while time.time() - start_time < budget_s - 1:
        corr = correlate(h_final, 1 - h_final, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_final.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h_final -= learning_rate * grad
        h_final = project_h(h_final, n_points)
        
        new_c5 = evaluate_h(h_final, dx)
        if new_c5 < c5_final:
            c5_final = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 500 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (h_final, c5_final, n_points)
