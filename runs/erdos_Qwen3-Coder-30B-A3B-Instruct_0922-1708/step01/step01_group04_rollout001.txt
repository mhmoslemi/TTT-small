import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=3):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def project_constraints(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to have sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def objective(h):
        return evaluate(h)
    
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Set up optimization
    h_opt = h_start.copy()
    best_c5 = evaluate(h_opt)
    best_h = h_opt.copy()
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s and no_improvement_count < max_no_improvement:
        iterations += 1
        
        # Try small perturbations
        if iterations % 10 == 0:
            pert_h = h_opt.copy()
            # Randomly perturb a few points
            n_perturb = max(1, n_points // 10)
            indices = np.random.choice(n_points, n_perturb, replace=False)
            for i in indices:
                pert_h[i] += np.random.uniform(-0.05, 0.05)
            pert_h = project_constraints(pert_h)
            c5_trial = evaluate(pert_h)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = pert_h.copy()
                h_opt = pert_h.copy()
                no_improvement_count = 0
                continue
        
        # Try optimization step
        try:
            # Minimize using L-BFGS-B
            result = minimize(
                objective,
                h_opt,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 10, 'ftol': 1e-6, 'gtol': 1e-6}
            )
            if result.success:
                h_new = result.x.copy()
                h_new = project_constraints(h_new)
                c5_new = evaluate(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_opt = h_new.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
                # Perturb if optimization fails
                h_new = h_opt.copy()
                n_perturb = max(1, n_points // 20)
                indices = np.random.choice(n_points, n_perturb, replace=False)
                for i in indices:
                    h_new[i] += np.random.uniform(-0.02, 0.02)
                h_new = project_constraints(h_new)
                c5_new = evaluate(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_opt = h_new.copy()
                    no_improvement_count = 0
        except Exception:
            no_improvement_count += 1
        
        # Apply smoothing periodically
        if iterations % 5 == 0:
            h_opt = smooth(h_opt, window=3)
            h_opt = project_constraints(h_opt)
    
    # Final check
    final_c5 = evaluate(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h, best_c5, n_points)
