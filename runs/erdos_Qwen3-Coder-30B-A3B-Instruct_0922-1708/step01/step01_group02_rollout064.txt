import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 47
        # Create a symmetric random starting point
        h = np.random.rand(n_points) * 0.5 + 0.25
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(conv)
        return max_val * dx
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Initialize population
    pop_size = 100
    population = []
    for _ in range(pop_size):
        individual = np.random.rand(n_points) * 0.5 + 0.25
        individual = project_h(individual)
        population.append(individual)
    
    # Add initial solution if available
    try:
        if 'initial_h_values' in globals():
            initial_individual = project_h(np.array(initial_h_values))
            population[0] = initial_individual
    except:
        pass
    
    # GA parameters
    crossover_rate = 0.7
    mutation_rate_start = 0.2
    elite_size = pop_size // 10
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    max_generations = 10000
    
    while generation < max_generations and (time.time() - start_time) < budget_s - 1:
        # Evaluate population
        fitness = []
        for individual in population:
            c5 = compute_c5(individual)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]
        
        # Selection: roulette wheel with linear scaling
        scaled_fitness = [1.0 / (1e-10 + f) for f in sorted_fitness]
        sum_fitness = sum(scaled_fitness)
        probabilities = [f / sum_fitness for f in scaled_fitness]
        
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection
            idx1 = np.random.choice(pop_size, p=probabilities)
            idx2 = np.random.choice(pop_size, p=probabilities)
            
            parent1 = sorted_population[idx1]
            parent2 = sorted_population[idx2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            mutation_rate = max(mutation_rate_start / np.sqrt(generation + 1), 0.01)
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.1)
                    child1[i] += noise
                    child2[i] += noise
                    
            # Project to feasible set
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population[:pop_size]
        generation += 1
    
    # Final check
    if best_h is None:
        # If no solution found, use an initial random solution
        best_h = h.copy()
        best_c5 = compute_c5(best_h)
    
    # Ensure final solution meets constraints
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
