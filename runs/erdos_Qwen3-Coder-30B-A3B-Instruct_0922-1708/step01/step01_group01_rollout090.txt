import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Early termination if we don't have enough time
    if budget_s < 10:
        return (np.full(n_points, 0.5), 0.5, n_points)
    
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Starting with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Define a reduced search space
    # Use piecewise constant with 5 segments with values 0, 0.5, 1
    num_segments = 5
    segment_values = [0, 0.5, 1]
    total_combinations = 3 ** num_segments
    
    best_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    
    # Determine how many combinations to sample
    max_samples = min(total_combinations, 1000)
    indices = list(range(total_combinations))
    np.random.shuffle(indices)
    indices = indices[:max_samples]
    
    # Enumerate combinations with fixed segment structure
    segment_length = n_points // num_segments
    if segment_length * num_segments != n_points:
        segment_length = n_points // num_segments + 1
    
    # Sample configurations using a smarter approach than exhaustive
    for i in range(min(max_samples, 1000)):
        # Sample segment values
        vals = []
        for j in range(num_segments):
            vals.append(np.random.choice(segment_values))
        
        # Construct piecewise constant h
        h_candidate = np.zeros(n_points)
        for j in range(num_segments):
            start_idx = j * segment_length
            end_idx = min((j + 1) * segment_length, n_points)
            h_candidate[start_idx:end_idx] = vals[j % len(vals)]
        
        # Project to meet constraints
        h_candidate = project_h(h_candidate, n_points)
        
        # Evaluate C5
        c5 = evaluate_h(h_candidate, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
            
        if time.time() - start_time > 0.9 * budget_s:
            break
    
    # Refine best solution with gradient descent
    h_refined = best_h.copy()
    c5_refined = best_c5
    learning_rate = 0.01
    patience = 0
    max_patience = 1000
    
    # Create a flexible search refinement function
    def compute_gradient(h, dx):
        eps = 1e-5
        grad = np.zeros_like(h)
        for idx in range(len(h)):
            h_plus = h.copy()
            h_plus[idx] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[idx] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[idx] = (c_plus - c_minus) / (2 * eps)
        return grad
    
    while time.time() - start_time < 0.95 * budget_s:
        grad = compute_gradient(h_refined, dx)
        h_refined -= learning_rate * grad
        h_refined = project_h(h_refined, n_points)
        new_c5 = evaluate_h(h_refined, dx)
        if new_c5 < c5_refined:
            c5_refined = new_c5
            patience = 0
        else:
            patience += 1
            if patience >= max_patience:
                break
        if patience > 0 and patience % 100 == 0:
            learning_rate *= 0.5
    
    # Final evaluation
    final_c5 = evaluate_h(h_refined, dx)
    
    return (h_refined, final_c5, n_points)
