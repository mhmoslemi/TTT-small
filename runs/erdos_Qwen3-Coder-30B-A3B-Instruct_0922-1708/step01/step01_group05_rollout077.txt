import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))

    # PSO parameters
    swarm_size = 100
    max_iter = 500
    w = 0.7   # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    # Initialize PSO particles with h values
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    gbest = None
    gbest_value = float('inf')
    
    # Generate initial swarm
    for _ in range(swarm_size):
        particle = h.copy()
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values) + np.random.normal(0, 0.05, n_points)
        else:
            particle = np.random.rand(n_points) * 0.5
        particle = np.clip(particle, 0, 1)
        particle = particle * (target_sum / np.sum(particle))
        particles.append(particle)
        velocities.append(np.random.rand(n_points) * 0.1)
        pbest.append(particle.copy())
        pbest_values.append(float('inf'))

    # Define helper functions
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(np.abs(corr)) * dx

    def evaluate_particle(h_vals):
        # Normalize h_vals to satisfy integral constraint
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        h_vals = np.clip(h_vals, 0, 1)
        return compute_C(h_vals)

    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            particle = particles[i]
            velocity = velocities[i]
            pbest_pos = pbest[i]
            
            # Evaluate current particle
            particle_value = evaluate_particle(particle)
            
            # Update personal best
            if particle_value < pbest_values[i]:
                pbest[i] = particle.copy()
                pbest_values[i] = particle_value
            
            # Update global best
            if particle_value < gbest_value:
                gbest_value = particle_value
                gbest = particle.copy()
            
            # Update velocity and position
            r1, r2 = np.random.rand(2, n_points)
            velocity = (w * velocity + 
                        c1 * r1 * (pbest_pos - particle) + 
                        c2 * r2 * (gbest - particle))
            
            particle = particle + velocity
            particle = np.clip(particle, 0, 1)
            particle = particle * (target_sum / np.sum(particle))
            
            particles[i] = particle
            velocities[i] = velocity

    # Refine with gradient-based method
    h_refined = gbest.copy()
    lr = 0.01
    max_grad_iter = 100
    prev_C = gbest_value
    no_improvement_count = 0
    
    for i in range(max_grad_iter):
        if time() - start_time > budget_s - 1:
            break
            
        def objective(h_vals):
            h_norm = h_vals * (target_sum / np.sum(h_vals))
            h_norm = np.clip(h_norm, 0, 1)
            return compute_C(h_norm)
            
        def gradient(h_vals):
            eps = 1e-5
            grad = np.zeros_like(h_vals)
            for j in range(len(h_vals)):
                h_plus = h_vals.copy()
                h_plus[j] += eps
                grad[j] = (objective(h_plus) - objective(h_vals)) / eps
            return grad
            
        grad = gradient(h_refined)
        h_refined = h_refined - lr * grad
        h_refined = np.clip(h_refined, 0, 1)
        current_C = evaluate_particle(h_refined)
        
        if current_C < prev_C:
            prev_C = current_C
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 10:
                lr *= 0.5
                no_improvement_count = 0
                
        if lr < 1e-8:
            break

    # Final evaluation
    final_C = evaluate_particle(h_refined)
    final_h = h_refined.tolist()
    
    return (final_h, final_C, n_points)
