import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):

    n_particles = 100
    max_iter = 5000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Initialize PSO particles - positions are node values at sample points
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    # Inherit from initial values or create new random particles
    for i in range(n_particles):
        if 'initial_h_values' in globals() and i == 0:
            pos = h.copy()
        else:
            pos = np.random.rand(n_points) * 0.5  # Uniform in [0, 0.5]
            # Ensure integral constraint
            pos = pos * (target_sum / np.sum(pos))
        particles.append(pos.copy())
        velocities.append(np.random.rand(n_points) * 0.01)
        pbest.append(pos.copy())
        pbest_fitness.append(float('inf'))
    
    # Swarm best
    gbest = h.copy()
    gbest_fitness = float('inf')
    
    # Evaluate initial particles
    for i in range(n_particles):
        fit = evaluate_C(particles[i], dx)
        if fit < pbest_fitness[i]:
            pbest[i] = particles[i].copy()
            pbest_fitness[i] = fit
        if fit < gbest_fitness:
            gbest = particles[i].copy()
            gbest_fitness = fit
    
    iter_count = 0
    # Main PSO loop
    while (time() - start_time < budget_s - 1) and iter_count < max_iter:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            # Update position
            particles[i] += velocities[i]
            # Clip to feasible range [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            # Rescale to meet integral constraint
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            
            # Evaluate fitness
            fitness = evaluate_C(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
                # Update global best
                if fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = fitness
        
        iter_count += 1

    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))

        eps = 1e-5
        grad = np.zeros_like(h_vals)
        C_ref = compute_C(h_vals)

            h_plus = h_vals.copy()
            h_plus[i] += eps
            # Ensure constraints
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C_ref) / eps
        return grad

        corr = np.correlate(h, 1-h, mode="full")

        return np.max(np.abs(corr))
    
    # Gradient descent refinement loop
    refine_time = time()
    while time() - refine_time < 100 and time() - start_time < budget_s - 1:
        # Get gradient
        grad = gradient_h(h)
        # Update
        h -= lr * grad
        # Bound and rescale
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Evaluate
        C = evaluate_C(h)
        if C < best_C:
            best_C = C
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 30:
                lr *= 0.5
                no_improvement_count = 0
    
    return (best_h.tolist(), best_C, n_points)

def evaluate_C(h_vals, dx):
    h = np.array(h_vals)
    corr = np.correlate(h, 1-h, mode="full")
    corr *= dx
    return np.max(np.abs(corr))
