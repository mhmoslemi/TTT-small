import numpy as np
from time import time
import scipy.optimize as so

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO Parameters
    n_particles = 100
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    best_particles = []
    best_values = []
    
    for _ in range(n_particles):
        # Parameterize h as values at nodes
        if 'initial_h_values' in globals():
            h_p = h.copy()
        else:
            h_p = np.random.rand(n_points)
        h_p = np.clip(h_p, 0, 1)
        h_p = h_p * (target_sum / np.sum(h_p))
        particles.append(h_p.copy())
        velocities.append(np.random.rand(n_points) * 0.01)
        best_particles.append(h_p.copy())
        best_values.append(float('inf'))
    
    global_best = h.copy()
    global_best_value = float('inf')
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def project_feasible(h_vals):
        h_proj = np.clip(h_vals, 0, 1)
        h_proj = h_proj * (target_sum / np.sum(h_proj))  # Rescale
        return h_proj
    
    # PSO loop
    max_pso_iter = 500
    pso_iter = 0
    
    while time() - start_time < max_pso_iter and pso_iter < max_pso_iter:
        pso_iter += 1
        
        for i in range(n_particles):
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                            cognitive * r1 * (best_particles[i] - particles[i]) +
                            social * r2 * (global_best - particles[i]))
            
            particles[i] += velocities[i]
            
            # Project to feasible space
            particles[i] = project_feasible(particles[i])
            
            # Evaluate
            current_C = compute_C(particles[i])
            
            # Update personal best
            if current_C < best_values[i]:
                best_values[i] = current_C
                best_particles[i] = particles[i].copy()
            
            # Update global best
            if current_C < global_best_value:
                global_best_value = current_C
                global_best = particles[i].copy()
    
    # Refinement with gradient-based method
    # Start with global best
    h = global_best.copy()
    best_C = global_best_value
    lr = 0.01
    
    # Gradient-based optimization for refinement
    max_refine_iter = 100
    refine_iter = 0
    no_improvement_count = 0
    
    def finite_diff_grad(h_vals, eps=1e-5):
        grad = np.zeros_like(h_vals)
        base_C = compute_C(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus)
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - base_C) / eps
        return grad
    
    while time() - start_time < budget_s - 1 and refine_iter < max_refine_iter:
        refine_iter += 1
        C = compute_C(h)
        grad = finite_diff_grad(h)
        
        # Update
        h -= lr * grad
        h = project_feasible(h)
        
        new_C = compute_C(h)
        if new_C < C:
            no_improvement_count = 0
            if new_C < best_C:
                best_C = new_C
                best_h = h.copy()
        else:
            no_improvement_count += 1
            if no_improvement_count > 10:
                lr *= 0.5
                no_improvement_count = 0
    
    # Final check
    final_h = best_h if 'best_h' in locals() else h
    final_C = compute_C(final_h)
    
    return (final_h.tolist(), final_C, n_points)
