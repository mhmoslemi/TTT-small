import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    # Genetic Algorithm Params
    population_size = 100
    mutation_rate = 0.2
    crossover_rate = 0.7
    generations = 0
    max_generations = 10000
    elite_count = population_size // 10
    
    # Initialize population
    population = []
    for _ in range(population_size):
        ind = np.random.rand(n_points)
        ind = project_h(ind)
        population.append(ind)
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        population[0] = project_h(h)
    
    best_individual = population[0].copy()
    best_c5 = compute_c5(best_individual)
    
    # GA Loop
    while generations < max_generations and (time.time() - start_time) < budget_s - 1:
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind)
            fitness.append(c5)
        
        # Track best individual
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_c5 = fitness[min_idx]
            best_individual = population[min_idx].copy()
        
        # Selection (Roulette Wheel)
        fitness = np.array(fitness)
        inverted_fitness = 1.0 / (fitness + 1e-10)
        total_fitness = np.sum(inverted_fitness)
        probabilities = inverted_fitness / total_fitness
        
        new_population = []
        elite_indices = np.argsort(fitness)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Select parents
            parent1_idx = np.random.choice(population_size, p=probabilities)
            parent2_idx = np.random.choice(population_size, p=probabilities)
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate / np.sqrt(generations + 1):
                        noise = np.random.normal(0, 0.1)
                        child[i] += noise
                child = project_h(child)
                new_population.append(child)
            
            if len(new_population) >= population_size:
                break
        
        # Update population
        population = new_population[:population_size]
        
        generations += 1
        
        # Adaptive mutation rate
        if generations % 100 == 0 and mutation_rate > 0.01:
            mutation_rate *= 0.99
    
    # Final evaluation
    final_c5 = compute_c5(best_individual)
    
    return (best_individual.tolist(), final_c5, len(best_individual))
