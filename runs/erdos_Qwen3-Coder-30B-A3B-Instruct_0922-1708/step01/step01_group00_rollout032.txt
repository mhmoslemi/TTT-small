import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Rescale to sum to n_points/2
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Objective function
    def objective(h):
        # Compute overlaps for all lags
        correlation = np.correlate(h, 1 - h, mode='full')
        # Take maximum correlation (over all lags)
        c5 = np.max(correlation) * dx
        return c5
        
    # Constraints for optimization
    def constraint_sum(h):
        return np.sum(h) * dx - 1.0
        
    # Bounds for variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    iteration = 0
    
    # Outer loop for refinement
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Optimization step using scipy minimize
        # Take a few iterations of gradient-based optimization
        try:
            result = minimize(objective,
                          h,
                          method='L-BFGS-B',
                          bounds=bounds,
                          constraints={'type': 'eq', 'fun': constraint_sum},
                          options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10})
            if result.success:
                h = result.x
                # Ensure sum constraint and bounds
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fall back to gradient descent with fixed step
                # Calculate gradient numerically
                eps = 1e-5
                grad = np.zeros_like(h)
                c5_base = objective(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c5_plus = objective(h_plus)
                    grad[i] = (c5_plus - c5_base) / eps
                # Update using gradient descent
                lr = 0.001
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except:
            # Fallback to gradient descent if optimization fails
            eps = 1e-5
            grad = np.zeros_like(h)
            c5_base = objective(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = objective(h_plus)
                grad[i] = (c5_plus - c5_base) / eps
            lr = 0.001
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
        # Update best solution if improved
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Occasionally refine and restart
        if iteration % 20 == 0:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            # Some random perturbation to escape local minima
            h = h + np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final computation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
