import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to the desired resolution
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Black-box optimization using gradient descent with adaptive steps
    def evaluate_and_gradient(h):
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5, correlation
    
    # Gradient computation
    def compute_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_c5, _ = evaluate_and_gradient(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            c5_plus, _ = evaluate_and_gradient(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        if iteration % 10 == 0:
            # Re-evaluate current best
            correlation = np.correlate(best_h, 1 - best_h, mode='full')
            c5_best = np.max(correlation) * dx
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = best_h.copy()
                
        # Compute gradient and update
        grad = compute_gradient(h)
        lr = 0.01
        
        # Adaptive step size
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally restart with the current best to escape local minima
        if iteration % 50 == 0:
            h = best_h.copy()
            # Add some noise to avoid getting stuck
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final check of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
