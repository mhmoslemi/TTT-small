import numpy as np
from scipy.optimize import line_search
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h, k, n_points):
    """Compute gradient w.r.t. h for the specific shift k."""
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + k) % n_points
        grad[i] = (1 - h[j]) * dx
    return grad

def line_search_step(h, grad, eta, n_points):
    """Perform line search to find optimal step size."""
    # Simple bracketing line search
    alpha_min, alpha_max = 0.0, eta
    alpha = alpha_max / 2.0
    best_alpha = alpha
    best_c5 = np.inf
    
    def eval_c5(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
        
    for _ in range(10):  # Max iterations for line search
        c5 = eval_c5(alpha)
        if c5 < best_c5:
            best_c5 = c5
            best_alpha = alpha
        if alpha == alpha_min or alpha == alpha_max:
            break
        # Quadratic interpolation or bisection
        alpha = (alpha_min + alpha_max) / 2.0
    
    return best_alpha

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using gradient descent with exact gradients."""
    np.random.seed(seed)
    n_points = 45
    h = initialize_h(n_points)
    
    start_time = time.time()
    eta = 0.1
    eta_min = 1e-6
    eta_max = 0.5
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    no_improvement_count = 0
    max_no_improvement = 1000
    
    # Gradient descent with exact gradient computation
    while time.time() - start_time < budget_s and no_improvement_count < max_no_improvement:
        # Compute correlations and find worst shift
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)  # shift relative to center
        # Using the found shift to compute gradient
        grad = compute_gradient(h, shift, n_points)
        
        # Perform line search for optimal step size
        alpha = line_search_step(h, grad, eta, n_points)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Check improvement
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
            # Increase learning rate
            eta = min(eta_max, eta * 1.2)
        else:
            no_improvement_count += 1
            # Decrease learning rate
            eta = max(eta_min, eta / 2.0)
    
    best_h = project_h(best_h, n_points)
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
