import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric function
    x = np.linspace(0, 2, n_points)
    # Create a symmetric, smooth function
    h = np.exp(-10 * (x - 0.5)**2) + 0.5 * np.exp(-10 * (x - 1.5)**2)
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO followed by gradient descent."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45  # odd number for symmetry
    total_time = 0
    
    # Phase 1: Particle Swarm Optimization
    start_time = time.time()
    swarm_size = 30
    max_iter = 10000
    cognitive_weight = 2.0
    social_weight = 2.0
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Create initial swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_score = []
    
    # Initialize with symmetric constraints: h[i] = h[n_points-1-i]
    for _ in range(swarm_size):
        h = initialize_h(n_points)
        # Ensure symmetry for odd n_points
        h_sym = h.copy()
        mid = n_points // 2
        for i in range(mid):
            h_sym[i] = (h[i] + h[n_points - 1 - i]) / 2.0
            h_sym[n_points - 1 - i] = h_sym[i]
        h = h_sym
        h = project_h(h, n_points)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        personal_best_score.append(evaluate_c5(h))
    
    # Global best
    gb_index = np.argmin(personal_best_score)
    global_best = personal_best[gb_index].copy()
    global_best_score = personal_best_score[gb_index]
    
    # PSO main loop
    omega_decay = 0.99
    iteration = 0
    while total_time < budget_s * 0.7 and iteration < max_iter:
        for i in range(swarm_size):
            # Update velocities
            r1, r2 = np.random.random(2)
            velocities[i] = (
                inertia_weight * velocities[i] +
                cognitive_weight * r1 * (personal_best[i] - swarm[i]) +
                social_weight * r2 * (global_best - swarm[i])
            )
            # Apply velocity limits
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            # Update positions
            swarm[i] += velocities[i]
            # Apply boundary conditions (clipping)
            swarm[i] = np.clip(swarm[i], 0, 1)
            # Maintain symmetry
            mid = n_points // 2
            for k in range(mid):
                swarm[i][n_points - 1 - k] = swarm[i][k]
            # Rescale and project
            swarm[i] = project_h(swarm[i], n_points)
            # Evaluate
            new_score = evaluate_c5(swarm[i])
            # Update personal best
            if new_score < personal_best_score[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_score[i] = new_score
                # Update global best
                if new_score < global_best_score:
                    global_best = swarm[i].copy()
                    global_best_score = new_score
        # Update inertia weight
        inertia_weight *= omega_decay
        iteration += 1
        total_time = time.time() - start_time
    
    # Phase 2: Gradient Descent
    # Use the global best from PSO as starting point
    best_h = global_best.copy()
    best_c5 = global_best_score
    start_time = time.time()
    # Fixed learning rate
    lr = 0.05
    max_iter_gd = 10000
    it = 0
    while total_time < budget_s and it < max_iter_gd:
        # Compute gradient with respect to h for the maximum overlap
        # This is simplified partial gradient w.r.t. one of the max shifts
        # We find k that gives max overlap, and take derivative w.rt. h at that k
        # Directly compute finite difference gradient
        h = best_h.copy()
        eps = 1e-4
        nabla = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            # Keep symmetry
            mid = n_points // 2
            h_plus[n_points - 1 - i] = h_plus[i]
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            # Keep symmetry
            h_minus[n_points - 1 - i] = h_minus[i]
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            nabla[i] = (c5_plus - c5_minus) / (2 * eps)
        # Update step
        h_new = h - lr * nabla
        # Keep symmetry
        mid = n_points // 2
        for i in range(mid):
            h_new[n_points - 1 - i] = h_new[i]
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        else:
            lr = lr * 0.99  # decrease learning rate
        
        total_time = time.time() - start_time
        it += 1
        
        if lr < 1e-8:
            break
    
    # Double-check final evaluation
    best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, n_points)
