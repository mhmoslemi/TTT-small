import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using Simulated Annealing and Gradient Descent"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal pattern
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    step_size = 0.1
    
    # Gradient Descent parameters
    gd_lr = 0.01
    gd_patience = 0
    max_gd_patience = 5000
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    
    # Simulated Annealing
    for iteration in range(max_iter):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Generate neighbor
        h_new = h + np.random.uniform(-step_size, step_size, n_points)
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_new.copy()
            best_h = h_new.copy()
            best_c5 = new_c5
            gd_patience = 0
        else:
            # Accept with probability based on temperature
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new.copy()
                gd_patience = 0
            else:
                gd_patience += 1
        
        # Cool down
        T *= alpha
        if T < 1e-8:
            T = 1e-8
            
        # Occasionally reset patience for GD
        if iteration % 100 == 0:
            gd_patience = 0
    
    # Gradient Descent refinement
    h = best_h.copy()
    while time.time() - start_time < budget_s - 1.0:
        # Compute current gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        base_c5 = evaluate_h(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with gradient
        h -= gd_lr * np.sign(grad)
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            gd_patience = 0
        else:
            gd_patience += 1
            
        if gd_patience > max_gd_patience:
            break
            
        if gd_patience % 1000 == 0:
            gd_lr *= 0.5
    
    return (best_h, best_c5, n_points)
