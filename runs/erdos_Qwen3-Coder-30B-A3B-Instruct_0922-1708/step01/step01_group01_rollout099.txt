import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def dual_optimization_phase(initial_h, n_points, budget_s, dx):
    """Dual optimization approach with adaptive updates"""
    start_time = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Initialize dual variables
    lambda_k = np.ones(2 * n_points - 1)
    learning_rate = 0.01
    
    while time.time() - start_time < 0.8 * budget_s:
        # Calculate current overlaps for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        max_corr = corr[max_shift_idx]
        
        # Update dual variables based on current solution
        # For maximization of min overlap
        # We penalize large overlaps by increasing dual weights
        lambda_k += learning_rate * (1 - corr)
        lambda_k = np.maximum(lambda_k, 0)  # Ensure non-negative
        
        # Update h by solving a relaxed version
        # We approximate the problem with linear penalties
        h_new = h.copy()
        eps = 1e-5
        
        # Simple gradient estimate in direction of minimizing overlaps
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)

            grad_h = (c_plus - c_minus) / (2 * eps)
            h_new[i] -= 0.01 * grad_h
            h_new[i] = max(0, min(1, h_new[i]))

        h_new = project_h(h_new, n_points)
        h = h_new
        
        # Update best solution
        current_c5 = evaluate_h(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Adaptive learning rate reduction
        if np.random.rand() < 0.01:
            learning_rate *= 0.98
    
    return best_h, best_c5

def gradient_descent_refinement(initial_h, n_points, budget_s, dx):
    """Local refinement using gradient descent"""
    start_time = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    learning_rate = 0.01
    patience = 0
    max_patience = 1000
    
    while time.time() - start_time < 0.2 * budget_s:
        # Calculate overlap
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        max_corr = corr[max_shift_idx]
        
        # Estimate gradient with finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update h
        h = h - learning_rate * grad
        h = project_h(h, n_points)
        
        # Track best
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
            
        # Adaptive learning rate
        if patience > 0 and patience % 500 == 0:
            learning_rate *= 0.9
    
    return best_h, best_c5

    # Gradient descent refinement
    final_h, final_c5 = gradient_descent_refinement(best_h, n_points, budget_s, dx)
