import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    # PSO parameters
    n_particles = 50
    max_iter = 10000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles and velocities
    particles = np.array([h.copy() for _ in range(n_particles)])
    velocities = np.random.uniform(-1, 1, (n_particles, n_points))
    pbest = np.array([p.copy() for p in particles])
    pbest_scores = np.array([compute_c5(p) for p in particles])
    
    # Initialize gbest
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        w = w_start - (w_start - w_end) * (iteration / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project onto feasible set
            particles[i] = project_h(particles[i])
            
            # Evaluate fitness
            current_score = compute_c5(particles[i])
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = current_score
                
                # Update global best
                if current_score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = current_score
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
        
        iteration += 1
        
        if no_improvement_count >= max_no_improvement:
            break
    
    return (gbest.tolist(), gbest_score, len(gbest))
