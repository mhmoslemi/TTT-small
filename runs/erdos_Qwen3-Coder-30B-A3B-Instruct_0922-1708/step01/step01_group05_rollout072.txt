    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create flat
    if 'initial_h_values' in globals():

    else:
        h = np.full(n_points, 0.5)
    
    # Ensure integral constraint
    h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_C = float('inf')
    
    # adaptive step size
    lr = 0.05
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def evaluate_and_update():
        nonlocal best_C, best_h
        C = compute_C(h)
        if C < best_C:
            best_C = C
            best_h = h.copy()
        return C
    
    # Initial evaluation
    previous_C = evaluate_and_update()
    
    iter_count = 0
    while time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        current_C = compute_C(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - current_C) / eps
        
        # Update h
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))  # Ensure constraint
        
        # Update learning rate
        new_C = evaluate_and_update()
        if new_C >= previous_C:
            lr *= 0.995  # Decrease learning rate
        else:
            lr = min(lr * 1.01, 0.5)  # Increase a bit if improving
        previous_C = new_C
        
        # Early termination
        if best_C < 0.3807:
            break
            
    return (best_h.tolist(), best_C, n_points)
