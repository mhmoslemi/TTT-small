import time
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run(seed=42, budget_s=600, **kwargs):

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))

    def project_h(h_vals):
        # Ensure integral constraint
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        # Ensure bounds
        h_vals = np.clip(h_vals, 0, 1)
        return h_vals

    def gradient_descent_step(h_vals, lr=0.01, max_iter=20):
        h_old = h_vals.copy()
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            C_old = compute_C(h_old)
            
            # Compute finite difference gradient
            grad = np.zeros_like(h_old)
            eps = 1e-5
            for i in range(len(h_old)):
                h_plus = h_old.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus)
                C_plus = compute_C(h_plus)
                grad[i] = (C_plus - C_old) / eps
              
            # Update
            h_new = h_old - lr * grad
            h_new = project_h(h_new)
            
            # Adaptive learning rate
            C_new = compute_C(h_new)
            if C_new < C_old:
                h_old = h_new
                lr *= 1.05  # Increase if improving
            else:
                lr *= 0.5   # Reduce if not improving
                if lr < 1e-6:
                    break
        return h_old

    # PSO parameters
    n_particles = 100
    max_iter = 5000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1

    particles = []
    best_positions = []
    velos = []
    best_C = float('inf')
    best_h = h.copy()

        else:
            p = np.random.rand(n_points)
        p = np.clip(p, 0, 1)
        p = p * (target_sum / np.sum(p))
        particles.append(p.copy())
        velos.append(np.random.rand(n_points) * v_max * 2 - v_max)
        C = compute_C(p)
        if C < best_C:
            best_C = C
            best_h = p.copy()
    
    # PSO main loop
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i, (pos, vel) in enumerate(zip(particles, velos)):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Update velocity
            r1, r2 = np.random.rand(2)
            pbest = particles[i]
            gbest = best_h
            
            vel = (w * vel + 
                   c1 * r1 * (pbest - pos) + 
                   c2 * r2 * (gbest - pos))
            
            # Clip velocity
            vel = np.clip(vel, -v_max, v_max)
            
            # Update position
            pos += vel
            
            # Clip position to [0, 2]
            pos = np.clip(pos, 0, 2)
            
            # Project to feasible h values
            pos = project_h(pos)
            
            # Store updated particle
            particles[i] = pos.copy()
            velos[i] = vel.copy()
            
            # Evaluate
            C = compute_C(pos)
            if C < best_C:
                best_C = C
                best_h = pos.copy()
    
    # Refinement with gradient descent on top PSO candidates
    candidates = [particles[i] for i in range(n_particles)]
    candidates = sorted(candidates, key=compute_C)
    
    ref_num = max(1, n_particles // 10)
    for i in range(ref_num):
        if time.time() - start_time > budget_s - 0.5:
            break
        ref_h = gradient_descent_step(candidates[i], lr=0.01, max_iter=100)
        C_ref = compute_C(ref_h)
        if C_ref < best_C:
            best_C = C_ref
            best_h = ref_h.copy()

    return (best_h.tolist(), best_C, n_points)
