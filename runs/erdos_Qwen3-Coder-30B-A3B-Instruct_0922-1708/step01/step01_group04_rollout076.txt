import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create initial construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute overlap
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    def constraint_eq(h):
        return np.sum(h) * dx - 1.0
    
    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_eq}
    
    # Track best solution
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < 500:
        iteration += 1
        
        # Try optimization
        try:
            result = minimize(
                objective,
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            
            if result.success:
                h_new = result.x
                # Project back to [0,1] and renormalize
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                c5_new = evaluate(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        except:
            no_improvement_count += 1
        
        # Periodic perturbation
        if iteration % 10 == 0:
            h_trial = best_h.copy()
            # Some random perturbations
            n_perturbed = max(1, n_points // 10)
            indices = np.random.choice(n_points, n_perturbed, replace=False)
            for i in indices:
                h_trial[i] += np.random.normal(0, 0.02)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
    return (best_h, best_c5, n_points)
