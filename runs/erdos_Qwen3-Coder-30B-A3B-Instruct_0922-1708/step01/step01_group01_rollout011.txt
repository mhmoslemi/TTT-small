import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def latin_hypercube_sampling(n_samples, n_points, seed=42):
    """Generate Latin Hypercube samples"""
    np.random.seed(seed)
    samples = []
    for _ in range(n_samples):
        # Generate a random permutation for each dimension
        sample = np.random.rand(n_points)
        # Scale to [0, 1) and add jitter
        sample = (np.arange(n_points) + sample) / n_points
        samples.append(sample)
    return samples

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h using the provided initial_h_values or create a sinusoidal one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)

    # Phase 1: Generate diverse initial candidates using LHS
    start_time = time.time()
    initial_candidates = []
    lhs_samples = latin_hypercube_sampling(50, n_points, seed)
    
    # Add the parent solution
    initial_candidates.append(h.copy())
    
    for sample in lhs_samples:
        candidate = sample * 0.8 + 0.1  # Scale to [0.1, 0.9]
        candidate = project_h(candidate, n_points)
        initial_candidates.append(candidate)
    
    # Evaluate all initial candidates
    candidate_scores = []
    for candidate in initial_candidates:
        score = evaluate_h(candidate, dx)
        candidate_scores.append(score)
    
    # Select top candidates
    sorted_indices = np.argsort(candidate_scores)
    top_candidates = [initial_candidates[i] for i in sorted_indices[:20]]
    best_score = candidate_scores[sorted_indices[0]]
    best_h = top_candidates[0].copy()
    
    # Phase 2: Evolution Strategy
    pop_size = 20
    offspring_per_parent = 5
    mutations_per_offspring = 3
    
    while time.time() - start_time < 0.6 * budget_s:
        # Create offspring
        offspring = []
        for parent in top_candidates:
            for _ in range(offspring_per_parent):
                # Mutate by slightly perturbing some elements
                child = parent.copy()
                for _ in range(mutations_per_offspring):
                    idx = np.random.randint(0, n_points)
                    # Small random perturbation
                    delta = np.random.normal(0, 0.05)
                    child[idx] += delta
                child = project_h(child, n_points)
                offspring.append(child)
        
        # Evaluate offspring
        all_candidates = top_candidates + offspring
        all_scores = []
        for candidate in all_candidates:
            score = evaluate_h(candidate, dx)
            all_scores.append(score)
        
        # Select top candidates
        sorted_indices = np.argsort(all_scores)
        top_candidates = [all_candidates[i] for i in sorted_indices[:pop_size]]
        current_best_score = all_scores[sorted_indices[0]]
        if current_best_score < best_score:
            best_score = current_best_score
            best_h = top_candidates[0].copy()

    # Phase 3: Gradient descent refinement
    h = best_h.copy()
    patience = 0
    max_patience = 5000
    learning_rate = 0.01
    step_size = 0.05
    
    while time.time() - start_time < budget_s:
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient w.r.t. h
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        update_step = np.sign(grad) * step_size
        h -= learning_rate * update_step
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_score:
            best_h = h.copy()
            best_score = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            step_size *= 0.95
            patience = 0
    
    return (best_h, best_score, n_points)
