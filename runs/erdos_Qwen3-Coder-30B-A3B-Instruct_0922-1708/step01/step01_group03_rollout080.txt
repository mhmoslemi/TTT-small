import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45  # odd for symmetry
    start_time = time.time()
    
    # Initialize particles with symmetric structure
    swarm_size = 30
    particles = []
    velocities = []
    
    # Initialize swarm with symmetrical functions
    for _ in range(swarm_size):
        h = initialize_h(n_points)
        # Ensure symmetry: h[i] = h[n_points-1-i] for i < n_points//2
        mid = n_points // 2
        for i in range(mid):
            h[n_points-1-i] = h[i]
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    cognitive_weight = 2.0
    social_weight = 2.0
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Track best particle
    best_particle = None
    best_c5 = float('inf')
    
    # PSO main loop
    iter_count = 0
    max_iter = 5000
    while iter_count < max_iter and time.time() - start_time < budget_s * 0.8:
        iter_count += 1
        
        for i, particle in enumerate(particles):
            # Enforce symmetry and constraints
            particle = project_h(particle, n_points)
            # Ensure middle values are equal in symmetric representation
            mid = n_points // 2
            for j in range(mid):
                particle[n_points-1-j] = particle[j]
            
            # Evaluate
            c5 = evaluate_c5(particle)
            
            # Update personal best
            if c5 < best_c5:
                best_c5 = c5
                best_particle = particle.copy()
            
            particles[i] = particle
            
        # Update velocities and positions
        for i in range(swarm_size):
            particle = particles[i]
            velocity = velocities[i]
            
            # Rand values for PSO update
            r1 = random.random()
            r2 = random.random()
            
            # Update velocity
            if best_particle is not None:
                cognitive = cognitive_weight * r1 * (particle - particle)
                social = social_weight * r2 * (best_particle - particle)
                velocity = inertia_weight * velocity + cognitive + social
            
            # Apply velocity limits
            velocity = np.clip(velocity, -velocity_max, velocity_max)
            velocities[i] = velocity
            
            # Update particle
            particle = particle + velocity
            particles[i] = particle

    # Refinement with gradient descent
    if best_particle is not None:
        h_refine = best_particle.copy()
        best_c5_refined = best_c5
        learning_rate = 0.05
        max_iter_gd = 5000
        iter_gd = 0
        no_improvement = 0
        
        while iter_gd < max_iter_gd and time.time() - start_time < budget_s * 0.95:
            iter_gd += 1
            
            # Compute gradient
            n = len(h_refine)
            dx = 2.0 / n
            # Finding maximum shift
            conv = np.correlate(h_refine, 1 - h_refine, mode="full")
            max_idx = np.argmax(conv)
            shift = max_idx - (n - 1)
            
            # Approximate gradient: finite difference
            eps = 1e-4
            grad = np.zeros(n)
            for i in range(n):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n)
                c5_plus = evaluate_c5(h_plus)
                h_minus = h_refine.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n)
                c5_minus = evaluate_c5(h_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with projected gradient descent
            h_new = h_refine - learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            # Ensure symmetry
            mid = n_points // 2
            for j in range(mid):
                h_new[n_points-1-j] = h_new[j]
            
            new_c5 = evaluate_c5(h_new)
            
            if new_c5 < best_c5_refined:
                best_c5_refined = new_c5
                h_refine = h_new
                no_improvement = 0
            else:
                no_improvement += 1
            
            if no_improvement > 1000:
                break
                
            if iter_gd % 100 == 0:
                if learning_rate > 1e-6:
                    learning_rate *= 0.9

        # Final check
        if best_c5_refined < best_c5:
            best_c5 = best_c5_refined
            best_particle = h_refine.copy()
    
    return (best_particle, best_c5, n_points)
