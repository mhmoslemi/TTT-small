import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def find_worst_shift(h):
    """Find shift that gives maximum overlap."""
    n = len(h)
    conv = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(conv)
    shift = max_idx - (n - 1)
    return shift

def compute_exact_gradient(h, shift, n_points):
    """Compute exact gradient for the worst shift."""
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    for i in range(n_points):
        idx = (i + shift) % n_points
        grad[i] = (1 - h[idx]) * dx
    return grad

def line_search(h, grad, max_step, n_points):
    """Find optimal step size using line search."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    try:
        res = optimize.minimize_scalar(objective, bounds=(0, max_step), method='bounded')
        return res.x if res.success else max_step
    except:
        return max_step

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45 
    h = initialize_h(n_points)
    
    # Set up parameters
    learning_rate = 0.1
    max_no_improvement = 1000
    no_improvement = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Precompute dx
    dx = 2.0 / n_points
    
    iteration = 0
    while time.time() - start_time < budget_s and no_improvement < max_no_improvement:
        iteration += 1
        
        # Find the worst shift
        worst_shift = find_worst_shift(best_h)
        
        # Compute exact gradient for this shift
        gradient = compute_exact_gradient(best_h, worst_shift, n_points)
        
        # Perform line search to find best step
        step_size = line_search(best_h, gradient, learning_rate, n_points)
        
        # Update h
        h_new = best_h - step_size * gradient
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improvement = 0
            
            # Adapt learning rate
            learning_rate = min(0.5, learning_rate * 1.2)
        else:
            no_improvement += 1
            
            # Adapt learning rate
            learning_rate = max(1e-6, learning_rate * 0.5)
            
    return (best_h, best_c5, n_points)
