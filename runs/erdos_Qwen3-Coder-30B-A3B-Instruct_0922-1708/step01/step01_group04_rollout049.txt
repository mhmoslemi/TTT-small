import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a reasonable starting construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        # Create a smoother version with some variation
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(2 * np.pi * x) * np.exp(-x**2 / 0.2)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize to have integral = 1
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)

    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def smooth_h(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth

    def perturb(h, delta=0.05):
        h_new = h.copy()
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def project_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Perform a few initial gradient steps to smooth the function
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)

    # Fast optimization approach: simulated annealing hybrid with gradient steps
    T = 1.0
    cooling_rate = 0.9999
    max_no_improvement = 100
    no_improvement = 0

    while time.time() - start_time < budget_s and no_improvement < max_no_improvement:
        # Try several perturbations in one iteration
        for _ in range(50):
            h_trial = perturb(best_h, delta=T * 0.2)
            h_trial = smooth_h(h_trial, window=3)
            c5_trial = evaluate(h_trial)
        
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement = 0
                break
            else:
                # Accept with probability based on temperature
                if np.random.rand() < np.exp(-(c5_trial - best_c5) / T):
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    no_improvement = 0
                    break
        else:
            no_improvement += 1

        # Cool down temperature
        T *= cooling_rate

        # Occasionally increase temperature if stagnated
        if no_improvement % 20 == 0 and T < 0.1:
            T *= 1.5

        if time.time() - start_time > budget_s * 0.95:
            break
            
    # Final smoothing
    best_h = smooth_h(best_h, window=5)
    best_h = project_feasible(best_h)
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
