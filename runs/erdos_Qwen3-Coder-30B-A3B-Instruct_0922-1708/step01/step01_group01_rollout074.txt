import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def dual_optimization_phase(initial_h, n_points, budget_s, dx):
    """Use a variant of alternating optimization with dual variables"""
    start_time = time.time()
    
    # Initialize
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Dual variables for each shift
    lambda_k = np.ones(2 * n_points - 1)  # length of full correlation
    
    # Parameters
    learning_rate = 0.1
    max_iter = 10000
    patience = 0
    max_patience = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Solve for h with fixed lambdas (skip solving the dual for efficiency)
        # Just do a local gradient-based update to improve C5
        # Evaluate correlation
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute sign of gradient more carefully
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update h
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        # Current evaluation
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.7
            patience = 0
    
    return best_h, best_c5

def gradient_descent_refinement(initial_h, n_points, budget_s, dx):
    """Perform gradient descent refinement with adaptive steps"""
    start_time = time.time()
    h = initial_h.copy()
    max_iter = 5000
    learning_rate = 0.01
    step_size = 0.1
    patience = 0
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > 0.3 * budget_s:
            break
            
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)

            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)

            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update h
        h = h - learning_rate * np.sign(grad) * step_size
        h = project_h(h, n_points)
        
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > 1000:
                break
        if patience > 0 and patience % 1000 == 0:
            step_size *= 0.95
    
    return best_h, best_c5
