import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother triangle-like function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h, worst_shift):
    """Compute gradient for the worst shift."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        # Compute gradient for shift k = worst_shift
        # The argument of h in the overlap is h[i] * (1 - h[i+k])
        # So d/dh[i] of this term when i+k is in bounds is -1 * (1 - h[i+k])
        k = worst_shift
        if 0 <= (i + k) % n < n:
            grad[i] = -1 * (1 - h[(i + k) % n])
    return grad

def line_search(h, grad, worst_shift, eta, n_points):
    """Perform line search to find optimal step size."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Try several alpha values to find good step size
    alphas = np.linspace(0, eta, 100)
    best_alpha = 0
    best_obj = objective(0)
    
    for alpha in alphas:
        obj = objective(alpha)
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    
    return best_alpha, best_obj

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function - uses gradient-based method with adaptive learning."""
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45  # Fixed for speed and consistent performance
    h = initialize_h(n_points)
    eta = 0.1  # Initial learning rate
    max_iter = 10000
    iter_no_improvement = 0
    max_no_improvement = 1000
    prev_c5 = evaluate_c5(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        if iter_no_improvement > max_no_improvement:
            break
            
        # Find worst shift
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute gradient for this shift
        grad = compute_gradient(h, shift)
        
        # Perform line search
        alpha, new_c5 = line_search(h, grad, shift, eta, n_points)
        
        # Update h
        h = h - alpha * grad
        h = project_h(h, n_points)
        new_c5 = evaluate_c5(h)
        
        # Update learning rate and iteration tracking
        if new_c5 < prev_c5:
            prev_c5 = new_c5
            iter_no_improvement = 0
            eta = min(eta * 1.1, 0.5)
        else:
            iter_no_improvement += 1
            eta = max(eta / 2, 1e-6)
            
    # Final evaluation
    final_c5 = evaluate_c5(h)
    return (h, final_c5, n_points)
