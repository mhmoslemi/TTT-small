import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Use previous construction if available
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: smooth symmetric function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h, shift, n_points):
    """Compute gradient for the specific shift that causes maximum overlap."""
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    for i in range(n_points):
        # Gradient at position i for shift k
        j = (i + shift) % n_points
        grad[i] = (1 - h[j]) * dx
    return grad

def line_search(h, grad, max_shift, n_points, eta):
    """Find optimal step size using line search."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Bracket the minimum
    a, b = 0.0, eta
    fa = objective(0.0)
    fb = objective(b)
    
    # If b is not a descent direction, adjust
    while fb >= fa and b < eta:
        b *= 2
        fb = objective(b)
    
    # Now perform line search using golden section
    phi = (1 + np.sqrt(5)) / 2
    resphi = 2 - phi
    
    # Initialize
    x1 = a
    x2 = b
    fx1 = objective(x1)
    fx2 = objective(x2)
    
    # Search for minimum
    while abs(x2 - x1) > 1e-8:
        x3 = x1 + resphi * (x2 - x1)
        x4 = x1 + phi * (x2 - x1)
        fx3 = objective(x3)
        fx4 = objective(x4)
        
        if fx3 < fx4:
            x2 = x4
            fx2 = fx4
        else:
            x1 = x3
            fx1 = fx3
    
    # Return optimal step size
    alpha = (x1 + x2) / 2
    return min(alpha, eta)

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function implementing gradient descent with line search."""
    np.random.seed(seed)
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    eta = 0.1  # Initial learning rate
    eta_min = 1e-6
    eta_max = 0.5
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s:
        # Find shift with maximum overlap
        conv = np.correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute gradient for that shift
        grad = compute_gradient(best_h, shift, n_points)
        
        # Line search to find optimal step size
        alpha = line_search(best_h, grad, shift, n_points, eta)
        
        # Update h
        h_new = best_h - alpha * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        
        # Accept improvement or adjust learning rate
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
            eta = min(eta * 1.2, eta_max)  # Increase learning rate
        else:
            no_improvement_count += 1
            eta = max(eta / 2, eta_min)  # Decrease learning rate
            
        if no_improvement_count > max_no_improvement:
            break
    
    # Final projection and evaluation
    final_h = project_h(best_h, n_points) 
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
