import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # PSO parameters
    n_particles = 50
    max_iter = 10000
    w_init = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(n_particles):
        # Random initialization
        h_candidate = np.random.rand(n_points) * 0.5 + 0.25
        h_candidate = project_h(h_candidate)
        particles.append(h_candidate.copy())
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
        pbest.append(h_candidate.copy())
        pbest_values.append(compute_c5(h_candidate))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iteration
    iteration = 0
    no_improvement_count = 0
    no_improvement_limit = 1000
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        w = w_init - (w_init - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and projection
            particles[i] = project_h(particles[i])
            
            # Evaluate
            current_value = compute_c5(particles[i])
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
            
            # Update global best
            if current_value < gbest_value:
                gbest = particles[i].copy()
                gbest_value = current_value
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        iteration += 1
        
        if no_improvement_count >= no_improvement_limit:
            break
    
    # Final check
    final_c5 = compute_c5(gbest)
    
    return (gbest.tolist(), final_c5, len(gbest))
