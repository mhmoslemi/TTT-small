from scipy.optimize import minimize
from scipy.signal import correlate
import time

        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap

    def construct_h_from_segments(segments, n_total):
        """Reconstruct full h from segment averages with smoothing"""
        # Create uniform segments
        segment_width = n_total // len(segments)
        h = np.zeros(n_total)
        for i, c in enumerate(segments):
            start_idx = i * segment_width
            end_idx = start_idx + segment_width
            h[start_idx:end_idx] = c
        
        # Apply Gaussian smoothing for continuity
        if len(segments) > 1:
            from scipy.ndimage import gaussian_filter1d
            h = gaussian_filter1d(h, sigma=1.0)
        
        # Normalize to integral = 1
        h = h * (n_total / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def optimize_segments(segments, n_total):
        """Optimize smooth segment representation"""
        def objective(c):
            h = construct_h_from_segments(c, n_total)
            return evaluate(h)
        
        def constraint_sum(c):
            return np.sum(c) - len(c) / 2

        # Set bounds for each segment
        bounds = [(0, 1) for _ in range(len(segments))]
        
        # Define constraints
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimize using SLSQP
        try:
            result = minimize(objective, segments, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
            if result.success:
                return result.x
        except:
            pass
        return segments
    
    # Start with better optimized segments
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    # Try different segmentation levels for optimization
    refinement_levels = [7, 14, 21, 28, 35, 42, 56, 63]  # Powers of 7 to keep it manageable
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0

    for level in refinement_levels:

            break
            
        # Initialize segments
        segments = np.full(level, 1.0)  # Start with uniform distribution
        segments = segments * (level / 2) / np.sum(segments)
        
        # Optimize segments
        optimized_segments = optimize_segments(segments, n_points)
        
        # Reconstruct h
        h_candidate = construct_h_from_segments(optimized_segments, n_points)
        c5_candidate = evaluate(h_candidate)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate

            no_improvement_count += 1

                    break

                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
                    no_improvement_count = 0
        
        if no_improvement_count > 50 or (time.time() - start_time > budget_s - 10):
            break
    
    return (best_h, best_c5, n_points)
