import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Optimizer parameters
    max_iter = 10000
    no_improvement_limit = 500
    lr = 0.01
    
    # Objective function
    def compute_C(h_vals):
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # Perturb h
    def perturb_h(h_vals, eps=1e-5):
        # Small random perturbation
        pert = np.random.normal(0, eps, len(h_vals))
        return np.clip(h_vals + pert, 0, 1)
    
    # Optimization loop
    best_h = h.copy()
    best_C = compute_C(h)
    no_improvement_count = 0
    iter_count = 0
    previous_C = best_C
    
    # For sinusoidal-based approach
    m = 10  # Number of cosine modes
    a = np.random.rand(m+1) * 0.5
    a[0] = 0.5  # Ensures integral constraint
    a[1::2] = 0  # Even symmetry
    
    def compute_h_from_a(a_coeffs):
        x = np.linspace(0, 2, n_points)
        h = a_coeffs[0] * np.ones_like(x)
        for k in range(1, len(a_coeffs)):
            h += a_coeffs[k] * np.cos(np.pi * k * x / 2)
        return np.clip(h, 0, 1)
    
    def evaluate_from_a():
        h_vals = compute_h_from_a(a)
        # Normalize
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return compute_C(h_vals)
    
    while time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        # Optimization step
        h_vals = compute_h_from_a(a)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        
        C = compute_C(h_vals)
        
        # Calculate gradient using finite differences
        grad = np.zeros_like(a)
        eps = 1e-5
        for i in range(len(a)):
            if abs(a[i]) < 1e-8: continue # Skip zero gradients for numerical stability
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus = compute_h_from_a(a_plus)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C) / eps
        
        # Update parameters
        a -= lr * grad
        a = np.clip(a, 0, 1)
        a[0] = 0.5  # Keep average value constraint
        
        # Evaluate
        new_C = evaluate_from_a()
        
        if new_C < best_C:
            best_C = new_C
            best_h = h_vals.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > no_improvement_limit:
            lr *= 0.95
            no_improvement_count = 0
            
        previous_C = new_C
    
    # Final binary refinement step with local search
    h_refined = best_h.copy()
    C_final = best_C
    
    def binary_refinement(h_vals, max_flips=2000):
        h_bin = (h_vals > 0.5).astype(float)
        # Adjust to match target sum
        current_sum = np.sum(h_bin)
        if current_sum != target_sum:
            # Flip the most beneficial bits
            diffs = h_vals - 0.5
            indices = np.argsort(np.abs(diffs))[:int(abs(current_sum - target_sum))]
            for i in indices:
                h_bin[i] = 1 - h_bin[i]
        
        # Binary local search
        current_C = compute_C(h_bin)
        best_C = current_C
        best_h = h_bin.copy()
        
        for _ in range(max_flips):
            improved = False
            for i in range(n_points):
                new_h = h_bin.copy()
                new_h[i] = 1 - new_h[i]
                new_C = compute_C(new_h)
                if new_C < best_C:
                    best_C = new_C
                    best_h = new_h.copy()
                    improved = True
            if not improved:
                break
        return best_h, best_C
    
    h_refined, C_final = binary_refinement(h_refined, 1000)
    
    return (h_refined.tolist(), C_final, n_points)
