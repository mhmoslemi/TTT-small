        particles.append(p_values.copy())
        velocities.append(np.random.rand(n_nodes) * 0.1 - 0.05)  # Small random velocities

    # Keep track of best particle
    g_best_particle = None
    g_best_score = float('inf')

        g_best_score = final_c5
