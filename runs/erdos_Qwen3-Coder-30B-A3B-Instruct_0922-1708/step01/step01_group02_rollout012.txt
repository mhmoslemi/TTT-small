import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        return project_h(h)
    
    # Parameters
    pop_size = 100
    generations = 0
    max_generations = 10000
    
    # Create initial population
    population = [create_individual() for _ in range(pop_size)]
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Genetic Algorithm
    mutation_rate = 0.2
    crossover_rate = 0.7
    elitism_size = pop_size // 10
    
    while generations < max_generations and (time.time() - start_time) < budget_s - 1:
        # Evaluate population
        fitness = [compute_c5(ind) for ind in population]
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fit[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fit[0]
        
        # Selection (roulette wheel)
        def select_parent():
            # Linear scaling to improve selection pressure
            scaled_fitness = [1.0 / (1e-10 + f) for f in sorted_fit]
            total_fit = sum(scaled_fitness)
            if total_fit <= 0:
                return sorted_pop[np.random.randint(0, pop_size)]
            probabilities = [f / total_fit for f in scaled_fitness]
            idx = np.random.choice(pop_size, p=probabilities)
            return sorted_pop[idx].copy()
        
        # Create new population
        new_pop = sorted_pop[:elitism_size]  # Elitism
        
        # Generate offspring
        while len(new_pop) < pop_size:
            if np.random.rand() < crossover_rate:
                parent1 = select_parent()
                parent2 = select_parent()
                
                # Crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = select_parent()
                child2 = select_parent()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    for i in range(n_points):
                        if np.random.rand() < 1.0 / np.sqrt(generations + 1):
                            noise = np.random.normal(0, 0.1)
                            child[i] += noise
                    child = project_h(child)
                new_pop.append(child)
                if len(new_pop) >= pop_size:
                    break
        
        population = new_pop[:pop_size]
        generations += 1
        
        # Adapt mutation rate
        mutation_rate = max(0.01, mutation_rate * 0.999)
    
    # Final quality check
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
