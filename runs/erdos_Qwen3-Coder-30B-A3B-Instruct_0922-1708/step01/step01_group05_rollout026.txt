import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO Parameters
    n_particles = 100
    w = 0.7   # Inertia weight
    c1 = 1.5  # Cognitive parameter
    c2 = 1.5  # Social parameter
    max_iter = 500
    
    # Initialize particle positions and velocities
    positions = np.random.rand(n_particles, n_points)
    velocities = np.random.rand(n_particles, n_points) * 0.1
    
    # Clip to bounds
    positions = np.clip(positions, 0, 1)
    
    # Initialize personal best
    pbest_positions = positions.copy()
    pbest_values = np.full(n_particles, float('inf'))
    
    # Global best
    gbest_value = float('inf')
    gbest_position = None
    
    # Helper to evaluate function
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def evaluate_particle(pos):
        h_vals = np.clip(pos, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))  # Rescale to integral constraint
        return compute_C(h_vals)
    
    # PSO loop
    iter_count = 0
    while time() - start_time < budget_s - 100 and iter_count < max_iter:
        iter_count += 1
        # Evaluate all particles
        for i in range(n_particles):
            val = evaluate_particle(positions[i])
            if val < pbest_values[i]:
                pbest_values[i] = val
                pbest_positions[i] = positions[i].copy()
            
            if val < gbest_value:
                gbest_value = val
                gbest_position = positions[i].copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - positions[i]) +
                            c2 * r2 * (gbest_position - positions[i]))
            positions[i] += velocities[i]
            
            # Keep in bounds
            positions[i] = np.clip(positions[i], 0, 1)
    
    # Refinement with gradient descent on the global best
    best_h = gbest_position.copy()
    
    # Rescale to make sum equal to target_sum
    best_h = best_h * (target_sum / np.sum(best_h))
    
    # Replace step size and iteration limit
    lr = 0.01
    max_refine_iter = 1000
    best_C = compute_C(best_h)
    no_improvement_count = 0
    iter_count = 0
    
    while time() - start_time < budget_s - 1 and iter_count < max_refine_iter:
        iter_count += 1
        # Compute gradient numerically
        eps = 1e-5
        grad = np.zeros_like(best_h)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            
            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (target_sum / np.sum(h_minus))
            C_minus = compute_C(h_minus)
            
            grad[i] = (C_plus - C_minus) / (2 * eps)
        
        # Gradient update
        best_h -= lr * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        
        # Evaluate
        new_C = compute_C(best_h)
        if new_C < best_C:
            best_C = new_C
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                lr *= 0.5
                no_improvement_count = 0
                
    return (best_h.tolist(), best_C, n_points)
