import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    n_dims = n_points
    max_iter = 5000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    for _ in range(n_particles):
        # Initialize with random values in [0,1]
        particle = np.random.rand(n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (target_sum / np.sum(particle))  # Rescale
        particles.append(particle.copy())
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
    
    # Global best
    global_best = h.copy()
    global_best_C = float('inf')
    
    # Optimization loop
    iter_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Evaluate current particle
            h_vals = particles[i]
            corr = np.correlate(h_vals, 1-h_vals, mode="full")
            corr *= dx
            C = np.max(np.abs(corr))
            
            # Update personal best
            if C < global_best_C:
                global_best_C = C
                global_best = h_vals.copy()
            
            # Update velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (h_vals - particles[i]) + 
                            social * r2 * (global_best - particles[i]))
            
            # Project velocities to valid range
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            
            # Update particles
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))  # Keep integral constraint
            
        iter_count += 1
    
    # Do gradient refinement on the best particles
    # Pick top 10% to refine
    n_refine = max(1, n_particles // 10)
    
    # Sort particles by their C value
    particle_Cs = []
    for i in range(n_particles):
        h_vals = particles[i]
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        C = np.max(np.abs(corr))
        particle_Cs.append((C, i))
    particle_Cs.sort()
    
    # Refine top particles
    best_h = global_best.copy()
    best_C = global_best_C
    
    for _, idx in particle_Cs[:n_refine]:
        h_refine = particles[idx].copy()
        lr = 0.01
        no_improvement_count = 0
        
        # Gradient descent for refinement
        for _ in range(200):
            if time() - start_time > budget_s - 1:
                break
                
            # Evaluate
            corr = np.correlate(h_refine, 1-h_refine, mode="full")
            corr *= dx
            C_current = np.max(np.abs(corr))
            
            # Estimate gradient by finite diff
            eps = 1e-5
            grad = np.zeros_like(h_refine)
            for j in range(n_points):
                h_plus = h_refine.copy()
                h_plus[j] += eps
                h_plus = h_plus * (target_sum / np.sum(h_plus))
                corr_plus = np.correlate(h_plus, 1-h_plus, mode="full")
                corr_plus *= dx
                C_plus = np.max(np.abs(corr_plus))
                grad[j] = (C_plus - C_current) / eps
            
            # Update
            h_refine -= lr * grad
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (target_sum / np.sum(h_refine))
            
            # Update best if improved
            corr = np.correlate(h_refine, 1-h_refine, mode="full")
            corr *= dx
            C_new = np.max(np.abs(corr))
            
            if C_new < best_C:
                best_C = C_new
                best_h = h_refine.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                if no_improvement_count >= 10:
                    lr *= 0.5
                    no_improvement_count = 0
    
    return (best_h.tolist(), best_C, n_points)
