import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a triangular distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to higher resolution
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        # Ensure integral constraint
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create triangular-like distribution
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 - np.abs(x - 1))  # Triangular
        h = h * (n_points / 2) / np.sum(h)
    
    # Clamp to [0,1] and rescale
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    
    # Adam optimizer variables
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        t += 1
        
        # Compute correlation and C5
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient analytically
        # For each position, we compute the gradient using the chain rule
        # and the fact that C5 = max_k (h * (1-h_k)), where h_k is h shifted by k
        grad = np.zeros_like(h)
        max_k = np.argmax(corr)
        
        # Forward pass to compute gradient contributions
        if max_k < len(h):
            # Case 1: Peak correlation is in the forward direction (not wrapped)
            # This means h[i] is correlated with (1-h[i+max_k]) 
            for i in range(len(h)):
                j = i + max_k
                if j < len(h):
                    grad[i] += 1.0 * dx  # Since h[i] contributes to K*sum(h[i](1-h[i+k]))
        else:
            # Case 2: Peak is in the backward, possibly wrapped part
            # This happens when using full mode correlation, and the peak appears in the earlier portion
            # So, max_k is actually the negative index offset here
            for i in range(len(h)):
                j = i - (max_k - len(h) + 1)
                if j >= 0 and j < len(h):
                    grad[i] += 1.0 * dx  # Contribution from (1-h[j])
        
        # Update with Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 1000 == 0:
            lr *= 0.5
    
    return (best_h.tolist(), best_c5, n_points)
