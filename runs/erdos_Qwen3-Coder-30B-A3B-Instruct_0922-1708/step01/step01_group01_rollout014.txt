import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def latin_hypercube_sampling(n_points, n_samples):
    """Generate Latin Hypercube Samples for initial population"""
    samples = []
    for _ in range(n_samples):
        # Generate sample points in [0, 1] with Latin hypercube property
        u = np.random.rand(n_points)
        # Shuffle each dimension
        np.random.shuffle(u)
        # Scale to [0, 2) and add slight randomness
        x = u * 2
        # Add small noise to avoid degenerate cases
        x += np.random.normal(0, 0.01, n_points)
        # Clip to [0, 2]
        x = np.clip(x, 0, 2)
        # Then sample from a sinusoidal-like shape
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        samples.append(h)
    return samples

    best_h = h.copy()

    pop.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(p, dx) for p in pop]

    best_c5 = min(fitness)

    # Phase 2: Evolutionary strategy with gradient phases
    n_elites = 20
    n_offspring = 5
    patience = 0
    max_patience = 1000
    
    while time.time() - start_time < 0.8 * budget_s:
        # Mutate elites
        new_pop = []
        for elite in population_h:
            for _ in range(n_offspring):
                # Create mutated version
                child = elite.copy()
                # Small random perturbations
                perturb = np.random.normal(0, 0.05, n_points)
                child += perturb
                child = project_h(child, n_points)
                new_pop.append(child)
        
        # Evaluate all
        population_h.extend(new_pop)
        fitness = [evaluate_h(p, dx) for p in population_h]
        
        # Keep best ones
        top_indices = np.argsort(fitness)[:n_elites]
        population_h = [population_h[i] for i in top_indices]

            best_h = population_h[np.argmin(fitness)].copy()
            patience = 0

    h = best_h.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 1000

        # Measure the worst overlap and calculate gradient

        eps = 1e-5

    return (best_h, best_c5, n_points)
