import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Helper function to compute C5
    def compute_C5(h_vals):
        h_vals = np.array(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # Helper function to make sure h stays in [0,1] and integrates to 1
    def project_h(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_sum = np.sum(h_vals) * dx
        if h_sum > 0:
            h_vals = h_vals * target_sum / h_sum
        return h_vals
    
    # PSO parameters
    n_particles = 100
    n_dims = n_points
    max_iter = 5000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest = None
    gbest_val = float('inf')
    
    # Initialize particles with some randomness around the starting h
    for i in range(n_particles):
        h_new = project_h(h + np.random.normal(0, 0.05, n_points))
        particles.append(h_new.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h_new.copy())
        pbest_vals.append(compute_C5(h_new))
        if pbest_vals[-1] < gbest_val:
            gbest_val = pbest_vals[-1]
            gbest = h_new.copy()
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (pbest[i] - particles[i]) + 
                           social * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Enforce integral constraint
            particles[i] = project_h(particles[i])
            
            # Evaluate
            current_val = compute_C5(particles[i])
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest_val = current_val
                    gbest = particles[i].copy()
    
    # Refinement with gradient descent on best particles
    refine_particles = 10  # Take top 10 particles
    top_indices = np.argsort(pbest_vals)[:refine_particles]
    
    best_h = gbest.copy()
    best_C = gbest_val
    
    for i in top_indices:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient descent on particle
        h_refine = pbest[i].copy()
        lr = 0.01
        prev_C = pbest_vals[i]
        stagnation_count = 0
        
        for step in range(200):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current C5
            current_C = compute_C5(h_refine)
            
            if current_C < best_C:
                best_C = current_C
                best_h = h_refine.copy()
                
            # Compute gradient
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_refine.copy()
                h_plus[j] += eps
                h_plus = project_h(h_plus)
                C_plus = compute_C5(h_plus)
                grad[j] = (C_plus - current_C) / eps
            
            # Update
            h_refine -= lr * grad
            h_refine = project_h(h_refine)
            
            # Check for stagnation
            if abs(current_C - prev_C) < 1e-8:
                stagnation_count += 1
                if stagnation_count > 10:
                    lr *= 0.5
                    stagnation_count = 0
            else:
                stagnation_count = 0
                
            prev_C = current_C
    
    return (best_h.tolist(), best_C, n_points)
