    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def project_to_feasible(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals
    
    def optimize_with_pso():
        # PSO parameters
        n_particles = 100
        n_iter = 5000
        w = 0.7
        c1 = 1.5
        c2 = 1.5
        
        # Initialize particles
        particles = []
        velocities = []
        best_positions = []
        best_scores = []
        
        for _ in range(n_particles):
            pos = np.random.rand(n_points) * 2  # positions in [0,2]
            vel = np.random.rand(n_points) * 0.01  # small initial velocities
            particles.append(np.copy(pos))
            velocities.append(np.copy(vel))
            best_positions.append(np.copy(pos))
            best_scores.append(float('inf'))
        
        # PSO main loop
        global_best_score = float('inf')
        global_best_h = None
        
        for iteration in range(n_iter):
            if time.time() - start_time > budget_s - 2:
                break

                # Update particle (only positions)
                velocities[i] = (w * velocities[i] + 
                                c1 * np.random.random() * (best_positions[i] - particles[i]) +
                                c2 * np.random.random() * (global_best_h - particles[i]))
                particles[i] += velocities[i]
                
                # Keep positions within [0,2]
                particles[i] = np.clip(particles[i], 0, 2)
                
                # Interpolate h values from positions
                # Since h is piecewise linear, use a proxy: direct interpolation
                # But we simplify by using a fixed basis for better integration control
                
                # Evaluate each particle
                h_vals = np.interp(np.linspace(0, 2, n_points), particles[i], 
                                  np.random.rand(len(particles[i])))  # Interpolation proxy
                h_vals = project_to_feasible(h_vals)
                score = compute_C(h_vals)
                
                if score < best_scores[i]:
                    best_scores[i] = score
                    best_positions[i] = particles[i].copy()
                    
                if score < global_best_score:
                    global_best_score = score
                    global_best_h = particles[i].copy()
                    
        return global_best_h, global_best_score

        stagnation_count = 0

            if time.time() - start_time > budget_s - 2:
                break

            grad = np.zeros_like(h)
            eps = 1e-5

                    stagnation_count = 0
