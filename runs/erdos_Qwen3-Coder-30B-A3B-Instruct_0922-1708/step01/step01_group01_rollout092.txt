import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using segment-based search with gradient refinement"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize with given starting point or a simple construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Simple construction
        h = np.ones(n_points) * 0.5
    
    # Segment-based optimization: use fewer degrees of freedom
    m = 5  # Number of segments
    segment_values = [0.0, 0.5, 1.0]
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute segment boundaries
    segment_length = n_points // m
    remainder = n_points % m
    
    # Generate all possible combinations of segment values
    start_time = time.time()
    candidates = list(product(segment_values, repeat=m))
    candidate_count = len(candidates)
    
    for i, seg_vals in enumerate(candidates):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Construct h from segments
        h_seg = np.zeros(n_points)
        start_idx = 0
        for j, val in enumerate(seg_vals):
            # Distribute points among segments
            end_idx = start_idx + segment_length + (1 if j < remainder else 0)
            h_seg[start_idx:end_idx] = val
            start_idx = end_idx
        
        # Normalize
        h_seg = project_h(h_seg, n_points)
        
        # Evaluate
        c5 = evaluate_h(h_seg, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_seg.copy()
    
    # Gradient descent refinement
    h = best_h.copy()
    best_h_refined = h.copy()
    best_c5_refined = best_c5
    learning_rate = 0.01
    last_improvement = 0
    patience_limit = 1000
    
    while time.time() - start_time < budget_s - 5:
        # Compute gradient
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for idx in range(n_points):
            h_plus = h.copy()
            h_plus[idx] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[idx] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[idx] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h = h - learning_rate * grad
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5_refined:
            best_c5_refined = new_c5
            best_h_refined = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            if last_improvement > patience_limit:
                break
        if last_improvement % 1000 == 0:
            learning_rate *= 0.5
    
    return (best_h_refined, best_c5_refined, n_points)
