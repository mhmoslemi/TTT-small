import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a double triangular pulse
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def gradient_descent_optimize(h, n_points, time_budget_s):
    """Perform gradient descent optimization with exact gradient computation."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Parameters
    eta = 0.1
    eta_min = 1e-6
    eta_max = 0.5
    max_iter = 10000
    patience = 1000
    iter_no_improvement = 0
    
    def find_worst_shift(h):
        """Find shift that gives maximum overlap."""
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        # Take max in absolute value correctly (shift between -n+1 and n-1)
        max_idx = np.argmax(conv)
        return max_idx - (n - 1)
    
    def compute_exact_gradient(h, shift):
        """Compute exact gradient for a specific shift."""
        n = len(h)
        dx = 2.0 / n
        grad = np.zeros(n)
        for i in range(n):
            j = (i + shift) % n
            grad[i] = (1 - h[j]) * dx
        return grad
    
    def line_search(h, grad, shift, initial_alpha=1.0):
        """Find optimal step size via line search."""
        def c5_at_alpha(alpha):
            h_new = h - alpha * grad
            h_new = project_h(h_new, len(h_new))
            return evaluate_c5(h_new)
        
        # We know gradient is with respect to worst shift.
        # Try to find good alpha using bracketed search.
        # Start at 1.0 and decrease
        alpha = initial_alpha
        c5_current = evaluate_c5(h)
        c5_new = c5_at_alpha(alpha)
        
        # If we're not improving, try smaller alpha
        while c5_new >= c5_current and alpha > 1e-8:
            alpha /= 2
            c5_new = c5_at_alpha(alpha)
        
        return alpha if c5_new < c5_current else 0.0
    
    iteration = 0
    while (time.time() - start_time < time_budget_s and 
           iteration < max_iter and 
           iter_no_improvement < patience):
        
        # Find shift that causes maximum overlap
        shift = find_worst_shift(best_h)
        
        # Compute exact gradient for that shift
        gradient = compute_exact_gradient(best_h, shift)
        
        # Line search to find good step size
        alpha = line_search(best_h, gradient, shift)
        
        if alpha > 0:
            # Perform update
            h_new = best_h - alpha * gradient
            h_new = project_h(h_new, n_points)
            new_c5 = evaluate_c5(h_new)
            
            # Accept update if it improves result
            if new_c5 < best_c5:
                best_h = h_new
                best_c5 = new_c5
                iter_no_improvement = 0
                eta = min(eta * 1.2, eta_max)
            else:
                iter_no_improvement += 1
                eta = max(eta / 2, eta_min)
        else:
            # No update direction found, reduce eta
            eta = max(eta / 2, eta_min)
            iter_no_improvement += 1
            
        iteration += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45  # Fixed small size for speed
    
    # Initialize
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Optimize using gradient descent with exact projections
    h_opt, c5_opt = gradient_descent_optimize(h, n_points, budget_s)
    
    # Double-check final result
    h_final = project_h(h_opt, n_points)
    c5_final = evaluate_c5(h_final)
    
    return (h_final, c5_final, n_points)
