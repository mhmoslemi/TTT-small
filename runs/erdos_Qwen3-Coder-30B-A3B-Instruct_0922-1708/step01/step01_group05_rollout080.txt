import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

    n_points = 70

    h = h * (target_sum / np.sum(h))

    n_particles = 100
    n_dims = n_points
    max_iter = 5000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    v_max = 0.1
    
    # PSO initialization
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for _ in range(n_particles):
        # Random particle with h in [0,1]
        h_p = np.random.rand(n_points)
        h_p = np.clip(h_p, 0, 1)
        # Satisfy integral constraint
        h_p = h_p * (target_sum / np.sum(h_p))
        particles.append(h_p.copy())
        velocities.append(np.random.rand(n_points) * 0.01)
        p_best.append(h_p.copy())
        p_best_vals.append(float('inf'))
    
    # Global best
    g_best = h.copy()
    g_best_val = float('inf')
    
    # Helper functions
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def enforce_constraints(h_vals):
        # Ensure h in [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Adjust to satisfy integral constraint
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals
    
    # PSO iteration
    iter_count = 0
    while time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        for i in range(n_particles):
            # Update velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (p_best[i] - particles[i]) + 
                            social * r2 * (g_best - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = enforce_constraints(particles[i])
            
            # Evaluate fitness
            curr_val = compute_C(particles[i])
            
            # Update personal best
            if curr_val < p_best_vals[i]:
                p_best_vals[i] = curr_val
                p_best[i] = particles[i].copy()
                
                # Update global best
                if curr_val < g_best_val:
                    g_best_val = curr_val
                    g_best = particles[i].copy()
        
        # Adjust inertia
        if iter_count % 100 == 0:
            inertia = max(0.4, inertia * 0.99)
        
    # Refinement with gradient descent on best particles
    refinement_particles = p_best[:max(1, n_particles//10)]
    best_h = g_best.copy()
    best_C = g_best_val
    
    # Gradient descent iteration with each top particle
    for i, h_init in enumerate(refinement_particles):
        if time() - start_time > budget_s - 2:
            break
            
        h_refined = h_init.copy()
        lr = 0.01
        iterations = 0
        
        while iterations < 200 and time() - start_time < budget_s - 2:
            # Evaluate current C
            C_current = compute_C(h_refined)
            
            # Compute gradient using finite difference
            grad = np.zeros_like(h_refined)
            eps = 1e-4
            for j in range(len(h_refined)):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = enforce_constraints(h_plus)
                C_plus = compute_C(h_plus)
                grad[j] = (C_plus - C_current) / eps
            
            # Update with gradients
            h_refined -= lr * grad
            h_refined = enforce_constraints(h_refined)
            
            # Check improvement
            C_new = compute_C(h_refined)
            if C_new < C_current:
                if C_new < best_C:
                    best_C = C_new
                    best_h = h_refined.copy()
                iterations += 1
            else:
                lr *= 0.5
                iterations += 1
    
    return (best_h.tolist(), best_C, n_points)
