import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Normalize to integral = 1
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)

    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def smooth(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth

    def local_perturb(h, delta=0.05):
        h_new = h.copy()
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    def simulated_anneal(h, temp, delta=0.1):
        h_new = h.copy()
        n_perturb = max(1, int(n_points * 0.15))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            if np.random.random() < temp:
                h_new[i] += np.random.uniform(-delta, delta)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    def make_segmented_h(c, n_segments, n_points):
        segment_size = n_points // n_segments
        h = np.ones(n_points)
        for i in range(n_segments):
            start = i * segment_size
            end = (i + 1) * segment_size if i < n_segments - 1 else n_points
            h[start:end] = c[i]
        return h

    # Initialize with fixed segment averages
    n_segments = 10
    c = np.ones(n_segments) * 0.5
    c_sum = np.sum(c) * (2 / n_segments)
    c = c * (n_points / 2) / c_sum

    # Start from initial h if available
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)

    # Trust-region parameters
    trust_radius = 0.1
    success_count = 0
    failure_count = 0
    no_improvement_count = 0
    temp = 0.1

    while time.time() - start_time < budget_s and no_improvement_count < 500:
        # Adaptive temperature for simulated annealing
        temp = max(0.001, temp * 0.9995)

        # Local improvement
        for _ in range(50):
            h_trial = local_perturb(best_h, delta=0.02)
            h_trial = smooth(h_trial, window=3)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                break

        # Simulated Annealing
        if np.random.random() < 0.3 or no_improvement_count > 100:
            h_trial = simulated_anneal(best_h, temp)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                
        # Early exit if no improvement recently
        no_improvement_count += 1

        # Occasionally increase trust radius
        if no_improvement_count < 50:
            trust_radius = min(0.5, trust_radius * 1.01)
        else:
            trust_radius = max(0.01, trust_radius * 0.95)

    # Final smoothing and normalize
    best_h = smooth(best_h, window=5)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)

    return (best_h, best_c5, n_points)
