from time import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Constants

    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Fourier basis setup
    m = 10  # Number of cosine modes (even symmetry)
    a = np.random.rand(m) * 0.5  # Coefficients for modes 1 to m-1
    a[0] = 0.5  # Ensure integral constraint
    a[1::2] = 0  # Only even modes for symmetry
    
    # Optimization loop (Gradient-based)
    best_h = h.copy()
    best_C = float('inf')
    no_improvement_count = 0
    lr = 0.01
    
    def compute_h_from_a(a_coeffs):
        # Build h(x) from cosine coefficients
        x = np.linspace(0, 2, n_points)
        h = a_coeffs[0] * np.ones_like(x)
        for k in range(1, len(a_coeffs)):
            h += a_coeffs[k] * np.cos(np.pi * k * x / 2)
        return np.clip(h, 0, 1)
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def evaluate_and_update():
        nonlocal best_C, best_h
        h_vals = compute_h_from_a(a)
        h_vals = h_vals * (target_sum / np.sum(h_vals))  # Rescale
        C = compute_C(h_vals)
        if C < best_C:
            best_C = C
            best_h = h_vals.copy()
        return C
    
    # Evaluate original starting point
    previous_C = evaluate_and_update()
    
    iter_count = 0
    while time() - start_time < budget_s - 1:
        iter_count += 1
        h_vals = compute_h_from_a(a)
        h_vals = h_vals * (target_sum / np.sum(h_vals))  # Rescale to sum constraint

        C = compute_C(h_vals)
        # Find worst shift k
        k_max = np.argmax(np.abs(np.correlate(h_vals, 1-h_vals, mode="full")))
        # Finite diff to compute gradient approx
        eps = 1e-5
        grad = np.zeros_like(a)
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus = compute_h_from_a(a_plus)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C) / eps
        
        # Gradient update
        a -= lr * grad
        a = np.clip(a, 0, 1)
        
        # Rescale to preserve integral condition
        if np.sum(h_vals) > 0:
            a[0] = 0.5  # keep integral constraint
            
        # Adaptive learning rate
        new_C = evaluate_and_update()
        if new_C >= previous_C:
            no_improvement_count += 1
            if no_improvement_count > 100:
                lr *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
        previous_C = new_C
        
    # Binary refinement to improve overlap
    h_final = compute_h_from_a(a)
    h_final = h_final * (target_sum / np.sum(h_final))  # Rescale to target sum
    
    # Threshold the function to get binary steps
    avg_h = np.mean(h_final)
    threshold = np.percentile(h_final, 50)  # All values below this will be 0, above 1  
    h_binary = (h_final > threshold).astype(float)
    
    # Ensure integral constraint for binary version
    actual_sum = np.sum(h_binary)
    target_binary_sum = target_sum
    
    # Adjust binary values to exactly match sum constraint
    if actual_sum < target_binary_sum:
        # Need to set more values to 1
        indices = np.argsort(h_final)
        for i in range(int(target_binary_sum - actual_sum)):
            idx = indices[-1*(i+1)]
            h_binary[idx] = 1.0
    elif actual_sum > target_binary_sum:
        # Need to set more values to 0
        indices = np.argsort(h_final)
        for i in range(int(actual_sum - target_binary_sum)):
            idx = indices[i]
            h_binary[idx] = 0.0
    
    # Final C5 evaluation
    final_C = compute_C(h_binary)
    
    return (h_binary.tolist(), final_C, n_points)
