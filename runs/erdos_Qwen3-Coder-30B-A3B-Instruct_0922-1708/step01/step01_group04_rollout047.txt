import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap

    def perturb(h, delta=0.05):
        # Create a perturbed version
        h_new = h.copy()
        # Perturb about 10% of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def smooth(h, window=3):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth

    def compute_gradient(h, eps=1e-4):
        """
        Compute finite difference gradient of the C5 functional
        """
        grad = np.zeros_like(h)
        base_c5 = evaluate(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad

    def trust_region_update(c, c_grad, trust_radius):
        """
        Update h using trust region method by optimizing over segment averages
        """
        # Compute directional derivative
        dotp = np.dot(c_grad, c_grad)
        if dotp == 0:
            return c.copy()
        
        # Descent direction: steepest descent
        
        # Reflect negative gradient to ensure steps towards decrease
        direction = -c_grad
        
        # Project to trust region center
        if np.linalg.norm(direction) > trust_radius:
            direction = direction * trust_radius / np.linalg.norm(direction)
        
        # Apply update
        new_c = c + direction
        # Keep within feasible bounds
        new_c = np.clip(new_c, 0, 1)
        # Adjust to keep sum = n_points/2
        new_c = new_c * (n_points / 2) / np.sum(new_c)
        
        return new_c

    # Initial construction
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    # Hyperparameters for optimization
    trust_radius = 0.1
    max_no_improvement = 100
    no_improvement_count = 0
    
    start_time = time.time()
    iterations = 0
    
    # Segment-based approach
    # Assume 10 segments for optimization, keep n_points >> segments for control
    
    m_segments = 10
    segment_averages = np.ones(m_segments) * (n_points / (2 * m_segments))  # Initial averages
    
    while time.time() - start_time < budget_s and no_improvement_count < max_no_improvement:
        iterations += 1
        # Create h from segment averages, replicate to match n_points
        # We will use each segment average over several points
        segments = []
        segment_lengths = []
        inv_segment_lengths = []
        
        # Uniformly expand the segment averages to fit into n_points
        segment_indices = np.linspace(0, n_points, m_segments + 1, dtype=int)
        for i in range(m_segments):
            start = segment_indices[i]
            end = segment_indices[i+1]
            length = end - start
            if length < 1:
                length = 1
            segment_lengths.append(length)
            inv_segment_lengths.append(1.0 / length)
            segments.append(np.full(length, segment_averages[i]))
        
        # Merge segments to match exactly n_points
        h_from_segs = np.concatenate(segments)
        # If too long, truncate; if too short, pad
        if len(h_from_segs) > n_points:
            h_from_segs = h_from_segs[:n_points]
        elif len(h_from_segs) < n_points:
            diff = n_points - len(h_from_segs)
            h_from_segs = np.pad(h_from_segs, (0, diff), mode='constant', constant_values=0)
        
        # Ensure sum is correct
        h_from_segs = h_from_segs * (n_points / 2) / np.sum(h_from_segs)
        
        # Evaluate current assignment
        current_c5 = evaluate(h_from_segs)
        
        # Compute gradient approximation
        try:
            c_grad = compute_gradient(h_from_segs)
            # Project down to segment-average space
            segment_grad = np.zeros(m_segments)
            pos = 0
            for i, length in enumerate(segment_lengths):
                segment_grad[i] = np.sum(c_grad[pos:pos+length]) * inv_segment_lengths[i]
                pos += length
            # Update segment averages using trust region
            new_segment_averages = trust_region_update(segment_averages, segment_grad, trust_radius)
            segment_averages = new_segment_averages
            
            # Check for improvement
            new_h_from_segs = np.clip(h_from_segs, 0, 1)
            new_h_from_segs = new_h_from_segs * (n_points / 2) / np.sum(new_h_from_segs)
            new_c5 = evaluate(new_h_from_segs)
            
            if new_c5 < current_c5:
                # Accept the improvement
                current_c5 = new_c5
                best_c5 = new_c5
                no_improvement_count = 0
                h_from_segs = new_h_from_segs
                best_h = h_from_segs.copy()
                # Reduce trust radius on success
                trust_radius = max(0.01, trust_radius * 0.99)
            else:
                no_improvement_count += 1
                # Increase trust radius if we're not improving much
                trust_radius = min(0.5, trust_radius * 1.01)
        
        except Exception as e:
            # If any computation fails, maintain best value
            pass
        
        # Occasionally try random perturbations
        if iterations % 20 == 0:
            h_trial = perturb(best_h, delta=0.05)
            h_trial = smooth(h_trial, window=3)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
    
    # Final smooth of best solution
    best_h = smooth(best_h, window=3)
    # Re-normalize
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Evaluate final
    final_c5 = evaluate(best_h)
    return (best_h, final_c5, n_points)
