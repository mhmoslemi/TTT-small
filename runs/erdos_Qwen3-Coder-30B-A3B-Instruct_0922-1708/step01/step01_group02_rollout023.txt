import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction with more refined shape
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle with slightly higher values
        mid = n_points // 2
        h[mid-4:mid+4] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    # Initialize PSO parameters
    n_particles = 50
    max_iterations = 10000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        # Start with slightly perturbed version of h
        h_init = h + np.random.normal(0, 0.05, n_points)
        h_init = project_h(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.uniform(-1, 1, n_points) * 0.1)
        pbest.append(h_init.copy())
        pbest_c5.append(compute_c5(h_init))
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # PSO main loop
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while iteration < max_iterations and (time.time() - start_time) < budget_s - 1:
        # Update inertia weight linearly
        w = w_start - (w_start - w_end) * (iteration / max_iterations)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = project_h(particles[i])
            
            # Evaluate current particle
            current_c5 = compute_c5(particles[i])
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                
                # Check if this is global best
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
                
        iteration += 1
        
        # Early stopping condition
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final projection and evaluation for best solution
    final_h = project_h(gbest)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
