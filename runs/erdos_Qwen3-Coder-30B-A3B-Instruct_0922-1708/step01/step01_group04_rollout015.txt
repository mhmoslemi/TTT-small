import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def coarse_to_fine(h_coarse, n_points, coarse_n):
        # Reconstruct fine h from coarse representation
        h_fine = np.repeat(h_coarse, n_points // coarse_n)
        h_fine = h_fine[:n_points]
        if len(h_fine) < n_points:
            h_fine = np.pad(h_fine, (0, n_points - len(h_fine)), 'constant', constant_values=h_fine[-1])
        
        # Smooth a bit
        kernel = np.exp(-np.linspace(-2, 2, 5)**2 / 2)
        kernel = kernel / np.sum(kernel)
        h_fine = np.convolve(h_fine, kernel, mode='same')
        
        # Re-normalize
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        return np.clip(h_fine, 0, 1)
    
    def solve_coarse(n_coarse):
        # Segment the problem
        h_coarse = np.ones(n_coarse)
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
        
        # Define constraints and bounds
        bounds = [(0, 1) for _ in range(n_coarse)]
        
        def obj(c):
            h = coarse_to_fine(c, n_points, n_coarse)
            return evaluate(h)
        
        def constraint(c):
            return np.sum(c) - n_coarse / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        
        # Optimize
        try:
            res = minimize(obj, h_coarse, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 1000, 'ftol': 1e-6})
            if res.success:
                return coarse_to_fine(res.x, n_points, n_coarse)
        except:
            pass
        return coarse_to_fine(h_coarse, n_points, n_coarse)
    
    start_time = time.time()
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    # Try coarse to fine approach
    m_values = [7, 14, 21, 28]
    for m in m_values:
        if time.time() - start_time > budget_s - 1:
            break
        try:
            h_trial = solve_coarse(m)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
        except:
            continue
    
    # Also try local perturbations
    no_improvement_count = 0
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        h_trial = best_h.copy()
        # Random perturb some points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_trial[i] += np.random.uniform(-0.02, 0.02)
        # Clip and re-normalize
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Smooth
        kernel = np.exp(-np.linspace(-2, 2, 5)**2 / 2)
        kernel = kernel / np.sum(kernel)
        h_trial = np.convolve(h_trial, kernel, mode='same')
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        c5_trial = evaluate(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
    
    return (best_h, best_c5, n_points)
