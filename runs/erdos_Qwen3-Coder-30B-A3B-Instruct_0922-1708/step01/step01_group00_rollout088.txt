import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to n_points resolution
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Define optimization variables: breakpoints and interval heights
    # Use fewer variables: fewer intervals
    K = 8  # Number of intervals
    # Initialize breakpoints (K+1 points: 0, b1, b2, ..., bK, 2)
    # We'll optimize the internal K breakpoints
    breakpoints = np.linspace(0, 2, K+1)
    h_vals = np.ones(K) * 0.5  # Initial interval heights
    
    # Precompute integrals for faster access
    x_vals = np.linspace(0, 2, n_points)
    
    # Store best solution
    best_breakpoints = breakpoints.copy()
    best_h_vals = h_vals.copy()
    best_c5 = float('inf')
    
    # Gradient descent on piecewise function
    lr_b = 0.005  # Breakpoint learning rate
    lr_h = 0.05   # Height learning rate
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Construct full h from intervals
        h_full = np.zeros(n_points)
        for i in range(K):
            start_idx = int(np.round(breakpoints[i] / dx))
            end_idx = int(np.round(breakpoints[i+1] / dx))
            h_full[start_idx:end_idx] = h_vals[i]
        
        if np.sum(h_full) * dx != 1:
            h_full = h_full * (1 / (np.sum(h_full) * dx))  # Rescale
        
        # Compute overlap integral
        c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_breakpoints = breakpoints.copy()
            best_h_vals = h_vals.copy()
            best_h = h_full.copy()
        
        # Compute gradients
        # Gradient w.r.t. breakpoints
        grad_b = np.zeros(K)
        eps = 1e-5
        
        for i in range(K):
            # Perturb breakpoint i
            bp_new = breakpoints.copy()
            bp_new[i] += eps
            bp_new = np.clip(bp_new, 0, 2)
            bp_new = np.sort(bp_new)  # Ensure order
            
            # Recompute h
            h_test = np.zeros(n_points)
            for j in range(K):
                start_idx = int(np.round(bp_new[j] / dx))
                end_idx = int(np.round(bp_new[j+1] / dx))
                h_test[start_idx:end_idx] = h_vals[j]
            h_test = h_test * (1 / (np.sum(h_test) * dx))
            
            c5_new = np.max(np.correlate(h_test, 1 - h_test, mode='full')) * dx
            
            grad_b[i] = (c5_new - c5) / eps
        
        # Gradient w.r.t. heights
        grad_h = np.zeros(K)
        for i in range(K):
            # Perturb h_vals[i]
            h_test = h_vals.copy()
            h_test[i] += eps
            h_test = np.clip(h_test, 0, 1)
            
            # Recompute h
            h_test_full = np.zeros(n_points)
            for j in range(K):
                start_idx = int(np.round(breakpoints[j] / dx))
                end_idx = int(np.round(breakpoints[j+1] / dx))
                h_test_full[start_idx:end_idx] = h_test[j]
            h_test_full = h_test_full * (1 / (np.sum(h_test_full) * dx))
            
            c5_new = np.max(np.correlate(h_test_full, 1 - h_test_full, mode='full')) * dx
            
            grad_h[i] = (c5_new - c5) / eps
        
        # Update breakpoints and heights
        breakpoints[:-1] -= lr_b * grad_b
        breakpoints = np.clip(breakpoints, 0, 2)
        breakpoints = np.sort(breakpoints)
        
        h_vals -= lr_h * grad_h
        h_vals = np.clip(h_vals, 0, 1)
        
        # Ensure integral constraint for h_vals
        total = np.sum(h_vals) * dx  # Weighted sum
        h_vals = h_vals * (1 / total)
        
        # Update full h
        h_full = np.zeros(n_points)
        for i in range(K):
            start_idx = int(np.round(breakpoints[i] / dx))
            end_idx = int(np.round(breakpoints[i+1] / dx))
            h_full[start_idx:end_idx] = h_vals[i]
        h_full = h_full * (1 / (np.sum(h_full) * dx))  # Double-check rescale
        
    return (best_h.tolist(), best_c5, n_points)
