import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Normalize to integral = 1
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def reconstruct_h(c, n_points):
        # Create h from segment averages
        m = len(c)
        segment_size = n_points // m
        remainder = n_points % m
        h = np.zeros(n_points)
        for i in range(m):
            start_idx = i * segment_size + min(i, remainder)
            end_idx = start_idx + segment_size + (1 if i < remainder else 0)
            h[start_idx:end_idx] = c[i]
        # Apply Gaussian smoothing
        if segment_size > 1:
            kernel = np.exp(-np.linspace(-2, 2, 5)**2 / 2)
            kernel = kernel / np.sum(kernel)
            h = np.convolve(h, kernel, mode='same')
        return np.clip(h, 0, 1)

    def optimize_segments(c0, n_points, max_iter=100):
        m = len(c0)  
        # Constraint: sum of c should equal m/2  
        def constraint_func(c):
            return np.sum(c) - m/2
        
        # Bounds for c
        bounds = [(0, 1) for _ in range(m)]
        
        # Objective function
        def objective(c):
            h = reconstruct_h(c, n_points)
            return evaluate(h)
        
        # Optimization
        result = minimize(objective, c0, method='SLSQP', bounds=bounds,
                         constraints={'type': 'eq', 'fun': constraint_func},
                         options={'maxiter': max_iter})
        
        if result.success:
            return result.x
        else:
            return c0
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    n_refinements = 0
    improvement_stall = 0
    
    while time.time() - start_time < budget_s and improvement_stall < 100:
        # Try different segment counts (7, 14, 21, ...)
        segment_counts = [7, 14, 21, 28, 35, 42, 49, 56, 63]
        for m in segment_counts:
            if m > n_points:
                break
                
            # Start with current best solution or initialize
            try:
                current_h = best_h
                # If we have a good solution, extract segment averages
                segment_size = n_points // m
                remainder = n_points % m
                c = np.zeros(m)
                for i in range(m):
                    start_idx = i * segment_size + min(i, remainder)
                    end_idx = start_idx + segment_size + (1 if i < remainder else 0)
                    c[i] = np.mean(current_h[start_idx:end_idx])
            except:
                c = np.ones(m) * (m / 2) / m  # Uniform initial guess
            
            # Run optimization
            c_opt = optimize_segments(c, n_points, max_iter=300)
            h_opt = reconstruct_h(c_opt, n_points)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)  # Renormalize
            c5_opt = evaluate(h_opt)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                improvement_stall = 0
            else:
                improvement_stall += 1
                
            # Check early termination if no progress
            if improvement_stall >= 50 and n_refinements > 1:
                break
                
        n_refinements += 1
        
        # Early exit if we're using large segment count and stability
        if m >= n_points:
            break
    
    return (best_h, best_c5, n_points)
