import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a symmetric function that's geometrically pleasing
    x = np.linspace(0, 2, n_points)
    # A smooth symmetric pulse
    h = 0.5 * (1 + np.sin(np.pi * x))
    h = project_h(h, n_points)
    return h

def gradient_step(h, n_points, eta=0.1):
    """Perform one gradient descent step with adaptive learning rate."""
    dx = 2.0 / n_points
    # Find the shift that gives maximum overlap
    conv = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(conv)
    shift = max_idx - (n_points - 1)
    
    # Compute the gradient explicitly for this maximum shift
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + shift) % n_points
        grad[i] = -dx * (1 - h[j])
    
    # Line search for optimal step size
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Try a few candidate alpha values if needed
    alphas = [eta, eta/2, eta/4, eta/8]
    best_alpha = eta
    best_val = objective(eta)
    for alpha in alphas:
        if alpha <= 0:
            continue
        val = objective(alpha)
        if val < best_val:
            best_alpha = alpha
            best_val = val
    
    # Perform the step with the chosen alpha
    h_new = h - best_alpha * grad
    h_new = project_h(h_new, n_points)
    return h_new, best_val, best_alpha

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # Use 45 points as per prior successful tests
    h = initialize_h(n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    eta = 0.1  # Initial learning rate
    iter_no_improvement = 0
    max_no_improvement = 1000
    min_improvement = 1e-6
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iter_no_improvement < max_no_improvement:
        # Gradient step
        h_new, new_c5, alpha = gradient_step(best_h, n_points, eta)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            iter_no_improvement = 0
            # Adjust learning rate back up on success
            eta = min(eta * 1.1, 0.5)
        else:
            iter_no_improvement += 1
        
        if abs(new_c5 - best_c5) < min_improvement:
            break
            
        iteration += 1
        
        # Reduce learning rate if no improvement in a while
        if iter_no_improvement % 50 == 0:
            eta = max(eta * 0.5, 1e-6)
    
    # Ensure final result is valid
    final_h = project_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
