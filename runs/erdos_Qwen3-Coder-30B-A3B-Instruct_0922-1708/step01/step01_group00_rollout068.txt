import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Genetic algorithm operations
        if iteration % 10 == 0:
            # Selection
            fitness = []
            population = []
            pop_size = 50
            
            # Generate population including current best
            for _ in range(pop_size - 1):
                # Perturb the current solution
                h_perturb = h + np.random.normal(0, 0.02, n_points)
                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
                population.append(h_perturb)
                
            population.append(h)
            
            # Evaluate fitness
            for candidate in population:
                corr = np.correlate(candidate, 1 - candidate, mode='full')
                fit = np.max(corr) * dx
                fitness.append(fit)
            
            # Select top 30%
            top_indices = np.argsort(fitness)[:pop_size // 3]
            selected = [population[i] for i in top_indices]
            
            # Crossover and mutation
            new_population = []
            for _ in range(pop_size):
                # Randomly select two parents
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                # Arithmetic crossover
                child = (parent1 + parent2) / 2
                # Mutation
                if np.random.rand() < 0.1:
                    noise = np.random.normal(0, 0.05, n_points)
                    child = child + noise
                # Clip and rescale
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            
            # Replace the population with the new one
            population = new_population
            h = population[0]  # Take the first element as the new current solution
            
        # Regular gradient-based update for fine tuning
        if iteration % 3 == 0:
            # Compute gradient approximation
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update h using gradient descent
            lr = 0.01
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
        
        # Early termination if no improvement for a while
        if iteration - last_improvement > 500 and iteration > 1000:
            break
    
    return (best_h.tolist(), best_c5, n_points)
