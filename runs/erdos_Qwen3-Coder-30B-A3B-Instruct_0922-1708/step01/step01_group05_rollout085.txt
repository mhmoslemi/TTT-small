import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Simple alternating construction
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 2 == 0 else 0
        # Scale to correct integral
        h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_C = float('inf')

    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def project_h(h_vals):
        # Clamp values to [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to match target sum
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals

    # Simulated Annealing
    T = 1.0
    alpha = 0.999
    cooling_threshold = 1000
    no_improvement_count = 0
    
    current_h = h.copy()
    current_C = compute_C(current_h)
    
    if current_C < best_C:
        best_C = current_C
        best_h = current_h.copy()
    
    # Main optimization loop
    it = 0
    while time() - start_time < budget_s - 1:
        it += 1
        # Generate neighbor by perturbing a random segment
        new_h = current_h.copy()
        # Perturb at a random index
        idx = np.random.randint(0, n_points)
        # Add small random perturbation
        new_h[idx] += np.random.normal(0, 0.05)
        # Project to feasible region
        new_h = project_h(new_h)
        
        new_C = compute_C(new_h)
        
        # Accept or reject
        if new_C < current_C or np.random.rand() < np.exp((current_C - new_C) / T):
            current_h = new_h
            current_C = new_C
            
        if new_C < best_C:
            best_C = new_C
            best_h = current_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Cooling schedule
        if no_improvement_count > cooling_threshold:
            T *= alpha
            no_improvement_count = 0
            
    return (best_h.tolist(), best_C, n_points)
