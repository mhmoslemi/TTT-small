import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from existing construction or create new
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Normalize
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Precompute correlation indices for efficiency
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Simulated Annealing with Trust Region approach
    def get_trust_region_step(c, grad, h, current_c5):
        # Simple trust region - compute step with limited size
        step_size = 0.05
        max_step = 0.25
        direction = - grad
        norm_direction = np.linalg.norm(direction)
        if norm_direction > 0:
            direction = direction / norm_direction
        step = step_size * direction
        # Clip step to trust region
        step = np.clip(step, -max_step, max_step)
        return step
    
    def project(c):
        # Ensure 0 <= c <= 1 and sum(c) = m/2
        c = np.clip(c, 0, 1)
        m = len(c)
        target_sum = m / 2
        current_sum = np.sum(c)
        if current_sum > 0:
            c = c * target_sum / current_sum
        else:
            c = np.ones(m) * target_sum / m
        return c
    
    # Compute initial h from segment averages
    def make_h_from_segments(c, n_points):
        m = len(c)
        segments = np.repeat(c, np.ceil(n_points / m).astype(int))[:n_points]
        return segments / np.sum(segments) * (n_points / 2)
    
    # Gradient approximation using finite differences
    def compute_gradient(c, h, current_c5, eps=1e-4):
        grad = np.zeros_like(c)
        for i in range(len(c)):
            c_plus = c.copy()
            c_plus[i] += eps
            c_plus = project(c_plus)
            h_plus = make_h_from_segments(c_plus, n_points)
            c5_plus = evaluate(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
        return grad
    
    def perturb_c(c, temp):
        # Perturb some segments with some probability
        new_c = c.copy()
        p = 0.3 * np.exp(-temp)  # decrease with temp
        for i in range(len(c)):
            if np.random.rand() < p:
                delta = np.random.normal(0, 0.1)
                new_c[i] += delta
        return project(new_c)
    
    def optimized_start(n_points):
        # Try to make a better initial construction
        segments = 10
        c_start = np.ones(segments) * 0.5
        # Ensure sum = segments/2
        c_start = c_start * (segments / 2) / np.sum(c_start)
        h_init = make_h_from_segments(c_start, n_points)
        # Smoothing for better behavior
        h_init = np.convolve(h_init, np.ones(5) / 5, mode='same')
        return c_start, h_init
    
    # Initialize
    if 'initial_h_values' in globals():
        h_current = h_start.copy()
        # Convert back to segments for optimization
        m = 10
        seg_length = n_points // m
        c_current = np.array([np.mean(h_current[i*seg_length:(i+1)*seg_length]) 
                             for i in range(m)])
    else:
        c_current, h_current = optimized_start(n_points)
    
    c_current = project(c_current)
    
    latest_c5 = evaluate(h_current)
    best_c5 = latest_c5
    best_c = c_current.copy()
    
    start_time = time.time()
    no_improvement = 0
    temperature = 1.0
    trust_radius = 0.1
    
    # Optimization loop
    while time.time() - start_time < budget_s and no_improvement < 100:
        # Compute gradient
        grad = compute_gradient(c_current, h_current, latest_c5)
        
        # Calculate trust region step
        step = get_trust_region_step(c_current, grad, h_current, latest_c5)
        
        # Update with trust region step
        c_new = c_current + step
        c_new = project(c_new)
        h_new = make_h_from_segments(c_new, n_points)
        c5_new = evaluate(h_new)
        
        # Accept or reject step based on trust region
        if c5_new < latest_c5:
            c_current = c_new
            h_current = h_new
            latest_c5 = c5_new
            no_improvement = 0
            if c5_new < best_c5:
                best_c = c_current.copy()
                best_c5 = c5_new
        else:
            # Decrease trust radius if failed
            trust_radius *= 0.5
            no_improvement += 1
        
        # Simulated Annealing Perturbation
        if np.random.rand() < 0.1 * np.exp(-temperature):
            c_pert = perturb_c(c_current, temperature)
            h_pert = make_h_from_segments(c_pert, n_points)
            c5_pert = evaluate(h_pert)
            if c5_pert < latest_c5:
                c_current = c_pert
                h_current = h_pert
                latest_c5 = c5_pert
                no_improvement = 0
                if c5_pert < best_c5:
                    best_c = c_pert.copy()
                    best_c5 = c5_pert
                    
        # Update temperature
        temperature += 0.05
        
        # Occasionally reset trust radius
        if no_improvement % 20 == 0:
            trust_radius = 0.1
    
    # Final reconstruction: expand segments back to full resolution
    final_h = make_h_from_segments(best_c, n_points)
    # Add smoothing for further refinement
    final_h = np.convolve(final_h, np.ones(5) / 5, mode='same')
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    final_c5 = evaluate(final_h)
    
    return (final_h, final_c5, n_points)
