import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Default construction - a symmetric piecewise constant function
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Create a plateau in the center
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    #PSO parameters
    n_particles = 50
    max_iter = 10000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    inertia_adapt = True
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        particle = project_h(np.random.rand(n_points) * 0.5 + 0.25)
        particles.append(particle.copy())
        velocities.append(np.random.rand(n_points) * 2 - 1)
        pbest.append(particle.copy())
        pbest_vals.append(compute_c5(particle))
    
    # Swarm global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # Initialize last_improvement iteration counter
    last_improvement = 0
    patience = 1000  # stop if no improvement in this many iterations
    
    # PSO loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Update inertia weight if needed
        if inertia_adapt:
            w = w_start - (w_start - w_end) * (iteration / max_iter)
        else:
            w = w_start
            
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            vel = (w * vel 
                   + c1 * r1 * (pbest[i] - particle)
                   + c2 * r2 * (gbest - particle))
            
            # Update position
            particle = particle + vel
            
            # Apply constraints
            particle = project_h(particle)
            
            # Evaluate current particle
            current_val = compute_c5(particle)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particle.copy()
                pbest_vals[i] = current_val
                
                # Update global best if needed
                if current_val < gbest_val:
                    gbest = particle.copy()
                    gbest_val = current_val
                    last_improvement = iteration
            
            # Save updated particle and velocity
            particles[i] = particle
            velocities[i] = vel
            
        # Early termination if no improvement
        if iteration - last_improvement > patience:
            break
            
    # Final evaluation
    final_h = project_h(gbest)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
