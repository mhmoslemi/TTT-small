import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    max_iter = 5000
    elite_size = pop_size // 5
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, len(h_base)), h_base)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        population.append(h_base.copy())
    
    # Add random initial individuals
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = iteration
        
        # Check for stagnation
        if iteration - last_improvement > 300:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Generate new population through crossover and mutation
        new_generation = elite[:]
        
        while len(new_generation) < pop_size:
            # Selection: tournament
            parent1 = elite[np.random.randint(0, len(elite))]
            parent2 = elite[np.random.randint(0, len(elite))]
            
            # Crossover: arithmetic mean
            alpha = np.random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation: Gaussian noise
            if np.random.random() < 0.1:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
            
            # Repair constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_generation.append(child)
            
        population = new_generation
        
    return (best_h.tolist(), best_c5, n_points)
