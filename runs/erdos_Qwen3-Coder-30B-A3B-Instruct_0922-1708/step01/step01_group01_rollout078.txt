import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using Simulated Annealing + Gradient Descent"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    step = 0.1
    
    # Gradient descent parameters
    gd_max_iter = 5000
    eta = 0.01
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    
    # Simulated Annealing
    for iteration in range(max_iter):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Mutation: perturb randomly
        h_new = h + np.random.uniform(-step, step, n_points)
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new.copy()
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        
        # Cooling
        T *= alpha
    
    # Gradient Descent on best solution
    h = best_h.copy()
    for iteration in range(gd_max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= eta * np.sign(grad)
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
        else:
            eta *= 0.999  # Adaptive learning rate
    
    return (best_h, best_c5, n_points)
