import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Normalize h_start to satisfy integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    dx = 2.0 / n_points

    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def optimize_segments(c, n_segments, n_points_total, dx):
        # Reconstruct h from segment averages
        segment_size = n_points_total // n_segments
        h_reconstructed = np.repeat(c, segment_size)
        if len(h_reconstructed) < n_points_total:
            h_reconstructed = np.pad(h_reconstructed, (0, n_points_total - len(h_reconstructed)), mode='constant')
        h_reconstructed = h_reconstructed[:n_points_total]
        
        # Apply Gaussian smoothing for continuity
        kernel = np.exp(-np.linspace(-2, 2, 5)**2 / 2)
        kernel = kernel / np.sum(kernel)
        h_smoothed = np.convolve(h_reconstructed, kernel, mode='same')
        h_smoothed = np.clip(h_smoothed, 0, 1)
        
        # Renormalize
        h_smoothed = h_smoothed * (n_points_total / 2) / np.sum(h_smoothed)
        
        return h_smoothed
    
    def objective(c, n_segments, n_points_total, dx):
        h = optimize_segments(c, n_segments, n_points_total, dx)
        return evaluate(h)
    
    def solve_for_segments(n_segments, h_init=None):
        if h_init is not None:
            # Initialize from previous solution
            segment_size = n_points // n_segments
            c_initial = np.array([np.mean(h_init[i:i+segment_size]) 
                                for i in range(0, n_points, segment_size)])
        else:
            # Start with uniform distribution
            c_initial = np.ones(n_segments) * (n_segments / 2) / n_segments
        
        # Define bounds (0 to 1 for all c_i)
        bounds = [(0, 1) for _ in range(n_segments)]
        
        # Define constraint: sum(c) = n_segments / 2
        def constraint_func(c):
            return np.sum(c) - n_segments / 2
        
        cons = {'type': 'eq', 'fun': constraint_func}
        
        # Optimize
        result = minimize(objective, c_initial, args=(n_segments, n_points, dx),
                         method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 500, 'ftol': 1e-8})
        
        if result.success:
            h_opt = optimize_segments(result.x, n_segments, n_points, dx)
        else:
            # Fallback to initial solution if optimization fails
            h_opt = h_init if h_init is not None else h_start
        
        return h_opt
    
    # Start optimization with current construction if available, or
    # a basic starting point
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    # Optimization loop with increasing resolution
    start_time = time.time()
    n_segments = 7
    no_improvement_count = 0
    iterations = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        # Try different segment counts (starting from 7, then 14, 21)
        segment_counts = []
        resolution = 7
        while resolution <= n_points:
            segment_counts.append(resolution)
            resolution *= 2
        
        # Try each segment count
        for seg_count in segment_counts:
            if seg_count > n_points:
                break
            
            # If segment count is same or larger than current resolution, optimize
            # If we already tried a finer resolution, skip here to avoid redundant work
            
            try:
                h_trial = solve_for_segments(seg_count, best_h if seg_count == n_segments else None)
                c5_trial = evaluate(h_trial)
                
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    no_improvement_count = 0
                    break  # Break inner loop, start over with new resolution if needed
                else:
                    no_improvement_count += 1
                    
            except Exception as e:
                no_improvement_count += 1
                continue  # Continue to next segment count or loop
        
        # Redefine resolution for next iteration
        n_segments = min(n_segments * 2, n_points)
        if n_segments > n_points:
            n_segments = n_points

    return (best_h, best_c5, n_points)
