import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Genetic Algorithm components
    pop_size = 50
    mutation_rate = 0.1
    elite_size = pop_size // 5
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        candidate = np.random.rand(n_points)
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        population.append(candidate)
    
    # Add the initial solution if it exists
    if 'initial_h_values' in globals():
        population[0] = h.copy()
    
    # Early stopping tracking
    last_improvement = 0
    patience = 300
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        if iteration % 100 == 0:
            if time.time() - start_time > budget_s - 5:
                break
            
        # Evaluate population
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            last_improvement = iteration
            
        # Early stopping for plateau
        if iteration - last_improvement > patience:
            break
            
        # Selection - keep top 20%
        elite = population[:elite_size]
        
        # Create new population through crossover and mutation
        new_population = elite.copy()
        
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = elite[np.random.randint(0, len(elite))]
            parent2 = elite[np.random.randint(0, len(elite))]
            
            # Crossover
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                
            # Repair constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population
        
    return (best_h.tolist(), best_c5, n_points)
