import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def optimize_segments(c_values, m, n_points, dx):
        # Reconstruct h from segment values
        segment_width = n_points // m
        h_reconstructed = np.zeros(n_points)
        for i in range(m):
            start_idx = i * segment_width
            end_idx = min((i + 1) * segment_width, n_points)
            h_reconstructed[start_idx:end_idx] = c_values[i]
        return h_reconstructed

    def objective(c_values, m, n_points, dx):
        h = optimize_segments(c_values, m, n_points, dx)
        return evaluate(h)

    def constraint_sum(c_values, m):
        return np.sum(c_values) - m / 2.0

    def run_optimization(initial_c_values, m, n_points, dx, max_iter=1000):
        bounds = [(0, 1) for _ in range(m)]
        cons = {'type': 'eq', 'fun': lambda x: constraint_sum(x, m)}
        
        result = minimize(
            objective, 
            initial_c_values, 
            args=(m, n_points, dx),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-8}
        )
        return result.x

    best_c5 = float('inf')
    best_h_final = None
    best_m = 7

    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 50
    max_segments = 21
    segments_to_try = [7, 9, 11, 13, 15, 17, 19, 21]

    # Try each segment count
    for m in segments_to_try:
        if time.time() - start_time > budget_s - 1:
            break

        # Initialize segment values
        initial_c = np.ones(m) * 0.5
        segments_per_dim = n_points // m
        # Set up with some variation to avoid local minima
        if m > 1:
            for i in range(m):
                if i % 2 == 0:
                    initial_c[i] = 0.3
                else:
                    initial_c[i] = 0.7

        # Run optimization
        result_c = run_optimization(initial_c, m, n_points, dx, max_iter=1000)
        h_temp = optimize_segments(result_c, m, n_points, dx)
        
        c5_temp = evaluate(h_temp)
        iterations += 1
        
        if c5_temp < best_c5:
            best_c5 = c5_temp
            best_h_final = h_temp.copy()
            best_m = m
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Small perturbation
        if time.time() - start_time < budget_s - 5:
            perturb_indices = np.random.choice(m, max(1, m // 10), replace=False)
            for i in perturb_indices:
                delta = np.random.uniform(-0.1, 0.1)
                result_c[i] += delta
                result_c[i] = np.clip(result_c[i], 0, 1)
            h_temp = optimize_segments(result_c, m, n_points, dx)
            c5_temp = evaluate(h_temp)
            
            if c5_temp < best_c5:
                best_c5 = c5_temp
                best_h_final = h_temp.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1

        if no_improvement_count >= max_no_improvement:
            break

    # Final smoothing
    smoothed_h = np.convolve(best_h_final, np.ones(3)/3, mode='same')
    smoothed_h = np.clip(smoothed_h, 0, 1)
    
    final_h = smoothed_h * (n_points / 2) / np.sum(smoothed_h)
    
    final_c5 = evaluate(final_h)

    return (final_h, final_c5, n_points)
