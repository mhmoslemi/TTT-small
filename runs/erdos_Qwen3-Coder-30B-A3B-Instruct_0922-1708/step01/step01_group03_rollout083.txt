import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: simple symmetric shape to start with
    h = np.ones(n_points) * 0.5
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use n_points = 45 for a reasonable balance
    n_points = 45
    h = initialize_h(n_points)
    
    start_time = time.time()
    
    # Phase 1: PSO with symmetry assumption
    # We work with symmetric functions h such that h[i] = h[n_points - 1 - i]
    # This reduces the number of decision variables to (n+1)//2
    
    # PSO parameters
    swarm_size = 30
    max_iter = 1000
    inertia_weight = 0.9
    cognitive_weight = 2.0
    social_weight = 2.0
    velocity_max = 0.2
    
    # For symmetric functions, we only optimize the first half
    half_size = (n_points + 1) // 2
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    gbest = None
    gbest_score = float('inf')
    
    # Initialize particles with symmetric structures
    for _ in range(swarm_size):
        # Start with base function and add small random noise
        h_part = np.ones(half_size) * 0.5  # Base symmetric structure
        for i in range(half_size):
            h_part[i] += (np.random.random() - 0.5) * 0.1  # add small noise
        # Mirror to full
        h_full = np.zeros(n_points)
        for i in range(half_size):
            h_full[i] = h_part[i]
            h_full[n_points - 1 - i] = h_part[i]
        h_full = project_h(h_full, n_points)
        particles.append(h_full.copy())
        velocities.append(np.zeros(n_points))
        score = evaluate_c5(h_full)
        pbest.append(h_full.copy())
        pbest_scores.append(score)
        if score < gbest_score:
            gbest = h_full.copy()
            gbest_score = score

    # Begin iterations
    iteration = 0
    iter_no_improvement = 0
    max_no_improvement = 1000

    while time.time() - start_time < budget_s * 0.9 and iteration < max_iter:
        iteration += 1
        if iteration % 100 == 0:
            pass  # progress update
        
        # Update inertia weight
        if iteration > 500:
            inertia_weight = max(0.4, 0.9 - (iteration - 500) * 0.001)
        
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            
            # PSO update
            r1, r2 = np.random.random(2)
            v = (inertia_weight * v + 
                 cognitive_weight * r1 * (pbest[i] - h) + 
                 social_weight * r2 * (gbest - h))
            
            # Clip velocity
            v = np.clip(v, -velocity_max, velocity_max)
            
            # Update position
            h = h + v
            h = project_h(h, n_points)
            
            # Update particle
            particles[i] = h
            score = evaluate_c5(h)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = h.copy()
                pbest_scores[i] = score
                if score < gbest_score:
                    gbest = h.copy()
                    gbest_score = score
                    iter_no_improvement = 0
                else:
                    iter_no_improvement += 1
            else:
                iter_no_improvement += 1
            
            # Early termination if no improvement
            if iter_no_improvement > max_no_improvement:
                break
        
        if iter_no_improvement > max_no_improvement:
            break
            
    # Phase 2: Projected Gradient Descent on best solution
    h = gbest.copy()
    best_score = gbest_score
    learning_rate = 0.05
    max_gd_iter = 10000
    gd_iter = 0
    
    def compute_gradient(h, shift_idx):
        # Compute gradient of C5 wrt the function h at a given shift
        n = len(h)
        dx = 2.0 / n
        grad = np.zeros(n)
        # Find which shift gives the maximum
        conv = np.correlate(h, 1 - h, mode="full")
        max_shift = np.argmax(conv) - (n - 1)
        # Now, gradient of the corresponding contribution to C5
        # This term is: h[i] * (1 - h[i + shift]) * dx
        # So d/dh[i] is (1 - h[i + shift]) * dx
        shift = max_shift
        for i in range(n):
            j = (i + shift) % n
            grad[i] = (1 - h[j]) * dx
        return grad
    
    while time.time() - start_time < budget_s * 0.99 and gd_iter < max_gd_iter:
        grad = compute_gradient(h, 0)  # Just a placeholder for a simple estimate
        # Calculate correlation to find the exact maximum contribution
        conv = np.correlate(h, 1 - h, mode="full")
        max_shift = np.argmax(conv) - (n_points - 1)
        # Gradient simplified calculation assumption:
        # For any given pair (i, i+shift) in the convolution
        # The contribution to the final C5 is h[i] * (1 - h[i+shift])
        # Thus assuming a different approach for numerical gradient:
        # Take average gradient from some close samples
        eps = 1e-5
        grad_avg = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            
            c5_plus = evaluate_c5(h_plus)
            c5_minus = evaluate_c5(h_minus)
            grad_avg[i] = (c5_plus - c5_minus) / (2 * eps)
        
        h_new = h - learning_rate * grad_avg
        h_new = project_h(h_new, n_points)
        new_score = evaluate_c5(h_new)
        
        if new_score < best_score:
            h = h_new
            best_score = new_score
            gd_iter = 0
        else:
            gd_iter += 1
            learning_rate *= 0.998  # Decrease learning rate over time
        
        if learning_rate < 1e-8 or gd_iter > 10000:
            break
    
    # Final validation
    h = project_h(h, n_points)
    final_c5 = evaluate_c5(h)
    
    return (h, final_c5, n_points)
