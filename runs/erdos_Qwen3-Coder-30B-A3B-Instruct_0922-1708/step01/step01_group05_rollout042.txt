    def compute_h(a_coeffs):
        x = np.linspace(0, 2, n_points)
        h_func = np.ones_like(x) * a_coeffs[0]
        for k in range(1, min(len(a_coeffs), m)):
            h_func += a_coeffs[k] * np.cos(np.pi * k * x / 2)
        h_func = np.clip(h_func, 0, 1)
        return h_func
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # Evaluate starting point
    h_vals = compute_h(a)
    h_vals = h_vals * (target_sum / np.sum(h_vals))  # Rescale to sum constraint

    best_C = current_C
    best_h = h_vals.copy()

        h_vals = compute_h(a_coeffs)

        C = compute_C(h_vals)
        return C, None  # We do not compute gradient analytically for simplicity

            h_binary[0] = 1.0
            h_binary[-1] = 0.0
