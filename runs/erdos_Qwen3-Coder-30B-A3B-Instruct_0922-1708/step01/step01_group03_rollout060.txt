import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def coordinatewise_optimize(h, n_points, max_iter=1000):
    """Optimize each coordinate individually with line search."""
    h = h.copy()
    dx = 2.0 / n_points
    prev_c5 = evaluate_c5(h)
    for _ in range(max_iter):
        improved = False
        for i in range(n_points):
            # Compute current contribution for this coordinate
            # This essentially computes the effect of changing h[i] on overlap
            h_current = h[i]
            def objective(h_i):
                h[i] = h_i
                h = project_h(h, n_points)
                return evaluate_c5(h)
                
            # Limit search for efficiency
            bounds = (0, 1)
            # Try few discrete values for quick search within bounds
            test_values = np.linspace(bounds[0], bounds[1], 10)
            best_value = h_current
            best_obj = objective(h_current)
            for val in test_values:
                obj_val = objective(val)
                if obj_val < best_obj:
                    best_obj = obj_val
                    best_value = val
            h[i] = best_value
            h = project_h(h, n_points)
            updated_c5 = evaluate_c5(h)
            if updated_c5 < prev_c5:
                improved = True
                prev_c5 = updated_c5
        if not improved:
            break
    return h, prev_c5

def nonlinear_cg_optimize(h, n_points, max_iter=1000):
    """Use nonlinear conjugate gradient method with projection."""
    h = h.copy()
    dx = 2.0 / n_points
    prev_c5 = evaluate_c5(h)
    grad = np.zeros(n_points)
    
    # Initialize conjugate direction
    dir = np.zeros(n_points)
    prev_dir = np.zeros(n_points)
    prev_grad = np.zeros(n_points)
    
    for iter in range(max_iter):
        if iter % 50 == 0 and iter > 0:
            # Recompute direction using Polak-Ribiere
            beta = np.dot(prev_grad - grad, grad) / np.dot(prev_grad, prev_grad)
            dir = -grad + beta * prev_dir
        else:
            dir = -grad
            
        # Line search in the direction of descent
        # test values
        step_size = 0.05
        best_step = 0.0
        best_c5 = prev_c5
        best_h = h.copy()
        
        for step in np.linspace(0, 2, 21):
            h_new = h + step * dir
            h_new = project_h(h_new, n_points)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_step = step
                best_h = h_new.copy()
        
        # Update
        h = best_h
        prev_c5 = best_c5
        grad = np.zeros(n_points)  # Recompute gradient ''
        
        # Approximate gradient as finite difference for improved efficiency
        # Even though not exact, can make some progress
        eps = 1e-6
        h_plus = h.copy()
        h_minus = h.copy()
        for i in range(n_points):
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_h(h_plus, n_points)
            h_minus = project_h(h_minus, n_points)
            c5_plus = evaluate_c5(h_plus)
            c5_minus = evaluate_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            h_plus[i] -= eps
            h_minus[i] += eps
        
        # Update for next iteration
        prev_dir = dir.copy()
        prev_grad = grad.copy()
        
        if iter % 10 == 0:
            pass
        
        # Stop if improvement is trivial
        if iter > 100 and abs(best_c5 - prev_c5) < 1e-10:
            break
            
    return h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45
    
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: Coordinate-wise optimization
    h_co, c5_co = coordinatewise_optimize(h, n_points, max_iter=1000)
    
    # Phase 2: Nonlinear CG optimization
    h_cg, c5_cg = nonlinear_cg_optimize(h_co, n_points, max_iter=1000)
    
    # Return best result
    final_h = h_cg if c5_cg < c5_co else h_co
    final_c5 = min(c5_cg, c5_co)
    
    # Ensure final result is valid - bug safeguard
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
