from scipy.signal import correlate

        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=3):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def project(h):
        # Ensure h is in [0,1]
        h = np.clip(h, 0, 1)
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Define the objective function for optimization
    def objective(h):
        return evaluate(h)
    
    def constraint_fun(h):
        return np.sum(h) * dx - 1.0  # Should be 0
    
    # Initialize
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iteration = 0

    best_h = project(best_h)
    best_c5 = evaluate(best_h)

        h_trial = best_h.copy()

        n_perturb = max(1, int(n_points * 0.1))

        h_trial = smooth(h_trial, window=3)
        h_trial = project(h_trial)
        c5_trial = evaluate(h_trial)

            best_c5 = c5_trial

            best_h = h_trial.copy()

            h_trial = best_h.copy()

            peak_idx = np.argmax(h_trial)
            left_shift = max(0, peak_idx - 2)

                    h_trial = best_h.copy()
                    h_trial[peak_idx] -= 0.05

                    h_trial = project(h_trial)

                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        best_h = h_trial.copy()
                        break
                        
    return (best_h, best_c5, n_points)
