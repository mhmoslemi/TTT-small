import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap

    def smooth(h, window=5):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth

    def perturb(h, delta=0.05):
        h_new = h.copy()
        # Perturb about 15% of points
        n_perturb = max(1, int(n_points * 0.15))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    def constraint_func(h):
        return np.sum(h) - n_points / 2

    # Define objective function
    def objective(h):
        return evaluate(h)

    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint for sum = N/2
    constraint = {'type': 'eq', 'fun': constraint_func}
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    min_delta = 0.005
    current_delta = 0.05
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        
        # Optimize using L-BFGS-B
        res = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                       constraints=constraint, options={'maxiter': 100})
        
        if res.success:
            h_opt = res.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            c5_opt = evaluate(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            # If optimization fails, perturb the current best
            no_improvement_count += 1
        
        # Apply perturbation every 10 iterations or if no improvement
        if iterations % 10 == 0 or no_improvement_count >= 1:
            h_trial = perturb(best_h, delta=current_delta)
            h_trial = smooth(h_trial, window=5)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                current_delta = max(min_delta, current_delta * 0.9)
            else:
                current_delta = max(min_delta, current_delta * 0.95)
                no_improvement_count += 1
    
    return (best_h, best_c5, n_points)
