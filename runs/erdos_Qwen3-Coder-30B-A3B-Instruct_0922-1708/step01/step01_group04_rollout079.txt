import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap

    def smooth(h, window=3):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth

    def project_constraints(h):
        # Ensure values are in [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to ensure integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return h

    def perturb(h, delta=0.1):
        # Create a perturbed version
        h_new = h.copy()
        # Perturb about 10% of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        return project_constraints(h_new)

    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    
    # First, try gradient-based optimization
    def objective(h):
        return evaluate(h)
    
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize.minimize with L-BFGS-B
    # We'll perturb periodically to escape local minima
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        # Try a few gradient-based steps
        for _ in range(5):
            try:
                # Make a simple step in direction of gradient apprx
                h_trial = best_h.copy()
                # Add small random noise to explore
                noise = np.random.normal(0, 0.01, n_points)
                h_trial += noise
                h_trial = project_constraints(h_trial)
                c5_trial = evaluate(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    no_improvement_count = 0
                    break
            except:
                pass
        
        # Try smoother updates
        h_trial = smooth(best_h, window=3)
        h_trial = project_constraints(h_trial)
        c5_trial = evaluate(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            no_improvement_count = 0        

        # Periodic perturbation
        if iterations % 10 == 0:
            h_trial = perturb(best_h)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
        
        no_improvement_count += 1
    
    return (best_h, best_c5, n_points)
