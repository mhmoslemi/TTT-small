import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap

    def smooth(h, window=3):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def project(h):
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return h

    def gradient(h, step=1e-6):
        # Estimating gradient by finite differences
        grad = np.zeros_like(h)
        base_val = evaluate(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += step
            h_plus = project(h_plus)
            grad[i] = (evaluate(h_plus) - base_val) / step
        return grad

    def trust_region_update(h, grad, trust_radius=0.1):
        # Simple trust region update
        delta_h = -grad * trust_radius / (np.linalg.norm(grad) + 1e-10)
        h_new = h + delta_h
        return project(h_new)

    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    trust_radius = 0.1

    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        # Gradient descent step
        grad = gradient(best_h)
        h_trial = trust_region_update(best_h, grad, trust_radius)
        h_trial = smooth(h_trial, window=3)
        
        c5_trial = evaluate(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            no_improvement_count = 0
            trust_radius = min(0.5, trust_radius * 1.2)  # Increase trust region
        else:
            no_improvement_count += 1
            trust_radius = max(1e-6, trust_radius * 0.5)  # Shrink trust region

        # Random perturbation to escape local minima
        if np.random.rand() < 0.1 and iterations % 5 == 0:
            h_pert = best_h.copy()
            indices = np.random.choice(n_points, max(1, n_points // 10), replace=False)
            for i in indices:
                h_pert[i] += np.random.normal(0, 0.02)
            h_pert = project(h_pert)
            c5_pert = evaluate(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

    # Final smoothing with Gaussian kernel
    kernel = np.exp(-np.linspace(-2, 2, 21)**2/2)
    kernel = kernel / np.sum(kernel)
    best_h = np.convolve(best_h, kernel, mode='same')
    
    return (best_h, best_c5, n_points)
