import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h, k, n_points):
    """Compute gradient w.r.t. h for shift k that gives maximum overlap."""
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + k) % n_points
        grad[i] = (1 - h[j]) * dx
    return grad

def line_search(h, grad, eta, n_points):
    """Perform line search to find optimal step size."""
    def obj(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    try:
        res = minimize_scalar(obj, bounds=(0, eta), method='bounded')
        if res.success:
            return res.x
    except:
        pass
    return 0.0

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Gradient descent with adaptive learning rate
    eta = 0.1
    eta_min = 1e-6
    eta_max = 0.5
    no_improvement = 0
    max_no_improvement = 1000
    iterations = 0
    
    while time.time() - start_time < budget_s and no_improvement < max_no_improvement:
        # Find shift with maximum overlap
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute gradient for this shift
        grad = compute_gradient(h, shift, n_points)
        
        # Line search to find best step size
        alpha = line_search(h, grad, eta, n_points)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Accept update if improvement
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            no_improvement = 0
            # Increase learning rate
            eta = min(eta * 1.2, eta_max)
        else:
            no_improvement += 1
            # Decrease learning rate
            eta = max(eta / 2, eta_min)
        
        iterations += 1
        
        # Occasionally refresh from initial if stuck
        if no_improvement > 100 and 'initial_h_values' in globals():
            h = initialize_h(n_points)
            no_improvement = 0
    
    # Final evaluation and projection
    final_h = project_h(h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
