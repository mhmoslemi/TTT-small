import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 5000
    mutation_rate = 0.1
    elite_size = int(0.2 * pop_size)
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        population.append(h0.copy())
    
    # Fill the rest with random samples
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Selection: top 20%
        elite = population[:elite_size]
        
        # Create new population through crossover and mutation
        new_pop = elite.copy()
        
        while len(new_pop) < pop_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
            parent1, parent2 = elite[idx1], elite[idx2]
            
            # Crossover: arithmetic mean
            alpha = np.random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            
            new_pop.append(child)
            
        population = new_pop[:pop_size]
        
    return (best_h.tolist(), best_c5, n_points)
