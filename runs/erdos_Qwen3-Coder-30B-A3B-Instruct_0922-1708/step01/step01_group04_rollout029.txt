import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def project_h(h):
        """Project h to satisfy constraints"""
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def smooth(h, window=3):
        """Apply smoothing"""
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def perturb(h, delta=0.1):
        """Perturb some points"""
        h_new = h.copy()
        n_perturb = max(1, int(n_points * 0.15))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        return project_h(h_new)
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    
    # Try multiple strategies
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        
        # Strategy 1: Gradient-like perturbation
        h_trial = perturb(best_h, delta=0.05)
        h_trial = smooth(h_trial, window=3)
        c5_trial = evaluate(h_trial)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Strategy 2: Random local search
        if iterations % 10 == 0:
            h_trial = h_start.copy()
            for _ in range(50):
                idx = np.random.randint(0, n_points)
                h_trial[idx] += np.random.uniform(-0.1, 0.1)
                h_trial = project_h(h_trial)
                c5_trial = evaluate(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    no_improvement_count = 0
                    break
    
    # Final validation and refinement
    best_h = project_h(best_h)
    final_c5 = evaluate(best_h)
    
    return (best_h, final_c5, n_points)
