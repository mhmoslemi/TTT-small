import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def construct_h_from_segments(c, n_points):
        # Reconstruct h from segment averages
        m = len(c)
        segments = np.repeat(c, n_points // m)
        if len(segments) < n_points:
            segments = np.pad(segments, (0, n_points - len(segments)), 'constant', constant_values=c[-1])
        elif len(segments) > n_points:
            segments = segments[:n_points]
        
        # Apply Gaussian smoothing
        smoothed = np.convolve(segments, np.exp(-np.linspace(-2, 2, 21)**2 / 2), mode='same')
        smoothed = np.clip(smoothed, 0, 1)
        return smoothed * (n_points / 2) / np.sum(smoothed)
    
    def optimize_segments(m, h_current):
        # Define objective for optimization
        def objective(c):
            h = construct_h_from_segments(c, n_points)
            return evaluate(h)
        
        # Define constraints and bounds
        bounds = [(0, 1) for _ in range(m)]
        cons = {'type': 'eq', 'fun': lambda c: np.sum(c) - m / 2}
        
        # Try to initialize with current solution if available
        try:
            # Starting from the previous segments
            if len(h_current) == n_points:
                # Get segment averages from the best solution so far
                segments = np.array_split(h_current, m)
                init_c = [np.mean(seg) for seg in segments]
            else:
                init_c = [0.5] * m
        except:
            init_c = [0.5] * m
        
        # Optimize
        try:
            result = minimize(objective, init_c, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:
                return result.x
        except:
            pass
        return init_c
    
    # Start with known solution or simple construction
    best_c = None
    best_h = h_start
    best_c5 = evaluate(h_start)
    
    start_time = time.time()
    refinement_levels = []
    
    # Generate refinement levels
    m = 7
    while m <= n_points:
        refinement_levels.append(m)
        if m < 21:
            m += 7
        else:
            m += 14
    
    refinement_idx = 0
    no_improvement = 0
    
    # Optimization loop
    while time.time() - start_time < budget_s and refinement_idx < len(refinement_levels):
        m = refinement_levels[refinement_idx]
        
        # Optimize segments
        if best_c is None:
            c = [0.5] * m
        else:
            # Interpolate best_c to m values
            if len(best_c) > m:
                c = best_c[:m]
            else:
                c = np.interp(np.linspace(0, len(best_c)-1, m), np.arange(len(best_c)), best_c)
        
        c_opt = optimize_segments(m, best_h)
        h_opt = construct_h_from_segments(c_opt, n_points)
        
        # Evaluate
        c5_new = evaluate(h_opt)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_opt.copy()
            best_c = c_opt.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= 100:
            refinement_idx += 1
            if refinement_idx >= len(refinement_levels):
                break
            no_improvement = 0
    
    # Final refinement if time allows
    if time.time() - start_time < budget_s * 0.9:
        # Try increasing resolution
        finest = refinement_levels[min(refinement_idx + 1, len(refinement_levels) - 1)]
        if finest > len(best_h):
            finer_h = construct_h_from_segments(best_c, finest)
            finer_c5 = evaluate(finer_h)
            if finer_c5 < best_c5:
                best_c5 = finer_c5
                best_h = finer_h
                n_points = finest
    
    return (best_h, best_c5, n_points)
