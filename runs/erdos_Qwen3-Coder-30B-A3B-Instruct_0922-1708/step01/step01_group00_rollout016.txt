import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to n_points resolution
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # After 300s without improvement, terminate early
        if iteration - last_improvement > 1000 and time.time() - start_time > 300:
            break
        
        # Crossover and Mutation (Genetic Algorithm)
        # Selection: top 20%
        pop_size = 50
        if iteration == 1:
            population = [h.copy()]
            for _ in range(pop_size - 1):
                h_mut = h + np.random.normal(0, 0.1, n_points)
                h_mut = np.clip(h_mut, 0, 1)
                h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
                population.append(h_mut)
        else:
            # Evaluate population
            population_fitness = []
            for individual in population:
                corr = np.correlate(individual, 1 - individual, mode='full')
                fitness = np.max(corr) * dx
                population_fitness.append(fitness)
            
            # Select top performers
            top_indices = np.argsort(population_fitness)[:pop_size//5]
            top_pop = [population[i] for i in top_indices]
            
            # Generate new population via crossover and mutation
            new_population = [h.copy()]
            while len(new_population) < pop_size:
                # Crossover two parents
                parent1 = top_pop[np.random.randint(len(top_pop))]
                parent2 = top_pop[np.random.randint(len(top_pop))]
                
                # Arithmetic crossover
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
                
                # Mutation
                if np.random.rand() < 0.1:
                    noise = np.random.normal(0, 0.05, n_points)
                    child += noise
                    child = np.clip(child, 0, 1)
                
                # Rescale
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            
            population = new_population
        
        # Select next individual
        h = population[np.random.randint(len(population))]
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
