import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h, k, n_points):
    """Compute exact gradient for shift k."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + k) % n_points
        grad[i] = -(1 - h[j])
    return grad

def line_search(h, grad, n_points, eta, dx):
    """Find optimal step size using line search."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Bracketing
    a, b = 0.0, eta
    fa = objective(a)
    fb = objective(b)
    
    # If we're going downhill, try increasing the interval
    while fb < fa:
        a, b = b, 2 * b
        fa, fb = fb, objective(b)
    
    # Golden section search in [a,b]
    gr = (1 + np.sqrt(5)) / 2
    c = (b - a) / (gr - 1) + a
    d = (b - a) / gr + a
    fc = objective(c)
    fd = objective(d)
    
    while abs(b - a) > 1e-8:
        if fc < fd:
            b, fb = d, fd
            d, fd = c, fc
            c = (b - a) / (gr - 1) + a
            fc = objective(c)
        else:
            a, fa = c, fc
            c, fc = d, fd
            d = (b - a) / gr + a
            fd = objective(d)
    
    return (a + b) / 2

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45  # Use a smaller number for faster runtime
    
    # Initialize
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Parameters
    eta = 0.1
    eta_min = 1e-6
    eta_max = 0.5
    max_iter = 10000
    patience = 1000
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    iter_no_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.99:
            break
            
        # Find worst shift
        dx = 2.0 / n_points
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute exact gradient
        grad = compute_gradient(h, shift, n_points)
        
        # Line search for optimal step
        alpha = line_search(h, grad, n_points, eta, dx)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Update best solution
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            iter_no_improvement = 0
            # Increase learning rate
            eta = min(eta * 1.2, eta_max)
        else:
            iter_no_improvement += 1
            # Reduce learning rate
            eta = max(eta / 2, eta_min)
            
        if iter_no_improvement >= patience:
            break
            
        h = h_new
    
    return (best_h, best_c5, n_points)
