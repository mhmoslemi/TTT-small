import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from parent construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Normalize to integral = 1
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth_h(h, sigma=1.0):
        # Apply Gaussian smoothing to make h continuous
        kernel = np.exp(-0.5 * np.arange(-3*sigma, 3*sigma+1)**2 / sigma**2)
        kernel = kernel / np.sum(kernel)
        h_smooth = np.convolve(h, kernel, mode='same')
        return np.clip(h_smooth, 0, 1)
    
    def segment_optimization(m):
        # Create m segments
        h_seg = np.ones(m) * 0.5
        lambda_ = 1e-4  # penalty for constraints
        
        def objective(c):
            # Reconstruct h from segments
            h_reconstructed = np.repeat(c, n_points // m)
            if len(h_reconstructed) < n_points:
                h_reconstructed = np.pad(h_reconstructed, (0, n_points - len(h_reconstructed)))
            h_reconstructed = h_reconstructed[:n_points]
            h_reconstructed = np.clip(h_reconstructed, 0, 1)
            # Apply smoothing
            h_smooth = smooth_h(h_reconstructed)
            return evaluate(h_smooth)
        
        def constraint_sum(c):
            return np.sum(c) - m/2
            
        def constraint_bounds(c):
            return np.concatenate([c, 1 - c])
        
        # Optimization constraints
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(m)]
        
        result = minimize(objective, h_seg, method='SLSQP', bounds=bounds, constraints=constraints, 
                          options={'maxiter': 100})
        if result.success:
            c_opt = result.x
            h_opt = np.repeat(c_opt, n_points // m)
            if len(h_opt) < n_points:
                h_opt = np.pad(h_opt, (0, n_points - len(h_opt)))
            h_opt = h_opt[:n_points]
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = smooth_h(h_opt)
            return h_opt
        else:
            return h_seg
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    
    # Try different resolutions
    segment_sizes = [7, 14, 21, 28, 35]
    for m in segment_sizes:
        if time.time() - start_time > budget_s - 5:
            break
        try:
            h_trial = segment_optimization(m)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
        except:
            continue
    
    # Final refinement with fine grid
    if time.time() - start_time < budget_s - 10:
        def refine_objective(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return evaluate(h)
            
        def pso_refinement():
            # Simple particle swarm optimization
            n_particles = 20
            max_iter = 100
            w, c1, c2 = 0.7, 1.5, 1.5
            
            particles = []
            velocities = []
            pbest = []
            pbest_vals = []
            
            for _ in range(n_particles):
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                particles.append(h.copy())
                velocities.append(np.random.rand(n_points) * 0.1)
                pbest.append(h.copy())
                pbest_vals.append(evaluate(h))
            
            gbest = pbest[np.argmin(pbest_vals)]
            gbest_val = np.min(pbest_vals)
            
            for _ in range(max_iter):
                if time.time() - start_time > budget_s - 5:
                    break
                for i in range(n_particles):
                    r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                    velocities[i] = (w * velocities[i] + 
                                     c1 * r1 * (pbest[i] - particles[i]) +
                                     c2 * r2 * (gbest - particles[i]))
                    particles[i] += velocities[i]
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                    
                    val = evaluate(particles[i])
                    if val < pbest_vals[i]:
                        pbest[i] = particles[i].copy()
                        pbest_vals[i] = val
                        if val < gbest_val:
                            gbest = particles[i].copy()
                            gbest_val = val
            return gbest, gbest_val
        
        try:
            h_refined, c5_refined = pso_refinement()
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
        except:
            pass
    
    return (best_h, best_c5, n_points)
