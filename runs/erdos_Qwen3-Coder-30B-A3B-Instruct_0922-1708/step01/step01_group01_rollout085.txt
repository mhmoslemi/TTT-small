import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def latin_hypercube_sample(n_points, n_samples):
    """Generate Latin Hypercube Samples"""
    samples = []
    for _ in range(n_samples):
        # Generate a random permutation for each dimension
        sample = np.random.random(n_points)
        # Sort and scale to [0, 1]
        sorted_sample = np.sort(sample)
        scaled_sample = sorted_sample / n_points
        # Add small jitter within bins
        jitter = np.random.random(n_points) / n_points
        # Ensure we don't exceed 1.0
        jitter = np.minimum(jitter, 1.0 - scaled_sample)
        final_sample = scaled_sample + jitter
        samples.append(final_sample)
    return np.array(samples)

def init_population(n_points, n_samples, initial_h=None):
    """Initialize population using LHS and possibly initial_h"""
    population = []
    if initial_h is not None:
        population.append(initial_h.copy())
    
    # Generate additional LHS samples
    lhs_samples = latin_hypercube_sample(n_points, n_samples - len(population))
    for sample in lhs_samples:
        h = project_h(sample, n_points)
        population.append(h)
    
    return population

def evolutionary_phase(population, n_points, budget_s, dx, start_time):
    """Perform evolutionary search on population"""
    n_elites = 20
    n_offspring = 5
    mutation_strength = 0.1
    max_generations = 10000
    generations = 0
    
    while time.time() - start_time < 0.7 * budget_s and generations < max_generations:
        generations += 1
        # Evaluate all individuals
        fitness = [evaluate_h(h, dx) for h in population]
        
        # Select top elites
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:n_elites]]
        
        # Generate offspring
        offspring = []
        for elite in elites:
            for _ in range(n_offspring):
                # Mutate
                child = elite + np.random.normal(0, mutation_strength, n_points)
                child = project_h(child, n_points)
                offspring.append(child)
        
        # Combine and keep best
        all_candidates = elites + offspring
        fitness = [evaluate_h(h, dx) for h in all_candidates]
        sorted_indices = np.argsort(fitness)
        population = [all_candidates[i] for i in sorted_indices[:n_elites]]
        
        if generations % 500 == 0:
            mutation_strength *= 0.95  # Reduce mutation rate over time
    
    return population

def gradient_descent_phase(h, n_points, budget_s, dx, start_time):
    """Refine solution using gradient descent"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    learning_rate = 0.01
    patience = 0
    max_patience = 2000
    
    while time.time() - start_time < 0.3 * budget_s:
        # Get the shift that gives max correlation
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = project_h(initial_h, n_points)
    else:
        # Initialize with sinusoidal pattern
        x = np.linspace(0, 2, n_points)
        initial_h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        initial_h = project_h(initial_h, n_points)
    
    start_time = time.time()
    
    # Initialize population
    population = init_population(n_points, 50, initial_h)
    
    # Evolutionary phase
    population = evolutionary_phase(population, n_points, budget_s, dx, start_time)
    
    # Select the best individual for gradient descent refinement
    fitness = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Gradient descent refinement
    final_h, final_c5 = gradient_descent_phase(best_h, n_points, budget_s, dx, start_time)
    
    return (final_h, final_c5, n_points)
