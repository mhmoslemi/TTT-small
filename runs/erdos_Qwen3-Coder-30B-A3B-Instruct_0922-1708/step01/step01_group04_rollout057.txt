import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def project_h(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Normalize to sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def perturb(h, delta=0.05):
        # Create a perturbed version
        h_new = h.copy()
        # Perturb about 10% of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        return project_h(h_new)
    
    def smooth(h, window=5):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    # Define constraint function
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    prev_c5 = best_c5
    delta = 0.05
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        
        # L-BFGS optimization
        try:
            result = minimize(
                lambda x: evaluate(x),
                best_h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            if result.success:
                h_opt = result.x
                h_opt = project_h(h_opt)
                c5_opt = evaluate(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        except:
            no_improvement_count += 1
            
        # Occasionally perturb and smooth
        if iterations % 10 == 0:
            h_trial = perturb(best_h, delta)
            h_trial = smooth(h_trial, window=5)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                delta = max(0.001, delta * 0.95)  # Adaptively decrease delta
            else:
                no_improvement_count += 1
        else:
            # Minor random perturbation to maintain diversity
            if np.random.random() < 0.1:
                h_trial = perturb(best_h, delta * 0.5)
                c5_trial = evaluate(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    no_improvement_count = 0
    
    return (best_h, best_c5, n_points)
