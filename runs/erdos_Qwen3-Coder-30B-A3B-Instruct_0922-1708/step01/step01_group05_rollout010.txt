        segments = 5
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = start + segment_size if i < segments - 1 else n_points
            if i % 2 == 0:
                h[start:end] = 1.0
            else:
                h[start:end] = 0.0
        # Slightly randomize for perturbation
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        
    h = h * (target_sum / np.sum(h))
    
    # Convert to list of segments for easier manipulation
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # Simulated Annealing parameters
    temp = 1.0
    alpha = 0.99
    max_iters = 50000
    iter_count = 0
    
    best_h = h.copy()
    best_C = compute_C(h)
    
    while time() - start_time < budget_s - 1 and iter_count < max_iters:
        # Generate neighbor solution by perturbing current solution
        h_new = h.copy()
        # Make small random changes
        idx = np.random.randint(n_points)
        change = np.random.normal(0, 0.05)
        h_new[idx] += change
        
        # Keep within bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Compute new C value
        C_new = compute_C(h_new)
        
        # Accept with probability
        if C_new < best_C or np.random.rand() < np.exp((best_C - C_new) / temp):
            h = h_new.copy()
            if C_new < best_C:
                best_C = C_new
                best_h = h.copy()
        
        # Update temperature
        temp *= alpha
        iter_count += 1
    
    # Local refinement with gradient-like method
    max_refine = 200
    iter_count = 0
    lr = 0.01
    no_improvement = 0
    
    while time() - start_time < budget_s - 1 and iter_count < max_refine:

            grad[i] = (C_plus - C_base) / eps

            best_C = C_new
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 50:
                lr *= 0.5
                no_improvement = 0
                
        iter_count += 1
    
    return (best_h.tolist(), best_C, n_points)
