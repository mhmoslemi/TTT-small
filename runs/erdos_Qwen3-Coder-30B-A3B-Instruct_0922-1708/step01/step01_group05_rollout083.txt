    particles = []
    velocities = []
    for _ in range(n_particles):
        # Random perturbation
        pos = np.random.rand(n_points) * 2.0  # Positions in [0,2]
        vel = np.random.rand(n_points) * 0.1 - 0.05  # Small random velocities
        
        # Values between 0 and 1
        val = np.random.rand(n_points)
        val = np.clip(val, 0, 1)
        
        # Rescale to sum constraint
        val = val * (target_sum / np.sum(val))
        
        particles.append((pos, val))
        velocities.append(vel)
    
    # Global best
    gbest_pos = None
    gbest_val = None
    gbest_C = float('inf')
    
    # PSO iterations
    for iter_num in range(n_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for p in range(n_particles):
            pos, val = particles[p]
            vel = velocities[p]
            
            # Evaluate current particle
            h_test = val.copy()
            C = compute_C(h_test)

            if C < gbest_C:

                gbest_pos = pos.copy()
                gbest_val = val.copy()
            
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            vel = (w * vel + 
                   c1 * r1 * (val - pos) + 
                   c2 * r2 * (gbest_val - pos))
            
            # Apply bounds for positions
            new_pos = pos + vel
            new_pos = np.clip(new_pos, 0, 2)
            
            # Update values with linear interpolation
            new_val = val.copy()

                grad[i] = (C_plus - C_base) / eps

    # Final evaluation
    return (h.tolist(), C, n_points)

def compute_C(h_vals):
    # Computes C_5 = max_k integral of h(x)(1-h(x+k))
    h = np.array(h_vals)
    corr = np.correlate(h, 1-h, mode="full")
    dx = 2.0 / len(h)
    return np.max(np.abs(corr)) * dx
