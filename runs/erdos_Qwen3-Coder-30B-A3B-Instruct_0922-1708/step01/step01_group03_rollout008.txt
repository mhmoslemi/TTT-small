import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # If initial_h_values is accessible in this scope, use it
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization
    h = np.ones(n_points) * 0.5
    return h

def particle_swarm_optimization(h, n_points, time_budget_s):
    """Run particle swarm optimization to find a good starting point."""
    start_time = time.time()
    
    swarm_size = 30
    cognitive_weight = 2.0
    social_weight = 2.0
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(swarm_size):
        individual = h.copy()
        individual = project_h(individual, n_points)
        particles.append(individual.copy())
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
        pbest.append(individual.copy())
        pbest_values.append(evaluate_c5(individual))
    
    # Global best
    global_best_idx = np.argmin(pbest_values)
    global_best = pbest[global_best_idx].copy()
    global_best_value = pbest_values[global_best_idx]
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.7:
        iteration += 1
        # Update inertia weight
        if iteration > 100:
            inertia_weight = max(0.4, 0.9 - 0.5 * iteration / 1000)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_weight * velocities[i] +
                             cognitive_weight * r1 * (pbest[i] - particles[i]) +
                             social_weight * r2 * (global_best - particles[i]))
            
            # Keep velocity in bounds
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update particle position
            particles[i] = particles[i] + velocities[i]
            
            # Project to valid constraints
            particles[i] = project_h(particles[i], n_points)
            
            # Evaluate particle
            current_value = evaluate_c5(particles[i])
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_value
        
        if iteration % 50 == 0:
            pass  # Progress update
    
    return global_best, global_best_value

def projected_gradient_descent(h, n_points, time_budget_s):
    """Run projected gradient descent to refine the solution."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.05
    max_no_improvement = 1000
    iter_no_improvement = 0
    
    def compute_gradient(h):
        """Compute gradient of C5 with respect to h."""
        n = len(h)
        dx = 2.0 / n
        # Find the shift that gives maximum value
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)  # shift in range [-n+1, n-1]
        
        # Construct gradient vector for this specific shift
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= i + shift < n:
                grad[i] = -dx
            else:
                # This case needs more careful treatment due to convolution wraparound
                # We recompute for all shifts (more robust but slower)
                pass
        return grad
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.3:
        # Compute gradient (We'll use finite difference instead for robustness)
        eps = 1e-5
        gradient = np.zeros(n_points)
        current_c5 = evaluate_c5(best_h)
        
        for i in range(n_points):
            # Finite difference to estimate gradient
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            gradient[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # GD step
        h_new = best_h - learning_rate * gradient
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Update if improvement
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            
        # Check early termination
        if iter_no_improvement > max_no_improvement:
            break
        
        iteration += 1
        if iteration % 500 == 0:
            learning_rate *= 0.9  # Decay learning rate
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 45  # Keep it small for faster optimization
    h = initialize_h(n_points)
    
    # Phase 1: Particle Swarm Optimization
    h_pso, c5_pso = particle_swarm_optimization(h, n_points, budget_s)
    
    # Phase 2: Projected Gradient Descent
    h_gd, c5_gd = projected_gradient_descent(h_pso, n_points, budget_s)
    
    # Return best result
    final_h = h_gd if c5_gd < c5_pso else h_pso
    final_c5 = min(c5_gd, c5_pso)
    
    # Final evaluation to make sure it's correct
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
