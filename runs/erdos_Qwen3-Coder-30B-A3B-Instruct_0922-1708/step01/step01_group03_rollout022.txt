import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangular pulses
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def line_search(h, grad, eta, n_points):
    """Perform line search to find optimal step size alpha"""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Bracketing
    a, b = 0.0, eta
    fa = objective(a)
    fb = objective(b)
    
    # Ensure a < b and fa >= fb
    if fa < fb:
        a, b = b, a
        fa, fb = fb, fa
    
    # Try to extend the interval
    c = 2 * b
    fc = objective(c)
    while fc < fb:
        a, b, c = b, c, 2*c
        fa, fb, fc = fb, fc, objective(2*c)
    
    # Golden section search on [a, c]
    phi = (1 + np.sqrt(5)) / 2
    resphi = 2 - phi
    
    x1 = resphi * a + phi * c
    x2 = phi * a + resphi * c
    f1 = objective(x1)
    f2 = objective(x2)
    
    while abs(c - a) > 1e-8:
        if f1 < f2:
            c, x2, f2 = x2, x1, f1
            x1 = resphi * a + phi * c
            f1 = objective(x1)
        else:
            a, x1, f1 = x1, x2, f2
            x2 = phi * a + resphi * c
            f2 = objective(x2)
    
    alpha = (a + c) / 2
    return alpha

def optimize_step(h, n_points, eta):
    """Perform one optimization step using exact gradient."""
    # Find shift that maximizes overlap
    dx = 2.0 / n_points
    conv = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(conv)
    shift = max_idx - (n_points - 1)
    
    # Compute the gradient at the worst shift
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + shift) % n_points
        grad[i] = (1 - h[j]) * dx
    
    # Line search for optimal step
    alpha = line_search(h, grad, eta, n_points)
    
    # Update h
    h_new = h - alpha * grad
    h_new = project_h(h_new, n_points)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Learning rate parameters
    eta = 0.1
    eta_min = 1e-6
    eta_max = 0.5
    no_improvement_count = 0
    max_no_improvement = 1000
    
    iterations = 0
    while time.time() - start_time < budget_s and no_improvement_count < max_no_improvement:
        h_new = optimize_step(best_h, n_points, eta)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
            # Increase learning rate if we're making good progress
            eta = min(eta * 1.2, eta_max)
        else:
            no_improvement_count += 1
            # Decrease learning rate if no improvement
            eta = max(eta / 2, eta_min)
        
        iterations += 1
        
        # Occasionally print progress
        if iterations % 100 == 0:
            pass  # Progress update
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    final_h = project_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
