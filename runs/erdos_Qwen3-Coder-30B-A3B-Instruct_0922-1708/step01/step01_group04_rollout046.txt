import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def smooth_h(h, window_size=3):
        if window_size < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window_size)/window_size, mode='same')
        return h_smooth

    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Segment-based optimization
    m = 10  # number of segments
    segment_length = n_points // m
    if segment_length == 0:
        segment_length = 1
    
    # Initialize segment averages
    c = np.ones(m) * 0.5
    c[0] = 0.2
    c[-1] = 0.8
    c = c * (m / 2) / np.sum(c)
    
    def reconstruct_h_from_c(c):
        h = np.zeros(n_points)
        for i in range(m):
            start = i * segment_length
            end = min((i + 1) * segment_length, n_points)
            h[start:end] = c[i]
        return h
    
    def get_gradient(c):
        h = reconstruct_h_from_c(c)
        # Compute correlation gradient with respect to c
        h_corr = np.zeros(n_points)
        for i in range(m):
            start = i * segment_length
            end = min((i + 1) * segment_length, n_points)
            h_corr[start:end] = 1 - h[start:end]  # derivative of -h
        # Gradient computation
        corr_full = correlate(h, h_corr, mode='full')
        indices = np.argmax(corr_full)
        max_corr = corr_full[indices]
        
        # Approximate gradient through finite diff
        grad = np.zeros(m)
        eps = 1e-4
        for i in range(m):
            c_plus = c.copy()
            c_plus[i] += eps
            c_plus = c_plus * (m / 2) / np.sum(c_plus)
            h_plus = reconstruct_h_from_c(c_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) * dx - max_corr) / eps
            
        return grad / np.linalg.norm(grad) if np.linalg.norm(grad) > 1e-10 else grad

    # Trust-region optimization
    trust_radius = 0.1
    no_improvement_count = 0
    best_c = c.copy()
    best_c5 = evaluate(reconstruct_h_from_c(best_c))
    
    start_time = time.time()
    iterations = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        # Update segment averages with trust region
        try:
            grad = get_gradient(best_c)
            delta = -grad * trust_radius
            # Ensure feasibility
            c_proposed = best_c + delta
            c_proposed = np.clip(c_proposed, 0, 1)
            c_proposed = c_proposed * (m / 2) / np.sum(c_proposed)
            
            # Evaluate new solution
            h_proposed = reconstruct_h_from_c(c_proposed)
            c5_proposed = evaluate(h_proposed)
            
            if c5_proposed < best_c5:
                best_c = c_proposed.copy()
                best_c5 = c5_proposed
                no_improvement_count = 0
                trust_radius = min(0.5, trust_radius * 1.2)
            else:
                trust_radius *= 0.5
                no_improvement_count += 1
        except:
            no_improvement_count += 1
            trust_radius *= 0.5

        # Occasionally apply simulated annealing
        if np.random.rand() < 0.05 and no_improvement_count < 10:
            i = np.random.randint(m)
            c[i] += np.random.normal(0, 0.05)
            c = np.clip(c, 0, 1)
            c = c * (m / 2) / np.sum(c)

    # Final reconstruction and smoothing
    h_final = reconstruct_h_from_c(best_c)
    h_final = smooth_h(h_final, window_size=3)
    h_final = project_h(h_final)
    final_c5 = evaluate(h_final)
    
    return (h_final, final_c5, n_points)
