import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a triangular distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to higher resolution
        h_original = h.copy()
        h = np.interp(np.linspace(0, 2, n_points), 
                      np.linspace(0, 2, len(h_original)), h_original)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Start with triangular-like shape
        x = np.linspace(0, 2, n_points)
        h = 1.5 - np.abs(x - 1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient exactly
        # For max correlation, find the index that gives maximum value
        max_idx = np.argmax(correlation)
        # The derivative of max correlation with respect to h[i]
        grad = np.zeros_like(h)
        
        # We are looking at full convolution h * (1 - h)
        # For correlation at k:
        # correlation[k] = sum_i h[i] * (1 - h[i+k])
        # So the gradient wrt h[i] is:
        # if k = max_idx: grad[i] = 1 - h[i + max_idx] if i + max_idx < n_points else 0?  
        # Need to refine this since the maximal correlation index is not directly tied
        # to single h[i] element. Let's compute exact behavior via finite diff or 
        # approximate sign of max partial derivative
        
        # Approximate gradient for each i by finite diff of the overlap at max
        eps = 1e-5
        base_correlation = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_correlation) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
            
        # Update with Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        
        # Update parameters
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale to stay feasible
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
