import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: triangular pulse
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h, shift, n_points):
    """Compute the exact gradient wrt h for a specific shift."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        grad[i] = (1 - h[(i + shift) % n_points])
    return grad

def line_search(h, grad, shift, n_points, eta, dx):
    """Find optimal step size along grad direction using line search."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Try to bracket minima first
    try:
        # Test few alpha values
        alpha_values = np.linspace(0, eta, 20)
        f_values = [objective(a) for a in alpha_values]
        
        # Find min
        min_idx = np.argmin(f_values)
        return alpha_values[min_idx]
    except Exception:
        return eta * 0.5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    
    # Parameters
    eta = 0.1        # Initial learning rate
    eta_min = 1e-6
    eta_max = 0.5
    patience = 1000  # For early stopping
    patience_count = 0
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Optimizer
    iteration = 0
    while time.time() - start_time < budget_s and patience_count < patience:
        iteration += 1
        
        # Find the shift that maximizes overlap
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)  # Convert to actual shift
        
        # Compute exact gradient
        grad = compute_gradient(h, shift, n_points)
        
        # Line search for optimal step size
        alpha = line_search(h, grad, shift, n_points, eta, 2.0 / n_points)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        
        # Update if better
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            patience_count = 0
            # Increase learning rate up to max
            eta = min(eta * 1.2, eta_max)
        else:
            patience_count += 1
            # Decrease learning rate
            eta = max(eta * 0.5, eta_min)
        
        # Log progress every 500 iterations
        if iteration % 500 == 0:
            pass
    
    # Final code validation
    h_final = project_h(h, n_points)
    c5_final = evaluate_c5(h_final)
    
    return (h_final, c5_final, n_points)
