import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize with the parent construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5

    # Normalize to integral = 1 
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Segment-based optimization with variable resolution
    def reconstruct_h_from_segments(c, n_total):
        # c: segment averages, n_total: total points
        m = len(c)
        # Each segment has n_points/m points
        segment_size = n_total // m
        if segment_size * m != n_total:
            segment_size = n_total // m + 1
        
        h = np.repeat(c, segment_size)
        h = h[:n_total]
        # Apply smoothing for continuity
        if segment_size > 1:
            # Gaussian smoothing
            kernel = np.exp(-np.linspace(-2, 2, 5)**2 / 2)
            kernel = kernel / np.sum(kernel)
            h = np.convolve(h, kernel, mode='same')
            h = h[:n_total]
        return np.clip(h, 0, 1)
    
    def optimize_segments(m, h_current, tolerance=1e-6):
        # Optimize m segment averages
        # Constraint: sum(c) = m/2
        def constraint_func(c):
            return np.sum(c) - m / 2
        
        # Objective: minimize max overlap
        def objective(c):
            h_reconstructed = reconstruct_h_from_segments(c, n_points)
            return evaluate(h_reconstructed)
        
        # Bounds for each c
        bounds = [(0, 1)] * m
        
        # Initial guess (uniform distribution)
        c_init = np.ones(m) * (m / 2) / m
        
        # Optimization
        result = minimize(objective, c_init, method='SLSQP', bounds=bounds,
                          constraints={'type': 'eq', 'fun': constraint_func},
                          tol=tolerance, options={'maxiter': 1000})
        
        if result.success:
            return result.x
        return c_init
    
    # Start with a coarse segmentation
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    # Try different segment counts
    segments_to_try = [7, 14, 21, 28, 35, 42, 49, 56, 63]
    segments_to_try = [s for s in segments_to_try if s <= n_points]
    
    # Process segments in order of increasing complexity
    for m in segments_to_try:
        if iteration > 100 or time.time() - start_time > budget_s - 5:
            break
            
        # Get current solution or use the default
        try:
            if iteration == 0:
                c_current = optimize_segments(m, best_h)
            else:
                # Reuse previous solution for warm start
                c_current = np.array([best_h[i:i+int(n_points/m)] for i in range(0, n_points, int(n_points/m))])
                c_current = np.mean(c_current, axis=1)
                c_current = np.clip(c_current, 0, 1)
                # Refine
                if len(c_current) < m:
                    c_current = np.pad(c_current, (0, m - len(c_current)), constant_values=0.5)
                c_current = optimize_segments(m, best_h)
        except:
            c_current = np.ones(m) * 0.5
            c_current = optimize_segments(m, best_h)
            
        iteration += 1
        h_candidate = reconstruct_h_from_segments(c_current, n_points)
        c5_candidate = evaluate(h_candidate)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Fine-grain refinement if time allows
        if time.time() - start_time < budget_s - 10 and no_improvement_count < 5 and m < n_points:
            # Double the resolution by duplicating segments
            new_m = min(m * 2, n_points)
            if new_m > m:
                try:
                    # Create the new segments
                    c_doubled = np.repeat(c_current, 2)
                    if len(c_doubled) > new_m:
                        c_doubled = c_doubled[:new_m]
                    elif len(c_doubled) < new_m:
                        c_doubled = np.pad(c_doubled, (0, new_m - len(c_doubled)), constant_values=0.5)

                    c_doubled = optimize_segments(new_m, best_h)
                    h_fine = reconstruct_h_from_segments(c_doubled, n_points)
                    c5_fine = evaluate(h_fine)
                    if c5_fine < best_c5:
                        best_c5 = c5_fine
                        best_h = h_fine.copy()
                except:
                    pass

    return best_h, best_c5, n_points
