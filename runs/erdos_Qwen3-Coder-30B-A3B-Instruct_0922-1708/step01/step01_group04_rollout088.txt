import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Fallback initial construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Normalize to integral = 1
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def segment_based_optimization(initial_segments, n_points, max_iter=200):
        # Segment-based optimization
        m = len(initial_segments)
        segment_width = n_points // m
        if segment_width * m < n_points:
            segment_width += 1
        
        # Initialize segment values
        segments = np.array(initial_segments, dtype=float)
        
        def reconstruct_h(segments):
            h = np.zeros(n_points)
            for i, s in enumerate(segments):
                start = i * segment_width
                end = min(start + segment_width, n_points)
                h[start:end] = s
            return h
        
        # Refined segment optimization
        def objective(segments):
            h = reconstruct_h(segments)
            return evaluate(h)
        
        # Constraint: sum(segments) * (2/m) = 1 => sum(segments) = m/2
        def constraint_func(segments):
            return np.sum(segments) - m/2
        
        cons = {'type': 'eq', 'fun': constraint_func}
        
        bounds = [(0, 1) for _ in range(m)]
        
        # Optimization loop
        best_segments = segments.copy()
        best_c5 = objective(best_segments)
        
        start_time = time.time()
        no_improvement_count = 0
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            try:
                result = minimize(objective, best_segments, method='SLSQP', 
                                bounds=bounds, constraints=cons, 
                                options={'maxiter': 50, 'ftol': 1e-6})
                if result.success:
                    new_segments = result.x
                    if np.all(new_segments >= 0) and np.all(new_segments <= 1):
                        # Ensure constraint satisfaction
                        sum_segments = np.sum(new_segments)
                        if sum_segments > 0:
                            new_segments = new_segments * (m/2) / sum_segments
                        h_recon = reconstruct_h(new_segments)
                        c5 = evaluate(h_recon)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_segments = new_segments
                            no_improvement_count = 0
                        else:
                            no_improvement_count += 1
                    else:
                        no_improvement_count += 1
                else:
                    no_improvement_count += 1
            except:
                no_improvement_count += 1
            
            if no_improvement_count > 50:
                break
                
            # Occasionally perturb segments
            if iteration % 20 == 0:
                perturb_indices = np.random.choice(m, max(1, m//5), replace=False)
                for i in perturb_indices:
                    delta = np.random.uniform(-0.05, 0.05)
                    new_val = best_segments[i] + delta
                    best_segments[i] = np.clip(new_val, 0, 1)
                # Renormalize
                sum_segments = np.sum(best_segments)
                if sum_segments > 0:
                    best_segments = best_segments * (m/2) / sum_segments
        
        return best_segments, best_c5
    
    # Start with initial construction
    m = min(7, n_points // 3)
    step_size = n_points // m
    if step_size * m < n_points:
        step_size += 1
        
    # Break initial_h_values into segments for optimization
    initial_segments = []
    for i in range(m):
        start = i * step_size
        end = min(start + step_size, n_points)
        avg_h = np.mean(h_start[start:end])
        initial_segments.append(avg_h)
        
    # Normalize initial segments to sum to m/2
    seg_sum = np.sum(initial_segments)
    if seg_sum > 0:
        initial_segments = [x * (m/2) / seg_sum for x in initial_segments]
    
    best_segments, best_c5 = segment_based_optimization(
        initial_segments, n_points, max_iter=150)
    
    # Reconstruct final h
    segment_width = n_points // m
    if segment_width * m < n_points:
        segment_width += 1
    h_final = np.zeros(n_points)
    for i, s in enumerate(best_segments):
        start = i * segment_width
        end = min(start + segment_width, n_points)
        h_final[start:end] = s
    
    # Ensure final constraints
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Final evaluation
    final_c5 = evaluate(h_final)
    
    return (h_final, final_c5, n_points)
