import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h_init = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h_init[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Genetic Algorithm parameters
    pop_size = 100
    generations = 2000
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = pop_size // 10
    
    # Initialize population
    population = []
    if 'h_init' in locals() and len(h_init) == n_points:
        population.append(h_init.copy())
    
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # GA main loop
    for gen in range(generations):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]
        
        # Elitism: keep top individuals
        next_gen = sorted_population[:elite_size]
        
        # Generate offspring
        while len(next_gen) < pop_size:
            # Selection (roulette wheel)
            wheel_fitness = [1.0 / (f + 1e-10) for f in sorted_fitness]
            wheel_fitness = np.array(wheel_fitness)
            wheel_fitness = wheel_fitness / np.sum(wheel_fitness)
            
            parents = np.random.choice(elite_size, size=2, p=wheel_fitness[:elite_size])
            parent1 = sorted_population[parents[0]]
            parent2 = sorted_population[parents[1]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            mutate_prob = mutation_rate / (1.0 + gen**0.5)
            for i in range(n_points):
                if np.random.rand() < mutate_prob:
                    child1[i] += np.random.normal(0, 0.1)
                if np.random.rand() < mutate_prob:
                    child2[i] += np.random.normal(0, 0.1)

            # Project to constraints
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            next_gen.extend([child1, child2])
        
        population = next_gen[:pop_size]
    
    # Final check
    final_h = project_h(best_h) if best_h is not None else project_h(population[0])
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
