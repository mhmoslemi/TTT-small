import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using Simulated Annealing + Gradient Descent"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    step_size = 0.1
    min_T = 1e-6
    
    # GD parameters
    gd_learning_rate = 0.01
    gd_patience = 0
    max_gd_patience = 1000
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    current_h = h.copy()
    current_c5 = best_c5
    
    # Annealing loop
    for iteration in range(max_iter):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Mutation step
        h_new = current_h + np.random.uniform(-step_size, step_size, n_points)
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept or reject based on simulated annealing
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = h_new
            current_c5 = new_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                gd_patience = 0
            else:
                gd_patience += 1
        else:
            gd_patience += 1
            
        # Cool down
        T = max(T * alpha, min_T)
        
        # Occasionally perform gradient descent
        if gd_patience > 100 and time.time() - start_time < 0.9 * budget_s:
            gd_patience = 0
            h = best_h.copy()
            for _ in range(100):
                if time.time() - start_time > 0.9 * budget_s:
                    break
                corr = correlate(h, 1 - h, mode="full")
                max_shift_idx = np.argmax(corr)
                worst_shift = max_shift_idx - (n_points - 1)
                
                # Gradient via finite differences
                eps = 1e-5
                grad = np.zeros(n_points)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = project_h(h_plus, n_points)
                    c_plus = evaluate_h(h_plus, dx)
                    
                    h_minus = h.copy()
                    h_minus[i] -= eps
                    h_minus = project_h(h_minus, n_points)
                    c_minus = evaluate_h(h_minus, dx)
                    
                    grad[i] = (c_plus - c_minus) / (2 * eps)
                
                # Update
                h -= gd_learning_rate * grad
                h = project_h(h, n_points)
                
                # Evaluate
                new_c5 = evaluate_h(h, dx)
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
                    gd_patience = 0
                else:
                    gd_patience += 1
                    
                if gd_patience > max_gd_patience:
                    break
                    
                if gd_learning_rate > 1e-5:
                    gd_learning_rate *= 0.999
    
    # Final gradient descent
    h = best_h.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s:
            break
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= 0.01 * grad
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
    
    return (best_h, best_c5, n_points)
