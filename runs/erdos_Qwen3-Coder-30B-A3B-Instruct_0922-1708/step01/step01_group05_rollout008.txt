import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 2000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
        
    # Rescale to meet integral constraint
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    max_iter_pso = 5000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Initialize particles around the current h with small random perturbations
    for _ in range(n_particles):
        h_p = h + np.random.normal(0, 0.05, n_points)
        h_p = np.clip(h_p, 0, 1)
        h_p = h_p * (target_sum / np.sum(h_p))
        particles.append(h_p.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Best global and particle trackers
    gbest_h = h.copy()
    gbest_C = float('inf')
    pbest_h = [p.copy() for p in particles]
    pbest_C = [float('inf')] * n_particles
    
    def compute_C(h_vals):
        """Compute C_5 = max_k integral of h(x)(1-h(x+k))"""
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def evaluate_and_update():
        nonlocal gbest_C, gbest_h
        C = compute_C(h_vals)
        if C < gbest_C:
            gbest_C = C
            gbest_h = h_vals.copy()
        return C
    
    # PSO iteration
    for iter_pso in range(max_iter_pso):
        if time() - start_time > budget_s - 2:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_h[i] - particles[i]) + 
                           c2 * r2 * (gbest_h - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            # Rescale to meet integral constraint
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            
            # Evaluate current particle
            C = compute_C(particles[i])
            
            # Update personal best
            if C < pbest_C[i]:
                pbest_C[i] = C
                pbest_h[i] = particles[i].copy()
                
        # Update global best
        for i in range(n_particles):
            if pbest_C[i] < gbest_C:
                gbest_C = pbest_C[i]
                gbest_h = pbest_h[i].copy()
    
    # Gradient descent refinement
    best_h_refined = gbest_h.copy()
    best_C_refined = gbest_C
    lr = 0.01
    no_improvement_count = 0
    max_refine_iter = 100
    
    for refine_iter in range(max_refine_iter):
        if time() - start_time > budget_s - 2:
            break
            
        # Compute gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        C_base = compute_C(best_h_refined)
        
        for j in range(n_points):
            h_plus = best_h_refined.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[j] = (C_plus - C_base) / eps
            
        # Update
        best_h_refined -= lr * grad
        best_h_refined = np.clip(best_h_refined, 0, 1)
        best_h_refined = best_h_refined * (target_sum / np.sum(best_h_refined))
        
        C_new = compute_C(best_h_refined)
        if C_new < best_C_refined:
            best_C_refined = C_new
        else:
            no_improvement_count += 1
            if no_improvement_count > 30:
                lr *= 0.8
                no_improvement_count = 0
                
    return (best_h_refined.tolist(), best_C_refined, n_points)
