import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    max_iters = 500
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize particle positions and velocities (flattened h values)
    particles = np.random.rand(n_particles, n_points)
    particles = np.clip(particles, 0, 1)
    for i in range(n_particles):
        particles[i] *= (target_sum / np.sum(particles[i]))
    
    velocities = np.zeros_like(particles)
    
    # Initialize best positions
    best_positions = particles.copy()
    best_values = np.full(n_particles, float('inf'))
    
    # Evaluate initial positions
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    for i in range(n_particles):
        C = compute_C(particles[i])
        if C < best_values[i]:
            best_values[i] = C
            best_positions[i] = particles[i].copy()
    
    # PSO main loop
    iter_count = 0
    while time() - start_time < budget_s - 1 and iter_count < max_iters:
        iter_count += 1
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (best_positions[i] - particles[i]) + 
                            social * r2 * (best_positions[np.argmin(best_values)] - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (target_sum / np.sum(particles[i]))
            
            # Evaluate
            C = compute_C(particles[i])
            if C < best_values[i]:
                best_values[i] = C
                best_positions[i] = particles[i].copy()
    
    # Gradient refinement on best position
    best_h = best_positions[np.argmin(best_values)].copy()
    best_C = np.min(best_values)
    lr = 0.01
    no_improvement_count = 0
    
    def evaluate_gradient(h_vals):
        # Approximate gradient using finite difference
        eps = 1e-5
        grad = np.zeros_like(h_vals)
        base_C = compute_C(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - base_C) / eps
        return grad
    
    # Gradient descent loop
    max_grad_iter = 100
    for _ in range(max_grad_iter):
        if time() - start_time > budget_s - 1:
            break
        h_vals = best_h.copy()
        grad = evaluate_gradient(h_vals)
        best_h -= lr * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        C_new = compute_C(best_h)
        if C_new < best_C:
            best_C = C_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 20:
                lr *= 0.5
                no_improvement_count = 0
    
    return (best_h.tolist(), best_C, n_points)
