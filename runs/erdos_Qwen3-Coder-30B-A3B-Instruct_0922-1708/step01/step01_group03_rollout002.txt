import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric with 2 bumps
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * (x - 0.5))) + 0.5 * (1 + np.sin(np.pi * (x - 1.5)))
    h = project_h(h, n_points)
    return h

def enforce_symmetry(h, n_points):
    """Enforce symmetry h[i] = h[n_points-1-i]."""
    k = n_points // 2
    for i in range(k):
        avg = (h[i] + h[n_points - 1 - i]) / 2
        h[i] = h[n_points - 1 - i] = avg
    return h

def optimize_with_symmetry(n_points, time_budget_s):
    """Optimize using symmetric initialization and gradient descent."""
    start_time = time.time()
    
    # Initialize with symmetric construction
    h = initialize_h(n_points)
    h = enforce_symmetry(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Parameters
    learning_rate = 0.1
    max_no_improvement = 1000
    iter_no_improvement = 0
    eta_min = 1e-6
    eta_max = 1.0
    
    def compute_exact_gradient(h, shift):
        """Compute exact gradient for given shift."""
        n = len(h)
        grad = np.zeros(n)
        # Convolution indices for shift
        idxs = np.arange(n)
        shifted_idxs = (idxs + shift) % n
        # For each element, compute contribution to the gradient of current shift
        for i in range(n):
            # If the shifted index is valid (i+shift >= 0 and < n)
            if 0 <= (i + shift) < n:
                grad[i] = 1 - h[i + shift]
            elif (i + shift) >= n:
                grad[i] = 1 - h[i + shift - n]
        return grad
    
    while time.time() - start_time < time_budget_s:
        # Find the shift causing max overlap
        conv = np.correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(conv)
        shift = max_shift_idx - (n_points - 1)
        
        # Compute exact gradient for that shift
        grad = compute_exact_gradient(h, shift)
        
        # Gradient descent update with learning rate adjustment
        h_new = h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        h_new = enforce_symmetry(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new)
        
        # Update if improvement
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            learning_rate = max(eta_min, learning_rate * 0.8)
        else:
            iter_no_improvement += 1
            learning_rate = min(eta_max, learning_rate * 1.2)
            
        if iter_no_improvement > max_no_improvement:
            break
    
    # Final verification
    final_h = enforce_symmetry(h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return final_h, final_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 45  # Odd to support symmetric structure
    h, c5 = optimize_with_symmetry(n_points, budget_s)
    
    # Ensure final result is valid
    h = enforce_symmetry(h, n_points)
    h = project_h(h, n_points)
    c5 = evaluate_c5(h)
    
    return (h, c5, n_points)
