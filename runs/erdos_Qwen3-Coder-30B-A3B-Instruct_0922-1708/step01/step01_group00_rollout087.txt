from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Set up optimization parameters
    max_iter = 1000
    patience = 50
    patience_count = 0
    prev_best = float('inf')
    lr_b = 0.01  # learning rate for breakpoints
    lr_h = 0.1   # learning rate for values

    # We will use a method to identify regions with potential maximum overlap
    points = np.linspace(0, 2, n_points)
    
    # Initialize structure with reduced granularity
    # Using 5 intervals for a simpler but effective approach
    n_intervals = 5
    interval_bounds = np.linspace(0, 2, n_intervals + 1)
    
    # We represent h as piecewise constant with interval values
    h_values = np.array([0.5] * n_intervals)
    
    # Initial evaluation
    def evaluate_h(h_values, bounds):
        # Reconstruct h from piecewise values
        h = np.zeros(n_points)
        for i in range(n_intervals):
            start_idx = int(bounds[i] / 2 * n_points)
            end_idx = int(bounds[i+1] / 2 * n_points)
            h[start_idx:end_idx] = h_values[i]
        
        # Ensure integral constraint
        total = np.sum(h) * dx
        h = h * (n_points / 2) / total
        
        return h
    
    def compute_c5(h_vals, bounds):
        h_recon = evaluate_h(h_vals, bounds)
        
        # Compute overlap integral as maximum correlation
        correlation = np.correlate(h_recon, 1 - h_recon, mode='full')
        c5 = np.max(correlation) * dx
        
        return c5
    
    # Simplified optimization with manual gradient-like updates
    iteration = 0
    
    # Simple gradient-like optimization with fixed step size
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        c5_current = compute_c5(h_values, interval_bounds)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            # Store the best structure
            best_h = evaluate_h(h_values, interval_bounds)
            patience_count = 0
        else:
            patience_count += 1
            if patience_count > patience:
                break
        
        # Update values using finite difference
        h_new = h_values.copy()
        eps = 1e-4
        
        for i in range(n_intervals):
            h_plus = h_values.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            
            p_plus = interval_bounds.copy()
            c5_plus = compute_c5(h_plus, p_plus)
            
            grad_h = (c5_plus - c5_current) / eps
            
            h_new[i] = max(0, min(1, h_values[i] - lr_h * grad_h))
        
        h_values = h_new
        
        # Update bounds with gradient (approximately)
        bounds_new = interval_bounds.copy()
        for i in range(1, n_intervals):
            if i == 1:

            else:
                bounds_new[i] = max(bounds_new[i-1], min(bounds_new[i+1], bounds_new[i] + lr_b * (1.0 if np.random.random() > 0.5 else -1.0)))
        
        interval_bounds = bounds_new

    return (best_h.tolist(), best_c5, n_points)
