import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def project_to_feasible(h):
        # Clip to bounds
        h = np.clip(h, 0, 1)
        # Normalize to ensure integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def smooth(h, window=5):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def perturb(h, delta=0.05):
        h_new = h.copy()
        # Perturb about 10% of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        # Clip and normalize
        h_new = project_to_feasible(h_new)
        return h_new
    
    def constraint_fun(h):
        return np.sum(h) - n_points / 2
    
    def objective(h):
        return evaluate(h)
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    perturb_delta = 0.05
    max_perturb_delta = 0.1
    
    # Set up constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_fun}
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        
        # Use L-BFGS-B with constraints for local optimization
        result = minimize(
            objective, 
            best_h, 
            method='L-BFGS-B', 
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 10}
        )
        
        if result.success:
            h_opt = result.x
            h_opt = project_to_feasible(h_opt)
            c5_opt = evaluate(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                no_improvement_count = 0
        else:
            # Fallback to perturbation if optimization fails
            pass
        
        # Every 10 iterations, try perturbation to escape local minima
        if iterations % 10 == 0:
            h_trial = perturb(best_h, delta=perturb_delta)
            h_trial = smooth(h_trial, window=5)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                # Reduce perturbation delta if we're making progress
                if perturb_delta > 0.001:
                    perturb_delta *= 0.9
            else:
                no_improvement_count += 1
                # Increase perturbation delta if stuck
                if perturb_delta < max_perturb_delta:
                    perturb_delta *= 1.1
        
        # Occasionally reinitialize the search
        if iterations % 100 == 0:
            h_reinit = np.random.rand(n_points)
            h_reinit = project_to_feasible(h_reinit)
            c5_reinit = evaluate(h_reinit)
            if c5_reinit < best_c5:
                best_c5 = c5_reinit
                best_h = h_reinit.copy()
                no_improvement_count = 0
    
    # Final check
    best_c5 = evaluate(best_h)
    return (best_h, best_c5, n_points)
