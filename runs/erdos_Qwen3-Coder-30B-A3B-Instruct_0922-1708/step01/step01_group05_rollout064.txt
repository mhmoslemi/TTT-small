import numpy as np
import random
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create inspired construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Construct step function with 6 segments
        h = np.zeros(n_points)
        segments = 6
        segment_size = n_points // segments
        for i in range(segments):
            start_idx = i * segment_size
            end_idx = start_idx + segment_size if i < segments - 1 else n_points
            if i % 2 == 0:
                h[start_idx:end_idx] = 1.0
            else:
                h[start_idx:end_idx] = 0.0
        # Ensure sum constraint
        h = h * (target_sum / np.sum(h))
    
    # Helper functions
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def project_h(h_vals):
        # Ensure h is in [0,1] and sum equals target_sum
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals
    
    def get_neighbor(h_vals):
        # Return a neighbor by slightly perturbing one segment
        h_new = h_vals.copy()
        # Choose a random segment and perturb it
        segment_size = n_points // 6
        segment_idx = random.randint(0, 5)
        start_idx = segment_idx * segment_size
        end_idx = start_idx + segment_size if segment_idx < 5 else n_points
        
        # Add small random noise
        noise = np.random.normal(0, 0.05, end_idx - start_idx)
        h_new[start_idx:end_idx] += noise
        h_new = project_h(h_new)
        return h_new
    
    # Simulated Annealing
    current_h = h.copy()
    current_C = compute_C(current_h)
    best_h = current_h.copy()
    best_C = current_C
    
    # Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-6
    iterations = 0
    max_iter = int(50000)
    
    # Adaptive cooling
    no_improvement = 0
    max_no_improvement = 500
    
    while iterations < max_iter and time() - start_time < budget_s - 1:
        # Generate neighbor
        new_h = get_neighbor(current_h)
        new_C = compute_C(new_h)
        
        # Accept or reject
        if new_C < current_C or np.random.rand() < np.exp(-(new_C - current_C) / T):
            current_h = new_h
            current_C = new_C
            
            if new_C < best_C:
                best_h = new_h.copy()
                best_C = new_C
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1
            
        # Cooling schedule
        T = max(min_T, T * alpha)
        if no_improvement > max_no_improvement:
            T = min(1.0, T * 1.2)
            no_improvement = 0
            
        iterations += 1
    
    # Gradient-based refinement (minimal)
    h_refined = best_h.copy()
    lr = 0.01
    for _ in range(200):
        if time() - start_time >= budget_s - 1:
            break
        grad = np.zeros_like(h_refined)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            C_plus = compute_C(project_h(h_plus))
            grad[i] = (C_plus - best_C) / eps
        h_refined = h_refined - lr * grad
        h_refined = project_h(h_refined)
        new_C = compute_C(h_refined)
        if new_C < best_C:
            best_h = h_refined.copy()
            best_C = new_C
        lr *= 0.999
    
    return (best_h.tolist(), best_C, n_points)
