import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric sigmoid-like function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.tanh(2 * (x - 1)))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def symmetric_project(h, n_points):
    """Project h to be symmetric around center."""
    k = (n_points - 1) // 2
    h_sym = h.copy()
    for i in range(k):
        avg = (h[i] + h[n_points - 1 - i]) / 2
        h_sym[i] = h_sym[n_points - 1 - i] = avg
    return h_sym

def gradient_descent_symmetric(h, n_points, time_budget_s, max_iter=None):
    """Run gradient descent with symmetric updates."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.1
    iter_no_improvement = 0
    max_no_improvement = 1000
    if max_iter is None:
        max_iter = 50000
    
    iteration = 0
    while iteration < max_iter and time.time() - start_time < time_budget_s:
        # Find shift that gives maximum overlap
        n = len(best_h)
        dx = 2.0 / n
        conv = np.correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(conv)
        shift_idx = max_idx - (n - 1)
        shift = shift_idx
        
        # Compute exact gradient for the current shift
        grad = np.zeros(n)
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                grad[i] = -dx
            else:
                # Handle wraparound (this should not occur with our full convolution)
                grad[i] = 0  # Shouldn't occur with valid shift
        
        # Apply gradient update maintaining symmetry
        h_new = best_h - learning_rate * grad
        h_new = symmetric_project(h_new, n_points)
        h_new = project_h(h_new, n_points)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            # Reduce learning rate if improvement is observed
            if learning_rate > 1e-6:
                learning_rate *= 0.99
        else:
            iter_no_improvement += 1
            # Increase learning rate if no improvement
            if learning_rate < 1.0:
                learning_rate *= 1.01
            else:
                learning_rate = 0.1  # Reset to default
            
        if iter_no_improvement > max_no_improvement:
            break
            
        iteration += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45  # Odd number for symmetry
    h = initialize_h(n_points)

    h_opt, c5_opt = gradient_descent_symmetric(h, n_points, budget_s)
    
    # Ensure final result is valid and symmetric
    h_opt = symmetric_project(h_opt, n_points)
    h_opt = project_h(h_opt, n_points)
    c5_opt = evaluate_c5(h_opt)
    
    return (h_opt, c5_opt, n_points)
