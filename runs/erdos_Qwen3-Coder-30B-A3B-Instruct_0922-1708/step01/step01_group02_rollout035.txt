        fitness_scores = []

        sorted_indices = np.argsort(fitness_scores)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_scores[i] for i in sorted_indices]

        new_population = sorted_population[:elitism_count]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection with roulette wheel
            def select_parent():
                # Tournament selection
                tournament_size = 3
                tournament_indices = np.random.choice(len(sorted_population), tournament_size)
                tournament_fitness = [sorted_fitness[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                return sorted_population[winner_idx]
            
            parent1 = select_parent()
            parent2 = select_parent()
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            mutation_p = mutation_rate / np.sqrt(generation + 1)
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.rand() < mutation_p:
                        noise = np.random.normal(0, 0.1)
                        child[i] += noise
