    h_vals = np.array([np.mean(h[int(breakpoints[i]*n_points/2):int(breakpoints[i+1]*n_points/2)]) 
                       for i in range(K)])
    
    # Prune zero or near-zero intervals and avoid too small intervals
    # This ensures we optimize only meaningful intervals
    valid_intervals = [i for i in range(K) if abs(h_vals[i]) > 1e-8 and 
                       abs(breakpoints[i+1] - breakpoints[i]) > 1e-6]
    
    # Use the valid intervals
    K_eff = len(valid_intervals)
    if K_eff < 2:
        K_eff = 2
        valid_intervals = list(range(K_eff))
    
    # Optimizer state for intervals (values + breakpoints)
    n_vars = K_eff + K_eff - 1  # K values + (K-1) breakpoints (except first and last fixed)

    start_time = time.time()
    iteration = 0
    
    # Simple gradient-free optimization of the combined function
    while time.time() - start_time < budget_s - 1 and iteration < 1000:
        iteration += 1
        
        # Reconstruct h from piecewise constant intervals
        h = np.zeros(n_points)

        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx

            best_c5 = c5
            best_vals = h_vals.copy()

        h_final = np.zeros(n_points)

            b1 = int(breakpoints[valid_intervals[i] + 1] * n_points / 2)
            if b1 > b0:
                h_final[b0:b1] = best_vals[i]
        
    # Ensure it integrates to 1
    integral = np.sum(h_final) * dx
    h_final = h_final * (n_points / 2) / integral
        
    # Final correlation check
    corr = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_final.tolist(), final_c5, n_points)
