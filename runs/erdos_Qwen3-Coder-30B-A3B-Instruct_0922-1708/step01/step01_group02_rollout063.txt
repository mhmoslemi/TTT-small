import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(conv)
        return max_val * dx
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.7
    elitism_rate = 0.1
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h.copy())
        remaining = population_size - 1
    else:
        remaining = population_size
    
    for _ in range(remaining):
        individual = np.random.rand(n_points)
        individual = project_h(individual)
        population.append(individual)
    population = np.array(population)
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    
    # Evolutionary process
    for gen in range(generations):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = np.array([compute_c5(ind) for ind in population])
        # Lower C5 is better, so we'll use inverse
        fitness = 1.0 / (fitness + 1e-10)
        
        # Track best solution
        min_idx = np.argmin([compute_c5(ind) for ind in population])
        if fitness[min_idx] > fitness[np.argmax(fitness)]:
            current_best = population[min_idx].copy()
            current_c5 = compute_c5(current_best)
            if current_c5 < best_c5:
                best_h = current_best.copy()
                best_c5 = current_c5
        
        # Selection (roulette wheel)
        fitness_sum = np.sum(fitness)
        if fitness_sum > 0:
            probs = fitness / fitness_sum
        else:
            probs = np.ones(len(fitness)) / len(fitness)
        
        selected_indices = np.random.choice(len(population), size=len(population), p=probs)
        selected_population = population[selected_indices]
        
        # Crossover and mutation
        new_population = []
        elite_count = int(elitism_rate * population_size)
        elite_indices = np.argsort(fitness)[-elite_count:]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(new_population) + 1 < population_size:
                idx1, idx2 = np.random.choice(len(selected_population), 2, replace=False)
                parent1, parent2 = selected_population[idx1], selected_population[idx2]
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutate children
                for child in [child1, child2]:
                    if np.random.rand() < mutation_rate / np.sqrt(gen + 1) or gen == 0:
                        noise = np.random.normal(0, 0.1, n_points)
                        child += noise
                        child = project_h(child)
                    new_population.append(child)
            else:
                # Just mutate a random parent
                parent = selected_population[np.random.randint(len(selected_population))]
                child = parent.copy()
                if np.random.rand() < mutation_rate / np.sqrt(gen + 1) or gen == 0:
                    noise = np.random.normal(0, 0.1, n_points)
                    child += noise
                    child = project_h(child)
                new_population.append(child)
        
        population = np.array(new_population[:population_size])
    
    # Final check
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
