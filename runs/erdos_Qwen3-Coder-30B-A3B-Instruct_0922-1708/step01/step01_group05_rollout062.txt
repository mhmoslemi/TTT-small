            start_idx = i * seg_size
            end_idx = start_idx + seg_size if i < segments - 1 else n_points
            # Alternate between 0 and 1
            if i % 2 == 0:
                h[start_idx:end_idx] = 1.0
            else:
                h[start_idx:end_idx] = 0.0
        # Smooth the transitions
        for i in range(1, n_points-1):
            h[i] = 0.5 * (h[i-1] + h[i+1])
        # Normalize sum
        h = h * (target_sum / np.sum(h))
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        # Return max of all positive and negative correlations
        return np.max(np.abs(corr))
    
    # Annealing parameters
    temp = 1.0
    alpha = 0.99
    min_temp = 1e-6
    max_iter = 50000
    
    best_h = h.copy()
    best_C = compute_C(h)
    
    iter_count = 0
    
    # Simulated Annealing
    while iter_count < max_iter and time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Generate neighbor
        h_new = h.copy()
        # Perturb a few points
        num_changes = np.random.randint(1, 10)

            delta = np.random.normal(0, 0.1)
            h_new[idx] += delta

        h_new = h_new * (target_sum / np.sum(h_new))

        C_new = compute_C(h_new)
        C_old = compute_C(h)

        delta = C_new - C_old
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):

            C = C_new
            if C < best_C:

                best_h = h.copy()
        
        # Cool down

                best_h = h_refined.copy()
                patience = 0
            else:
                patience += 1
                if patience > 20:
                    lr *= 0.5
                    patience = 0
    
    return (best_h.tolist(), best_C, n_points)
