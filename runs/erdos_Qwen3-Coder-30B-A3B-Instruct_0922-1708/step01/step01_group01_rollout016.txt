import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using reduced-dimensional search"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h from available construction or use a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Simple initialization
        h = np.ones(n_points) * 0.5
    
    # Reduced dimensional search parameters
    m = 5  # number of segments
    segment_values = [0.0, 0.5, 1.0]
    
    # Get all combinations of segment values
    combinations = list(product(segment_values, repeat=m))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Exhaustive search on combinations
    for combo in combinations:
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Create piecewise constant function from segment values
        segment_length = n_points // m
        h_candidate = np.zeros(n_points)
        
        for i, value in enumerate(combo):
            start_idx = i * segment_length
            end_idx = start_idx + segment_length
            if i == m - 1:
                end_idx = n_points
            h_candidate[start_idx:end_idx] = value
        
        # Project to correct sum
        h_candidate = project_h(h_candidate, n_points)
        
        # Compute C5
        c5_val = evaluate_h(h_candidate, dx)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_candidate.copy()
    
    # Gradient descent refinement of best candidates
    def refine_h(h_init, max_iter=1000):
        h = h_init.copy()
        c5_best = evaluate_h(h, dx)
        learning_rate = 0.01
        patience = 0
        max_patience = 1000
        
        for i in range(max_iter):
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Compute finite difference gradient 
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[j] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[j] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h -= learning_rate * grad
            h = project_h(h, n_points)
            
            # Evaluate
            c5_new = evaluate_h(h, dx)
            if c5_new < c5_best:
                c5_best = c5_new
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    break
                if patience % 500 == 0:
                    learning_rate *= 0.5
                    
        return h, c5_best
    
    # Refinement of best solution
    final_h, final_c5 = refine_h(best_h)
    
    return (final_h, final_c5, n_points)
