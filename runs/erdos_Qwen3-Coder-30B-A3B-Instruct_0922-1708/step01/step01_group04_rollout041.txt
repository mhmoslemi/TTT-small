import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    dx = 2.0 / n_points

    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def segment_averages(h, m):
        """Convert h to m segment averages."""
        n = len(h)
        segment_size = n // m
        if segment_size == 0:
            return h[:m]
        averages = []
        for i in range(m):
            start = i * segment_size
            end = (i + 1) * segment_size if i < m - 1 else n
            averages.append(np.mean(h[start:end]))
        return np.array(averages)
    
    def reconstruct_h(c, n_points):
        """Reconstruct h from segment averages."""
        segment_size = n_points // len(c)
        if segment_size == 0:
            return np.full(n_points, c[0])
        h = np.repeat(c, segment_size)
        if len(h) > n_points:
            h = h[:n_points]
        elif len(h) < n_points:
            h = np.pad(h, (0, n_points - len(h)), mode='edge')
        return h
    
    def smooth_h(h, sigma=1.0):
        """Apply Gaussian smoothing."""
        kernel = np.exp(-0.5 * np.linspace(-3*sigma, 3*sigma, int(6*sigma + 1))**2 / sigma**2)
        kernel = kernel / np.sum(kernel)
        return np.convolve(h, kernel, mode='same')
    
    def optimize_segments(m, h_init):
        """Optimize m segment averages."""
        def objective(c):
            h = reconstruct_h(c, n_points)
            h = smooth_h(h, sigma=1.0)
            return evaluate(h)
        
        def constraint_eq(c):
            return np.sum(c) - m/2
        
        bounds = [(0, 1) for _ in range(m)]
        constraints = {'type': 'eq', 'fun': constraint_eq}
        
        # Initial guess
        c_init = segment_averages(h_init, m)
        if len(c_init) < m:
            c_init = np.pad(c_init, (0, m - len(c_init)), mode='edge')
        
        try:
            result = minimize(objective, c_init, method='SLSQP', bounds=bounds, 
                              constraints=constraints, options={'maxiter': 200})
            if result.success:
                return result.x
        except:
            pass
        return c_init
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    no_improvement_count = 0
    current_m = 7
    
    # Hierarchical approach: start with coarse resolution
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        if current_m > n_points:
            break
            
        # Run optimization on current resolution
        c_opt = optimize_segments(current_m, best_h)
        h_opt = reconstruct_h(c_opt, n_points)
        h_opt = smooth_h(h_opt, sigma=1.0)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        c5_opt = evaluate(h_opt)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Double resolution for next iteration if time allows
        if current_m * 2 <= n_points and time.time() - start_time < budget_s * 0.9:
            current_m *= 2
        else:
            break

    return (best_h, best_c5, n_points)
