import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 10000
    
    # Initialize h from initial_h_values or create flat
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def project_feasible(h_vals):
        # Ensure 0 <= h <= 1
        h_vals = np.clip(h_vals, 0, 1)
        # Adjust to satisfy integral constraint
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals

    # PSO parameters
    n_particles = 100
    max_iter_pso = 500
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocity_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    for _ in range(n_particles):
        particle = h.copy() + np.random.normal(0, 0.01, n_points)
        particle = project_feasible(particle)
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
        pbest.append(particle.copy())
        pbest_vals.append(compute_C(particle))
    
    # Global best
    best_idx = np.argmin(pbest_vals)
    gbest = pbest[best_idx].copy()
    gbest_val = pbest_vals[best_idx]
    
    # PSO loop
    iter_count = 0
    while time() - start_time < max_iter_pso and iter_count < max_iter:
        iter_count += 1
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (pbest[i] - particles[i]) +
                social * r2 * (gbest - particles[i])
            )
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            # Update position
            particles[i] += velocities[i]
            # Project to feasible set
            particles[i] = project_feasible(particles[i])
            # Evaluate
            c = compute_C(particles[i])
            # Update personal best
            if c < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = c
                # Update global best
                if c < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = c
        
        # Early termination if improved enough
        if gbest_val < 0.381:
            break
            
    # Refine with gradient descent
    best_h = gbest.copy()
    best_C = gbest_val
    
    def obj(h_vals):
        # Function to minimize
        return compute_C(h_vals)
        
    def grad(h_vals):
        # Approximate gradient with finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vals)
        base_val = obj(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus)
            grad[i] = (obj(h_plus) - base_val) / eps
        return grad
    
    # LSQ optimization with gradient descent
    opt_h = best_h.copy()
    lr = 0.01
    no_improvement_count = 0
    prev_C = float('inf')
    
    while time() - start_time < budget_s - 1:
        C = obj(opt_h)
        if C < best_C:
            best_C = C
            best_h = opt_h.copy()
            
        if C >= prev_C:
            no_improvement_count += 1
            if no_improvement_count > 20:
                lr *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
            
        prev_C = C
        g = grad(opt_h)
        opt_h -= lr * g
        opt_h = project_feasible(opt_h)
        
        # Early termination
        if best_C < 0.38085:
            break
            
    return (best_h.tolist(), best_C, n_points)
