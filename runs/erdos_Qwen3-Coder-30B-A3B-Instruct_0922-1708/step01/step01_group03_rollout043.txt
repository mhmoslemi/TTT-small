import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smooth symmetric function
    x = np.linspace(0, 2, n_points)
    # Create a symmetric pulse-like function
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient_for_shift(h, shift, n_points):
    """Compute gradient with respect to h for a specific shift."""
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + shift) % n_points
        grad[i] = dx * (1 - h[j])
    return grad

def line_search(h, grad, max_step=0.1):
    """Find optimal step size via simple line search."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Simple grid search for step size
    step_sizes = np.linspace(0, max_step, 20)
    for step in step_sizes:
        h_trial = h - step * grad
        h_trial = project_h(h_trial, len(h))
        c5 = evaluate_c5(h_trial)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_trial.copy()
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    
    # Initialize parameters
    eta = 0.1
    min_eta = 1e-6
    no_improvement_count = 0
    max_no_improvement = 50
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    while time.time() - start_time < budget_s - 2:
        # Find the shift that causes maximum overlap
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)  # this gives shift in range [-n+1, n-1]
        
        # Compute gradient for this shift
        grad = compute_gradient_for_shift(h, shift, n)
        
        # Perform line search to find optimal step size
        h_trial, c5_trial = line_search(h, grad, eta)
        
        # Update if improvement
        if c5_trial < best_c5:
            h = h_trial
            best_c5 = c5_trial
            no_improvement_count = 0
            # Reset learning rate
            eta = 0.1
        else:
            no_improvement_count += 1
            # Reduce learning rate
            eta = max(eta / 2, min_eta)
        
        # Early stopping condition
        if no_improvement_count >= max_no_improvement or eta < min_eta:
            break
    
    # Final evaluation
    final_h = project_h(h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
