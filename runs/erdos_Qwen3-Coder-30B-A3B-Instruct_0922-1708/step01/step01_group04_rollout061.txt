import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    dx = 2.0 / n_points
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)

    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def segment_based_optimization(h, m, max_iter=200):
        h_reconstructed = np.zeros(n_points)
        segment_width = n_points // m
        if segment_width == 0:
            segment_width = 1
        
        # Initialize segment values
        c = np.ones(m) * 0.5
        c[0] = 0.2
        c[-1] = 0.8
        c = np.clip(c, 0, 1)
        c = c * (m / 2) / np.sum(c)
        
        def objective(params):
            # Reconstruct h
            h_test = np.zeros(n_points)
            for i in range(m):
                start_idx = i * segment_width
                end_idx = min((i+1) * segment_width, n_points)
                h_test[start_idx:end_idx] = params[i]
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            return evaluate(h_test)
        
        # Optimization constraints
        def constraint_sum(params):
            return np.sum(params) - m / 2
        
        # Bounds
        bounds = [(0, 1) for _ in range(m)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        options = {'maxiter': max_iter, 'disp': False}
        
        # Initial reconstruction
        h_reconstructed = np.zeros(n_points)
        for i in range(m):
            start_idx = i * segment_width
            end_idx = min((i+1) * segment_width, n_points)
            h_reconstructed[start_idx:end_idx] = c[i]
        h_reconstructed = np.clip(h_reconstructed, 0, 1)
        h_reconstructed = h_reconstructed * (n_points / 2) / np.sum(h_reconstructed)
        
        # Optimize segment values
        res = minimize(objective, c, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        if res.success:
            c = res.x
        else:
            # fallback to initialization if optimization fails
            c = np.clip(c, 0, 1)
            c = c * (m / 2) / np.sum(c)
        
        h_reconstructed = np.zeros(n_points)
        for i in range(m):
            start_idx = i * segment_width
            end_idx = min((i+1) * segment_width, n_points)
            h_reconstructed[start_idx:end_idx] = c[i]
        h_reconstructed = np.clip(h_reconstructed, 0, 1)
        h_reconstructed = h_reconstructed * (n_points / 2) / np.sum(h_reconstructed)
        return h_reconstructed

    def smooth(h, window=3):
        # Apply simple moving average
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def perturb_segment(h, size=10, delta=0.1):
        # Perturb a segment of the histogram
        h_new = h.copy()
        length = len(h_new)
        start_idx = np.random.randint(0, max(1, length - size))
        end_idx = min(start_idx + size, length)
        h_new[start_idx:end_idx] += np.random.uniform(-delta, delta, end_idx - start_idx)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    # Start optimization
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    no_improvement_count = 0
    iteration = 0
    m = 7

    while time.time() - start_time < budget_s and no_improvement_count < 100:
        # Alternate approaches
        if iteration % 2 == 0:
            # Use piecewise optimization
            h_trial = segment_based_optimization(best_h, m, 100)
        else:
            # Use perturbation with smoothing
            h_trial = perturb_segment(best_h, 10, 0.1)
            h_trial = smooth(h_trial, 3)
        
        c5_trial = evaluate(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Periodically increase resolution
        if no_improvement_count > 50 and m < 21:
            m += 2
            best_h = segment_based_optimization(best_h, m, 100)
            best_c5 = evaluate(best_h)
        
        iteration += 1
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final check
    best_c5 = evaluate(best_h)
    # Normalize to make sure it's valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    return (best_h, best_c5, n_points)
