import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def optimize_coordinate(h, n_points, i):
    """Optimize coordinate i by finding the best h[i] value."""
    def objective(h_val):
        h_copy = h.copy()
        h_copy[i] = h_val
        h_copy = project_h(h_copy, n_points)
        return evaluate_c5(h_copy)
    
    # Find the optimal value for h[i] in [0,1]
    result = minimize_scalar(objective, bounds=(0, 1), method='bounded')
    if result.success:
        return result.x
    return h[i]

def coordinate_wise_optimization(h, n_points, max_iter=5000):
    """Perform coordinate-wise optimization."""
    h = h.copy()
    for iteration in range(max_iter):
        if iteration % 1000 == 0:
            pass  # Progress update
        improved = False
        for i in range(n_points):
            old_h_i = h[i]
            new_h_i = optimize_coordinate(h, n_points, i)
            if abs(new_h_i - old_h_i) > 1e-6:
                h[i] = new_h_i
                improved = True
        if not improved:
            break
    return h

def compute_gradient(h, n_points):
    """Compute gradient of C5 with respect to h."""
    # Simplified gradient estimation by finite differences
    eps = 1e-5
    grad = np.zeros(n_points)
    base_c5 = evaluate_c5(h)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def nonlinear_conjugate_gradient(h, n_points, max_iter=5000):
    """Perform nonlinear conjugate gradient optimization."""
    h = h.copy()
    grad = compute_gradient(h, n_points)
    d = -grad
    for iteration in range(max_iter):
        if iteration % 1000 == 0:
            pass  # Progress update

        # Line search for step size
        def objective(alpha):
            h_new = h + alpha * d
            h_new = project_h(h_new, n_points)
            return evaluate_c5(h_new)
            
        # Use scipy to find optimal alpha
        try:
            result = minimize_scalar(objective, bounds=(0, 10), method='bounded')
            alpha = result.x
        except:
            alpha = 0.1
            
        # Update h
        h_new = h + alpha * d
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_c5(h_new)
        
        # Check for convergence
        if abs(c5_new - evaluate_c5(h)) < 1e-8:
            break
            
        h = h_new
        
        # Compute new gradient
        grad_new = compute_gradient(h, n_points)
        
        # Polak-Ribière formula
        beta = max(0, np.dot(grad_new - grad, grad_new) / np.dot(grad, grad))
        
        # Update search direction
        d = -grad_new + beta * d
        grad = grad_new
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    
    # Phase 1: Coordinate-wise optimization
    h = coordinate_wise_optimization(h, n_points, max_iter=3000)
    
    # Phase 2: Nonlinear Conjugate Gradient
    h = nonlinear_conjugate_gradient(h, n_points, max_iter=2000)
    
    # Final evaluation
    final_c5 = evaluate_c5(h)
    
    # Ensure final result is valid
    h = project_h(h, n_points)
    final_c5 = evaluate_c5(h)
    
    return (h, final_c5, n_points)
