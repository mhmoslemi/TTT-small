import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using reduced dimensional search"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Reduced dimensional representation: m segments of constant values
    m = 5  # Number of segments 
    possible_values = [0.0, 0.5, 1.0]
    segment_length = n_points // m
    remainder = n_points % m
    
    # Generate all combinations of segment values
    combinations = list(product(possible_values, repeat=m))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Exhaustive search over segment combinations
    for i, seg_vals in enumerate(combinations):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Create h from segments
        h_candidate = np.zeros(n_points)
        for j, val in enumerate(seg_vals):
            start_idx = j * segment_length + (j if j < remainder else remainder)
            end_idx = start_idx + segment_length + (1 if j < remainder else 0)
            h_candidate[start_idx:end_idx] = val
        
        # Project to satisfy the constraint
        h_candidate = project_h(h_candidate, n_points)
        
        # Evaluate
        c5 = evaluate_h(h_candidate, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
    
    # Gradient descent refinement on top candidates
    # Sort combinations by their score
    candidates = []
    for seg_vals in combinations[:100]:  # Top 100 candidates
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        h_candidate = np.zeros(n_points)
        for j, val in enumerate(seg_vals):
            start_idx = j * segment_length + (j if j < remainder else remainder)
            end_idx = start_idx + segment_length + (1 if j < remainder else 0)
            h_candidate[start_idx:end_idx] = val
        h_candidate = project_h(h_candidate, n_points)
        c5 = evaluate_h(h_candidate, dx)
        candidates.append((c5, h_candidate))
    
    candidates.sort(key=lambda x: x[0])
    top_candidates = candidates[:10]
    
    for c5, h_candidate in top_candidates:
        if time.time() - start_time > budget_s:
            break
        h_refined = h_candidate.copy()
        learning_rate = 0.01
        last_improvement = 0
        
        for step in range(1000):
            if time.time() - start_time > budget_s:
                break
                
            corr = correlate(h_refined, 1 - h_refined, mode="full")
            max_shift_idx = np.argmax(corr)
            worst_shift = max_shift_idx - (n_points - 1)
            
            # Compute gradient via finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            for k in range(n_points):
                h_plus = h_refined.copy()
                h_plus[k] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_refined.copy()
                h_minus[k] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[k] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h_refined -= learning_rate * grad
            h_refined = project_h(h_refined, n_points)
            
            new_c5 = evaluate_h(h_refined, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_refined.copy()
                last_improvement = 0
            else:
                last_improvement += 1
                if last_improvement > 100:
                    break
                    
            if step % 500 == 0 and last_improvement > 50:
                learning_rate *= 0.5
    
    return (best_h, best_c5, n_points)
