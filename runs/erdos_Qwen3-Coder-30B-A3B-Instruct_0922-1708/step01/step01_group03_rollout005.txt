import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Access global initial_h_values if available
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use symmetric construction
    h = np.zeros(n_points)
    # Place a symmetric triangular pulse
    center = n_points // 2
    width = n_points // 3
    for i in range(max(0, center - width), min(n_points, center + width)):
        h[i] = 1.0 - abs(i - center) / width
    # Ensure the sum is correct
    h = project_h(h, n_points)
    return h

def compute_gradient(h, n_points, shift_index):
    """Compute gradient of C5 with respect to h at a specific shift."""
    # Compute correlation for given shift index
    # Return gradient for all positions for a fixed shift key
    dx = 2.0 / n_points
    # Use finite differences to approximate gradient
    eps = 1e-5
    grad = np.zeros(n_points)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus)

        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_h(h_minus, n_points)
        c5_minus = evaluate_c5(h_minus)
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45  # odd for symmetry
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: Particle Swarm Optimization
    swarm_size = 30
    max_iterations = 2000
    cognitive_weight = 2
    social_weight = 2
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(swarm_size):
        particle = h.copy()
        particle = project_h(particle, n_points)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_values.append(evaluate_c5(particle))
    
    # Global best
    best_idx = np.argmin(pbest_values)
    gbest = pbest[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    # PSO iterations
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.7:
            break
        for i in range(swarm_size):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = cognitive_weight * r1 * (pbest[i] - particles[i])
            social = social_weight * r2 * (gbest - particles[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_h(particles[i], n_points)
            
            # Evaluate
            current_value = evaluate_c5(particles[i])
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
        
        # Decrease inertia weight over time
        inertia_weight = 0.9 - (0.5 * iteration / max_iterations)
    
    # Phase 2: Gradient Descent on best solution from PSO
    h_gd = gbest.copy()
    learning_rate = 0.05
    max_no_improvement = 1000
    iter_no_improvement = 0
    best_c5 = gbest_value
    
    while time.time() - start_time < budget_s * 0.95:
        current_c5 = evaluate_c5(h_gd)
        if current_c5 < best_c5:
            best_c5 = current_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
        
        if iter_no_improvement > max_no_improvement:
            break
            
        # Compute gradient for current solution
        # We identify the index of the maximum correlation shift first
        n = len(h_gd)
        dx = 2.0 / n
        conv = np.correlate(h_gd, 1 - h_gd, mode="full")
        best_shift_idx = np.argmax(conv)
        shift = best_shift_idx - (n - 1)
        # Use the gradient computation
        grad = compute_gradient(h_gd, n, shift)
        
        # Update
        h_new = h_gd - learning_rate * grad
        h_new = project_h(h_new, n)
        h_gd = h_new
        
        # Decrease learning rate
        if iter_no_improvement % 100 == 0 and learning_rate > 1e-6:
            learning_rate *= 0.9
    
    # Final evaluation
    final_h = h_gd
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
