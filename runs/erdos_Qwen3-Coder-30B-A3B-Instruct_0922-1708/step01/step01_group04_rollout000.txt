import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available

        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=3):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def project_h(h):
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Re-normalize to satisfy integral constraint
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Define objective and constraint
    def obj(h):
        return evaluate(h)
    
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Optimization parameters
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    
    # Start with a simple gradient-free approach using basin hopping
    # Use scipy.optimize.minimize with L-BFGS-B
    # But since we need to support bounds and constraints, we use trust-constr
    
    # First we try a few different initializations with small perturbations
    current_h = best_h.copy()
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        # Try various random perturbations to escape local optima
        for _ in range(5):
            h_trial = current_h.copy()
            # Random perturbation
            h_trial += np.random.normal(0, 0.02, n_points)
            h_trial = project_h(h_trial)
            # Apply smoothing
            h_trial = smooth(h_trial, window=3)
            h_trial = project_h(h_trial)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                break
        else:
            no_improvement_count += 1
            
        # Alternative: random restarts with better smoothing
        if iterations % 10 == 0 and no_improvement_count > 5:
            # Random restart
            new_h = np.random.uniform(0.1, 0.9, n_points)
            new_h = project_h(new_h)
            # Apply smoothing to get a better baseline
            new_h = smooth(new_h, window=3)
            new_h = project_h(new_h)
            c5_new = evaluate(new_h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = new_h.copy()
                no_improvement_count = 0
                
        # Additional smoothing step
        current_h = smooth(best_h.copy(), window=3)
        current_h = project_h(current_h)
                
    return (best_h, best_c5, n_points)
