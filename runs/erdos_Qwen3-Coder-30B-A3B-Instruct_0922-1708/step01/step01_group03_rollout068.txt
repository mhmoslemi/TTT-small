import numpy as np
from scipy import optimize
import time


def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx


def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h


def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h


def compute_gradient(h, worst_shift):
    """Compute the gradient of C5 w.r.t. h at the worst shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        j = (i + worst_shift) % n
        grad[i] = dx * (1 - h[j])
    return grad


def line_search(h, grad, max_alpha=0.1):
    """Perform line search to find optimal step size alpha."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, len(h_new))
        return evaluate_c5(h_new)
    
    # Use scipy.optimize to minimize over alpha
    result = optimize.minimize_scalar(objective, bounds=(0, max_alpha), method='bounded')
    return result.x if result.success else max_alpha


def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    h = project_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Parameters
    learning_rate = 0.1
    min_lr = 1e-6
    max_iter = 10000
    improvement_threshold = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find worst shift (where maximum overlap occurs)
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        worst_shift = max_idx - (n - 1)
        
        # Compute gradient at worst shift
        grad = compute_gradient(h, worst_shift)
        
        # Line search for optimal step size
        alpha = line_search(h, grad, max_alpha=learning_rate)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            # Reset learning rate
            learning_rate = 0.1
        else:
            # Reduce learning rate if no improvement
            learning_rate *= 0.5
            if learning_rate < min_lr:
                break
                
    return (best_h, best_c5, n_points)
