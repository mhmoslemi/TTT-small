import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    except NameError:
        n_points = 47
        h_init = np.ones(n_points) * 0.5
        mid = n_points // 2
        h_init[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(conv)
        return max_val * dx
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Population-based approach with genetic algorithm
    population_size = 100
    population = []
    for _ in range(population_size):
        h = np.random.rand(n_points) * 0.5 + 0.25  # Random in [0.25, 0.75]
        h = project_h(h)
        population.append(h)
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        population[0] = h_init.copy()
    
    def fitness(h_vec):
        return -compute_c5(h_vec)  # Negative because we minimize
    
    def select_parent(population, fitnesses):
        # Roulette wheel selection
        max_fitness = max(fitnesses)
        if max_fitness == 0:
            # Avoid division by zero
            weights = [1.0] * len(fitnesses)
        else:
            weights = [max_fitness - f + 1e-10 for f in fitnesses]
        # Normalize weights
        weights = np.array(weights)
        weights = weights / np.sum(weights)
        return population[np.random.choice(len(population), p=weights)]
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(h_vec, generation):
        # Mutation rate
        mutation_rate = 0.2 / np.sqrt(generation + 1)
        # Add noise
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1)
                h_vec[i] += noise
        # Project back to constraints
        return project_h(h_vec)
    
    best_h = None
    best_c5 = float('inf')
    
    # Initial evaluation to find best starting point
    for individual in population:
        c5 = compute_c5(individual)
        if c5 < best_c5:
            best_c5 = c5
            best_h = individual.copy()
    
    generation = 0
    while (time.time() - start_time) < budget_s - 1:
        generation += 1
        fitnesses = [fitness(ind) for ind in population]
        
        # Selection
        new_population = []
        sorted_indices = np.argsort(fitnesses)
        # Elitism: keep top 10%
        elite_count = max(1, population_size // 10)
        for i in range(elite_count):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate rest via selection -> crossover -> mutation
        while len(new_population) < population_size:
            parent1 = select_parent(population, fitnesses)
            parent2 = select_parent(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, generation)
            child2 = mutate(child2, generation)
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Update best
        for individual in population:
            c5 = compute_c5(individual)
            if c5 < best_c5:
                best_c5 = c5
                best_h = individual.copy()
    
    return (best_h.tolist(), best_c5, len(best_h))
