import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a symmetric construction based on sine waves
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def gradient_step(h, n_points, eta):
    """Perform one gradient descent step for the worst-case shift."""
    dx = 2.0 / n_points
    # Find the shift that gives the maximum overlap
    conv = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(conv)
    shift = max_idx - (n_points - 1)
    
    # Compute the gradient for the worst shift
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i - shift) % n_points
        grad[i] = dx * (1 - h[j])
    
    # Line search to find optimal step size
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Find optimal alpha using golden section search or simple line search
    alpha_max = eta
    try:
        result = optimize.minimize_scalar(objective, bounds=(0, alpha_max), method='bounded')
        alpha = result.x
    except:
        alpha = alpha_max / 2
    
    # Update h
    h_new = h - alpha * grad
    h_new = project_h(h_new, n_points)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45  # Fixed as per suggestion
    h = initialize_h(n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Main loop
    eta = 0.1  # Initial learning rate
    iter_no_improvement = 0
    max_no_improvement = 1000
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Perform gradient step for the worst-case shift
        h_new = gradient_step(h, n_points, eta)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            iter_no_improvement = 0
            # Reset learning rate
            eta = 0.1  # Reset to initial value
        else:
            iter_no_improvement += 1
            # Reduce learning rate if no improvement
            eta = max(eta / 2, 1e-6)
        
        # Break if no improvement for a while
        if iter_no_improvement > max_no_improvement:
            break
        
        h = h_new
        
        # Periodic update
        if iteration % 100 == 0:
            pass

    # Final projection and evaluation
    final_h = project_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
