from scipy.signal import correlate
import time
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth_h(h, sigma=0.5):
        # Apply Gaussian smoothing
        window = np.arange(-3*sigma, 3*sigma+1)
        kernel = np.exp(-0.5 * (window/sigma)**2)
        kernel = kernel / np.sum(kernel)
        h_smooth = np.convolve(h, kernel, mode='same')
        return h_smooth
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Define optimization variables as segment averages
    n_segments = 15
    # Initialize with better starting values
    if 'initial_h_values' in globals():
        # Use provided initial construction
        h_ref = initial_h_values
        # Convert to segment averages
        segment_indices = np.linspace(0, n_points, n_segments+1, dtype=int)
        segment_averages = []
        for i in range(n_segments):
            start, end = segment_indices[i], segment_indices[i+1]
            seg_mean = np.mean(h_ref[start:end])
            segment_averages.append(seg_mean)
    else:
        segment_averages = [0.5] * n_segments
        
    # Ensure initial sum matches constraint
    avg_sum = sum(segment_averages)
    if abs(avg_sum - n_segments/2) > 1e-10:
        scale_factor = (n_segments/2) / avg_sum
        segment_averages = [x * scale_factor for x in segment_averages]

    def get_h_from_segments(segs):
        # Expand segments to full array
        segment_indices = np.linspace(0, n_points, len(segs)+1, dtype=int)
        h_full = np.zeros(n_points)
        for i in range(len(segs)):
            start, end = segment_indices[i], segment_indices[i+1]
            h_full[start:end] = segs[i]
        return h_full

        h = get_h_from_segments(segs)
        return evaluate(h)

        grad = []
        for i in range(len(segs)):
            segs_plus = segs.copy()
            segs_plus[i] += eps
            segs_minus = segs.copy()
            segs_minus[i] -= eps
            f_plus = compute_objective(segs_plus)
            f_minus = compute_objective(segs_minus)
            grad.append((f_plus - f_minus) / (2 * eps))
        return np.array(grad)

        # Simple trust region step: try to move in direction of gradient
        # without overshooting
        step_size = min(0.1, 1.0 / (1.0 + np.linalg.norm(grad)))
        new_segs = segs - step_size * grad

        new_segs = np.clip(new_segs, 0, 1)

        return new_segs

            scale = (n_segments/2) / sum_segs
            new_segs = [x * scale for x in new_segs]
        
        # Evaluate new solution
        new_c5 = compute_objective(new_segs)
        
        # Decision: accept if better or accept with probability based on temp
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_segs = new_segs.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Optional: anneal temperature
        temp *= temp_decay

        # After a few steps, try perturbing more aggressively
        if iter_count % 20 == 0:
            # Occasionally try a different exploration
            new_segs_temp = [x + (np.random.rand() - 0.5) * temp * 0.2 for x in best_segs]

            if abs(sum_temp - n_segments/2) > 1e-10:

                new_segs_temp = [x * scale for x in new_segs_temp]

                best_c5 = test_c5
                best_segs = new_segs_temp.copy()
                no_improvement_count = 0
    
    # Final reconstruction with smoothing
    final_h = get_h_from_segments(best_segs)
    final_h = smooth_h(final_h)
    final_h = project_h(final_h)
    final_c5 = evaluate(final_h)
    
    return (final_h, final_c5, n_points)
