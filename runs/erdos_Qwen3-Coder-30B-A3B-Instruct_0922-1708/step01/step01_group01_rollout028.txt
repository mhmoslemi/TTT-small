import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Simulated Annealing
    start_time = time.time()
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    step = 0.1
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    current_h = h.copy()
    current_c5 = evaluate_h(current_h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.01
    patience = 0
    max_patience = 1000
    
    for i in range(max_iter):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Simulated Annealing step
        h_new = current_h + np.random.uniform(-step, step, n_points)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        delta = c5_new - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = h_new
            current_c5 = c5_new
            
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        T *= alpha
        
        if i % 100 == 0:
            # Occasionally perform GD step
            if time.time() - start_time > 0.8 * budget_s:
                break
                
            # Compute gradient via finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            for j in range(n_points):
                h_plus = current_h.copy()
                h_plus[j] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = current_h.copy()
                h_minus[j] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[j] = (c_plus - c_minus) / (2 * eps)
            
            # Update with gradient
            current_h -= learning_rate * grad
            current_h = project_h(current_h, n_points)
            current_c5 = evaluate_h(current_h, dx)
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    # Reduce learning rate
                    learning_rate *= 0.5
                    patience = 0
    
    # Final optimization with GD
    current_h = best_h.copy()
    current_c5 = best_c5
    for i in range(5000):
        if time.time() - start_time > budget_s:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = current_h.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[j] = (c_plus - c_minus) / (2 * eps)
        
        # Update with gradient
        current_h -= learning_rate * grad
        current_h = project_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        if i % 1000 == 0:
            learning_rate *= 0.8
    
    return (best_h, best_c5, n_points)
