import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Initialize sinusoidal coefficients (even modes only)
    m = 10
    a = np.random.rand(m) * 0.5
    a[0] = 0.5
    a[1::2] = 0  # Even modes only for symmetry
    
    # Use a faster algorithm: direct optimization on small set of parameters
    def compute_h_from_a(a_coeffs):
        x = np.linspace(0, 2, n_points)
        h = a_coeffs[0] * np.ones_like(x)
        for k in range(1, len(a_coeffs)):
            h += a_coeffs[k] * np.cos(np.pi * k * x / 2)
        return np.clip(h, 0, 1)
    
    def compute_C(h_vals):
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    # Direct optimization using L-BFGS with sub-sampled data
    def objective(params):
        # Reshape parameters to coeffs
        a_coeffs = params.copy()
        a_coeffs[0] = 0.5  # Enforce integral constraint
        h_vals = compute_h_from_a(a_coeffs)
        h_scaled = h_vals * (target_sum / np.sum(h_vals))
        return compute_C(h_scaled)
    
    def solve():
        best_h = h.copy()
        best_C = float('inf')
        iter_count = 0
        
        # Optimization using L-BFGS
        options = {'maxiter': 500, 'disp': False}
        
        # Start from smooth approx to avoid local minima
        for _ in range(5):
            # Random shuffle for robustness
            a_init = np.random.rand(m) * 0.5
            a_init[0] = 0.5
            a_init[1::2] = 0
            
            res = so.minimize(objective, a_init, method='L-BFGS-B', 
                            bounds=[(0, 1)] * m, options=options)
            
            if not res.success:
                continue
                
            a_coeffs = res.x.copy()
            a_coeffs[0] = 0.5
            h_vals = compute_h_from_a(a_coeffs)
            h_scaled = h_vals * (target_sum / np.sum(h_vals))
            C = compute_C(h_scaled)
            
            if C < best_C:
                best_C = C
                best_h = h_scaled.copy()
                
        return best_h, best_C
    
    # Get initial best solution
    h_best, C_best = solve()
    
    # Refinement with binary search
    binary_iter = 0
    while time() - start_time < budget_s - 1:
        # Binary refinement step
        while time() - start_time < budget_s - 1 and binary_iter < 300:
            best_step_h = h_best.copy()
            best_step_C = C_best
            
            # Try random thresholds
            for _ in range(50):
                threshold = np.random.rand() * 0.8 + 0.1
                h_binary = (h_best > threshold).astype(float)
                h_binary = h_binary * (target_sum / np.sum(h_binary))
                C_test = compute_C(h_binary)
                
                if C_test < best_step_C:
                    best_step_C = C_test
                    best_step_h = h_binary.copy()
                    
            if best_step_C < C_best:
                C_best = best_step_C
                h_best = best_step_h.copy()
                
            binary_iter += 1
            
        # Try to optimize the sinusoidal fit once more
        h_best, C_best = solve()
        
    return (h_best.tolist(), C_best, n_points)
