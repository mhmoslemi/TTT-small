import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def segment_based_optimization(h, m=7, iterations=50):
        # Divide h into m segments
        segment_width = n_points // m
        remainder = n_points % m
        
        # Initialize segment averages
        c = np.ones(m) * (m / 2) / m
        if m > 1:
            # Distribute the target sum among segments
            c = np.ones(m) * (m / 2) / m

        def reconstruct_h(c_vals):
            h_reconstructed = np.zeros(n_points)
            for i in range(m):
                start_idx = i * segment_width + min(i, remainder)
                end_idx = start_idx + segment_width + (1 if i < remainder else 0)
                h_reconstructed[start_idx:end_idx] = c_vals[i]
            return h_reconstructed

        def objective(c_vals):
            h_reconstructed = reconstruct_h(c_vals)
            return evaluate(h_reconstructed)

        def constraint_func(c_vals):
            return np.sum(c_vals) - m / 2

        # Optimization
        constraints = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1)] * m

        for _ in range(iterations):
            try:
                res = minimize(objective, c, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 5})
                if res.success:
                    c = res.x
                else:
                    break
            except:
                break

        h_final = reconstruct_h(c)
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        return h_final

    # Main loop
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    no_improvement_count = 0
    max_segments = 21
    current_segments = 7
    iterations = 0

    while time.time() - start_time < budget_s and no_improvement_count < 50:
        iterations += 1
        # Try segment-based optimization
        try:
            h_trial = segment_based_optimization(best_h, m=current_segments, iterations=20)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        except:
            no_improvement_count += 1

        # Occasionally increase resolution
        if iterations % 100 == 0 and current_segments < max_segments:
            current_segments += 2
            # Interpolate from current to higher resolution
            h_interp = np.interp(np.linspace(0, n_points - 1, n_points), np.linspace(0, n_points - 1, len(best_h)), best_h)
            best_h = h_interp
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_c5 = evaluate(best_h)

    # Final evaluation
    final_c5 = evaluate(best_h)
    
    return (best_h, final_c5, n_points)
