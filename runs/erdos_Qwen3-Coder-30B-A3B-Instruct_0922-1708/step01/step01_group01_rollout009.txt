import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO + Gradient Descent"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w = 0.7
    c1 = c2 = 1.5
    inertia_decrease = 0.99
    max_stagnation = 500
    min_swarm_size = 20
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            # 90% from random perturbation of initial, 10% uniform random
            if np.random.rand() < 0.1:
                particle = np.random.rand(n_points)
            else:
                particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_h(particle, n_points)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        score = evaluate_h(particle, dx)
        pbest.append(particle.copy())
        pbest_scores.append(score)
    
    # Global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    start_time = time.time()
    iteration = 0
    stagnation_counter = 0
    last_improvement = 0
    adaptive_swarm_size = swarm_size
    
    while time.time() - start_time < 0.8 * budget_s and iteration < max_iter:
        iteration += 1
        
        # Update swarm size based on stagnation
        if stagnation_counter > max_stagnation and adaptive_swarm_size > min_swarm_size:
            adaptive_swarm_size = max(min_swarm_size, adaptive_swarm_size - 2)
            # Repopulate with new particles
            for i in range(adaptive_swarm_size - len(swarm)):
                if np.random.rand() < 0.1:
                    new_h = np.random.rand(n_points)
                else:
                    new_h = h + np.random.normal(0, 0.05, n_points)
                new_h = project_h(new_h, n_points)
                swarm.append(new_h)
                velocities.append(np.zeros(n_points))
                score = evaluate_h(new_h, dx)
                pbest.append(new_h.copy())
                pbest_scores.append(score)
        
        for i in range(len(swarm)):
            particle = swarm[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocity = (w * velocity + 
                       c1 * r1 * (pbest[i] - particle) + 
                       c2 * r2 * (gbest - particle))
            
            # Update position
            new_particle = particle + velocity
            new_particle = project_h(new_particle, n_points)
            
            # Evaluate
            new_score = evaluate_h(new_particle, dx)
            
            # Update personal best
            if new_score < pbest_scores[i]:
                pbest[i] = new_particle.copy()
                pbest_scores[i] = new_score
                
                # Check if global best
                if new_score < gbest_score:
                    gbest = new_particle.copy()
                    gbest_score = new_score
                    stagnation_counter = 0
                    last_improvement = iteration
                else:
                    stagnation_counter += 1
            
            # Update particle and velocity
            swarm[i] = new_particle
            velocities[i] = velocity
        
        if iteration % 10 == 0:
            w = max(0.3, w * inertia_decrease)
    
    # Refine with gradient descent
    h_refine = gbest.copy()
    best_refined = gbest_score
    initial_lr = 0.01
    lr = initial_lr
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s and patience < max_patience:
        corr = correlate(h_refine, 1 - h_refine, mode="full")
        max_shift_idx = np.argmax(corr)
        shift = max_shift_idx - (n_points - 1)
        
        # Gradient computation for worst case
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_refine.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_refine.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h_refine -= lr * np.sign(grad)
        h_refine = project_h(h_refine, n_points)
        
        new_score = evaluate_h(h_refine, dx)
        if new_score < best_refined:
            best_refined = new_score
            patience = 0
        else:
            patience += 1
            if patience % 1000 == 0:
                lr *= 0.5
                if lr < 1e-6:
                    break
    
    # Return final result
    return (h_refine, best_refined, n_points)
