import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    
    # Use a symmetric initialization
    x = np.linspace(0, 2, n_points)
    # Create two bumps
    h = 0.5 * (1 + np.sin(np.pi * (x - 0.5))) + 0.5 * (1 + np.sin(np.pi * (x - 1.5)))
    h = project_h(h, n_points)
    return h

def enforce_symmetry(h):
    """Enforce symmetry: h[i] = h[n-1-i] for all valid i."""
    n = len(h)
    if n % 2 == 1:
        mid = n // 2
        for i in range(mid):
            avg = (h[i] + h[n-1-i]) / 2
            h[i] = avg
            h[n-1-i] = avg
    else:
        mid = n // 2
        for i in range(mid):
            avg = (h[i] + h[n-1-i]) / 2
            h[i] = avg
            h[n-1-i] = avg
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using symmetric gradient descent."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # odd number ensures symmetry
    h = initialize_h(n_points)
    h = enforce_symmetry(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Main optimization loop
    learning_rate = 0.1
    max_no_improvement = 1000
    iter_no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        # Find the shift that gives max overlap
        conv = np.correlate(best_h, 1 - best_h, mode="full")
        max_shift_idx = np.argmax(conv)
        max_shift = max_shift_idx - (n_points - 1)
        
        # Compute exact gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_shift
            if j >= 0 and j < n_points:
                grad[i] = (1 - best_h[j])
            elif j < 0:
                grad[i] = (1 - best_h[j + n_points])
            else:  # j >= n_points
                grad[i] = (1 - best_h[j - n_points])
        
        # Apply gradient update
        h_new = best_h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        h_new = enforce_symmetry(h_new)
        
        # Check if this helps
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            if learning_rate > 1e-6:
                learning_rate *= 0.95  # Decrease learning rate
        else:
            iter_no_improvement += 1
            learning_rate *= 2.0  # Increase learning rate
            if learning_rate > 1.0:
                learning_rate = 1.0
        
        if iter_no_improvement > max_no_improvement:
            break
            
        iteration += 1
        if iteration % 500 == 0:
            pass  # Progress checkpoint
    
    # Final validation
    final_h = project_h(best_h, n_points)
    final_h = enforce_symmetry(final_h)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
