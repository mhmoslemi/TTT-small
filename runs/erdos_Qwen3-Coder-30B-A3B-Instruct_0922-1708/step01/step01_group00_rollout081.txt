import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to desired resolution if needed
        if len(h) != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, len(h)), h)
            h = h_interp
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Rescale to meet constraint
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    def compute_c5(h_vals):
        # Compute using full correlation with mode='full'
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    def objective(h_vals):
        return compute_c5(h_vals)
    
    def constraint_eq(h_vals):
        # Integral constraint: sum(h) * dx = 1
        return np.sum(h_vals) * dx - 1.0
    
    # Use scipy.optimize with SLSQP
    # Set bounds [0, 1] for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    cons = {'type': 'eq', 'fun': constraint_eq}
    
    # Use a mix of approaches to improve convergence
    # Try several restarts with different sampling strategies
    
    restarts = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 5 and restarts < 10:
        restarts += 1
        current_h = h.copy()
        
        # Amplitude tuning
        if restarts > 1:
            # Randomly modify a few points to escape local optima
            for i in range(n_points // 20):
                idx = np.random.randint(0, n_points)
                current_h[idx] = np.random.random()
            
        # Normalize
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # For better optimization, try to find better local minimum
        opt_result = minimize(objective, current_h, method='SLSQP', 
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 2000, 'ftol': 1e-6})
        
        if opt_result.success:
            h_opt = opt_result.x
            c5_opt = compute_c5(h_opt)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                last_improvement = time.time()
                
        # Update h for next restart from current best
        h = best_h.copy()
        
        # Occasionally try a different sampling
        if restarts % 3 == 0:
            # Add small perturbations
            h += 0.01 * np.random.random(n_points) - 0.005
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Early stopping if no improvement for 300 seconds
        if time.time() - last_improvement > 300 and restarts > 3:
            break
            
    return (best_h.tolist(), best_c5, n_points)
