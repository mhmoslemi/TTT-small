import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create basic construction with clustering
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        # Cluster high values
        h_start[10:20] = 0.9
        h_start[45:55] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=5):
        # Apply smoothing with moving average
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def local_search(h, radius=0.1, iterations=50):
        # Random local search around current solution
        current_h = h.copy()
        current_c5 = evaluate(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        
        for _ in range(iterations):
            h_trial = current_h.copy()
            # Perturb a few points
            indices = np.random.choice(n_points, max(1, int(n_points * 0.1)), replace=False)
            for i in indices:
                h_trial[i] += np.random.uniform(-radius, radius)
            # Clip and normalize
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                
        return best_h, best_c5
    
    def adaptive_search(h, min_radius=0.01, max_radius=0.2):
        # Use progressive refinement
        current_h = h.copy()
        current_c5 = evaluate(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        
        # Try increasing radii
        radii = np.logspace(np.log10(min_radius), np.log10(max_radius), 5)
        for radius in radii:
            h_trial, c5_trial = local_search(current_h, radius, 100)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                
        return best_h, best_c5
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        
        # Perturb the current solution
        h_trial = best_h.copy()
        
        # Do localized perturbations
        # Jiggle some points
        n_jiggle = max(1, int(n_points * 0.15))
        indices = np.random.choice(n_points, n_jiggle, replace=False)
        for i in indices:
            h_trial[i] += np.random.uniform(-0.05, 0.05)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Apply smoothing
        h_trial = smooth(h_trial, window=5)
        
        # Local search to further optimize
        h_opt, c5_opt = adaptive_search(h_trial, 0.01, 0.1)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Occasionally use a more aggressive search
        if iterations % 10 == 0:
            h_rand = np.random.uniform(0, 1, n_points)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
            # Smooth it
            h_rand = smooth(h_rand, window=3)
            h_trial, c5_trial = adaptive_search(h_rand, 0.01, 0.1)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
    
    # Final cleanup and estimation
    # Ensure the final h is normalized exactly to sum = n_points/2
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Apply final smoothing
    final_h = smooth(best_h, window=5)
    final_c5 = evaluate(final_h)
    
    # Double-check and finalize
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate(final_h)
    
    return (final_h, final_c5, n_points)
