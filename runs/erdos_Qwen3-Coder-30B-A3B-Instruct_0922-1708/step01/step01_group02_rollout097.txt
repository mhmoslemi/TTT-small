import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 47
        h = np.ones(n_points) * 0.5
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(conv)
        return max_val * dx

    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec

    h = project_h(h)
    
    # PSO parameters
    n_particles = 50
    max_iter = 10000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    inertia = w_start
    
    # Initialize particles
    particles = [project_h(np.random.rand(n_points) * 0.5 + 0.25) for _ in range(n_particles)]
    velocities = [np.random.uniform(-1, 1, n_points) for _ in range(n_particles)]
    
    # Track best positions
    pbest = [p.copy() for p in particles]
    pbest_scores = [compute_c5(p) for p in particles]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        inertia = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible space
            particles[i] = project_h(particles[i])
            
            # Evaluate fitness
            current_score = compute_c5(particles[i])
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = current_score
                
                # Update global best
                if current_score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = current_score
        
        if iteration % 100 == 0:
            if time.time() - start_time > budget_s - 1:
                break

    # Final check and return
    final_h = project_h(gbest)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
