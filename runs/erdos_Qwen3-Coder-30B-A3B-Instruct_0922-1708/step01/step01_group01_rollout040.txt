import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using hybrid approach"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        h = project_h(h, n_points)
    else:
        # Initialize with simple construction
        h = np.ones(n_points) * 0.5
    
    # Phase 1: Exhaustive search over segment combinations
    # Represent h as piecewise constant with few segments
    m_segments = 5  # Number of segments for reduced search
    segment_values = [0.0, 0.5, 1.0]
    
    # Generate all segment value combinations
    combinations = list(product(segment_values, repeat=m_segments))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try a limited number of combinations based on time budget
    max_combinations = min(100, len(combinations))
    step = max(1, len(combinations) // max_combinations)
    
    for i, combo in enumerate(combinations[::step]):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Construct h from segments
        segment_length = n_points // m_segments
        h_temp = np.zeros(n_points)
        
        for j in range(m_segments):
            start_idx = j * segment_length
            end_idx = min((j + 1) * segment_length, n_points)
            h_temp[start_idx:end_idx] = combo[j]
        
        # Project to meet constraints
        h_temp = project_h(h_temp, n_points)
        
        # Evaluate
        c5 = evaluate_h(h_temp, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_temp.copy()
    
    # Phase 2: Gradient descent refinement of best candidates
    # Start with top candidates from the search
    candidates = [best_h]
    
    # Add some noise to explore nearby solutions
    for _ in range(5):
        if time.time() - start_time > 0.9 * budget_s:
            break
        h_noisy = best_h + np.random.normal(0, 0.01, n_points)
        h_noisy = project_h(h_noisy, n_points)
        candidates.append(h_noisy)
    
    # Refine each candidate
    best_refined_h = best_h.copy()
    best_refined_c5 = best_c5
    
    for h_candidate in candidates:
        h_refined = h_candidate.copy()
        max_patience = 1000
        patience = 0
        learning_rate = 0.01
        
        for _ in range(10000):  # Max iterations per candidate
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Compute gradient via finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_refined.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h_refined -= learning_rate * grad
            h_refined = project_h(h_refined, n_points)
            
            new_c5 = evaluate_h(h_refined, dx)
            if new_c5 < best_refined_c5:
                best_refined_c5 = new_c5
                best_refined_h = h_refined.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    break
            if patience > 0 and patience % 100 == 0:
                learning_rate *= 0.5
                patience = 0
    
    return (best_refined_h, best_refined_c5, n_points)
