import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sample(n_points, n_samples):
    """Generate Latin Hypercube Samples"""
    samples = np.zeros((n_samples, n_points))
    for i in range(n_points):
        # Generate random points in each interval
        intervals = np.linspace(0, 1, n_points + 1)
        lower = intervals[i]
        upper = intervals[i + 1]
        samples[:, i] = np.random.uniform(lower, upper, n_samples)
    return samples

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Phase 1: Multi-start LHS sampling
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Generate LHS samples
    lhs_samples = lhs_sample(n_points, 50)
    for i in range(len(lhs_samples)):
        h_trial = lhs_samples[i]
        h_trial = project_h(h_trial, n_points)
        c5_trial = evaluate_h(h_trial, dx)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial

    # Phase 2: Evolutionary strategy with local refinement
    population_size = 20
    offspring_per_parent = 5
    max_generations = 1000
    
    # Initialize population from best sample + random variants
    population = [best_h.copy()]
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = project_h(h_rand, n_points)
        population.append(h_rand)
        
    for gen in range(max_generations):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Evaluate population
        fitness = [evaluate_h(h, dx) for h in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:population_size // 2]]
        
        # Generate offspring
        offspring = []
        for parent in elites:
            for _ in range(offspring_per_parent):
                child = parent + 0.1 * np.random.randn(n_points)
                child = project_h(child, n_points)
                offspring.append(child)
                
        # New population: elites + offspring
        population = elites + offspring[:population_size - len(elites)]
    
    # Phase 3: Gradient-based refinement
    h = best_h.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 1000
    
    while time.time() - start_time < 0.95 * budget_s:
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
            
        # Update
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 100 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (best_h, best_c5, n_points)
