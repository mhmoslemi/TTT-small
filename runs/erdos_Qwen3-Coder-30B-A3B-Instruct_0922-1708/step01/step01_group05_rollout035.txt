import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create from scratch
    if 'initial_h_values' in globals():

    else:
        # Create a periodic-like construction
        h = np.zeros(n_points)
        # Make it resemble a step function with reduced overlap
        segments = 5
        segment_size = n_points // segments
        for i in range(segments):
            start = i * segment_size
            end = start + segment_size if i < segments - 1 else n_points
            if i % 2 == 0:
                h[start:end] = 1.0
            else:
                h[start:end] = 0.0
        # Normalize
        h = h * (target_sum / np.sum(h))
    
    # Store the best solution found
    best_h = h.copy()
    best_C = float('inf')
    
    def compute_C(h_vals):
        """Compute the maximum overlap integral."""
        h = np.array(h_vals)
        # Cross-correlation of h with (1-h) using full mode
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(np.abs(corr)) * dx

    def project_h(h_vals):
        """Project h to feasible region and renormalize."""
        h_proj = np.clip(h_vals, 0, 1)
        h_proj = h_proj * (target_sum / np.sum(h_proj))
        return h_proj
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-6
    max_iter = 50000
    iter_count = 0
    
    # Start with the initial solution
    current_h = h.copy()
    current_C = compute_C(current_h)
    if current_C < best_C:
        best_C = current_C
        best_h = current_h.copy()
    
    # Temperature decay function
    def update_temp(T_current, iter_num):
        new_T = T_current * alpha
        if new_T < min_T:
            return min_T
        return new_T

    # Main SA loop
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Generate a new candidate
        candidate_h = current_h.copy()
        
        # Make a small perturbation
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.1)
        candidate_h[idx] += delta
        
        # Project candidate to feasible space
        candidate_h = project_h(candidate_h)
        
        # Compute new overlap
        candidate_C = compute_C(candidate_h)
        
        # Compute acceptance probability
        if candidate_C < current_C:
            accept = True
        else:
            # Calculate acceptance probability
            delta_C = candidate_C - current_C
            if T > 0:
                accept_prob = np.exp(-delta_C / T)
                accept = np.random.rand() < accept_prob
            else:
                accept = False
        
        # Accept or reject the move
        if accept:
            current_h = candidate_h
            current_C = candidate_C
        
        # Update best solution
        if current_C < best_C:
            best_C = current_C
            best_h = current_h.copy()
        
        # Update temperature
        T = update_temp(T, iter_count)

    # Gradient-based refinement
    lr = 0.01
    no_improvement_steps = 0
    max_no_improvement = 100
    refinement_steps = 200  # 200 steps for refinement

            break
            
        current_h = best_h.copy()
        
        # Compute gradient with finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        base_C = compute_C(current_h)
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C(h_plus)
            grad[j] = (C_plus - base_C) / eps
        
        # Clip gradient and update
        grad = np.clip(grad, -10, 10)
        new_h = current_h - lr * grad
        new_h = project_h(new_h)
        
        new_C = compute_C(new_h)
        
        if new_C < best_C:
            best_C = new_C
            best_h = new_h.copy()
            no_improvement_steps = 0
        else:
            no_improvement_steps += 1
            
        # Adaptive learning rate
        if no_improvement_steps >= max_no_improvement:
            lr *= 0.5
            no_improvement_steps = 0

    return (best_h.tolist(), best_C, n_points)
