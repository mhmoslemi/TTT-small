import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO + gradient descent"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)

    # PSO initialization
    n_particles = 50
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    # Generate initial population
    for i in range(n_particles):
        if i == 0:
            # Use initial solution for first particle
            particle = h.copy()
        else:
            # Perturb initial solution or random
            if np.random.rand() < 0.1:
                particle = np.random.rand(n_points)
            else:
                particle = h + np.random.normal(0, 0.05, n_points)
        particle = project_h(particle, n_points)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        pbest_positions.append(particle.copy())
        pbest_values.append(evaluate_h(particle, dx))

    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    stagnation_count = 0
    max_stagnation = 500
    min_particles = 20
    
    start_time = time.time()
    
    # PSO iteration
    for iteration in range(max_iter):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] = project_h(particles[i] + velocities[i], n_points)
            
            # Evaluate
            current_value = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_value < gbest_value:
                    gbest_value = current_value
                    gbest_position = particles[i].copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        
        # Adapt swarm size if stagnating
        if stagnation_count > max_stagnation and n_particles > min_particles:
            n_particles = max(min_particles, int(n_particles * 0.8))
            particles = particles[:n_particles]
            velocities = velocities[:n_particles]
            pbest_positions = pbest_positions[:n_particles]
            pbest_values = pbest_values[:n_particles]
            stagnation_count = 0
            # Reinitialize smaller swarm
            for i in range(n_particles, 50):
                particle = h + np.random.normal(0, 0.05, n_points)
                particle = project_h(particle, n_points)
                particles.append(particle)
                velocities.append(np.zeros(n_points))
                pbest_positions.append(particle.copy())
                pbest_values.append(evaluate_h(particle, dx))
            # Update gbest
            gbest_idx = np.argmin(pbest_values)
            gbest_position = pbest_positions[gbest_idx].copy()
            gbest_value = pbest_values[gbest_idx]
    
    # Gradient descent refinement on gbest
    h_refined = gbest_position.copy()
    best_c5 = gbest_value
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s - 1 and patience < max_patience:
        # Find worst overlap
        corr = correlate(h_refined, 1 - h_refined, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient with finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_refined.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with gradient descent (coordinate-wise)
        old_h = h_refined.copy()
        for i in range(n_points):
            h_refined[i] -= learning_rate * np.sign(grad[i])
        
        h_refined = project_h(h_refined, n_points)
        
        new_c5 = evaluate_h(h_refined, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience % 1000 == 0:
                learning_rate *= 0.5
        
        # Early stopping
        if abs(np.sum(old_h - h_refined)) < 1e-8:
            break

    return (h_refined, best_c5, n_points)
