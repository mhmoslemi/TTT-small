    n_particles = 100
    max_iter = 5000
    
    # Initialize particles with node values
    particles = []
    velocities = []
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle_h = np.array(initial_h_values)
            # Perturb a bit
            particle_h += np.random.normal(0, 0.05, n_points)
        else:
            particle_h = np.random.rand(n_points)

        particles.append(particle_h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Global best
    gbest_h = h.copy()
    gbest_C = float('inf')
    
    # PSO parameters
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def project_h(h_vals):
        # Project to [0,1] and enforce integral constraint
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals
    
    # Main PSO loop
    for iter_idx in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        for i, particle_h in enumerate(particles):
            # Evaluate particle
            particle_h = project_h(particle_h)
            C = compute_C(particle_h)
            
            # Update personal best
            if C < gbest_C:
                gbest_C = C
                gbest_h = particle_h.copy()
            
            # Update velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (particle_h - particles[i]) + 
                            c2 * r2 * (gbest_h - particles[i]))
            
            particles[i] += velocities[i]
            
            # Keep within bounds
            particles[i] = np.clip(particles[i], 0, 1)

    ref_start = time.time()

    best_C = gbest_C

        return compute_C(h_vals)

        for i in range(len(h_vals)):

            h_plus = project_h(h_plus)
            C_plus = objective(h_plus)
            grad[i] = (C_plus - base_C) / eps
        return grad
    
    # Gradient descent refinement on best particle
