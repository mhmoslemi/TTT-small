import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create flat function
    if 'initial_h_values' in globals():

    else:

    # Ensure the integral constraint
    h = h * (target_sum / np.sum(h))
    
    # Optimization loop using scipy.optimize.minimize
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def objective(h_vals):
        return compute_C(h_vals)
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    # Bounds for h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization parameters
    max_iter = 1000
    best_C = float('inf')
    best_h = h.copy()
    
    # Use SLSQP method which deals with bounds and constraints
    try:
        result = so.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_iter, 'ftol': 1e-6, 'eps': 1e-5},
            callback=lambda x: None  # Suppress callback
        )
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (target_sum / np.sum(h_opt))
            C_opt = compute_C(h_opt)
            if C_opt < best_C:
                best_C = C_opt
                best_h = h_opt.copy()
    except Exception:
        pass
    
    # Local refinement around the current best solution
    iter_count = 0
    while time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Random perturbation
        perturbation = np.random.normal(0, 0.01, n_points)
        h_pert = best_h + perturbation
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (target_sum / np.sum(h_pert))
        
        # Compute new C
        C_new = compute_C(h_pert)
        
        if C_new < best_C:
            best_C = C_new
            best_h = h_pert.copy()
            
        # Periodically recompute to avoid local minimum
        if iter_count % 100 == 0:
            try:
                result = so.minimize(
                    objective,
                    best_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-5}
                )
                if result.success:
                    h_refined = result.x
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined * (target_sum / np.sum(h_refined))
                    C_refined = compute_C(h_refined)
                    if C_refined < best_C:
                        best_C = C_refined
                        best_h = h_refined.copy()
            except Exception:
                pass

    return (best_h.tolist(), best_C, n_points)
