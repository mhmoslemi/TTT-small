import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h 
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure integral constraint
    h = h * (target_sum / np.sum(h))
    
    # Optimization loop
    best_h = h.copy()
    best_C = float('inf')
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def project_h(h_vals):
        # Clip to [0,1] and rescale to match integral constraint
        h_proj = np.clip(h_vals, 0, 1)
        h_proj = h_proj * (target_sum / np.sum(h_proj))
        return h_proj

    def finite_diff_grad(h_vals, eps=1e-5):
        grad = np.zeros_like(h_vals)
        base_C = compute_C(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - base_C) / eps
        return grad
    
    iter_count = 0
    lr = 0.01
    no_improvement_count = 0
    
    while time() - start_time < budget_s - 1:
        iter_count += 1
        h = project_h(h)
        C = compute_C(h)
        
        if C < best_C:
            best_C = C
            best_h = h.copy()
        
        # Compute gradient
        grad = finite_diff_grad(h)
        
        # Update
        h_new = h - lr * grad
        h = project_h(h_new)
        
        # Adaptive learning rate
        new_C = compute_C(h)
        if new_C >= C:
            no_improvement_count += 1
            if no_improvement_count > 100:
                lr *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
            
    return (best_h.tolist(), best_C, n_points)
