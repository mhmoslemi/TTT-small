import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using reduced-dimension search"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial construction from parent
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Use simple construction as fallback
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
        
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Reduced representation: piecewise constant with 5 segments
    m = 5  # number of segments
    values = [0.0, 0.5, 1.0]  # possible values for segments
    total_combinations = 3 ** m
    
    # Parameters
    max_candidates = min(100, total_combinations)  # limit candidates to prevent timeout
    candidates = []
    
    # Generate combinations
    for combination in product(values, repeat=m):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Create h from segments
        segment_length = n_points // m
        h_candidate = np.zeros(n_points)
        for i in range(m):
            start_idx = i * segment_length
            end_idx = (i + 1) * segment_length if i < m - 1 else n_points
            h_candidate[start_idx:end_idx] = combination[i]
            
        # Project to satisfy constraints
        h_candidate = project_h(h_candidate, n_points)
        
        # Evaluate
        c5 = evaluate_h(h_candidate, dx)
        candidates.append((c5, h_candidate))
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
    
    # Sort candidates and refine top ones
    candidates.sort(key=lambda x: x[0])
    top_candidates = candidates[:min(10, len(candidates))]
    
    # Gradient-based refinement
    for c5_val, h_refine in top_candidates:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        h_tmp = h_refine.copy()
        patience = 0
        lr = 0.01
        
        for _ in range(5000):
            if time.time() - start_time > 0.95 * budget_s:
                break
                
            # Compute gradient via finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h_tmp.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_tmp.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h_tmp -= lr * grad
            h_tmp = project_h(h_tmp, n_points)
            
            # Evaluate
            new_c5 = evaluate_h(h_tmp, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_tmp.copy()
                patience = 0
                lr = 0.01  # reset learning rate
            else:
                patience += 1
                if patience > 500:
                    break  # early stopping
    
    return (best_h, best_c5, n_points)
