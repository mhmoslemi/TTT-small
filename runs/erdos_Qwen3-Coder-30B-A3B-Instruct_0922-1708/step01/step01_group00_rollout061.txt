import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Detect intervals by looking at local gradients
    def detect_intervals(h_vals):
        grad = np.gradient(h_vals)
        # Find sign changes or large magnitude changes
        intervals = []
        current_start = 0
        last_sign = np.sign(grad[0]) if grad[0] != 0 else 0
        for i in range(1, len(grad)):
            sign = np.sign(grad[i]) if grad[i] != 0 else 0
            if sign != last_sign and sign != 0:
                intervals.append((current_start, i))
                current_start = i
            last_sign = sign
        intervals.append((current_start, len(h_vals)))
        return intervals
    
    # Refine h as piecewise constant function with K intervals
    intervals = detect_intervals(h)
    K = min(len(intervals), 10)
    breakpoints = [0]
    for i in range(K-1):
        breakpoints.append(intervals[i][1])
    breakpoints.append(n_points)
    
    # Initialize values for intervals
    h_vals = np.array([np.mean(h[breakpoints[i]:breakpoints[i+1]]) for i in range(K)])
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals * (n_points / 2) / np.sum(h_vals * np.diff(breakpoints))
    
    # Set up the optimization
    def compute_c5(h_vals, breakpoints):
        h = np.zeros(n_points)
        for i in range(K):
            start = breakpoints[i]
            end = breakpoints[i+1]
            h[start:end] = h_vals[i]
            
        # Compute max overlap with correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def objective(params):
        # params = [h1, h2, ..., hK, b1, b2, ..., bK-1]
        h_pieces = params[:K]
        bp = params[K:]
        bp = np.clip(bp, 0, n_points - 1)
        bp = np.sort(bp)
        bp = np.concatenate([[0], bp, [n_points]])
        
        # Compute h using piecewise constant
        h = np.zeros(n_points)
        for i in range(K):
            start = int(bp[i])
            end = int(bp[i+1])
            if end > start:
                h[start:end] = h_pieces[i]
                
        # Rescale to satisfy integral constraint
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def gradient(params):
        # Compute numerical gradient
        eps = 1e-6
        grad = np.zeros_like(params)
        base_obj = objective(params)
        for i in range(len(params)):
            params_p = params.copy()
            params_p[i] += eps
            grad[i] = (objective(params_p) - base_obj) / eps
        return grad
    
    # Initialize optimizer state
    init_params = np.concatenate([h_vals, [(breakpoints[i] + breakpoints[i+1]) / 2 for i in range(K-1)]])
    
    start_time = time.time()
    best_c5 = float('inf')
    best_params = init_params.copy()
    iteration = 0
    
    while time.time() - start_time < budget_s - 2:
        iteration += 1
        try:
            grad = gradient(init_params)
            lr = 0.001
            
            # Update parameters
            init_params = init_params - lr * grad
            
            # Clip / reproject to valid range
            init_params[:K] = np.clip(init_params[:K], 0, 1)
            for i in range(K-1):
                init_params[K+i] = max(0, min(n_points-1, init_params[K+i]))
                
            # Apply constraints
            bp = np.concatenate([[0], np.sort(init_params[K:]), [n_points]])
            h_vals_new = np.array([np.mean(h[bp[i]:bp[i+1]]) for i in range(K)]) * 0.5
            h_vals_new = np.clip(h_vals_new, 0, 1)
            if np.sum(h_vals_new * np.diff(bp)) != 0:
                h_vals_new = h_vals_new * np.sum(h_vals_new * np.diff(bp)) / (n_points / 2)
            
            init_params[:K] = h_vals_new
            
            # Validate solution
            computed_c5 = objective(init_params)
            if computed_c5 < best_c5:
                best_c5 = computed_c5
                best_params = init_params.copy()
                
        except Exception as e:
            break
            
    # Extract final result
    best_h_vals = best_params[:K]
    bp_final = np.concatenate([[0], np.sort(best_params[K:]), [n_points]])
    
    # Reconstruct final h and compute final c5
    final_h = np.zeros(n_points)
    for i in range(K):
        start = int(bp_final[i])
        end = int(bp_final[i+1])
        if end > start:
            final_h[start:end] = best_h_vals[i]
    scale = (n_points / 2) / np.sum(final_h)
    final_h = final_h * scale
    final_c5 = compute_c5(best_h_vals, bp_final.astype(int).tolist())
        
    return (final_h.tolist(), final_c5, n_points)
