import numpy as np
from scipy import optimize
import time

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangle
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using gradient-based method with line search."""
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45  # Optimize with smaller number for speed
    
    # Initialize h
    h = initialize_h(n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Gradient-based optimization with line search
    eta = 0.1  # Initial learning rate
    max_iter = 10000
    iter_no_improvement = 0
    max_no_improvement = 1000
    min_eta = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.99:
            break
            
        # Find current worst shift
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        worst_shift = max_idx - (n - 1)  # Convert to actual shift
        
        # Compute gradient: dC5/dh[i] = dx * (1 - h[i + k mod n])
        grad = np.zeros(n)
        for i in range(n):
            j = (i - worst_shift) % n  # Index for the term (1 - h[i + k])
            grad[i] = dx * (1 - h[j])
        
        # Line search for optimal step size
        def objective(alpha):
            h_new = h - alpha * grad
            h_new = project_h(h_new, n_points)
            return evaluate_c5(h_new)
        
        # Find alpha using golden section search or simple adaptive approach
        alpha_max = eta
        if abs(np.linalg.norm(grad)) > 1e-12:
            # Use line search
            try:
                result = optimize.minimize_scalar(objective, bounds=(0, alpha_max), method='bounded')
                alpha = result.x
                if result.success:
                    alpha = max(0, min(alpha_max, alpha))
                else:
                    alpha = alpha_max / 2
            except:
                alpha = alpha_max / 2
        else:
            alpha = alpha_max / 2
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Accept update if improvement
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            if eta < 1:  # Increase learning rate if previously reduced
                eta *= 1.1
        else:
            iter_no_improvement += 1
            if iter_no_improvement > 50:
                eta *= 0.5
                iter_no_improvement = 0
                if eta < min_eta:
                    break
        
        # Ensure eta does not go below threshold
        if eta < min_eta:
            eta = min_eta
    
    # Final check and return
    final_h = project_h(h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
