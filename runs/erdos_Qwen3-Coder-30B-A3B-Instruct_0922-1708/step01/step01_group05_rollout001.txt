import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    max_iter = 500
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    # Particle positions (as parameters for piecewise linear h)
    particles = np.random.rand(n_particles, n_points)
    particles = np.clip(particles, 0, 1)
    velocities = np.random.rand(n_particles, n_points) * 0.1 - 0.05
    pbest = particles.copy()
    pbest_fitness = np.full(n_particles, float('inf'))
    
    # Global best
    gbest_idx = 0
    gbest_fitness = float('inf')
    
    def compute_h_from_params(params):
        h_values = params.copy()
        h_values = np.clip(h_values, 0, 1)
        # Rescale to meet integral condition
        current_sum = np.sum(h_values) * dx
        if current_sum > 0:
            h_values = h_values * (target_sum / current_sum)
        return h_values
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def evaluate_particle(pos):
        h_vals = compute_h_from_params(pos)
        C = compute_C(h_vals)
        return C
    
    # PSO loop
    for iter_count in range(max_iter):
        if time() - start_time > budget_s - 5:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            fitness = evaluate_particle(particles[i])
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest[i] = particles[i].copy()
            
            if fitness < gbest_fitness:
                gbest_fitness = fitness
                gbest_idx = i
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) +
                           c2 * r2 * (pbest[gbest_idx] - particles[i]))
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
    
    # Refinement with gradient descent
    ref_start_time = time()
    best_h = compute_h_from_params(pbest[gbest_idx])
    best_C = gbest_fitness
    
    # Simple gradient-based refinement
    lr = 0.01
    max_refine_iter = 200
    no_improvement_count = 0
    
    def finite_diff_grad(h_vals, eps=1e-5):
        grad = np.zeros_like(h_vals)
        base_C = compute_C(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = compute_h_from_params(h_plus)
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - base_C) / eps
        return grad
    
    for iter_count in range(max_refine_iter):
        if time() - ref_start_time > 100 or time() - start_time > budget_s - 2:
            break
            
        grad = finite_diff_grad(best_h)
        best_h -= lr * grad
        best_h = compute_h_from_params(best_h)
        
        # Check for convergence
        new_C = compute_C(best_h)
        if new_C < best_C:
            best_C = new_C
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 20:
                lr *= 0.5
                no_improvement_count = 0
    
    return (best_h.tolist(), best_C, n_points)
