import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO Parameters
    swarm_size = 50
    max_iter = 1000
    w = 0.7
    c1 = c2 = 1.5
    inertia = w
    max_stagnation = 500
    stagnation_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    for i in range(swarm_size):
        if i == 0:
            particle_h = h.copy()
        else:
            # Random from initial or small perturbation
            if np.random.rand() < 0.1:
                particle_h = np.random.rand(n_points)
            else:
                particle_h = h + np.random.normal(0, 0.05, n_points)
            particle_h = project_h(particle_h, n_points)
        particles.append(particle_h)
        velocities.append(np.zeros(n_points))
        score = evaluate_h(particle_h, dx)
        pbest.append(particle_h.copy())
        pbest_scores.append(score)
        if score < best_c5:
            best_c5 = score
            best_h = particle_h.copy()
    
    gbest = best_h.copy()
    gbest_score = best_c5
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < 0.9 * budget_s and iteration < max_iter:
        iteration += 1
        for i in range(swarm_size):
            particle_h = particles[i]
            velocity = velocities[i]
            particle_pbest = pbest[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocity = (inertia * velocity + 
                        c1 * r1 * (particle_pbest - particle_h) + 
                        c2 * r2 * (gbest - particle_h))
            
            # Update position
            new_h = particle_h + velocity
            new_h = project_h(new_h, n_points)
            
            # Evaluate new position
            new_score = evaluate_h(new_h, dx)
            
            # Update personal best
            if new_score < pbest_scores[i]:
                pbest[i] = new_h.copy()
                pbest_scores[i] = new_score
                if new_score < gbest_score:
                    gbest = new_h.copy()
                    gbest_score = new_score
                    stagnation_count = 0
                    if new_score < best_c5:
                        best_c5 = new_score
                        best_h = new_h.copy()
            else:
                stagnation_count += 1
            
            # Update particle
            particles[i] = new_h
            velocities[i] = velocity
            
        # Adapt swarm size if stagnating
        if stagnation_count > max_stagnation and swarm_size > 20:
            swarm_size = max(20, int(swarm_size * 0.8))
            # Reinitialise swarm
            new_particles = particles[-swarm_size:]
            new_velocities = velocities[-swarm_size:]
            new_pbest = pbest[-swarm_size:]
            new_pbest_scores = pbest_scores[-swarm_size:]
            for i in range(swarm_size):
                if i >= len(new_particles):
                    if np.random.rand() < 0.1:
                        new_h = np.random.rand(n_points)
                    else:
                        new_h = h + np.random.normal(0, 0.05, n_points)
                    new_h = project_h(new_h, n_points)
                    new_particles.append(new_h)
                    new_velocities.append(np.zeros(n_points))
                    score = evaluate_h(new_h, dx)
                    new_pbest.append(new_h.copy())
                    new_pbest_scores.append(score)
                    if score < gbest_score:
                        gbest = new_h.copy()
                        gbest_score = score
            particles = new_particles
            velocities = new_velocities
            pbest = new_pbest
            pbest_scores = new_pbest_scores
            
        # Adjust inertia
        if iteration % 100 == 0:
            inertia = max(0.4, w * 0.95)
    
    # Final gradient phase
    h = gbest.copy()
    best_h = h.copy()
    best_c5 = gbest_score
    max_iter = 5000
    lr = 0.01
    patience = 0
    max_patience = 5000
    
    for i in range(max_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient component-wise
        eps = 1e-5
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[j] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= lr * grad
        h = project_h(h, n_points)
        
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience >= max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            lr *= 0.5
            patience = 0

    return (best_h, best_c5, n_points)
