import scipy.optimize as so
from time import time

    start_time = time()

    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():

    else:

    # PSO Parameters
    n_particles = 100
    max_iter = 1000
    w = 0.7  # Inertia weight
    c1 = 1.5  # Cognitive parameter
    c2 = 1.5  # Social parameter
    
    # Particle positions and velocities (as parameters for piecewise linear h)
    particles = np.random.rand(n_particles, n_points)
    velocities = np.random.rand(n_particles, n_points) * 0.01
    pbest_positions = particles.copy()
    pbest_scores = np.full(n_particles, float('inf'))
    
    # Global best
    gbest_score = float('inf')
    gbest_position = None
    
    # PSO main loop
    for iter_count in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
            
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to satisfy integral constraint
            h_candidate = particles[i] * (target_sum / np.sum(particles[i]))
            
            # Calculate C_5
            corr = np.correlate(h_candidate, 1-h_candidate, mode="full")
            corr *= dx
            score = np.max(np.abs(corr))
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = particles[i].copy()
                
            # Update global best
            if score < gbest_score:
                gbest_score = score
                gbest_position = particles[i].copy()
    
    # Do final gradient-based refinement on the best candidate
    if gbest_position is not None:
        h_refined = gbest_position * (target_sum / np.sum(gbest_position))
        
        def objective(h_vals):
            corr = np.correlate(h_vals, 1-h_vals, mode="full")
            corr *= dx
            return np.max(np.abs(corr))
            
        # Simple gradient descent refinement
        lr = 0.01
        for _ in range(1000):
            if time() - start_time > budget_s - 1:
                break
            h_vals = np.clip(h_refined, 0, 1)
            
            # Trick: approximate the gradient via finite differences
            grad = np.zeros_like(h_vals)
            eps = 1e-5
            base_score = objective(h_vals)
            
            for j in range(len(h_vals)):
                h_plus = h_vals.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (target_sum / np.sum(h_plus))
                score_plus = objective(h_plus)
                grad[j] = (score_plus - base_score) / eps
            
            # Update with gradient descent
            h_refined -= lr * grad
            h_refined = np.clip(h_refined, 0, 1)
            
            # Ensure integral constraint
            h_refined = h_refined * (target_sum / np.sum(h_refined))
            
            # Adaptive learning rate
            if lr > 1e-6:
                lr *= 0.999
            
        # Final evaluation
        final_h = np.clip(h_refined, 0, 1)
        final_h = final_h * (target_sum / np.sum(final_h))
        final_score = objective(final_h)
        
        return (final_h.tolist(), final_score, n_points)
    
    # Return best from PSO if no refinement was done
    final_h = np.clip(gbest_position, 0, 1)
    final_h = final_h * (target_sum / np.sum(final_h))
    corr = np.correlate(final_h, 1-final_h, mode="full")
    corr *= dx
    final_score = np.max(np.abs(corr))
    
    return (final_h.tolist(), final_score, n_points)
