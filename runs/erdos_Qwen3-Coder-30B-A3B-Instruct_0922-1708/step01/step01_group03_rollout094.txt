import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Use initial_h_values if available
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangular pulse
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using gradient-based approach with line search."""
    np.random.seed(seed)
    n_points = 45
    h = initialize_h(n_points)
    
    start_time = time.time()
    
    # Initial evaluation
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Method parameters
    eta = 0.1      # Initial learning rate
    min_eta = 1e-6  # Minimum learning rate
    max_iter = 10000
    improvement_threshold = 1e-6
    iter_no_improvement = 0
    max_no_improvement = 2000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.99:
            break
            
        # Compute the full convolution
        conv = np.correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute gradient we're interested in
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + shift) % n_points
            grad[i] = (1 - best_h[j])
        
        # Compute gradient magnitude for normalization
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            break
        
        # Normalize gradient
        grad = grad / grad_norm
        
        # Line search for optimal step size
        def line_search_objective(alpha):
            h_try = best_h - alpha * grad
            h_try = project_h(h_try, n_points)
            return evaluate_c5(h_try)
        
        def backtrack_line_search():
            alpha = eta
            current_c5 = line_search_objective(0)
            # Try decreasing alpha until we find an improvement
            while alpha > min_eta:
                new_c5 = line_search_objective(alpha)
                if new_c5 < current_c5:
                    return alpha
                alpha *= 0.5
            return alpha
        
        alpha = backtrack_line_search()
        
        # Update h and project
        h_new = best_h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Update best if improved
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            
        # Adaptive learning rate adjustment
        if iter_no_improvement > 50:
            eta = max(eta / 2, min_eta)
            iter_no_improvement = 0
            
        if iter_no_improvement > max_no_improvement:
            break
            
        if abs(new_c5 - best_c5) < improvement_threshold:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
