import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Initialize PSO parameters
    n_particles = 50
    max_iter = 10000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        particle = project_h(np.random.rand(n_points) * 0.5 + 0.25)
        particles.append(particle.copy())
        velocities.append(np.random.rand(n_points) * 2 - 1)
        pbest.append(particle.copy())
        pbest_vals.append(compute_c5(particle))
    
    # Global best
    best_idx = np.argmin(pbest_vals)
    gbest = pbest[best_idx].copy()
    gbest_val = pbest_vals[best_idx]
    
    # PSO main loop
    iteration = 0
    no_improvement_count = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = project_h(particles[i])
            
            # Evaluate fitness
            current_val = compute_c5(particles[i])
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Check if new global best
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
        
        iteration += 1
        
        # Early termination if no improvement for many iterations
        if no_improvement_count > 1000:
            break
    
    # Final check
    final_c5 = compute_c5(gbest)
    
    return (gbest.tolist(), final_c5, len(gbest))
