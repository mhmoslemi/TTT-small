import numpy as np
from scipy import optimize
import time

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangle function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using gradient-based minimax approach."""
    np.random.seed(seed)
    time_start = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    
    # Initialize parameters
    eta = 0.1  # Initial learning rate
    min_eta = 1e-6
    max_iter = 10000
    min_improvement = 1e-6
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Gradient computation function for a specific shift
    def compute_gradient(h, k):
        """Compute gradient component for shift k."""
        n = len(h)
        dx = 2.0 / n
        grad = np.zeros(n)
        for i in range(n):
            j = (i + k) % n
            grad[i] = dx * (1 - h[j])
        return grad
    
    # Line search for optimal step size
    def line_search(h, grad, eta_max):
        """Perform line search to find optimal step size."""
        def objective(alpha):
            h_new = h - alpha * grad
            h_new = project_h(h_new, len(h_new))
            return evaluate_c5(h_new)
        
        # Search in range [0, eta_max]
        try:
            result = optimize.minimize_scalar(objective, bounds=(0, eta_max), method='bounded')
            alpha = result.x
        except:
            # Fallback if optimization fails
            alpha = eta_max / 2
        return alpha
    
    iteration = 0
    while time.time() - time_start < budget_s * 0.95 and iteration < max_iter:
        iteration += 1
        
        # Find the shift that maximizes overlap
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        best_shift = max_idx - (n - 1)  # shift in range [-n+1, n-1]
        
        # Compute gradient for current best shift
        grad = compute_gradient(h, best_shift)
        
        # Perform line search
        alpha = line_search(h, grad, eta)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        
        # Update if improvement
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            eta = min(eta * 2, 1.0)  # Increase learning rate
        else:
            eta = max(eta / 2, min_eta)  # Decrease learning rate
            
        # Stopping condition
        if abs(best_c5 - new_c5) < min_improvement:
            break
    
    return (best_h, best_c5, n_points)
