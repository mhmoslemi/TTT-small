import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sample(n_points, n_samples):
    """Generate Latin Hypercube Sample"""
    samples = []
    for _ in range(n_samples):
        sample = np.random.rand(n_points)
        # Shuffle each dimension independently
        for i in range(n_points):
            sample[i] = np.random.rand()
        samples.append(sample)
    return samples

    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or create sinusoidal

        h = project_h(h, n_points)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Phase 1: Latin Hypercube Sampling
    n_samples = 50
    n_elites = 20
    
    # Generate LHS samples
    lhs_samples = lhs_sample(n_points, n_samples)
    candidates = []
    
    for sample in lhs_samples:
        h_candidate = project_h(sample, n_points)
        c5 = evaluate_h(h_candidate, dx)
        candidates.append((h_candidate, c5))
    
    # Sort candidates by C5
    candidates.sort(key=lambda x: x[1])
    elites = candidates[:n_elites]
    
    # Phase 2: Evolution Strategy
    n_offspring = 5
    max_gen = 5000
    gen = 0
    
    while time.time() - start_time < 0.6 * budget_s and gen < max_gen:
        gen += 1
        new_candidates = []
        
        for h_elite, _ in elites:
            for _ in range(n_offspring):
                # Create offspring by small mutation
                h_mut = h_elite + np.random.normal(0, 0.05, n_points)
                h_mut = project_h(h_mut, n_points)
                c5 = evaluate_h(h_mut, dx)
                new_candidates.append((h_mut, c5))
        
        # Merge elites and offspring
        all_candidates = elites + new_candidates
        all_candidates.sort(key=lambda x: x[1])
        elites = all_candidates[:n_elites]
        
        # Update best
        if elites[0][1] < best_c5:
            best_h = elites[0][0].copy()
            best_c5 = elites[0][1]
    
    # Phase 3: Gradient Descent
    h = best_h.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s - 10:
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Approximate gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with gradient step direction
        h -= learning_rate * np.sign(grad)
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:

            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (best_h, best_c5, n_points)
