import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values.copy()
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=5):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def project_to_feasible(h):
        # Clip to bounds
        h = np.clip(h, 0, 1)
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def perturb(h, delta=0.05):
        h_new = h.copy()
        # Perturb about 10% of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        return project_to_feasible(h_new)
    
    def objective(h):
        return evaluate(h)
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_func}
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    perturb_delta = 0.05
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        # Use L-BFGS-B to minimize with constraints
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            h_trial = result.x
        else:
            # Fallback to naive perturbation if optimization fails
            h_trial = perturb(best_h, delta=perturb_delta)
        
        h_trial = project_to_feasible(h_trial)
        h_trial = smooth(h_trial, window=5)
        c5_trial = evaluate(h_trial)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            no_improvement_count = 0
            perturb_delta = max(0.01, perturb_delta * 0.99)
        else:
            no_improvement_count += 1
        
        # Occasionally perturb to escape local minima
        if iterations % 10 == 0:
            h_perturbed = perturb(best_h, delta=perturb_delta)
            h_perturbed = smooth(h_perturbed, window=5)
            c5_perturbed = evaluate(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                no_improvement_count = 0
                perturb_delta = max(0.01, perturb_delta * 0.99)
    
    return (best_h, best_c5, n_points)
