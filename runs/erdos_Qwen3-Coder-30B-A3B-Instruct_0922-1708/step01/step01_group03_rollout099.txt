import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h, k, n_points):
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + k) % n_points
        grad[i] = 1 - h[j]
    return grad

def line_search(h, grad, n_points, max_step=0.5):
    def c5_along_line(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Bracketing search to find suitable alpha
    alpha_min, alpha_max = 0.0, max_step
    f_min = c5_along_line(alpha_min)
    f_max = c5_along_line(alpha_max)
    
    if f_min <= f_max:
        # Local minimum at alpha_min, just return it
        return alpha_min
    
    while f_max < f_min:
        alpha_min *= 2
        f_min = f_max
        alpha_max = alpha_min * 2
        f_max = c5_along_line(alpha_max)
    
    # Use golden section search for precision
    phi = (1 + np.sqrt(5)) / 2
    resphi = 2 - phi
    
    a, b = 0.0, alpha_max
    v = a + resphi * (b - a)
    w = b - resphi * (b - a)
    fv = c5_along_line(v)
    fw = c5_along_line(w)
    
    for _ in range(100):  # Limit iterations
        if abs(b - a) < 1e-10:
            break
        if fv < fw:
            b = w
            w = v
            v = a + resphi * (b - a)
            fv = fw
            fw = c5_along_line(v)
        else:
            a = v
            v = w
            w = b - resphi * (b - a)
            fw = fv
            fv = c5_along_line(w)
    
    return (a + b) / 2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    h = initialize_h(n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    eta = 0.1
    eta_min = 1e-6
    eta_max = 0.5
    max_no_improvement = 1000
    no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        # Find key correlation shift that gives maximum overlap
        conv = np.correlate(best_h, 1 - best_h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute exact gradient for that shift
        grad = compute_gradient(best_h, shift, n_points)
        
        # Line search for optimal step size
        alpha = line_search(best_h, grad, n_points, eta)
        if alpha == 0.0:
            break
            
        h_new = best_h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improvement = 0
            eta = min(eta * 1.2, eta_max)
        else:
            no_improvement += 1
            if no_improvement > max_no_improvement:
                break
            eta = max(eta / 2, eta_min)
            if eta < eta_min:
                break
    
    # If we found a better solution, use it; otherwise return initial
    if no_improvement < max_no_improvement and iteration > 1:
        final_h = best_h
        final_c5 = evaluate_c5(final_h)
    else:
        final_h = h
        final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
