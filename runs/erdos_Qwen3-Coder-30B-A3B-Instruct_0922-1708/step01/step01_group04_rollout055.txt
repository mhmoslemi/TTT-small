import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def objective(h):
        return evaluate(h)
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Define constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Initialize best solution
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    no_improvement_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iteration += 1
        
        # L-BFGS-B optimization step
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 100})
        
        if result.success:
            h_new = result.x
        else:
            # If optimization failed, just use a small perturbation
            h_new = best_h.copy() + np.random.normal(0, 0.01, n_points)
        
        # Enforce bounds and normalization
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5 value
        c5_new = evaluate(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Occasionally perturb to escape local minima
        if iteration % 10 == 0:
            h_perturbed = best_h.copy()
            # Perturb a portion of points
            num_perturb = max(1, n_points // 10)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for i in indices:
                h_perturbed[i] += np.random.uniform(-0.05, 0.05)
            
            # Normalize and clip
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            c5_perturbed = evaluate(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                no_improvement_count = 0
    
    return (best_h, best_c5, n_points)
