import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create flat
    if 'initial_h_values' in globals():

    else:
        h = np.ones(n_points) * 0.5
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Optimization loop (Projected Gradient Descent)
    best_h = h.copy()
    best_C = float('inf')
    no_improvement_count = 0
    lr = 0.01
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def compute_gradient(h_vals):
        # Compute gradient w.r.t. h for current worst case shift
        # Find worst shift k
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        k_max = np.argmax(np.abs(corr))
        # Gradient w.r.t h for this shift
        grad = np.zeros_like(h_vals)
        # For shift k_max, derivative of h[i]*(1-h[i+k_max]) w.r.t h[i] is (1-h[i+k_max]) 
        # and w.r.t. h[i+k_max] is -h[i]
        for i in range(len(h_vals)):
            if i + k_max < len(h_vals):
                grad[i] += (1 - h_vals[i + k_max])
            if i - k_max >= 0:
                grad[i] -= h_vals[i - k_max]
        return grad * dx
        
    def project_and_scale(h_vals):
        # Clip h_vals to [0,1] and rescale to meet integral constraint
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals
    
    def evaluate_and_update():
        nonlocal best_C, best_h
        C = compute_C(h_vals)
        if C < best_C:
            best_C = C
            best_h = h_vals.copy()
        return C
    
    # Evaluate initial point
    previous_C = evaluate_and_update()
    
    iter_count = 0
    while time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute gradient for current solution
        grad = compute_gradient(h)
        # Update h in direction of negative gradient
        h_new = h - lr * grad
        h_new = project_and_scale(h_new)
        
        # Estimate new value using linear approximation with gradient
        C_new_estimate = previous_C - lr * np.sum(grad * (h_new - h))
        
        # If improvement is not seen with the linear estimate, revert to projected gradient 
        if C_new_estimate >= previous_C:
            # Do a line search to find better step size (simplified version)
            best_step = 0.0
            best_C_val = previous_C
            for step in np.linspace(0.0, 0.1, 11):
                h_trial = h - step * grad
                h_trial = project_and_scale(h_trial)
                C_trial = compute_C(h_trial)
                if C_trial < best_C_val:
                    best_C_val = C_trial
                    best_step = step
            if best_step > 0:
                h = h - best_step * grad
                h = project_and_scale(h)
                C_new = best_C_val
            else:
                h = h_new
                C_new = compute_C(h)
        else:
            h = h_new
            C_new = C_new_estimate
            
        # Adaptive learning rate
        if C_new >= previous_C:
            no_improvement_count += 1
            if no_improvement_count > 100:
                lr *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
        previous_C = C_new
        
    return (best_h.tolist(), best_C, n_points)
