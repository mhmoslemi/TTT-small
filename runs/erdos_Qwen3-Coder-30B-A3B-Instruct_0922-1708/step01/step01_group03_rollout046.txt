import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:

    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient(h, k, n_points):
    """Compute the gradient dC5/dh_i for the specific shift k."""
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    for i in range(n_points):
        grad[i] = (1 - h[(i + k) % n_points]) * dx
    return grad

def line_search(h, grad, k, n_points, eta):
    """Perform line search to find optimal step size."""
    def obj(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Bracket search
    a, b = 0.0, eta
    c = 2.0 * b
    
    # Find an interval [a,b] where obj(a) >= obj(b) <= obj(c)
    while obj(b) > obj(c):
        a, b = b, c
        c = 2.0 * c
    
    # Optimize in [a,b], with obj(a) >= obj(b) <= obj(c)
    res = minimize_scalar(obj, bounds=(a, b), method='bounded')
    return res.x

        # Compute gradient for that shift
        grad = compute_gradient(best_h, shift, n_points)

        alpha = line_search(best_h, grad, shift, n_points, eta)

        h_new = best_h - alpha * grad
