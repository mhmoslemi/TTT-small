import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=3):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def project(h):
        # Project h to feasible region
        h = np.clip(h, 0, 1)
        # Rescale to meet constraint
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def perturb(h, delta=0.05):
        # Create a perturbed version
        h_new = h.copy()
        # Perturb about 20% of points
        n_perturb = max(1, int(n_points * 0.2))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        return project(h_new)
    
    def objective(h):
        return evaluate(h)
    
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Set up optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    patience = 50
    
    # Gradient-based optimization with resetting
    while time.time() - start_time < budget_s and no_improvement_count < patience:
        iterations += 1
        
        # Try gradient-based optimization
        try:
            result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                              options={'maxiter': 10})
            if result.success:
                h_candidate = project(result.x)
                c5_candidate = evaluate(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        except:
            no_improvement_count += 1
        
        # Occasionally perturb to avoid local minimum
        if iterations % 10 == 0 and no_improvement_count > 5:
            h_perturbed = perturb(best_h)
            h_smoothed = smooth(h_perturbed, window=5)
            c5_perturbed = evaluate(h_smoothed)
            if c5_perturbed < best_c5:
                best_h = h_smoothed
                best_c5 = c5_perturbed
                no_improvement_count = 0
        
        # Random restart occasionaly
        if no_improvement_count > 20 and time.time() - start_time < budget_s * 0.9:
            # Initialize with random feasible solution
            h_new = np.random.rand(n_points)
            h_new = project(h_new)
            c5_new = evaluate(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                no_improvement_count = 0
    
    # Final check
    final_h = project(best_h)
    final_c5 = evaluate(final_h)
    
    return (final_h, final_c5, n_points)
