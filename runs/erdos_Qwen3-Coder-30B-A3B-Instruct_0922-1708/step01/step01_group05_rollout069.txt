import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Fourier basis setup
    m = 10  # Number of cosine modes (even symmetry)
    a = np.random.rand(m) * 0.5
    a[0] = 0.5
    a[1::2] = 0  # Only even modes for symmetry
    phi = np.random.rand(m) * 2 * np.pi
    
    # Optimization loop
    best_h = h.copy()
    best_C = float('inf')
    no_improvement_count = 0
    lr = 0.01
    
    def compute_h_from_a_phi(a_coeffs, phi_coeffs):
        # Build h(x) from cosine coefficients
        x = np.linspace(0, 2, n_points)
        h = a_coeffs[0] * np.ones_like(x)
        for k in range(1, len(a_coeffs)):
            h += a_coeffs[k] * np.cos(np.pi * k * x / 2 + phi_coeffs[k])
        return np.clip(h, 0, 1)
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def evaluate_and_update():
        nonlocal best_C, best_h
        h_vals = compute_h_from_a_phi(a, phi)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        C = compute_C(h_vals)
        if C < best_C:
            best_C = C
            best_h = h_vals.copy()
        return C
    
    # Evaluate original starting point
    previous_C = evaluate_and_update()
    
    iter_count = 0
    while time() - start_time < budget_s - 1:
        iter_count += 1
        h_vals = compute_h_from_a_phi(a, phi)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        C = compute_C(h_vals)
        
        # Compute gradient approx using finite differences
        grad_a = np.zeros_like(a)
        grad_phi = np.zeros_like(phi)
        eps = 1e-5
        
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus = compute_h_from_a_phi(a_plus, phi)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad_a[i] = (C_plus - C) / eps
        
        for i in range(len(phi)):
            phi_plus = phi.copy()
            phi_plus[i] += eps
            h_plus = compute_h_from_a_phi(a, phi_plus)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad_phi[i] = (C_plus - C) / eps
            
        # Gradient update
        a -= lr * grad_a
        phi -= lr * grad_phi
        
        # Clip to maintain bounds
        a = np.clip(a, 0, 1)
        phi = phi % (2 * np.pi)
        
        # Rescale to preserve integral
        a[0] = 0.5
        
        # Adaptive learning rate
        new_C = evaluate_and_update()
        if new_C >= previous_C:
            no_improvement_count += 1
            if no_improvement_count > 100:
                lr *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
        previous_C = new_C
        
    # Final refinement using threshold-based binary search
    h_final = compute_h_from_a_phi(a, phi)
    h_final = h_final * (target_sum / np.sum(h_final))
    
    # Do simple threshold optimization
    def threshold_objective(threshold):
        h_bin = (h_final > threshold).astype(float)
        h_bin = h_bin * (target_sum / np.sum(h_bin))  # normalize
        if np.sum(h_bin) == 0:
            return float('inf')
        return compute_C(h_bin)
    
    # Grid search for optimal threshold
    best_threshold = 0.5
    best_bin_C = compute_C((h_final > 0.5).astype(float))
    step_size = 0.05
    for t in np.arange(0, 1, step_size):
        h_test = (h_final > t).astype(float)
        h_test = h_test * (target_sum / np.sum(h_test))
        C_test = compute_C(h_test)
        if C_test < best_bin_C:
            best_bin_C = C_test
            best_threshold = t
    
    final_h = (h_final > best_threshold).astype(float)
    final_h = final_h * (target_sum / np.sum(final_h))
    
    # Final check
    C_final = compute_C(final_h)
    final_h = np.clip(final_h, 0, 1)
    
    if C_final < best_C:
        best_C = C_final
        best_h = final_h.copy()
    
    return (best_h.tolist(), best_C, n_points)
