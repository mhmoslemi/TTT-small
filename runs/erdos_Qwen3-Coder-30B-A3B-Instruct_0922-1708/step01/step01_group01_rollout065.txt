import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with uniform values
        h = np.ones(n_points) * 0.5
    
    # Segment-based search
    m = 5  # number of segments
    segment_values = [0.0, 0.5, 1.0]
    total_candidates = 3 ** m
    
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Generate combinations of segment values
    segment_combinations = []
    def generate_combinations(current, remaining):
        if remaining == 0:
            segment_combinations.append(current.copy())
            return
        for v in segment_values:
            current.append(v)
            generate_combinations(current, remaining - 1)
            current.pop()
    
    generate_combinations([], m)
    
    # Process combinations
    for i, seg_vals in enumerate(segment_combinations):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Build h from segments
        h_candidate = np.zeros(n_points)
        segment_length = n_points // m
        remainder = n_points % m
        
        pos = 0
        for j, val in enumerate(seg_vals):
            length = segment_length + (1 if j < remainder else 0)
            h_candidate[pos:pos+length] = val
            pos += length
        
        h_candidate = project_h(h_candidate, n_points)
        c5 = evaluate_h(h_candidate, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
    
    # Refinement with gradient descent
    h_refined = best_h.copy()
    h_refined = project_h(h_refined, n_points)
    best_c5_refined = evaluate_h(h_refined, dx)
    
    # Adaptive gradient descent
    learning_rate = 0.01
    patience = 0
    max_patience = 1000
    
    # Check if we have time for refinement
    if time.time() - start_time < 0.9 * budget_s:
        while time.time() - start_time < 0.95 * budget_s:
            corr = correlate(h_refined, 1 - h_refined, mode="full")
            max_shift_idx = np.argmax(corr)
            worst_shift = max_shift_idx - (n_points - 1)
            
            # Compute gradient by finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for idx in range(n_points):
                h_plus = h_refined.copy()
                h_plus[idx] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_refined.copy()
                h_minus[idx] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[idx] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h_refined -= learning_rate * grad
            h_refined = project_h(h_refined, n_points)
            
            new_c5 = evaluate_h(h_refined, dx)
            if new_c5 < best_c5_refined:
                best_c5_refined = new_c5
                best_h = h_refined.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    break
            if patience > 0 and patience % 500 == 0:
                learning_rate *= 0.5
                patience = 0
    
    return (best_h, best_c5_refined, n_points)
