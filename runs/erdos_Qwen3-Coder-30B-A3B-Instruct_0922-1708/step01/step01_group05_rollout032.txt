from time import time

    start_time = time()

    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 5000

    else:

    # PSO parameters
    n_particles = 100
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_vel = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    for _ in range(n_particles):
        h_p = h.copy() + np.random.normal(0, 0.05, n_points) * (1 - 2 * np.random.random(n_points))
        h_p = np.clip(h_p, 0, 1)
        h_p = h_p * (target_sum / np.sum(h_p))  # normalize
        particles.append(h_p.copy())
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
    
    # Global best
    g_best = h.copy()
    g_best_C = float('inf')
    
    def compute_C(h_vals):
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        return np.max(np.abs(corr)) * dx
        
    # PSO main loop
    for _ in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
        for i in range(n_particles):
            # Evaluate fitness
            C = compute_C(particles[i])
            if C < g_best_C:
                g_best_C = C
                g_best = particles[i].copy()
                
            # Update personal best
            if C < compute_C(particles[i]):
                # Already using current
                pass
            
            # Update velocity
            r1, r2 = np.random.random(2), np.random.random(2)
            for j in range(n_points):
                velocities[i][j] = (
                    inertia * velocities[i][j] +
                    cognitive * r1[j] * (particles[i][j] - particles[i][j]) +
                    social * r2[j] * (g_best[j] - particles[i][j])
                )
                velocities[i][j] = np.clip(velocities[i][j], -max_vel, max_vel)
            
            # Update position
            for j in range(n_points):
                particles[i][j] += velocities[i][j]
                particles[i][j] = np.clip(particles[i][j], 0, 1)
            
            # Normalize
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
    
    # Final refinement with gradient-based method for best particles
    best_particles = []
    for i in range(n_particles):
        C = compute_C(particles[i])
        best_particles.append((C, particles[i].copy()))
    best_particles.sort()
    
    best_h = g_best.copy()  # Start with global best
    best_C = g_best_C
    
    # Try gradient descent on several top PSO results
    for i in range(min(10, len(best_particles))):
        h_ref = best_particles[i][1]
        
        # Simple gradient-based refinement
        lr = 0.01
        for _ in range(200):
            if time() - start_time > budget_s - 1:
                break
            # Compute gradient using finite differences
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (target_sum / np.sum(h_plus))
                C_plus = compute_C(h_plus)
                
                h_minus = h_ref.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (target_sum / np.sum(h_minus))
                C_minus = compute_C(h_minus)
                
                grad[j] = (C_plus - C_minus) / (2*eps)
            
            # Update step
            h_ref = h_ref - lr * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (target_sum / np.sum(h_ref))
            
            C_new = compute_C(h_ref)
            if C_new < best_C:
                best_C = C_new
                best_h = h_ref.copy()
                
            # Reduce learning rate if no improvement
            if C_new >= compute_C(h_ref):
                lr *= 0.95
            if lr < 1e-6:
                break
                
    return (best_h.tolist(), best_C, n_points)
