import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    stagnation_limit = 500
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Generate initial particles
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            # Add small random noise to initial solution
            particle = h + np.random.normal(0, 0.02, n_points)
        particle = project_h(particle, n_points)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_values.append(evaluate_h(particle, dx))
    
    # Global best
    best_idx = np.argmin(pbest_values)
    gbest = pbest[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    # PSO iteration
    start_time = time.time()
    iter_count = 0
    stagnation = 0
    
    while time.time() - start_time < 0.7 * budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = project_h(particles[i] + velocities[i], n_points)
            
            # Evaluate and update personal best
            current_value = evaluate_h(particles[i], dx)
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
                    stagnation = 0
        
        # Adaptive swarm size
        if stagnation > stagnation_limit:
            n_particles = max(20, int(n_particles * 0.8))
            stagnation = 0
        else:
            stagnation += 1
            
        # Occasionally refresh swarm
        if iter_count % 100 == 0:
            for i in range(n_particles // 5):
                j = np.random.randint(0, n_particles)
                particles[j] = project_h(h + np.random.normal(0, 0.05, n_points), n_points)
                pbest[j] = particles[j].copy()
                pbest_values[j] = evaluate_h(particles[j], dx)
    
    # Gradient descent refinement
    h = gbest.copy()
    best_h = h.copy()
    best_c5 = gbest_value
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < 0.9 * budget_s and patience < max_patience:
        # Find worst shift
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update step
        h -= learning_rate * np.sign(grad)
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience % 1000 == 0:
                learning_rate *= 0.5
                patience = 0
    
    return (best_h, best_c5, n_points)
