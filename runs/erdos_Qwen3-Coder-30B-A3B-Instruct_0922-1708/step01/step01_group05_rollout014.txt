import numpy as np
import scipy.optimize as so
from time import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create a better starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Try a more structured approach: sine-based with proper normalization
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(np.pi * x)  # Start with sinusoidal
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    # Helper functions
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def project_to_feasible(h_vals):
        # Ensure h values are in [0,1] and sum to target_sum
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals
    
    def anneal_step(h_vals, temperature):
        # Generate neighbor by slightly perturbing a random segment
        h_new = h_vals.copy()
        idx = random.randint(0, n_points - 1)
        # Perturb a value with small noise
        h_new[idx] += random.gauss(0, 0.05)
        # Ensure it stays within bounds
        h_new[idx] = np.clip(h_new[idx], 0, 1)
        return project_to_feasible(h_new)
    
    # Simulated Annealing section
    current_h = h.copy()
    current_C = compute_C(current_h)
    best_h = current_h.copy()
    best_C = current_C
    
    temp = 1.0
    alpha = 0.999
    cooldown = 0.999
    max_iter = int(50000)  # About 400 seconds
    iter_count = 0
    
    for i in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor
        new_h = anneal_step(current_h, temp)
        new_C = compute_C(new_h)
        
        # Accept or reject
        delta = new_C - current_C
        if delta < 0 or random.random() < np.exp(-delta / temp):
            current_h = new_h
            current_C = new_C
            
        # Update best
        if current_C < best_C:
            best_h = current_h.copy()
            best_C = current_C
            
        # Cooling schedule
        temp *= alpha
        if i % 1000 == 0:
            temp *= cooldown
            
        iter_count += 1
    
    # Gradient Descent refinement
    def compute_gradient(h_vals, epsilon=1e-4):
        grad = np.zeros_like(h_vals)
        base_C = compute_C(h_vals)
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += epsilon
            h_plus = project_to_feasible(h_plus)
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - base_C) / epsilon
        return grad
    
    # Gradient descent settings
    lr = 0.01
    max_steps = 200
    no_improvement = 0
    for step in range(max_steps):
        if time() - start_time > budget_s - 1:
            break
            
        grad = compute_gradient(best_h)
        best_h = best_h - lr * grad
        best_h = project_to_feasible(best_h)
        new_C = compute_C(best_h)
        
        if new_C < best_C:
            best_C = new_C
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 10:
                lr *= 0.5
                no_improvement = 0
            
    return (best_h.tolist(), best_C, n_points)
