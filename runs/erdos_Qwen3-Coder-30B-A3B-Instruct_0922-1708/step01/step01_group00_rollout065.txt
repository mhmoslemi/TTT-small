import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    mutation_rate = 0.1
    elite_size = pop_size // 5
    max_no_improve = 300
    eval_interval = 100
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_init = np.clip(h_init, 0, 1)
        population.append(h_init)
    
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    no_improve_count = 0
    
    def evaluate_h(h):
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Track evaluation count
    eval_count = 0
    
    while time.time() - start_time < budget_s - 1:
        eval_count += 1
        
        # Evaluate fitness of population
        fitness = []
        for h in population:
            c5 = evaluate_h(h)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Early termination
        if no_improve_count > max_no_improve:
            break
            
        # Selection: keep top elite_size
        elites = population[:elite_size]
        
        # Crossover and mutation to create new population
        new_population = elites.copy()
        
        while len(new_population) < pop_size:
            # Tournament selection
            idx1 = np.random.randint(0, elite_size)
            idx2 = np.random.randint(0, elite_size)
            
            # Arithmetic crossover
            parent1 = elites[idx1]
            parent2 = elites[idx2]
            alpha = np.random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child = child + noise
                
            # Repair
            child = np.clip(child, 0, 1) 
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
            
        population = new_population
        
        # Periodic logging
        if eval_count % eval_interval == 0:
            pass # Simplified logging
    
    return (best_h.tolist(), best_c5, n_points)
