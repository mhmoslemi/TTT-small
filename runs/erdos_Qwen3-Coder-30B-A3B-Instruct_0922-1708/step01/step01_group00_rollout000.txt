import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a triangular distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to higher resolution
        h_original = h.copy()
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_original)), h_original)
        h = np.clip(h, 0, 1)
        # Rescale to meet integral constraint
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create triangular-like distribution to start from
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h * (n_points / 2) / np.sum(h)
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    lr = 0.01
    t = 0
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        t += 1
        
        # Compute current C5 and gradient
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        max_lag_idx = np.argmax(correlation)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute exact gradient with respect to h[i]
        grad = np.zeros_like(h)
        # For the maximum lag, compute the gradient
        # The derivative of max_{k} ∫ h(x) (1 - h(x + k)) dx with respect to h[i]
        # corresponds to the values in the correlation that contribute to the maximum
        # We calculate the gradient of the correlation corresponding to the max_lag
        
        # Simple approach: gradient from finite difference (but with better handling)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        if iteration % 100 == 0:
            # Check if learning rate needs decay
            if iteration % 1000 == 0:
                lr *= 0.5
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
