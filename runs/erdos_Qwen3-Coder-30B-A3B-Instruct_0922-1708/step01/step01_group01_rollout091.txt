import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with a simple construction
        h = np.full(n_points, 0.5)
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    
    # Phase 1: Exhaustive search on piecewise constant functions
    # Use 5 segments with values in {0, 0.5, 1}
    m = 5  # number of segments
    values = [0, 0.5, 1]
    total_candidates = 3 ** m
    
    segment_combinations = list(product(values, repeat=m))
    
    # Early termination if we don't have much time
    if 0.7 * budget_s < 10:  # roughly 70% for this phase
        segment_combinations = segment_combinations[:100]  # reduce search space

    top_candidates = []
    
    for combo in segment_combinations:
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Create piecewise constant function
        h_candidate = np.zeros(n_points)
        segment_length = n_points // m
        remainder = n_points % m
        
        idx = 0
        for i, value in enumerate(combo):
            # Distribute remainder to first few segments
            actual_length = segment_length + (1 if i < remainder else 0)
            h_candidate[idx:idx + actual_length] = value
            idx += actual_length

        # Adjust to meet sum constraint
        h_candidate = project_h(h_candidate, n_points)
        
        c5 = evaluate_h(h_candidate, dx)
        top_candidates.append((c5, h_candidate))
    
    # Sort by cost and take top 10
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:10]
    
    # Phase 2: Gradient descent refinement
    for _, initial_h in top_candidates:
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        h = initial_h.copy()
        best_refined = h.copy()
        best_refined_c5 = evaluate_h(best_refined, dx)
        learning_rate = 0.01
        patience = 0
        max_patience = 1000
        
        # Gradient descent
        while time.time() - start_time < 0.95 * budget_s:
            # Compute overlap for all shifts and find worst one
            corr = correlate(h, 1 - h, mode="full")
            max_shift_idx = np.argmax(corr)
            worst_shift = max_shift_idx - (n_points - 1)
            
            # Compute gradient w.r.t. h using finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h -= learning_rate * grad
            h = project_h(h, n_points)
            
            # Evaluate
            new_c5 = evaluate_h(h, dx)
            if new_c5 < best_refined_c5:
                best_refined = h.copy()
                best_refined_c5 = new_c5
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    break
            if patience > 0 and patience % 1000 == 0:
                learning_rate *= 0.5
                patience = 0
        
        if best_refined_c5 < best_c5:
            best_c5 = best_refined_c5
            best_h = best_refined.copy()
    
    return (best_h, best_c5, n_points)
