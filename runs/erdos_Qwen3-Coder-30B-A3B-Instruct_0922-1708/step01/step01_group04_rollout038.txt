import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points

    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def perturb(h, delta=0.05):
        h_new = h.copy()
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    def smooth(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth

    def segment_optimize(h, n_segments=7):
        segment_size = n_points // n_segments
        if segment_size == 0:
            segment_size = 1
        segment_values = np.array([np.mean(h[i*segment_size:(i+1)*segment_size]) 
                                  for i in range(n_segments)])
        # Ensure sum constraint
        segment_sum = np.sum(segment_values)
        segment_values = segment_values * (n_segments / 2) / segment_sum
        
        def obj_func(segments):
            segments = np.clip(segments, 0, 1)
            # Reconstruct h from segments
            h_reconstructed = np.repeat(segments, segment_size)
            # Pad or truncate to n_points
            if len(h_reconstructed) > n_points:
                h_reconstructed = h_reconstructed[:n_points]
            elif len(h_reconstructed) < n_points:
                h_reconstructed = np.pad(h_reconstructed, (0, n_points - len(h_reconstructed)))
                # Make sure we have the right sum
                h_reconstructed = h_reconstructed * (n_points / 2) / np.sum(h_reconstructed)
            # Compute C5
            return evaluate(h_reconstructed)
        
        # Optimize using scipy
        result = minimize(obj_func, segment_values, method='L-BFGS-B', bounds=[(0,1)]*n_segments)
        segments_opt = np.clip(result.x, 0, 1)
        # Reconstruct
        h_opt = np.repeat(segments_opt, segment_size)
        if len(h_opt) > n_points:
            h_opt = h_opt[:n_points]
        elif len(h_opt) < n_points:
            h_opt = np.pad(h_opt, (0, n_points - len(h_opt)))
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        return h_opt

    best_h = h_start.copy()
    best_c5 = evaluate(best_h)

    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    n_segments = 7
    max_segments = 21
    
    while time.time() - start_time < budget_s and no_improvement_count < 50:
        iterations += 1
        # Try segment-based optimization first
        h_opt = segment_optimize(best_h, n_segments)
        c5_opt = evaluate(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Occasionally perturb or increase resolution
        if iterations % 20 == 0:
            # Perturb
            h_trial = perturb(best_h)
            h_trial = smooth(h_trial, window=3)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0

        # Increase resolution
        if (no_improvement_count > 20) and (n_segments < max_segments):
            n_segments += 2
            no_improvement_count = 0
            
    return (best_h, best_c5, n_points)
