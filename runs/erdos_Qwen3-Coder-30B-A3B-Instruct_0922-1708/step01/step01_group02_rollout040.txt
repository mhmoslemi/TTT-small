import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Simulated Annealing with better neighborhood search
    def neighbor(h_vec, T):
        h_new = h_vec.copy()
        # Randomly choose index and perturb
        idx = np.random.randint(0, len(h_vec))
        delta = np.random.normal(0, 0.01 * T)
        h_new[idx] += delta
        return project_h(h_new)
    
    # Annealing schedule
    T = 1.0
    Tmin = 1e-6
    alpha = 0.95
    max_iter = 50000
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        h_new = neighbor(h, T)
        c5_new = compute_c5(h_new)
        
        # Metropolis criterion
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
        # Cooling
        if T > Tmin:
            T *= alpha
            
        # Occasionally restart from best solution
        if iteration % 1000 == 0:
            h = best_h.copy()
    
    # Final refinement using scipy.optimize
    def objective(h_vec):
        return compute_c5(h_vec)
    
    # Constraints
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple random initializations for refinement
    for _ in range(10):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        h_trial = best_h + np.random.normal(0, 0.01, n_points)
        h_trial = project_h(h_trial)
        
        try:
            result = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=cons, 
                            options={'maxiter': 100, 'ftol': 1e-8})
            if result.success:
                c5_trial = compute_c5(result.x)
                if c5_trial < best_c5:
                    best_h = result.x.copy()
                    best_c5 = c5_trial
        except:
            pass
            
    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
