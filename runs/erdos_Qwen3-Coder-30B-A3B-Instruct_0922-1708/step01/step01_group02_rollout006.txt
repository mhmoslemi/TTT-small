import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    max_iterations = int(1e6)
    min_T = 1e-8
    
    # Main Simulated Annealing loop
    for i in range(max_iterations):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Save current state
        h_current = h.copy()
        c5_current = compute_c5(h_current)
        
        # Generate candidate by perturbing h
        h_candidate = h_current + np.random.normal(0, T * 0.1, n_points)
        h_candidate = project_h(h_candidate)
        c5_candidate = compute_c5(h_candidate)
        
        # Accept or reject based on Metropolis criterion
        delta = c5_candidate - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_candidate
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
        
        # Cool down temperature
        T = max(T * alpha, min_T)
    
    # Final projection and evaluation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
