import numpy as np
from scipy.signal import correlate
import time
from scipy.stats import qmc
import math

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sampling(n_points, n_samples):
    """Generate Latin Hypercube Samples"""
    sampler = qmc.LatinHypercube(d=1, seed=42)
    samples = sampler.random(n=n_samples)
    return samples.flatten() * 2

def initialization_phase(n_points, n_samples=50):
    """Initialize population using LHS and starting from current solution"""
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = project_h(initial_h, n_points)
    else:

        x = np.linspace(0, 2, n_points)
        initial_h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        initial_h = project_h(initial_h, n_points)
    
    population = [initial_h.copy()]
    
    # Generate LHS samples
    lhs_samples = lhs_sampling(n_points, n_samples - 1)
    for sample_val in lhs_samples:
        h = np.random.rand(n_points) * 0.5
        h = project_h(h, n_points)
        population.append(h)
    
    return population

def evolution_strategy_phase(population, n_points, dx, budget_s, start_time):
    """Evolution Strategy: mutate best solutions"""
    n_elites = 20
    n_offspring = 5
    best_population = population
    
    while time.time() - start_time < 0.7 * budget_s:
        # Evaluate all individuals
        fitness = [evaluate_h(h, dx) for h in best_population]
        
        # Keep top performers
        elite_indices = np.argsort(fitness)[:n_elites]
        elites = [best_population[i] for i in elite_indices]
        
        # Generate offspring
        offspring = []
        for elite in elites:
            for _ in range(n_offspring):
                # Small random perturbation
                mutation = np.random.normal(0, 0.05, n_points)
                child = elite + mutation
                child = project_h(child, n_points)
                offspring.append(child)
        
        # Combine elites and offspring
        combined = elites + offspring
        fitness = [evaluate_h(h, dx) for h in combined]
        
        # Keep best
        best_indices = np.argsort(fitness)[:n_elites]
        best_population = [combined[i] for i in best_indices]
    
    return best_population

def gradient_descent_phase(best_h, n_points, dx, budget_s, start_time):
    """Refine the best solution with gradient descent"""
    h = best_h.copy()
    best_c5 = evaluate_h(h, dx)
    learning_rate = 0.01
    
    while time.time() - start_time < 0.3 * budget_s:
        # Compute correlation
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h = h - learning_rate * np.sign(grad)
        h = project_h(h, n_points)
        
        # Evaluate

            best_c5 = new_c5
            best_h = h.copy()
        
        # Adapt learning rate
        if learning_rate > 1e-6:
            learning_rate *= 0.9995
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population using LHS
    population = initialization_phase(n_points)
    
    # Start time tracking
    start_time = time.time()
    
    # Evolution Strategy
    best_population = evolution_strategy_phase(population, n_points, dx, budget_s, start_time)
    
    # Extract the best solution
    fitness = [evaluate_h(h, dx) for h in best_population]
    best_idx = np.argmin(fitness)
    best_h = best_population[best_idx]
    best_c5 = fitness[best_idx]
    
    # Refine with gradient descent
    refined_h, refined_c5 = gradient_descent_phase(best_h, n_points, dx, budget_s, start_time)
    
    return (refined_h, refined_c5, n_points)
