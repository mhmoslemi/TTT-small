    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # PSO parameters
    n_particles = 100
    n_dims = n_points
    max_iter = 5000
    
    # Initialize particles
    particles = []
    velocities = []
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
        else:
            h_init = np.random.rand(n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (target_sum / np.sum(h_init))
        particles.append(h_init.copy())
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
    
    # Best particle tracking
    best_particles = [p.copy() for p in particles]
    best_C_values = [compute_C(p) for p in particles]
    global_best_idx = np.argmin(best_C_values)
    global_best_C = best_C_values[global_best_idx]
    global_best_h = best_particles[global_best_idx].copy()
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_dims), np.random.rand(n_dims)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (best_particles[i] - particles[i]) + 
                            c2 * r2 * (global_best_h - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Renormalize to sum constraint
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            
            # Evaluate particle
            C = compute_C(particles[i])
            
            # Update personal best
            if C < best_C_values[i]:
                best_C_values[i] = C
                best_particles[i] = particles[i].copy()
                
                # Update global best
                if C < global_best_C:
                    global_best_C = C
                    global_best_h = particles[i].copy()
    
    # Gradient descent refinement of top particles
    n_refine = min(10, n_particles)
    top_indices = np.argsort(best_C_values)[:n_refine]

        h = h_start.copy()
        
        # Scale h to satisfy constraint
        h = h * (target_sum / np.sum(h))
        
        learning_rate = 0.01
        no_improvement_count = 0
        
        for step in range(max_steps):
            if time() - start_time > budget_s - 1:
                break
                
            # Compute current C
            C = compute_C(h)
            
            # Approximate gradient with finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for j in range(len(h)):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = h_plus * (target_sum / np.sum(h_plus))
                C_plus = compute_C(h_plus)
                grad[j] = (C_plus - C) / eps
            
            # Update step with projection onto feasible set
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            
            # Adaptive learning rate
            new_C = compute_C(h)
            if new_C >= C:
                no_improvement_count += 1
                if no_improvement_count > 50:
                    learning_rate *= 0.5
                    no_improvement_count = 0
            else:
                no_improvement_count = 0

        return h, compute_C(h)
    
    # Refine the global best with gradient
    refined_h, refined_C = gradient_descent_refinement(global_best_h)
    
    if refined_C < global_best_C:
        return (refined_h.tolist(), refined_C, n_points)
    else:
        return (global_best_h.tolist(), global_best_C, n_points)
