import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Create a more structured approach with breakpoints
    K = 10  # Number of intervals
    breakpoints = np.linspace(0, 2, K+1)
    
    # Convert h to interval values
    interval_values = np.zeros(K)
    
    def get_h_from_intervals(interval_values, breakpoints):
        # Create piecewise constant h from intervals
        h = np.zeros(n_points)
        for i in range(K):
            start_idx = int(breakpoints[i] / dx)
            end_idx = int(breakpoints[i+1] / dx)
            h[start_idx:end_idx] = interval_values[i]
        return h
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    # Initial assignment of interval values
    for i in range(K):
        start_idx = int(breakpoints[i] / dx)
        end_idx = int(breakpoints[i+1] / dx)
        interval_values[i] = np.mean(h[start_idx:end_idx])
    
    # Gradient-based optimization with periodic parameter updates
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current C5
        h_current = get_h_from_intervals(interval_values, breakpoints)
        c5 = compute_c5(h_current)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_current.copy()
            last_improvement = iteration
            
        # Small perturbation for gradient estimation
        eps = 1e-5
        grad = np.zeros_like(interval_values)
        
        for i in range(K):
            interval_values_perturbed = interval_values.copy()
            interval_values_perturbed[i] += eps
            
            h_perturbed = get_h_from_intervals(interval_values_perturbed, breakpoints)
            c5_perturbed = compute_c5(h_perturbed)
            
            grad[i] = (c5_perturbed - c5) / eps
        
        # Update interval values
        lr = 0.02
        interval_values -= lr * grad
        
        # Clip interval values to [0,1]
        interval_values = np.clip(interval_values, 0, 1)
        
        # Rescale to ensure sum = n_points/2
        h_rescaled = get_h_from_intervals(interval_values, breakpoints)
        h_rescaled = h_rescaled * (n_points / 2) / np.sum(h_rescaled)
        
        # Update breakpoints every 50 iterations for structure refinement
        if iteration % 50 == 0:
            # Dynamically adjust breakpoints based on derivative
            h_detailed = get_h_from_intervals(interval_values, breakpoints)
            for i in range(1, K):
                # Adjust breakpoints according to average value
                start_idx = int(breakpoints[i-1] / dx)
                end_idx = int(breakpoints[i] / dx)
                mid_idx = (start_idx + end_idx) // 2
                breakpoints[i] = min(2.0, max(0.0, mid_idx * dx))
        
        # Keep increasing break point order with small adjustments
        for i in range(1, K):
            breakpoints[i] = max(breakpoints[i-1] + 0.001, breakpoints[i])
            
        # Early termination if no improvement for 300s
        if iteration - last_improvement > 300:
            break
            
    return (best_h.tolist(), best_c5, n_points)
