import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sample(n_points, n_samples):
    """Generate Latin Hypercube Samples"""
    samples = []
    for _ in range(n_samples):
        sample = np.random.rand(n_points)
        # Shuffle each dimension
        for i in range(n_points):
            sample[i] = (sample[i] + np.random.rand()) / n_points
        samples.append(sample)
    return samples

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    start_time = time.time()

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Phase 1: Latin Hypercube Sampling
    best_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    
    if time.time() - start_time < 0.3 * budget_s:
        # Generate LHS samples
        lhs_samples = lhs_sample(n_points, 50)
        candidates = []
        for sample in lhs_samples:
            h_candidate = project_h(sample, n_points)
            c5 = evaluate_h(h_candidate, dx)
            candidates.append((h_candidate, c5))
        
        # Sort and keep top candidates
        candidates.sort(key=lambda x: x[1])
        top_candidates = [h for h, _ in candidates[:20]]
        
        # Evaluate top candidates
        for h_cand in top_candidates:
            c5 = evaluate_h(h_cand, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cand.copy()
    
    # Phase 2: Evolution Strategy
    if time.time() - start_time < 0.6 * budget_s:
        population = [best_h.copy()]
        for _ in range(19):  # Create 20 total candidates
            h = np.random.rand(n_points)
            h = project_h(h, n_points)
            population.append(h)
        
        while time.time() - start_time < 0.85 * budget_s:
            # Evaluate population
            fitness = [evaluate_h(h, dx) for h in population]
            
            # Track best
            min_fitness_idx = np.argmin(fitness)
            if fitness[min_fitness_idx] < best_c5:
                best_h = population[min_fitness_idx].copy()
                best_c5 = fitness[min_fitness_idx]
            
            # Create offspring (elitism + mutation)
            new_population = [population[min_fitness_idx].copy()]
            for i in range(19):
                parent = population[np.random.randint(0, 20)]
                child = parent.copy()
                # Mutation - small random change
                mut_idx = np.random.randint(0, n_points)
                child[mut_idx] = np.random.rand()
                child = project_h(child, n_points)
                new_population.append(child)
            
            population = new_population
    
    # Phase 3: Gradient Descent with adaptive learning rate
    if time.time() - start_time < budget_s:
        h = best_h.copy()
        best_h = h.copy()
        best_c5 = evaluate_h(h, dx)
        learning_rate = 0.01
        patience = 0
        max_patience = 1000
        
        while time.time() - start_time < budget_s and patience < max_patience:
            # Compute overlap for all shifts and find worst one
            corr = correlate(h, 1 - h, mode="full")
            max_shift_idx = np.argmax(corr)
            worst_shift = max_shift_idx - (n_points - 1)
            
            # Compute gradient w.r.t. h
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h -= learning_rate * np.sign(grad)
            h = project_h(h, n_points)
            
            # Evaluate
            new_c5 = evaluate_h(h, dx)
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                patience = 0
            else:
                patience += 1
                if patience % 1000 == 0:
                    learning_rate *= 0.5
            if patience > 0 and patience % 100 == 0:
                learning_rate *= 0.95
            
    return (best_h, best_c5, n_points)
