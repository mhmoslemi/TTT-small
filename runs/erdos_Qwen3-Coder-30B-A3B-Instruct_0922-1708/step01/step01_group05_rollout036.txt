import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create from scratch
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a reasonable starting point: segments of alternating 0's and 1's 
        h = np.zeros(n_points)
        segment_length = n_points // 6
        for i in range(0, n_points, segment_length*2):
            end = min(i + segment_length, n_points)
            h[i:end] = 1.0
        # Scale to meet constraint
        h = h * (target_sum / np.sum(h))
        
    # Synchronize h to be feasible
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))

    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def project_h(h_vals):
        # Ensure h_vals sums to target_sum and is in [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return h_vals
    
    def neighbor(h_vals, temperature):
        # Generate neighbor by slightly modifying random segments
        h_new = h_vals.copy()
        # Pick random segment to modify
        segment_length = max(1, n_points // 20)
        start = np.random.randint(0, n_points - segment_length)
        end = min(start + segment_length, n_points)
        
        # Modify that segment
        delta = np.random.normal(0, 0.05 * temperature)
        h_new[start:end] += delta
        h_new = project_h(h_new)
        return h_new
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.995
    min_T = 1e-6
    max_iter = 50000
    current_C = compute_C(h)
    
    best_h = h.copy()
    best_C = current_C
    
    # Annealing loop
    for i in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
        h_new = neighbor(h, T)
        new_C = compute_C(h_new)
        delta = new_C - current_C
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_C = new_C
            if current_C < best_C:
                best_C = current_C
                best_h = h.copy()
        T = max(min_T, T * alpha)
    
    # Refine with gradient descent
    lr = 0.01
    no_improvement_count = 0
    max_no_improvement = 300
    
    def compute_gradient(h_vals, epsilon=1e-5):
        grad = np.zeros_like(h_vals)
        base_C = compute_C(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += epsilon
            h_plus = project_h(h_plus)
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - base_C) / epsilon
        return grad
    
    # Gradient descent refinement step
    for i in range(200):
        if time() - start_time > budget_s - 1:
            break
        grad = compute_gradient(best_h)
        best_h -= lr * grad
        best_h = project_h(best_h)
        new_C = compute_C(best_h)
        if new_C < best_C:
            best_C = new_C
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                lr *= 0.5
                no_improvement_count = 0
                if lr < 1e-6:
                    break
    
    # Final valid check
    final_h = project_h(best_h)
    final_C = compute_C(final_h)
    
    return (final_h.tolist(), final_C, n_points)
