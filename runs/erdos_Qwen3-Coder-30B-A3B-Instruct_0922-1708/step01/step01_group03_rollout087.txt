import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangular pulse
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def line_search(h, i, conv_full, shift_len, n_points):
    """Perform line search to optimize h[i] while maintaining sum constraint."""
    # Compute current offset contribution
    current_sum = np.sum(h)
    delta = (n_points / 2) - current_sum
    if abs(delta) < 1e-10:
        return h[i]
    
    # Solve for optimal h[i] that minimizes max overlap
    # This is really a convex optimization problem in h[i] over [0,1] with constraint
    # For small problem, we can do line search on 1D
    def obj(hi):
        new_h = h.copy()
        new_h[i] = hi
        # Use a simple approximation for derivative by finite difference
        # Only for local optimization around current h
        try:
            # Compute new conv
            test_h = project_h(new_h, n_points)
            # Consider just the current contribution to max overlap
            conv = np.correlate(test_h, 1 - test_h, mode="full")
            # Find max overlap
            max_overlap_val = np.max(conv)
            return max_overlap_val
        except:
            return 1e10  # High value if invalid
    
    # Bounded optimization
    bounds = [0, 1]
    try:
        result = optimize.minimize_scalar(obj, bounds=bounds, method='bounded')
        return max(0, min(1, result.x))
    except:
        return h[i]  # fallback

def coordinate_wise_optimization(h, n_points, time_budget_s):
    """Perform coordinate-wise optimization."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    n_iterations = 0
    max_iter = 5000
    
    # Precomputed conv for faster access

            # Compute new value
            new_h = best_h.copy()
            
            # Simple search approach:
            # Determine which shift has max overlap
            conv = np.correlate(best_h, 1 - best_h, mode="full")
            best_shift_index = np.argmax(conv)
            max_shift = best_shift_index - (n_points - 1)  # shift in [-n_points+1, n_points-1]
            
            # Compute contribution of each h[i] to the maximum overlap shift
            # As this is complex, do automatic optimization:
            # Use simple line search or grid search
            test_vals = np.linspace(0, 1, 100)
            best_val = best_h[i]
            best_obj = evaluate_c5(best_h)  # initial
            
            new_h = best_h.copy()
            for val in test_vals:
                new_h[i] = val
                new_h = project_h(new_h, n_points)

def nonlinear_cg_optimization(h, n_points, time_budget_s):
    """Perform nonlinear conjugate gradient optimization."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Precompute full convolution for efficiency
    conv_full = np.correlate(best_h, 1 - best_h, mode="full")
    shift_len = len(conv_full)
    
    # Polak-Ribière CG with momentum
    x = best_h.copy()
    g = np.zeros(n_points)  # gradient (approximated)
    d = -g.copy()  # search direction
    beta = 0
    
    max_iter = 2000
    iteration = 0
    momentum = 0.9
    
    while time.time() - start_time < time_budget_s and iteration < max_iter:
        iteration += 1
        
        # Compute current gradient approximatively
        eps = 1e-5
        g_old = g.copy()
        g = np.zeros(n_points)
        for i in range(n_points):
            h_plus = x.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            h_minus = x.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            g[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Conjugate direction update (Polak-Ribière)
        if iteration == 1:
            d = -g
        else:
            beta = max(0, np.dot(g - g_old, g) / np.dot(g_old, g_old))
            d = -g + beta * d
        
        # Line search for step size
        def phi(alpha):
            new_x = x + alpha * d
            new_x = project_h(new_x, n_points)
            return evaluate_c5(new_x)
        
        # Scalar optimization
        alpha_opt = 0.1
        try:
            res = optimize.minimize_scalar(phi, bounds=(0, 1), method='bounded')
            alpha_opt = res.x
        except:
            pass
        
        # Update
        x = x + alpha_opt * d
        x = project_h(x, n_points)
        
        new_c5 = evaluate_c5(x)
        if new_c5 < best_c5:
            best_h = x.copy()
            best_c5 = new_c5
        else:
            # Loose the momentum if not improving
            momentum = min(0.99, momentum)
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45
    
    # Initialize
    h = initialize_h(n_points)
    
    start_time = time.time()
    
    # Phase 1: Coordinate-wise optimization
    h_phase1, c5_phase1 = coordinate_wise_optimization(h, n_points, budget_s * 0.8)
    
    # Phase 2: Nonlinear CG
    h_phase2, c5_phase2 = nonlinear_cg_optimization(h_phase1, n_points, budget_s * 0.2)
    
    # Final check
    final_h = h_phase2 if c5_phase2 < c5_phase1 else h_phase1
    final_c5 = min(c5_phase2, c5_phase1)
    
    # Ensure project
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
