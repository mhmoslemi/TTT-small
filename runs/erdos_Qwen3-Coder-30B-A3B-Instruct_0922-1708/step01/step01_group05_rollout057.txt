import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points) * 0.5
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    n_dims = n_points
    max_iter = 5000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Initialize particles and personal bests
    particles = np.random.rand(n_particles, n_dims) * 0.5
    particles = np.clip(particles, 0, 1)
    velocities = np.random.rand(n_particles, n_dims) - 0.5
    
    # Normalize each particle to satisfy integral constraint
    for i in range(n_particles):
        particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
    
    pbest = particles.copy()
    pbest_C = np.full(n_particles, np.inf)
    
    # Global best
    gbest_idx = 0
    gbest_C = np.inf
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            # Update velocity and position with bounds
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (particles[gbest_idx] - particles[i]))
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Renormalize to satisfy integral constraint
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            
            # Evaluate current particle
            h_vals = particles[i]
            corr = np.correlate(h_vals, 1-h_vals, mode="full")
            C = np.max(np.abs(corr)) * dx
            
            if C < pbest_C[i]:
                pbest_C[i] = C
                pbest[i] = h_vals.copy()
                
            if C < gbest_C:
                gbest_C = C
                gbest_idx = i
                
    # Refinement with gradient descent
    best_h = pbest[gbest_idx].copy()
    
    def compute_C(h_vals):
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        return np.max(np.abs(corr)) * dx
    
    def grad_C(h_vals):
        # Approximate gradient
        eps = 1e-6
        grad = np.zeros_like(h_vals)
        base_C = compute_C(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - base_C) / eps
        return grad
    
    lr = 0.01
    ls_count = 0
    for i in range(200):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_C = compute_C(best_h)
        grad = grad_C(best_h)
        best_h -= lr * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        
        new_C = compute_C(best_h)
        if new_C >= current_C:
            ls_count += 1
            if ls_count > 10:
                lr *= 0.5
                ls_count = 0
        else:
            ls_count = 0
            
    # Final evaluation
    final_C = compute_C(best_h)
    
    return (best_h.tolist(), final_C, n_points)
