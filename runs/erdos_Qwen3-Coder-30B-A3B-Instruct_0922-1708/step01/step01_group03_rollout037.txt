import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother symmetric function
    x = np.linspace(0, 2, n_points)
    # Create a symmetric shape with triangular pulses
    h = 0.5 * (1 + np.sin(np.pi * x)) * (1 + np.sin(np.pi * (x - 1))) 
    # Ensure it's bounded and scaled appropriately
    h = np.clip(h, 0, 1)
    h = project_h(h, n_points)
    return h

def line_search(h, i, shift_idx, old_c5, n_points, dx):
    """Perform line search to find optimal h[i] value."""
    # Use optimization to find best value
    def objective(new_h_i):
        # Copy h and update i-th element
        h_temp = h.copy()
        h_temp[i] = new_h_i
        # Ensure it still sums to n_points / 2
        # Adjust all other elements proportionally, but that will be complex
        # So we use the fixed sum constraint in the final result by projecting
        h_temp = project_h(h_temp, n_points)
        # Compute new C5
        new_c5 = evaluate_c5(h_temp)
        return new_c5
    
    # bracketing
    old_val = h[i]
    # Try a small shift
    low = max(0, old_val - 0.1)
    high = min(1, old_val + 0.1) if old_val < 1 else old_val
    if low == high:
        return old_val
    try:
        res = optimize.minimize_scalar(objective, bounds=(low, high), method='bounded')
        if res.success:
            return res.x
    except:
        pass
    return old_val

def coordinate_wise_optimization(h, n_points, max_iter=5000, time_budget_s=500):
    """Optimize h coordinate-wise with line search."""
    start_time = time.time()
    dx = 2.0 / n_points
    old_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = old_c5
    improvement_count = 0
    no_improvement = 0

    for iteration in range(max_iter):
        if time.time() - start_time > time_budget_s:
            break
        improvement = False
        for i in range(n_points):
            if time.time() - start_time > time_budget_s:
                break
            old_h_i = h[i]
            # Try to find a better value for h[i]
            new_h_i = line_search(h, i, 0, old_c5, n_points, dx)
            if abs(new_h_i - old_h_i) > 1e-6:
                # Actually update
                h[i] = new_h_i
                h = project_h(h, n_points)
                new_c5 = evaluate_c5(h)
                if new_c5 < old_c5 - 1e-10:
                    improvement = True
                    old_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    improvement_count += 1
                    no_improvement = 0
                elif new_c5 > old_c5:
                    h[i] = old_h_i  # revert changes
                else:
                    # Small numerical errors
                    pass
            if time.time() - start_time > time_budget_s:
                break
        if not improvement:
            no_improvement += 1
            if no_improvement > 100:
                break
    return best_h, best_c5

def conjugate_gradient_optimization(h, n_points, max_iter=5000, time_budget_s=500):
    """Optimize using conjugate gradient method."""
    start_time = time.time()
    dx = 2.0 / n_points
    old_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = old_c5
    x = h.copy()
    old_grad = np.zeros_like(x)
    d = np.zeros_like(x)
    no_improvement = 0
    improvement_count = 0
    
    def compute_gradient(x, eps=1e-5):
        """Compute finite difference gradient."""
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            x_plus = project_h(x_plus, n_points)
            c5_plus = evaluate_c5(x_plus)
            x_minus = x.copy()
            x_minus[i] -= eps
            x_minus = project_h(x_minus, n_points)
            c5_minus = evaluate_c5(x_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    # Initialize first direction
    grad = compute_gradient(x)
    d = -grad
    old_grad = grad.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_budget_s:
            break
            
        # Line search along direction d
        def phi(alpha):
            x_new = x + alpha * d
            x_new = project_h(x_new, n_points)
            return evaluate_c5(x_new)
        
        # Find optimal alpha
        try:
            res = optimize.minimize_scalar(phi, bounds=(0, 10), method='bounded')
            if res.success:
                alpha = res.x
            else:
                alpha = 0.1
        except:
            alpha = 0.1
            
        # Update x
        x_new = x + alpha * d
        x_new = project_h(x_new, n_points)
        new_c5 = evaluate_c5(x_new)
        
        if new_c5 < old_c5:
            improvement = True
            old_c5 = new_c5
            x = x_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = x.copy()
                improvement_count += 1
            no_improvement = 0
        else:
            improvement = False
            no_improvement += 1
        
        # Conjugate gradient update
        new_grad = compute_gradient(x)
        beta = max(0, np.dot(new_grad - old_grad, new_grad) / np.dot(old_grad, old_grad) + 1e-10)
        d = -new_grad + beta * d
        old_grad = new_grad.copy()
        
        if no_improvement > 500:
            break

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45
    # Start with a reasonably good starting point
    h = initialize_h(n_points)
    start_time = time.time()
    
    # First, try coordinate-wise optimization
    h1, c5_1 = coordinate_wise_optimization(h, n_points, max_iter=2500, time_budget_s=budget_s * 0.7)
    
    # Then try conjugate gradient
    h2, c5_2 = conjugate_gradient_optimization(h1, n_points, max_iter=2500, time_budget_s=budget_s * 0.3)
    
    # Return best result
    final_h = h2 if c5_2 < c5_1 else h1
    final_c5 = min(c5_2, c5_1)
    
    # Ensure final result is valid
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
