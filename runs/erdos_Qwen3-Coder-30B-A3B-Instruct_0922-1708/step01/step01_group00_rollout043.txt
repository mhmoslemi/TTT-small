import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to desired resolution
        x_old = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_old, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Genetic algorithm parameters
    pop_size = 50
    mutation_rate = 0.1
    elite_size = 10
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_new = h + np.random.normal(0, 0.05, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        population.append(h_new)
    
    last_improvement_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate population
        fitnesses = []
        for h_ind in population:
            correlation = np.correlate(h_ind, 1 - h_ind, mode='full')
            c5 = np.max(correlation) * dx
            fitnesses.append(c5)
        
        # Update best
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
            last_improvement_time = time.time()
        
        # Early termination if no improvement
        if time.time() - last_improvement_time > 300:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            idx1, idx2 = np.random.randint(0, pop_size, 2)
            if fitnesses[idx1] < fitnesses[idx2]:
                selected.append(population[idx1])
            else:
                selected.append(population[idx2])
        
        # Crossover and mutation
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate rest through crossover and mutation
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            
            # Crossover - arithmetic mean
            child = (parent1 + parent2) / 2.0
            
            # Mutation
            mask = np.random.random(n_points) < mutation_rate
            child += np.random.normal(0, 0.05, n_points) * mask
            
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population
        
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
