import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=3):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def objective(h):
        return evaluate(h)
    
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Set up optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Initial point
    current_h = h_start.copy()
    best_h = current_h.copy()
    best_c5 = evaluate(current_h)
    
    start_time = time.time()
    
    # Optimize with L-BFGS-B
    iteration = 0
    patience = 0
    max_patience = 50
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Try optimizing with scipy
        try:
            res = minimize(
                objective, 
                current_h, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 10}
            )
            
            if res.success:
                new_h = res.x
                # Ensure constraints are met
                new_h = np.clip(new_h, 0, 1)
                # Rescale to integral = 1
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                current_c5 = evaluate(new_h)
                if current_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = current_c5
                    patience = 0
                else:
                    patience += 1
                current_h = new_h
            else:
                # Fall back to perturbation if optimization fails
                patience += 1
                current_h = np.clip(current_h + np.random.normal(0, 0.01, n_points), 0, 1)
                current_h = current_h * (n_points / 2) / np.sum(current_h)
                
        except Exception:
            # If optimization fails, use perturbation
            patience += 1
            current_h = np.clip(current_h + np.random.normal(0, 0.01, n_points), 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # Add periodic exploration
        if iteration % 30 == 0:
            # Add some noise and smooth
            noise = np.random.normal(0, 0.02, n_points)
            current_h = np.clip(current_h + noise, 0, 1)
            current_h = smooth(current_h, 5)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            
        # Early stopping if no improvement for too long
        if patience > max_patience:
            break
    
    return (best_h, best_c5, n_points)
