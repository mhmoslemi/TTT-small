import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def segment_based_optimization(h_start, m_segments=7, max_iter=200):
        # Initialize segments
        segment_width = n_points // m_segments
        if segment_width * m_segments < n_points:
            segment_width += 1
        
        segments = []
        for i in range(m_segments):
            start_idx = i * segment_width
            end_idx = min((i + 1) * segment_width, n_points)
            segment_avg = np.mean(h_start[start_idx:end_idx])
            segments.append(max(0, min(1, segment_avg)))
        
        # Ensure constraint: sum of segments = m_segments / 2
        total = sum(segments)
        if total > 0:
            scale = (m_segments / 2) / total
            segments = [s * scale for s in segments]
        
        def reconstruct_h(segments):
            h = np.zeros(n_points)
            segment_width = n_points // len(segments)
            if segment_width * len(segments) < n_points:
                segment_width += 1
            for i, s in enumerate(segments):
                start_idx = i * segment_width
                end_idx = min((i + 1) * segment_width, n_points)
                h[start_idx:end_idx] = s
            return h
        
        def objective(segments):
            h = reconstruct_h(segments)
            return evaluate(h)
        
        def constraint(segments):
            return sum(segments) - (len(segments) / 2)
        
        cons = {'type': 'eq', 'fun': constraint}
        bounds = [(0, 1) for _ in segments]
        
        # Optimize segments
        start_time = time.time()
        best_c5 = float('inf')
        best_segments = segments.copy()
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s:
                break
                
            # Optimize using SLSQP
            res = minimize(objective, segments, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 5})
            if res.success:
                segments = list(res.x)
                h = reconstruct_h(segments)
                c5 = evaluate(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_segments = segments.copy()
            
            # Perturb a portion of the segments
            for i in np.random.choice(len(segments), max(1, len(segments)//10), replace=False):
                segments[i] += np.random.uniform(-0.1, 0.1)
                segments[i] = max(0, min(1, segments[i]))
            
            # Normalize constraint
            total = sum(segments)
            if total > 0:
                scale = (len(segments) / 2) / total
                segments = [s * scale for s in segments]
            
            # Smooth the segments
            if len(segments) >= 3:
                smoothed = np.convolve(segments, [1/3, 1/3, 1/3], mode='same')
                segments = smoothed.tolist()
        
        best_h = reconstruct_h(best_segments)
        return best_h, best_c5
    
    # Try different segment counts and faster initial search
    best_c5 = float('inf')
    best_h = h_start.copy()
    current_points = n_points
    
    # Try with several segment sizes
    segment_counts = [5, 7, 9, 11]
    
    start_time = time.time()
    for m in segment_counts:
        if time.time() - start_time > budget_s * 0.9:
            break
            
        h_optimized, c5_optimized = segment_based_optimization(h_start, m, 100)
        if c5_optimized < best_c5:
            best_c5 = c5_optimized
            best_h = h_optimized.copy()
    
    # Final refinement
    if time.time() - start_time < budget_s * 0.1:
        # Optimize full resolution with a more focused approach
        def refinement_objective(h):
            return evaluate(h)
        
        # Early stopping optimization
        def optimize_with_refinement(h_start):
            h = h_start.copy()
            best_h = h.copy()
            best_c5 = evaluate(h)
            
            iterations = 0
            no_improvement = 0
            
            while no_improvement < 50 and time.time() - start_time < budget_s:
                iterations += 1
                # Small local perturbations
                h_trial = h.copy()
                indices = np.random.choice(n_points, max(1, n_points//10), replace=False)
                for i in indices:
                    h_trial[i] += np.random.uniform(-0.02, 0.02)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)

                c5_trial = evaluate(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    no_improvement = 0
                else:
                    no_improvement += 1
                
                h = best_h.copy()
                
            return best_h, best_c5
        
        final_h, final_c5 = optimize_with_refinement(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    return (best_h, best_c5, n_points)
