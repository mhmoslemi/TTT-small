import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 47
        # Create symmetric construction with plateau
        h = np.full(n_points, 0.5)
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(conv)
        return max_val * dx

    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Genetic algorithm parameters
    population_size = 100
    generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.7
    elitism_count = population_size // 10
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0 and 'initial_h_values' in globals():
            individual = h.copy()
        else:
            individual = np.random.random(n_points)
        individual = project_h(individual)
        population.append(individual)
    
    best_h = None
    best_c5 = float('inf')
    
    # GA Loop
    for gen in range(generations):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for individual in population:
            c5 = compute_c5(individual)
            fitness_scores.append(c5)
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness_scores[0]
        
        # Elitism - keep best individuals
        new_population = population[:elitism_count]
        
        # Roulette wheel selection
        def roulette_selection():
            # Linear scaling to prevent degeneracy
            min_fitness = min(fitness_scores)
            max_fitness = max(fitness_scores)
            if max_fitness == min_fitness:
                return population[np.random.randint(0, population_size)]
            fitness_scaled = [1.0 / (1e-10 + f - min_fitness) for f in fitness_scores]
            total_fitness = sum(fitness_scaled)
            if total_fitness == 0:
                return population[np.random.randint(0, population_size)]
            probabilities = [f / total_fitness for f in fitness_scaled]
            idx = np.random.choice(population_size, p=probabilities)
            return population[idx]
        
        # Generate offspring
        for _ in range(population_size - elitism_count):
            if np.random.random() < crossover_rate:
                parent1 = roulette_selection()
                parent2 = roulette_selection()
                # Single point crossover
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = population[np.random.randint(0, population_size)].copy()
                child2 = population[np.random.randint(0, population_size)].copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate / (1.0 + gen**0.5):
                    child1[i] += np.random.normal(0, 0.1)
                if np.random.random() < mutation_rate / (1.0 + gen**0.5):
                    child2[i] += np.random.normal(0, 0.1)
            
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.extend([child1, child2])
        
        # Keep only population_size individuals
        population = new_population[:population_size]
        
        # Adapt mutation rate
        mutation_rate = max(0.01, mutation_rate * 0.999)
    
    # Final check with best found solution
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
