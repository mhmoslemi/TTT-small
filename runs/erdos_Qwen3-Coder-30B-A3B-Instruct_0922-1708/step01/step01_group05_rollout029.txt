import numpy as np
from time import time
import scipy.optimize as so

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    max_iter = 1000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles and velocities
    particles = np.random.rand(n_particles, n_points)
    particles = np.clip(particles, 0, 1)
    velocities = np.random.rand(n_particles, n_points) * 0.1 - 0.05
    
    # Personal best
    pbest = particles.copy()
    pbest_vals = np.full(n_particles, float('inf'))
    
    # Global best
    gbest_idx = 0
    gbest = particles[0].copy()
    gbest_val = float('inf')
    
    # Optimization loop
    iter_count = 0
    while time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Update particles
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocity = (inertia * velocity + 
                       cognitive * r1 * (pbest[i] - particle) + 
                       social * r2 * (gbest - particle))
            
            # Update particle
            particle += velocity
            particle = np.clip(particle, 0, 1)
            
            # Rescale to meet constraint
            particle = particle * (target_sum / np.sum(particle))
            
            # Evaluate
            corr = np.correlate(particle, 1-particle, mode="full")
            corr *= dx
            C = np.max(np.abs(corr))
            
            # Update personal best
            if C < pbest_vals[i]:
                pbest[i] = particle.copy()
                pbest_vals[i] = C
                
                # Update global best
                if C < gbest_val:
                    gbest = particle.copy()
                    gbest_val = C
                    gbest_idx = i
            
            particles[i] = particle
            velocities[i] = velocity
            
        iter_count += 1
    
    # Refine with gradient descent
    h_refined = gbest.copy()
    best_C = gbest_val
    lr = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    
    def compute_C(h_vals):
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def gradient_step(h_vals, lr):
        # Compute gradient using finite differences
        grad = np.zeros_like(h_vals)
        eps = 1e-5
        base_C = compute_C(h_vals)
        
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = h_plus * (target_sum / np.sum(h_plus))  # rescale
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - base_C) / eps
        
        # Update using gradient ascent
        h_new = h_vals - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))  # rescale
        
        return h_new
    
    # Gradient descent iterations
    while time() - start_time < budget_s - 1:
        h_new = gradient_step(h_refined, lr)
        C_new = compute_C(h_new)
        
        if C_new < best_C:
            h_refined = h_new
            best_C = C_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                lr *= 0.5
                no_improvement_count = 0
        
        if lr < 1e-8:
            break
    
    return (h_refined.tolist(), best_C, n_points)
