import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use symmetric triangular pulse
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_exact_gradient(h, k, n_points):
    """Compute exact gradient for shift k."""
    grad = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(n_points):
        # Gradient contribution from shift k
        j = (i + k) % n_points
        grad[i] = (1 - h[j]) * dx
    return grad

def line_search(h, grad, max_k, n_points, eta):
    """Perform line search to find optimal step size."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Bracketing search with bounds [0, eta]
    try:
        result = minimize_scalar(objective, bounds=(0, eta), method='bounded')
        return result.x if result.success else 0.0
    except:
        return 0.0

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using gradient descent with exact gradients."""
    np.random.seed(seed)
    n_points = 45  # Use smaller number for faster runtime
    h = initialize_h(n_points)
    
    start_time = time.time()
    
    # Initialize parameters
    eta = 0.1
    eta_min = 1e-6
    eta_max = 0.5
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    no_improvement = 0
    max_no_improvement = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s and no_improvement < max_no_improvement:
        # Find shift that gives maximum overlap
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        max_k = max_idx - (n_points - 1)  # Convert to shift index
        
        # Compute exact gradient for this shift
        grad = compute_exact_gradient(h, max_k, n_points)
        
        # Perform line search to find optimal step size
        alpha = line_search(h, grad, max_k, n_points, eta)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improvement = 0
            # Increase learning rate
            eta = min(eta * 1.2, eta_max)
        else:
            no_improvement += 1
            # Decrease learning rate
            eta = max(eta / 2, eta_min)
        
        h = h_new
        iteration += 1
        
        # Safety break for very large iterations
        if iteration > 10000:
            break
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
