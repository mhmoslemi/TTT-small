import numpy as np
from scipy import optimize
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # If initial_h_values is defined globally, use it
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric with triangular pulse
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def compute_gradient_at_shift(h, shift, n_points, dx):
    """Compute gradient w.r.t. h for the specific shift that maximizes overlap."""
    # Clip shift to valid range (-n+1 to n-1)
    shift = int(shift)
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + shift) % n_points
        grad[i] = -dx * (1 - h[j])
    return grad

def line_search(h, grad, eta, n_points, dx):
    """Find optimal step size using line search."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Use scipy to find minimum
    try:
        res = optimize.minimize_scalar(objective, bounds=(0, eta), method='bounded')
        alpha = res.x
    except:
        alpha = 0.0
    return min(alpha, eta)

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45  # Try smaller size for faster optimization
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Parameters
    eta = 0.1  # Initial learning rate
    tolerance = 1e-6
    max_iter = 10000
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    dx = 2.0 / n_points
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95 and iteration < max_iter:
        # Find shift causing maximum overlap
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)  # Compute actual shift
        
        # Compute gradient for this shift
        grad = compute_gradient_at_shift(h, shift, n_points, dx)
        
        # Line search for optimal step size
        alpha = line_search(h, grad, eta, n_points, dx)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Check for improvement
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            eta = 0.1  # Reset learning rate
        else:
            eta *= 0.5  # Reduce learning rate
            if eta < 1e-6:
                break
                
        iteration += 1
        
        # Ensure feasibility
        h = project_h(h, n_points)
    
    # Final evaluation
    final_c5 = evaluate_c5(h)
    
    return (h, final_c5, n_points)
