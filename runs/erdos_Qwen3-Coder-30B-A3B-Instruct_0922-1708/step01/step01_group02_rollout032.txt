import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(conv) * dx

    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec

    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    T0 = 1.0
    alpha = 0.999
    T = T0
    max_iter = 10000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_current = h.copy()
        c5_current = compute_c5(h_current)
        
        # Perturb h with Gaussian noise scaled by temperature
        perturbation = np.random.normal(0, T * 0.1, n_points)
        h_candidate = h_current + perturbation
        h_candidate = project_h(h_candidate)
        
        c5_candidate = compute_c5(h_candidate)
        delta = c5_candidate - c5_current
        
        # Accept or reject based on Metropolis criterion
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate
                
        T = alpha * T

    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
