import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a symmetric function
    x = np.linspace(0, 2, n_points)
    h = np.sin(np.pi * x) * 0.5 + 0.5  # Range [0,1]
    h = project_h(h, n_points)
    return h

def get_shift_index(h, shift):
    """Get the index for circular shift."""
    n = len(h)
    return (np.arange(n) + shift) % n

def compute_gradient(h, shift):
    """Compute exact gradient for a specific shift."""
    n = len(h)
    dx = 2.0 / n
    # Gradient[i] = d/dh[i] (sum over all j of h[j] * (1 - h[(j+shift) % n]))
    # = 1 - h[(i + shift) % n]
    grad = 1 - h[get_shift_index(h, shift)]
    return grad * dx

def line_search(h, grad, shift, eta, n_points):
    """Perform line search to find optimal step size."""
    def objective(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Bracketing
    a, b = 0.0, eta
    # We want to find minimum, so we're looking for the alpha that minimizes objective
    try:
        res = minimize_scalar(objective, bounds=(0, eta), method='bounded')
        alpha_opt = res.x
        return alpha_opt
    except:
        # fallback
        return eta * 0.5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    h = initialize_h(n_points)
    
    # Current best
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Gradient descent settings
    eta = 0.1
    eta_min = 1e-6
    eta_max = 0.5
    max_no_improvement = 10000
    no_improvement = 0
    iteration = 0
    learning_rate = 0.01

    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        # Identify worst shift
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)

        # Compute gradient for this shift
        grad = compute_gradient(h, shift)
        
        # Line search for optimal step size
        alpha = line_search(h, grad, shift, eta, n_points)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        
        # Check improvement
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            no_improvement = 0
            # Increase learning rate if improvement happens
            if eta < eta_max:
                eta *= 1.2
        else:
            no_improvement += 1
            # Decrease learning rate if no improvement
            eta = max(eta_min, eta * 0.5)
            
        h = h_new
        iteration += 1
        
        # Ensure numerical stability and prevent overflow
        if iteration % 1000 == 0:
            if no_improvement > 500:
                # Random perturbation
                h = project_h(h + np.random.normal(0, 0.01, n_points), n_points)
        
    return (best_h, best_c5, n_points)
