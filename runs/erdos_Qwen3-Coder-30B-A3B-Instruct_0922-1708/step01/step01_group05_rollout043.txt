import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Optimization using scipy minimize with bounds
    best_h = h.copy()
    best_C = float('inf')
    
    # Objective function using correlate
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    # Constraints function
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    # Optimization with L-BFGS-B
    def objective(h_vals):
        return compute_C(h_vals)
    
    # Initial evaluation
    C = compute_C(h)
    if C < best_C:
        best_C = C
        best_h = h.copy()
    
    # Bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Main optimization loop
    no_improvement_count = 0
    max_no_improvement = 100
    iteration = 0
    
    while time() - start_time < budget_s - 1:
        iteration += 1
        
        # Do one optimization step using L-BFGS-B
        result = so.minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                             constraints=constraint, options={'maxiter': 1})
        
        # Update h if successful
        if result.success:
            h = result.x
            # Re-normalize to satisfy sum constraint
            h = h * (target_sum / np.sum(h))
            
            # Evaluate objective
            C = compute_C(h)
            if C < best_C:
                best_C = C
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Random perturbation if no improvement
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            no_improvement_count = 0
            
    # Final evaluation of best solution
    final_C = compute_C(best_h)
    
    return (best_h.tolist(), final_C, n_points)
