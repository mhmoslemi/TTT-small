import numpy as np
import time
from scipy.signal import correlate
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(conv)
        return max_val * dx
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        population.append(project_h(h.copy()))
    else:
        # Create random individuals
        for _ in range(population_size - 1):
            h_rand = np.random.rand(n_points)
            population.append(project_h(h_rand))
    
    if len(population) < population_size:
        # fill up to required size
        for _ in range(population_size - len(population)):
            h_rand = np.random.rand(n_points) 
            population.append(project_h(h_rand))
    
    # GA parameters
    crossover_rate = 0.7
    mutation_rate = 0.2
    generations = 0
    max_generations = 3000
    
    # Track best individual
    best_h = None
    best_c5 = float('inf')
    
    while generations < max_generations and (time.time() - start_time) < budget_s - 1:
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind)
            fitness.append(c5)
        
        # Track best
        for i, c5 in enumerate(fitness):
            if c5 < best_c5:
                best_c5 = c5
                best_h = population[i].copy()
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Selection: roulette wheel
        # Convert to fitness values (lower is better)
        inv_fitness = [1.0 / (f + 1e-10) for f in fitness]  # avoid division by zero
        total_fitness = sum(inv_fitness)
        if total_fitness == 0:
            # Fallback to uniform selection
            selection_probs = [1.0 / len(population)] * len(population)
        else:
            selection_probs = [f / total_fitness for f in inv_fitness]
            
        new_population = population[:10]  # Elitism: keep best 10
        
        # Generate offspring
        while len(new_population) < population_size:
            # Select parents
            parent1_idx = np.random.choice(len(population), p=selection_probs)
            parent2_idx = np.random.choice(len(population), p=selection_probs) 
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if random.random() < crossover_rate:
                # Single-point crossover
                crossover_point = random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
                
            # Mutate offspring
            mutation_strength = mutation_rate / (1.0 + generations**0.5)
            if random.random() < mutation_strength:
                for i in range(n_points):
                    if random.random() < 0.1:  # 10% chance to mutate
                        noise = np.random.normal(0, 0.1)
                        child1[i] += noise
                child1 = project_h(child1)
                
            if random.random() < mutation_strength:
                for i in range(n_points):
                    if random.random() < 0.1:  # 10% chance to mutate
                        noise = np.random.normal(0, 0.1)
                        child2[i] += noise
                child2 = project_h(child2)
                
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        population = new_population
        generations += 1
        
        # Early exit based on time
        if (time.time() - start_time) > budget_s - 1:
            break
    
    # Final validation
    if best_h is None:
        # fallback to initial solution
        final_h = project_h(h)
        final_c5 = compute_c5(final_h)
    else:
        final_h = project_h(best_h)
        final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
