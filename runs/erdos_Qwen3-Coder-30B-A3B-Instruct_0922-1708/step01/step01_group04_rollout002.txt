import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def smooth(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def perturb(h, delta=0.05):
        h_new = h.copy()
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def objective(h):
        return evaluate(h)
    
    def project_constraints(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    patience = 0
    max_patience = 100
    
    # Basic local search
    while time.time() - start_time < budget_s and patience < max_patience:
        iterations += 1
        h_trial = perturb(best_h, delta=0.05)
        h_trial = smooth(h_trial, window=3)
        c5_trial = evaluate(h_trial)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            patience = 0
        else:
            patience += 1
            
        # Every 10 iterations, try a more aggressive perturbation
        if iterations % 10 == 0:
            h_trial = perturb(best_h, delta=0.1)
            h_trial = smooth(h_trial, window=5)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                patience = 0
                
    return (best_h, best_c5, n_points)
