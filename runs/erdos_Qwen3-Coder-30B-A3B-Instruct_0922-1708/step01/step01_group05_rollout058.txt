import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    n_dims = n_points
    max_iter = 5000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    for _ in range(n_particles):
        particle = h.copy() + np.random.normal(0, 0.01, n_points)
        particle = np.clip(particle, 0, 1)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Global best
    gbest = h.copy()
    gbest_C = float('inf')
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def normalize_h(h_vals):
        return h_vals * (target_sum / np.sum(h_vals))
    
    # PSO loop
    iter_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s - 2:
            break
        iter_count += 1
        for i, particle in enumerate(particles):
            # Update velocity
            r1, r2 = np.random.rand(n_dims), np.random.rand(n_dims)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (gbest - particle) +
                c2 * r2 * (particles[np.random.randint(0, n_particles)] - particle)
            )
            # Update position
            particle = particle + velocities[i]
            # Keep in bounds
            particle = np.clip(particle, 0, 1)
            particles[i] = particle
            
            # Evaluate
            particle = normalize_h(particle)
            C = compute_C(particle)
            if C < gbest_C:
                gbest_C = C
                gbest = particle.copy()
    
    # Refinement with gradient descent on top PSO solutions
    top_particles = sorted(particles, key=compute_C)[:n_particles//10]
    best_refined_h = gbest.copy()
    best_refined_C = gbest_C
    
    for particle in top_particles:
        if time() - start_time > budget_s - 2:
            break
        # Initialize with the top particle
        h = particle.copy()
        
        # Gradient descent
        lr = 0.01
        for _ in range(200):
            if time() - start_time > budget_s - 2:
                break
            h = normalize_h(h)
            C = compute_C(h)
            # Estimate gradient via finite difference
            eps = 1e-4
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = normalize_h(h_plus)
                C_plus = compute_C(h_plus)
                grad[j] = (C_plus - C) / eps
            
            # Update step
            h -= lr * grad
            h = np.clip(h, 0, 1)
            
            # Check improvement
            new_C = compute_C(h)
            if new_C < best_refined_C:
                best_refined_C = new_C
                best_refined_h = h.copy()
            else:
                lr *= 0.95  # Reduce learning rate
    
    return (best_refined_h.tolist(), best_refined_C, n_points)
