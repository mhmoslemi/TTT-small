import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        # Initialize with a simple construction that satisfies constraints
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def get_segments(h, m):
        # Segment h into m parts and compute averages
        segments = []
        segment_size = n_points // m
        for i in range(m):
            start_idx = i * segment_size
            end_idx = (i + 1) * segment_size if i < m - 1 else n_points
            segment_avg = np.mean(h[start_idx:end_idx])
            segments.append(segment_avg)
        return np.array(segments)
    
    def reconstruct_h(segments, m):
        # Reconstruct h from segments using simple uniform assignment
        h = np.zeros(n_points)
        segment_size = n_points // m
        for i in range(m):
            start_idx = i * segment_size
            end_idx = (i + 1) * segment_size if i < m - 1 else n_points
            h[start_idx:end_idx] = segments[i]
        return h
    
    def smooth_h(h, sigma=1.0):
        # Apply Gaussian smoothing
        if len(h) < 3:
            return h
        # Simple Gaussian-like smoothing
        kernel = np.exp(-np.arange(-3, 4)**2 / (2 * sigma**2))
        kernel = kernel / np.sum(kernel)
        return np.convolve(h, kernel, mode='same')
    
    # Begin optimization loop at varying resolutions
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    # Try different segmentation sizes
    segment_sizes = [7, 14, 21, 28, 35, 42, 49, 56, 63] # up to n_points
    found_improvement = True
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        found_improvement = False
        for m in segment_sizes:
            if time.time() - start_time >= budget_s - 1:
                break
            iter_count += 1
            # Get current best segments
            segments = get_segments(best_h, m)
            
            # Optimization function
            def obj(seg_vals):
                reconstructed = reconstruct_h(seg_vals, m)
                # Apply smoothing for continuity
                smoothed = smooth_h(reconstructed)
                # Enforce constraints
                smoothed = np.clip(smoothed, 0, 1)
                # Normalize to integral 1
                smoothed = smoothed * (n_points / 2) / np.sum(smoothed)
                return evaluate(smoothed)
            
            # Constraints: segments must be feasible
            bounds = [(0, 1) for _ in range(m)]
            # Add constraint for sum of segments
            def constraint_func(seg_vals):
                return np.sum(seg_vals) - m/2
            
            cons = {'type': 'eq', 'fun': constraint_func}
            
            try:
                res = minimize(obj, segments, method='SLSQP', bounds=bounds, constraints=cons, 
                               options={'maxiter': 500, 'ftol': 1e-8})
                if res.success:
                    new_segments = res.x
                    new_h = reconstruct_h(new_segments, m)
                    new_h = smooth_h(new_h)
                    new_h = np.clip(new_h, 0, 1)
                    # Normalize
                    new_h = new_h * (n_points / 2) / np.sum(new_h)
                    new_c5 = evaluate(new_h)
                    
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h.copy()
                        found_improvement = True
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
                else:
                    no_improvement_count += 1
            except Exception as e:
                no_improvement_count += 1
                
        if not found_improvement:
            no_improvement_count += 1
    
    return (best_h, best_c5, n_points)
