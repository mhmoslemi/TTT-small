import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangular pulse
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x) + np.sin(np.pi * (x - 1)))
    h = project_h(h, n_points)
    return h

def line_search(h, grad, max_step, n_points, dx):
    """Find optimal step size using bracketed line search."""
    def c5_at_alpha(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Bracketing
    a, b = 0.0, max_step
    fa = c5_at_alpha(a)
    fb = c5_at_alpha(b)
    
    # If b is better, keep expanding
    while fb < fa:
        a, b = b, 2 * b
        fa, fb = fb, c5_at_alpha(b)
    
    # Bisection within the bracket
    for _ in range(20):
        c = (a + b) / 2
        fc = c5_at_alpha(c)
        if fc < fa:
            b, fb = c, fc
        else:
            a, fa = c, fc
    
    return (a + b) / 2

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45  # Use smaller size for faster runtime
    h = initialize_h(n_points)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Parameters
    eta = 0.1  # initial learning rate
    eta_min = 1e-6
    eta_max = 0.5
    patience = 0
    max_patience = 1000
    dx = 2.0 / n_points
    
    while time.time() - start_time < budget_s - 1:
        # Compute the full correlation and find worst shift
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        worst_shift = max_idx - (n_points - 1)
        
        # Compute gradient for the worst shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + worst_shift) % n_points
            grad[i] = (1 - h[j]) * dx
        
        # Line search to find optimal step
        alpha = line_search(h, grad, eta, n_points, dx)
        
        # Update with step
        h = h - alpha * grad
        h = project_h(h, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 0
            eta = min(eta * 1.2, eta_max)
        else:
            patience += 1
            eta = max(eta / 2, eta_min)
            
        if patience > max_patience:
            break
    
    return (best_h, best_c5, n_points)
