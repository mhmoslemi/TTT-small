import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalization
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def reconstruct_h(c, m):
        # Reconstruct h from segment values
        h = np.zeros(n_points)
        segment_width = n_points // m
        remainder = n_points % m
        for i in range(m):
            start = i * segment_width + min(i, remainder)
            end = start + segment_width + (1 if i < remainder else 0)
            h[start:end] = c[i]
        return h
    
    def optimize_segments(c0, m):
        # Optimize segment values to reduce C5
        def objective(c):
            h = reconstruct_h(c, m)
            return evaluate(h)
        
        def constraint(c):
            return np.sum(c) - m / 2  # Sum should be m/2
        
        bounds = [(0, 1) for _ in range(m)]
        cons = {'type': 'eq', 'fun': constraint}
        
        result = minimize(objective, c0, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'disp': False, 'maxiter': 100})
        
        return result.x if result.success else c0
    
    def perturb(c, frac=0.1, delta=0.1):
        # Perturb a fraction of the variables
        n = len(c)
        indices = np.random.choice(n, max(1, int(n * frac)), replace=False)
        c_new = c.copy()
        for i in indices:
            c_new[i] += np.random.uniform(-delta, delta)
        c_new = np.clip(c_new, 0, 1)
        # Renormalize
        c_new = c_new * (len(c) / 2) / np.sum(c_new)
        return c_new
    
    def smooth(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    best_c5 = float('inf')
    best_h = h_start.copy()
    best_c = None
    
    # Start with fewer segments, increase resolution
    num_segs = 7
    max_segs = 21
    no_improvement = 0
    iter_count = 0
    
    while time.time() - start_time < budget_s and no_improvement < 50:
        iter_count += 1
        
        # Reconstruct initial
        c_current = np.ones(num_segs) * 0.5  # Default segment values
        if best_c is not None:
            # Interpolate from previous result
            c_current = np.interp(np.linspace(0, num_segs-1, len(best_c)), 
                                  np.arange(len(best_c)), best_c)
            if len(c_current) != num_segs:
                c_current = np.interp(np.linspace(0, num_segs-1, num_segs),
                                      np.linspace(0, len(c_current)-1, len(c_current)),
                                      c_current)
        
        # Optimize segment values
        c_opt = optimize_segments(c_current, num_segs)
        h_opt = reconstruct_h(c_opt, num_segs)
        
        # Smooth the result
        h_opt = smooth(h_opt, window=3)
        
        # Ensure it's normalized
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        h_opt = np.clip(h_opt, 0, 1)
        
        c5_opt = evaluate(h_opt)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_c = c_opt.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Occasionally perturb
        if iter_count % 5 == 0:
            c_pert = perturb(best_c, frac=0.1, delta=0.1)
            # Rerun optimization with perturbed segment values
            h_pert = reconstruct_h(c_pert, num_segs)
            h_pert = smooth(h_pert, window=3)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            h_pert = np.clip(h_pert, 0, 1)
            c5_pert = evaluate(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                best_c = c_pert.copy()
                no_improvement = 0
        
        # Increase resolution if no improvement
        if no_improvement > 20 and num_segs < max_segs:
            num_segs += 2
            no_improvement = 0
            # Reduce iteration count for increase
            iter_count = 0
    
    return (best_h, best_c5, n_points)
