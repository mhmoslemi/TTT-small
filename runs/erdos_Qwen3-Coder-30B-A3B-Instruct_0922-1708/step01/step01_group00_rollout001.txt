import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a triangular distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to higher resolution
        h_orig = h.copy()
        h = np.interp(np.linspace(0, 2, n_points), 
                      np.linspace(0, 2, len(h_orig)), 
                      h_orig)
    else:
        # Create a triangular-like distribution
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(x - 1)
        h = np.clip(h, 0, 1)
    
    # Normalize to integral = 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.01
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
        
        # Compute gradient using finite differences with better sampling
        eps = 1e-5
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration+1))
        v_hat = v / (1 - beta2**(iteration+1))
        
        # Update parameters
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 1000 == 0 and iteration > last_improvement:
            lr *= 0.5
            if lr < 1e-6:
                lr = 1e-6
                
        # Early stopping check
        if iteration - last_improvement > 2000:
            break
    
    return (best_h.tolist(), best_c5, n_points)
