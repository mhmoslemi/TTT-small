import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def segment_based_optimization(h_start, m, max_iter=500):
        # Initialize segments
        segment_width = n_points // m
        if segment_width * m < n_points:
            segment_width += 1
        segment_indices = [i * segment_width for i in range(m)] + [n_points]
        
        # Start with uniform segments
        c_values = np.ones(m) * 0.5
        
        def get_h_from_c(c_vals):
            h_vals = np.zeros(n_points)
            for i in range(m):
                start_idx = segment_indices[i]
                end_idx = segment_indices[i+1]
                h_vals[start_idx:end_idx] = c_vals[i]
            return h_vals
        
        def objective(c_vals):
            h = get_h_from_c(c_vals)
            return evaluate(h)
        
        # Constraint: sum of c * segment_width = 1 * n_points / 2
        def constraint_func(c_vals):
            return np.sum(c_vals) * segment_width - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(m)]
        
        result = minimize(objective, c_values, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': max_iter})
        c_opt = result.x
        h_opt = get_h_from_c(c_opt)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        return h_opt
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    no_improvement_count = 0
    m = 7  # starting number of segments
    
    while time.time() - start_time < budget_s and no_improvement_count < 50:
        # Perform segment-based optimization
        h_trial = segment_based_optimization(best_h, m)
        
        # Perturb a few values to escape local minima
        perturb_indices = np.random.choice(m, max(1, m // 10), replace=False)
        perturb_amount = 0.05
        for i in perturb_indices:
            h_trial = segment_based_optimization(best_h, m, 50)
            break  # Only use the optimized version once
        
        # Apply smoothing
        h_trial_smooth = np.convolve(h_trial, np.ones(3)/3, mode='same')
        h_trial_smooth = np.clip(h_trial_smooth, 0, 1)
        h_trial_smooth = h_trial_smooth * (n_points / 2) / np.sum(h_trial_smooth)
        
        c5_trial = evaluate(h_trial_smooth)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial_smooth.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Increase resolution if no improvement for a while
        if no_improvement_count >= 20:
            if m < 21:  # max segments
                m += 2
                no_improvement_count = 0
    
    # Final refinement at highest resolution
    h_final = segment_based_optimization(best_h, m, 200)
    c5_final = evaluate(h_final)
    
    # Return the best solution
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final
    
    return (best_h, best_c5, n_points)
