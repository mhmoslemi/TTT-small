import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(conv)
        return max_val * dx
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(n_particles):
        # Initialize particles around the current best if available, otherwise randomly
        if i == 0:
            h_init = h.copy()
        else:
            h_init = np.random.rand(n_points) * 0.5 + 0.25  # More diverse initialization
            
        h_init = project_h(h_init)
        particles.append(h_init)
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
        pbest.append(h_init.copy())
        pbest_vals.append(compute_c5(h_init))
    
    # Set global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    max_iter = 10000
    no_improve_iter = 0
    max_no_improve = 1000
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * (iteration / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_h(particles[i])  # Ensure feasibility
            
            # Evaluate new position
            current_val = compute_c5(particles[i])
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best if needed
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
                    no_improve_iter = 0
                else:
                    no_improve_iter += 1
                    
        if no_improve_iter >= max_no_improve:
            break
            
    # Final refinement using local search
    best_h = gbest.copy()
    best_c5 = gbest_val
    
    # Try different mutations of the best solution to improve
    for _ in range(20):
        if (time.time() - start_time) > budget_s - 1:
            break
        # Random perturbation of the solution
        h_perturb = best_h.copy()
        idx = np.random.randint(0, n_points)
        change = np.random.normal(0, 0.02)
        h_perturb[idx] += change
        
        # Project to constraints
        h_perturb = project_h(h_perturb)
        
        # Check if this improves C5
        c5_perturb = compute_c5(h_perturb)
        if c5_perturb < best_c5:
            best_h = h_perturb
            best_c5 = c5_perturb
    
    return (best_h.tolist(), best_c5, len(best_h))
