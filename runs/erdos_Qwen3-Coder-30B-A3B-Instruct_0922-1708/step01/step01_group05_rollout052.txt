import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from globals or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    max_iter = 5000
    w = 0.7  # Inertia weight
    c1 = 1.5  # Cognitive parameter
    c2 = 1.5  # Social parameter
    
    # Initialize PSO particles (params = h values)
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest = h.copy()
    gbest_val = float('inf')
    
    # Initialize particles
    for _ in range(n_particles):
        particle = h + np.random.normal(0, 0.01, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (target_sum / np.sum(particle))
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        pbest.append(particle.copy())
        pbest_vals.append(float('inf'))
        
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    iter_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            
            # Evaluate fitness
            fitness = compute_C(particles[i])
            
            # Update personal best
            if fitness < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = fitness
                
            # Update global best
            if fitness < gbest_val:
                gbest = particles[i].copy()
                gbest_val = fitness
                
        iter_count += 1
        
    # Refinement with gradient descent
    best_h = gbest.copy()
    best_C = gbest_val
    
    # Gradient descent refinement - start from gbest
    lr = 0.01
    no_improvement_count = 0
    current_h = best_h.copy()
    
    def compute_C_grad(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def compute_gradient(h_vals):
        # Approximate gradient via finite differences
        eps = 1e-5
        grad = np.zeros_like(h_vals)
        base_C = compute_C_grad(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = h_plus * (target_sum / np.sum(h_plus))  # Rescale
            C_plus = compute_C_grad(h_plus)
            grad[i] = (C_plus - base_C) / eps
        return grad
    
    for _ in range(1000):
        if time() - start_time > budget_s - 1:
            break
            
        grad = compute_gradient(current_h)
        current_h -= lr * grad
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (target_sum / np.sum(current_h))
        
        new_C = compute_C_grad(current_h)
        if new_C < best_C:
            best_C = new_C
            best_h = current_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 100:
                lr *= 0.5
                no_improvement_count = 0
    
    return (best_h.tolist(), best_C, n_points)
