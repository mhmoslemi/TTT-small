import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given step function h"""
    correlation = correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def rescale_h(h, target_sum=1.0):
    """Rescale h to have the target sum"""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate_individual(h, mutation_rate=0.1, sigma=0.05):
    """Mutate an individual by adding Gaussian noise"""
    h_new = h.copy()
    n_points = len(h)
    indices_to_mutate = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
    noise = np.random.normal(0, sigma, len(indices_to_mutate))
    h_new[indices_to_mutate] += noise
    return np.clip(h_new, 0, 1)

def crossover(parent1, parent2):
    """Arithmetic crossover between two parents"""
    alpha = np.random.random()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2  # because dx = 2/n_points, we want sum(h) * dx = 1
    
    pop_size = 50
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            # Interpolate to correct resolution
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = rescale_h(h_init, target_sum)
        population.append(h_init)
    
    # Fill remaining population with random individuals
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = rescale_h(h, target_sum)
        population.append(h)
    
    # Evaluation function
    def evaluate_population(pop):
        fitness = []
        for h in pop:
            fit = evaluate_h(h, dx)
            fitness.append(fit)
        return np.array(fitness)
    
    # Main GA loop
    start_time = time.time()
    best_fitness = float('inf')
    best_individual = None
    no_improvement = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate current population
        fitness = evaluate_population(population)
        
        # Track best individual
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_fitness:
            best_fitness = fitness[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            no_improvement = 0
            last_improvement = time.time()
        else:
            no_improvement += 1
            
        if no_improvement > 100 and time.time() - last_improvement > 300:
            break
            
        # Selection: keep top 20%
        num_keep = max(5, int(pop_size * 0.2))
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:num_keep]]
        
        # Generate new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Select two parents
            p1_idx, p2_idx = np.random.choice(num_keep, 2, replace=False)
            p1, p2 = selected[p1_idx], selected[p2_idx]
            
            # Crossover
            child = crossover(p1, p2)
            
            # Mutation
            child = mutate_individual(child)
            
            # Repair
            child = np.clip(child, 0, 1)
            child = rescale_h(child, target_sum)
            
            new_population.append(child)
            
        population = new_population
    
    # Final verification
    if best_individual is not None:
        final_fitness = evaluate_h(best_individual, dx)
        return (best_individual.tolist(), final_fitness, n_points)

        # Fallback to simple optimization if GA didn't work well
        # Use a slightly modified gradient descent with better heuristics
        n_points = 45
        dx = 2.0 / n_points
        
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = h / np.sum(h) * (n_points / 2)
        else:
            h = np.ones(n_points) * 0.5
            
        best_h = h.copy()
        best_c5 = float('inf')
        
        for iteration in range(10000):
            if time.time() - start_time > budget_s - 1:
                break
                
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Use a slightly more sophisticated gradient estimation
            # with adaptive learning rate and momentum
            lr = 0.01
            eps = 1e-5
            
            grad = np.zeros_like(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Apply momentum with better parameter tuning
            h_new = h - lr * grad
            
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new

        return (best_h.tolist(), best_c5, n_points)
