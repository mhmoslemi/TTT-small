import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Create piecewise constant representation
    # Start with 10 intervals
    K = 10
    breakpoints = np.linspace(0, 2, K+1)
    
    # Initialize interval values to match h integral
    interval_values = np.ones(K) * 0.5
    
    # Convert h to interval representation for easier updating
    def h_from_intervals(breakpoints, interval_values):
        h_vals = np.zeros(n_points)
        for i in range(K):
            start_idx = int(breakpoints[i] / dx)
            end_idx = int(breakpoints[i+1] / dx)
            h_vals[start_idx:end_idx] = interval_values[i]
        # Make sure last point is covered
        h_vals[-1] = interval_values[-1]
        return h_vals
    
    def compute_C5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # Gradient computation
    def compute_gradients(breakpoints, interval_values):
        grad_breakpoints = np.zeros(K)
        grad_values = np.zeros(K)
        
        h_vals = h_from_intervals(breakpoints, interval_values)
        C5 = compute_C5(h_vals)
        
        eps = 1e-5
        
        # Compute gradient w.r.t. interval values
        for i in range(K):
            interval_values_new = interval_values.copy()
            interval_values_new[i] += eps
            h_vals_new = h_from_intervals(breakpoints, interval_values_new)
            C5_new = compute_C5(h_vals_new)
            grad_values[i] = (C5_new - C5) / eps
            
        # Compute gradient w.r.t. breakpoints
        for i in range(K):
            breakpoints_new = breakpoints.copy()
            breakpoints_new[i] += eps
            if i > 0:
                breakpoints_new[i] = max(breakpoints_new[i], breakpoints[i-1] + 1e-6)
            if i < K-1:
                breakpoints_new[i] = min(breakpoints_new[i], breakpoints[i+1] - 1e-6)
            h_vals_new = h_from_intervals(breakpoints_new, interval_values)
            C5_new = compute_C5(h_vals_new)
            grad_breakpoints[i] = (C5_new - C5) / eps
            
        return grad_breakpoints, grad_values
    
    # Optimization
    start_time = time.time()
    best_C5 = float('inf')
    best_breakpoints = breakpoints.copy()
    best_interval_values = interval_values.copy()
    iteration = 0
    
    lr_bp = 0.01
    lr_iv = 0.1
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Flatten current state
        h_vals = h_from_intervals(breakpoints, interval_values)
        C5 = compute_C5(h_vals)
        
        if C5 < best_C5:
            best_C5 = C5
            best_breakpoints = breakpoints.copy()
            best_interval_values = interval_values.copy()
            
        # Gradient computation
        grad_b, grad_v = compute_gradients(breakpoints, interval_values)
        
        # Update interval values
        interval_values -= lr_iv * grad_v
        interval_values = np.clip(interval_values, 0, 1)
        
        # Update breakpoints
        breakpoints -= lr_bp * grad_b
        # Ensure monotonicity
        for i in range(1, K):
            breakpoints[i] = max(breakpoints[i], breakpoints[i-1] + 1e-6)
        breakpoints = np.clip(breakpoints, 0, 2)
        
        # Maintain the integral constraint
        h_vals = h_from_intervals(breakpoints, interval_values)
        integral = np.sum(h_vals) * dx
        scale_factor = 1.0 / integral * (n_points / 2)
        interval_values *= scale_factor
    
    # Final conversion to the required format
    final_h = h_from_intervals(best_breakpoints, best_interval_values)
    return (final_h.tolist(), best_C5, n_points)
