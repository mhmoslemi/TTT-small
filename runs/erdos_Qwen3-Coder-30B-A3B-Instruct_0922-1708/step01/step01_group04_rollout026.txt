import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def project(h):
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def perturb(h, strength=0.1):
        h_new = h.copy()
        # Perturb about 10% of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-strength, strength)
        return project(h_new)
    
    # Initial setup
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    patience = 100
    
    while time.time() - start_time < budget_s and no_improvement_count < patience:
        iterations += 1
        
        # Try multiple perturbations
        for _ in range(10):
            h_trial = perturb(best_h, strength=0.05)
            h_trial = smooth(h_trial, window=3)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                break
        else:
            no_improvement_count += 1
    
    # Final refinement with optimization if time allows
    if time.time() - start_time < 0.8 * budget_s:
        # Re-initialize with current best
        h0 = best_h.copy()
        # Try gradient-based optimization for final refinement
        def objective(h):
            return evaluate(h)
            
        def constraint(h):
            return np.sum(h) * dx - 1.0
            
        cons = {'type': 'eq', 'fun': constraint}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = minimize(objective, h0, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                              options={'ftol': 1e-8, 'gtol': 1e-6})
            if result.success:
                h_opt = project(result.x)
                c5_opt = evaluate(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass
    
    return (best_h, best_c5, n_points)
