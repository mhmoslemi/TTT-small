import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Use initial_h_values if available
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangular pulses
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def coordinate_wise_optimization(h, n_points, time_budget_s):
    """Coordinate-wise optimization with line search."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Parameters
    step_size = 0.05
    max_iter = 5000
    iter_no_improvement = 0
    max_no_improvement = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_budget_s * 0.8:
            break
            
        improvement = False
        for i in range(n_points):
            # Save current value
            old_val = best_h[i]
            old_c5 = best_c5
            
            # Try increasing value
            new_val_inc = min(1.0, old_val + step_size)
            if new_val_inc != old_val:
                best_h[i] = new_val_inc
                best_h = project_h(best_h, n_points)
                new_c5 = evaluate_c5(best_h)
                if new_c5 < old_c5:
                    improvement = True
                    old_c5 = new_c5
                else:
                    best_h[i] = old_val
                    best_h = project_h(best_h, n_points)
            
            # Try decreasing value
            new_val_dec = max(0.0, old_val - step_size)
            if new_val_dec != old_val:
                best_h[i] = new_val_dec
                best_h = project_h(best_h, n_points)
                new_c5 = evaluate_c5(best_h)
                if new_c5 < old_c5:
                    improvement = True
                    old_c5 = new_c5
                else:
                    best_h[i] = old_val
                    best_h = project_h(best_h, n_points)
            
            if improvement:
                best_c5 = old_c5
                iter_no_improvement = 0
            else:
                iter_no_improvement += 1
                if iter_no_improvement > max_no_improvement:
                    break
        
        if not improvement:
            iter_no_improvement += 1
            if iter_no_improvement > max_no_improvement:
                break
        else:
            iter_no_improvement = 0
            
        if step_size > 1e-6 and iteration > 100:
            step_size *= 0.95
            
    return best_h, best_c5

def nonlinear_conjugate_gradient(h, n_points, time_budget_s):
    """Nonlinear conjugate gradient descent."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Parameters
    max_iter = 5000
    tolerance = 1e-6
    max_no_improvement = 500
    
    # Initialize
    old_grad = np.zeros(n_points)
    direction = np.zeros(n_points)
    
    iteration = 0
    iter_no_improvement = 0
    
    while iteration < max_iter and time.time() - start_time < time_budget_s * 0.9:
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Polak-Ribière formula for conjugate direction
        if iteration == 0:
            direction = -grad
        else:
            beta = np.dot(grad, grad - old_grad) / np.dot(old_grad, old_grad)
            beta = max(0, beta)
            direction = -grad + beta * direction
        
        # Line search to find optimal step size
        def line_search_func(alpha):
            new_h = best_h + alpha * direction
            new_h = project_h(new_h, n_points)
            return evaluate_c5(new_h)
        
        # Simple backtracking line search
        alpha = 1.0
        c5_current = evaluate_c5(best_h)
        while True:
            new_h = best_h + alpha * direction
            new_h = project_h(new_h, n_points)
            new_c5 = evaluate_c5(new_h)
            if new_c5 < c5_current:
                break
            alpha *= 0.5
            if alpha < 1e-10:
                break
        
        # Update
        best_h = best_h + alpha * direction
        best_h = project_h(best_h, n_points)
        new_c5 = evaluate_c5(best_h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
        
        old_grad = grad
        iteration += 1
        
        if iter_no_improvement > max_no_improvement:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: Coordinate-wise optimization
    h1, c5_1 = coordinate_wise_optimization(h, n_points, budget_s)
    
    # Phase 2: Nonlinear conjugate gradient
    h2, c5_2 = nonlinear_conjugate_gradient(h1, n_points, budget_s)
    
    # Return best result
    final_h = h2 if c5_2 < c5_1 else h1
    final_c5 = min(c5_2, c5_1)
    
    # Ensure final result is valid
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
