import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def latin_hypercube_sampling(n_points, n_samples):
    """Generate Latin Hypercube samples for initial population"""
    samples = []
    for _ in range(n_samples):
        # Generate uniform random samples in [0, 1]
        u = np.random.rand(n_points)
        # Generate a random permutation for each dimension
        perm = np.random.permutation(n_points)
        # Create a Latin Hypercube sample
        sample = (perm + u) / n_points
        samples.append(sample)
    return samples

def generate_initial_population(n_points, n_samples, initial_h):
    """Generate diverse initial population"""
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = project_h(h_parent, n_points)
        population = [h_parent]
    else:
        # Initialize with sinusoidal pattern
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
        population = [h]
    
    # Add Latin Hypercube samples
    lhs_samples = latin_hypercube_sampling(n_points, n_samples - len(population))
    for sample in lhs_samples:
        h = sample * (n_points / 2)  # Scale to correct sum
        h = project_h(h, n_points)
        population.append(h)
    
    return population

def evolution_strategy_phase(initial_h, n_points, budget_s, dx):
    """Evolution strategy with elites and offspring"""
    start_time = time.time()
    
    n_samples = 50
    n_elites = 20
    n_offspring = 5
    
    # Generate initial population
    population = generate_initial_population(n_points, n_samples, initial_h)
    
    # Evaluate all candidates
    fitness = [evaluate_h(h, dx) for h in population]
    
    # Track best solution
    min_fitness_idx = np.argmin(fitness)
    best_h = population[min_fitness_idx].copy()
    best_c5 = fitness[min_fitness_idx]
    
    while time.time() - start_time < 0.5 * budget_s:
        # Select elites
        elite_indices = np.argsort(fitness)[:n_elites]
        elites = [population[i] for i in elite_indices]
        
        # Generate offspring for each elite
        offspring = []
        for elite in elites:
            for _ in range(n_offspring):
                # Small random perturbation
                mut_strength = 0.1
                noise = np.random.normal(0, mut_strength, n_points)
                child = elite + noise
                child = project_h(child, n_points)
                offspring.append(child)
        
        # Combine elites and offspring
        population = elites + offspring
        
        # Re-evaluate
        fitness = [evaluate_h(h, dx) for h in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
    
    return best_h, best_c5

def gradient_descent_refinement(h, n_points, budget_s, dx):
    """Refine the best solution using gradient descent"""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < 0.5 * budget_s:
        # Compute the correlation to find the worst shift
        corr = correlate(best_h, 1 - best_h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient approximate using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with sign of gradient to reduce oscillations
        new_h = best_h - learning_rate * np.sign(grad)
        new_h = project_h(new_h, n_points)
        
        new_c5 = evaluate_h(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Evolution strategy phase
    best_h, best_c5 = evolution_strategy_phase(h, n_points, budget_s, dx)
    
    # Gradient descent refinement
    refined_h, refined_c5 = gradient_descent_refinement(best_h, n_points, budget_s, dx)
    
    return (refined_h, refined_c5, n_points)
