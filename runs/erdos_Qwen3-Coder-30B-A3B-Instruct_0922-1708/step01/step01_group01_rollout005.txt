import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w = 0.7
    c1 = c2 = 1.5
    inertia = w
    max_stagnation = 500
    min_swarm_size = 20
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(swarm_size):
        if np.random.rand() < 0.1:
            particle = np.random.rand(n_points)
        else:
            particle = h + np.random.normal(0, 0.05, n_points)
        particle = project_h(particle, n_points)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_vals.append(evaluate_h(particle, dx))
    
    # Global best tracking
    gb_idx = np.argmin(pbest_vals)
    gbest = pbest[gb_idx].copy()
    gbest_val = pbest_vals[gb_idx]
    
    # PSO iteration
    start_time = time.time()
    stagnation = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Update swarm size if stagnated
        if stagnation > max_stagnation:
            swarm_size = max(min_swarm_size, int(swarm_size * 0.8))
            if swarm_size < min_swarm_size:
                swarm_size = min_swarm_size
            stagnation = 0
            # Reinitialize swarm
            new_swarm = []
            new_velocities = []
            new_pbest = []
            new_pbest_vals = []
            for _ in range(swarm_size):
                if np.random.rand() < 0.1:
                    particle = np.random.rand(n_points)
                else:
                    particle = gbest + np.random.normal(0, 0.05, n_points)
                particle = project_h(particle, n_points)
                new_swarm.append(particle)
                new_velocities.append(np.zeros(n_points))
                new_pbest.append(particle.copy())
                new_pbest_vals.append(evaluate_h(particle, dx))
            swarm = new_swarm
            velocities = new_velocities
            pbest = new_pbest
            pbest_vals = new_pbest_vals
            gb_idx = np.argmin(pbest_vals)
            gbest = pbest[gb_idx].copy()
            gbest_val = pbest_vals[gb_idx]
            last_improvement = iteration
            
        # Update particles
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (pbest[i] - swarm[i]) + 
                            c2 * r2 * (gbest - swarm[i]))
            # Update position
            swarm[i] = project_h(swarm[i] + velocities[i], n_points)
            # Evaluate
            current_val = evaluate_h(swarm[i], dx)
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = current_val
                if current_val < gbest_val:
                    gbest = swarm[i].copy()
                    gbest_val = current_val
                    last_improvement = iteration
                    stagnation = 0
            else:
                stagnation += 1
                
        # Decrease inertia if no improvement for a while
        if iteration - last_improvement > 200:
            inertia = max(0.3, w * 0.995)
    
    # Gradient descent refinement
    h = gbest.copy()
    best_h = h.copy()
    best_c5 = gbest_val
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s:
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute finite difference gradient for worst shift
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with sign of gradient
        update_direction = np.sign(grad)
        h -= learning_rate * update_direction
        h = project_h(h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > 2000:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (best_h, best_c5, n_points)
