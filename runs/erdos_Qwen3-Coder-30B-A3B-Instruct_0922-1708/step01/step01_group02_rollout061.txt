import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 47
        # Create symmetric initial construction
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 5000
    mutation_rate = 0.2
    crossover_rate = 0.7
    elitism_count = population_size // 10
    
    # Initialize population
    population = []
    population.append(h.copy())
    for i in range(population_size - 1):
        h_init = np.random.rand(n_points)
        h_init = project_h(h_init)
        population.append(h_init)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    def compute_population_c5s(population):
        return [compute_c5(ind) for ind in population]
    
    def roulette_selection(population_c5s):
        # Convert to selection weights (inverse of C5)
        weights = [1.0 / (c5 + 1e-12) for c5 in population_c5s]
        total_weight = sum(weights)
        if total_weight <= 0:
            # Fallback to uniform selection
            return np.random.choice(len(population_c5s), size=2)
        probabilities = [w / total_weight for w in weights]
        return np.random.choice(len(population_c5s), size=2, p=probabilities)
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(individual, generation):
        rate = mutation_rate / np.sqrt(generation + 1)
        if np.random.rand() > rate:
            return individual.copy()
        child = individual.copy()
        noise_std = 0.1
        for i in range(n_points):
            if np.random.rand() < rate:
                child[i] += np.random.normal(0, noise_std)
        return project_h(child)
    
    # Main GA loop
    for gen in range(generations):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Compute fitness for entire population
        c5_values = compute_population_c5s(population)
        
        # Update best solution
        min_c5_idx = np.argmin(c5_values)
        if c5_values[min_c5_idx] < best_c5:
            best_h = population[min_c5_idx].copy()
            best_c5 = c5_values[min_c5_idx]
        
        # Sort population by fitness
        sorted_indices = np.argsort(c5_values)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_c5s = [c5_values[i] for i in sorted_indices]
        
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elitism_count):
            new_population.append(sorted_population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent_indices = roulette_selection(sorted_c5s)
            parent1, parent2 = population[parent_indices[0]], population[parent_indices[1]]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, gen)
            child2 = mutate(child2, gen)
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
