import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def line_search_update(h, i, shift, n_points):
    """Performs line search to compute optimal update for h[i]."""
    dx = 2.0 / n_points
    # We must compute gradients of correlation with respect to h[i] for fixed shift
    # For simplicity, we do finite difference to approximate this
    eps = 1e-5
    h_plus = h.copy()
    h_plus[i] += eps
    h_plus = project_h(h_plus, n_points)
    
    h_minus = h.copy()
    h_minus[i] -= eps
    h_minus = project_h(h_minus, n_points)
    
    c5_plus = evaluate_c5(h_plus)
    c5_minus = evaluate_c5(h_minus)
    
    grad = (c5_plus - c5_minus) / (2 * eps)
    
    # Line search step size
    step_size = 0.05
    new_h = h.copy()
    new_h[i] = h[i] - step_size * grad
    new_h = project_h(new_h, n_points)
    
    # Check if this is actually an improvement
    new_c5 = evaluate_c5(new_h)
    if new_c5 < evaluate_c5(h):
        return new_h
    else:
        return h

def coordinate_wise_update(h, n_points, max_iter=1000):
    """Update coordinates with coordinate-wise optimization."""
    old_h = h.copy()
    for _ in range(max_iter):
        improvement = False
        for i in range(n_points):
            new_h = line_search_update(old_h, i, i, n_points)
            if not np.allclose(old_h, new_h):
                old_h = new_h
                improvement = True
        if not improvement:
            break
    return old_h

def conjugate_gradient_update(h, n_points, max_iter=1000):

    def func(h):
        return evaluate_c5(h)
    
    def grad(h):

        g = np.zeros_like(h)

            h_plus[i] += eps

            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)

        return g

    x = h.copy()
    g = grad(x)
    d = -g

                x_new = x + alpha * d
                x_new = project_h(x_new, n_points)

            x_new = x + alpha * d
