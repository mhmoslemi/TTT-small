import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # PSO parameters
    n_particles = 50
    max_iter = 10000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        # Initialize particle
        new_h = np.random.rand(n_points) * 0.5  # Start with smaller values
        new_h = project_h(new_h)
        particles.append(new_h.copy())
        velocities.append(np.random.uniform(-1, 1, n_points) * 0.1)
        pbest.append(new_h.copy())
        pbest_vals.append(compute_c5(new_h))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iteration
    for iter_num in range(max_iter):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Update inertia
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = project_h(particles[i])
            
            # Evaluate
            current_val = compute_c5(particles[i])
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
        
        # Early stopping
        if iter_num % 1000 == 0 and iter_num > 0:
            # Check if we're not improving significantly
            if abs(gbest_val - best_c5) < 1e-6:
                pass  # Continue
            else:
                best_c5 = gbest_val
                best_h = gbest.copy()
    
    # Final check with best
    final_c5 = compute_c5(gbest)
    final_h = project_h(gbest)
    
    return (final_h.tolist(), final_c5, len(final_h))
