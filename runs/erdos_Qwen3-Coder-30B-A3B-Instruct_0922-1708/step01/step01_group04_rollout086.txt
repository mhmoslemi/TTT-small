import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def segment_based_optimization(h_init, n_segments=7):
        # Create segments
        segment_size = n_points // n_segments
        remainder = n_points % n_segments
        
        # Initialize segment values
        segment_means = np.ones(n_segments) * 0.5
        segment_means[0] = 0.2
        segment_means[-1] = 0.8
        
        # Constraint: sum of segment means should be n_segments / 2
        target_sum = n_segments / 2.0
        current_sum = np.sum(segment_means)
        if current_sum > 0:
            segment_means = segment_means * target_sum / current_sum
        
        def objective(segments):
            # Recreate h from segments
            h = np.zeros(n_points)
            for i in range(n_segments):
                start = i * segment_size
                end = start + segment_size
                if i == n_segments - 1:
                    end = n_points
                
                h[start:end] = segments[i]
            
            # Evaluate overlap
            return evaluate(h)
        
        def constraint_sum(segments):
            return np.sum(segments) - target_sum
        
        # Optimize
        bounds = [(0, 1) for _ in range(n_segments)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(objective, segment_means, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000})
        
        # Reconstruct h
        h = np.zeros(n_points)
        for i in range(n_segments):
            start = i * segment_size
            end = start + segment_size
            if i == n_segments - 1:
                end = n_points
            h[start:end] = res.x[i]
        
        return h, res.fun
    
    # Try optimization with different number of segments
    min_c5 = float('inf')
    best_h = h_start.copy()
    
    start_time = time.time()
    
    # Try different segment counts
    for n_segments in [5, 7, 9, 11, 13]:
        if time.time() - start_time > budget_s - 10:
            break
            
        h_opt, c5 = segment_based_optimization(h_start, n_segments)
        
        if c5 < min_c5:
            min_c5 = c5
            best_h = h_opt.copy()
    
    # Fine-tune with local search if we still have time
    if time.time() - start_time < budget_s - 5:
        # Keep the best of previous and do full local refinement
        h_final = best_h.copy()
        current_c5 = min_c5
        n_perturb = max(1, int(n_points * 0.1))
        
        # Try random perturbations for a while
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_trial = h_final.copy()

            for i in indices:
                h_trial[i] += np.random.uniform(-0.05, 0.05)
            
            # Clip and normalize
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            c5_trial = evaluate(h_trial)
            if c5_trial < current_c5:
                current_c5 = c5_trial
                h_final = h_trial.copy()
                best_h = h_final.copy()
                
    return (best_h, min_c5, n_points)
