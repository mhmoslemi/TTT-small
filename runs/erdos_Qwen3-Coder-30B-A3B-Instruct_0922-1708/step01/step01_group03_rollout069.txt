import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # If initial_h_values is accessible in this scope
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    
    # Default: sin-based initialization
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45  # Use smaller n_points for faster runtimes
    h = initialize_h(n_points)
    
    # Set initial hyperparameters for gradient descent
    eta = 0.1  # Initial learning rate
    min_eta = 1e-6
    max_iter = 10000
    improvement_threshold = 1e-6
    
    # Store the best values
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        iteration += 1
        
        # Find the shift that causes maximum overlap
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        max_shift = max_idx - (n - 1)  # shift is in [-n+1, n-1]
        
        # Compute gradient for this specific shift.
        # Gradient component for h[i]: d/dh[i] (h[i]*(1-h[i+k])) = 1 - h[i+k]
        gradient = np.zeros(n)
        for i in range(n):
            # i + max_shift (mod n) is the variable that gets multiplied with h[i]
            j = (i + max_shift) % n
            gradient[i] = 1 - h[j]
        gradient *= dx
        
        # Line search to find optimal step size alpha
        def line_search_objective(alpha):
            h_new = h - alpha * gradient
            h_new = project_h(h_new, n_points)
            return evaluate_c5(h_new)
        
        # Try different alpha values to find the minimum
        alphas = np.logspace(-4, 0, 20)
        best_alpha = 0
        best_val = line_search_objective(0)
        
        for alpha in alphas:
            val = line_search_objective(alpha)
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Update h with the best alpha found
        h_new = h - best_alpha * gradient
        h_new = project_h(h_new, n_points)
        
        new_c5 = evaluate_c5(h_new)
        
        # Update if better
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Adapt learning rate
        if no_improvement_count > 10:
            eta = max(eta / 2, min_eta)
            no_improvement_count = 0

        # Early stop if improvement is small
        if (iteration > 100) and (abs(best_c5 - evaluate_c5(h)) < improvement_threshold):
            break

    # Final evaluation and projection
    final_h = project_h(h, n_points)
    final_c5 = evaluate_c5(final_h)

    return (final_h, final_c5, n_points)
