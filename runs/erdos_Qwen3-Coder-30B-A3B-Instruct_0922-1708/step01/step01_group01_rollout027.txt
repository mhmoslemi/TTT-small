import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal pattern
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Simulated Annealing
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    T = 1.0
    alpha = 0.999
    max_iter = 10000
    step = 0.1
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > 0.8 * budget_s:
            break
            
        # Mutation
        h_prime = h + np.random.uniform(-step, step, n_points)
        h_prime = project_h(h_prime, n_points)
        
        # Evaluate
        c5_prime = evaluate_h(h_prime, dx)
        
        # Accept or reject
        delta = c5_prime - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_prime.copy()
            if c5_prime < best_c5:
                best_h = h_prime.copy()
                best_c5 = c5_prime
                
        # Cooling
        T *= alpha
        
        # Occasionally run GD refinement
        if i % 100 == 0:
            h_refined = best_h.copy()
            for _ in range(50):
                if time.time() - start_time > 0.8 * budget_s:
                    break
                eps = 1e-5
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = project_h(h_plus, n_points)
                    c_plus = evaluate_h(h_plus, dx)
                    
                    h_minus = h_refined.copy()
                    h_minus[j] -= eps
                    h_minus = project_h(h_minus, n_points)
                    c_minus = evaluate_h(h_minus, dx)
                    
                    grad[j] = (c_plus - c_minus) / (2 * eps)
                
                h_refined -= 0.01 * grad
                h_refined = project_h(h_refined, n_points)
                new_c5 = evaluate_h(h_refined, dx)
                if new_c5 < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = new_c5
                else:
                    break
    
    return (best_h, best_c5, n_points)
