import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import itertools

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction or create one
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Ensure it's within bounds and normalized
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def perturb(h, delta=0.05):
        h_new = h.copy()
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def smooth(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    
    # Switch to a more sophisticated approach after initial phase
    use_sophisticated = False
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        if not use_sophisticated and iterations > 200:
            use_sophisticated = True
        
        if use_sophisticated:
            # Try more diverse perturbations
            for _ in range(50):
                h_trial = perturb(best_h, 0.03)
                h_trial = smooth(h_trial, window=5)
                c5_trial = evaluate(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    no_improvement_count = 0
                    break
            else:
                # Random restarts if no improvement
                if no_improvement_count % 20 == 0:
                    h_trial = np.random.rand(n_points)
                    h_trial = np.clip(h_trial, 0, 1)
                    h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                    c5_trial = evaluate(h_trial)
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        best_h = h_trial.copy()
                        no_improvement_count = 0
                no_improvement_count += 1
        else:
            # Original approach
            for _ in range(100):
                h_trial = perturb(best_h)
                h_trial = smooth(h_trial, window=3)
                c5_trial = evaluate(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    no_improvement_count = 0
                    break
            else:
                no_improvement_count += 1
    
    # Final smoothing
    h_final = smooth(best_h, window=3)
    c5_final = evaluate(h_final)
    
    return (h_final, c5_final, n_points)
