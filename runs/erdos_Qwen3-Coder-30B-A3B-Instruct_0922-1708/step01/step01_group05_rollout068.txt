import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Fourier basis setup
    m = 10  # Number of cosine modes (even symmetry)
    a = np.random.rand(m) * 0.5  # Coefficients for modes 1 to m-1
    a[0] = 0.5  # Ensure integral constraint
    a[1::2] = 0  # Only even modes for symmetry
    phi = np.random.rand(m) * 2 * np.pi
    
    # Optimization loop (Gradient-based)
    best_h = h.copy()
    best_C = float('inf')
    no_improvement_count = 0
    lr = 0.01
    
    def compute_h_from_a_phi(a_coeffs, phi_coeffs):
        # Build h(x) from cosine coefficients
        x = np.linspace(0, 2, n_points)
        h = a_coeffs[0] * np.ones_like(x)
        for k in range(1, len(a_coeffs)):
            h += a_coeffs[k] * np.cos(np.pi * k * x / 2 + phi_coeffs[k])
        return np.clip(h, 0, 1)
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def evaluate_and_update():
        nonlocal best_C, best_h
        h_vals = compute_h_from_a_phi(a, phi)
        # Rescale to satisfy integral constraint
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        C = compute_C(h_vals)
        if C < best_C:
            best_C = C
            best_h = h_vals.copy()
        return C
    
    # Evaluate original starting point
    previous_C = evaluate_and_update()
    
    iter_count = 0
    while time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute current h from a, phi
        h_vals = compute_h_from_a_phi(a, phi)
        h_vals = h_vals * (target_sum / np.sum(h_vals))  # Rescale to sum constraint

        C = compute_C(h_vals)
        
        # Find worst shift k
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        k_max = np.argmax(np.abs(corr))
        
        # Compute gradient with finite difference
        eps = 1e-6
        grad_a = np.zeros_like(a)
        grad_phi = np.zeros_like(phi)
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus = compute_h_from_a_phi(a_plus, phi)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad_a[i] = (C_plus - C) / eps
        
        for i in range(len(phi)):
            phi_plus = phi.copy()
            phi_plus[i] += eps
            h_plus = compute_h_from_a_phi(a, phi_plus)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad_phi[i] = (C_plus - C) / eps
        
        # Update parameters
        a -= lr * grad_a
        a = np.clip(a, 0, 1)
        phi -= lr * grad_phi
        phi = phi % (2 * np.pi)
        
        # Rescale to preserve integral condition
        a[0] = 0.5
        
        # Adaptive learning rate
        new_C = evaluate_and_update()
        if new_C >= previous_C:
            no_improvement_count += 1
            if no_improvement_count > 50:
                lr *= 0.90
                no_improvement_count = 0
        else:
            no_improvement_count = 0
        previous_C = new_C
        
    # Refinement: convert to step function
    h_final = best_h.copy()
    h_bin = np.zeros(n_points)
    # Threshold to make it binary
    h_sorted = np.sort(h_final)[::-1]
    threshold_idx = np.argmax(np.cumsum(h_sorted) >= target_sum) 
    thresh = h_sorted[threshold_idx] if threshold_idx < len(h_sorted) else 0.5
    h_bin = (h_final > thresh).astype(float)
    h_bin = h_bin * (target_sum / np.sum(h_bin))  # rescale integral
    
    # Final evaluation
    final_C = compute_C(h_bin)
    return (h_bin.tolist(), final_C, n_points)
