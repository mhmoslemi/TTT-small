import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # PSO Parameters
    n_particles = 50
    max_iter = 10000
    inertia_w = 0.9
    inertia_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles and velocities
    particles = np.array([project_h(np.random.rand(n_points)) for _ in range(n_particles)])
    velocities = np.random.uniform(-1, 1, (n_particles, n_points))
    
    # Track personal bests and global best
    pbest = particles.copy()
    pbest_c5 = np.array([compute_c5(p) for p in particles])
    
    # Find initial global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = particles[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # PSO iteration
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        inertia_w = inertia_end + (inertia_w - inertia_end) * (1.0 - iter_count / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_h(particles[i])
            
            # Evaluate new position
            current_c5 = compute_c5(particles[i])
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
            
            # Early stopping if no improvement in last 1000 iterations
            if iter_count > 1000 and abs(gbest_c5 - best_c5) < 1e-8:
                break
                
        best_c5 = min(best_c5, gbest_c5)
    
    return (gbest.tolist(), gbest_c5, n_points)
