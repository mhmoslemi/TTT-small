import numpy as np
from time import time
import scipy.optimize as so

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Project h to satisfy integral constraint
    h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    n_particles = 100
    max_iter = 1000
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Initialize particle positions with random perturbations of h
    for i in range(n_particles):
        pos = h + np.random.normal(0, 0.05, n_points)
        pos = np.clip(pos, 0, 1)
        pos = pos * (target_sum / np.sum(pos))  # Normalize
        particles.append(pos.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(pos.copy())
        pbest_vals.append(float('inf'))
    
    # Global best
    gbest = h.copy()
    gbest_val = float('inf')
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def evaluate_particle(pos):
        # Ensure feasibility
        pos = np.clip(pos, 0, 1)
        pos = pos * (target_sum / np.sum(pos))
        return compute_C(pos)
    
    # Main PSO loop
    iter_count = 0
    while time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        for i in range(n_particles):
            pos = particles[i]
            vel = velocities[i]
            
            # Evaluate current position
            current_val = evaluate_particle(pos)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = pos.copy()
                pbest_vals[i] = current_val
            
            # Update global best
            if current_val < gbest_val:
                gbest = pos.copy()
                gbest_val = current_val
            
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            vel = w * vel + c1 * r1 * (pbest[i] - pos) + c2 * r2 * (gbest - pos)
            new_pos = pos + vel
            
            # Apply constraints
            new_pos = np.clip(new_pos, 0, 1)
            if np.sum(new_pos) > 0:
                new_pos = new_pos * (target_sum / np.sum(new_pos))
            
            particles[i] = new_pos.copy()
            velocities[i] = vel.copy()
        
        # Early stopping check
        if iter_count % 10 == 0 and gbest_val < 0.381:
            break
    
    # Now refine using gradient descent
    refined_h = gbest.copy()
    
    def objective(h_vals):
        return compute_C(h_vals)
        
    def gradient(h_vals):
        eps = 1e-6
        grad = np.zeros_like(h_vals)
        base_val = objective(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            grad[i] = (objective(h_plus) - base_val) / eps
        return grad
    
    lr = 0.01
    no_improvement_count = 0
    best_refined = refined_h.copy()
    best_refined_val = gbest_val
    
    # Gradient descent for refinement
    for i in range(1000):
        if time() - start_time > budget_s - 1:
            break
            
        g = gradient(refined_h)
        # Project gradient to stay feasible
        proj_g = g - np.mean(g)  # Subtract mean to maintain sum constraint
        
        # Update
        refined_h = refined_h - lr * proj_g
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = refined_h * (target_sum / np.sum(refined_h))
        
        new_val = objective(refined_h)
        if new_val < best_refined_val:
            best_refined = refined_h.copy()
            best_refined_val = new_val
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > 50:
            lr *= 0.5
            no_improvement_count = 0
            
        if best_refined_val < 0.3807:
            break

    return (best_refined.tolist(), best_refined_val, n_points)
