import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangular pulse
    x = np.linspace(0, 2, n_points)
    h = np.clip(np.sin(np.pi * x) + np.sin(np.pi * (x - 1)), 0, 1)
    h = project_h(h, n_points)
    return h

def gradient_step(h, n_points, eta):
    """Compute one gradient step to reduce the maximum overlap."""
    dx = 2.0 / n_points
    # Compute all correlations
    conv = np.correlate(h, 1 - h, mode="full")
    # Find the shift k that causes maximum overlap
    max_idx = np.argmax(conv)
    shift = max_idx - (n_points - 1)
    
    # Compute the gradient with respect to h for this shift
    # d/dh[i] (sum_j h[j] * (1 - h[j + k])) = 1 - h[i + k]
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + shift) % n_points
        grad[i] = - (1 - h[j]) * dx
    
    # Update h with gradient descent
    h_new = h - eta * grad
    h_new = project_h(h_new, n_points)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Initial learning rate
    eta = 0.1
    max_no_improvement = 10000
    iter_no_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s and iter_no_improvement < max_no_improvement:
        iteration += 1
        
        # Perform one gradient step
        h_new = gradient_step(best_h, n_points, eta)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            # Reset learning rate on improvement
            eta = 0.1
        else:
            iter_no_improvement += 1
            # Reduce learning rate if no improvement
            eta = max(eta / 2, 1e-6)
        
        if iter_no_improvement % 100 == 0:
            # Periodic check of time
            if time.time() - start_time > budget_s * 0.95:
                break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h, final_c5, n_points)
