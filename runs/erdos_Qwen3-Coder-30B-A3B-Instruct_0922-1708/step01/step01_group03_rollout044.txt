import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangular pulse
    x = np.linspace(0, 2, n_points)
    h = np.clip(np.sin(np.pi * x) + np.sin(np.pi * (x - 1)), 0, 1)
    h = project_h(h, n_points)
    return h

def compute_gradient(h, k, n_points):
    """Compute gradient at shift k for the correlation."""
    dx = 2.0 / n_points
    # For shift k, the gradient of the correlation w.r.t. h[i] 
    # is (1 - h[(i+k) % n_points]) * dx
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + k) % n_points
        grad[i] = dx * (1 - h[j])
    return grad

def line_search(h, grad, n_points, max_alpha=0.1):
    """Perform line search to find optimal step size."""
    def c5_func(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Try a range of alpha values
    alphas = np.linspace(0, max_alpha, 100)
    c5_values = [c5_func(alpha) for alpha in alphas]
    best_idx = np.argmin(c5_values)
    return alphas[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using gradient-based method."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    
    # Initial evaluation
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Parameters
    eta = 0.1
    min_eta = 1e-6
    max_iter = 10000
    iter_no_improvement = 0
    max_no_improvement = 100
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Find shift causing maximum overlap
        dx = 2.0 / n_points
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute gradient for this shift
        grad = compute_gradient(h, shift, n_points)
        
        # Line search for optimal step size
        alpha = line_search(h, grad, n_points, eta)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Accept update if it improves C5
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            # Reset learning rate
            eta = 0.1
        else:
            iter_no_improvement += 1
            # Reduce learning rate if no improvement
            eta = max(eta / 2, min_eta)
            
        # Stop if no improvement for too long
        if iter_no_improvement > max_no_improvement:
            break
            
        # Periodically update best solution
        if new_c5 < best_c5:
            best_h = h.copy()
            
    return (best_h, best_c5, n_points)
