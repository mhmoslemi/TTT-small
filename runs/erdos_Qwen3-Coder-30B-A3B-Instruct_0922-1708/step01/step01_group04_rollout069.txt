import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
from math import sqrt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with initial_h_values or create basic construction
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Normalize to integral = 1
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def segment_averages(h, m):
        """Get m segment averages of h"""
        segment_size = n_points // m
        if segment_size == 0:
            segment_size = 1
        c = []
        for i in range(m):
            start_idx = i * segment_size
            end_idx = min((i + 1) * segment_size, n_points)
            segment = h[start_idx:end_idx]
            c.append(np.mean(segment) if len(segment) > 0 else 0)
        return np.array(c)
    
    def reconstruct_h(c, m):
        """Reconstruct h from segment averages c"""
        segment_size = n_points // m
        if segment_size == 0:
            segment_size = 1
        h = np.zeros(n_points)
        for i in range(m):
            start_idx = i * segment_size
            end_idx = min((i + 1) * segment_size, n_points)
            h[start_idx:end_idx] = c[i]
        return h
    
    def smooth_h(h, sigma=1.0):
        """Apply Gaussian smoothing"""
        if len(h) < 3:
            return h
        # Create Gaussian kernel
        kernel_size = min(21, len(h))
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.exp(-np.arange(-(kernel_size//2), kernel_size//2 + 1)**2 / (2 * sigma**2))
        kernel /= np.sum(kernel)
        # Convolve
        h_smooth = np.convolve(h, kernel, mode='same')
        return h_smooth
    
    # Multi-resolution optimization
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    best_n_points = n_points
    
    # Try different resolution levels
    resolutions = [7, 14, 21, 35, 49, 70]  # m values
    
    for m in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Start optimization from current best
        c_start = segment_averages(best_h, m)
        
        def obj(c):
            h_recon = reconstruct_h(c, m)
            h_smooth = smooth_h(h_recon)
            # Normalize
            h_smooth = h_smooth * (n_points / 2) / np.sum(h_smooth)
            return evaluate(h_smooth)
        
        # Constraints
        def constraint_sum(c):
            return np.sum(c) - m/2
        
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(m)]
        
        # Optimize
        try:
            result = minimize(obj, c_start, method='SLSQP', bounds=bounds, constraints=cons, 
                              options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6})
            if result.success:
                c_opt = result.x
                h_recon = reconstruct_h(c_opt, m)
                h_smooth = smooth_h(h_recon)
                h_smooth = h_smooth * (n_points / 2) / np.sum(h_smooth)
                c5_new = evaluate(h_smooth)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_smooth.copy()
                    best_n_points = n_points
        except:
            pass
    
    return (best_h, best_c5, best_n_points)
