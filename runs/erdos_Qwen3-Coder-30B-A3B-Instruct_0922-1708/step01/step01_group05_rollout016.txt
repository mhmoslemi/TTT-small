import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Initialize parameters for sinusoidal basis
    m = 10
    a = np.random.rand(m) * 0.5
    a[0] = 0.5
    phi = np.random.rand(m) * 2 * np.pi
    
    # Optimization loop (Gradient-based)
    best_h = h.copy()
    best_C = float('inf')
    no_improvement_count = 0
    lr = 0.01
    
    def compute_h_from_params(a_coeffs, phi_coeffs):
        x = np.linspace(0, 2, n_points)
        h = a_coeffs[0] * np.ones_like(x)
        for k in range(1, len(a_coeffs)):
            h += a_coeffs[k] * np.cos(np.pi * k * x / 2 + phi_coeffs[k])
        return np.clip(h, 0, 1)
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def rescale_h(h_vals):
        return h_vals * (target_sum / np.sum(h_vals))
        
    def evaluate_and_update():
        nonlocal best_C, best_h
        h_vals = compute_h_from_params(a, phi)
        h_vals = rescale_h(h_vals)
        C = compute_C(h_vals)
        if C < best_C:
            best_C = C
            best_h = h_vals.copy()
        return C
    
    # Evaluate initial point
    prev_C = evaluate_and_update()
    
    iter_count = 0
    while time() - start_time < budget_s - 1:
        iter_count += 1
        h_vals = compute_h_from_params(a, phi)
        h_vals = rescale_h(h_vals)
        C = compute_C(h_vals)
        
        # Find worst shift k
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        k_max = np.argmax(np.abs(corr))
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad_a = np.zeros_like(a)
        grad_phi = np.zeros_like(phi)
        
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus = compute_h_from_params(a_plus, phi)
            h_plus = rescale_h(h_plus)
            C_plus = compute_C(h_plus)
            grad_a[i] = (C_plus - C) / eps
            
        for i in range(len(phi)):
            phi_plus = phi.copy()
            phi_plus[i] += eps
            h_plus = compute_h_from_params(a, phi_plus)
            h_plus = rescale_h(h_plus)
            C_plus = compute_C(h_plus)
            grad_phi[i] = (C_plus - C) / eps
        
        # Update parameters
        a -= lr * grad_a
        phi -= lr * grad_phi
        
        # Clip to valid range
        a = np.clip(a, 0, 1)
        phi = phi % (2 * np.pi)
        
        # Adaptive learning rate
        new_C = evaluate_and_update()
        if new_C >= prev_C:
            no_improvement_count += 1
            if no_improvement_count > 100:
                lr *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
        prev_C = new_C
        
    # Final adjustment with binary refinement
    h_final = compute_h_from_params(a, phi)
    h_final = rescale_h(h_final)
    C_final = compute_C(h_final)
    
    # Binary refinement: threshold for exact discrete solution
    h_binary = np.zeros_like(h_final)
    h_sorted = np.sort(h_final)[::-1]
    threshold_idx = np.searchsorted(h_sorted, 0.5, side="right")  # Aproximate median 
    if threshold_idx == 0:
        threshold = 0.5
    else:
        threshold = np.mean(h_sorted[:threshold_idx])
    h_binary = (h_final > threshold).astype(float)
    h_binary = rescale_h(h_binary)
    C_binary = compute_C(h_binary)
    
    final_h = h_binary if C_binary < C_final else h_final
    final_C = C_binary if C_binary < C_final else C_final
    
    return (final_h.tolist(), final_C, n_points)
