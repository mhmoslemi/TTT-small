from scipy.signal import correlate
import time
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d

        c[0] = 0.2
        c[-1] = 0.8

        total_mass = np.sum(c * segment_lengths) * dx
        c = c * (n_points / 2) / total_mass

        segment_indices = []
        start_idx = 0
        for length in segment_lengths:
            end_idx = int(start_idx + length)
            segment_indices.append(np.arange(int(start_idx), end_idx))
            start_idx = end_idx
        
        # Trust region variables
        trust_radius = 0.1
        last_c = c.copy()
        last_eval = evaluate(make_h_from_c(c, segment_indices, n_points, dx))
        best_c = c.copy()
        best_eval = last_eval
        no_improvement_count = 0
        temp = 1.0
        temp_decay = 0.99
        
        start_time = time.time()
        iteration = 0
        
        while time.time() - start_time < budget_s - 1 and no_improvement_count < 1000:
            iteration += 1
            # Construct h from segments
            h = make_h_from_c(c, segment_indices, n_points, dx)
            # Evaluate current
            current_eval = evaluate(h)
            
            # Compute approximate gradient using finite differences
            epsilon = 1e-4
            grad = np.zeros(segments)

                c_plus[i] += epsilon
                if c_plus[i] > 1: c_plus[i] = 1
                h_plus = make_h_from_c(c_plus, segment_indices, n_points, dx)
                eval_plus = evaluate(h_plus)
                grad[i] = (eval_plus - current_eval) / epsilon
            
            # Update step with trust region
            step = -grad
            step_norm = np.linalg.norm(step)
            if step_norm > trust_radius:
                step = step * trust_radius / step_norm
            
            # Apply step
            c_new = c + step

            if total_mass != 0:

            h_new = make_h_from_c(c_new, segment_indices, n_points, dx)

            else:

                no_improvement_count += 1

                    best_eval = new_eval
                    best_c = c_new.copy()
                    c = c_new
                    no_improvement_count = 0
                else:
                    # Accept with probability
                    if np.random.rand() < np.exp(-(new_eval - best_eval) / temp):
                        c = c_new
                        no_improvement_count = 0
            
            temp *= temp_decay
            
        # Final reconstruction
        final_h = make_h_from_c(best_c, segment_indices, n_points, dx)
        final_h = gaussian_filter1d(final_h, 0.5)
        # Ensure constraints are met
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_eval = evaluate(final_h)
        
        return final_h, final_eval, n_points
    
    def make_h_from_c(c_vals, segment_indices, n_points, dx):

        for i, idxs in enumerate(segment_indices):
            h[idxs] = c_vals[i]
        return h
