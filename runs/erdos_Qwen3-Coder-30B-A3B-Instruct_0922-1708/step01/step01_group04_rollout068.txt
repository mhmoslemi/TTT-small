import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8

    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    def reconstruct_h(c, n_points):
        # Each segment in c is replicated to fill n_points
        m = len(c)
        segment_size = n_points // m
        remainder = n_points % m
        h = np.zeros(n_points)
        for i in range(m):
            start_idx = i * segment_size + min(i, remainder)
            end_idx = start_idx + segment_size + (1 if i < remainder else 0)
            h[start_idx:end_idx] = c[i]
        return h

    def optimize_segments(m, h_start):
        # Number of segments
        c_start = np.ones(m) * (m / 2) / m
        if len(h_start) >= m:
            # Initialize from existing solution
            segment_size = len(h_start) // m
            for i in range(m):
                start_idx = i * segment_size
                end_idx = start_idx + segment_size
                c_start[i] = np.mean(h_start[start_idx:end_idx])
        else:
            # Initialize with equal values
            c_start = np.ones(m) * (m / 2) / m
            
        # Define constraints and bounds
        bounds = [(0, 1) for _ in range(m)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - m/2}]

        # Objective function
        def obj(c):
            h = reconstruct_h(c, n_points)
            return evaluate(h)
        
        # Optimization
        res = minimize(obj, c_start, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 100})
        if res.success:
            return reconstruct_h(res.x, n_points)
        else:
            return reconstruct_h(c_start, n_points)

    # Start the hierarchical optimization
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    # Start with smaller segments and gradually increase
    segment_sizes = [7, 14, 21, 28, 35, 42, 49, 56, 63]  # up to n_points
    segment_sizes = [s for s in segment_sizes if s <= n_points]

    for m in segment_sizes:
        if time.time() - start_time > budget_s - 10:  # Leave some margin
            break
            
        try:
            h_current = optimize_segments(m, best_h)
            c5_current = evaluate(h_current)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        except Exception as e:
            no_improvement_count += 1  # Failure in optimization
            
        iteration += 1
        if no_improvement_count >= 100:
            break

    # Return best solution found
    return (best_h, best_c5, n_points)
