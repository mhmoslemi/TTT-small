import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def latin_hypercube_sampling(n_points, n_samples):
    """Generate Latin Hypercube samples"""
    samples = []
    for _ in range(n_samples):
        # Generate samples from uniform distribution
        x = np.random.rand(n_points)
        # Shuffle each dimension
        for i in range(n_points):
            x[i] = np.random.rand()
        # Apply LHS transformation
        sample = np.sort(x)
        # Scale to [0, 2]
        sample = sample * 2
        # Map to [0, 1] and project to constraints
        h = sample / 2.0
        h = project_h(h, n_points)
        samples.append(h)
    return samples

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    
    # Phase 1: Latin Hypercube Sampling
    samples = latin_hypercube_sampling(n_points, 50)
    samples.append(h)
    sample_c5 = [evaluate_h(s, dx) for s in samples]
    
    # Select best samples
    sorted_indices = np.argsort(sample_c5)
    elites = [samples[i] for i in sorted_indices[:20]]
    elite_c5 = [sample_c5[i] for i in sorted_indices[:20]]
    
    # Phase 2: Evolution Strategy
    while time.time() - start_time < 0.7 * budget_s:
        offspring = []
        for elite in elites:
            for _ in range(5):
                # Perturb elite with small random changes
                child = elite.copy()
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] += np.random.normal(0, 0.05)
                child = project_h(child, n_points)
                offspring.append(child)
        
        # Evaluate offspring
        offspring_c5 = [evaluate_h(s, dx) for s in offspring]
        
        # Combine elites and offspring
        all_solutions = elites + offspring
        all_c5 = elite_c5 + offspring_c5
        
        # Keep best
        sorted_indices = np.argsort(all_c5)
        elites = [all_solutions[i] for i in sorted_indices[:20]]
        elite_c5 = [all_c5[i] for i in sorted_indices[:20]]
        
        # Update global best
        new_best_idx = np.argmin(elite_c5)
        if elite_c5[new_best_idx] < best_c5:
            best_h = elites[new_best_idx].copy()
            best_c5 = elite_c5[new_best_idx]
    
    # Phase 3: Gradient Descent
    h = best_h.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    min_improvement = 1e-5
    
    while time.time() - start_time < budget_s - 1:
        prev_c5 = evaluate_h(h, dx)
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update step with sign of gradient
        h -= learning_rate * np.sign(grad)
        h = project_h(h, n_points)
        
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (best_h, best_c5, n_points)
