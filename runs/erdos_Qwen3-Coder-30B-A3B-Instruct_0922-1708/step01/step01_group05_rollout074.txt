    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():

    else:

    # Ensure sum constraint
    h = h * (target_sum / np.sum(h))
    
    # Optimization parameters
    lr = 0.01
    max_iter = 10000
    no_improvement_patience = 100
    best_C = float('inf')
    best_h = h.copy()
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def project_h(h_vals):
        # Clip and rescale to maintain constraints
        h_proj = np.clip(h_vals, 0, 1)
        h_proj = h_proj * (target_sum / np.sum(h_proj))
        return h_proj
        
    def evaluate():
        nonlocal best_C, best_h
        C = compute_C(h)
        if C < best_C:
            best_C = C
            best_h = h.copy()
        return C
    
    previous_C = evaluate()
    no_improvement_count = 0
    
    # Gradient descent loop
    iter_count = 0
    while time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Compute current overlap integral and find worst shift
        h_vals = h.copy()
        C = compute_C(h_vals)
        
        # Compute finite differences to estimate gradient
        eps = 1e-5
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C) / eps
        
        # Update
        h = h - lr * grad
        h = project_h(h)
        
        # Adaptive learning rate
        new_C = evaluate()
        if new_C >= previous_C:
            no_improvement_count += 1
            if no_improvement_count > no_improvement_patience:
                lr *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
        previous_C = new_C
        
        # Early stopping if target achieved
        if best_C <= 0.3807:
            break
