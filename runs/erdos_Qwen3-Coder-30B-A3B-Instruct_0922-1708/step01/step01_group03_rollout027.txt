import numpy as np
from scipy import optimize
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # If initial_h_values is available, use it
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    
    # Default initialization: symmetric with Gaussian bumps
    x = np.linspace(0, 2, n_points)
    # Create symmetric function with two bumps
    h = np.zeros(n_points)
    for i in range(n_points):
        xi = x[i]
        # Two Gaussian bumps centered at 0.5 and 1.5, truncated to [0,1]
        h[i] = max(0, min(1, np.exp(-5*(xi - 0.5)**2) + np.exp(-5*(xi - 1.5)**2)))
    
    h = project_h(h, n_points)
    return h

def run_gradient_descent(h, n_points, time_budget_s, learning_rate=0.1):
    """Run gradient-based optimization with symmetry and projection."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    iter_no_improvement = 0
    max_no_improvement = 20000
    min_learning_rate = 1e-6
    max_learning_rate = 1.0
    
    # Create masks for symmetric updates
    midpoint = n_points // 2
    # Only update first half, then mirror to second half
    update_indices = list(range(midpoint + 1))
    
    while time.time() - start_time < time_budget_s:
        # Find the shift k that gives max overlap
        conv = np.correlate(current_h, 1 - current_h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute gradient exact for convolution
        # Derivative of ∫ h(x)(1 - h(x+k)) dx w.r.t. h_i:
        # If i+k < n_points: 1 - h[i+k]
        # else: 1 - h[i+k-n_points]
        dx = 2.0 / n_points
        gradient = np.zeros(n_points)
        for i in range(n_points):
            k = (shift + i) % n_points
            if shift + i < n_points:
                gradient[i] = (1 - current_h[k]) * dx
            else:
                # in case of wraparound, we should use the shifted view properly 
                # The mode="full" corresponds to full convolution
                k_offset = shift + i - (n_points - 1)
                if 0 <= k_offset < n_points:
                    gradient[i] = (1 - current_h[k_offset]) * dx
        
        # Update with gradient ascent (but we want to minimize)
        # we take negative gradient for descent
        new_h = current_h - learning_rate * (-gradient)
        
        # Project new h to valid space
        new_h = project_h(new_h, n_points)
        
        # Apply symmetry: h[2k - i] = h[i] for i < k
        for i in range(midpoint):
            new_h[2*midpoint - i] = new_h[i]
            
        new_c5 = evaluate_c5(new_h)
        
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            iter_no_improvement = 0
            # Adjust learning rate
            if learning_rate > min_learning_rate:
                learning_rate *= 0.95
        else:
            iter_no_improvement += 1
            learning_rate *= 1.1
            if learning_rate > max_learning_rate:
                learning_rate = max_learning_rate
                
        # Track best result
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        if iter_no_improvement > max_no_improvement:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main execution function."""
    np.random.seed(seed)
    n_points = 45  # Odd number to support symmetric structure
    
    # Initialize
    h = initialize_h(n_points)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    
    # Run optimization
    h_opt, c5_opt = run_gradient_descent(h, n_points, budget_s)
    
    # Final check
    h_opt = project_h(h_opt, n_points)
    c5_opt = evaluate_c5(h_opt)
    
    return (h_opt, c5_opt, n_points)
