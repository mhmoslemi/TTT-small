import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric and smooth
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def line_search(h, i, shift_idx, dt=0.01):
    """Perform line search for coordinate i to minimize C5 along a direction."""
    orig_h = h[i]
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    for sign in [-1, 1]:
        for t in [dt, 2*dt, 5*dt]:
            h[i] = orig_h + sign * t
            h = project_h(h, len(h))
            new_c5 = evaluate_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    h[i] = best_h[i]
    return h, best_c5

def coordinate_wise_optimization(h, n_points, time_budget_s):
    """Optimize using coordinate-wise updates."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    iter_count = 0
    while time.time() - start_time < time_budget_s * 0.6:
        improved = False
        for i in range(n_points):
            iter_count += 1
            if iter_count % 1000 == 0:
                pass  # Progress update

            h_new = best_h.copy()
            # Approximate optimal value in direction of coordinate i
            # Use a simple line-search with a few steps
            h_new, new_c5 = line_search(h_new, i, 0)
            if new_c5 < best_c5:
                best_h = h_new
                best_c5 = new_c5
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def nonlinear_conjugate_gradient(h, n_points, time_budget_s):
    """Run nonlinear conjugate gradient (Polak-Ribière) optimization."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    grad_prev = None
    dir_prev = None
    iter_count = 0
    max_iter = 20000

    while time.time() - start_time < time_budget_s * 0.4 and iter_count < max_iter:
        iter_count += 1
        if iter_count % 1000 == 0:
            pass  # Progress update

        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        base_c5 = evaluate_c5(best_h)

        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps

        # Polak-Ribière CG direction
        if iter_count == 1:
            dir = -grad
        else:
            beta = np.dot(grad, grad - grad_prev) / np.dot(grad_prev, grad_prev)
            beta = max(0, beta)
            dir = -grad + beta * dir_prev

        # Line search in direction dir
        alpha_best = 0.0
        best_c5_search = best_c5
        best_h_search = best_h.copy()
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.2]:
            h_trial = best_h + alpha * dir
            h_trial = project_h(h_trial, n_points)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5_search:
                best_c5_search = c5_trial
                best_h_search = h_trial.copy()
                alpha_best = alpha

        # Update
        if best_c5_search < best_c5:
            best_h = best_h_search
            best_c5 = best_c5_search
            grad_prev = grad
            dir_prev = dir
        else:
            break

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    n_points = 45  # Using a smaller number for faster runtime
    
    # Initialize
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: Coordinate-wise optimization
    h1, c5_1 = coordinate_wise_optimization(h, n_points, budget_s)
    
    # Phase 2: Nonlinear CG
    h2, c5_2 = nonlinear_conjugate_gradient(h1, n_points, budget_s)
    
    # Return best result
    final_h = h2 if c5_2 < c5_1 else h1
    final_c5 = min(c5_2, c5_1)
    
    # Final projection to ensure feasibility
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
