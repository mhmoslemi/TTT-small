import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using segmented search + gradient refinement"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_h(h_init, n_points)
    else:
        # Default initialization
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h_init = project_h(h_init, n_points)
    
    # Set up segmented search - use 5 segments
    n_segments = 5
    segment_values = [0.0, 0.5, 1.0]
    segment_combinations = list(product(segment_values, repeat=n_segments))
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    num_combinations = len(segment_combinations)
    
    # Early termination flag
    elapsed = time.time() - start_time
    if elapsed > 0.8 * budget_s:
        return (best_h, best_c5, n_points)
    
    # Exhaustive search over segment combinations
    segment_size = n_points // n_segments
    remainder = n_points % n_segments
    
    for combo in segment_combinations:
        # Build piecewise constant h
        h = np.zeros(n_points)
        for i, seg_val in enumerate(combo):
            start_idx = i * segment_size + min(i, remainder)
            end_idx = start_idx + segment_size + (1 if i < remainder else 0)
            h[start_idx:end_idx] = seg_val
        
        # Project to satisfy constraints
        h = project_h(h, n_points)
        c5 = evaluate_h(h, dx)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Early termination check
        elapsed = time.time() - start_time
        if elapsed > 0.8 * budget_s:
            break
    
    # Gradient descent refinement of top candidates
    # Use top 10 candidates from our enumeration
    # For efficiency, we do gradient descent on a smaller set
    refine_candidates = [best_h]
    
    max_refinements = min(10, num_combinations)
    candidate_scores = []
    
    # Rebuild a subset for gradient refinement more efficiently
    for _ in range(min(10, num_combinations)):
        h_sub = h_init.copy()
        # Randomly modify from best to get varied candidates
        for i in range(n_points):
            if np.random.rand() < 0.2:  # 20% chance to modify
                h_sub[i] = np.random.rand()
        h_sub = project_h(h_sub, n_points)
        c5 = evaluate_h(h_sub, dx)
        refine_candidates.append(h_sub)
        candidate_scores.append(c5)
    
    # Sort by performance and refine top ones
    sorted_indices = np.argsort(candidate_scores)
    top_candidates = [refine_candidates[i] for i in sorted_indices[:5]]
    
    # Refinement by gradient descent
    max_refinement_time = 0.2 * budget_s
    refinement_start = time.time()
    
    for h_refine in top_candidates:
        h = h_refine.copy()
        current_c5 = evaluate_h(h, dx)
        lr = 0.01
        patience = 0
        max_patience = 1000
        
        while time.time() - refinement_start < max_refinement_time and patience < max_patience:
            # Compute approximate gradient via finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update
            h -= lr * grad
            h = project_h(h, n_points)
            
            new_c5 = evaluate_h(h, dx)
            if new_c5 < current_c5:
                current_c5 = new_c5
                patience = 0
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                patience += 1
                if patience % 100 == 0:
                    lr *= 0.5  # Adaptive learning rate decay
    
    return (best_h, best_c5, n_points)
