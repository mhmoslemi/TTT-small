import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create custom
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a smooth piecewise constant function with segments
        h = np.zeros(n_points)
        segments = 5
        seg_size = n_points // segments
        for i in range(segments):
            start = i * seg_size
            end = (i + 1) * seg_size if i < segments - 1 else n_points
            h[start:end] = 1 if i % 2 == 0 else 0
        # Smooth it a bit
        h = np.clip(h + np.random.normal(0, 0.1, n_points), 0, 1)
    
    # Normalize to sum constraint
    h = h * (target_sum / np.sum(h))
    
    # Simulated Annealing parameters
    temp = 1.0
    alpha = 0.99
    min_temp = 1e-6
    max_iter = int(50000)
    
    # Gradient descent parameters
    lr = 0.01
    gd_steps = 200
    
    def compute_C(h_vals):
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    def perturb_h(h_vals, temp_scale):
        h_new = h_vals.copy()
        # Select random segment to perturb
        segment = np.random.randint(0, n_points)
        # Adjust value
        delta = np.random.normal(0, 0.01 * temp_scale)
        h_new[segment] = np.clip(h_new[segment] + delta, 0, 1)
        # Renormalize to keep sum constraint
        h_new = h_new * (target_sum / np.sum(h_new))
        return h_new
    
    best_h = h.copy()
    best_C = compute_C(h)
    
    # Annealing loop
    iter_count = 0
    while time() - start_time < budget_s * 0.6 and iter_count < max_iter:
        iter_count += 1
        h_new = perturb_h(h, temp)
        C_new = compute_C(h_new)
        C_old = compute_C(h)
        
        if C_new < C_old or np.random.rand() < np.exp(-(C_new - C_old) / temp):
            h = h_new
            if C_new < best_C:
                best_C = C_new
                best_h = h.copy()
        
        temp = max(temp * alpha, min_temp)
    
    # Gradient descent refinement
    def finite_diff_grad(h_vals, eps=1e-4):
        grad = np.zeros_like(h_vals)
        C_base = compute_C(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - C_base) / eps
        return grad
    
    h = best_h.copy()
    no_improvement_count = 0
    
    for _ in range(gd_steps):
        if time() - start_time > budget_s * 0.9:
            break
        grad = finite_diff_grad(h)
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        C_new = compute_C(h)
        if C_new < best_C:
            best_C = C_new
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 20:
                lr /= 2.0
                no_improvement_count = 0
    
    return (best_h.tolist(), best_C, n_points)
