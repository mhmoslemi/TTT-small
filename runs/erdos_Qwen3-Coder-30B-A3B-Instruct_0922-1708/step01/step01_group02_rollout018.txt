import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Use differential evolution for better global search
    bounds = [(0, 1) for _ in range(n_points)]
    
    def obj_func(h_vec):
        h_vec = project_h(h_vec)
        return compute_c5(h_vec)
    
    # Set up DE optimization with bounds
    result = differential_evolution(
        obj_func,
        bounds,
        maxiter=1000,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    final_h = project_h(result.x)
    final_c5 = compute_c5(final_h)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    # If we have time, refine with local optimization
    if (time.time() - start_time) < budget_s - 5:
        # Local refinement with bounds
        try:
            from scipy.optimize import minimize
            
            # Define optimization with bounds
            def local_obj(h_vec):
                return compute_c5(h_vec)
            
            # Apply bounds and constraints in scipy
            constraints = [
                {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                {'type': 'ineq', 'fun': lambda x: x.min()},
                {'type': 'ineq', 'fun': lambda x: 1 - x.max()}
            ]
            
            bounds_opt = [(0, 1) for _ in range(n_points)]
            
            res = minimize(
                local_obj,
                best_h,
                method='SLSQP',
                bounds=bounds_opt,
                constraints=constraints,
                options={'disp': False, 'maxiter': 1000},
                callback=lambda x: None
            )
            
            refined_h = project_h(res.x)
            refined_c5 = compute_c5(refined_h)
            
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
                
        except:
            pass  # continue with current best if error
    
    return (best_h.tolist(), best_c5, len(best_h))
