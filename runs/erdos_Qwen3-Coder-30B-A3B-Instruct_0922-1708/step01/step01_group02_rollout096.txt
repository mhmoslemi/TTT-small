import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    # PSO parameters
    n_particles = 50
    max_iter = 10000
    inertia_start = 0.9
    inertia_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(n_particles):
        part = project_h(np.random.rand(n_points) * 0.5 + 0.25)
        particles.append(part)
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
        pbest_positions.append(part.copy())
        pbest_values.append(compute_c5(part))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO main loop
    for iter_num in range(max_iter):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Adaptive inertia
        inertia = inertia_start - (inertia_start - inertia_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = project_h(particles[i])
            
            # Evaluate
            current_value = compute_c5(particles[i])
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_value < gbest_value:
                    gbest_value = current_value
                    gbest = particles[i].copy()
    
    # Final evaluation
    final_h = project_h(gbest)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
