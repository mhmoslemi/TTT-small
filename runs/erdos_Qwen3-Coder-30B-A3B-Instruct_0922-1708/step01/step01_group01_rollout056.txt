import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO + Gradient Descent"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    swarm_size = n_particles
    stagnation_count = 0
    max_stagnation = 500
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # 10% random, 90% from initial solution
    for _ in range(swarm_size):
        if np.random.rand() < 0.1:
            particle = np.random.rand(n_points)
        else:
            particle = h + np.random.normal(0, 0.05, n_points)
        particle = project_h(particle, n_points)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_values.append(evaluate_h(particle, dx))
    
    # Global best
    best_idx = np.argmin(pbest_values)
    gbest = pbest[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    start_time = time.time()
    current_iter = 0
    
    while current_iter < max_iter and time.time() - start_time < 0.8 * budget_s:
        current_iter += 1
        stagnation_count += 1
        
        # Update particles
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] = project_h(swarm[i] + velocities[i], n_points)
            
            # Evaluate
            current_value = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = swarm[i].copy()
                pbest_values[i] = current_value
                if current_value < gbest_value:
                    gbest = swarm[i].copy()
                    gbest_value = current_value
                    stagnation_count = 0
    
        # Adapt swarm size if stuck
        if stagnation_count > max_stagnation:
            if swarm_size > 20:
                swarm_size = max(20, int(swarm_size * 0.8))
                # Reinitialize smaller swarm
                swarm = []
                velocities = []
                pbest = []
                pbest_values = []
                for _ in range(swarm_size):
                    if np.random.rand() < 0.1:
                        particle = np.random.rand(n_points)
                    else:
                        particle = gbest + np.random.normal(0, 0.05, n_points)
                    particle = project_h(particle, n_points)
                    swarm.append(particle)
                    velocities.append(np.zeros(n_points))
                    pbest.append(particle.copy())
                    pbest_values.append(evaluate_h(particle, dx))
                best_idx = np.argmin(pbest_values)
                gbest = pbest[best_idx].copy()
                gbest_value = pbest_values[best_idx]
                stagnation_count = 0
    
    # Gradient descent on best solution
    h = gbest.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    while time.time() - start_time < 0.9 * budget_s:
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (best_h, best_c5, n_points)
