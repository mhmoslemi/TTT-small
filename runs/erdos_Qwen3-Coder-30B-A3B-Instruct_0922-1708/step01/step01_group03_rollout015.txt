```python
import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Use initial construction
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use symmetric function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  
    h = project_h(h, n_points)
    return h

def compute_gradient(h, k, n_points):
    """Compute gradient for shift k."""
    dx = 2.0 / n_points
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + k) % n_points
        grad[i] = -dx * (1 - h[j])
    return grad

def line_search(h, grad, learning_rate, n_points):
    """Perform line search to find the best step size."""
    dx = 2.0 / n_points
    best_c5 = evaluate_c5(h)
    best_alpha = 0.0
    eps = 1e-8
    
    # Try a few alpha values
    alphas = np.linspace(0, learning_rate, 100)
    for alpha in alphas:
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5 - eps:
            best_c5 = c5
            best_alpha = alpha
    return best_alpha

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45  
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Initialize learning rate
    learning_rate = 0.1
    min_learning_rate = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Find the shift that maximizes overlap
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        k = max_idx - (n_points - 1)  # convert to shift
        
        # Compute gradient for this shift
        grad = compute_gradient(h, k, n_points)
        
        # Line search for optimal step size
        alpha = line_search(h, grad, learning_rate, n_points)
        
        # Update h
        h = h - alpha * grad
        h = project_h(h, n_points)
        
        # Evaluate new c5
        new_c5 = evaluate_c5(h)
        
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
        # Adaptive learning rate
        if iteration % 10 == 0:
            if abs(new_c5 - best_c5) < 1e-6:
                learning_rate *= 0.5
                if learning_rate < min_learning_rate:
                    break
    
    return (best_h, best_c5, n_points)
```
