import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    start_time = time.time()
    
    # Use a stochastic optimization approach with simulated annealing inspired method
    temp = 1.0
    min_temp = 1e-6
    decay = 0.9995
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Generate a neighbor solution
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate neighbor
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept with probability based on temperature and difference
        if c5_new < c5 or np.random.rand() < np.exp((c5 - c5_new) / temp):
            h = h_new
            
        # Cool down
        temp = max(min_temp, temp * decay)
        
    if best_h is None:
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
