    # Constants
    n_points = 70

    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

    else:
        h = np.full(n_points, 0.5)

    h = h * (target_sum / np.sum(h))
    
    # Objective function to minimize C_5
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # PSO parameters
    n_particles = 100
    n_dims = n_points
    max_iter = 5000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            # Start near initial_h_values but with some noise
            particle = h + np.random.normal(0, 0.05, n_points)
        else:
            particle = np.random.rand(n_points)
        particle = np.clip(particle, 0, 1)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.1, 0.1, n_dims))
    
    # Track global best
    best_particle = h.copy()
    best_C = compute_C(best_particle)
    
    # PSO iteration
    for iteration in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
            
        for i, particle in enumerate(particles):
            # Ensure particle stays within bounds
            particle = np.clip(particle, 0, 1)
            
            # Rescale to satisfy integral constraint
            particle = particle * (target_sum / np.sum(particle))
            
            # Evaluate current particle
            current_C = compute_C(particle)
            
            # Update personal best if better
            if current_C < best_C:
                best_C = current_C
                best_particle = particle.copy()
            
            # Update velocities and positions
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (particle - particles[i]) +
                            social * r2 * (best_particle - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
    
    # Refinement with gradient descent on top PSO solutions
    # Take top 10% of particles for refinement
    particle_scores = [compute_C(p) for p in particles]
    sorted_indices = np.argsort(particle_scores)
    top_indices = sorted_indices[:max(1, n_particles // 10)]
    
    best_h = best_particle.copy()
    best_C_refined = best_C
    
    for idx in top_indices:
        if time() - start_time > budget_s - 1:
            break
            
        h_refined = particles[idx].copy()
        h_refined = h_refined * (target_sum / np.sum(h_refined))
        
        lr = 0.01
        stagnation_count = 0
        prev_C = compute_C(h_refined)
        
        for _ in range(200):  # Max 200 gradient steps
            if time() - start_time > budget_s - 1:
                break
                
            # Compute finite difference gradient
            grad = np.zeros(n_points)
            eps = 1e-5
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (target_sum / np.sum(h_plus))
                C_plus = compute_C(h_plus)
                grad[j] = (C_plus - prev_C) / eps
            
            # Update with gradient and learning rate
            h_refined -= lr * grad
            
            # Clip values
            h_refined = np.clip(h_refined, 0, 1)
            
            # Rescale to satisfy integral constraint
            h_refined = h_refined * (target_sum / np.sum(h_refined))
            
            # Update learning rate if no improvement
            C_new = compute_C(h_refined)
            if C_new >= prev_C:
                stagnation_count += 1
                if stagnation_count > 10:
                    lr *= 0.5
                    stagnation_count = 0
            else:
                stagnation_count = 0
            prev_C = C_new
            
            # Update best solution
            if prev_C < best_C_refined:
                best_C_refined = prev_C
                best_h = h_refined.copy()
    
    return (best_h.tolist(), best_C_refined, n_points)
