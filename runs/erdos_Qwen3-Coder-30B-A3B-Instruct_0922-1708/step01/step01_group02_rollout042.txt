import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:

        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Use simulated annealing for better exploration
    def anneal_step(h_vec, temp):
        h_new = h_vec.copy()
        # Perturb one random element
        i = np.random.randint(0, len(h_vec))
        delta = np.random.normal(0, temp * 0.1)
        h_new[i] += delta
        return project_h(h_new)
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    temp = 0.1
    alpha = 0.99
    min_temp = 1e-6
    
    iterations = 0
    max_iterations = 10000
    
    while iterations < max_iterations and (time.time() - start_time) < budget_s - 1:
        # Try a neighbor
        new_h = anneal_step(current_h, temp)
        new_c5 = compute_c5(new_h)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        temp *= alpha
        
        iterations += 1
        
        if (time.time() - start_time) > budget_s - 1:
            break
    
    # Final refinement using gradient-based method on best result
    def objective(h_vec):
        return compute_c5(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Use scipy optimization on the best found solution
    try:
        result = minimize(objective, best_h, method='SLSQP', 
                         constraints={'type': 'eq', 'fun': constraint_sum},
                         bounds=[(0, 1) for _ in range(n_points)],
                         options={'maxiter': 100, 'ftol': 1e-8})
        if result.success:
            refined_h = project_h(result.x)
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
