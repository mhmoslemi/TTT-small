import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    # Normalize to integral = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Compute dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=5):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def perturb(h, delta=0.05):
        # Create a perturbed version
        h_new = h.copy()
        # Perturb about 10% of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    def obj_and_grad(h):
        # Compute the objective value and its gradient
        # We'll use a smooth approximation of the max operation
        # But still compute the actual max for evaluation
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        
        # For gradient: we need to compute the gradient of the maximal element
        # This is an approximation, since max is not differentiable
        # We compute gradient of the overlap with respect to h
        # We approximate it by choosing one element of the correlation that is maximal
        
        # Fortunately for optimization, we're doing local search, so don't need exact gradient
        # Return approximate gradient by finite differences 
        # (Not very accurate but good enough for local search)
        return max_overlap, np.zeros_like(h)  # placeholder gradient
    
    best_h = h_start.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    perturb_delta = 0.05
    
    # Gradient-based optimizer callback (approximate)
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iterations += 1
        
        # Try several perturbations
        for _ in range(50):
            h_trial = perturb(best_h, delta=perturb_delta)
            h_trial = smooth(h_trial, window=5)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                break
        else:
            no_improvement_count += 1
            
        # Occasionally decrease perturbation strength
        if iterations % 20 == 0 and perturb_delta > 0.001:
            perturb_delta *= 0.95
    
    return (best_h, best_c5, n_points)
