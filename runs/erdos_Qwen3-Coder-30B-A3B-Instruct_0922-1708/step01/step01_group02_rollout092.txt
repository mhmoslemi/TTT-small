import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize, Bounds
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Simple gradient-free optimization using Nelder-Mead
    def objective(h_vec):
        return compute_c5(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds
    bnds = Bounds([0]*n_points, [1]*n_points)
    
    # Optimization with restarts
    max_restarts = 10
    restart_count = 0
    
    while (time.time() - start_time) < budget_s - 1 and restart_count < max_restarts:
        # Random restart
        if restart_count > 0:
            # Generate a new random feasible vector
            h_new = np.random.rand(n_points)
            h_new = project_h(h_new)
        else:
            h_new = h.copy()
            
        # Minimize using Nelder-Mead (gradient free)
        res = minimize(objective, h_new, method='Nelder-Mead', 
                       options={'maxiter': 1000, 'disp': False})
        
        if res.success:
            h_opt = project_h(res.x)
            c5_opt = compute_c5(h_opt)
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

        restart_count += 1
        # Early termination
        if (time.time() - start_time) > budget_s - 1:
            break

    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
