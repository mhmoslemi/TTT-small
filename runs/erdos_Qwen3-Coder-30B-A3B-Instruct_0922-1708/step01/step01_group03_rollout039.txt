import numpy as np
from scipy.optimize import minimize_scalar
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Access initial_h_values via globals
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric function with triangular pulses
    x = np.linspace(0, 2, n_points)
    h = np.maximum(0, 1 - np.abs(x - 0.5)) + np.maximum(0, 1 - np.abs(x - 1.5))
    h = project_h(h, n_points)
    return h

def line_search(h, i, delta, current_c5, n_points):
    """1D line search to find optimal h[i] value."""
    def c5_func(alpha):
        h_new = h.copy()
        h_new[i] += alpha * delta
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Find suitable alpha range
    alpha_min, alpha_max = -h[i], 1 - h[i]
    try:
        # Use scalar minimization to find optimal step
        result = minimize_scalar(c5_func, bounds=(alpha_min, alpha_max), method='bounded')
        return result.x
    except:
        return 0.0

def coordinate_wise_optimization(h, n_points, max_iter=5000, tolerance=1e-6):
    """Coordinate-wise optimization with line search."""
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    old_c5 = best_c5
    
    for iteration in range(max_iter):
        improved = False
        # Shuffle indices to avoid bias
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for i in indices:
            # Try both positive and negative changes
            for delta in [-0.05, 0.05]:
                new_alpha = line_search(best_h, i, delta, best_c5, n_points)
                if abs(new_alpha) > 1e-10:
                    h_new = best_h.copy()
                    h_new[i] += new_alpha * delta
                    h_new = project_h(h_new, n_points)
                    new_c5 = evaluate_c5(h_new)
                    if new_c5 < best_c5:
                        best_h = h_new
                        best_c5 = new_c5
                        improved = True
                        break
            if improved:
                break
        
        if not improved:
            if abs(best_c5 - old_c5) < tolerance:
                break
            old_c5 = best_c5

    return best_h, best_c5

def nonlinear_cg_optimization(h, n_points, max_iter=1000, tolerance=1e-6):
    """Nonlinear conjugate gradient optimization."""
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)

    # Compute initial gradient approximation
    def compute_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad

    # CG update
    g = compute_gradient(best_h)
    d = -g
    old_grad = g.copy()
    old_c5 = best_c5

    for iteration in range(max_iter):
        # Directional line search
        def directional_c5(alpha):
            h_new = best_h + alpha * d
            h_new = project_h(h_new, n_points)
            return evaluate_c5(h_new)

        # Minimize in direction d
        result = minimize_scalar(directional_c5)
        alpha = result.x

        # Update
        old_h = best_h.copy()
        best_h += alpha * d
        best_h = project_h(best_h, n_points)
        new_c5 = evaluate_c5(best_h)
        
        # Polak-Ribière
        if iteration > 0:
            g = compute_gradient(best_h)
            beta = np.dot(g - old_grad, g) / np.dot(old_grad, old_grad)
            beta = max(0, beta)
            d = -g + beta * d
            old_grad = g.copy()

        if abs(new_c5 - old_c5) < tolerance:
            break
        old_c5 = new_c5

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n_points=45 to keep it fast
    n_points = 45
    h = initialize_h(n_points)
    
    # Phase 1: Coordinate-wise optimization
    if time.time() - start_time < budget_s * 0.7:
        h, c5 = coordinate_wise_optimization(h, n_points, max_iter=2000)
    else:
        c5 = evaluate_c5(h)
        
    # Phase 2: Nonlinear CG
    if time.time() - start_time < budget_s * 0.9:
        h, c5 = nonlinear_cg_optimization(h, n_points, max_iter=1000)
    else:
        c5 = evaluate_c5(h)
    
    # Final check
    h = project_h(h, n_points)
    c5 = evaluate_c5(h)
    
    return (h, c5, n_points)
