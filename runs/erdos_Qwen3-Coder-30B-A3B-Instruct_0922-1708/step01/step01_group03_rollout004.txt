import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def project_symmetric(h, n_points):
    """Project to satisfy symmetry and constraints."""
    h = project_h(h, n_points)
    # Enforce symmetry: h[i] = (h[i] + h[n-1-i]) / 2
    k = n_points // 2
    for i in range(k):
        h[i] = h[n_points - 1 - i] = (h[i] + h[n_points - 1 - i]) / 2
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values).copy()
            return project_symmetric(h, n_points)
    except:
        pass
    # Default initialization: symmetric function
    x = np.linspace(0, 2, n_points)
    # Create two symmetric bumps for initial construction
    h = 0.5 * (1 + np.sin(np.pi * (x - 0.5))) * (x < 1) + \
        0.5 * (1 + np.sin(np.pi * (x - 1.5))) * (x >= 1)
    h = project_symmetric(h, n_points)
    return h

def compute_gradient(h, shift):
    """Compute gradient at given shift."""
    n = len(h)
    G = np.zeros(n)
    dx = 2.0 / n
    for i in range(n):
        j = (i + shift) % n
        G[i] = (1 - h[j]) * dx
    return G

def find_max_shift(h):
    """Find the shift that gives maximum overlap."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(conv)
    shift = max_idx - (n - 1)
    return shift

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    time_start = time.time()
    
    n_points = 45  # odd for symmetry
    h = initialize_h(n_points)
    h = project_symmetric(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Optimization parameters
    eta = 0.1
    max_no_improvement = 1000
    iter_no_improvement = 0
    iteration = 0
    
    while time.time() - time_start < budget_s * 0.95:
        iteration += 1
        # Find shift that causes maximum overlap
        shift = find_max_shift(best_h)
        
        # Compute gradient
        gradient = compute_gradient(best_h, shift)
        
        # Update h
        h_new = best_h - eta * gradient
        h_new = project_symmetric(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            # Reduce learning rate
            eta *= 0.99
        else:
            iter_no_improvement += 1
            # Increase learning rate
            eta *= 1.01
            if eta > 1:
                eta = 1.0
        
        if iter_no_improvement > max_no_improvement or eta < 1e-6:
            break
    
    # Final check
    final_h = project_symmetric(best_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
