import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.999
    T = T0
    max_iterations = 10000
    
    # SA loop
    for iteration in range(max_iterations):
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Store current state
        h_current = h.copy()
        c5_current = compute_c5(h_current)
        
        # Perturb h with Gaussian noise scaled by temperature
        noise_scale = T * 0.1
        perturbation = np.random.normal(0, noise_scale, n_points)
        h_proposed = h_current + perturbation
        
        # Project to feasible set
        h_proposed = project_h(h_proposed)
        
        # Compute C5 for proposed solution
        c5_proposed = compute_c5(h_proposed)
        
        # Accept or reject based on Metropolis criterion
        delta_c5 = c5_proposed - c5_current
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_proposed
            c5 = c5_proposed
        else:
            c5 = c5_current
        
        # Update best solution
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Cool down temperature
        T = alpha * T
        
    # Final check and return
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
