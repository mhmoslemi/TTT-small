from scipy.optimize import minimize
from scipy.signal import correlate
import time

    h_start = np.clip(h_start, 0, 1)

    h_start = h_start * (n_points / 2) / np.sum(h_start)

    dx = 2.0 / n_points
    
    def evaluate(h):
        # Compute correlation with 1-h
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def smooth(h, window=5):
        # Simple moving average smoothing
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def perturb(h, delta=0.05):
        # Create a perturbed version
        h_new = h.copy()
        # Perturb about 10% of points
        n_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    def gradient_descent_step(h, lr=1e-3, max_iter=100):
        # Simple gradient descent to minimize C5
        h_new = h.copy()
        prev_c5 = evaluate(h_new)
        
        for _ in range(max_iter):
            # Estimate gradient by finite differences
            grad = np.zeros_like(h)
            eps = 1e-6
            for i in range(len(h)):
                h_plus = h_new.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)

                grad[i] = (c5_plus - prev_c5) / eps
                
            # Update step with clipping and normalization
            h_new = h_new - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)

        h_opt = gradient_descent_step(best_h, lr=1e-2)

        # Reduce perturbation strength if no improvement for a while
        if no_improvement_count > 50:
            perturb_delta *= 0.95
            
    return (best_h, best_c5, n_points)
