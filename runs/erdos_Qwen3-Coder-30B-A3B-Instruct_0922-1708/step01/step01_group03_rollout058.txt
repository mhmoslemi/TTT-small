import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangular pulse
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    # Create symmetric triangular pulse
    center1, center2 = 0.5, 1.5
    width = 0.5
    for i in range(n_points):
        xi = x[i]
        h[i] = max(0, 1 - abs(xi - center1) / width)
        if abs(xi - center2) < width:
            h[i] = max(h[i], 1 - abs(xi - center2) / width)
    h = project_h(h, n_points)
    return h

def run_pso_phase(h, n_points, time_budget_s):
    """Run Particle Swarm Optimization to find a good starting point."""
    start_time = time.time()
    swarm_size = 30
    cognitive_weight = 2.0
    social_weight = 2.0
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Initialize swarm with symmetric particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    
    for _ in range(swarm_size):
        particle = initialize_h(n_points)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
        personal_best.append(particle.copy())
        personal_best_scores.append(evaluate_c5(particle))
    
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.7:
        iteration += 1
        if iteration % 50 == 0:
            pass
        
        for i in range(swarm_size):
            particle = particles[i]
            vel = velocities[i]
            
            # Update velocity
            r1, r2 = random.random(), random.random()
            cognitive = cognitive_weight * r1 * (personal_best[i] - particle)
            social = social_weight * r2 * (global_best - particle)
            vel = inertia_weight * vel + cognitive + social
            
            # Clip velocity
            vel = np.clip(vel, -velocity_max, velocity_max)
            velocities[i] = vel
            
            # Update particle
            particle = particle + vel
            particle = project_h(particle, n_points)
            particles[i] = particle
            
            # Evaluate new score
            score = evaluate_c5(particle)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particle.copy()
                personal_best_scores[i] = score
            
            # Update global best
            if score < global_best_score:
                global_best = particle.copy()
                global_best_score = score
        
        # Decrease inertia weight over time
        inertia_weight = 0.9 - 0.5 * (time.time() - start_time) / (time_budget_s * 0.7)
        inertia_weight = max(0.4, inertia_weight)
    
    return global_best, global_best_score

def run_gradient_descent_phase(h, n_points, time_budget_s):
    """Run gradient descent with projected updates."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.05
    max_iter_no_improvement = 1000
    iter_no_improvement = 0
    
    # Find maximum shift K
    def find_max_shift(h):
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)
        return shift
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.3:
        iteration += 1
        
        # Compute gradient estimate
        shift = find_max_shift(best_h)
        n = len(best_h)
        dx = 2.0 / n
        
        # Gradient computation
        grad = np.zeros(n)
        eps = 1e-6
        for i in range(n):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update using gradient descent
        h_new = best_h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Update if improvement
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            
        if iter_no_improvement > max_iter_no_improvement:
            break
        
        # Adjust learning rate
        if iteration % 100 == 0:
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 45
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: PSO
    h_pso, c5_pso = run_pso_phase(h, n_points, budget_s)
    
    # Phase 2: Gradient Descent
    h_gd, c5_gd = run_gradient_descent_phase(h_pso, n_points, budget_s)
    
    # Return best result
    final_h = h_gd if c5_gd < c5_pso else h_pso
    final_c5 = min(c5_gd, c5_pso)
    
    # Ensure final result is valid and re-evaluate
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
