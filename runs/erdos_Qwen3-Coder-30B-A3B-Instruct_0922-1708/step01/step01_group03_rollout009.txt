import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def pso_phase(h, n_points, time_budget_s):
    """Run Particle Swarm Optimization to find good starting point."""
    start_time = time.time()
    
    # PSO parameters
    swarm_size = 30
    max_iter = 10000
    cognitive_weight = 2
    social_weight = 2
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            # Mutate previous particle slightly
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_h(particle, n_points)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        score = evaluate_c5(particle)
        personal_best.append(particle.copy())
        personal_best_scores.append(score)
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.8 and iteration < max_iter:
        iteration += 1
        # Update inertia weight
        current_inertia = inertia_weight * (1 - iteration / max_iter)
        
        for i in range(swarm_size):
            particle = swarm[i]
            velocity = velocities[i]
            
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = cognitive_weight * r1 * (personal_best[i] - particle)
            social = social_weight * r2 * (global_best - particle)
            velocity = current_inertia * velocity + cognitive + social
            
            # Clip velocity
            velocity = np.clip(velocity, -velocity_max, velocity_max)
            velocities[i] = velocity
            
            # Update position
            particle = particle + velocity
            # Ensure bounds
            particle = np.clip(particle, 0, 1)
            # Rescale to correct sum
            current_sum = np.sum(particle)
            if current_sum != 0:
                particle = particle * (n_points / 2) / current_sum
            
            swarm[i] = particle
            
            # Evaluate
            score = evaluate_c5(particle)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particle.copy()
                personal_best_scores[i] = score
                
                # Update global best
                if score < global_best_score:
                    global_best = particle.copy()
                    global_best_score = score
    
    return global_best, global_best_score

def gradient_descent_phase(h, n_points, time_budget_s):
    """Run gradient descent to refine the solution."""
    start_time = time.time()
    
    def compute_gradient(h, dx=1e-5):
        """Compute gradient using finite differences."""
        current_c5 = evaluate_c5(h)
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += dx
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= dx
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * dx)
        
        return grad
    
    def find_max_overlap_shift(h):
        """Find the shift with max overlap to guide gradient."""
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        return max_idx - (n_points - 1)
    
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    learning_rate = 0.05
    iteration = 0
    no_improvement = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < time_budget_s * 0.2 and no_improvement < max_no_improvement:
        iteration += 1
        # Compute gradient
        gradient = compute_gradient(current_h)
        
        # Update step
        new_h = current_h - learning_rate * gradient
        new_h = project_h(new_h, n_points)
        new_c5 = evaluate_c5(new_h)
        
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Reduce learning rate for stability
        if iteration % 500 == 0:
            learning_rate *= 0.9
            if learning_rate < 1e-6:
                break
    
    return current_h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45
    
    # Initialize
    h = initialize_h(n_points)
    
    start_time = time.time()
    
    # Phase 1: PSO
    h_pso, c5_pso = pso_phase(h, n_points, budget_s)
    
    # Phase 2: Gradient Descent
    h_gd, c5_gd = gradient_descent_phase(h_pso, n_points, budget_s)
    
    # Return best result
    final_h = h_gd if c5_gd < c5_pso else h_pso
    final_c5 = min(c5_gd, c5_pso)
    
    # Ensure final result is valid
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
