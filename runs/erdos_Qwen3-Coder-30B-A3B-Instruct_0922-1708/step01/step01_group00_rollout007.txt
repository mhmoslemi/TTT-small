import numpy as np
from cvxpy import *
import time

    import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available (a bit larger than previously used)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h with initial_h_values if available
    if 'initial_h_values' in globals():
        h_vals = np.array(initial_h_values)
        if len(h_vals) != n_points:
            # interpolate to the target length
            x_orig = np.linspace(0, 2, len(h_vals))
            x_new = np.linspace(0, 2, n_points)
            h_vals = np.interp(x_new, x_orig, h_vals)
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
    else:
        h_vals = np.ones(n_points) * 0.5
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        
    best_h = h_vals.copy()
    best_c5 = float('inf')
    
    # Time-aware search loop
    iteration = 0
    last_improvement = start_time
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute objective directly using numpy
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5_val = np.max(correlation) * dx
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_vals.copy()
            last_improvement = time.time()
            
        # Check for early termination (no improvement in 300s)
        if time.time() - last_improvement > 300:
            break
            
        # Perform gradient-based optimization steps with fixed steps
        # This is heuristic and imperfect, but tends to work fine here
        old_h = h_vals.copy()
        eps = 1e-3
        
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            h_plus = old_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5_val) / eps
            
        # Gradient descent step
        lr = 0.03
        h_new = old_h - lr * grad
        
        # Clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_vals = h_new
        
        # Occasionally apply binary rounding for sparsity if we're making progress
        if iteration % 20 == 0 and time.time() - start_time < budget_s - 10:
            # Rounding and scaling to binary
            h_binary = (h_vals > 0.5).astype(float)
            h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
            h_vals = h_binary
            
    return (best_h.tolist(), best_c5, n_points)
