import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(conv)
        return max_val * dx
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Optimization using Nelder-Mead (more robust than SGD)
    max_iterations = 10000
    iterations = 0
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a more robust local method
    while iterations < max_iterations and (time.time() - start_time) < budget_s - 1:
        # Use a more effective random restart strategy
        if iterations % 100 == 0 and iterations > 0:
            # Random restart with small perturbation
            h = project_h(h + np.random.normal(0, 0.05, n_points))
        
        # Perform optimization step
        try:
            res = minimize(
                lambda x: compute_c5(x),
                h,
                method='Nelder-Mead',
                bounds=bounds,
                options={'maxiter': 10, 'adaptive': True}
            )
            if res.success:
                h = project_h(res.x)
        except:
            # Fall back to simple gradient-like update
            pass
        
        # Update best if improved
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        iterations += 1
        
        # Early termination condition
        if (time.time() - start_time) > budget_s - 1:
            break
    
    # Final verification and projection
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
