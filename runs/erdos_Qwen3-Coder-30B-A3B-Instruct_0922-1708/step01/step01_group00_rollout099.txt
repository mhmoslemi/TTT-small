import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    # Use a structured optimization approach
    # Define parameters: breakpoints and interval values
    # We'll optimize just 5 breakpoints for better efficiency
    n_breakpoints = 5
    breakpoints = np.linspace(0, 2, n_breakpoints + 2)[1:-1]  # Exclude 0 and 2
    interval_values = np.ones(n_breakpoints + 1) * 0.5  # Initialize with uniform values
    
    def reconstruct_h(breakpoints, interval_values):
        """Reconstruct h from breakpoints and interval values"""
        h = np.zeros(n_points)
        breakpoints_full = np.concatenate([[0], breakpoints, [2]])
        for i in range(len(interval_values)):
            start_idx = int(breakpoints_full[i] / dx)
            end_idx = int(breakpoints_full[i+1] / dx)
            h[start_idx:end_idx] = interval_values[i]
        return h
    
    def compute_c5(breakpoints, interval_values):
        """Compute C5 from breakpoints and interval values"""
        h = reconstruct_h(breakpoints, interval_values)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Gradient-based optimization
    def optimize_breakpoints_and_values():
        nonlocal h, best_h, best_c5, breakpoints, interval_values
        
        # Set up optimization variables
        params = np.concatenate([breakpoints, interval_values])
        
        def objective(params):
            # Unpack
            new_breakpoints = params[:n_breakpoints]
            new_interval_values = params[n_breakpoints:]
            
            # Ensure valid breakpoints (in order)
            new_breakpoints = np.clip(new_breakpoints, 0, 2)
            new_breakpoints = np.sort(new_breakpoints)
            
            # Ensure valid interval values (in [0,1])
            new_interval_values = np.clip(new_interval_values, 0, 1)
            
            # Compute C5
            c5 = compute_c5(new_breakpoints, new_interval_values)
            return c5
        
        # For initial parameters, just evaluate the current state
        c5 = objective(params)
        
        if c5 < best_c5:
            best_c5 = c5
            # Reconstruct best h
            best_h = reconstruct_h(breakpoints, interval_values)
        
        # Do gradient descent
        # Here, we'll use a simple coordinate descent approach
        # more sophisticated optimizers could be used with scipy.optimize.minimize
        
        learning_rate = 0.05
        for i in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Perturb each parameter
            for j in range(len(params)):
                old_param = params[j]
                eps = 1e-5
                
                # Try small positive perturbation
                params[j] = old_param + eps
                if j < n_breakpoints:
                    # Ensure valid breakpoint order
                    params[:n_breakpoints] = np.sort(params[:n_breakpoints])
                c5_plus = objective(params)
                
                # Try small negative perturbation
                params[j] = old_param - eps
                if j < n_breakpoints:
                    # Ensure valid breakpoint order
                    params[:n_breakpoints] = np.sort(params[:n_breakpoints])
                c5_minus = objective(params)
                
                # Estimate gradient
                grad = (c5_plus - c5_minus) / (2 * eps)
                
                # Update parameter with gradient descent
                params[j] = old_param - learning_rate * grad
                
                # Ensure valid bounds for breakpoints
                if j < n_breakpoints:
                    params[j] = np.clip(params[j], 0, 2)
                
                # Ensure valid bounds for interval values
                if j >= n_breakpoints:
                    params[j] = np.clip(params[j], 0, 1)
                
                # Recompute objective and check improvement
                c5_new = objective(params)
                if c5_new < c5:
                    c5 = c5_new
                    if c5 < best_c5:
                        best_c5 = c5
                        # Reconstruct best h
                        new_breakpoints = params[:n_breakpoints]
                        new_interval_values = params[n_breakpoints:]
                        best_h = reconstruct_h(new_breakpoints, new_interval_values)
                else:
                    # Revert
                    params[j] = old_param
            
            # Periodically rescale h to maintain sum constraint
            h_reconstructed = reconstruct_h(params[:n_breakpoints], params[n_breakpoints:])
            h_reconstructed = h_reconstructed * (n_points / 2) / np.sum(h_reconstructed)
            params[:n_breakpoints] = np.sort(params[:n_breakpoints])
            param_interval_values = params[n_breakpoints:]
            param_interval_values = np.clip(param_interval_values, 0, 1)
        
        # Final reconstruction
        breakpoints = np.sort(params[:n_breakpoints])
        interval_values = np.clip(params[n_breakpoints:], 0, 1)
        
    while time.time() - start_time < budget_s - 1:
        optimize_breakpoints_and_values()
        
        # Recompute integral for final check
        final_h = reconstruct_h(breakpoints, interval_values)
        correlation = np.correlate(final_h, 1 - final_h, mode='full')
        final_c5 = np.max(correlation) * dx
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
            
        # Periodic check if solution is still improving
        if time.time() - start_time - last_improvement > 300:
            # Start a new, potentially better search
            # Initialize from scratch instead of relying on previous parameters
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                h = h / np.sum(h) * (n_points / 2)
            else:
                h = np.ones(n_points) * 0.5
            # Reset optimization
            breakpoints = np.linspace(0, 2, n_breakpoints + 2)[1:-1]  # Exclude 0 and 2
            interval_values = np.ones(n_breakpoints + 1) * 0.5  # Initialize with uniform values
            last_improvement = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
