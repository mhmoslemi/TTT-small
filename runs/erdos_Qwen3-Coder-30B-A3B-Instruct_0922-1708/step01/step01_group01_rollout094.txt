import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values or sinusoidal
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Reduce search space by using piecewise constant function with few segments
    # We'll use 5 segments and values {0, 0.5, 1}
    m = 5  # number of segments
    segment_values = [0, 0.5, 1]
    total_combinations = 3**m
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    # Exhaustive search over segment combinations
    for i in range(total_combinations):
        # Convert combination index to segment values
        segment_vals = []
        temp_i = i
        for j in range(m):
            segment_vals.append(segment_values[temp_i % 3])
            temp_i //= 3
        
        # Construct h from segments
        h_candidate = np.zeros(n_points)
        segment_length = n_points // m
        remainder = n_points % m
        
        for j in range(m):
            start_idx = j * segment_length + min(j, remainder)
            end_idx = start_idx + segment_length + (1 if j < remainder else 0)
            h_candidate[start_idx:end_idx] = segment_vals[j]
        
        h_candidate = project_h(h_candidate, n_points)
        c5_candidate = evaluate_h(h_candidate, dx)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
        
        if time.time() - start_time > 0.8 * budget_s:
            break
    
    # Refine with gradient descent
    h_refined = best_h.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 1000
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 5 and patience < max_patience:
        corr = correlate(h_refined, 1 - h_refined, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_refined.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        h_refined -= learning_rate * grad
        h_refined = project_h(h_refined, n_points)
        
        new_c5 = evaluate_h(h_refined, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_refined.copy()
            patience = 0
            last_improvement = time.time()
        else:
            patience += 1
            if patience > 0 and patience % 100 == 0:
                learning_rate *= 0.9
                patience = 0
    
    return (best_h, best_c5, n_points)
