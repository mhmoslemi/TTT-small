import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function with hybrid approach"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Use a different initialization
        h = np.ones(n_points) * 0.5
    
    # Use a more structured approach
    # Represent h as piecewise constant with low number of segments
    n_segments = 5
    segment_values = [0.0, 0.5, 1.0]
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    
    # Exhaustive search over possible segment combinations (fast phase)
    if time.time() - start_time < 0.5 * budget_s:
        segment_combinations = list(product(segment_values, repeat=n_segments))
        candidate_count = len(segment_combinations)
        
        # For efficiency, sample a subset of combinations
        sampled_combinations = segment_combinations[:min(100, candidate_count)]
        
        for seg_vals in sampled_combinations:
            # Construct piecewise constant h
            h_candidate = np.zeros(n_points)
            segment_length = n_points // n_segments
            remainder = n_points % n_segments
            
            idx = 0
            for i in range(n_segments):
                length = segment_length + (1 if i < remainder else 0)
                h_candidate[idx:idx+length] = seg_vals[i % len(seg_vals)]
                idx += length
            
            # Project to satisfy constraints
            h_candidate = project_h(h_candidate, n_points)
            
            # Evaluate C5
            c5 = evaluate_h(h_candidate, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate.copy()
    
    # Gradient-based refinement on top candidates (refinement phase) 
    if time.time() - start_time < 0.9 * budget_s:
        # Refine the best solution found so far
        h_refined = best_h.copy()
        best_refined_c5 = best_c5
        learning_rate = 0.01
        patience = 0
        max_patience = 1000
        
        while time.time() - start_time < budget_s and patience < max_patience:
            # Compute current overlap
            corr = correlate(h_refined, 1 - h_refined, mode="full")
            max_shift_idx = np.argmax(corr)
            worst_shift = max_shift_idx - (n_points - 1)
            
            # Compute gradient numerically using finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_refined.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus, n_points)
                c_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update with gradient descent
            h_refined -= learning_rate * grad
            h_refined = project_h(h_refined, n_points)
            
            # Evaluate updated solution
            new_c5 = evaluate_h(h_refined, dx)
            if new_c5 < best_refined_c5:
                best_refined_c5 = new_c5
                best_h = h_refined.copy()
                patience = 0
            else:
                patience += 1
                if patience % 100 == 0:
                    learning_rate *= 0.5
            
        best_c5 = best_refined_c5
    
    return (best_h, best_c5, n_points)
