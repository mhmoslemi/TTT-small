import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Get dx
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Segment-based approach
    def segment_to_full(segments, n_points):
        m = len(segments)
        segment_width = n_points // m
        remainder = n_points % m
        h = np.zeros(n_points)
        offset = 0
        for i, s in enumerate(segments):
            width = segment_width + (1 if i < remainder else 0)
            h[offset:offset+width] = s
            offset += width
        return h
    
    def optimize_segments(segments, n_points, dx):
        def obj(c):
            h = segment_to_full(c, n_points)
            return evaluate(h)
        
        # Bounds and constraints
        bounds = [(0, 1) for _ in range(len(segments))]
        
        def constraint(c):
            return np.sum(c) - len(c)/2  # Sum should equal m/2
            
        cons = {'type': 'eq', 'fun': constraint}
        
        # Optimize
        res = minimize(obj, segments, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'ftol': 1e-8, 'maxiter': 500})
        return res.x
    
    def smooth_h(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    # Start with few segments
    n_segments = 7
    segments = np.array([0.5] * n_segments)
    
    best_segments = segments.copy()
    best_c5 = evaluate(segment_to_full(segments, n_points))
    best_h = segment_to_full(segments, n_points)
    
    start_time = time.time()
    no_improvement_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s and no_improvement_count < 100:
        iteration += 1
        
        # Optimize segments
        new_segments = optimize_segments(best_segments.copy(), n_points, dx)
        
        # Add randomness occasionally
        if iteration % 20 == 0:
            perturb_indices = np.random.choice(n_segments, max(1, n_segments // 10), replace=False)
            for i in perturb_indices:
                new_segments[i] += np.random.uniform(-0.1, 0.1)
            new_segments = np.clip(new_segments, 0, 1)
            # Re-normalize to maintain sum constraint
            new_segments = new_segments * (n_segments / 2) / np.sum(new_segments)
        
        # Reconstruct h
        new_h = segment_to_full(new_segments, n_points)
        # Apply smoothing
        new_h = smooth_h(new_h, window=3)
        # Re-normalize
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        new_c5 = evaluate(new_h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_segments = new_segments.copy()
            best_h = new_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Occasionally increase resolution
        if iteration % 50 == 0 and n_segments < 21:
            n_segments += 2
            # Interpolate
            old_segments = best_segments.copy()
            new_segments = np.zeros(n_segments)
            for i in range(n_segments):
                j = i * len(old_segments) // n_segments
                new_segments[i] = old_segments[j]
            best_segments = new_segments.copy()
    
    return (best_h, best_c5, n_points)
