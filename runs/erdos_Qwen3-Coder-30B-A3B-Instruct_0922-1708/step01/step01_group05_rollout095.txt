import numpy as np
import scipy.optimize as so
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create flat
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
        
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Optimization loop using scipy minimize
    best_h = h.copy()
    best_C = float('inf')
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def objective(h_vals):
        return compute_C(h_vals)
    
    def constraint_fun(h_vals):
        return np.sum(h_vals) - target_sum
    
    # Define bounds (0 <= h[i] <= 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use scipy.optimize.minimize with SLSQP
    constraints = {'type': 'eq', 'fun': constraint_fun}
    
    iter_count = 0
    while time() - start_time < budget_s - 1:
        iter_count += 1
        try:
            # Use scipy.optimize.minimize with SLSQP method
            res = so.minimize(
                objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            if res.success:
                h = np.clip(res.x, 0, 1)
                # Re-scale to meet integral constraint
                h = h * (target_sum / np.sum(h))
                C = compute_C(h)
                if C < best_C:
                    best_C = C
                    best_h = h.copy()
        except Exception:
            pass  # Fallback to previous best
            
        # Occasionally reinitialize with a random perturbation
        if iter_count % 20 == 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (target_sum / np.sum(h))
    
    return (best_h.tolist(), best_C, n_points)
