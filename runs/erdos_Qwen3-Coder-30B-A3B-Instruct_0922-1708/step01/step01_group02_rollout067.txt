import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Use a more sophisticated approach: local search with adaptive steps
    n_iter = 0
    max_iter = 1000
    
    while n_iter < max_iter and (time.time() - start_time) < budget_s - 1:
        # Random perturbation of h
        h_new = h.copy()
        # Select random indices to perturb
        n_perturb = max(1, n_points // 20)
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        for i in indices:
            # Perturbation with small step in [-0.05, 0.05]
            delta = np.random.uniform(-0.05, 0.05)
            h_new[i] += delta
        
        # Project to feasible set
        h_new = project_h(h_new)
        
        # Compute new C5
        c5_new = compute_c5(h_new)
        
        # Accept if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new.copy()
            # Reset step size if improvement
        else:
            # Occasionally accept worse solutions for escape from local minima
            if np.random.rand() < 0.05:
                h = h_new.copy()
        
        n_iter += 1
        if (time.time() - start_time) > budget_s - 1:
            break
    
    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
