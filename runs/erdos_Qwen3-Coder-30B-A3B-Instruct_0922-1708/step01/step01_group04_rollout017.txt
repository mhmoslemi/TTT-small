import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.2
        h_start[-1] = 0.8
    
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Segment-based approach for efficient optimization
    def segment_optimization(max_segments=100, min_segments=7):
        best_h = h_start.copy()
        best_c5 = evaluate(best_h)
        start_time = time.time()
        segments = min_segments
        no_improvement_count = 0
        
        # Helper to construct h from segments
        def h_from_segments(c):
            h_vals = np.repeat(c, n_points // segments)
            # Handle remainder
            if len(h_vals) < n_points:
                h_vals = np.pad(h_vals, (0, n_points - len(h_vals)), mode='edge')
            h_vals = h_vals[:n_points]
            # Normalize
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            return h_vals
        
        # Helper to smooth
        def smooth_h(h_vals, sigma=1):
            # Apply gaussian filter
            kernel = np.exp(-0.5 * np.arange(-3*sigma, 3*sigma+1)**2 / sigma**2)
            kernel = kernel / np.sum(kernel)
            h_smooth = np.convolve(h_vals, kernel, mode='same')
            return h_smooth
        
        # Optimization loop
        while segments <= max_segments and time.time() - start_time < budget_s - 10:
            # Create initial segment averages
            c0 = np.full(segments, 0.5 * (n_points / 2) / segments)
            
            # Define objective for segments
            def seg_objective(c):
                h_vals = h_from_segments(c)
                return evaluate(h_vals)
            
            # Constraints
            def constraint_sum(c):
                return np.sum(c) - (n_points / 2)
            
            # Bounds
            bounds = [(0, 1) for _ in range(segments)]
            constraints = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(
                    seg_objective,
                    c0,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 1000, 'ftol': 1e-8}
                )
                
                if res.success:
                    c_opt = res.x
                    h_opt = h_from_segments(c_opt)
                    h_opt = smooth_h(h_opt, sigma=1)
                    c5_opt = evaluate(h_opt)
                    
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
                
            except Exception:
                pass
                
            # Increase segments or stop if no progress
            if no_improvement_count > 20:
                break
            segments = min(segments * 2, max_segments)
        
        return best_h, best_c5
    
    # Run segment optimization
    h_final, c5_final = segment_optimization(max_segments=100)
    
    # Final full optimization pass
    def full_optimize(h_init):
        dx = 2.0 / len(h_init)
        
        def obj(h):
            corr = correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        # Simple perturbation based refinement
        h_best = h_init.copy()
        c5_best = obj(h_best)
        
        # Try small perturbations
        for _ in range(2000):
            h_trial = h_best.copy()
            # Perturb few points
            n_perturb = max(1, len(h_trial) // 20)
            indices = np.random.choice(len(h_trial), n_perturb, replace=False)
            for i in indices:
                pert = np.random.normal(0, 0.01)
                h_trial[i] = np.clip(h_trial[i] + pert, 0, 1)
            # Normalize
            h_trial = h_trial * (len(h_trial) / 2) / np.sum(h_trial)
            c5_trial = obj(h_trial)
            
            if c5_trial < c5_best:
                h_best = h_trial.copy()
                c5_best = c5_trial
        
        return h_best, c5_best
    
    # Final refinement
    h_final, c5_final = full_optimize(h_final)
    
    # Ensure it's valid
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    return (h_final, c5_final, n_points)
