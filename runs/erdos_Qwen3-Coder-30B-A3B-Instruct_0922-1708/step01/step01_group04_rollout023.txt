import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_start = initial_h_values
        n_points = len(h_start)
    except NameError:
        # Create a basic construction with fewer points for faster optimization
        n_points = 60
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.1
        h_start[-1] = 0.9
        
    # Ensure bounds and normalize
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    start_time = time.time()
    
    def evaluate(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def smooth(h, window=3):
        if window < 2:
            return h
        h_smooth = np.convolve(h, np.ones(window)/window, mode='same')
        return h_smooth
    
    def perturb(h, delta=0.03):
        h_new = h.copy()
        n_perturb = max(1, int(n_points * 0.15))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    no_improvement_count = 0
    iterations = 0
    max_iterations = 100000
    
    while (time.time() - start_time < budget_s and 
           no_improvement_count < 100 and
           iterations < max_iterations):
        iterations += 1
        # Try multiple perturbations
        for _ in range(50):
            h_trial = perturb(best_h, 0.03)
            h_trial = smooth(h_trial, window=3)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
                break
        else:
            no_improvement_count += 1
        # Occasionally apply more aggressive perturbations
        if np.random.random() < 0.05:
            h_trial = perturb(best_h, 0.08)
            h_trial = smooth(h_trial, window=5)
            c5_trial = evaluate(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                no_improvement_count = 0
    
    # Final validation
    best_h = project(best_h)
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
