import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangular pulse
    x = np.linspace(0, 2, n_points)
    h = np.clip(np.sin(np.pi * x) + np.sin(np.pi * (x - 1)), 0, 1)
    h = project_h(h, n_points)
    return h

def line_search(h, grad, eta, n_points, dx):
    """Find optimal step size using line search."""
    def c5_at_alpha(alpha):
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        return evaluate_c5(h_new)
    
    # Bracket the minimum
    a, b = 0.0, eta
    fa = c5_at_alpha(a)
    fb = c5_at_alpha(b)
    if fa <= fb:
        # Try increasing b
        for _ in range(5):
            b *= 2
            fb = c5_at_alpha(b)
            if fb < fa:
                fa = fb
            else:
                break
    else:
        # Try decreasing a
        for _ in range(5):
            a /= 2
            fa = c5_at_alpha(a)
            if fa < fb:
                fb = fa
            else:
                break
    
    # Refine using scipy minimize_scalar
    result = optimize.minimize_scalar(c5_at_alpha, bounds=(a, b), method='bounded')
    return result.x if result.success else eta

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using gradient descent with adaptive steps."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number for faster runtime
    n_points = 45
    h = initialize_h(n_points)
    dx = 2.0 / n_points
    
    # Parameters for gradient descent
    eta = 0.1  # initial learning rate
    eta_min = 1e-6
    eta_max = 0.5
    max_no_improvement = 1000
    iter_no_improvement = 0
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s and iter_no_improvement < max_no_improvement:
        # Find shift that gives maximum overlap
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Compute gradient at that shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            grad[i] = (1 - h[(i + shift) % n_points]) * dx
        
        # Line search
        alpha = line_search(h, grad, eta, n_points, dx)
        
        # Update
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        # Update step size
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            iter_no_improvement = 0
            eta = min(eta * 1.2, eta_max)
        else:
            iter_no_improvement += 1
            eta = max(eta / 2, eta_min)
        
        iteration += 1
        
        if eta < eta_min:
            break
    
    # Final validation
    final_c5 = evaluate_c5(best_h)
    return (best_h, final_c5, n_points)
