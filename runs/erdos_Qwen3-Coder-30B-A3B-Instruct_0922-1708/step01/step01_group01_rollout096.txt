import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Dual optimization with refinement
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Initialize dual variables
    lambda_k = np.ones(2 * n_points - 1)
    dual_step = 0.01
    max_iter = 20000
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > 0.9 * budget_s:
            break
            
        # Fix dual variables, optimize h
        # Create overlap matrix and solve for new h
        # Objective: minimize sum_k lambda_k * (1 - h_i * h_{i+k})
        # This is a quadratic optimization with bounds
        # Use projected gradient descent
        
        # Compute current overlaps
        corr = correlate(h, 1 - h, mode="full")
        current_max_idx = np.argmax(corr)
        current_max_shift = current_max_idx - (n_points - 1)
        
        # Simple gradient approximation
        eps = 1e-4
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            # Forward difference
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            # Backward difference
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
            
        # Update h
        h -= 0.01 * grad
        h = project_h(h, n_points)
        
        # Update dual variables
        new_corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(new_corr)
        if max_idx > 0:
            lambda_k[max_idx] += dual_step * (1.0 - new_corr[max_idx])
        
        # Update best solution
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
    # Final gradient descent refinement
    for i in range(10000):
        if time.time() - start_time > budget_s:
            break
            
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[j] = (c_plus - c_minus) / (2 * eps)
        
        h -= 0.001 * grad
        h = project_h(h, n_points)
        
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
    return (best_h, best_c5, n_points)
