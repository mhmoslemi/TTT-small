import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Using a smaller number of points for faster optimization
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Use a separate optimization routine with better convergence
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    def constraint_bounds(h_vec):
        return np.concatenate([h_vec, 1 - h_vec])
    
    # Use scipy minimize with constraints for better optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization
    x0 = h.copy()
    
    try:
        # L-BFGS-B often works better with bounds
        result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                         constraints=constraints, options={'maxiter': 1000})
        h_final = result.x
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        c5_final = objective(h_final)
        
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
            
    except:
        pass
    
    # Perform additional local search
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 1500:
        iteration += 1
        # Local gradient update
        h_current = best_h.copy()
        
        # Slightly perturb and evaluate
        eps = 1e-4
        gradient = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = objective(h_plus)
            gradient[i] = (c5_plus - objective(h_current)) / eps
        
        # Update with adaptive learning rate        
        lr = 0.01
        h_new = h_current - lr * gradient
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        if objective(h_new) < objective(best_h):
            best_h = h_new.copy()
        
    return (best_h.tolist(), objective(best_h), n_points)
