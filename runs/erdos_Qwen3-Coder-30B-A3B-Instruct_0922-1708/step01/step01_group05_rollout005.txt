        # Node values
        values = np.random.rand(n_points)
        values = np.clip(values, 0, 1)
        # Normalize values to satisfy integral constraint
        values = values * (target_sum / np.sum(values))
        particles.append((positions, values))
        velocities.append((np.random.rand(n_points) * 0.1, np.random.rand(n_points) * 0.1))
    
    # Global best
    gbest = (np.array(particles[0][0]), np.array(particles[0][1]))
    gbest_C = float('inf')
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
        
    def evaluate_particle(pos, val):
        # Interpolate values at the grid points from node positions and values
        # Use piecewise linear interpolation
        x_grid = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_grid, pos, val, left=0, right=0)
        h_interp = np.clip(h_interp, 0, 1)
        # Ensure integral constraint
        h_interp = h_interp * (target_sum / np.sum(h_interp))
        return compute_C(h_interp)

            C = evaluate_particle(pos, val)

            r1, r2 = np.random.rand(2, n_points)
