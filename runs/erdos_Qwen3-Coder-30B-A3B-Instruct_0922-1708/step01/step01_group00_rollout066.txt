    population = []

        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    last_improvement = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        num_generations += 1
        
        # Evaluate fitness of population
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()

        if stagnation_count > 300:
            break
            
        # Select elite individuals
        elite_indices = np.argsort(fitness)[:elite_size]

        # Create new population through crossover and mutation
        new_population = elite_individuals.copy()
