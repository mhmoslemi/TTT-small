import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to n_points resolution
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(h)), h)
        h = h_interp
        h = h * (n_points / 2) / np.sum(h)  # Rescale to sum to n_points/2
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Genetic algorithm parameters
    pop_size = 50
    elite_size = 10
    mutation_rate = 0.1
    sigma = 0.05
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_candidate = np.random.rand(n_points) * 0.5
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        h_candidate = np.clip(h_candidate, 0, 1)
        population.append(h_candidate)
    
    # Add the initial solution if available
    if 'initial_h_values' in globals():
        population = [h] + population[1:]
    
    last_improvement = 0
    stagnant_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        if iteration % 100 == 0:
            if best_c5 < 0.38085:
                break
        
        # Evaluate fitness of population
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            last_improvement = iteration
            stagnant_count = 0
        else:
            stagnant_count += 1
            
        if stagnant_count > 300:
            break
            
        # Select elite
        elite = population[:elite_size]
        
        # Create new population through crossover and mutation
        new_population = elite.copy()
        
        while len(new_population) < pop_size:
            # Tournament selection
            idx1 = np.random.randint(0, elite_size)
            idx2 = np.random.randint(0, elite_size)
            parent1 = elite[idx1]
            parent2 = elite[idx2] if idx2 != idx1 else elite[(idx1 + 1) % elite_size]
            
            # Crossover (arithmetic mean)
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, sigma)
            
            # Repair constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
    return (best_h.tolist(), best_c5, n_points)
