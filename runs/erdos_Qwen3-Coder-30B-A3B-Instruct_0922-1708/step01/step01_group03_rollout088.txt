import numpy as np
from scipy import optimize
import time
import math

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric triangle
    x = np.linspace(0, 2, n_points)
    h = np.maximum(0, 1 - np.abs(x - 1))
    h = project_h(h, n_points)
    return h

def line_search(h, i, direction, max_step=0.1):
    """Perform line search along direction for coordinate i"""
    n = len(h)
    c5_before = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5_before
    
    # Try small steps to find improvement
    step = max_step
    for _ in range(10):
        h_new = h.copy()
        h_new[i] += step * direction
        h_new = project_h(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        step *= 0.5
        
    return best_h, best_c5

def coordinate_wise_optimization(h, n_points, max_iter=5000):
    """Optimize h using coordinate-wise updates"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    for iteration in range(max_iter):
        improved = False
        for i in range(n_points):
            # Compute directional derivative (approximate)
            h_plus = current_h.copy()
            h_plus[i] += 1e-4
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = current_h.copy()
            h_minus[i] -= 1e-4
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            grad_i = (c5_plus - c5_minus) / (2e-4)
            
            # Try to decrease along gradient
            direction = -grad_i
            new_h, new_c5 = line_search(current_h, i, direction)
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                improved = True
                
        if not improved:
            break
            
    return current_h, current_c5

def conjugate_gradient_optimization(h, n_points, max_iter=5000):
    """Optimize with nonlinear conjugate gradient method"""
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    
    # Initial gradient estimation
    prev_grad = np.zeros(n_points)
    prev_h = current_h.copy()
    d = -prev_grad  # Initial direction
    
    for iteration in range(max_iter):
        # Perform line search along d
        step = 0.01
        for _ in range(20):  # Try several steps
            h_new = current_h + step * d
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_c5(h_new)
            if c5_new < current_c5:
                current_h = h_new
                current_c5 = c5_new
                break
            step *= 0.5
            
        # Compute new gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = current_h.copy()
            h_plus[i] += 1e-4
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = current_h.copy()
            h_minus[i] -= 1e-4
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2e-4)
        
        # Polak-Ribiere conjugate gradient update
        if iteration > 0:
            beta = max(0, np.dot(grad - prev_grad, grad) / np.dot(prev_grad, prev_grad))
        else:
            beta = 0
        d = -grad + beta * d
        prev_grad = grad
        prev_h = current_h.copy()
        
        # Check for convergence
        if np.linalg.norm(d) < 1e-6:
            break
            
    return current_h, current_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    
    # Initialize
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: Coordinate-wise optimization
    h_opt1, c5_opt1 = coordinate_wise_optimization(h, n_points, max_iter=3000)
    
    # Phase 2: Conjugate gradient optimization
    h_opt2, c5_opt2 = conjugate_gradient_optimization(h_opt1, n_points, max_iter=2000)
    
    # Return best result
    final_h = h_opt2 if c5_opt2 < c5_opt1 else h_opt1
    final_c5 = min(c5_opt2, c5_opt1)
    
    # Ensure final result is valid
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
