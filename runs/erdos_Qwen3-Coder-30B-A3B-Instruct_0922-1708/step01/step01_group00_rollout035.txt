from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points

    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use a more sophisticated optimization approach based on gradient information
    # and structured initialization to avoid local minima
    
    # Initialize with a piecewise constant approach
    K = 10  # number of intervals
    breakpoints = np.linspace(0, 2, K+1)
    h_values = np.ones(K) * 0.5  # initial values
    
    # Optimization strategy: alternate between breaking and optimizing
    # Also keep track of convergence
    old_c5 = float('inf')
    patience = 0
    max_patience = 50
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Convert to full array with correct resolution
        h_full = np.zeros(n_points)
        for i in range(K):
            start_idx = int(breakpoints[i] / dx)
            end_idx = int(breakpoints[i+1] / dx)
            h_full[start_idx:end_idx] = h_values[i]
        
        # Ensure integral is preserved
        h_full = h_full * (n_points / 2) / np.sum(h_full) 
        
        # Compute C5
        correlation = np.correlate(h_full, 1 - h_full, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_full.copy()
            
        # Early stopping based on improvement
        if abs(old_c5 - c5) / old_c5 < 1e-4:
            patience += 1
        else:
            patience = 0
        old_c5 = c5
        
        if patience > max_patience:
            break
            
        # Optimization step: perturb the breakpoints and h_values
        # Sample to generate a better solution
        new_breakpoints = breakpoints.copy()
        new_h_values = h_values.copy()
        
        # Perturb breakpoints slightly
        for i in range(1, K):
            if np.random.rand() < 0.5:

                new_breakpoints[i] = max(0, min(2, new_breakpoints[i] + perturbation))

        new_breakpoints = np.sort(new_breakpoints)
        # Recompute middle breakpoint positions

            start_idx = int(new_breakpoints[i] / dx)
            end_idx = int(new_breakpoints[i+1] / dx)

            # Ensure bounds and normalization
            norm_factor = sum(temp_h) * (dx)  
            temp_h = temp_h * (n_points / 2) / norm_factor

                start_idx = int(new_breakpoints[j] / dx)

            new_h_values[i] -= lr_h * grad_h

        new_h_values = new_h_values * (n_points / 2) / np.sum(new_h_values * dx)
        
        # Update policies
        h_values = new_h_values.copy()
        breakpoints = new_breakpoints.copy()
    
    return (best_h.tolist(), best_c5, n_points)
