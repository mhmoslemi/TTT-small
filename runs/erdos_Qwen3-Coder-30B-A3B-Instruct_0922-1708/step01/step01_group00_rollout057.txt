import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use a gradient-based approach with improved heuristic
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iteration - last_improvement < 300:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Compute approximate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        lr = 0.01 / (1.0 + 0.001 * iteration)
        
        # Update h using gradient descent
        h_new = h - lr * grad
        
        # Clip and rescale to maintain constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Add small perturbation to avoid local minima
        if iteration % 10 == 0:
            noise = np.random.normal(0, 1e-5, n_points)
            h_new += noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally resample and restart if stagnating
        if iteration % 100 == 0 and iteration > last_improvement + 100:
            h = np.clip(h + np.random.normal(0, 1e-4, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            # Reset last improvement checkpoint
            last_improvement = iteration
            
    # Final refinement: try a few iterations of local search
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Local search in small neighborhoods
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        h_new[idx] = np.clip(h[idx] + np.random.normal(0, 1e-3), 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
    # Post-processing: ensure the final h is feasible
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Final validation
    correlation = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (h_final.tolist(), final_c5, n_points)
