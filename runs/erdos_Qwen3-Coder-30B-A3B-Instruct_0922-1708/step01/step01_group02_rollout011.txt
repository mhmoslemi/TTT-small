import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 47
        # Initialize with symmetric construction
        h = np.ones(n_points) * 0.5
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        max_val = np.max(conv)
        return max_val * dx
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Population-based GA approach
    population_size = 100
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = population_size // 10
    
    # Initialize population
    population = []
    for _ in range(population_size):
        individual = np.random.rand(n_points)
        individual = project_h(individual)
        population.append(individual)
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        population[0] = project_h(h)
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    max_generations = 10000
    
    while generation < max_generations and (time.time() - start_time) < budget_s - 1:
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Elitism: keep top individuals
        elite = population[:elite_size]
        
        # Create offspring
        new_population = elite[:]
        while len(new_population) < population_size:
            # Selection (roulette wheel)
            inv_fitness = 1.0 / (np.array(fitness) + 1e-10)
            total_fitness = np.sum(inv_fitness)
            probs = inv_fitness / total_fitness
            parents_idx = np.random.choice(population_size, size=2, p=probs)
            parent1, parent2 = population[parents_idx[0]], population[parents_idx[1]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:point], parent2[point:]])
                child2 = np.concatenate([parent2[:point], parent1[point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate / (1 + generation**0.5):
                        noise = np.random.normal(0, 0.1)
                        child[i] += noise
                child = project_h(child)
                new_population.append(child)
        
        # Trim to population size
        population = new_population[:population_size]
        
        generation += 1
        
        # Adaptive mutation rate
        if generation % 100 == 0:
            mutation_rate = max(0.01, mutation_rate * 0.98)
    
    # Final check
    if best_h is not None:
        final_c5 = compute_c5(best_h)
        return (best_h.tolist(), final_c5, len(best_h))
    
    # Fallback to initial if nothing better found
    final_h = project_h(h)
    final_c5 = compute_c5(final_h)
    return (final_h.tolist(), final_c5, len(final_h))
