```python
import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using PSO + gradient descent"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # PSO parameters
    swarm_size = 50
    w = 0.7
    c1 = c2 = 1.5
    max_iter = 1000
    stagnation_limit = 500
    min_swarm_size = 20
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    # Generate initial swarm
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            # 10% from random, 90% from perturbed initial
            if np.random.rand() < 0.1:
                particle = np.random.rand(n_points)
            else:
                particle = h + np.random.normal(0, 0.05, n_points)
            particle = project_h(particle, n_points)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        score = evaluate_h(particle, dx)
        pbest.append(particle.copy())
        pbest_scores.append(score)
    
    # Global best tracking
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    # PSO main loop
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < 0.7 * budget_s and iteration < max_iter:
        iteration += 1
        
        # Update each particle
        for i in range(len(swarm)):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] = project_h(swarm[i] + velocities[i], n_points)
            
            # Evaluate
            score = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = swarm[i].copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = swarm[i].copy()
                    gbest_score = score
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
        
        # Adapt swarm size if no improvement
        if no_improvement_count > stagnation_limit and len(swarm) > min_swarm_size:
            # Remove worst half of swarm
            scores = [evaluate_h(p, dx) for p in swarm]
            worst_indices = np.argsort(scores)[-len(swarm)//2:]
            for idx in sorted(worst_indices, reverse=True):
                del swarm[idx]
                del velocities[idx]
                del pbest[idx]
                del pbest_scores[idx]
            no_improvement_count = 0
    
    # Gradient descent phase
    final_h = gbest.copy()
    final_c5 = gbest_score
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < 0.95 * budget_s:
        # Compute overlap for all shifts and find worst one
        corr = correlate(final_h, 1 - final_h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient w.r.t. h
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = final_h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = final_h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        final_h = final_h - learning_rate * np.sign(grad)
        final_h = project_h(final_h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(final_h, dx)
        if new_c5 < final_c5:
            final_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (final_h, final_c5, n_points)
```
