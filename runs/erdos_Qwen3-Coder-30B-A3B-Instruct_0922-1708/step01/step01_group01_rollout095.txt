import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def dual_optimization_phase(initial_h, n_points, budget_s, dx):
    """Dual optimization approach to improve the solution"""
    start_time = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Initialize dual variables
    lambda_vals = np.ones(2 * n_points - 1)
    learning_rate = 0.01
    
    while time.time() - start_time < 0.8 * budget_s:
        # Fix dual variables and optimize h
        # Use a simple gradient ascent step for dual variables based on current overlap
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Gradient step in dual space
        lambda_vals += learning_rate * (1 - corr)
        lambda_vals = np.maximum(lambda_vals, 0)

        # Since we can't directly solve the optimization due to non-convexity,
        # we just do a simple adaptive gradient descent step on h
        
        # Compute empirical gradient via finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update h in direction of negative gradient
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        # Update best solution if found
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:

        # Adaptive learning rate
        if np.random.rand() < 0.05:
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                learning_rate = 1e-6
    
    return best_h, best_c5

def local_refinement_phase(h, n_points, budget_s, dx):
    """Perform local gradient-based refinement"""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    learning_rate = 0.01
    
    while time.time() - start_time < 0.2 * budget_s:
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update h
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        # Re-evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Adapt learning rate
        if np.random.rand() < 0.05:
            learning_rate *= 0.95
            
    return best_h, best_c5
