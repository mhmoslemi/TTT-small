import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def lhs_sample(n_points, n_samples):
    samples = []
    for _ in range(n_samples):
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        samples.append(h)
    return samples

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    
    # Phase 1: LHS Sampling
    samples = lhs_sample(n_points, 50)
    fitness = [evaluate_h(s, dx) for s in samples]
    top_indices = np.argsort(fitness)[:20]
    population_h = [samples[i] for i in top_indices]
    
    # Phase 2: Evolution Strategy
    while time.time() - start_time < 0.7 * budget_s:
        new_population = []
        for h_ind in population_h:
            for _ in range(5):
                h_new = h_ind + np.random.normal(0, 0.05, n_points)
                h_new = project_h(h_new, n_points)
                new_population.append(h_new)
        
        all_pop = population_h + new_population
        fitness = [evaluate_h(s, dx) for s in all_pop]
        top_indices = np.argsort(fitness)[:20]
        population_h = [all_pop[i] for i in top_indices]
        
        current_best = evaluate_h(population_h[0], dx)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population_h[0].copy()
    
    # Phase 3: Gradient Descent
    h = best_h.copy()
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < budget_s:
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with sign of gradient to reduce oscillation
        h -= learning_rate * np.sign(grad)
        h = project_h(h, n_points)
        
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return (best_h, best_c5, n_points)
