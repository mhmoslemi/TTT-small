import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: symmetric function with one peak
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    # Two peaks around 0.5 and 1.5
    h += 0.5 * np.exp(-((x - 0.5) ** 2) / 0.1)
    h += 0.5 * np.exp(-((x - 1.5) ** 2) / 0.1)
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function using particle swarm optimization and gradient descent."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45  # odd for symmetry
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Initialize swarm
    swarm_size = 30
    particles = []
    velocities = []
    pbests = []
    pbest_scores = []
    gbest = h.copy()
    gbest_score = evaluate_c5(h)
    
    # Initialize particle positions and velocities
    for i in range(swarm_size):
        particle = initialize_h(n_points)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
        pbests.append(particle.copy())
        pbest_scores.append(evaluate_c5(particle))
        
        if pbest_scores[i] < gbest_score:
            gbest = pbests[i].copy()
            gbest_score = pbest_scores[i]
    
    # PSO parameters
    cognitive_weight = 2.0
    social_weight = 2.0
    inertia_weight = 0.9
    velocity_max = 0.2
    
    # PSO main loop
    for iteration in range(10000):  # Maximum iterations
        if time.time() - start_time > budget_s * 0.7:
            break
            
        for i in range(swarm_size):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_weight * r1 * (pbests[i] - particle)
            social = social_weight * r2 * (gbest - particle)
            new_velocity = inertia_weight * velocity + cognitive + social
            
            # Clip velocity
            new_velocity = np.clip(new_velocity, -velocity_max, velocity_max)
            
            # Update particle
            new_particle = particle + new_velocity
            
            # Project to feasible region
            new_particle = project_h(new_particle, n_points)
            
            # Evaluate
            new_score = evaluate_c5(new_particle)
            
            # Update personal best
            if new_score < pbest_scores[i]:
                pbests[i] = new_particle.copy()
                pbest_scores[i] = new_score
                if new_score < gbest_score:
                    gbest = new_particle.copy()
                    gbest_score = new_score
            
            # Update particle and velocity
            particles[i] = new_particle
            velocities[i] = new_velocity
            
        # Decrease inertia weight
        inertia_weight = 0.9 - (0.5 * iteration / 10000)
    
    # Gradient descent refinement
    h_refined = gbest.copy()
    
    def compute_gradient(h, dx):
        # Compute gradient for the current solution
        n = len(h)
        conv = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)
        
        grad = np.zeros(n)
        eps = 1e-5
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n)
            c5_plus = evaluate_c5(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n)
            c5_minus = evaluate_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    # Refine with gradient descent
    learning_rate = 0.05
    patience = 1000
    patience_counter = 0
    old_score = evaluate_c5(h_refined)
    
    for _ in range(10000):  # Maximum iterations
        if time.time() - start_time > budget_s:
            break
        
        grad = compute_gradient(h_refined, 2.0 / n_points)
        new_h = h_refined - learning_rate * grad
        new_h = project_h(new_h, n_points)
        new_score = evaluate_c5(new_h)
        
        if new_score < old_score:
            h_refined = new_h
            old_score = new_score
            patience_counter = 0
            if learning_rate > 1e-6:
                learning_rate *= 0.995
        else:
            patience_counter += 1
            if patience_counter > patience:
                break
    
    final_c5 = evaluate_c5(h_refined)
    return (h_refined, final_c5, n_points)
