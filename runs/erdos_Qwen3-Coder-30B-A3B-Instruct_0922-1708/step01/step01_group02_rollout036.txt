import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add a plateau in the middle
        mid = n_points // 2
        h[mid-5:mid+5] = 0.75
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    # Genetic algorithm
    population_size = 100
    generations = 0
    max_generations = 5000
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:  # First individual is the initial solution
            individual = h.copy()
        else:
            individual = np.random.rand(n_points)
            individual = project_h(individual)
        population.append(individual)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    mutation_rate = 0.2
    crossover_rate = 0.7
    elitism_rate = 0.1
    
    current_population = population
    
    while generations < max_generations and (time.time() - start_time) < budget_s - 1:
        # Evaluate fitness for current population
        fitness_scores = []
        for individual in current_population:
            c5 = compute_c5(individual)
            fitness_scores.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        sorted_population = [current_population[i] for i in sorted_indices]
        sorted_fitness = [fitness_scores[i] for i in sorted_indices]
        
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]
        
        # Selection using roulette wheel
        # Convert fitness to scores (lower is better)
        inv_fitness = [1.0/(f + 1e-10) for f in sorted_fitness]
        total_fitness = sum(inv_fitness)
        probabilities = [f/total_fitness for f in inv_fitness]
        
        # Create new population
        new_population = []
        
        # Elitism
        num_elites = int(elitism_rate * population_size)
        for i in range(num_elites):
            new_population.append(sorted_population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1_idx = np.random.choice(population_size, p=probabilities)
            parent2_idx = np.random.choice(population_size, p=probabilities)
            
            parent1 = sorted_population[parent1_idx]
            parent2 = sorted_population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points)
                offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                offspring1 = parent1.copy()
                offspring2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate / (1 + generations**0.5):
                    noise = np.random.normal(0, 0.1)
                    offspring1[i] += noise
                    offspring2[i] += noise
                    
            offspring1 = project_h(offspring1)
            offspring2 = project_h(offspring2)
            
            new_population.append(offspring1)
            if len(new_population) < population_size:
                new_population.append(offspring2)
        
        # Trim to exact population size
        current_population = new_population[:population_size]
        
        generations += 1
        
        # Decrease mutation rate over time
        if generations % 100 == 0 and mutation_rate > 0.01:
            mutation_rate *= 0.95
            
    # Final evaluation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
