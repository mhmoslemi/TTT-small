import scipy.optimize as so
from time import time

    n_particles = 100
    n_dims = n_points
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iter = 1000
    
    # Initialize particles
    particles = np.random.rand(n_particles, n_dims)
    particles = np.clip(particles, 0, 1)
    for i in range(n_particles):
        particles[i] *= (target_sum / np.sum(particles[i]))
    
    velocities = np.zeros((n_particles, n_dims))
    
    # Initialize personal best
    personal_best = particles.copy()
    personal_best_scores = np.full(n_particles, float('inf'))
    
    # Initialize global best
    global_best_idx = 0
    global_best_score = float('inf')
    
    # Optimization loop
    iter_count = 0
    while time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        # Evaluate particles
        for i in range(n_particles):
            particle = particles[i]
            # Compute overlap score
            corr = np.correlate(particle, 1-particle, mode="full")
            score = np.max(np.abs(corr)) * dx
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = particle.copy()
            if score < global_best_score:
                global_best_score = score
                global_best_idx = i
                
        # Update particles
        r1, r2 = np.random.rand(2)
        for i in range(n_particles):
            # Compute new velocities
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            # Update particles
            particles[i] += velocities[i]
            # Apply bounds
            particles[i] = np.clip(particles[i], 0, 1)
            # Projection onto constraint
            particles[i] = particles[i] * (target_sum / np.sum(particles[i]))
            
    # Refinement with gradient descent on global best
    h_best = particles[global_best_idx].copy()
    
    # Make sure we have decent starting values
    h = h_best.copy()
    
    # Gradient descent refinement
    lr = 0.01
    max_refine_iter = 500
    no_improvement_count = 0
    best_C = global_best_score
    
    def compute_C(h_vals):
        corr = np.correlate(h_vals, 1-h_vals, mode="full")
        return np.max(np.abs(corr)) * dx
        
    def gradient(h_vals):
        # Approximate gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h_vals)
        base_C = compute_C(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C_plus = compute_C(h_plus)
            grad[i] = (C_plus - base_C) / eps
        return grad
    
    # Refinement loop
    for i in range(max_refine_iter):
        if time() - start_time >= budget_s - 1:
            break
        grad = gradient(h)
        # Update with constrained gradient
        h -= lr * grad
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Compute new score
        new_C = compute_C(h)
        if new_C < best_C:
            best_C = new_C
            h_best = h.copy()
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                lr *= 0.5
                no_improvement_count = 0
        # Early stopping if learning rate gets too small
        if lr < 1e-8:
            break
    
    return (h_best.tolist(), best_C, n_points)
