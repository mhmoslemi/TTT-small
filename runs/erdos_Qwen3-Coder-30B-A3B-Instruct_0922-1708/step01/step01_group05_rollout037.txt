import numpy as np
import time
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a simple construction
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        
    # Ensure sum constraint
    h = h * (target_sum / np.sum(h))
    
    def compute_C(h_vals):
        # Compute C_5 = max_k integral of h(x)(1-h(x+k))
        h = np.array(h_vals)
        corr = np.correlate(h, 1-h, mode="full")
        corr *= dx
        return np.max(np.abs(corr))
    
    # Objective function for optimization (minimize C_5)
    def objective(h_vals):
        return compute_C(h_vals)
    
    # Bounds for each variable [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint function for sum
    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum
    
    # Define constraint in scipy format
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Use differential evolution for global optimization
    result = differential_evolution(
        objective,
        bounds,
        args=(),
        constraints=constraint,
        seed=seed,
        maxiter=1000,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        atol=1e-6,
        rtol=1e-6
    )
    
    h_opt = result.x
    c5_bound = result.fun
    
    # Ensuring final solution satisfies constraints
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (target_sum / np.sum(h_opt))
    c5_bound = compute_C(h_opt)
    
    # Refinement with local optimization if time allows
    if time.time() - start_time < budget_s - 10:
        # Use scipy's minimize for further refinement
        from scipy.optimize import minimize
        
        # Define bounds and constraints
        bnds = [(0, 1) for _ in range(n_points)]
        
        # Gradient-based refinement
        def refined_objective(h_vals):
            return compute_C(h_vals)
        
        # Run local optimization
        local_result = minimize(
            refined_objective,
            h_opt,
            method='L-BFGS-B',
            bounds=bnds,
            options={'maxiter': 500}
        )
        
        if local_result.success:
            h_opt = local_result.x
            c5_bound = compute_C(h_opt)
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (target_sum / np.sum(h_opt))
            c5_bound = compute_C(h_opt)
    
    return (h_opt.tolist(), c5_bound, n_points)
