import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def repair_h(h, n_points):
    """Repair h to satisfy constraints using quadratic projection."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) > 1e-8:
        # Project onto the feasible set: sum = target_sum, bounds [0,1]
        # This is a quadratic programming problem: min ||x - x0||^2 s.t. sum(x) = target, 0 <= x <= 1
        # Solution: clip and rescale
        h = np.clip(h, 0, 1)
        if current_sum > target_sum:
            # Reduce values with largest components
            sort_indices = np.argsort(h)[::-1]
            remaining = target_sum
            for i in sort_indices:
                if remaining <= 0:
                    h[i] = 0
                else:
                    if h[i] >= remaining:
                        h[i] = remaining
                        remaining = 0
                    else:
                        remaining -= h[i]
                        h[i] = 0
        else:
            # Increase values with smallest components  
            sort_indices = np.argsort(h)
            remaining = target_sum - current_sum
            for i in sort_indices:
                if remaining <= 0:
                    break
                if h[i] < 1:
                    increase = min(1 - h[i], remaining)
                    h[i] += increase
                    remaining -= increase
    return h

def evaluate_h(h, dx):
    """Evaluate C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_step(h, dx, epsilon=1e-6):
    """Approximate gradient of C5 with respect to h."""
    n_points = len(h)
    grad = np.zeros(n_points)
    c5_base = evaluate_h(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = repair_h(h_plus, n_points)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / epsilon
    
    return grad

    dx = 2.0 / n_points
    start_time = time.time()

        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with feasible random solution
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
    
    # Restart from best known solution
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    max_iter = 20000
    step_size = 0.01
    perturb_rate = 0.05
    patience = 500
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Adaptive step size
        step_size = max(1e-6, step_size * 0.9999)
        
        # Gradient descent step
        grad = gradient_step(h_current, dx)
        h_new = h_current - step_size * grad
        
        # Perturb some elements
        num_perturb = max(1, int(perturb_rate * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            delta = np.random.uniform(-0.05, 0.05)
            h_new[idx] += delta
            
        # Repair feasibility
        h_new = repair_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)

        if c5_new < c5_current:

            # Accept worse solution with low probability
            h_current = h_new
            c5_current = c5_new
            
        # Adaptive perturbation rate
        if i - last_improvement > patience:
            perturb_rate = min(0.2, perturb_rate * 1.1)
            last_improvement = i
    
    return (h_best.tolist(), c5_best, n_points)
