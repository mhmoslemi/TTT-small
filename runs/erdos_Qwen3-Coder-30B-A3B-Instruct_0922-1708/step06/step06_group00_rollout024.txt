import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Nesterov accelerated proximal gradient
    gamma = 0.01
    beta = 0.9
    max_iter = 10000
    
    # Variables for Nesterov acceleration
    v = np.zeros_like(h)
    h_old = h.copy()
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
        
        # Compute all correlations
        corr_full = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr_full) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Identify the maximum lag
        max_lag_idx = np.argmax(corr_full)
        k = max_lag_idx - (len(h) - 1)
        
        # Compute gradient for the specific shift k
        grad = np.zeros_like(h)
        if k >= 0:
            # h[i] contributes to overlap with h[i+k] 
            for i in range(len(h) - k):
                grad[i] -= (1 - h[i + k]) * dx
        else:
            # h[i-k] contributes to overlap with h[i]
            for i in range(-k, len(h)):
                grad[i - k] -= (1 - h[i]) * dx

        # Update with Nesterov acceleration
        v = beta * v + grad
        h_nesterov = h - gamma * v
        h = np.clip(h_nesterov, 0, 1)
        
        # Enforce sum constraint
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / h_sum

        # Perturbation to escape local minima
        if iteration % 1000 == 0 and iteration > 0:
            perturb = np.random.random(n_points) * 0.01
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
