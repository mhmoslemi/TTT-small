import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Distribute the difference to keep values in [0,1]
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def crossover(parent1, parent2):
    """Create offspring via uniform crossover."""
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.1):
    """Apply mutation to a candidate."""
    h = h.copy()
    mask = np.random.rand(len(h)) < mutation_rate
    if np.any(mask):
        noise = np.random.normal(0, noise_std, len(h))
        h[mask] += noise[mask]
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points) * 0.5
        
    # Initialize population
    population_size = 50
    population = []
    for _ in range(population_size):
        h = h_init.copy()
        h = repair_h(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = []
    for h in population:
        value = evaluate_h(h, dx)
        fitnesses.append(value)
    
    best_h = population[np.argmin(fitnesses)]
    best_value = min(fitnesses)
    
    # Evolution parameters
    generations = 1000
    elite_ratio = 0.2
    mutation_rate = 0.1
    noise_std = 0.1
    learning_rate = 0.001
    reduction_factor = 0.95
    
    # Genetic algorithm loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        elite_count = int(elite_ratio * population_size)
        
        # Create offspring
        new_population = []
        new_fitnesses = []
        
        # Elitism
        for i in range(elite_count):
            new_population.append(population[sorted_indices[i]])
            
        # Crossover + Mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = sorted_indices[np.random.randint(elite_count)]
            parent2_idx = sorted_indices[np.random.randint(elite_count)]
            child = crossover(population[parent1_idx], population[parent2_idx])
            child = mutate(child, mutation_rate, noise_std)
            child = repair_h(child, n_points)
            
            value = evaluate_h(child, dx)
            new_population.append(child)
            new_fitnesses.append(value)
            
        # Replace population
        population = new_population[:]
        fitnesses = new_fitnesses[:]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_value:
            best_value = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        # Gradient refinement for top candidate
        refinement_count = 0
        top_h = population[current_best_idx]
        top_fitness = fitnesses[current_best_idx]
        
        while refinement_count < 100 and time.time() - start_time < budget_s - 10:
            # Compute finite difference gradient
            grad = np.zeros(n_points)
            eps = 1e-6
            for j in range(n_points):
                h_plus = top_h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - top_fitness) / eps
            
            # Update
            top_h = top_h - learning_rate * grad
            top_h = repair_h(top_h, n_points)
            corr = correlate(top_h, 1 - top_h, mode='full')
            top_fitness = np.max(corr) * dx
            
            if top_fitness < best_value:
                best_value = top_fitness
                best_h = top_h.copy()
            
            refinement_count += 1
            
        # Adaptive parameters
        if gen % 100 == 0 and gen > 0:
            mutation_rate *= reduction_factor
            noise_std *= reduction_factor
            learning_rate *= reduction_factor
            
    return (best_h.tolist(), best_value, n_points)
