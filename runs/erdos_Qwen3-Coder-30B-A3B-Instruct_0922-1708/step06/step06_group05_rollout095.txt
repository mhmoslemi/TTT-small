import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using FFT-based correlation."""
    # UsingFFT for efficient computation
    corr = signal.convolve(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Start with a simple construction
        h = np.ones(n_points) * 0.5
    
    # Ensure the integral constraint
    h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Parameters
    lr = 0.001
    perturb_rate = 0.05
    max_iter = 10000
    stagnation_limit = 200
    stagnation_count = 0
    decay_factor = 0.999
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite difference
        eps = 1e-5
        grad = np.zeros(n_points)
        h_eval = evaluate_h(h, dx)
        
        # Compute gradient w.r.t. each point
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - h_eval) / eps
        
        # Update
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        # Evaluate
        c5 = evaluate_h(h, dx)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Random perturbation occasionally
        if np.random.rand() < perturb_rate or stagnation_count > stagnation_limit:
            indices = np.random.choice(n_points, size=int(n_points * perturb_rate), replace=False)
            h[indices] += np.random.normal(0, 0.005, len(indices))
            h = repair_h(h, n_points)
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
                
        # Adaptive learning rate decay
        if iteration % 100 == 0:
            lr *= decay_factor
            if lr < 1e-6:
                lr = 1e-6
                
        # Occasionally enforce integral constraint via projection
        if iteration % 100 == 0:
            h = h * (n_points / 2) / np.sum(h)
    
    # Intensive final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 2:
            break
        # Much finer gradient estimation
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
    return (best_h.tolist(), best_c5, n_points)
