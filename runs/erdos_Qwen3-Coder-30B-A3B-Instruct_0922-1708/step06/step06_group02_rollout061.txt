import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = repair_h(h, n_points)
    
    # Initialize tracking
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Shift grid
    shift_grid = np.linspace(-2, 2, 100)
    
    # Parameters
    learning_rate = 0.01
    perturbation_frequency = 50
    max_iter = 10000
    patience = 100
    patience_counter = 0
    last_improvement = 0
    adaptive_lr = True
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        corr = corr * dx
        
        # Find the maximum overlap
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        
        # Find corresponding shift
        shift = shift_grid[max_idx % len(shift_grid)]
        
        # Compute gradient for this shift
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            # Gradient approximation
            if j + int(shift/dx) < n_points and j + int(shift/dx) >= 0:
                grad[j] = -2 * (1 - h[j + int(shift/dx)])
            if j - int(shift/dx) >= 0 and j - int(shift/dx) < n_points:
                grad[j] -= 2 * h[j - int(shift/dx)]
        
        # Apply gradient update
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        
        # Evaluate
        current_c5 = evaluate_h(h, dx)
        
        # Update best if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
            last_improvement = i
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if adaptive_lr and i > 0 and i % 100 == 0 and i > last_improvement + 100:
            learning_rate *= 0.9
            
        # Perturb occasionally
        if i % perturbation_frequency == 0 and i > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = repair_h(h, n_points)
            
        if patience_counter > patience:
            break
            
    return (best_h.tolist(), best_c5, n_points)
