import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute finite difference gradient of C5."""
    n_points = len(h)
    grad = np.zeros(n_points)
    h_base = h.copy()
    c5_base = evaluate_h(h_base, dx)
    
    for i in range(n_points):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n_points)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    
    return grad

def trust_region_step(h, dx, grad, trust_radius=1.0, max_iter=100):
    """Solve trust-region subproblem using projected gradient descent."""
    n_points = len(h)
    d = np.zeros(n_points)
    
    # Initial step: gradient descent without trust region
    d = -grad * 0.01
    
    # Project onto trust region
    dist = np.linalg.norm(d)
    if dist > trust_radius:
        d = d * trust_radius / dist
    
    for _ in range(max_iter):
        # Make sure it's feasible
        h_trial = h + d
        h_trial = repair_h(h_trial, len(h))
        
        # Check if it improves the objective
        c5_trial = evaluate_h(h_trial, dx)
        
        if c5_trial < evaluate_h(h, dx):
            break
            
        # Reduce step size
        d = d * 0.5
        if np.linalg.norm(d) < 1e-12:
            break
    
    return d

    else:
        # Initialize with a low-discrepancy sequence (Halton-style)
        h_init = np.zeros(n_points)

            h_init[i] = (i + 0.5) / n_points
        h_init = repair_h(h_init, n_points)

    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()

    trust_radius = 1.0
    trust_radius_min = 1e-4
    trust_radius_max = 1.0
    alpha = 0.9
    beta = 0.1
    
    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        
        # Compute gradient
        grad = compute_gradient(h_current, dx, eps=1e-6)

        d = trust_region_step(h_current, dx, grad.copy(), trust_radius)

        h_new = h_current + d
