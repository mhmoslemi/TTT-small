def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):

        h_plus[i] += eps

        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        
        h_minus = h_orig.copy()

        h_minus = repair_h(h_minus, n_points)

        c5_minus = np.max(corr_minus) * dx
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

    # Solve: min 0.5 * d^T H d + grad^T d, s.t. ||d|| <= trust_radius, bounds, sum(d) = 0

    # project gradient onto feasible space and scale to trust radius
    
    d = -grad
    d = d - np.mean(d)  # enforce sum to zero
    d_norm = np.linalg.norm(d)
    
    if d_norm > trust_radius:

    # Ensure bounds feasibility

    # Rescale to satisfy bounds
    scale = 1.0

    # Ensure sum(d) = 0
    d = d - np.mean(d)
    
    # Check if within trust region
    if np.linalg.norm(d) > trust_radius:
        d = d * trust_radius / np.linalg.norm(d)
    
    return d
