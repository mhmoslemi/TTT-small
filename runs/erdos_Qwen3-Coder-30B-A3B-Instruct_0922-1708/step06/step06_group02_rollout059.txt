import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def interpolate_h(h_coarse, n_points):
    """Interpolate coarse h to fine resolution."""
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_points)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine

def mutate_h(h, mutation_rate):
    """Mutate h with Gaussian noise."""
    noise = np.random.normal(0, mutation_rate, len(h))
    h_mut = h + noise
    return repair_h(h_mut, len(h))

def crossover_h(h1, h2, crossover_rate=0.3):
    """Crossover two h vectors."""
    mask = np.random.rand(len(h1)) < crossover_rate
    h_child = np.where(mask, h1, h2)
    return h_child

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define resolution stages
    resolutions = [10, 20, 40, 100]
    max_generations = [100, 50, 50, 50]
    mutation_rates = [0.1, 0.08, 0.05, 0.02]
    
    # Start with initial solution or create one
    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values)
        n_points = len(h_best)

        else:

            x_target = np.linspace(0, 2, 100)
            h_coarse = np.interp(x_target, x_orig, h_best)
    else:
        h_coarse = np.random.rand(10)
        h_coarse = h_coarse * 5 / np.sum(h_coarse)
        
    best_h = None
    best_value = float('inf')
    
    # Coarse optimization
    n_points = 10
    dx = 2.0 / n_points
    pop_size = 50
    
    # Initialize populations
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Optimization loop
    for gen in range(max_generations[0]):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            val = evaluate_h(h, dx)
            fitness.append(val)
            if val < best_value:
                best_value = val
                best_h = h.copy()
        
        # Selection (tournament)
        selected = []
        tourn_size = 3
        for _ in range(pop_size // 2):
            indices = np.random.choice(len(population), tourn_size)
            best_idx = indices[np.argmin([fitness[i] for i in indices])]
            selected.append(population[best_idx].copy())
        
        # Crossover and mutation
        new_pop = []
        
        # Elitism: keep top 5
        elite_indices = np.argsort(fitness)[:5]
        for idx in elite_indices:
            new_pop.append(population[idx].copy())
        
        # Generate offspring
        num_offspring = pop_size - 5
        for _ in range(num_offspring):
            if len(selected) >= 2:
                parent1 = selected[np.random.randint(len(selected))]
                parent2 = selected[np.random.randint(len(selected))]
                child = crossover_h(parent1, parent2)
                child = mutate_h(child, mutation_rates[0])
                new_pop.append(child)
            else:
                h = np.random.rand(n_points)
                h = repair_h(h, n_points)
                new_pop.append(h)
                
        population = new_pop
    
    # Refinement steps
    for res_idx, (n_points, gen, mut_rate) in enumerate(zip(resolutions[1:], max_generations[1:], mutation_rates[1:])):
        dx = 2.0 / n_points
        # Interpolate best coarse solution
        h_fine = interpolate_h(best_h, n_points)
        h_fine = repair_h(h_fine, n_points)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        # Get top solutions from previous stage
        pop_size = 50
        population = [h_fine]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = repair_h(h, n_points)
            population.append(h)
        
        # Refinement
        for gen_count in range(gen):
            if time.time() - start_time > budget_s - 10:
                break
                
            fitness = []
            for h in population:
                val = evaluate_h(h, dx)
                fitness.append(val)
                if val < best_value:
                    best_value = val
                    best_h = h.copy()
            
            # Selection
            selected = []
            tourn_size = 3
            for _ in range(pop_size // 2):
                indices = np.random.choice(len(population), tourn_size)
                best_idx = indices[np.argmin([fitness[i] for i in indices])]
                selected.append(population[best_idx].copy())
            
            # Crossover and mutation
            new_pop = []
            
            # Elitism
            elite_indices = np.argsort(fitness)[:5]
            for idx in elite_indices:
                new_pop.append(population[idx].copy())
            
            # Generate offspring
            num_offspring = pop_size - 5
            for _ in range(num_offspring):
                if len(selected) >= 2:
                    parent1 = selected[np.random.randint(len(selected))]
                    parent2 = selected[np.random.randint(len(selected))]
                    child = crossover_h(parent1, parent2)
                    child = mutate_h(child, mut_rate)
                    new_pop.append(child)
                else:
                    h = np.random.rand(n_points)
                    h = h * (n_points / 2) / np.sum(h)
                    h = repair_h(h, n_points)
                    new_pop.append(h)
                    
            population = new_pop
    
    # Local refinement with gradient descent
    if best_h is not None:
        h_current = best_h.copy()
        current_value = best_value
        
        # Use a more adaptive learning rate
        learning_rate = 0.001
        stable_steps = 0
        max_stable = 50
        
        for i in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            eps = 1e-6
            grad = np.zeros(len(h_current))
            for j in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, len(h_current))
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
                
            h_current = h_current - learning_rate * grad
            h_current = repair_h(h_current, len(h_current))
            corr = correlate(h_current, 1 - h_current, mode='full')
            current_value = np.max(corr) * dx
            
            if current_value < best_value:
                best_value = current_value
                best_h = h_current.copy()
                stable_steps = 0
                learning_rate = min(learning_rate * 1.05, 0.01)  # Increase rate
            else:
                stable_steps += 1
                if stable_steps > max_stable:
                    learning_rate *= 0.9  # Decrease if no progress
                    stable_steps = 0
                if learning_rate < 1e-8:
                    break
    
    return (best_h.tolist(), best_value, len(best_h))
