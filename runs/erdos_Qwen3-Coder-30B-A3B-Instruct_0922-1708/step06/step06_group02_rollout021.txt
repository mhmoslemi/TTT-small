import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = repair_h(h, n_points)
    else:
        h = np.ones(n_points) * 0.5
    
    # Start with coarse grid optimization (10, 20, 40, 100 points)
    coarse_points = [10, 20, 40, 100]
    
    # Grid level 1 (10 points)
    if len(coarse_points) > 0:
        n_coarse = coarse_points[0]
        h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), h)
        h_coarse = repair_h(h_coarse, n_coarse)
        
        # Augmented Lagrangian with adaptive penalty
        lambda_lag = 0.0
        rho = 1.0
        max_iter = 200
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient with finite differences
            grad = np.zeros(n_coarse)
            eps = 1e-6
            for j in range(n_coarse):
                h_plus = h_coarse.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_coarse)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * (2.0 / n_coarse)
                grad[j] = (c5_plus - evaluate_h(h_coarse, 2.0 / n_coarse)) / eps
            
            # Update h
            h_new = h_coarse - 0.01 * grad
            h_new = repair_h(h_new, n_coarse)
            
            # Update Lagrange multiplier
            constraint_violation = np.sum(h_new) - n_coarse / 2
            lambda_lag += rho * constraint_violation
            h_coarse = h_new
            
            # Adaptive penalty
            if abs(constraint_violation) > 1e-4:
                rho *= 2.0
                
            # Interpolate back to full resolution
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
            h = repair_h(h, n_points)
    
    # Refine on full grid using gradient descent
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    iter_count = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 5 and no_improvement < 100:
        iter_count += 1
        eps = 1e-6
        grad = np.zeros(n_points)
        
        # Compute gradient using finite differences
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
        
        # Update h
        h_new = h - 0.001 * grad
        h_new = repair_h(h_new, n_points)
        
        current_value = evaluate_h(h_new, dx)
        
        if current_value < best_value:
            best_value = current_value
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        h = h_new
        
    return (best_h.tolist(), best_value, n_points)
