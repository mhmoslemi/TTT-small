import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.full(n_points, 0.5)
        # Add slight Gaussian noise
        noise = np.random.normal(0, 0.1, n_points)
        h_init = h_init + noise
        h_init = repair_h(h_init, n_points)
    
    best_h = h_init.copy()
    best_value = evaluate_h(best_h, dx)
    
    # Gradient-based optimization with LR decay
    learning_rate = 0.001
    max_iter = 10000
    last_improvement = 0
    no_improve_count = 0
    patience = 200
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient with finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        current_c5 = evaluate_h(best_h, dx)
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
            
        # Update
        best_h = best_h - learning_rate * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        
        if new_c5 < best_value:
            best_value = new_c5
            last_improvement = iter_count
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Adaptive learning rate
        if no_improve_count > 10:
            learning_rate *= 0.95
            no_improve_count = 0
            
        if iter_count - last_improvement > patience:
            # Inject some random perturbations
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            h_perturb = repair_h(h_perturb, n_points)
            c5_perturb = evaluate_h(h_perturb, dx)
            if c5_perturb < best_value:
                best_h = h_perturb
                best_value = c5_perturb
                last_improvement = iter_count
        
        # Early stopping if good enough
        if best_value < 0.3807:
            break
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        grad = np.zeros(n_points)
        eps = 1e-6
        current_c5 = evaluate_h(best_h, dx)
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
            
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        
        if new_c5 < best_value:
            best_value = new_c5
    
    return (best_h.tolist(), best_value, n_points)
