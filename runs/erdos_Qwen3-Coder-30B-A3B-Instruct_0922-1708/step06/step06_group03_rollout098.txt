import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of C5 w.r.t. h at the maximizing lag k_max."""
    n_points = len(h)
    grad = np.zeros(n_points)
    for i in range(n_points):
        # For lag k_max, the gradient contribution from term h[i]*(1-h[i-k_max])
        # is -1*(1-h[(i-k_max)%n_points]) if we're at the max
        grad[i] = -1 * (1 - h[(i - k_max) % n_points])
    return grad * dx

def compute_max_lag(h):
    """Compute the lag k that maximizes the overlap integral."""
    n_points = len(h)
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    # Only consider lags that correspond to elements in the full convolution
    k_max = np.argmax(corr)
    # Account for the fact that full convolution gives lags from -(n_points-1) to (n_points-1)
    # but we are interested in the actual positions
    # So k_max corresponds to a lag of k_max - (n_points - 1)
    return k_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Set up optimization parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    decay_factor = 0.999
    perturb_every = 100
    improve_every = 200
    max_iter = 100000
    progress_count = 0
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute max lag
        k_max = compute_max_lag(current_h)
        
        # Compute gradient
        grad = compute_gradient(current_h, dx, k_max)
        
        # Compute perturbation
        if i % perturb_every == 0:
            # Perturb elements where gradient is small
            abs_grad = np.abs(grad)
            weak_indices = np.where(abs_grad < np.percentile(abs_grad, 30))[0]
            if len(weak_indices) > 0:
                for idx in weak_indices:
                    if np.random.rand() < 0.1:
                        delta = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
                        current_h[idx] += delta
                
        # Update using gradient
        current_h = current_h - learning_rate * grad
        
        # Repair
        current_h = repair_h(current_h, n_points)
        
        # Evaluate
        current_c5 = evaluate_h(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            progress_count = 0
            last_improvement = i
        else:
            progress_count += 1
                
        # Update learning rate and perturbation magnitude
        if i % 500 == 0:
            learning_rate *= 0.99
            perturbation_magnitude *= 0.99
        if progress_count > improve_every:
            learning_rate = max(1e-6, learning_rate * 0.9)
            progress_count = 0

    # Final refinement
    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
            
        k_max = compute_max_lag(best_h)
        grad = compute_gradient(best_h, dx, k_max)
        
        best_h = best_h - 0.00001 * grad
        best_h = repair_h(best_h, n_points)
        current_c5 = evaluate_h(best_h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5

    return (best_h.tolist(), best_c5, n_points)
