import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def project_h(h, n_points):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    dx = 2.0 / n_points
    start_time = time.time()

        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Use a projected gradient descent approach
    learning_rate = 5e-4
    max_iter = 10000
    progress_count = 0
    max_progress = 500
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        
        # Compute gradient for each component
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        # Update using projected gradient descent
        new_h = current_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new point
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept the new point if it improves
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                progress_count = 0
            else:
                progress_count += 1
        else:
            progress_count += 1
            
        # Adaptive learning rate
        if progress_count > max_progress:
            learning_rate *= 0.99
            progress_count = 0
        
        # Update learning rate if it's too small
        if learning_rate < 1e-8:
            learning_rate = 1e-4
            
        # Occasionally restart from a random point to avoid local minima
        if i % 1000 == 0 and i > 0 and current_c5 > best_c5 + 1e-6:
