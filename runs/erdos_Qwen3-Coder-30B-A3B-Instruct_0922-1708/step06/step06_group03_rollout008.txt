import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent approach with projected gradient descent
    # Initialize dual variables
    lambda_k = np.zeros(2 * n_points - 1)
    # Step size and learning rate
    lr = 5e-4
    decay = 0.99
    max_iter = 10000
    progress_count = 0
    max_progress = 500
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlations for all lags
        corr = correlate(current_h, 1 - current_h, mode='full')
        
        # Update dual variables using projected gradients
        # The constraint is that the maximum correlation should be minimized
        # We want to push lambda_k towards the direction that decreases the current max
        max_k = np.argmax(corr)
        lambda_k += lr * (corr - np.max(corr))
        lambda_k = np.maximum(lambda_k, 0)  # Non-negativity constraint
        
        # Gradient with respect to h
        grad_h = np.zeros(n_points)
        for i in range(n_points):
            # Compute the gradient component for h[i]
            # This is an approximation using a small finite difference
            for k in range(2 * n_points - 1):
                if k < n_points:
                    # Forward correlation
                    if i + k < n_points:
                        grad_h[i] += lambda_k[k] * (1 - current_h[i + k]) 
                else:
                    # Backward correlation
                    if i - (k - n_points + 1) >= 0:
                        grad_h[i] += lambda_k[k] * (1 - current_h[i - (k - n_points + 1)])
        
        # Update h using gradient descent
        current_h = current_h - lr * grad_h
        current_h = repair_h(current_h, n_points)
        
        # Compute new C5 value
        current_c5 = evaluate_h(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            progress_count = 0
        else:
            progress_count += 1
            
        # Reduce learning rate occasionally
        if it % 500 == 0 and it > 0:
            lr *= decay
            
        # Early stopping if no progress for many iterations
        if progress_count > max_progress:
            break
    
    # Final refinement with gradient descent
    h_refined = best_h.copy()
    best_c5_refined = best_c5
    
    lr = 1e-4
    for _ in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5_refined) / eps
            
        h_refined = h_refined - lr * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_c5_refined = np.max(corr) * dx
        
        if best_c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = best_c5_refined
            
        if abs(best_c5 - best_c5_refined) < 1e-12:
            break
    
    return (best_h.tolist(), best_c5, n_points)
