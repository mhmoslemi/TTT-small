import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h - use initial_h_values if provided
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_value = evaluate_h(best_h, dx)
    
    # Hierarchical optimization: start with coarse grid
    coarse_levels = [10, 20, 40, 100]
    h_current = h_init.copy()
    h_current = repair_h(h_current, n_points)
    
    for level_idx, levels in enumerate(coarse_levels):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Create coarse representation
        x_coarse = np.linspace(0, 2, levels)
        x_fine = np.linspace(0, 2, n_points)
        
        if level_idx == 0:
            h_coarse = h_current.copy()
        else:
            # Interpolate from previous coarse solution
            f = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
            h_coarse = f(x_coarse)
            
        # Optimize the coarse representation
        h_coarse = repair_h(h_coarse, levels)
        
        # Simple optimization using gradient method on coarse grid
        def objective(coarse_vals):
            h_interp = interp1d(x_coarse, coarse_vals, kind='linear', fill_value='extrapolate')
            h_fine = h_interp(x_fine)
            h_fine = repair_h(h_fine, n_points)
            return evaluate_h(h_fine, dx)
            
        def gradient(coarse_vals):
            h_interp = interp1d(x_coarse, coarse_vals, kind='linear', fill_value='extrapolate')
            h_fine = h_interp(x_fine)
            h_fine = repair_h(h_fine, n_points)
            eps = 1e-6
            grad = np.zeros_like(coarse_vals)
            for i in range(len(coarse_vals)):
                coarse_plus = coarse_vals.copy()
                coarse_plus[i] += eps
                h_interp_plus = interp1d(x_coarse, coarse_plus, kind='linear', fill_value='extrapolate')
                h_fine_plus = h_interp_plus(x_fine)
                h_fine_plus = repair_h(h_fine_plus, n_points)
                corr_plus = correlate(h_fine_plus, 1 - h_fine_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - objective(coarse_vals)) / eps
            return grad
        
        # Optimize using scipy minimize with L-BFGS
        try:
            result = minimize(objective, h_coarse, method='L-BFGS-B', bounds=[(0, 1)]*len(h_coarse), 
                              jac=gradient, options={'maxiter': 100, 'ftol': 1e-8})
            h_coarse = result.x
            h_coarse = repair_h(h_coarse, levels)
        except:
            pass
            
        # Interpolate back to fine grid
        if level_idx < len(coarse_levels) - 1:  # Not the final level
            f = interp1d(x_coarse, h_coarse, kind='linear', fill_value='extrapolate')
            h_current = f(x_fine)
        else:
            h_current = h_coarse
            
        h_current = repair_h(h_current, n_points)
        new_value = evaluate_h(h_current, dx)
        if new_value < best_value:
            best_value = new_value
            best_h = h_current.copy()
    
    # Final refinement
    h_refined = best_h.copy()
    current_value = best_value
    
    # Intensive gradient descent on the full resolution
    def objective_full(h_vals):
        h_fine = repair_h(h_vals.copy(), n_points)
        return evaluate_h(h_fine, dx)
        
    def gradient_full(h_vals):
        h_fine = repair_h(h_vals.copy(), n_points)
        eps = 1e-6
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - objective_full(h_vals)) / eps
        return grad
    
    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
        try:
            # Simple gradient descent
            grad = gradient_full(h_refined)
            h_refined = h_refined - 0.001 * grad
            h_refined = repair_h(h_refined, n_points)
            
            new_value = evaluate_h(h_refined, dx)
            if new_value < current_value:
                current_value = new_value
                best_h = h_refined.copy()
        except:
            break
            
    return (best_h.tolist(), best_value, n_points)
