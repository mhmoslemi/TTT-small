import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) * dx = 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initial construction: triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1 - np.abs(x - 1)
        h_init = project_h(h_init, n_points, dx)
    
    # Coordinate-wise gradient descent with projection
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    eta = 0.01
    eps = 1e-6
    max_iter = 100000
    plateau_count = 0
    plateau_threshold = 100
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic coordinate-wise updates
        for idx in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[idx] += eps
            h_plus = project_h(h_plus, n_points, dx)
            c5_plus = evaluate_h(h_plus, dx)
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[idx] -= eps
            h_minus = project_h(h_minus, n_points, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            # Gradient estimate
            gradient = (c5_plus - c5_minus) / (2 * eps)
            
            # Update coordinate
            h_new = h_current.copy()
            h_new[idx] = h_current[idx] - eta * gradient
            h_new = project_h(h_new, n_points, dx)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept if improvement or with probability
            if c5_new < c5_current or np.random.rand() < 0.1:
                h_current = h_new
                c5_current = c5_new
                
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    plateau_count = 0
                else:
                    plateau_count += 1
            else:
                plateau_count += 1
        
        # Adaptive learning rate
        if plateau_count > plateau_threshold:
            eta = max(1e-6, eta * 0.9)
            plateau_count = 0
        
        # Periodic perturbation
        if i % 500 == 0:
            perturb_indices = np.random.choice(n_points, size=max(1, n_points//20), replace=False)
            for idx in perturb_indices:
                h_current[idx] += np.random.normal(0, 0.01)
            h_current = project_h(h_current, n_points, dx)
            c5_current = evaluate_h(h_current, dx)
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        # Gradient descent step
        grad = np.zeros(n_points)
        for j in range(n_points):
            eps = 1e-6
            h_plus = h_best.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points, dx)
            c5_plus = evaluate_h(h_plus, dx)
            h_minus = h_best.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus, n_points, dx)
            c5_minus = evaluate_h(h_minus, dx)
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        h_best = h_best - 0.001 * grad
        h_best = project_h(h_best, n_points, dx)
        c5_best = evaluate_h(h_best, dx)
        
        if time.time() - start_time > budget_s - 5:
            break
    
    return (h_best.tolist(), c5_best, n_points)
