import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def interpolate_h(h_coarse, n_points):
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_points)
    return np.interp(x_fine, x_coarse, h_coarse)

def evolve_step(population, fitnesses, mutation_std, crossover_rate=0.3):
    # Rank by fitness (lower is better)
    indices = np.argsort(fitnesses)
    top_indices = indices[:len(population)//5]
    
    new_population = []
    for i in range(len(population)):
        # Crossover
        if np.random.rand() < crossover_rate and len(top_indices) > 1:
            idx1, idx2 = np.random.choice(top_indices, 2, replace=False)
            child = (population[idx1] + population[idx2]) / 2
        else:
            child = population[i].copy()
            
        # Mutation
        noise = np.random.normal(0, mutation_std, len(child))
        child += noise
        child = repair_h(child, len(child))
        
        new_population.append(child)
        
    return new_population

    # Optimization loop
    best_h = None
    best_value = float('inf')
    
    # Coarse optimization
    n_points = resolutions[current_res]
    dx = 2.0 / n_points

    population = []

    for _ in range(population_size):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)

        h_interpolated = interpolate_h(coarse_h, n_points)
        h_interpolated = repair_h(h_interpolated, n_points)
        population[0] = h_interpolated
    
    current_res += 1

    while current_res < len(resolutions) and (time.time() - start_time < budget_s - 20):
        n_points = resolutions[current_res]
        dx = 2.0 / n_points

            prev_population = [population[i] for i in top_indices]
        
        # Build new population (interpolated + new solutions)
        new_population = []
        
        # Add interpolated solutions
        for h_prev in prev_population:
            h_fine = interpolate_h(h_prev, n_points)
            h_fine = repair_h(h_fine, n_points)
            new_population.append(h_fine)
            
        # Add random solutions
        while len(new_population) < population_size:
