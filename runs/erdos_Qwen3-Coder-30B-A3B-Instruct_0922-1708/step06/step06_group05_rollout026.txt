import numpy as np
import time
import cvxpy as cp
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use higher resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Use CVXPY for convex relaxation if possible
    try:
        # Try convex relaxation approach
        h_var = cp.Variable(n_points)
        # Minimize sum of squared correlations (relaxation of max)
        # Convolution can be approximated
        objective = cp.Minimize(cp.sum_squares(h_var @ (1 - h_var)))
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-6)
        
        if prob.status == cp.OPTIMAL:
            h = h_var.value
            h = repair_h(h, n_points)
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    except:
        # Fall back to gradient descent
        pass
    
    # Optimizer state
    lr = 0.001
    momentum = 0.9
    velocity = np.zeros(n_points)
    stagnation_count = 0
    max_stagnation = 100
    
    for iteration in range(50000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        c5_ref = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_ref) / eps
        
        # Update with momentum
        velocity = momentum * velocity - lr * grad
        h = h + velocity
        h = repair_h(h, n_points)
        
        c5 = evaluate_h(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Restart with new random perturbation if stagnating
        if stagnation_count > max_stagnation:
            h = h + np.random.normal(0, 0.001, n_points)
            h = repair_h(h, n_points)
            stagnation_count = 0
            
    # Final refinement with Adam optimizer
    try:
        from scipy.optimize import minimize
        def obj(h_flat):
            h = h_flat.reshape(-1)
            return evaluate_h(h, dx)
        def grad_func(h_flat):
            h = h_flat.reshape(-1)
            eps = 1e-6
            g = np.zeros_like(h_flat)
            for i in range(len(h_flat)):
                h_plus = h_flat.copy()
                h_plus[i] += eps
                g[i] = (obj(h_plus) - obj(h_flat)) / eps
            return g.flatten()
            
        # Initialize with best solution
        h_init = best_h.copy()
        res = minimize(obj, h_init, method='L-BFGS-B', jac=grad_func,
                       bounds=[(0, 1)] * n_points, options={'maxiter': 500})
        if res.success:
            h_final = res.x
            h_final = repair_h(h_final, n_points)
            c5_final = evaluate_h(h_final, dx)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final
    except:
        # Fall back to local refinement
        pass

    return (best_h.tolist(), best_c5, n_points)
