import numpy as np
import time
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def create_halton_sequence(n_points):
    """Create a Halton sequence for structured initialization."""
    engine = qmc.Halton(d=1, seed=42)
    points = engine.random(n_points)
    # Scale to [0, 2]
    return points.flatten() * 2

def create_sobol_sequence(n_points):
    """Create a Sobol sequence for structured initialization."""
    engine = qmc.Sobol(d=1, seed=42)
    points = engine.random(n_points)
    # Scale to [0, 2]
    return points.flatten() * 2

def bin_to_step_function(x, n_points):
    """Convert a sequence of points to step function over [0,2]."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for point in x:
        if 0 <= point <= 2:
            idx = int(point / dx)
            if idx < n_points:
                h[idx] += 1
    h = h / np.sum(h) * (n_points / 2)  # Normalize to sum to n_points/2
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute finite difference gradient."""
    grad = np.zeros_like(h)
    c5_base = evaluate_h(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, len(h))
        c5_plus = evaluate_h(h_plus, dx)
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = repair_h(h_minus, len(h))
        c5_minus = evaluate_h(h_minus, dx)
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Try different initialization strategies
    init_methods = [
        lambda: bin_to_step_function(create_halton_sequence(n_points), n_points),
        lambda: bin_to_step_function(create_sobol_sequence(n_points), n_points),
        lambda: np.random.rand(n_points),
        lambda: np.ones(n_points) * 0.5
    ]
    
    # Use initial_h_values if available
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            x_orig = np.linspace(0, 2, len(initial_h))
            x_target = np.linspace(0, 2, n_points)
            initial_h = np.interp(x_target, x_orig, initial_h)
        initial_h = repair_h(initial_h, n_points)
    
    num_seeds = 10
    for seed_offset in range(num_seeds):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Initialize
        h = initial_h.copy() if initial_h is not None else init_methods[seed_offset % len(init_methods)]()
        h = repair_h(h, n_points)
        
        # Optimize using gradient-based method
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        step_size = 0.01
        patience = 0
        max_patience = 20
        for iter_count in range(5000):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute gradient
            grad = compute_gradient(h, dx)
            
            # Update step
            h_new = h - step_size * grad
            h_new = repair_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                patience = 0
                step_size = min(0.1, step_size * 1.05)  # Increase step size if improving
            else:
                patience += 1
                step_size *= 0.95  # Reduce step size if stuck
                if patience > max_patience:
                    # Start over with different initialization
                    h = init_methods[seed_offset % len(init_methods)]()
                    h = repair_h(h, n_points)
                    c5 = evaluate_h(h, dx)
                    patience = 0
                    
    # Final refinement with more iterations if time allows
    if best_h is not None:
        h = best_h.copy()
        c5 = best_c5
        
        for _ in range(5000):
            if time.time() - start_time > budget_s - 5:
                break
                
            grad = compute_gradient(h, dx)
            h_new = h - 0.005 * grad
            h_new = repair_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                
        best_h = h
        best_c5 = c5
    
    return (best_h.tolist(), best_c5, n_points)
