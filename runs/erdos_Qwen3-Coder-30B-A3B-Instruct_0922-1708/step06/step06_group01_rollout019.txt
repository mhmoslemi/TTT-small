import numpy as np
import time
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def halton_sequence(n, seed=42):
    """Generate Halton sequence for initialization."""
    sampler = qmc.Halton(d=1, scramble=True, seed=seed)
    points = sampler.random(n)
    return points.flatten()

def sobol_sequence(n, seed=42):
    """Generate Sobol sequence for initialization."""
    sampler = qmc.Sobol(d=1, scramble=True, seed=seed)
    points = sampler.random(n)
    return points.flatten()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize tracking variables
    best_h = None
    best_c5 = float('inf')
    
    # Use multiple random seeds for initialization
    seeds = [seed + i for i in range(10)]
    
    # Initialization phase
    for i, init_seed in enumerate(seeds):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate low-discrepancy sequence
        try:
            halton_points = halton_sequence(n_points, init_seed)
            sobol_points = sobol_sequence(n_points, init_seed)
            
            # Create centered step function by binning and applying sigmoid
            h_init = np.zeros(n_points)
            bin_width = 2.0 / n_points
            for j in range(n_points):
                x = j * bin_width + bin_width/2
                # Centered sigmoid transformation
                h_init[j] = 1.0 / (1.0 + np.exp(-10 * (x - 1.0)))
            
            # Apply different initializations
            if i % 2 == 0:
                # Use Halton points for shaping
                h_init = np.interp(halton_points, [0, 1], [0, 1])
            else:
                # Use Sobol points for shaping
                h_init = np.interp(sobol_points, [0, 1], [0, 1])
                
            # Normalize and clip
            h_init = repair_h(h_init, n_points)
            
        except Exception:
            # Fallback to random
            h_init = np.random.rand(n_points)
            h_init = repair_h(h_init, n_points)
        
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
    
    # Main optimization phase with better repair and gradient steps
    if best_h is None:
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
        best_h = h_init.copy()
        best_c5 = evaluate_h(best_h, dx)
    
    # Gradient-based refinement with adaptation
    for iter_count in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute numerical gradient
        epsilon = 1e-6
        grad = np.zeros(n_points)
        h_base = best_h.copy()
        
        for j in range(n_points):
            h_plus = h_base.copy()
            h_plus[j] += epsilon
            h_plus = repair_h(h_plus, n_points)
            
            h_minus = h_base.copy()
            h_minus[j] -= epsilon
            h_minus = repair_h(h_minus, n_points)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update step
        step_size = 0.01
        best_h = best_h - step_size * grad
        best_h = repair_h(best_h, n_points)
        
        # Evaluate and update best
        current_c5 = evaluate_h(best_h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5

    return (best_h.tolist(), float(best_c5), n_points)
