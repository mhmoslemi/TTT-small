import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a different approach: Poisson spike
        h_init = np.zeros(n_points)
        spike_positions = np.random.poisson(0.5, n_points)
        for i in range(n_points):
            if spike_positions[i] > 0:
                h_init[i] = np.random.uniform(0.5, 1.0)
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimization parameters
    learning_rate = 5e-4
    max_iter = 50000
    cooldown_period = 500
    cooldown_factor = 0.99
    progress_stall = 0
    max_progress_stall = 1000
    
    # Gradient-based optimization with projection
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        h = current_h.copy()
        eps = 1e-6
        grad = np.zeros(n_points)
        h_plus = h.copy()
        
        for j in range(n_points):
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
            h_plus[j] = h[j]  # reset
            
        # Update
        current_h = current_h - learning_rate * grad
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)
        
        # Adaptive learning rate
        if i % cooldown_period == 0 and i > 0:
            learning_rate *= cooldown_factor
            
        # Track best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            progress_stall = 0
        else:
            progress_stall += 1
            
        # Early stopping
        if progress_stall > max_progress_stall:
            break
    
    # Final refinement using gradient ascent (in case we're in a suboptimal region)
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:

        eps = 1e-6
        grad = np.zeros(n_points)

        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
        
    return (best_h.tolist(), best_c5, n_points)
