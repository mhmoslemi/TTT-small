import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Formulate as QP using CVXPY
    h = cp.Variable(n_points)
    t = cp.Variable(1)
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
    
    # Compute overlap constraints: for each shift k, we have the
    # overlap integral = sum_{i} h[i] * (1 - h[i+k]) * dx
    # We need to handle circular correlation correctly for negative and positive shifts
    
    max_lag = n_points - 1
    total_constraints = 0
    
    # Efficiently check all valid lags
    for k in range(-max_lag, max_lag + 1):
        # Construct overlap expression for shift k
        # For negative k: h[i] * (1 - h[i-k]) for valid i
        # For positive k: h[i] * (1 - h[i+k]) for valid i
        
        if k < 0:
            # h[i]* (1 - h[i-k]) where i-k >= 0
            i_valid = np.arange(-k, n_points)
            j_valid = i_valid - k
            # Add constraint sum(h[i] * (1 - h[j])) <= t * dx
            constraint_expr = cp.sum(cp.multiply(h[i_valid], 1 - h[j_valid]))
        elif k > 0:
            # h[i]* (1 - h[i+k]) where i+k < n_points
            i_valid = np.arange(0, n_points - k)
            j_valid = i_valid + k
            constraint_expr = cp.sum(cp.multiply(h[i_valid], 1 - h[j_valid]))
        else:
            # k == 0: h[i] * (1 - h[i])
            i_valid = np.arange(n_points)
            constraint_expr = cp.sum(cp.multiply(h[i_valid], 1 - h[i_valid]))
        
        # Normalize by dx so overlap normalized to integral
        constraints.append(constraint_expr <= t)
        total_constraints += 1
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solving with MOSEK
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s-10)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_bound = t.value * dx
        else:
            # fallback to initial solution
            h_opt = h_init
            c5_bound = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx
    except:
        # If solver fails, return initial solution
        h_opt = h_init
        c5_bound = np.max(np.correlate(h_init, 1 - h_init, mode="full")) * dx

    # Final verification
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    c5_verify = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx

    return (h_opt.tolist(), c5_verify, n_points)
