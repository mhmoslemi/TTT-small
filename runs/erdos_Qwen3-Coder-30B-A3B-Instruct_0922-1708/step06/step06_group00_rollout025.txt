import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h = np.ones(n_points) * 0.5
    
    # Convert to binary (step function representation)
    h_binary = (h > 0.5).astype(float)
    current_sum = np.sum(h_binary)
    
    # Ensure sum constraint
    if abs(current_sum - n_points/2) > 1e-8:
        # Adjust to get exactly n_points/2 non-zeroes
        # Identify indices with value 0 or 1
        idx_0 = np.where(h_binary == 0)[0]
        idx_1 = np.where(h_binary == 1)[0]
        
        to_add = int(n_points/2 - current_sum)
        if to_add > 0:
            # Pick indices to flip from 0 to 1
            # Try to pick elements that are likely to improve the overlap
            # First, get all current indices that are 0
            # Then, flip the first 'to_add' ones
            if len(idx_0) >= to_add:
                flip_indices = idx_0[:to_add]
            else:
                flip_indices = idx_0  # Take all remaining 0s
            for i in flip_indices:
                h_binary[i] = 1
        else:
            # More 1s than needed
            to_remove = int(current_sum - n_points/2)
            if len(idx_1) >= to_remove:
                flip_indices = idx_1[:to_remove]
            else:
                flip_indices = idx_1  # Take all 1s
            for i in flip_indices:
                h_binary[i] = 0

    # Track the best solution found
    best_h = h_binary.copy()
    best_c5 = float('inf')
    
    # Optimization parameters
    temp = 0.1
    cooling_rate = 0.999
    max_iter = 50000
    tabu_list = []
    tabu_size = 50
    
    # Main optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
        
        # Generate a neighbor using bit-flip
        h_candidate = h_binary.copy()
        # Randomly select one bit to flip
        flip_idx = np.random.randint(0, n_points)
        h_candidate[flip_idx] = 1 - h_candidate[flip_idx]
        
        # Ensure we're maintaining the correct sum
        current_sum = np.sum(h_candidate)
        if abs(current_sum - n_points/2) > 1e-8:
            # If sum is not correct, adjust by flipping the opposite bit
            # But we want exactly n_points/2 ones
            # So find both correct and wrong positions and flip accordingly
            ones_needed = int(n_points/2)
            idx_zero = np.where(h_candidate == 0)[0]
            idx_one = np.where(h_candidate == 1)[0]
            current_ones = len(idx_one)
            if current_ones > ones_needed:
                # Need to flip (current_ones - ones_needed) ones
                if len(idx_one) > (current_ones - ones_needed):
                    flip_indices = np.random.choice(idx_one, size=current_ones - ones_needed, replace=False)
                    h_candidate[flip_indices] = 0
            elif current_ones < ones_needed:
                # Need to flip (ones_needed - current_ones) zeros
                if len(idx_zero) > (ones_needed - current_ones):
                    flip_indices = np.random.choice(idx_zero, size=ones_needed - current_ones, replace=False)
                    h_candidate[flip_indices] = 1

        # Compute current overlap
        corr = correlate(h_binary, 1 - h_binary, mode='full')
        c5_current = np.max(corr) * dx
        corr_candidate = correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr_candidate) * dx

        # Evaluate acceptability
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
        
        # Acceptance criterion
        if c5_candidate < c5_current:
            h_binary = h_candidate.copy()
        else:
            # Accept with probability based on temperature
            delta = c5_candidate - c5_current
            if np.random.rand() < np.exp(-delta / temp):
                h_binary = h_candidate.copy()
        
        # Temperature update
        temp *= cooling_rate
        
        # Add current state to tabu list
        state_tuple = tuple(h_binary)
        if state_tuple in tabu_list:
            tabu_list.remove(state_tuple)
        tabu_list.append(state_tuple)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
    
    # Final check of best solution
    corr_best = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_best) * dx
    
    # Ensure result is within constraints
    h_final = best_h.copy()
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
