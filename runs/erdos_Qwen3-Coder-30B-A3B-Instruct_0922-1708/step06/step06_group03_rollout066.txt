import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Gradient descent parameters
    learning_rate = 1e-4
    decay_rate = 0.99
    perturb_freq = 500
    perturb_magnitude = 0.05
    perturb_decay = 0.999
    progress_count = 0
    max_progress = 100
    
    for i in range(100000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute the current overlap
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            # For the convolution at k_max
            # dC5/dh_j = (2/n_points) * sum_k indicator(k == k_max) * (1 - h[(j - k) mod n_points])
            # But we're just focusing on k_max term
            k = k_max
            idx = (j - k) % n_points
            grad[j] = (1 - current_h[idx])
        
        # Apply gradient descent
        new_h = current_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept the new solution if better or with some probability
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                progress_count = 0
            else:
                progress_count += 1
        else:
            progress_count += 1
        
        # Apply perturbations occasionally
        if i % perturb_freq == 0 and progress_count > 50:
            # Add some random noise
            noise = np.random.uniform(-perturb_magnitude, perturb_magnitude, n_points)
            current_h = current_h + noise
            current_h = repair_h(current_h, n_points)
            current_c5 = evaluate_h(current_h, dx)
            progress_count = 0
            perturb_magnitude *= perturb_decay
        
        # Adjust learning rate and perturbation magnitude
        if i % 100 == 0:
            learning_rate *= decay_rate
            
        # Periodically check for improvement
        if progress_count > max_progress:
            learning_rate *= 0.9
            progress_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
