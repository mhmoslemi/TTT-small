import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create a simple initial solution
        h_init = np.ones(n_points) * 0.5
        h_init = repair_h(h_init, n_points)
    
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Gradient descent with adaptive learning rate
    max_iter = 10000
    lr = 5e-4
    patience = 50
    patience_counter = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        # Update h
        current_h = current_h - lr * grad
        current_h = repair_h(current_h, n_points)
        
        # Evaluate new C5
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if patience_counter > patience:
            lr *= 0.9
            patience_counter = 0
            
        # Early stop if learning rate too small
        if lr < 1e-8:
            break
    
    # Final refinement with simulated annealing on best solution
    current_h = best_h.copy()
    current_c5 = best_c5
    
    T = 0.1 * current_c5 if current_c5 > 0 else 0.01
    alpha = 0.99
    sa_iter = 5000
    
    for i in range(sa_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate neighbor
        neighbor_h = current_h.copy()
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for idx in indices:
            delta = np.random.uniform(-0.1, 0.1)
            neighbor_h[idx] += delta
            
        # Repair neighbor
        neighbor_h = repair_h(neighbor_h, n_points)
        neighbor_c5 = evaluate_h(neighbor_h, dx)
        
        # Accept or reject
        delta = neighbor_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = neighbor_h
            current_c5 = neighbor_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        T *= alpha
        
    return (best_h.tolist(), best_c5, n_points)
