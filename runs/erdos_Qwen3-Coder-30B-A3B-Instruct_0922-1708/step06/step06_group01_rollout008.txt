import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set: [0,1]^n with sum h = n/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a triangular pattern
        x = np.linspace(0, 2, n_points)
        h_init = np.maximum(0, 1 - np.abs(x - 1) / 0.5)
        h_init = project_h(h_init, n_points)
    
    # Frank-Wolfe parameters
    max_iter = 10000
    eta_max = 1.0
    patience_limit = 1000
    grad_eps = 1e-6
    
    # Current solution
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    last_improvement = 0
    patience = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += grad_eps
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_current) / grad_eps
        
        # Solve linear subproblem: min_d <grad, d> s.t. 0 <= d <= 1, sum(d) = 0
        # We can solve this by setting d = argmin <grad, d> = -sign(grad) with sum = 0
        # However, we also need to project onto the constraint set [0,1]^n
        d = -np.sign(grad)
        # Adjust so sum(d) = 0
        d = d - np.mean(d)
        
        # Ensure d in feasible direction (dir derivative must be negative or zero)
        dir_deriv = np.dot(grad, d)
        if dir_deriv >= 0:
            d = np.zeros(n_points)
        
        # Line search to find optimal step size
        eta = eta_max
        h_new = h_current + eta * d
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Reduce eta until improvement is found
        while (c5_new >= c5_current and eta > 1e-10):
            eta *= 0.5
            h_new = h_current + eta * d
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
        
        # Update
        h_current = h_new
        c5_current = c5_new
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            patience = 0
        else:
            patience += 1
            
        if patience > patience_limit:
            break
    
    # Final refinement with local search
    h_final = h_best.copy()
    c5_final = c5_best
    
    # Try several gradient descent steps with adaptive step size
    for _ in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
            
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += grad_eps
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_final) / grad_eps
            
        # Update with gradient descent
        step_size = 0.001
        h_try = h_final - step_size * grad
        h_try = project_h(h_try, n_points)
        corr_try = correlate(h_try, 1 - h_try, mode='full')
        c5_try = np.max(corr_try) * dx
        
        if c5_try < c5_final:
            h_final = h_try
            c5_final = c5_try
        else:
            # Try smaller step
            step_size *= 0.1
            if step_size < 1e-10:
                break
            h_try = h_final - step_size * grad
            h_try = project_h(h_try, n_points)
            corr_try = correlate(h_try, 1 - h_try, mode='full')
            c5_try = np.max(corr_try) * dx
            if c5_try < c5_final:
                h_final = h_try
                c5_final = c5_try
    
    # Final evaluation
    c5_final = evaluate_h(h_final, dx)
    
    return (h_final.tolist(), c5_final, n_points)
