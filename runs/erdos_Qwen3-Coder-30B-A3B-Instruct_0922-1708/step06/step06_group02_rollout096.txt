import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_current = h_init.copy()
    else:
        h_current = np.ones(n_points) * 0.5
    
    # Initialize lambda and penalty
    lambda_lag = 0.0
    rho = 1.0
    
    # Multi-resolution optimization
    resolutions = [10, 20, 50, 100]
    best_h = None
    best_value = float('inf')
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
        # Interpolate to current resolution
        if res < n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_current = np.interp(x_new, x_orig, h_current)
            n_points_curr = res
            dx_curr = 2.0 / n_points_curr
        else:
            n_points_curr = n_points
            dx_curr = dx
        
        # Augmented Lagrangian optimization
        max_iter = 200
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros(n_points_curr)
            eps = 1e-6
            
            # Compute current value
            current_value = evaluate_h(h_current, dx_curr)
            
            # Gradient computation
            for j in range(n_points_curr):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points_curr)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx_curr
                grad[j] = (c5_plus - current_value) / eps
            
            # Update with augmented Lagrangian step
            h_new = h_current - 0.001 * grad
            # Apply augmented Lagrangian penalty
            h_new = h_new - lambda_lag * (np.sum(h_new) - n_points_curr / 2)
            h_new = repair_h(h_new, n_points_curr)
            
            # Update Lagrange multiplier
            constraint_violation = np.sum(h_new) - n_points_curr / 2
            lambda_lag += rho * constraint_violation
            
            # Adaptive penalty
            if abs(constraint_violation) > 1e-4:
                rho *= 2
            
            h_current = h_new
            current_value_new = evaluate_h(h_current, dx_curr)
            
            if current_value_new < best_value:
                best_value = current_value_new
                best_h = h_current.copy()
    
    # Final intensive gradient descent refinement
    if best_h is not None:
        h_refined = best_h.copy()
        current_value = best_value
        
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
                
            h_refined = h_refined - 0.0001 * grad
            h_refined = repair_h(h_refined, n_points)
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            current_value = np.max(corr) * dx
            
            if current_value < best_value:
                best_value = current_value
                best_h = h_refined.copy()
    
    # Final check and return
    return (best_h.tolist(), best_value, n_points)
