import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_gradient(h, k, dx, n_points):
    """Compute gradient of the overlap at shift k."""
    grad = np.zeros(n_points)
    for j in range(n_points):
        # Approximate gradient using finite differences
        # Perturb h[j] slightly and compute the change in overlap
        h_plus = h.copy()
        eps = 1e-6
        h_plus[j] += eps
        h_plus = repair_h(h_plus, n_points)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        grad[j] = (c5_plus - c5) / eps
    return grad

        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize uniformly
        h = np.ones(n_points) * 0.5
    
    # Ensure it meets the sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    learning_rate = 0.001
    min_learning_rate = 1e-6
    max_iter = 10000
    patience = 500
    no_improvement_count = 0
    
    shift_grid_size = 100
    shift_grid = np.linspace(-2, 2, shift_grid_size)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Compute all correlations for all shifts in the grid
        correlations = []
        all_k = []
        
        for k in shift_grid:
            # Compute overlap at this shift using circular shift
            if k >= 0:
                shifted_h = np.concatenate((h[int(k/dx):], h[:int(k/dx)]))
            else:
                shifted_h = np.concatenate((h[int(-k/dx):], h[:int(-k/dx)]))
            
            # Compute correlation with 1-h
            corr = np.sum(h * (1 - shifted_h))
            correlations.append(corr)
            all_k.append(k)
        
        # Find the shift that maximizes C5 for the current h
        max_idx = np.argmax(correlations)
        worst_k = all_k[max_idx]
        
        # Compute actual gradient by finite differences at current h for the worst shift
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            
            # Compute overlap at worst_k with h_plus
            if worst_k >= 0:
                shifted_h_plus = np.concatenate((h_plus[int(worst_k/dx):], h_plus[:int(worst_k/dx)]))
            else:
                shifted_h_plus = np.concatenate((h_plus[int(-worst_k/dx):], h_plus[:int(-worst_k/dx)]))
            
            corr_plus = np.sum(h_plus * (1 - shifted_h_plus))
            
            # Compute overlap at current h at the worst_k
            if worst_k >= 0:
                shifted_h = np.concatenate((h[int(worst_k/dx):], h[:int(worst_k/dx)]))

        h = h - learning_rate * grad

        # Evaluate new C5
        current_value = evaluate_h(h, dx)
        
        # Update best solution
        if current_value < best_value:

            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Reduce learning rate if no improvement
        if no_improvement_count > patience:
            learning_rate = max(learning_rate * 0.95, min_learning_rate)
            no_improvement_count = 0
            
        # Occasionally add a small random perturbation to escape local optima

            pert_factor = 0.001
            noise = np.random.normal(0, pert_factor, n_points)
            h = h + noise
            h = repair_h(h, n_points)
    
    return (best_h.tolist(), best_value, n_points)
