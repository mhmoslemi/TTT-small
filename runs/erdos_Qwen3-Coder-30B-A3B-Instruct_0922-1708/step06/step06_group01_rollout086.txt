import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

        h_init = repair_h(h_init, n_points)

        # Initialize with a piecewise linear construction that's known to work well
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a simple shape: raise to 1 near middle, then drop down
        h_init = np.clip(1.0 - np.abs(x - 1.0) * 2.0, 0, 1)
        h_init = repair_h(h_init, n_points)
    
    # Initialize with the starting candidate
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    eta = 0.01  # Step size
    epsilon = 1e-6  # finite difference step
    max_iter = 20000  # max iterations
    plateau_window = 100  # iterations to wait for improvement
    last_improvement = 0
    plateau_count = 0
    
    # Coordinate descent with gradient approximation
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update each coordinate in sequence
        h_new = h_current.copy()
        
        # Cycle through coordinates
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices:
            # Compute gradient estimate
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            
            # Perturb forward
            h_plus[j] += epsilon
            h_plus = repair_h(h_plus, n_points)
            
            # Perturb backward
            h_minus[j] -= epsilon
            h_minus = repair_h(h_minus, n_points)
            
            # Evaluate C5 at both points
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Estimate gradient
            grad_j = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update
            h_new[j] = h_current[j] - eta * grad_j
            
        # Project and clamp
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current:
