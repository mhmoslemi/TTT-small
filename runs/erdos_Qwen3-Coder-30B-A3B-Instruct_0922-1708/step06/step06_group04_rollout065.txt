import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.full(n_points, 0.5)
        h_init[0] = 0.4
        h_init[1] = 0.6
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Gradient descent parameters
    learning_rate = 0.0001
    max_iter = 5000
    patience = 100
    best_h = h_init.copy()
    best_value = evaluate_h(best_h, dx)
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute finite difference gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        # Update step with clipping and re-normalization
        new_h = best_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_value = evaluate_h(new_h, dx)
        
        # Accept improvement or update best
        if new_value < best_value:
            best_h = new_h
            best_value = new_value
            last_improvement = i
        
        # Adaptive learning rate decay
        if i - last_improvement > 50:
            learning_rate *= 0.995
            
        # Diversity injection
        if i - last_improvement > patience:
            # Add small Gaussian noise
            noise = np.random.normal(0, 0.01, n_points)
            new_h = best_h + noise
            new_h = repair_h(new_h, n_points)
            new_value = evaluate_h(new_h, dx)
            if new_value < best_value:
                best_h = new_h
                best_value = new_value
                last_improvement = i
    
    # Final gradient refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        best_value = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_value, n_points)
