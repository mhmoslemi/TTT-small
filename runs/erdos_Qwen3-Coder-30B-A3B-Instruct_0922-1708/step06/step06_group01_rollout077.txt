import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints using quadratic programming."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Project onto the constraint sum = n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) > 1e-8:
        # Use a simple projection approach
        delta = (target_sum - current_sum) / n_points
        h = h + delta
        
        # Clip again after adjustment
        h = np.clip(h, 0, 1)
        
        # Ensure exact sum
        current_sum = np.sum(h)
        adjustment = (target_sum - current_sum) / n_points
        h = h + adjustment
        h = np.clip(h, 0, 1)
        
    return h
