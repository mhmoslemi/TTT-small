import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def interpolate_to_resolution(h_coarse, n_points):
    """Interpolate h from coarse to fine resolution."""
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_points)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine

def es_mutation(h, std):
    """Apply mutation with given standard deviation."""
    mutation = np.random.normal(0, std, len(h))
    h_muted = h + mutation
    return h_muted

def es_selection(population, values, selection_rate=0.2):
    """Select top individuals based on their objective values."""
    indices = np.argsort(values)
    selected_indices = indices[:int(len(indices)*selection_rate)]
    return [population[i] for i in selected_indices]

def es_crossover(parent1, parent2, crossover_rate=0.3):
    """Perform uniform crossover between two parents."""
    child = np.copy(parent1)
    mask = np.random.rand(len(parent1)) < crossover_rate
    child[mask] = parent2[mask]
    return child

def optimize_stage(h_coarse, n_points, generations, mutation_std, selection_rate=0.2, crossover_rate=0.3):
    """Optimize at a specific resolution."""
    population = [h_coarse.copy()]
    for _ in range(5):  # Add 5 random individuals
        rand_h = np.random.rand(n_points)
        rand_h = rand_h * (n_points / 2) / np.sum(rand_h)
        population.append(rand_h)
        
    dx = 2.0 / n_points
    
    for _ in range(generations):
        # Evaluate population
        values = [evaluate_h(h, dx) for h in population]
        
        # Selection
        selected = es_selection(population, values, selection_rate)
        if len(selected) < 2:
            break
            
        # Crossover and mutation to create new population
        new_population = selected.copy()
        
        while len(new_population) < len(population):
            parent1 = np.random.choice(selected)
            parent2 = np.random.choice(selected)
            child = es_crossover(parent1, parent2, crossover_rate)
            child = es_mutation(child, mutation_std)
            child = repair_h(child, n_points)
            new_population.append(child)
            
        population = new_population[:len(population)]
        
    # Return the best individual
    values = [evaluate_h(h, dx) for h in population]
    best_index = np.argmin(values)
    return population[best_index], values[best_index]
