import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Proximal gradient with Nesterov acceleration
    gamma = 0.05
    beta = 0.9
    max_iter = 10000
    
    # For Nesterov acceleration
    v = np.zeros(n_points)
    h_acc = h.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_h(h_vec):
        h_proj = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_proj)
        if abs(h_sum) > 1e-10:
            h_proj = h_proj * (n_points / 2) / h_sum
        return h_proj

    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current C5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr) - (len(h) - 1)
        
        # Compute gradient for worst shift
        grad = np.zeros_like(h)
        
        # For each i, compute the partial derivative w.r.t. h[i]
        for i in range(len(h)):
            # This is the gradient for the overlap term at shift k_max
            if 0 <= i + k_max < len(h):
                grad[i] -= (1 - h[i + k_max]) * dx
        
        # Update with Nesterov acceleration
        h_new = h - gamma * grad
        h_new = project_h(h_new)
        
        # Nesterov acceleration
        v = beta * v + (h_new - h)
        h = h_new + beta * v
        
        # Re-project
        h = project_h(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
