import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Grid for shifts
    shift_grid = np.linspace(-2, 2, 100)
    step_size = 2.0 / 99
    
    iterations = 0
    no_improvement_count = 0
    best_stagnant_value = best_value
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 5:
        iterations += 1
        
        # Compute correlation for all shifts
        correlations = []
        for k in shift_grid:
            # Compute the integral for shift k
            k_ind = int(k / step_size)
            if k_ind < 0:
                k_ind = -k_ind
                # Shift h backwards
                shifted = np.roll(h, k_ind)
                corr = np.sum(h * (1 - shifted)) * dx
            elif k_ind >= n_points:
                # Shift h forwards
                shifted = np.roll(h, -k_ind)
                corr = np.sum(h * (1 - shifted)) * dx
            else:
                # Standard case with shift k_ind
                shifted = np.roll(h, k_ind)
                corr = np.sum(h * (1 - shifted)) * dx
            correlations.append(corr)
        
        max_corr = max(correlations)
        worst_k_idx = np.argmax(correlations)
        worst_k = shift_grid[worst_k_idx]
        
        # Compute gradient at worst_k
        gradient = np.zeros(n_points)
        k_ind = int(worst_k / step_size)
        if k_ind < 0:
            k_ind = -k_ind
            # Gradient for h[j] with respect to shift k in negative direction
            for j in range(n_points):
                left_idx = (j + k_ind) % n_points
                right_idx = (j - k_ind) % n_points
                if j + k_ind < n_points:
                    gradient[j] = -2 * h[left_idx] * dx
                else:
                    gradient[j] = -2 * h[right_idx] * dx
        elif k_ind > 0:
            # Gradient for h[j] with respect to shift k in positive direction
            for j in range(n_points):
                left_idx = (j - k_ind) % n_points
                right_idx = (j + k_ind) % n_points
                if j - k_ind >= 0:
                    gradient[j] = -2 * h[left_idx] * dx
                else:
                    gradient[j] = -2 * h[right_idx] * dx
        else:
            # No shift -- should not happen
            for j in range(n_points):
                gradient[j] = -2 * h[j] * dx
                
        # Gradient descent
        learning_rate = 0.01
        
        h += learning_rate * gradient
        h = repair_h(h, n_points)
        
        # Periodic perturbation to escape local minima
        if iterations % 50 == 0:
            perturbation = np.random.normal(0, 0.001, n_points)
            h += perturbation
            h = repair_h(h, n_points)
            
        # Update best solution
        current_value = evaluate_h(h, dx)
        if current_value < best_value:
            best_value = current_value
            best_h = h.copy()
            best_stagnant_value = best_value
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping if stagnation
        if no_improvement_count >= 500:
            break
            
    return (best_h.tolist(), best_value, n_points)
