import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradients(h, dx, max_idx, n_points):
    """Compute gradient of maximum overlap with respect to h."""
    grad = np.zeros(n_points)
    eps = 1e-6
    
    # For each position compute the gradient
    for i in range(n_points):
        # Perturb h[i] by eps
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n_points)
        
        # Compute new overlap maximum
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        
        grad[i] = (c5_plus - evaluate_h(h, dx)) / eps
    
    return grad

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    learning_rate = 1e-4
    
    # Main optimization loop
    for i in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find maximum overlap index
        corr = correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        
        # Compute gradient of the maximum overlap
        grad = compute_gradients(current_h, dx, max_idx, n_points)
        
        # Update with gradient descent
        current_h = current_h - learning_rate * grad
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Adaptive learning rate
        if i % 100 == 0:
            learning_rate *= 0.99
    
    # Final refinement with smaller learning rate
    for i in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        corr = correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        grad = compute_gradients(best_h, dx, max_idx, n_points)
        best_h = best_h - 1e-5 * grad
        best_h = repair_h(best_h, n_points)
        best_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
