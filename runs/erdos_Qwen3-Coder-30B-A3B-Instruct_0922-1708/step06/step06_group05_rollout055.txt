import numpy as np
import time
import cvxpy as cp
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use larger resolution for better accuracy
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization with integral constraint
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Convex relaxation approach for initial solution
    if time.time() - start_time < budget_s * 0.3:
        try:
            # Create CVXPY problem with regularization
            h_var = cp.Variable(n_points)
            objective = cp.Minimize(
                cp.sum_squares(cp.convolution(h_var, 1 - h_var, method='fft')) + 
                0.1 * cp.sum(h_var) * cp.sum(h_var)  # Regularization
            )
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h = np.array(h_var.value).reshape(-1)
                h = repair_h(h, n_points)
                c5 = evaluate_h(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            pass
    
    # Refinement using projected gradient with adaptive learning rate
    max_iter = 10000
    lr_initial = 0.01
    beta = 0.9
    stagnation_limit = 100
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Use FFT to compute gradients efficiently
        # finite difference for gradient approximation
        eps = 1e-6
        grad = np.zeros(n_points)
        h_eval = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - h_eval) / eps
            
        # Adaptive learning rate
        lr = lr_initial / (1 + (iteration ** beta))
        
        # Update
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        # Evaluate
        c5 = evaluate_h(h, dx)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Random perturbation occasionally
        if np.random.rand() < 0.05 or stagnation_count > stagnation_limit:
            indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
            h[indices] += np.random.normal(0, 0.005, len(indices))
            h = repair_h(h, n_points)
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
                
    return (best_h.tolist(), best_c5, n_points)
