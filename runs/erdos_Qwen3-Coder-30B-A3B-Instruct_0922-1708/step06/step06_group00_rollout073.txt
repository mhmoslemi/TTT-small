import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Start with a smooth step function
        x = np.linspace(0, 2, n_points)
        h = np.clip(1.0 - np.abs(x - 1.0), 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Parameters for proximal gradient
    learning_rate = 0.01
    momentum = 0.9
    max_iter = 10000
    
    # Nesterov acceleration variables
    v = np.zeros(n_points)
    y = h.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # For adaptive learning rate
    adaptive_lr = learning_rate
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlations for all lags
        corr = correlate(y, 1 - y, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = y.copy()
        
        # Identify the maximum lag
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(y) - 1)
        
        # Compute gradient w.r.t. the maximum overlap term
        grad = np.zeros_like(y)
        if k >= 0:
            # h[i] contributes to overlap at position k iff i + k < n
            for i in range(len(y) - k):
                grad[i] -= (1 - y[i + k]) * dx
        else:
            # k < 0, so shift is negative
            for i in range(-k, len(y)):
                grad[i - k] -= (1 - y[i]) * dx
        
        # Update with momentum (Nesterov acceleration)
        v = momentum * v + grad
        y = y - adaptive_lr * v
        
        # Proximal step: enforce box constraints [0,1]
        y = np.clip(y, 0, 1)
        
        # Rescale to satisfy sum constraint
        y_sum = np.sum(y)
        if abs(y_sum) > 1e-8:
            y = y * (n_points / 2) / y_sum
        
        # Adapt learning rate
        if iteration % 100 == 0 and iteration > 0:
            # Simple adaptation: decrease if no improvement
            # Check recent performance (simple heuristic)
            pass
        
        # Occasionally introduce perturbations to escape local minima
        if iteration % 500 == 0 and iteration > 0:
            perturb = np.random.random(n_points) * 0.02
            y = y + perturb
            y = np.clip(y, 0, 1)
            y_sum = np.sum(y)
            if abs(y_sum) > 1e-8:
                y = y * (n_points / 2) / y_sum
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
