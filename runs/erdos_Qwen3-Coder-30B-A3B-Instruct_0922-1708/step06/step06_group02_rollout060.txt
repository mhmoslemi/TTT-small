import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_gradient(h, dx, k_shift):
    """Compute gradient of overlap at specific shift k."""
    n_points = len(h)
    grad = np.zeros(n_points)
    
    # For each position, compute how changing it affects overlap at k_shift
    # Approximate gradient using finite differences or direct method
    
    # Constant for speed computing change: 2*dx because of correlation derivative
    for i in range(n_points):
        # Simplify gradient approx: 
        # d/dh_i [int h(x)(1-h(x+k)) dx] = 1 - 2*h(i+k) where valid indices
        if i + k_shift < n_points and i + k_shift >= 0:
            grad[i] += -2.0 * dx * h[i + k_shift]
        if i - k_shift >= 0 and i - k_shift < n_points:
            grad[i] += -2.0 * dx * h[i - k_shift]
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise use uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Optimization parameters
    learning_rate = 0.01
    max_iter = 10000
    tolerance = 1e-6
    no_improvement_patience = 500
    shake_interval = 50
    grid_size = 100
    k_grid = np.linspace(-2, 2, grid_size)
    avg_improvement = 0.0
    
    # Iterate
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find worst shift (shift where C(k) is maximum)
        corr_values = []
        for k in k_grid:
            if k >= 0:
                shifted = np.concatenate([h[int(k/dx):], h[:int(k/dx)]])
            else:
                shifted = np.concatenate([h[int(-k/dx):], h[:int(-k/dx)]])
            corr = np.sum(h * (1 - shifted)) * dx
            corr_values.append(corr)
            
        max_k_idx = np.argmax(corr_values)
        worst_shift = k_grid[max_k_idx]
        
        # Compute gradient at worst shift
        grad = compute_gradient(h, dx, int(worst_shift/dx))
        
        # Update h
        h_new = h - learning_rate * grad
        h_new = repair_h(h_new, n_points)  # Enforce constraints
        
        # Evaluate new solution
        new_value = evaluate_h(h_new, dx)
        
        # Update best solution if improved
        if new_value < best_value:
            best_h = h_new.copy()
            best_value = new_value
            avg_improvement = (avg_improvement * 0.9 + (best_value - new_value) * 0.1)
            learning_rate = max(1e-5, learning_rate * 0.999)  # Decrease learning rate
        else:
            avg_improvement = avg_improvement * 0.9
            
        # Set h to new solution
        h = h_new
        
        # Perturb occasionally to escape local minima
        if iteration % shake_interval == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            h_perturbed = h + perturb
            h_perturbed = repair_h(h_perturbed, n_points)
            new_val = evaluate_h(h_perturbed, dx)
            if new_val < best_value:
                best_h = h_perturbed.copy()
                best_value = new_val
                h = h_perturbed
        elif iteration % 100 == 0:
            # Periodically adjust learning rate
            if avg_improvement < tolerance:
                learning_rate = min(0.1, learning_rate * 1.1)
                
    return (best_h.tolist(), best_value, n_points)
