import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy constraints using quadratic programming."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Project onto feasible set
        diff = target_sum - current_sum
        # Simple projection: adjust values while keeping in [0,1]
        h = h + diff / n_points
        # Clamp and renormalize
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    return h

    start_time = time.time()

        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
    
    # Gradient descent with adaptive step
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Adaptive parameters
    step_size = 0.01
    perturb_rate = 0.05
    patience = 1000
    last_improvement = 0
    
    for i in range(20000):  # reducible to budget
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient approximation via finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_current) / eps
            
        # Update using gradient
        h_new = h_current - step_size * grad
        
        # Perturb some elements
        num_perturb = max(1, int(perturb_rate * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.01)
            
        # Repair feasibility
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject based on improvement
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            # Accept with some probability
            if np.random.rand() < np.exp(-(c5_new - c5_current) / (0.01 * max(1, i/1000))):
                h_current = h_new
                c5_current = c5_new

            perturb_rate *= 1.01
