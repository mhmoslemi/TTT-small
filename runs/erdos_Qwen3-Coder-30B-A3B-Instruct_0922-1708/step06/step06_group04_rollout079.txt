import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Define variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        t >= 0
    ]
    
    # Set up overlap constraints
    # For each lag j from -n_points+1 to n_points-1, compute the overlap
    # We'll focus on non-circular convolution indices that make sense
    
    max_lag = n_points - 1
    for lag in range(-max_lag, max_lag + 1):
        # This is an overlap at shift k = -lag
        # We only consider valid range of indices
        if lag >= 0:
            # h[i] * (1 - h[i-lag]) for valid indices
            expr1 = h[:n_points-lag]
            expr2 = 1 - h[lag:]
            overlap = cp.sum(expr1 * expr2)
        elif lag < 0:
            # h[i] * (1 - h[i-lag]) for valid indices
            expr1 = h[-lag:]
            expr2 = 1 - h[:-lag]
            overlap = cp.sum(expr1 * expr2)
        else:
            overlap = 0
            
        # Add constraint: overlap <= t * (n_points / 2) * dx
        # This is because in the discrete case, the normalization factor is 1/dx
        constraints.append(overlap <= t * (n_points / 2))
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Try to solve
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_result = h.value
            c5_bound = t.value
        else:
            # Fall back to initial solution
            h_result = h_init
            c5_bound = evaluate_h(h_init, dx)
    except:
        # Fall back to initial solution
        h_result = h_init
        c5_bound = evaluate_h(h_init, dx)
    
    # Fine-tune with gradient descent
    def compute_grad(h_vec, eps=1e-6):
        grad = np.zeros_like(h_vec)
        base_c5 = evaluate_h(h_vec, dx)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    h_refined = h_result.copy()
    prev_c5 = c5_bound
    
    for i in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        grad = compute_grad(h_refined)
        h_refined = h_refined - 0.001 * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        new_c5 = evaluate_h(h_refined, dx)
        if new_c5 < prev_c5:
            prev_c5 = new_c5
            h_result = h_refined.copy()
    
    return (h_result.tolist(), prev_c5, n_points)
