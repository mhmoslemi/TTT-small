import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    T0 = 1.0
    alpha = 0.99
    lr = 0.01
    n_points = 20
    dx = 2.0 / n_points
    T = T0
    best_h = None
    best_c5 = float('inf')
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():

        # Interpolate to match n_points

        h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        # Perturb
        h_new = h + np.random.normal(0, T, n_points)
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Annealing
        T *= alpha
        iter_count += 1
        
        # Resolution change
        if iter_count % 500 == 0 and n_points < 200:
            n_points *= 2
            dx = 2.0 / n_points
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
            h = repair_h(h, n_points)
            # Re-optimize at new resolution
            lr = 0.01
            for _ in range(100):
                if time.time() - start_time > budget_s - 5:
                    break
                eps = 1e-6
                grad = np.zeros(n_points)
                c5_eval = evaluate_h(h, dx)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = repair_h(h_plus, n_points)
                    c5_plus = evaluate_h(h_plus, dx)
                    grad[i] = (c5_plus - c5_eval) / eps
                h = h - lr * grad
                h = repair_h(h, n_points)
                c5 = evaluate_h(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            # Reset learning rate
            lr = 0.01
        elif iter_count % 1000 == 0:
            lr *= 0.1
            
    # Final refinement
    for _ in range(200):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)

            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)

            grad[i] = (c5_plus - c5_eval) / eps
        best_h = best_h - 0.001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
