import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, max_k):
    """Compute gradient of C5 with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        # For the specific lag that maximizes overlap
        # This is an approximation to the gradient
        # The true gradient involves the derivative of the max operation
        # For simplicity, we approximate it with the derivative of the 
        # overlap function at that lag.
        grad[i] = 0.0
        for k in range(n):
            # Handle wraparound
            idx = (i - k) % n
            grad[i] += (1 - h[idx])  # This is a simplified version
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimization parameters
    learning_rate = 1e-4
    perturb_rate = 0.05
    max_iter = 100000
    progress_count = 0
    max_progress = 1000
    perturb_every = 500
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute current gradient by numerical approximation
        # We will use the correlation to find the lag that maximizes overlap,
        # and compute a gradient using the derivative of that specific term
        corr = correlate(current_h, 1 - current_h, mode='full')
        max_k = np.argmax(corr)
        k_max = max_k - (n_points - 1)  # convert from full to valid index
        
        # Estimate gradient - for the specific k_max
        grad = np.zeros(n_points)
        for j in range(n_points):
            idx = (j - k_max) % n_points
            grad[j] = (1 - current_h[idx])
        
        # Normalize the gradient
        grad = grad * dx
        
        # Apply gradient descent
        new_h = current_h - learning_rate * grad
        
        # Add some random perturbations periodically
        if i % perturb_every == 0:
            perturb_mag = perturb_rate * (1 - (i / max_iter))
            perturb = np.random.normal(0, perturb_mag, n_points)
            new_h = new_h + perturb
            
        # Repair the solution
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept if better
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                progress_count = 0
            else:
                progress_count += 1
                
        else:
            # Even if worse, occasionally accept to escape local minima
            if np.random.rand() < 0.01:
                current_h = new_h
                current_c5 = new_c5
                
        # Adapt learning rate and perturbation
        if progress_count > max_progress:
            learning_rate *= 0.99
            perturb_rate *= 0.99
            progress_count = 0
            
        # Decrease learning rate over time
        if i % 100 == 0 and i > 0:
            learning_rate *= 0.99
            
    return (best_h.tolist(), best_c5, n_points)
