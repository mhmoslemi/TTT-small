import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50000
    tabu_size = 50
    initial_temp = 0.1
    cooling_rate = 0.999
    
    # Initialize from parent or random binary
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = np.round(h_init).astype(int)
    else:
        h = np.random.randint(0, 2, n_points)
    
    # Ensure sum = n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points // 2
    if current_sum != target_sum:
        # Flip bits to balance sum
        flips = target_sum - current_sum
        if flips > 0:
            # Need to flip 0s to 1s
            zero_indices = np.where(h == 0)[0]
            if len(zero_indices) >= flips:
                flip_indices = np.random.choice(zero_indices, flips, replace=False)
                h[flip_indices] = 1
        elif flips < 0:
            # Need to flip 1s to 0s
            one_indices = np.where(h == 1)[0]
            if len(one_indices) >= -flips:
                flip_indices = np.random.choice(one_indices, -flips, replace=False)
                h[flip_indices] = 0
    
    # Convert to float for processing
    h_float = h.astype(float)
    
    # Tabu list to prevent cycling
    tabu_list = []
    
    # Compute initial overlap
    corr = correlate(h_float, 1 - h_float, mode='full')
    current_c5 = np.max(corr) * dx
    
    best_h = h_float.copy()
    best_c5 = current_c5
    
    temperature = initial_temp
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping one bit
        candidate_h = h_float.copy()
        flip_idx = np.random.randint(0, n_points)
        candidate_h[flip_idx] = 1 - candidate_h[flip_idx]
        
        # Project to [0,1] and ensure sum = n_points/2
        candidate_h = np.clip(candidate_h, 0, 1)
        h_sum = np.sum(candidate_h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            candidate_h = candidate_h * target_sum / h_sum
        
        # Avoid tabu moves
        if tuple(candidate_h.round(10)) in tabu_list:
            continue
            
        # Compute new overlap
        corr = correlate(candidate_h, 1 - candidate_h, mode='full')
        candidate_c5 = np.max(corr) * dx
        
        # Accept or reject
        if candidate_c5 < current_c5:
            h_float = candidate_h.copy()
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_h = candidate_h.copy()
                best_c5 = candidate_c5
        else:
            # Accept with probability based on temperature
            delta = candidate_c5 - current_c5
            if np.random.rand() < np.exp(-delta / temperature):
                h_float = candidate_h.copy()
                current_c5 = candidate_c5
        
        # Update tabu list
        tabu_list.append(tuple(h_float.round(10)))
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
            
        # Cool down temperature
        temperature *= cooling_rate
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
