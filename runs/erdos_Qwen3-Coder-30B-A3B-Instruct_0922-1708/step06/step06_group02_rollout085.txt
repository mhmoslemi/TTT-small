import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h = h_init.copy()
    else:
        h = np.ones(n_points) * 0.5  # Simple initialization
        h = h * (n_points / 2) / np.sum(h)
    
    # Initial best solution
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Optimization parameters
    learning_rate = 0.01
    max_iter = 5000
    no_improvement_limit = 100
    perturb_freq = 50
    max_shifts = 200
    
    # Generate shift grid
    shifts = np.linspace(-2, 2, max_shifts)
    
    iter_count = 0
    no_improvement_count = 0
    last_best = best_value
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 5:
        # Compute overlaps for all shifts in grid to find worst case
        # Create matrix form for efficient computation
        h_extended = np.concatenate([h, h[:n_points-1]])  # Periodic extension
        worst_overlap = -1
        worst_k_index = 0
        
        # Compute overlaps for a subset of shifts
        subset_indices = np.linspace(0, len(shifts)-1, min(50, len(shifts)), dtype=int)
        for k_idx in subset_indices:
            k = shifts[k_idx]
            k_int = int(np.round(k / dx))
            if k_int >= 0:
                overlap = np.sum(h * h_extended[k_int:k_int+n_points])
            else:
                overlap = np.sum(h_extended[-k_int:-k_int+n_points] * h)
            
            # Convert to actual overlap
            overlap_actual = overlap * dx
            if overlap_actual > worst_overlap:
                worst_overlap = overlap_actual
                worst_k_index = k_idx
                
        # Refine around the worst shift
        k_refined = shifts[worst_k_index]
        k_indices = np.arange(max(0, worst_k_index-10), min(len(shifts), worst_k_index+11))
        
        worst_overlap_refined = -1
        best_k_refined = k_refined
        
        for k_idx in k_indices:
            k = shifts[k_idx]
            k_int = int(np.round(k / dx))
            if k_int >= 0:
                overlap = np.sum(h * h_extended[k_int:k_int+n_points])
            else:
                overlap = np.sum(h_extended[-k_int:-k_int+n_points] * h)
            
            overlap_actual = overlap * dx
            if overlap_actual > worst_overlap_refined:
                worst_overlap_refined = overlap_actual
                best_k_refined = k
                
        # Compute gradient with respect to worst shift
        target_k = best_k_refined
        k_int = int(np.round(target_k / dx))
        gradient = np.zeros(n_points)
        
        # Compute gradient approximation
        for i in range(n_points):
            # Forward difference
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus = repair_h(h_plus, n_points)
            
            # Set up the correlation for the new h_plus
            h_plus_extended = np.concatenate([h_plus, h_plus[:n_points-1]])

                overlap_plus = np.sum(h_plus * h_plus_extended[k_int:k_int+n_points])

                overlap_plus = np.sum(h_plus_extended[-k_int:-k_int+n_points] * h_plus)
                
            h_original_extended = np.concatenate([h, h[:n_points-1]])
            if k_int >= 0:
                overlap_original = np.sum(h * h_original_extended[k_int:k_int+n_points])
            else:
                overlap_original = np.sum(h_original_extended[-k_int:-k_int+n_points] * h)
                
            grad = (overlap_plus - overlap_original) / 1e-6
            gradient[i] = -grad  # Negative gradient because we minimize
        
        # Update h
        h = h - learning_rate * gradient
        
        # Repair h to satisfy constraints
        h = repair_h(h, n_points)
        
        # Check for improvement
        current_value = evaluate_h(h, dx)
        if current_value < best_value:
            best_value = current_value
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Periodic perturbations to escape local minima
        if iter_count % perturb_freq == 0 and iter_count > 0:
            # Add small random noise
            noise = np.random.normal(0, 0.001, n_points)
            h = h + noise
            h = repair_h(h, n_points)
            
        # Adaptive learning rate decay
        if no_improvement_count > 50:
            learning_rate *= 0.9
            
        iter_count += 1
        
        # Termination conditions
        if no_improvement_count > no_improvement_limit:
            break
    
    return (best_h.tolist(), best_value, n_points)
