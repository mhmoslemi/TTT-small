import numpy as np
import time
from scipy.signal import correlate
from scipy.spatial.distance import cdist

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def halton_sequence(n, base=2):
    """Generate Halton sequence for given base."""
    result = np.zeros(n)
    for i in range(n):
        f = 1.0
        r = 0.0
        while i > 0:
            f /= base
            r += (i % base) * f
            i //= base
        result[i] = r
    return result

def sobol_sequence(n):
    """Generate Sobol sequence."""

    # For now, we'll use pseudo-random with good distribution
    return np.random.rand(n)

def generate_initial_h(n_points, seed_val):
    """Generate step function h from low-discrepancy sequence."""
    np.random.seed(seed_val)
    # Use Halton sequence
    x = halton_sequence(n_points, 2)
    # Map to [0, 2]
    x = x * 2
    # Interpolate to step function
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(n_points):
        # Place a step at each point
        idx = int(x[i] / dx)
        if idx < n_points:
            h[idx] = 1.0
    
    # Add some randomness to prevent degeneracy
    for i in range(n_points):
        if np.random.rand() < 0.1:
            h[i] = np.random.rand()
    
    # Make it feasible
    h = repair_h(h, n_points)
    return h

            h_init = generate_initial_h(n_points, seed + i)
            init_h_list.append(h_init)
        
        # Evaluate all and pick best
        best_init_value = float('inf')
        best_init_h = None

    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()

    # Gradient-based optimization with adaptive step size
    max_iter = 50000
    step_size = 0.01
    patience = 500
    last_improvement = 0
    eps = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break

        grad = np.zeros(n_points)

            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)

            h_minus = h_current.copy()

            c5_minus = np.max(corr_minus) * dx
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Gradient descent step
        h_new = h_current - step_size * grad
        h_new = repair_h(h_new, n_points)

        c5_new = evaluate_h(h_new, dx)
