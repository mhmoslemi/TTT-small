import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def interpolate_h(h_coarse, n_points):
    """Interpolate coarse solution to target resolution."""
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_target = np.linspace(0, 2, n_points)
    return np.interp(x_target, x_coarse, h_coarse)

def mutate_h(h, std):
    """Apply Gaussian mutation to h with given standard deviation."""
    noise = np.random.normal(0, std, len(h))
    return h + noise

def crossover_h(h1, h2, prob=0.5):
    """Perform crossover between two solutions."""
    mask = np.random.rand(len(h1)) < prob
    return np.where(mask, h1, h2)

def evolution_strategy(n_points, dx, budget_s):
    """Evolutionary Strategy optimization with multi-resolution."""
    start_time = time.time()
    
    # Coarse resolution for initial search
    coarse_points = 10
    pop_size = 50
    max_generations = 100
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        coarse_h = np.interp(np.linspace(0, 2, coarse_points), 
                            np.linspace(0, 2, len(h_init)), h_init)
    else:
        coarse_h = np.random.rand(coarse_points)
        coarse_h = coarse_h * (coarse_points / 2) / np.sum(coarse_h)
    
    population = [coarse_h.copy() for _ in range(pop_size)]
    
    best_h = None
    best_value = float('inf')
    resolution = coarse_points
    
    # Optimization loop with resolution increases
    res_levels = [10, 20, 40, 100]
    res_idx = 0
    
    while res_idx < len(res_levels):
        if time.time() - start_time > budget_s - 5:
            break
            
        target_points = res_levels[res_idx]
        
        if target_points > resolution:
            # Interpolate from previous resolution
            new_population = []
            for h in population:
                h_interp = interpolate_h(h, target_points)
                new_population.append(h_interp)
            population = new_population
            resolution = target_points
        else:
            # Use existing solutions
            pass
        
        # Update parameters based on resolution
        mutation_std = 0.1 / np.sqrt(target_points)
        selection_pressure = 0.2
        crossover_rate = 0.3
        
        for gen in range(max_generations):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Evaluate population
            fitness = []
            for h in population:
                h_repaired = repair_h(h, target_points)
                value = evaluate_h(h_repaired, dx)
                fitness.append(value)
                
                if value < best_value:
                    best_value = value
                    best_h = h_repaired.copy()
            
            # Sort by fitness
            sorted_indices = np.argsort(fitness)
            elite_size = max(1, int(selection_pressure * pop_size))
            elite_indices = sorted_indices[:elite_size]
            elite = [population[i] for i in elite_indices]
            
            # Update population
            new_population = elite.copy()
            
            # Add mutated copies
            for _ in range(pop_size - len(elite)):
                parent = elite[np.random.randint(0, len(elite))]
                mutated = mutate_h(parent, mutation_std)
                mutated = repair_h(mutated, target_points)
                new_population.append(mutated)
                
                # Possibly do crossover
                if np.random.rand() < crossover_rate:
                    parent2 = elite[np.random.randint(0, len(elite))]
                    crossed = crossover_h(mutated, parent2)
                    crossed = repair_h(crossed, target_points)
                    new_population.append(crossed)
            
            population = new_population
            
        res_idx += 1
    
    # Final refinement with gradient descent on best solution
    if best_h is not None:
        h_refined = best_h.copy()
        current_value = best_value
        learning_rate = 0.001
        
        for i in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
                
            eps = 1e-6
            grad = np.zeros(len(h_refined))
            for j in range(len(h_refined)):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, len(h_refined))
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
            
            h_refined = h_refined - learning_rate * grad
            h_refined = repair_h(h_refined, len(h_refined))
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            current_value = np.max(corr) * dx
            
            if current_value < best_value:
                best_value = current_value
                best_h = h_refined.copy()
    
    return best_h, best_value

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with 100 points
    n_points = 100
    dx = 2.0 / n_points
    
    best_h, best_value = evolution_strategy(n_points, dx, budget_s)
    
    return (best_h.tolist(), best_value, n_points)
