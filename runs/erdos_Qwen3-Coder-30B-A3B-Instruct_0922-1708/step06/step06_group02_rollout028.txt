import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    
    # hierarchical optimization
    coarse_points = [10, 20, 40, 100]
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    for coarse_n in coarse_points:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Create coarse representation
        if coarse_n < n_points:
            # Interpolate to coarse grid
            x_fine = np.linspace(0, 2, n_points)
            x_coarse = np.linspace(0, 2, coarse_n)
            h_coarse = np.interp(x_coarse, x_fine, h)
            h_coarse = h_coarse * (coarse_n / 2) / np.sum(h_coarse)
        else:
            h_coarse = h.copy()
            
        # Optimize coarse representation
        def objective_coarse(h_coarse):
            # Resample to full resolution
            x_coarse = np.linspace(0, 2, len(h_coarse))
            x_fine = np.linspace(0, 2, n_points)
            h_fine = np.interp(x_fine, x_coarse, h_coarse)
            h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
            return evaluate_h(h_fine, dx)
            
        def constraint_func(h_coarse):
            return np.sum(h_coarse) - coarse_n / 2
            
        # Create optimization bounds
        bounds = [(0, 1) for _ in range(len(h_coarse))]
        
        # Optimize with constraints
        try:
            result = minimize(objective_coarse, h_coarse, method='L-BFGS-B', 
                            bounds=bounds, options={'maxiter': 200})
            if result.success:
                h_coarse = result.x
                # Resample to fine grid
                x_coarse = np.linspace(0, 2, len(h_coarse))
                x_fine = np.linspace(0, 2, n_points)
                h = np.interp(x_fine, x_coarse, h_coarse)
                h = repair_h(h, n_points)
                current_value = evaluate_h(h, dx)
                if current_value < best_value:
                    best_value = current_value
                    best_h = h.copy()
        except:
            pass
            
        # Continue with fine-grid optimization on the best result so far
        h = best_h.copy()
    
    # Final intensive local optimization using scipy
    def evaluate_gradient(h):
        eps = 1e-6
        grad = np.zeros_like(h)
        base_value = evaluate_h(h, dx)
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            value_plus = evaluate_h(h_plus, dx)
            grad[j] = (value_plus - base_value) / eps
        return grad

    # Iterative gradient descent
    h = best_h.copy()
    current_value = best_value
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = evaluate_gradient(h)
        h = h - 0.001 * grad
        h = repair_h(h, n_points)
        new_value = evaluate_h(h, dx)
        if new_value < current_value:
            current_value = new_value
            best_h = h.copy()
    
    return (best_h.tolist(), current_value, n_points)
