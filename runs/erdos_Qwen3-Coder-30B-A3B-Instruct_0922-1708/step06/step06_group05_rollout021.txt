import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound using FFT-based convolution."""
    # Using full convolution to get all shifts
    corr = signal.correlate(h, 1 - h, mode='full')
    # Get maximum overlap
    max_overlap = np.max(corr)
    return max_overlap * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    else:
        # Start with a simple construction
        h = np.ones(n_points) * 0.5
        
    # Ensure integral constraint
    h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Hyperparameters
    lr = 0.001
    perturb_rate = 0.05
    max_iter = 10000
    stagnation_limit = 100
    stagnation_count = 0
    
    # Compute initial overlaps for gradient targeting
    corr = signal.correlate(h, 1 - h, mode='full')

        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute gradient with respect to max overlap shift
        eps = 1e-5
        grad = np.zeros(n_points)
