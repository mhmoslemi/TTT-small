import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a more structured approach
        h_init = np.random.rand(n_points) * 0.5
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent parameters
    dual_vars = np.ones(2 * n_points - 1)  # For all lags
    dual_vars /= np.sum(dual_vars)  # Ensure sum is 1
    dual_step_size = 0.1
    temp = 0.1
    learning_rate = 5e-4
    patience = 0
    max_patience = 100
    
    for iteration in range(30000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all overlaps once
        overlaps = correlate(current_h, 1 - current_h, mode='full')
        
        # Update dual variables using variational inference
        # λ_k ∝ exp(-overlap_k / temp)
        new_dual_vars = np.exp(-overlaps / temp)
        new_dual_vars /= np.sum(new_dual_vars)
        dual_vars = (1 - dual_step_size) * dual_vars + dual_step_size * new_dual_vars
        
        # Compute gradient with respect to h
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Gradient contribution from all lags
            for k in range(-n_points+1, n_points):
                if 0 <= i - k < n_points:
                    # Shift of k corresponds to index (i-k) in original
                    grad[i] += dual_vars[k + n_points - 1]
        
        # Update h with gradient descent
        current_h = current_h - learning_rate * grad
        current_h = repair_h(current_h, n_points)
        
        # Recompute objective
        current_c5 = evaluate_h(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            
        # Adaptive learning rate and temperature
        if iteration % 500 == 0:
            learning_rate *= 0.99
        if iteration % 100 == 0:
            temp *= 0.999
            
        # Reset if we're stuck
        if patience > max_patience:
            current_h = h_init.copy()
            patience = 0
            if iteration > 2000:
                # Increase temperature to escape local optima
                temp = max(0.01, temp * 1.1)

    # Final verification
    best_c5 = evaluate_h(best_h, dx)
    best_h = repair_h(best_h, n_points)  # Ensure final repair
    return (best_h.tolist(), best_c5, n_points)
