import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = repair_h(h, n_points)
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Hierarchical optimization approach
    resolutions = [10, 20, 40, 100]
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Coarsen if needed
        if res < n_points:
            h_coarse = np.array([np.mean(h[i*2:(i+1)*2]) for i in range(res)])
            h_coarse = h_coarse * (res / 2) / np.sum(h_coarse)
        else:
            h_coarse = h.copy()
        
        # Gradient-based optimization on coarse h
        def objective(h_coarse_flat):
            h_full = np.repeat(h_coarse_flat, 2)[:n_points]
            h_full = repair_h(h_full, n_points)
            return evaluate_h(h_full, dx)
        
        def grad_objective(h_coarse_flat):
            h_full = np.repeat(h_coarse_flat, 2)[:n_points]
            h_full = repair_h(h_full, n_points)
            eps = 1e-6
            grad = np.zeros_like(h_coarse_flat)
            for i in range(len(h_coarse_flat)):
                h_plus = h_coarse_flat.copy()
                h_plus[i] += eps
                h_plus_full = np.repeat(h_plus, 2)[:n_points]
                h_plus_full = repair_h(h_plus_full, n_points)
                c5_plus = evaluate_h(h_plus_full, dx)
                grad[i] = (c5_plus - objective(h_coarse_flat)) / eps
            return grad
        
        # Optimize using scipy
        try:
            result = minimize(objective, h_coarse, method='L-BFGS-B', 
                              bounds=[(0, 1)] * len(h_coarse),
                              jac=grad_objective, options={'maxiter': 100})
            if result.success:
                h_coarse = result.x
        except:
            pass
        
        # Refine to full resolution
        if res < n_points:
            h_full = np.repeat(h_coarse, 2)[:n_points]
            h = repair_h(h_full, n_points)
        else:
            h = h_coarse
        
        # Update best if better
        current_value = evaluate_h(h, dx)
        if current_value < best_value:
            best_value = current_value
            best_h = h.copy()
            
    # Final intensive gradient descent on full resolution
    def final_objective(h_flat):
        h_full = repair_h(h_flat, n_points)
        return evaluate_h(h_full, dx)
    
    def final_grad(h_flat):
        h_full = repair_h(h_flat, n_points)
        eps = 1e-6
        grad = np.zeros_like(h_flat)
        for i in range(n_points):
            h_plus = h_flat.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - final_objective(h_flat)) / eps
        return grad
    
    try:
        result = minimize(final_objective, best_h, method='L-BFGS-B',
                          bounds=[(0, 1)] * n_points,
                          jac=final_grad, options={'maxiter': 500})
        if result.success:
            h = result.x
    except:
        pass
        
    # Final consistency check and refinement
    h = repair_h(h, n_points)
    final_value = evaluate_h(h, dx)
    
    # If we improved, return the better result
    if final_value < best_value:
        best_value = final_value
        best_h = h.copy()
        
    return (best_h.tolist(), best_value, n_points)
