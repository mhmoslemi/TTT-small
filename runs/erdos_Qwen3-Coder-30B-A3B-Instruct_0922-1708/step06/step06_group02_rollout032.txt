import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def interpolate_h(h_old, n_new):
    """Interpolate h from old resolution to new resolution."""
    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

    start_time = time.time()
    
    # Initial resolution and parameters
    resolutions = [10, 20, 40, 100]
    population_size = 50
    elite_size = 5
    generations_per_stage = 50
    mutation_std_base = 0.1
    
    # Initialize coarsest resolution
    n_points = resolutions[0]
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init.copy()]
    else:
        population = []
    
    # Fill population with random solutions
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = None
    best_value = float('inf')
    
    # Evaluate initial population
    for i, h in enumerate(population):
        value = evaluate_h(h, dx)
        if value < best_value:
            best_value = value
            best_h = h.copy()
            
    # Evolutionary stages
    for res_index, n_points in enumerate(resolutions):
        if time.time() - start_time > budget_s - 10:
            break
        dx = 2.0 / n_points
        
        if res_index > 0:
            # Refine previous best solutions
            prev_best_h = best_h.copy()
            new_population = [prev_best_h.copy()]
            new_population.extend([interpolate_h(prev_best_h, n_points) for _ in range(elite_size - 1)])
            # Add new random candidates
            for _ in range(population_size - elite_size):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                new_population.append(h)
            population = new_population
        
        # Mutation scale decay based on resolution
        mutation_std = mutation_std_base / np.sqrt(n_points)
        
        # Optimization loop for this resolution
        for gen in range(generations_per_stage):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Evaluate all individuals
            values = []
            for h in population:
                value = evaluate_h(h, dx)
                values.append(value)
                if value < best_value:
                    best_value = value
                    best_h = h.copy()
            
            # Sort by fitness
            sorted_indices = np.argsort(values)
            elite_indices = sorted_indices[:elite_size]
            elite = [population[i] for i in elite_indices]
            
            # Create new population through mutation and crossover
            new_population = elite.copy()
            
            # Crossover and mutation
            while len(new_population) < population_size:
                # Select parents
                parent1_idx = np.random.choice(elite_indices)
                parent2_idx = np.random.choice(elite_indices)
                parent1 = population[parent1_idx]
                parent2 = population[parent2_idx]
                
                # Crossover
                child = np.zeros(n_points)
                crossover_prob = 0.3
                for i in range(n_points):
                    if np.random.rand() < crossover_prob:
                        child[i] = parent2[i]
                    else:
                        child[i] = parent1[i]
                
                # Mutation
                noise = np.random.normal(0, mutation_std, n_points)
                child = child + noise
                child = repair_h(child, n_points)
                new_population.append(child)
                
            population = new_population
    
    # Final gradient descent on best solution
    if best_h is not None and time.time() - start_time < budget_s - 5:
        h_refined = best_h.copy()
        current_value = best_value
        
        # Try to refine with gradient descent
        lr = 0.001
        for i in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            eps = 1e-6
            grad = np.zeros(n_points)
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
                
            h_refined = h_refined - lr * grad
            h_refined = repair_h(h_refined, n_points)
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            current_value = np.max(corr) * dx
            
            if current_value < best_value:
                best_value = current_value
                best_h = h_refined.copy()
    
    return (best_h.tolist(), best_value, len(best_h))
