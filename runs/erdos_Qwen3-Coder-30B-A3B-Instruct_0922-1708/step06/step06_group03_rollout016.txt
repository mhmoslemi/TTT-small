import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of C5 with respect to h."""
    n_points = len(h)
    grad = np.zeros(n_points)
    
    # For each position j, compute contribution to gradient
    for j in range(n_points):
        # Only the term where k = k_max contributes to the gradient
        # dC5/dh_j = dx * sum_k (indicator(k == k_max) * (1 - h[(j-k) mod n_points]))
        # This simplifies to: dx * (1 - h[(j - k_max) mod n_points])
        k = k_max
        idx = (j - k) % n_points
        grad[j] = dx * (1 - h[idx])
    
    return grad

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    learning_rate = 1e-4
    perturb_magnitude = 0.05
    recent_improvements = 0
    stagnation_count = 0
    
    # Main optimization loop
    for iteration in range(1000000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap values to find maximum
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap_value = corr[k_max]
        
        # Compute gradient
        grad = compute_gradient(current_h, dx, k_max)
        
        # Update h using gradient descent
        new_h = current_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject the update
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            recent_improvements = 0
            stagnation_count = 0
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            recent_improvements += 1
            stagnation_count += 1
            
        # Apply random perturbations periodically and when stuck
        if stagnation_count > 500 or (recent_improvements > 100 and np.random.rand() < 0.05):
            perturb_indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
            for idx in perturb_indices:
                perturb = np.random.uniform(-perturb_magnitude, perturb_magnitude)
                new_h[idx] += perturb
            new_h = repair_h(new_h, n_points)
            new_c5 = evaluate_h(new_h, dx)
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                recent_improvements = 0
                stagnation_count = 0

                    best_c5 = current_c5
        
        # Adaptive learning rate and perturbation magnitude
        if iteration % 100 == 0:
            learning_rate *= 0.99
        if iteration % 500 == 0:
            perturb_magnitude *= 0.999

        if recent_improvements > 500:

    return (best_h.tolist(), best_c5, n_points)
