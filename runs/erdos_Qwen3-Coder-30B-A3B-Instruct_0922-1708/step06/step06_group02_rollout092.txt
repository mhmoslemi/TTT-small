import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def crossover(parent1, parent2):
    """Create a child from two parents by averaging."""
    child = (parent1 + parent2) / 2.0
    return child

def mutate(h, mutation_rate=0.1):
    """Mutate h by adding Gaussian noise."""
    mask = h > 1e-6  # Only mutate significant values
    noise = np.random.normal(0, 0.05, len(h))
    mutated = h + noise * mutation_rate
    mutated = np.clip(mutated, 0, 1)
    return mutated

def initialize_population(n_points, population_size, initial_h):
    """Initialize a population with feasible solutions."""
    population = []
    population.append(initial_h)
    for _ in range(population_size - 1):
        # Start with a random feasible solution
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    return population

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5  # Default initialization
    h_init = repair_h(h_init, n_points)
    
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    elite_size = int(population_size * 0.2)
    
    # Initialize population
    population = initialize_population(n_points, population_size, h_init)
    
    best_h = None
    best_value = float('inf')
    
    # Evaluate initial population
    fitness_scores = []
    for h in population:
        value = evaluate_h(h, dx)
        fitness_scores.append(value)
        if value < best_value:
            best_value = value
            best_h = h.copy()
    
    # Main GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        elite_indices = sorted_indices[:elite_size]
        population = [population[i] for i in elite_indices]
        fitness_scores = [fitness_scores[i] for i in elite_indices]
        
        # If no improvement for a while, Reduce mutation rate
        if gen > 50 and gen % 10 == 0:
            mutation_rate = max(mutation_rate * 0.95, 0.01)
            
        # Generate offspring
        new_population = population.copy()
        while len(new_population) < population_size:
            # Selection with tournament
            parent1 = population[np.random.choice(elite_size)]
            parent2 = population[np.random.choice(elite_size)]
            
            # Crossover
            child = crossover(parent1, parent2)
            # Mutate
            child = mutate(child, mutation_rate)
            child = repair_h(child, n_points)
            
            new_population.append(child)
            
        # Replace population
        population = new_population[:population_size]
        
        # Evaluate and update best
        fitness_scores = []
        for h in population:
            value = evaluate_h(h, dx)
            fitness_scores.append(value)
            if value < best_value:
                best_value = value
                best_h = h.copy()
                
        # Gradient refinement on best
        if gen % 20 == 0 and best_h is not None:
            h_refined = best_h.copy()
            current_value = best_value
            
            # Simple gradient descent
            for _ in range(20):
                if time.time() - start_time > budget_s - 5:
                    break
                eps = 1e-5
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_value) / eps
                    
                h_refined = h_refined - 0.0005 * grad
                h_refined = repair_h(h_refined, n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                current_value = np.max(corr) * dx
                
                if current_value < best_value:
                    best_value = current_value
                    best_h = h_refined.copy()
    
    return (best_h.tolist(), best_value, n_points)
