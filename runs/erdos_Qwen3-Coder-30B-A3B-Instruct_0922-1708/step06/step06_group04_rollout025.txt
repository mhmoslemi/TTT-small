import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # CVXPY optimization
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2,
        t >= 0
    ]
    
    # Compute all valid overlaps
    max_lag = n_points - 1
    for lag in range(-max_lag, max_lag + 1):
        # Construct the overlap expression
        if lag >= 0:
            # h[i] * (1 - h[i - lag]) for valid i
            i_indices = np.arange(lag, n_points)
            j_indices = np.arange(0, n_points - lag)
        else:
            # h[i] * (1 - h[i - lag]) for valid i
            i_indices = np.arange(0, n_points + lag)
            j_indices = np.arange(-lag, n_points)
        
        if len(i_indices) > 0:
            overlap_expr = cp.sum(cp.multiply(h_var[i_indices], 1 - h_var[j_indices])) * dx
            constraints.append(overlap_expr <= t)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
        else:
            h_opt = h_init
    except:
        h_opt = h_init
    
    # Refine with local search if needed
    best_h = h_opt.copy()
    best_value = evaluate_h(best_h, dx)
    
    for iter_count in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Local gradient-based refinement
        eps = 1e-6
        grad = np.zeros(n_points)
        current_c5 = best_value
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        # Update using gradient
        best_h = best_h - 0.001 * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        new_c5 = evaluate_h(best_h, dx)
        
        if new_c5 < best_value:
            best_value = new_c5
    
    return (best_h.tolist(), best_value, n_points)
