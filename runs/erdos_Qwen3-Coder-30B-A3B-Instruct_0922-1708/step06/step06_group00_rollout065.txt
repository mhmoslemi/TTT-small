import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Use CVXPY for convex optimization
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Barrier method parameters
    mu = 1.0
    mu_factor = 1.1
    max_iter = 500
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Create and solve the convex optimization problem
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Add barrier term
        barrier = -mu * cp.sum(cp.log(h_var) + cp.log(1 - h_var))
        objective = cp.Minimize(t + barrier)
        
        # Add overlap constraints
        for k in range(n_points):
            # Compute overlap for shift k
            if k == 0:
                overlap = cp.sum(cp.multiply(h_var, 1 - h_var))
            else:
                overlap = cp.sum(cp.multiply(h_var[:-k], 1 - h_var[k:]))
            constraints.append(t >= overlap * dx)
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new_sum = np.sum(h_new)
                if abs(h_new_sum - n_points / 2) > 1e-8:
                    h_new = h_new * (n_points / 2) / h_new_sum
                h = h_new
                
                # Recompute C5
                corr = correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                else:
                    # If no improvement, reduce mu to relax constraints
                    mu *= 0.9
        except:
            pass
        
        # Increase mu for stricter barrier
        mu *= mu_factor
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
