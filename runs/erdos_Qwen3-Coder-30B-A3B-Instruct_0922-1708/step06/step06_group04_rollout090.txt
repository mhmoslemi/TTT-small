import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    else:
        # Create a smooth initialization that respects constraints
        h_init = np.ones(n_points) * 0.5
        # Add slight variations to reduce symmetry
        h_init += np.random.normal(0, 0.05, n_points)
    
    # Ensure sum = n_points / 2
    h_init = repair_h(h_init, n_points)
    
    # Begin optimization with gradient descent
    h = h_init.copy()
    current_c5 = evaluate_h(h, dx)
    best_h = h.copy()

        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
