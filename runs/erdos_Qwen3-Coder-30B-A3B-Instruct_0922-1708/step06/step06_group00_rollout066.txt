import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Barrier method parameters
    mu = 1.0
    max_iter = 1000
    barrier_factor = 1.1
    tolerance = 1e-6
    
    # Store best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Variables for optimization
    h_var = cp.Variable(n_points)
    t_var = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t_var)
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # Add overlap constraints
    for k in range(n_points):
        # Compute overlap at shift k
        overlap_expr = 0
        for i in range(n_points - k):
            overlap_expr += h_var[i] * (1 - h_var[i+k])
        overlap_expr = overlap_expr * dx
        constraints.append(t_var >= overlap_expr)
    
    # Create problem
    problem = cp.Problem(objective, constraints)
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Update barrier parameter
        mu *= barrier_factor
        
        # Modify constraints to include barrier function
        try:
            # Solve the problem
            problem.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, reltol=1e-6, abstol=1e-6)
            
            if problem.status == cp.OPTIMAL or problem.status == cp.OPTIMAL_INACCURATE:
                h_new = h_var.value
                c5 = t_var.value
                
                # Update best solution if it's better
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                
                # Update h
                h = h_new.copy()
                
                # Project to feasible region
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if abs(h_sum - n_points/2) > 1e-8:
                    h = h * (n_points/2) / h_sum
                    
        except Exception as e:
            # Fallback to gradient-based update
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Simple gradient-like update
            h += 0.0001 * (0.5 - h)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points/2) / h_sum
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
