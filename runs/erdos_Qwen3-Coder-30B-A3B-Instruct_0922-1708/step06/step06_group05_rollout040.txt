import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    pop_size = 50
    F = 1.0
    CR = 0.9
    max_generations = 1000
    stagnation_limit = 100
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        population.append(repair_h(h, n_points))
    else:
        population.append(np.ones(n_points) * 0.5)
    
    # Fill rest of population
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    
    # Evolution loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        new_population = []
        fitnesses = []
        
        # Evaluate current population
        for h in population:
            c5 = evaluate_h(h, 2.0 / n_points)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Sort and keep top half
        sorted_indices = np.argsort(fitnesses)
        top_half = sorted_indices[:pop_size//2]
        new_population = [population[i] for i in top_half]
        
        # Generate new individuals using DE
        while len(new_population) < pop_size:
            # Select three distinct random individuals
            a, b, c = np.random.choice(len(population), 3, replace=False)
            
            # Mutation
            v = population[a] + F * (population[b] - population[c])
            v = np.clip(v, 0, 1)
            v = repair_h(v, n_points)
            
            # Crossover
            trial = []
            for i in range(n_points):
                if np.random.rand() < CR:
                    trial.append(v[i])
                else:
                    trial.append(population[a][i])
            
            new_population.append(np.array(trial))
        
        population = new_population
        
        # Adaptive parameter adjustment
        if stagnation_count > stagnation_limit:
            F = min(1.5, F + 0.1)
            stagnation_count = 0
            
        # Check for final refinement
        if generation == 300 and best_h is not None:
            # Refine at higher resolution
            n_points = 200
            dx = 2.0 / n_points
            # Interpolate best solution
            x_orig = np.linspace(0, 2, 100)
            x_target = np.linspace(0, 2, 200)
            best_h_200 = np.interp(x_target, x_orig, best_h)
            best_h_200 = repair_h(best_h_200, n_points)
            best_c5 = evaluate_h(best_h_200, dx)
            best_h = best_h_200
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_h(best_h, 2.0 / n_points)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
