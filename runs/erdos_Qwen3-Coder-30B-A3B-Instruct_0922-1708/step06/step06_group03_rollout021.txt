import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Gradient descent with adaptive perturbation
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    last_improvement = 0
    iteration = 0
    max_no_improvement = 200
    
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        
        # Compute overlap values and find the maximum
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient
        grad = np.zeros(n_points)
        # Correct way to compute gradient w.r.t to h[j]
        for j in range(n_points):
            grad[j] = 0
            for k in range(len(corr)):
                # The contribution to overlap at lag k from h[j]
                # comes from h[j] * (1 - h[(j-k)%n_points])
                idx = (j - k) % n_points
                grad[j] += (1 - current_h[idx]) * (1 if k == k_max else 0)
        grad *= 2 * dx / n_points
        
        # Apply gradient descent
        h_candidate = current_h - learning_rate * grad
        
        # Add structured perturbations to escape local minima
        if iteration % 100 == 0:
            # Identify indices where gradient is small 
            indices = np.where(np.abs(grad) < 0.01)[0]
            # Add perturbations to these indices
            for idx in indices:
                if np.random.rand() < 0.5:  # 50% chance to perturb
                    perturbation = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
                    h_candidate[idx] += perturbation
        
        # Repair candidate
        h_candidate = repair_h(h_candidate, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_h(h_candidate, dx)
        
        # Accept the move if better or with some probability
        if candidate_c5 < current_c5 or np.random.rand() < 0.1:
            current_h = h_candidate
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                last_improvement = 0
            else:
                last_improvement += 1
        else:
            last_improvement += 1
            
        # Adjust hyperparameters
        if iteration % 100 == 0:
            learning_rate *= 0.99
        if iteration % 500 == 0:
            perturbation_magnitude *= 0.999
            
        # Check for early stopping
        if last_improvement > max_no_improvement:
            if iteration > 1000:
                break
    
    # Final refinement
    h_refined = best_h.copy()
    best_c5_refined = best_c5
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5_refined) / eps
        h_refined = h_refined - 0.0001 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_c5_refined = np.max(corr) * dx
        if best_c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = best_c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
