import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_gradient(h, dx):
    """Compute gradient of C5 with respect to h using finite differences."""
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-6
    
    base_value = evaluate_h(h, dx)
    
    for j in range(n):
        h_plus = h.copy()
        h_plus[j] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        grad[j] = (c5_plus - base_value) / eps
        
    return grad

def backtracking_line_search(h, grad, dx, alpha=0.5, beta=0.8, max_iter=10):
    """Perform backtracking line search."""
    t = 1.0
    for _ in range(max_iter):
        h_new = h - t * grad
        h_new = repair_h(h_new, len(h))
        new_value = evaluate_h(h_new, dx)
        if new_value <= evaluate_h(h, dx) - alpha * t * np.dot(grad, grad):
            return h_new, t
        t *= beta
    return h, t

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Multi-resolution optimization
    resolutions = [10, 20, 40, 100]
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Downsample for coarser resolution
        if res < n_points:
            h_coarse = np.array([np.mean(h[i*res//n_points:(i+1)*res//n_points]) 
                               for i in range(res)])
            h_coarse = h_coarse * (res / 2) / np.sum(h_coarse)
        else:
            h_coarse = h.copy()
            
        # Optimization for current resolution
        max_iter = 1000
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
                
            grad = compute_gradient(h_coarse, dx/res)
            h_coarse, _ = backtracking_line_search(h_coarse, grad, dx/res)
            
            # Interpolate back to finer resolution
            if res < n_points:
                h_new = np.zeros(n_points)
                for i in range(res):
                    start_idx = i * n_points // res
                    end_idx = (i + 1) * n_points // res
                    h_new[start_idx:end_idx] = h_coarse[i]
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_coarse = h_new.copy()
                
            current_value = evaluate_h(h_coarse, dx)
            if current_value < best_value:
                best_value = current_value
                best_h = h_coarse.copy()
    
    # Intensive local optimization
    h_final = best_h.copy()
    lr = 0.001
    no_improvement_count = 0
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 5:

        grad = compute_gradient(h_final, dx)
        h_new, _ = backtracking_line_search(h_final, grad, dx, alpha=0.5)
        
        current_value = evaluate_h(h_new, dx)
        if current_value < best_value:
            best_value = current_value
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        h_final = h_new.copy()
        
        # Adaptive learning rate

            lr *= 0.5
            no_improvement_count = 0
            
        # Convergence check
        if no_improvement_count > 100:
            break
            
    return (best_h.tolist(), best_value, n_points)
