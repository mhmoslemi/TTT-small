import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Use a two-stage approach for improvement
    # Stage 1: Convex relaxation with CVXPY
    try:
        h_vars = cp.Variable(n_points)
        # Relaxation: minimize sum of squared correlations
        # This is a proxy for minimizing max correlation
        # We'll use a weaker but more manageable constraint:
        # minimize max_k (h * (1-h))_k subject to bounds and integral
        # Try solve relaxation first
        h_prob = cp.Problem(
            cp.Minimize(cp.sum_squares(cp.conv((1 - h_vars), h_vars, mode='full'))),
            [
                h_vars >= 0,
                h_vars <= 1,
                cp.sum(h_vars) == n_points / 2
            ]
        )
        h_prob.solve(solver=cp.CVXOPT, verbose=False, atol=1e-6, rtol=1e-6)
        if h_prob.status == cp.OPTIMAL:
            h = h_vars.value
            h = repair_h(h, n_points)
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    except:
        pass  # Fall back to gradient-based method
    
    # Stage 2: Gradient descent with adaptive learning
    def compute_grad(h, dx):
        eps = 1e-6
        grad = np.zeros(n_points)
        h_eval = evaluate_h(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - h_eval) / eps
        return grad
    
    lr = 0.001
    max_iter = 50000
    stagnation_limit = 200
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        grad = compute_grad(h, dx)
        
        # Update
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        # Evaluate
        c5 = evaluate_h(h, dx)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive learning rate
        if stagnation_count > stagnation_limit:
            lr *= 0.9
            stagnation_count = 0
            
        # Occasionally increase LR to escape local minima
        if np.random.rand() < 0.01:
            lr = np.minimum(lr * 1.1, 0.01)
    
    # Refinement with stable gradient descent
    lr = 0.0001
    for _ in range(1000):
        if time.time() - start_time > budget_s - 2:
            break
        grad = compute_grad(best_h, dx)
        best_h = best_h - lr * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
    return (best_h.tolist(), best_c5, n_points)
