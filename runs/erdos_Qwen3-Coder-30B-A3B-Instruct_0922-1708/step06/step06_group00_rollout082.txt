import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 50
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iterations):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap with all shifts
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the maximum overlap lag
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h) - 1)
        
        # Formulate the convex subproblem
        # Minimize t subject to overlap_k <= t
        # overlap_k = 1 - (2 / n_points) * sum(h_i * h_{i+k})
        
        # Create the quadratic program
        h_vars = cp.Variable(n_points)
        t = cp.Variable(1)
        
        # Objective
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = [cp.sum(h_vars) == n_points / 2]
        constraints += [0 <= h_vars, h_vars <= 1]
        
        # Add constraint for the worst shift
        # overlap_k = sum(h_i * h_{i+k}) = 1 - t * n_points / 2
        # So we need sum(h_i * h_{i+k}) <= 1 - t * n_points / 2
        
        # Build the quadratic term efficiently for shift k
        if k == 0:
            # Self-overlap
            quadratic_expr = cp.sum_squares(h_vars)
            constraints += [quadratic_expr <= n_points / 2 - t * n_points / 2]
        else:
            # Cross-correlation for arbitrary shift k
            # We need to be careful to handle boundary conditions
            # For a shift k, we correlate h with h shifted by k
            # Build effective indices
            if k > 0:
                # h shifted right (h_{i+k})
                # h[i] * h[i+k] for i in range(len(h) - k)
                qform = 0
                for i in range(len(h) - k):
                    qform += h_vars[i] * h_vars[i + k]
                for i in range(k):
                    qform += h_vars[i] * h_vars[len(h) - k + i]
                constraints += [qform <= n_points / 2 - t * n_points / 2]
            else:
                # h shifted left (h_{i+k})
                k_abs = -k
                qform = 0
                for i in range(len(h) - k_abs):
                    qform += h_vars[i + k_abs] * h_vars[i]
                for i in range(k_abs):
                    qform += h_vars[i + k_abs] * h_vars[len(h) - k_abs + i]
                constraints += [qform <= n_points / 2 - t * n_points / 2]
        
        # Solve subproblem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                if h_new is not None:
                    h = h_new
                    h = np.clip(h, 0, 1)
            else:
                # Fall back to gradient descent if CVXPY fails
                pass
        except:
            # Fallback to gradient descent if error in optimization
            pass
        
        # Refine with gradient descent
        learning_rate = 0.01
        step_size = 1e-5
        for _ in range(10):
            # Calculate current overlap
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Compute simple approx gradient for each point
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += step_size
                h_plus = np.clip(h_plus, 0, 1)
                h_sum = np.sum(h_plus)
                h_plus = h_plus * (n_points / 2) / h_sum
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / step_size
            
            # Update
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
