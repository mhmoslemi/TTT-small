import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def interpolate_to_resolution(h_coarse, n_points):
    """Interpolate coarse h to finer resolution."""
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_points)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine

def mutate(h, mutation_scale):
    """Add Gaussian noise to h."""
    noise = np.random.normal(0, mutation_scale, len(h))
    return h + noise

def selection(population, values, n_select):
    """Select top individuals."""
    indices = np.argsort(values)[:n_select]
    return [population[i] for i in indices]

def crossover(parent1, parent2, crossover_rate=0.3):
    """Create offspring via crossover."""
    offspring = parent1.copy()
    mask = np.random.rand(len(parent1)) < crossover_rate
    offspring[mask] = parent2[mask]
    return offspring

def evolve_population(population, values, n_points, mutation_scale):
    """Evolve a population."""
    # Selection
    n_select = max(5, len(population) // 4)
    selected = selection(population, values, n_select)
    
    # Create new population
    new_population = selected.copy()
    while len(new_population) < len(population):
        parent1 = selected[np.random.randint(len(selected))]
        parent2 = selected[np.random.randint(len(selected))]
        offspring = crossover(parent1, parent2)
        offspring = mutate(offspring, mutation_scale)
        offspring = repair_h(offspring, n_points)
        new_population.append(offspring)
    
    return new_population

    best_h = None
    best_value = float('inf')
    best_n_points = 10
    
    # Multi-resolution approach
    for res_idx, n_points in enumerate(resolutions):
        dx = 2.0 / n_points
        
        # Interpolate to finer resolution
        if n_points > len(h_coarse):
            h_fine = interpolate_to_resolution(h_coarse, n_points)

            h_fine = h_coarse[:n_points]
        
        # Population parameters
        population_size = 50
