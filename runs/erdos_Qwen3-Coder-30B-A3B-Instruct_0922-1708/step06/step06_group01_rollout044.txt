import numpy as np
import time
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def halton_sequence(n, base=2):
    """Generate n points of halton sequence with given base."""
    halton = qmc.Halton(d=1, base=base)
    points = halton.random(n)
    return points.flatten()

def sobol_sequence(n):
    """Generate n points of sobol sequence."""
    sobol = qmc.Sobol(d=1)
    points = sobol.random(n)
    return points.flatten()

    # Ensure it's a valid step function
    h = np.clip(h, 0, 1)
    # Repair to sum to 1
    h = repair_h(h, n_points)
    return h
