import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        diff = target_sum - current_sum
        # Distribute the difference
        h = h + diff / n_points
        h = np.clip(h, 0, 1)
        # Re-adjust total sum
        current_sum = np.sum(h)
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Optimization parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 100
    no_improvement = 0
    shift_grid_size = 100
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Get correlation values for different shifts
        corr = correlate(h, 1 - h, mode='full')
        shift_indices = np.arange(len(corr))
        actual_shifts = (shift_indices - (len(corr) - 1) // 2) * (2.0 / (len(corr) - 1))
        
        # Find the k that gives maximum overlap
        max_idx = np.argmax(corr)
        worst_k_idx = max_idx
        worst_k = actual_shifts[worst_k_idx]
        max_corr = corr[worst_k_idx]
        
        # Compute gradient with respect to h
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Approximate gradient using finite differences
            idx_plus = i + worst_k_idx - (len(corr) - 1) // 2
            idx_minus = i - worst_k_idx + (len(corr) - 1) // 2
            if 0 <= idx_plus < n_points:
                grad[i] += 1 - h[idx_plus]
            if 0 <= idx_minus < n_points:
                grad[i] += 1 - h[idx_minus]
        
        # Adjust search direction - we want to minimize the correlation
        grad = -grad
        
        # Take gradient step
        h_new = h + learning_rate * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new value
        new_value = evaluate_h(h_new, dx)
        
        # If improvement, update
        if new_value < best_value:
            h = h_new.copy()
            best_value = new_value
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Occasionally perturb to escape local minima
        if iter_count % 50 == 0 and no_improvement > 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + perturbation
            h = repair_h(h, n_points)
            new_value = evaluate_h(h, dx)
            if new_value < best_value:
                best_value = new_value

        # Reduce learning rate if stuck
        if no_improvement > patience:
            learning_rate *= 0.9
            no_improvement = 0

    return (best_h.tolist(), best_value, n_points)
