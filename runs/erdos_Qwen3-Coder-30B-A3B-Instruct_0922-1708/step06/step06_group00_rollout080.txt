import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 30
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Cutting plane method
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
        
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the worst shift
        max_lag_idx = np.argmax(corr)
        k_worst = max_lag_idx - (n_points - 1)
        
        # Convex optimization subproblem
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Objective
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = []
        constraints.append(cp.sum(h_var) == n_points / 2)
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # The key constraint for worst overlap
        # For shift k_worst, we have overlap_k = sum h_i * (1 - h_{i+k})
        # This expands to: overlap_k = 1 - sum h_i * h_{i+k}
        # Minimize t such that t >= overlap_k
        
        # Build the constraint manually for efficiency
        if k_worst >= 0:
            # Overlap for this k
            term1 = cp.sum(cp.multiply(h_var[:n_points-k_worst], 
                                    1 - h_var[k_worst:]))
            constraints.append(term1 <= 1 - t)
        else:
            # k_worst < 0
            term1 = cp.sum(cp.multiply(h_var[-k_worst:], 
                                    1 - h_var[:n_points+k_worst]))
            constraints.append(term1 <= 1 - t)
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        # Try different solvers
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            # prob.solve(solver=cp.ECOS, verbose=False)
            # prob.solve(solver=cp.GLPK, verbose=False)
        except:
            # Fallback plans
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except:
                try:
                    prob.solve(solver=cp.GLPK, verbose=False)
                except:
                    # Solving with just bounds each iteration when sophisticated solvers fail
                    pass
        
        # Extract h from solution if valid
        if prob.status == cp.OPTIMAL:
            h_new = h_var.value
            if h_new is not None and np.all(h_new >= 0) and np.all(h_new <= 1)
                # rescale
                h = h_new * (n_points / 2) / np.sum(h_new)
        else:
            # Just use a step-sequence or random if we can't solve
            # Try to make a step towards a well spread solution
            h = np.clip(0.5 + 0.1 * np.sin(np.linspace(0, 2 * np.pi, n_points)), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            if np.isnan(h).any() or np.isinf(h).any():
                h = np.ones(n_points) * 0.5
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
