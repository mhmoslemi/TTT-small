import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Hyperparameters
    learning_rate = 5e-4
    temperature = 0.1
    decay_rate = 0.999
    dual_update_freq = 10
    stagnation_patience = 1000
    stagnation_count = 0
    
    # Dual variables initialization (for max overlap)
    dual_vars = np.ones(2 * n_points - 1)  # size equals number of lags
    dual_vars /= np.sum(dual_vars)  # normalize to probabilities
    
    # Main optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        
        # Compute all overlaps and their corresponding lags
        corr = correlate(current_h, 1 - current_h, mode='full')
        overlaps = corr * dx
        indices = np.arange(len(overlaps))
        
        # Update dual variables (variance reduction approach)
        if iter_count % dual_update_freq == 0:
            # Compute current weights as exponential of negative overlap divided by temperature
            exp_weights = np.exp(-overlaps / temperature)
            dual_vars = exp_weights / np.sum(exp_weights)
            
            # Decrease temperature over time (simulated annealing-like)
            temperature *= decay_rate
        
        # Compute gradient of Lagrangian L = sum(dual_vars * overlaps) + lambda * (sum(h) - target)
        # Gradient w.r.t. h[i] = sum_k dual_vars[k] * (1 - h[i-k]) if k <= i else 0
        gradient = np.zeros(n_points)
        for i in range(n_points):
            for k in range(len(overlaps)):
                lag = k - n_points + 1
                if 0 <= i - lag < n_points:
                    gradient[i] += dual_vars[k] * (1 - current_h[i - lag])
        
        # Update h with gradient descent
        current_h = current_h - learning_rate * gradient
        current_h = repair_h(current_h, n_points)
        
        # Evaluate new current solution
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adaptive learning rate decay
        if stagnation_count > stagnation_patience:
            learning_rate *= 0.9
            stagnation_count = 0
            if learning_rate < 1e-7:
                break
        
        # Early exit condition
        if stagnation_count > stagnation_patience * 2:
            break
    
    # Ensure final result is valid
    final_h = repair_h(best_h, n_points)
    final_c5 = evaluate_h(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
