import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = (h_init > 0.5).astype(float)
    else:
        h = np.ones(n_points) * 0.5
        h = (h > 0.5).astype(float)
    
    # Ensure the initial sum is correct
    h_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(h_sum - target_sum) > 1e-8:
        # Adjust the binary vector to get the exact sum
        diff = int(target_sum - h_sum)
        if diff > 0:
            # Need to turn some 0s to 1s
            zero_indices = np.where(h == 0)[0]
            for i in range(diff):
                if i < len(zero_indices):
                    h[zero_indices[i]] = 1
        elif diff < 0:
            # Need to turn some 1s to 0s
            one_indices = np.where(h == 1)[0]
            for i in range(-diff):
                if i < len(one_indices):
                    h[one_indices[i]] = 0
    
    # Check solution is valid
    assert abs(np.sum(h) - target_sum) < 1e-8, "Invalid sum"
    assert all(x in [0, 1] for x in h), "Not binary"
    
    # Solve using mixed-integer programming
    # Define the MIP problem
    h_vars = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_vars) == target_sum]
    
    # For each shift k, we compute the overlap constraint
    # The valid indices for shift k are i such that 0 <= i + k < n_points
    # i ranges from max(0, -k) to min(n_points - 1, n_points - 1 - k)
    for k in range(-n_points + 1, n_points):
        # Get valid i indices for this k
        # i + k must be within [0, n_points - 1]
        # so i must be within [max(0, -k), min(n_points - 1, n_points - 1 - k)]
        i_min = max(0, -k)
        i_max = min(n_points - 1, n_points - 1 - k)
        if i_min <= i_max:
            # Sum of h_vars[i] * (1 - h_vars[i + k]) over valid i
            sum_expr = 0
            for i in range(i_min, i_max + 1):
                sum_expr += h_vars[i] * (1 - h_vars[i + k])
            constraints.append(sum_expr <= t * target_sum)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Use CBC solver with time limit
    solve_time = budget_s - 10
    try:
        prob.solve(solver=cp.CBC, time_limit=solve_time, presolve=True, 
                   verbose=False, maximum_seconds=solve_time)
    except Exception as e:
        print(f"MIP solver error: {e}")
    
    # Extract best integer solution (if available)
    h_best = np.array([round(x.value) for x in h_vars]).astype(float)
    
    # Check if we got a better solution
    corr = correlate(h_best, 1 - h_best, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Return result
    return (h_best.tolist(), final_c5, n_points)
