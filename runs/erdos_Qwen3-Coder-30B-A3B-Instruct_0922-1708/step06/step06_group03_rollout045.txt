import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max, n_points):
    """Compute gradient of C5 w.r.t. h at k_max."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        # For each i, we look at the contribution to overlap at k_max
        j = (i - k_max) % n_points
        grad[i] = -dx * (1 - h[j])
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Parameters
    lr = 1e-4
    pert_mag = 0.05
    decay_lr = 0.99
    decay_pert = 0.999
    patience = 200
    patience_counter = 0
    last_improvement = 0
    
    for iter_num in range(100000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap values
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient
        grad = compute_gradient(current_h, dx, k_max, n_points)
        
        # Apply gradient descent step
        new_h = current_h - lr * grad
        
        # Apply structured perturbations
        if iter_num % 100 == 0 and iter_num > 0:
            # Find indices with small gradient magnitude
            grad_magnitude = np.abs(grad)
            low_grad_indices = np.where(grad_magnitude < np.percentile(grad_magnitude, 30))[0]
            if len(low_grad_indices) > 0:
                perturbation = np.zeros(n_points)
                for idx in np.random.choice(low_grad_indices, min(10, len(low_grad_indices)), replace=False):
                    perturbation[idx] = np.random.uniform(-pert_mag, pert_mag)
                new_h += perturbation
        
        # Repair the new solution
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                last_improvement = iter_num
                patience_counter = 0
            else:
                patience_counter += 1
        else:
            patience_counter += 1
            
        # Adapt learning rate and perturbation magnitude
        if iter_num % 100 == 0:
            lr *= decay_lr
            pert_mag *= decay_pert
            
        # Check patience
        if patience_counter > patience:
            lr = max(lr * 0.5, 1e-8)
            pert_mag = max(pert_mag * 0.9, 1e-6)
            patience_counter = 0
            
    return (best_h.tolist(), best_c5, n_points)
