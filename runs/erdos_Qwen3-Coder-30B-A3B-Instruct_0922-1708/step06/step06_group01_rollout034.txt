import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1, sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fw_step(h, grad, n_points, dx):
    """Frank-Wolfe linear subproblem solution"""
    # Find direction d that minimizes grad*d subject to constraints
    # d in [0,1]^n, sum(d) = 0
    # choose d to minimize grad*d = sum(grad[i] * d[i])
    # This is equivalent to finding the direction with maximum negative gradient
    
    # Assuming convexity, solution is at vertices of feasible set.
    # We want d such that d[i] = 1 for i where grad[i] < 0,
    # and d[i] = 0 for i where grad[i] > 0, but sum(d)=0 constraint.
    # So consider lazy selection.
    # Simple approach: use gradient sign
    d = np.zeros_like(h)
    for i in range(n_points):
        if grad[i] < 0:
            d[i] = 1
        elif grad[i] > 0:
            d[i] = -1
    # Enforce sum(d) = 0
    d_sum = np.sum(d)
    if abs(d_sum) > 1e-10:
        # Re-scale: we want d sum to be 0.
        # Adjust scale so that the constraint is met
        d = d - d_sum / n_points
    # Clip to [0,1] and renormalize to sum 0
    d = np.clip(d, -1, 1)
    d = d - np.mean(d)  # Ensure mean is 0 to make sum zero (approximately)
    return d

def line_search(h, d, grad, n_points, dx, max_eta=1.0):
    """Perform backtracking line search to find step size"""
    eta = max_eta
    current_c5 = evaluate_h(h, dx)
    
    for _ in range(20):
        h_new = h + eta * d
        h_new = project_to_feasible(h_new, n_points)
        new_c5 = evaluate_h(h_new, dx)
        if new_c5 < current_c5:
            return eta, h_new
        eta *= 0.5
    return eta, h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_to_feasible(h_init, n_points)
    else:
        # Initial structure based on heuristic
        h_init = np.zeros(n_points)
        # Simple linear increase pattern
        for i in range(n_points):
            h_init[i] = min(i / (n_points/2), 1.0)
        h_init = project_to_feasible(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Frank-Wolfe parameters
    max_iter = 10000
    patience = 1000
    last_improvement = 0
    epsilon = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute approximate gradient using finite differences
        grad = np.zeros(n_points)
        h_plus = h_current.copy()
        for j in range(n_points):
            h_plus[j] += epsilon
            h_plus = project_to_feasible(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_current) / epsilon
            h_plus[j] -= epsilon
    
        # Frank-Wolfe step
        d = fw_step(h_current, grad, n_points, dx)
        
        # Line search
        eta, h_new = line_search(h_current, d, grad, n_points, dx)
        
        # Update
        h_current = h_new
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        if i - last_improvement > patience:
            break
            
    # Final evaluation of best h
    final_c5 = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), final_c5, n_points)
