import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, 2.0 / n_points)
    
    # Annealing parameters
    T0 = 1.0
    alpha = 0.99
    T = T0
    lr = 0.01
    step = 0
    max_iter_per_res = 1000
    
    # Multi-scale approach
    scales = [20, 40, 80, 200]
    current_scale_idx = 0
    current_n = scales[current_scale_idx]
    h_current = h.copy()
    
    # Initialize at first scale
    if current_n != n_points:
        x_orig = np.linspace(0, 2, n_points)
        x_target = np.linspace(0, 2, current_n)
        h_current = np.interp(x_target, x_orig, h_current)
    h_current = repair_h(h_current, current_n)
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 5:
        # Simulate annealing step
        h_new = h_current.copy()
        noise = np.random.normal(0, T, current_n)
        h_new += noise
        
        # Repair
        h_new = repair_h(h_new, current_n)
        
        # Evaluate
        c5_new = evaluate_h(h_new, 2.0 / current_n)

        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):

    return (best_h.tolist(), best_c5, len(best_h))
