import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or create binary approximation
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Convert to binary step function with exactly n_points/2 ones
    h_binary = (h > 0.5).astype(float)
    ones_count = np.sum(h_binary)
    if ones_count != n_points / 2:
        # Adjust to get exactly n_points/2 ones
        indices = np.where(h_binary == 1)[0]
        if ones_count > n_points / 2:
            # Need to turn some 1s to 0s
            to_flip = int(ones_count - n_points / 2)
            flip_indices = np.random.choice(indices, to_flip, replace=False)
            h_binary[flip_indices] = 0
        else:
            # Need to turn some 0s to 1s
            zeros_indices = np.where(h_binary == 0)[0]
            to_flip = int(n_points / 2 - ones_count)
            flip_indices = np.random.choice(zeros_indices, to_flip, replace=False)
            h_binary[flip_indices] = 1
    
    # Tabu search parameters
    max_iter = 50000
    tabu_size = 50
    tabu_list = []
    temperature = 0.1
    cooling_rate = 0.999
    
    # Initial evaluation
    corr = correlate(h_binary, 1 - h_binary, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h_binary.copy()
    best_c5 = current_c5
    
    # Simulated annealing loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping one bit randomly
        candidate_h = h_binary.copy()
        flip_idx = np.random.randint(0, n_points)
        candidate_h[flip_idx] = 1 - candidate_h[flip_idx]
        
        # Maintain exactly n_points/2 ones
        ones_count = np.sum(candidate_h)
        if ones_count != n_points / 2:
            if ones_count > n_points / 2:
                # Flip a 1 back to 0
                ones_idx = np.where(candidate_h == 1)[0]
                flip_back = np.random.choice(ones_idx)
                candidate_h[flip_back] = 0
            else:
                # Flip a 0 back to 1
                zeros_idx = np.where(candidate_h == 0)[0]
                flip_back = np.random.choice(zeros_idx)
                candidate_h[flip_back] = 1
        
        # Check if neighbor is in tabu list
        candidate_key = tuple(candidate_h.astype(int))
        if candidate_key in tabu_list:
            # With low probability, accept anyway
            if np.random.rand() > 0.9:
                pass
            else:
                continue
        
        # Compute new overlap
        corr = correlate(candidate_h, 1 - candidate_h, mode='full')
        candidate_c5 = np.max(corr) * dx
        
        # Accept or reject
        if candidate_c5 < current_c5:
            current_c5 = candidate_c5
            h_binary = candidate_h.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = candidate_h.copy()
        else:
            # Accept with probability based on temperature
            delta = candidate_c5 - current_c5
            if np.random.rand() < np.exp(-delta / temperature):
                current_c5 = candidate_c5
                h_binary = candidate_h.copy()
        
        # Update temperature
        temperature *= cooling_rate
        
        # Update tabu list
        tabu_list.append(candidate_key)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
    
    # Final validation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Ensure we return exactly n_points values
    if len(best_h) != n_points:
        best_h = best_h[:n_points] if len(best_h) > n_points else np.pad(best_h, (0, n_points - len(best_h)))
    
    return (best_h.tolist(), final_c5, n_points)
