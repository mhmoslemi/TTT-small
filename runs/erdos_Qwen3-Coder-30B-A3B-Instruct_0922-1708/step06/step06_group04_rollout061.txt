import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum=50.0):
    """Project h to feasible region: 0 <= h <= 1 and sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from available parent construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to 50
    h = project_h(h, 50.0)
    
    # Add small perturbation
    h += np.random.normal(0, 0.05, n_points)
    h = project_h(h, 50.0)
    
    # Optimization parameters
    learning_rate = 0.1
    lambda_reg = 0.01  # L1 regularization strength
    max_iter = 1000
    min_improvement = 1e-4
    patience = 100
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    last_improvement = 0
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient numerically
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, 50.0)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - best_c5) / eps
        
        # Add L1 regularization gradient
        reg_grad = lambda_reg * np.sign(h)
        grad_total = grad + reg_grad
        
        # Update with backtracking line search
        h_new = h - learning_rate * grad_total
        h_new = project_h(h_new, 50.0)
        
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject update with backtracking
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            last_improvement = iter_count
            learning_rate *= 0.95  # Decrease learning rate if improvement
        else:
            learning_rate *= 1.1  # Increase learning rate
            learning_rate = min(learning_rate, 1.0)
            
        # If no improvement, try restarting with more sparsity
        if iter_count - last_improvement > patience:
            # Perturb and restart
            h = h + np.random.normal(0, 0.02, n_points)
            h = project_h(h, 50.0)
            last_improvement = iter_count
            lambda_reg = min(lambda_reg * 1.2, 0.1)  # Increase regularization gradually
        
        # Ensure learning rate is in reasonable bounds
        learning_rate = np.clip(learning_rate, 1e-4, 1.0)
    
    return (best_h.tolist(), best_c5, n_points)
