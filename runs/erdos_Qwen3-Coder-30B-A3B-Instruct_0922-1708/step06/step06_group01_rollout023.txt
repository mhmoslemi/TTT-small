import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.linalg import qr

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute gradient using central differences."""
    n = len(h)
    grad = np.zeros(n)
    h_ref = h.copy()
    for i in range(n):
        h_plus = h_ref.copy()
        h_minus = h_ref.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = repair_h(h_plus, n)
        h_minus = repair_h(h_minus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps) * dx
    return grad

def trust_region_subproblem(H, g, delta, bounds):
    """Solve trust-region subproblem: min g^T d + 0.5 d^T H d s.t. ||d|| <= delta"""
    # Using scipy.optimize.minimize with bounds
    def objective(d):
        return g.dot(d) + 0.5 * d.dot(H.dot(d))
    
    def constraint(d):
        return delta - np.linalg.norm(d)
    
    # Initialize
    d0 = np.zeros(len(g))
    
    # Solve constrained optimization problem
    res = minimize(
        objective, d0, 
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraint},
        options={'disp': False, 'tol': 1e-8}
    )
    return res.x if res.success else d0

        h_init = repair_h(h_init, n_points)
    
    # Initialize trust region parameters
    trust_radius = 0.5
    trust_radius_max = 1.0
    trust_radius_min = 1e-6
    delta = trust_radius
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    bounds = [(0, 1) for _ in range(n_points)]

    perturb_freq = 100
    step_count = 0
    
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Periodic perturbation
        if i % perturb_freq == 0:
            h_current = h_current + np.random.normal(0, 0.01, n_points)

        grad = compute_gradient(h_current, dx)
