import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent optimization
    # Initialize dual variables
    lambda_k = np.zeros(2 * n_points - 1)
    step_size = 5e-4
    decay_factor = 0.99
    max_iter = 50000
    no_improve_count = 0
    max_no_improve = 1000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlations
        corr = correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Update dual variables
        lambda_k = np.maximum(0, lambda_k + step_size * (corr - max_val))
        
        # Compute gradient using dual variables
        grad_h = np.zeros(n_points)
        for k in range(2 * n_points - 1):
            if k < n_points:
                # Forward convolution component
                for i in range(n_points - k):
                    grad_h[i] += lambda_k[k] * (1 - current_h[i + k])
            else:
                # Backward convolution component
                for i in range(k - n_points + 1, n_points):
                    grad_h[i] += lambda_k[k] * (1 - current_h[i - (k - n_points + 1)])
        
        # Update h with projected gradient descent
        current_h = current_h - step_size * grad_h
        current_h = repair_h(current_h, n_points)
        
        # Update step size
        if i % 500 == 0:
            step_size *= decay_factor
            
        # Evaluate new C5
        new_c5 = evaluate_h(current_h, dx)
        
        # Update best if improved
        if new_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = new_c5
            no_improve_count = 0
            # Refine with local descent
            for _ in range(50):
                if time.time() - start_time > budget_s - 5:
                    break
                eps = 1e-6
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = current_h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - new_c5) / eps
                current_h = current_h - 0.0001 * grad
                current_h = repair_h(current_h, n_points)
                new_c5 = evaluate_h(current_h, dx)
                if new_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = new_c5
        
        else:
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                break
    
    # Final verification
    best_h = repair_h(best_h, n_points)
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
