import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Barrier method parameters
    mu = 1.0
    max_iter = 500
    convergence_threshold = 1e-6
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # CVXPY formulation
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add soft constraints for overlaps
        # For each valid lag k
        for k in range(n_points):
            # Compute the overlap for lag k
            lags = np.arange(k, min(k + n_points - k, n_points))
            if len(lags) > 0:
                overlap_expr = cp.sum(cp.multiply(h_var[:len(lags)], 1 - h_var[k:k+len(lags)])) * dx
                constraints.append(t >= overlap_expr)
        
        # Objective
        objective = cp.Minimize(t - mu * cp.sum(cp.entr(h_var) + cp.entr(1 - h_var)))
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        try:
            # Solver choice for efficiency
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h = np.maximum(0, np.minimum(1, h_var.value))
                h_sum = np.sum(h)
                target_sum = n_points / 2
                if abs(h_sum - target_sum) > 1e-8:
                    h = h * target_sum / h_sum
                
                # Update mu
                mu *= 1.1
            else:
                # Revert to previous if infeasible
                h = best_h.copy()
                mu *= 1.1
                
        except:
            h = best_h.copy()
            mu *= 1.1

    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
