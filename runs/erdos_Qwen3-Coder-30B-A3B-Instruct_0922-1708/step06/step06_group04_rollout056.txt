import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize swarm
    population_size = 100
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    # Initial solution from parent if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize particles with some from initial solution
    particles = [h_init.copy()]
    velocities = [np.random.uniform(-0.5, 0.5, n_points)]
    
    for i in range(1, population_size):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Evaluate initial particles
    fitness = [evaluate_h(h, dx) for h in particles]
    pbest = particles[:]
    pbest_fitness = fitness[:]
    
    # Global best
    best_idx = np.argmin(pbest_fitness)
    gbest = pbest[best_idx].copy()
    gbest_fitness = pbest_fitness[best_idx]
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    iter_count = 0
    stagnation_counter = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate fitness
            current_fitness = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = current_fitness
                
                # Update global best
                if current_fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = current_fitness
                    stagnation_counter = 0
            
        stagnation_counter += 1
        if stagnation_counter > 50:
            # Inject some random particles
            for _ in range(10):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles.append(h)
                velocities.append(np.random.uniform(-0.5, 0.5, n_points))
                fitness_val = evaluate_h(h, dx)
                pbest.append(h)
                pbest_fitness.append(fitness_val)
                if fitness_val < gbest_fitness:
                    gbest = h.copy()
                    gbest_fitness = fitness_val
                    
            # Keep only best population_size particles
            combined_fitness = pbest_fitness + [evaluate_h(h, dx) for h in particles[population_size:]]
            indices = np.argsort(combined_fitness)[:population_size]
            new_particles = []
            new_velocities = []
            new_pbest = []
            new_pbest_fitness = []
            
            for idx in indices:
                if idx < len(pbest):
                    new_particles.append(pbest[idx])
                    new_velocities.append(velocities[idx])
                    new_pbest.append(pbest[idx])
                    new_pbest_fitness.append(pbest_fitness[idx])
                else:
                    new_particles.append(particles[idx])
                    new_velocities.append(velocities[idx])
                    new_pbest.append(particles[idx])
                    new_pbest_fitness.append(combined_fitness[idx])
                    
            particles = new_particles
            velocities = new_velocities
            pbest = new_pbest
            pbest_fitness = new_pbest_fitness
            
            best_idx = np.argmin(pbest_fitness)
            gbest = pbest[best_idx].copy()
            gbest_fitness = pbest_fitness[best_idx]
            
            stagnation_counter = 0
            
        iter_count += 1
    
    # Gradient refinement on best solution
    h_refined = gbest.copy()
    prev_c5 = gbest_fitness
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr) * dx
            grad[j] = (c5_plus - prev_c5) / eps
        
        h_refined = h_refined - 0.0001 * grad
        h_refined = repair_h(h_refined, n_points)
        prev_c5 = evaluate_h(h_refined, dx)
        
        if prev_c5 < gbest_fitness:
            gbest = h_refined.copy()
            gbest_fitness = prev_c5
    
    return (gbest.tolist(), gbest_fitness, n_points)
