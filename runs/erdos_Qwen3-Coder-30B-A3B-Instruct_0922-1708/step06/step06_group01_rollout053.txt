import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints using quadratic programming."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Solve min ||x - x0||^2 subject to sum(x) = target_sum, 0 <= x <= 1
        # Using Lagrange multipliers and projection
        diff = target_sum - current_sum
        if diff > 0:
            # Need to increase values
            max_increase = np.min([1.0 - h, diff], axis=0)
            indices = np.argsort(max_increase)[::-1]
            cum_sum = 0
            for i in indices:
                if cum_sum + max_increase[i] > diff:
                    h[i] += diff - cum_sum
                    break
                else:
                    h[i] += max_increase[i]
                    cum_sum += max_increase[i]
        else:
            # Need to decrease values
            max_decrease = np.min([h, -diff], axis=0)
            indices = np.argsort(max_decrease)[::-1]
            cum_sum = 0
            for i in indices:
                if cum_sum + max_decrease[i] > -diff:
                    h[i] -= (-diff) - cum_sum
                    break
                else:
                    h[i] -= max_decrease[i]
                    cum_sum += max_decrease[i]
    return h

    dx = 2.0 / n_points

        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)

    # Gradient-based optimization with adaptive learning rate and perturbations
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    patience = 0
    patience_limit = 1000
    step_size = 0.01
    perturb_rate = 0.05
    
    max_iter = 20000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_current) / eps

        # Update using gradient descent
        h_new = h_current - step_size * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement or accept with small probability
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                patience = 0
            else:
                patience += 1
        else:
            # With probability, accept worse solution to escape local optima
            if np.random.rand() < 0.1:
                h_current = h_new
                c5_current = c5_new
                patience = 0
            else:
                patience += 1
                
        # Perturb some elements randomly
        num_perturb = max(1, int(perturb_rate * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)

            delta = np.random.uniform(-0.02, 0.02)
            h_current[idx] += delta

            step_size = min(0.1, step_size * 1.01)
        else:

        # Adaptive perturb rate
