import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize best solution
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Shift grid
    shift_grid = np.linspace(-2, 2, 100)
    
    # Optimization parameters
    learning_rate = 0.01
    max_iter = 5000
    no_improvement_limit = 500
    perturbation_interval = 50
    perturbation_scale = 0.001
    
    last_best = best_value
    no_improvement_count = 0
    iter_count = 0
    
    # Setup full correlation structure for speed
    full_corr_indices = np.arange(-n_points + 1, n_points)
    
    # Main optimization loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        # Find worst shift
        worst_idx = np.argmax(corr)
        worst_k = full_corr_indices[worst_idx]
        
        # Compute gradient for the specificworst_k
        grad = np.zeros(n_points)
        if worst_k >= 0:
            # For k >= 0: h[i] contributes to correlation at shift k as h[i] * (1 - h[i+k])
            for i in range(n_points - worst_k):
                grad[i] -= 1 - h[i + worst_k]
                if i < n_points - worst_k:
                    grad[i + worst_k] -= 1 - h[i]
        else:
            # For k < 0: h[i] contributes to correlation at shift k as h[i] * (1 - h[i+k])
            for i in range(-worst_k, n_points):
                grad[i] -= 1 - h[i + worst_k]
                if i - worst_k >= 0 and i - worst_k < n_points:
                    grad[i - worst_k] -= 1 - h[i]
        
        # Apply gradient step
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        
        # Evaluate current solution
        current_value = evaluate_h(h, dx)
        
        # Update best solution if improved
        if current_value < best_value:
            best_value = current_value
            best_h = h.copy()
            last_best = best_value
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Gradually decrease learning rate if no improvement
            if no_improvement_count > 100:
                learning_rate *= 0.95

        if iter_count % perturbation_interval == 0 and iter_count > 0:
            noise = np.random.normal(0, perturbation_scale, n_points)

        # Adaptively refine the grid

    final_value = evaluate_h(best_h, dx)
