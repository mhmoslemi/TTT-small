import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def interpolate_h(h_coarse, n_points):
    """Interpolate coarse h to target resolution."""
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_target = np.linspace(0, 2, n_points)
    h_target = np.interp(x_target, x_coarse, h_coarse)
    return h_target

def mutate_h(h, std):
    """Mutate h with Gaussian noise."""
    noise = np.random.normal(0, std, len(h))
    h_mut = h + noise
    return h_mut

def crossover_h(h1, h2, rate=0.3):
    """Crossover two h vectors."""
    mask = np.random.rand(len(h1)) < rate
    h_new = np.where(mask, h1, h2)
    return h_new

def optimize_stage(h_candidates, n_points, dx, generations=50):
    """Optimize a stage."""
    best_value = float('inf')
    best_h = None
    
    for _ in range(generations):
        # Evaluate population
        values = [evaluate_h(h, dx) for h in h_candidates]
        
        # Update best
        for h, v in zip(h_candidates, values):
            if v < best_value:
                best_value = v
                best_h = h.copy()
                
        # Selection (top 20%)
        indices = np.argsort(values)[:int(len(values) * 0.2)]
        selected = [h_candidates[i] for i in indices]
        
        # Create new population
        new_candidates = selected.copy()
        
        # Crossover and mutation
        while len(new_candidates) < len(h_candidates):
            h1, h2 = np.random.choice(selected, 2, replace=False)
            if np.random.rand() < 0.3:
                h_new = crossover_h(h1, h2)
            else:
                h_new = h1.copy()
            
            h_new = mutate_h(h_new, 0.1)
            h_new = repair_h(h_new, n_points)
            new_candidates.append(h_new)
            
        h_candidates = new_candidates
        
    return best_h, best_value

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population sizes
    initial_resolutions = [10, 20, 40, 100]
    max_iter = 200
    
    # Initialize coarse h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != 10:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, 10)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * 5.0 / np.sum(h_init)
    else:
        h_init = np.ones(10) * 0.5
    
    # Scale by target sum
    h_init = h_init * 5.0 / np.sum(h_init)
    
    # Multi-resolution optimization
    resolution = 10
    best_h = h_init.copy()
    best_value = evaluate_h(best_h, 2.0 / resolution)
    
    for i, res in enumerate(initial_resolutions):
        if time.time() - start_time > budget_s - 10:
            break
            
        dx = 2.0 / res
        
        # Interpolate to current resolution
        if res == 10:
            h_current = h_init.copy()
        else:
            h_current = interpolate_h(best_h, res)
            
        # Initialize population
        pop_size = 50
        if i == 0:
            h_candidates = [h_current.copy()]
            # Add some random ones
            for _ in range(pop_size - 1):
                h_rand = np.random.rand(res)
                h_rand = h_rand * (res / 2) / np.sum(h_rand)
                h_candidates.append(h_rand)
        else:
            # Use elites from previous stage
            h_candidates = [best_h.copy()]
            # Add random ones
            for _ in range(pop_size - 1):
                h_rand = np.random.rand(res)
                h_rand = h_rand * (res / 2) / np.sum(h_rand)
                h_candidates.append(h_rand)
                
        # Optimize this resolution
        if res < 100:
            h_opt, v_opt = optimize_stage(h_candidates, res, dx, 50)
        else:
            # Final stage - more detailed optimization
            h_opt, v_opt = optimize_stage(h_candidates, res, dx, 100)
            
        best_h = h_opt.copy()
        best_value = v_opt
        
        if i == len(initial_resolutions) - 1:
            break
            
    # Final gradient descent refinement for best_h at 100 points
    if len(best_h) != 100:
        best_h = interpolate_h(best_h, 100)
    
    dx = 2.0 / 100
    h_final = best_h.copy()
    current_value = best_value
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(100)
        for j in range(100):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, 100)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_value) / eps
            
        h_final = h_final - 0.0001 * grad
        h_final = repair_h(h_final, 100)
        corr = correlate(h_final, 1 - h_final, mode='full')
        current_value = np.max(corr) * dx
        
        if current_value < best_value:
            best_value = current_value
            best_h = h_final.copy()
    
    return (best_h.tolist(), best_value, 100)
