import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of C5 with respect to h."""
    n_points = len(h)
    grad = np.zeros(n_points)
    for i in range(n_points):
        # The gradient contribution from the maximum overlap term
        j = (i - k_max) % n_points
        grad[i] = (1 - h[j])
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    current_h = h_init.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_h(current_h, dx)
    current_c5 = best_c5
    
    # Optimization parameters
    learning_rate = 1e-4
    decay_rate = 0.99
    perturb_rate = 0.05
    perturb_decay = 0.999
    stagnation_count = 0
    max_stagnation = 1000
    gradient_update_freq = 10
    
    for i in range(1000000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap for all lags k
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        current_c5 = max_overlap * dx

        # Compute gradient
        if i % gradient_update_freq == 0:
            grad = compute_gradient(current_h, dx, k_max)
            
            # Gradient descent step
            current_h = current_h - learning_rate * grad
            current_h = repair_h(current_h, n_points)
            
            # Update learning rate
            learning_rate *= decay_rate
            
            # Check improvement
            new_c5 = evaluate_h(current_h, dx)
            if new_c5 < current_c5:
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
            
            # Apply random perturbations periodically to escape local minima
            if stagnation_count > max_stagnation or i % 500 == 0:
                h_perturbed = current_h.copy()
                perturb_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                for idx in perturb_indices:
                    delta = np.random.uniform(-perturb_rate, perturb_rate)
                    h_perturbed[idx] += delta
                h_perturbed = repair_h(h_perturbed, n_points)
                new_c5 = evaluate_h(h_perturbed, dx)
                if new_c5 < current_c5:
                    current_h = h_perturbed
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
                    stagnation_count = 0
                
                # Reduce perturbation magnitude
                perturb_rate *= perturb_decay
                stagnation_count = 0
                
        # Update current_h if we've made progress
        new_c5 = evaluate_h(current_h, dx)
        if new_c5 < current_c5:
            current_c5 = new_c5

    h_refined = best_h.copy()

        h_refined = h_refined - learning_rate * grad
        h_refined = repair_h(h_refined, n_points)
        best_c5 = evaluate_h(h_refined, dx)
        if best_c5 < best_c5:
            best_h = h_refined.copy()
            best_c5 = best_c5
            
    return (best_h.tolist(), best_c5, n_points)
