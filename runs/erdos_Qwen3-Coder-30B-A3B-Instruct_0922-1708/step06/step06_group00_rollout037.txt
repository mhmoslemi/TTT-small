import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h = (h_init > 0.5).astype(int)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-6:
            # Adjust to have exactly n_points / 2 ones
            num_ones = int(n_points / 2)
            zero_indices = np.where(h == 0)[0]
            one_indices = np.where(h == 1)[0]
            if len(one_indices) > num_ones:
                # Turn some 1s into 0s
                inds = np.random.choice(one_indices, len(one_indices) - num_ones, replace=False)
                h[inds] = 0
            elif len(one_indices) < num_ones:
                # Turn some 0s into 1s
                inds = np.random.choice(zero_indices, num_ones - len(one_indices), replace=False)
                h[inds] = 1
    else:
        # Initialize with uniform binary pattern
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # MIP solver setup
    # Define optimization variables
    h_vars = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_vars) == n_points / 2]
    
    # Loop through all shifts - efficient way for MIP
    shifts = range(-n_points + 1, n_points)
    
    # For each shift k, compute overlap constraint
    for k in shifts:
        if abs(k) >= n_points:
            continue
        # Valid indices where both i and i+k are in [0, n_points-1]
        if k >= 0:
            valid_i = list(range(n_points - k))
            valid_j = list(range(k, n_points))
        else:
            valid_i = list(range(-k, n_points))
            valid_j = list(range(n_points + k))
        
        # Sum over valid pairs
        overlap_sum = cp.sum(cp.multiply(h_vars[valid_i], 1 - h_vars[valid_j]))
        constraints.append(overlap_sum <= t * (n_points / 2))
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CBC, time_limit=budget_s - 10, gap_limit=0.01)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_mip = h_vars.value
            if h_mip is not None:
                h = np.round(h_mip).astype(int)
            else:
                # Fallback if no solution computed
                h = h
        else:
            # Fallback to previous best if no optimal solution
            pass
    except Exception as e:
        # If MIP solver fails, just keep the binary pattern
        pass
    
    # Ensure correct sum
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-6:
        # Adjust to make sum equal to n_points / 2
        target_sum = int(n_points / 2)
        diff = int(current_sum - target_sum)
        if diff > 0:
            # Remove some ones
            zero_positions = np.where(h == 0)[0]
            if len(zero_positions) < diff:
                one_positions = np.where(h == 1)[0]
                change_idx = np.random.choice(one_positions, diff, replace=False)
                h[change_idx] = 0
            else:
                change_idx = np.random.choice(zero_positions, diff, replace=False)
                h[change_idx] = 1
        elif diff < 0:
            # Add some ones
            one_positions = np.where(h == 1)[0]
            if len(one_positions) < -diff:
                zero_positions = np.where(h == 0)[0]
                change_idx = np.random.choice(zero_positions, -diff, replace=False)
                h[change_idx] = 1
            else:
                change_idx = np.random.choice(one_positions, -diff, replace=False)
                h[change_idx] = 0
    
    # Final evaluation
    cor = correlate(h, 1 - h, mode='full')
    c5 = np.max(cor) * dx
    
    # Final h should be in binary form
    h_final = h.astype(float)
    
    return (h_final.tolist(), c5, n_points)
