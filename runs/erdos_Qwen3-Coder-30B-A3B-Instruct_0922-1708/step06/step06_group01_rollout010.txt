import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a piecewise linear function
        x = np.linspace(0, 2, n_points)
        h_init = np.zeros(n_points)
        # Create a two-part piecewise linear function
        idx1 = x <= 1
        idx2 = x > 1
        h_init[idx1] = 2 * x[idx1]
        h_init[idx2] = 2 - 2 * x[idx2]
        h_init = repair_h(h_init, n_points)
    
    # Coordinate descent with projection
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Parameters for optimization
    eta = 0.01
    eps = 1e-6
    patience = 100
    last_improvement = 0
    max_iter = 100000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic coordinate descent
        for j in range(n_points):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient using finite differences
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad = (c5_plus - c5_minus) / (2 * eps)
            
            # Update coordinate
            h_current[j] = h_current[j] - eta * grad
            h_current[j] = max(0, min(1, h_current[j]))
            
        # Repair and evaluate
        h_current = repair_h(h_current, n_points)
        c5_current = evaluate_h(h_current, dx)
        
        # Update best if improved
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            eta = min(0.1, eta * 1.1)
            last_improvement = i
            
        # Occasionally perturb to escape local minima
        if i % 1000 == 0 and i > 0:
            for k in range(n_points):
                if np.random.rand() < 0.01:

            c5_current = evaluate_h(h_current, dx)
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
