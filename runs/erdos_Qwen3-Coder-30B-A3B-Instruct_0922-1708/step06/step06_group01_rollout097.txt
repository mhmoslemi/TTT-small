import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute gradient of C5 using finite differences."""
    n_points = len(h)
    grad = np.zeros(n_points)
    c5_base = evaluate_h(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n_points)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    
    return grad

def trust_region_step(h, dx, grad, trust_radius=1.0, max_iter=100):
    """Compute a trust region step using projected gradient descent."""
    n_points = len(h)
    h_new = h.copy()
    
    # Projected gradient descent in trust region
    for _ in range(max_iter):
        # Compute projected gradient
        d = -grad
        
        # Project to trust region
        norm_d = np.linalg.norm(d)
        if norm_d > trust_radius:
            d = d * trust_radius / norm_d
        
        # Check if step is feasible
        h_trial = h_new + d
        h_trial = repair_h(h_trial, n_points)
        
        # Accept step if it improves the objective
        c5_trial = evaluate_h(h_trial, dx)
        c5_current = evaluate_h(h_new, dx)
        
        if c5_trial < c5_current:
            h_new = h_trial
            break
        else:
            # Reduce step size
            d = d * 0.5
            h_trial = h_new + d
            h_trial = repair_h(h_trial, n_points)
            c5_trial = evaluate_h(h_trial, dx)
            if c5_trial < c5_current:
                h_new = h_trial
                break
    
    return h_new
