import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5

    # Gradient descent optimization
    learning_rate = 1e-4
    max_iter = 10000
    decay_factor = 0.99
    decay_every = 100

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        current_corr = correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(current_corr)
        max_val = current_corr[max_idx]
        
        # Compute gradient for each component using finite differences
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[j] = (corr_plus[max_idx] - max_val) / eps
        
        # Update with gradient descent
        new_h = current_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)
        
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Adaptive learning rate decay
        if i > 0 and i % decay_every == 0:
            learning_rate *= decay_factor
            
    # Final postprocessing with a few iterations of refinement
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        current_corr = correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(current_corr)
        max_val = current_corr[max_idx]
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[j] = (corr_plus[max_idx] - max_val) / eps
        
        new_h = best_h - 0.001 * grad
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
