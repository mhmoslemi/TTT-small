import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # CVXPY optimization
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2, t >= 0]
    
    # Compute overlap constraints for all valid lags
    # For a non-circular convolution of h with (1-h), valid lags j in [-n_points+1, n_points-1]
    # But we only need to check where the convolution makes sense
    max_lag = n_points - 1
    for j in range(-max_lag, max_lag + 1):
        # Compute indices such that h[i] * (1 - h[i-j]) is valid
        left_indices = np.arange(max(0, -j), min(n_points, n_points-j))
        right_indices = np.arange(max(0, j), min(n_points, n_points+j))
        if len(left_indices) > 0 and len(right_indices) > 0:
            if j < 0:
                # h[i] * (1 - h[i-j]) where i-j >= 0
                # so we consider i in [0, n_points-1] such that i-j >=0 and i-j < n_points
                lhs = h_var[left_indices]
                rhs = 1 - h_var[right_indices]
            else:
                # h[i] * (1 - h[i-j]) where i-j < n_points
                # so we consider i in [0, n_points-1] such that i ≥ 0 and i-j < n_points
                lhs = h_var[right_indices]
                rhs = 1 - h_var[left_indices]
            
            if len(lhs) > 0:
                # Ceil(2*dx) because of discretization effect
                # Approximate the integral as sum: dx * (h[i] * (1 - h[i-j])) = sum over i
                # Use approximate dot product within index bounds
                # Overlap function will be around N * dx in scale
                overlap_constraint = cp.sum(cp.multiply(lhs, rhs)) <= t * dx * n_points
                constraints.append(overlap_constraint)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h_var.value
            c5_sol = t.value * dx
        else:
            h_sol = h_init
            c5_sol = evaluate_h(h_init, dx)
    except Exception:
        h_sol = h_init
        c5_sol = evaluate_h(h_init, dx)
        
    # Return best result
    return (h_sol.tolist(), c5_sol, n_points)
