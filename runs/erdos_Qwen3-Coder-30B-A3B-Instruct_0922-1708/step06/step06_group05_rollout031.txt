import numpy as np
import time
import cvxpy as cp
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Convex relaxation stage
    if time.time() - start_time < budget_s - 100:
        # Use CVXPY for convex relaxation of the problem
        h_cvx = cp.Variable(n_points)
        # Minimize a quadratic approximation to the max term
        # We try to find h that minimizes sum of squares of correlations
        # This is a convex relaxation approximating the original problem
        
        # Sum of squared correlations
        corr_expr = cp.norm(cp.conv(h_cvx, 1 - h_cvx, mode='full'), 2) ** 2
        # Constrain the sum
        constraints = [
            cp.sum(h_cvx) == n_points / 2,
            h_cvx >= 0,
            h_cvx <= 1
        ]
        
        prob = cp.Problem(cp.Minimize(corr_expr), constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h = h_cvx.value
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                best_h = h.copy()
                best_c5 = evaluate_h(h, dx)
        except Exception:
            pass

    # Refinement with gradient-based method
    max_iter = 5000
    lr = 0.01
    beta = 0.9
    tolerance = 1e-6
    last_improvement = 0
    stagnation_count = 0
    
    corr_cache = None
    last_c5 = best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute gradient using finite difference
        eps = 1e-6
        grad = np.zeros(n_points)
        h_eval = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - h_eval) / eps
                
        # Update step
        # Adapt learning rate based on iteration
        current_lr = lr / (1 + iteration ** beta)
        h = h - current_lr * grad
        h = repair_h(h, n_points)
        
        # Evaluate new C5
        c5 = evaluate_h(h, dx)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early termination if no improvement in a while
        if stagnation_count > 100:
            break
            
    return (best_h.tolist(), best_c5, n_points)
