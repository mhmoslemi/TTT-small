```python
import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum=50.0):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Make sure sum is exactly 50
    h = project_h(h, 50.0)
    
    # Parameters
    lambda_reg = 0.01  # sparsity regularization
    alpha = 0.1        # learning rate
    eps = 1e-6         # finite difference step
    max_iter = 1000
    patience = 100
    best_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    
    iter_count = 0
    no_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_eps = h.copy()
            h_eps[i] += eps
            h_eps = project_h(h_eps, 50.0)
            c5_eps = evaluate_h(h_eps, dx)
            grad[i] = (c5_eps - current_c5) / eps

        # Add L1 regularization gradient (sign of h)
        reg_grad = lambda_reg * np.sign(h)
        
        # Update
        h_new = h - alpha * (grad + reg_grad)
        h_new = project_h(h_new, 50.0)
        
        # Evaluate new solution
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept with backtracking
        if new_c5 <= current_c5:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
            # Decrease learning rate if improvement
            alpha = max(1e-4, alpha * 0.995)
        else:
            # Increase learning rate if no improvement
            alpha = min(1.0, alpha * 1.05)
            no_improvement += 1
            
        iter_count += 1
        
        # Early termination if no improvement for too long
        if no_improvement > patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
```
