import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Annealing parameters
    T0 = 1.0
    alpha = 0.99
    T = T0
    lr = 0.01
    max_iter = 10000
    
    # Multi-scale parameters
    scale_up_every = 500
    max_scale = 4  # up to 20 * 2^4 = 320 points
    
    # Initial scale
    current_scale = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Double resolution periodically
        if iteration > 0 and iteration % scale_up_every == 0 and current_scale < max_scale:
            current_scale += 1
            n_points *= 2
            dx = 2.0 / n_points
            
            # Interpolate best_h to new grid
            x_orig = np.linspace(0, 2, len(best_h))
            x_target = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_target, x_orig, best_h)
            h = repair_h(h_interp, n_points)
            
            # Re-initialize for new scale
            best_h = h.copy()
            best_c5 = evaluate_h(h, dx)
            
        # Perturb using simulated annealing
        h_new = h + np.random.normal(0, T, n_points)
        h_new = repair_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Cool down
        T = max(T * alpha, 1e-6)

        if iteration % 100 == 0:
            eps = 1e-6
            grad = np.zeros(n_points)
            c5_eval = evaluate_h(h, dx)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                grad[i] = (c5_plus - c5_eval) / eps
            h = h - lr * grad
            h = repair_h(h, n_points)
            new_c5 = evaluate_h(h, dx)
            if new_c5 < best_c5:

    for _ in range(1000):

            best_c5 = new_c5
