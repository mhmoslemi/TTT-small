import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50000
    tabu_size = 50
    
    # Initialize h from initial_h_values or random binary
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init.copy()
        # Ensure sum is exactly n_points/2
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Convert to binary representation with exact support size
    def to_binary(h_vec):
        # Binary approximation with exactly n_points/2 ones
        sorted_indices = np.argsort(h_vec)[::-1]  # descending order
        binary = np.zeros(n_points)
        binary[sorted_indices[:n_points//2]] = 1.0
        return binary
    
    # Convert to continuous representation to satisfy constraint
    def to_continuous(binary_vec):
        h_new = binary_vec.copy()
        # Normalize sum to be exactly n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Compute overlap
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Tabu list to store last tabu_size moves
    tabu_list = []
    
    # Current best solution
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Simulated annealing variables
    temperature = 0.1
    cooling_rate = 0.999
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping one bit in binary representation
        binary_current = to_binary(current_h)
        candidate_binary = binary_current.copy()
        flip_idx = np.random.randint(0, n_points)
        
        # Tabu check
        if flip_idx in tabu_list:
            continue
            
        candidate_binary[flip_idx] = 1 - candidate_binary[flip_idx]
        candidate_h = to_continuous(candidate_binary)
        
        # Compute new overlap
        candidate_c5 = compute_c5(candidate_h)
        
        # Accept if better
        if candidate_c5 < current_c5:
            current_h = candidate_h
            current_c5 = candidate_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
            # Add to tabu list
            tabu_list.append(flip_idx)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        else:
            # Accept with probability based on temperature
            delta = candidate_c5 - current_c5
            if np.random.random() < np.exp(-delta / temperature):
                current_h = candidate_h
                current_c5 = candidate_c5
                
                # Update best solution
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    
                # Add to tabu list
                tabu_list.append(flip_idx)
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)
        
        # Decrease temperature
        temperature *= cooling_rate
        
        # Occasionally reinitialize for escaping local optima
        if iteration % 1000 == 0 and iteration > 0:
            current_h = to_continuous(to_binary(np.random.random(n_points)))
            current_c5 = compute_c5(current_h)
            temperature = max(temperature * 0.9, 1e-6)
    
    return (best_h.tolist(), best_c5, n_points)
