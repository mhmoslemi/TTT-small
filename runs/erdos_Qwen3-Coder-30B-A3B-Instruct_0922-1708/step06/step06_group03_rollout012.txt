import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()

    # Dual ascent parameters
    dual_vars = np.ones(2 * n_points - 1)  # λ_k for all lags
    dual_vars /= np.sum(dual_vars)  # normalize
    temperature = 0.1
    learning_rate = 5e-4
    dual_update_freq = 10
    stagnation_count = 0
    max_stagnation = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        
        # Compute overlaps for all lags
        corr = correlate(current_h, 1 - current_h, mode='full')
        overlaps = corr * dx
        
        # Update dual variables using variational inference
        if iteration % dual_update_freq == 0:
            dual_vars = np.exp(-overlaps / temperature)
            dual_vars /= np.sum(dual_vars)
            temperature *= 0.999
            
        # Compute gradient w.r.t h
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(-n_points + 1, n_points):

                    grad[i] += dual_vars[k + n_points - 1] * (1 - current_h[i - k])

        current_h -= learning_rate * grad
        current_h = repair_h(current_h, n_points)

            stagnation_count += 1

            current_h = h_init.copy() + np.random.normal(0, 0.001, n_points)
            current_h = repair_h(current_h, n_points)
            current_c5 = evaluate_h(current_h, dx)
            stagnation_count = 0

        if iteration % 500 == 0:
            learning_rate *= 0.99
            
    return (best_h.tolist(), best_c5, n_points)
