import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def interpolate_to_resolution(h_coarse, n_points):
    x_src = np.linspace(0, 2, len(h_coarse))
    x_tgt = np.linspace(0, 2, n_points)
    return np.interp(x_tgt, x_src, h_coarse)

def es_mutation(h, mutation_scale):
    noise = np.random.normal(0, mutation_scale, len(h))
    h_new = h + noise
    return repair_h(h_new, len(h))

def es_selection(population, values, k=10):
    indices = np.argsort(values)[:k]
    return [population[i] for i in indices]

def es_crossover(parent1, parent2):
    if np.random.rand() < 0.3:
        mask = np.random.rand(len(parent1)) < 0.5
        child = np.where(mask, parent1, parent2)
        return child
    else:
        return parent1

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution ES approach
    resolutions = [10, 20, 40, 100]  # coarse to fine
    initial_candidates = 50
    generations_per_stage = 50
    survival_rate = 0.2
    elite_count = int(initial_candidates * survival_rate)
    
    # Initialize coarse solution
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        if len(h_coarse) != resolutions[0]:
            x_orig = np.linspace(0, 2, len(h_coarse))
            x_target = np.linspace(0, 2, resolutions[0])
            h_coarse = np.interp(x_target, x_orig, h_coarse)
        h_coarse = repair_h(h_coarse, resolutions[0])
    else:
        h_coarse = np.random.rand(resolutions[0])
        h_coarse = repair_h(h_coarse, resolutions[0])
    
    best_h = None
    best_value = float('inf')
    current_resolution = resolutions[0]
    
    for stage, resolution in enumerate(resolutions):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Interpolate to target resolution
        h_current = interpolate_to_resolution(h_coarse, resolution)
        current_resolution = resolution
        
        # Create initial population
        population = [h_current.copy()]
        for _ in range(initial_candidates - 1):
            if stage > 0:
                h_new = es_mutation(h_current, 0.1 / np.sqrt(resolution))
            else:
                h_new = np.random.rand(resolution)
                h_new = repair_h(h_new, resolution)
            population.append(h_new)
        
        # Evolutionary optimization
        for gen in range(generations_per_stage):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Evaluate population
            values = []
            for h in population:
                value = evaluate_h(h, 2.0 / resolution)
                values.append(value)
                if value < best_value:
                    best_value = value
                    best_h = h.copy()
            
            # Selection
            elite = es_selection(population, values, elite_count)
            
            # Crossover and mutation
            new_population = elite.copy()
            while len(new_population) < initial_candidates:
                parent1 = np.random.choice(elite)
                parent2 = np.random.choice(elite)
                child = es_crossover(parent1, parent2)
                child = es_mutation(child, 0.1 / np.sqrt(resolution))
                new_population.append(child)
            
            population = new_population
            
        # Update coarse solution for next stage
        h_coarse = interpolate_to_resolution(population[0], resolutions[0]) if stage < len(resolutions) - 1 else population[0]
    
    # Final local refinement with gradient descent
    h_final = best_h.copy()
    dx = 2.0 / current_resolution
    value_final = best_value
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(current_resolution)
        for j in range(current_resolution):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, current_resolution)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - value_final) / eps
        
        h_final = h_final - 0.001 * grad
        h_final = repair_h(h_final, current_resolution)
        corr = correlate(h_final, 1 - h_final, mode='full')
        value_final = np.max(corr) * dx
        
        if value_final < best_value:
            best_value = value_final
            best_h = h_final.copy()
    
    return (best_h.tolist(), best_value, current_resolution)
