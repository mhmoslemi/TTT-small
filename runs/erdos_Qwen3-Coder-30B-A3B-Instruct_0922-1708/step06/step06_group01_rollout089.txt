import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with structured waveform
        x = np.linspace(0, 2, n_points)
        h_init = np.zeros(n_points)
        # Sine wave with period 2
        h_init = 0.5 + 0.5 * np.sin(np.pi * x)
        # Make sure it sums to 1
        h_init = repair_h(h_init, n_points)
    
    # Coordinate descent with gradient approximation
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Parameters
    eta = 0.01
    epsilon = 1e-6
    max_iter = 100000
    patience = 100
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic coordinate-wise update
        for j in range(n_points):
            # Compute gradient with finite differences
            h_plus = h_current.copy()
            h_plus[j] += epsilon
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[j] -= epsilon
            h_minus = repair_h(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)
            
            # Estimate gradient
            grad = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update
            h_new = h_current[j] - eta * grad
            h_new = np.clip(h_new, 0, 1)
            h_current[j] = h_new
            
        # Re-project non-negativity and sum constraint
        h_current = repair_h(h_current, n_points)
        
        # Evaluate current
        c5_current = evaluate_h(h_current, dx)
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            eta *= 0.95
            if eta < 1e-6:
                eta = 1e-6
            last_improvement = i
            
    return (h_best.tolist(), c5_best, n_points)
