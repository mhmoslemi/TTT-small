import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Convert to binary representation with correct sum
    def to_binary(h_values):
        h_copy = h_values.copy()
        h_copy = np.clip(h_copy, 0, 1)
        target_sum = n_points / 2
        current_sum = np.sum(h_copy)
        if abs(current_sum - target_sum) > 1e-8:
            h_copy = h_copy * target_sum / current_sum
        
        # Convert to binary with ~n_points/2 ones
        sorted_indices = np.argsort(h_copy)[::-1]
        binary_h = np.zeros(n_points)
        for i in range(int(target_sum)):
            binary_h[sorted_indices[i]] = 1
        
        return binary_h
    
    # Convert back to continuous representation
    def to_continuous(binary_h, target_sum=n_points/2):
        # Normalize to have target_sum
        h = binary_h.copy().astype(float)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-8:
            h = h * target_sum / current_sum
        return h
    
    # Initial binary representation
    binary_h = to_binary(h)
    
    # Simulated annealing with tabu search
    best_binary = binary_h.copy()
    best_c5 = float('inf')
    
    # Temperature schedule
    temp = 0.1
    cooling_rate = 0.999
    max_iter = 50000
    tabu_size = 50
    tabu_list = []
    
    # Compute initial C5
    h_continuous = to_continuous(binary_h)
    corr = correlate(h_continuous, 1 - h_continuous, mode='full')
    current_c5 = np.max(corr) * dx
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_binary = binary_h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping one bit
        neighbor_binary = binary_h.copy()
        flip_idx = np.random.randint(0, n_points)
        
        # Keep track of recent moves to avoid tabu
        if len(tabu_list) >= tabu_size:
            tabu_list.pop(0)
        
        while flip_idx in tabu_list:
            flip_idx = np.random.randint(0, n_points)
            
        neighbor_binary[flip_idx] = 1 - neighbor_binary[flip_idx]
        tabu_list.append(flip_idx)
        
        # Convert neighbor to continuous
        neighbor_h = to_continuous(neighbor_binary)
        
        # Compute C5 for neighbor
        corr = correlate(neighbor_h, 1 - neighbor_h, mode='full')
        neighbor_c5 = np.max(corr) * dx
        
        # Accept or reject
        delta = neighbor_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            binary_h = neighbor_binary.copy()
            current_c5 = neighbor_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_binary = binary_h.copy()
        
        # Update temperature
        temp *= cooling_rate
    
    # Final conversion and evaluation
    final_h = to_continuous(best_binary)
    corr = correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (final_h.tolist(), final_c5, n_points)
