import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, k_max, n_points):
    """Compute gradient of C5 with respect to h."""
    grad = np.zeros(n_points)
    for j in range(n_points):
        # Compute contribution to gradient at position j
        # for the maximum overlap at lag k_max
        grad[j] = 1 - h[(j - k_max) % n_points]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Start from initial construction if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Gradient descent parameters
    learning_rate = 1e-4
    decay_rate = 0.99
    perturbation_magnitude = 0.05
    decay_perturbation = 0.999
    patience = 0
    max_patience = 100
    perturb_every = 500
    
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        
        # Compute current overlaps
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Compute gradient
        grad = compute_gradient(current_h, k_max, n_points)
        
        # Update h using gradient descent
        h_new = current_h - learning_rate * grad
        h_new = repair_h(h_new, n_points)
        
        # Compute new C5
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept or reject the update
        if new_c5 < current_c5:
            current_h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            
        # Apply perturbations occasionally to escape local optima
        if iteration % perturb_every == 0 and patience > 50:
            perturbation = np.random.uniform(-perturbation_magnitude, perturbation_magnitude, n_points)
            h_perturbed = current_h + perturbation
            h_perturbed = repair_h(h_perturbed, n_points)
            c5_perturbed = evaluate_h(h_perturbed, dx)
            
            if c5_perturbed < current_c5:
                current_h = h_perturbed
                current_c5 = c5_perturbed
                patience = 0
        
        # Adjust hyperparameters
        if iteration % 100 == 0:
            learning_rate *= decay_rate
            perturbation_magnitude *= decay_perturbation
            
        # Early stopping if no improvement for a long time
        if patience > max_patience:
            break
    
    # Final refinement
    h_refined = best_h.copy()
    best_c5_refined = best_c5

    for _ in range(1000):
