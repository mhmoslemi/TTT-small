import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_to_feasible(h_init, n_points)
    else:
        # Initialize with a structured pattern - triangular wave
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 - np.abs(x - 1))  # Triangular midrange waveform
        h_init = project_to_feasible(h_init, n_points)

    # Frank-Wolfe algorithm
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    patience = 0
    max_patience = 1000
    alpha = 0.95
    eta_max = 1.0
    eps = 1e-6

    # Frank-Wolfe subproblem solver
    def solve_linear_subproblem(grad, h, n_points):
        # This is a rough estimate of what the projected gradient step 
        # would look like, taking into account bounds and sum constraint
        d = -grad
        # Project d onto feasible set with sum=0 constraint
        # This projection approximates the Frank-Wolfe linear subproblem
        d_sum = np.sum(d)
        d = d - d_sum / n_points  # Ensure ∑d = 0
        # Check if d is in descent direction and feasible
        if np.dot(grad, d) >= 0:
            d = -grad + np.dot(grad, grad) / np.dot(grad, grad) * grad  # fallback
            d_sum = np.sum(d)
            d = d - d_sum / n_points  # Ensure sum=0
        return d

    for i in range(100000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = project_to_feasible(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Solve linear subproblem
        d = solve_linear_subproblem(grad, h_current, n_points)
        
        # Line search to find optimal step size
        eta = eta_max
        while eta > 1e-10:
            h_new = h_current + eta * d
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            if c5_new < c5_current:
                break
            eta *= 0.5
        
        if eta <= 1e-10:
            eta = eta_max
            
        h_current = h_current + eta * d
        h_current = project_to_feasible(h_current, n_points)
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            break
            
        # Adaptive step size adjustment
        eta_max = max(1e-5, eta_max * alpha)

    return (h_best.tolist(), c5_best, n_points)
