import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Formulate as convex optimization
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2, t >= 0]
    
    # Compute overlaps for all valid lags
    max_lag = n_points - 1
    for lag in range(-max_lag, max_lag + 1):
        if lag >= 0:
            # h[i] * (1 - h[i-lag]) for i >= lag
            overlap_sum = cp.sum(cp.multiply(h[lag:], 1 - h[:n_points-lag]))
        else:
            # h[i] * (1 - h[i-lag]) for i < -lag
            lag = -lag
            overlap_sum = cp.sum(cp.multiply(h[:n_points-lag], 1 - h[lag:]))
        
        # Add constraint: overlap <= t
        constraints.append(overlap_sum <= t * n_points / 2)
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_bound = t.value * dx
        else:
            # Fallback to gradient-based refinement on initial guess
            h_opt = h_init.copy()
            c5_bound = evaluate_h(h_init, dx)
            
            # Simple gradient descent refinement
            for _ in range(1000):
                if time.time() - start_time > budget_s - 10:
                    break
                eps = 1e-6
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_opt.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c5_plus = evaluate_h(h_plus, dx)
                    grad[j] = (c5_plus - c5_bound) / eps
                
                h_opt = h_opt - 0.01 * grad
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                new_c5 = evaluate_h(h_opt, dx)
                if new_c5 < c5_bound:
                    c5_bound = new_c5
    except:
        # Revert to last known good solution
        h_opt = h_init.copy()
        c5_bound = evaluate_h(h_init, dx)
    
    # Final validation
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    c5_bound = evaluate_h(h_opt, dx)
    
    return (h_opt.tolist(), c5_bound, n_points)

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
