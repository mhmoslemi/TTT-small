import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_best = h_init.copy()
    else:
        h_best = np.ones(n_points) * 0.5
    
    h_best = repair_h(h_best, n_points)
    c5_best = evaluate_h(h_best, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 1000
    elite_size = 10
    mutation_rate = 0.1
    crossover_rate = 0.5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = h_best + np.random.normal(0, 0.05, n_points)
            h = repair_h(h, n_points)
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Main loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < c5_best:
            c5_best = fitness[0]
            h_best = population[0].copy()
        
        # Create new population
        new_population = population[:elite_size]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate:
                # Crossover
                parent1, parent2 = np.random.choice(elite_size, 2, replace=False)
                alpha = np.random.rand()
                child = alpha * population[parent1] + (1 - alpha) * population[parent2]
            else:
                # Mutation
                parent = np.random.choice(elite_size)
                child = population[parent].copy()
                mask = np.random.rand(n_points) < mutation_rate
                if np.any(mask):
                    noise = np.random.normal(0, 0.05, n_points)
                    child[mask] += noise[mask]
            
            child = repair_h(child, n_points)
            new_population.append(child)
        
        population = new_population
    
    # Final gradient refinement
    h_refined = h_best.copy()
    c5_refined = c5_best
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - c5_refined) / eps
        
        h_refined = h_refined - 0.0001 * grad
        h_refined = repair_h(h_refined, n_points)
        c5_refined = evaluate_h(h_refined, dx)
        
        if c5_refined < c5_best:
            c5_best = c5_refined
            h_best = h_refined.copy()
    
    return (h_best.tolist(), c5_best, n_points)
