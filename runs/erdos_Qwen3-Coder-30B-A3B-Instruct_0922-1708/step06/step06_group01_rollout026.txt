import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to the feasible set using quadratic programming."""
    # Constraint: sum(h) = n_points / 2, h[i] in [0,1]
    target_sum = n_points / 2.0
    
    # Initialize
    h_proj = h.copy()
    
    # Scale to satisfy sum constraint
    current_sum = np.sum(h_proj)
    if abs(current_sum - target_sum) > 1e-8:
        h_proj = h_proj * target_sum / current_sum
    
    # Clip to [0, 1]
    h_proj = np.clip(h_proj, 0, 1)
    
    # Use a simple iterative projection if needed (sparse version)
    # For robustness on budget, we'll use a simpler approach
    # based on solving a simpler LP or simply clip, project
    # We'll use a more stable version  
    h_proj = np.clip(h_proj, 0, 1)
    
    # Re-scale to ensure sum is correct
    current_sum = np.sum(h_proj)
    if abs(current_sum - target_sum) > 1e-8:
        h_proj = h_proj * target_sum / current_sum
    
    return h_proj

def compute_gradient(h, dx, eps=1e-6):
    """Compute approximate gradient using finite differences."""
    grad = np.zeros_like(h)
    h_base = h.copy()
    c5_base = evaluate_h(h_base, dx)
    
    for i in range(len(h)):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    
    return grad
