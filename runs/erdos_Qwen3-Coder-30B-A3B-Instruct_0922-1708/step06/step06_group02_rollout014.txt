import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Precompute shift grid
    shift_grid = np.linspace(-2, 2, 100)
    
    # Gradient ascent parameters
    learning_rate = 0.001
    max_iter = 10000
    adaptive_lr = True
    lr_decay = 0.5
    lr_patience = 100
    
    last_improvement = 0
    lr = learning_rate
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all correlations
        corr_full = correlate(h, 1 - h, mode='full')
        # Get the max correlation for each shift
        corr_shifted = np.array([corr_full[len(corr_full)//2 + int(k/dx)] for k in shift_grid if abs(k) <= 2])
        
        # Find worst k
        worst_idx = np.argmax(corr_shifted)
        worst_k = shift_grid[worst_idx]
        
        # Compute gradient w.r.t. h
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Check neighbors with offset
            k_idx = int(worst_k / dx)
            if 0 <= i - k_idx < n_points:
                grad[i] += -2 * (1 - h[i - k_idx])
            if 0 <= i + k_idx < n_points:
                grad[i] += -2 * h[i + k_idx]
        
        # Update h
        h_new = h + lr * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate
        new_value = evaluate_h(h_new, dx)
        
        if new_value < best_value:
            best_value = new_value
            best_h = h_new.copy()
            last_improvement = iteration
            if adaptive_lr:
                lr = learning_rate
        else:
            # Update h back to previous best
            h = h_new.copy()
            
        # Adaptive learning rate
        if adaptive_lr and iteration - last_improvement > lr_patience:
            lr *= lr_decay
            last_improvement = iteration
        
        # Periodic perturbation to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            pert = np.random.normal(0, 0.001, n_points)
            h = h + pert
            h = repair_h(h, n_points)
    
    return (best_h.tolist(), best_value, n_points)
