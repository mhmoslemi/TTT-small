import numpy as np
import time
import cvxpy as cp
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    # Use scipy.signal.correlate for efficient computation
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Rescale to meet integral constraint exactly
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use high resolution for better accuracy
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Define variables for convex optimization
    h_var = cp.Variable(n_points)
    # Constraints for h_var
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # Use a more robust quadratic relaxation approach
    # Minimize ||h ⋆ (1-h)||_2^2 with constraint that max over shifts is bounded
    
    # Gradient descent with adaptive learning rate
    lr = 0.001
    lr_decay = 0.999
    max_iter = 10000
    stagnation_limit = 200
    stagnation_count = 0
    last_improvement = 0
    
    # Direct gradient computation with finite differences
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate current value and gradient
        c5_eval = evaluate_h(h, dx)
        
        # Gradient estimation
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        
        # Update
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        # Evaluate
        c5 = evaluate_h(h, dx)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
            last_improvement = iteration
        elif iteration - last_improvement > stagnation_limit or (
             abs(c5 - best_c5) < 1e-8 and iteration > 100):
            # Adaptively reduce learning rate and increase perturbation
            lr *= lr_decay
            # Add small random perturbation
            perturb_indices = np.random.choice(n_points, size=max(1, n_points//20), replace=False)
            h[perturb_indices] += np.random.normal(0, lr/10, len(perturb_indices))
            h = repair_h(h, n_points)
            stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Occasionally restart to avoid local optima
        if stagnation_count > 50 and iteration > 1000:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
    
    # Final intensive refinement with smaller learning rate
    lr_final = 0.0001
    for _ in range(5000):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        best_h = best_h - lr_final * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
    return (best_h.tolist(), best_c5, n_points)
