import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints using quadratic programming approach."""
    # Clip to [0, 1] first
    h = np.clip(h, 0, 1)
    
    # Project onto the feasible set with sum = n_points / 2
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    
    if abs(current_sum - target_sum) > 1e-8:
        # Use the projection formula for sum constraint
        # h_new = h + (target_sum - sum(h)) / n_points * ones
        h = h + (target_sum - current_sum) / n_points
        
        # Re-clip to [0, 1]
        h = np.clip(h, 0, 1)
        
    return h
