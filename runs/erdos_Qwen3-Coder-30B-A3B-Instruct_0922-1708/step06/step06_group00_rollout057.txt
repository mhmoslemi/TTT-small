import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution and time
    best_h = h.copy()
    best_c5 = float('inf')
    
    iter_count = 0
    last_improvement = 0
    
    while (time.time() - start_time) < budget_s - 5 and iter_count - last_improvement < 10:
        iter_count += 1
        
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iter_count

        # Sample random points to vary h
        # Do binary optimization for sparsity
        z = (h > 0.5).astype(float)
        h_new = z.copy()
        
        # Mutate with small perturbation
        for i in range(n_points):
            if np.random.random() < 0.1:
                h_new[i] = np.random.random()
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if abs(h_new_sum - n_points / 2) > 1e-8:
            h_new = h_new * (n_points / 2) / h_new_sum
            
        # Test with new h
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < c5:
            h = h_new.copy()
        else:
            # Try alternating
            h_alternate = h.copy()
            for i in range(0, n_points, 2):
                if h_alternate[i] > 0.5:
                    h_alternate[i] = 0.5
                else:
                    h_alternate[i] = 0.75
            h_alternate = np.clip(h_alternate, 0, 1)
            h_alternate_sum = np.sum(h_alternate)
            if abs(h_alternate_sum - n_points / 2) > 1e-8:
                h_alternate = h_alternate * (n_points / 2) / h_alternate_sum
                
            corr_alt = correlate(h_alternate, 1 - h_alternate, mode='full')
            c5_alt = np.max(corr_alt) * dx
            if c5_alt < c5:
                h = h_alternate.copy()


        # Gurobi-like local search -- Random flips for better local search
        if np.random.random() < 0.3:
            h_prime = h.copy()
            # Flip a few values
            for i in range(np.random.randint(1, 10)):
                j = np.random.randint(0, n_points)
                if h_prime[j] > 0.5:
                    h_prime[j] = 0.0
                else:
                    h_prime[j] = 1.0
            h_prime = np.clip(h_prime, 0, 1)
            h_prime_sum = np.sum(h_prime)
            if abs(h_prime_sum - n_points / 2) > 1e-8:
                h_prime = h_prime * (n_points / 2) / h_prime_sum
            
            corr_prime = correlate(h_prime, 1 - h_prime, mode='full')
            c5_prime = np.max(corr_prime) * dx
            
            if c5_prime < c5:
                h = h_prime.copy()

    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
