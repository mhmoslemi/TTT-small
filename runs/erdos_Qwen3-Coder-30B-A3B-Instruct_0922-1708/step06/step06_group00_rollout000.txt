import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction or binary random
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.random.rand(n_points)
    
    # Convert to binary approximation
    h_bin = (h > 0.5).astype(float)
    
    # Ensure sum = n_points / 2
    current_sum = np.sum(h_bin)
    target_sum = n_points / 2
    
    if current_sum != target_sum:
        diff = int(target_sum - current_sum)
        if diff > 0:
            # Need to flip diff 0s to 1s
            zero_indices = np.where(h_bin == 0)[0]
            flip_indices = np.random.choice(zero_indices, diff, replace=False)
            h_bin[flip_indices] = 1
        elif diff < 0:
            # Need to flip |diff| 1s to 0s
            one_indices = np.where(h_bin == 1)[0]
            flip_indices = np.random.choice(one_indices, -diff, replace=False)
            h_bin[flip_indices] = 0
    
    # Simulated annealing parameters
    temp = 0.1
    cooling_rate = 0.999
    max_iter = 50000
    tabu_list = []
    tabu_size = 50
    
    # Evaluate initial solution
    corr = correlate(h_bin, 1 - h_bin, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h_bin.copy()
    best_c5 = current_c5
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping one bit
        flip_idx = np.random.randint(0, n_points)
        h_candidate = h_bin.copy()
        h_candidate[flip_idx] = 1 - h_candidate[flip_idx]
        
        # Ensure constraint ∑h = n_points / 2 by flipping another bit if needed
        current_sum = np.sum(h_candidate)
        if current_sum != target_sum:
            # Flip another bit to restore sum
            if current_sum > target_sum:
                # Must flip a 1 to 0
                ones_indices = np.where(h_candidate == 1)[0]
                if len(ones_indices) > 0:
                    flip_idx2 = np.random.choice(ones_indices)
                    h_candidate[flip_idx2] = 0
            else:
                # Must flip a 0 to 1
                zeros_indices = np.where(h_candidate == 0)[0]
                if len(zeros_indices) > 0:
                    flip_idx2 = np.random.choice(zeros_indices)
                    h_candidate[flip_idx2] = 1
        
        # Check tabu list
        flip_tuple = tuple(sorted([flip_idx, flip_idx2] if 'flip_idx2' in locals() else [flip_idx]))
        if flip_tuple in tabu_list:
            continue
        
        # Compute overlap
        corr = correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(corr) * dx
        
        # Accept or reject based on SA criteria
        if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / temp):
            h_bin = h_candidate
            current_c5 = candidate_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = h_bin.copy()
                best_c5 = current_c5
        
        # Update tabu list
        if len(tabu_list) >= tabu_size:
            tabu_list.pop(0)
        tabu_list.append(flip_tuple)
        
        # Cool down
        temp *= cooling_rate
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
