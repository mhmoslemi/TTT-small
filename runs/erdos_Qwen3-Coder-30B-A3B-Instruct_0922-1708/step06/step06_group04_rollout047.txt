import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a sieve-like pattern in frequency domain
    freqs = np.fft.fftfreq(n_points, 1.0 / n_points)
    # High frequencies have lower amplitudes (sieve pattern)
    amplitudes = np.exp(-0.1 * np.abs(freqs)**1.5)
    amplitudes[0] = 0  # DC component should be zero for the overlap integral  
    # Construct frequency-domain signal
    real_part = np.random.normal(0, 1, n_points//2 + 1)
    imag_part = np.random.normal(0, 1, n_points//2 + 1)
    freq_signal = real_part + 1j * imag_part
    freq_signal = freq_signal * amplitudes
    # Inverse FFT to get real signal
    h = np.fft.irfft(freq_signal, n_points)
    # Normalize to meet constraint
    h = project_h(h, n_points)
    return h

    dx = 2.0 / n_points

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = fourier_initialization(n_points)
    
    # Tabu Search parameters
    tabu_list = []
    tabu_size = 50
    max_iter = 2000
    perturbation_magnitude = 0.2
    
    # Initialize current solution
    h_current = h_init.copy()
    current_value = evaluate_overlap(h_current, dx)
    
    best_h = h_current.copy()
    best_value = current_value
    
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate candidate by perturbing current solution
        h_candidate = h_current.copy()
        # Randomly perturb some points  
        indices = np.random.choice(n_points, size=n_points//5, replace=False)
        for i in indices:
            delta = np.random.normal(0, perturbation_magnitude)
            h_candidate[i] += delta
        
        # Enforce constraints
        h_candidate = project_h(h_candidate, n_points)
        
        # Check if move is in tabu list
        move_key = tuple(sorted([(i, round(h_candidate[i], 5)) for i in indices]))
        if move_key in tabu_list:
            # Small chance to accept tabu move
            if np.random.random() < 0.1:
                pass
            else:
                # Generate a new candidate to avoid tabu
                h_candidate = h_current + np.random.normal(0, perturbation_magnitude, n_points)
                h_candidate = project_h(h_candidate, n_points)
        else:
            # Add to tabu list
            tabu_list.append(move_key)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        
        # Evaluate candidate
        candidate_value = evaluate_overlap(h_candidate, dx)
        
        # Accept new solution if better
        if candidate_value < current_value:
            h_current = h_candidate.copy()
            current_value = candidate_value
            
            if candidate_value < best_value:
                best_h = h_candidate.copy()
                best_value = candidate_value
        
        iter_count += 1
    
    # Gradient descent refinement of best solution
    h_refined = best_h.copy()
    prev_value = best_value
    learning_rate = 0.0001
    gradient_iters = 500
    
    for _ in range(gradient_iters):
        if time.time() - start_time > budget_s - 10:
            break
        # Numerical gradient estimation
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for j in range(n_points):

            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        # Update
        h_refined = h_refined - learning_rate * grad
        h_refined = project_h(h_refined, n_points)
        prev_value = evaluate_overlap(h_refined, dx)
        
        if prev_value < best_value:
            best_h = h_refined.copy()
            best_value = prev_value
    
    return (best_h.tolist(), best_value, n_points)
