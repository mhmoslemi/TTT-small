import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Helper function to compute C5
    def compute_c5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
        
    # Constraint function for sum
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Objective function to minimize (C5)
    def objective(h_vals):
        return compute_c5(h_vals)
        
    # Create constraints dictionary
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    max_iter = 1000
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Try different optimization approaches
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 10:
            break
            
        # Method 1: L-BFGS-B
        try:
            result = minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-6},
                callback=lambda x: None if (time.time() - start_time) < budget_s - 10 else None
            )
            
            if result.success:
                h = result.x
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            pass
            
        # Project h back to constraints
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / h_sum
            
        # Simple local search
        for i in range(n_points):
            if (time.time() - start_time) > budget_s - 10:
                break
            # Try small perturbations
            h_trial = h.copy()
            delta = 0.01
            h_trial[i] += delta
            if h_trial[i] > 1:
                h_trial[i] = 1
            h_trial_sum = np.sum(h_trial)
            h_trial = h_trial * (n_points / 2) / h_trial_sum
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                
    # Final validation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h.tolist(), final_c5, n_points)
