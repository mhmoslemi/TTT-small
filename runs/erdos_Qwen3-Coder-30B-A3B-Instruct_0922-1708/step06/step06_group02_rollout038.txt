import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        if len(h_current) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_current = np.interp(x_target, x_orig, h_current)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    else:
        # Initialize uniformly
        h_current = np.ones(n_points) * 0.5
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_value = evaluate_h(h_current, dx)
    
    # Perturbation and learning rate
    learning_rate = 0.01
    max_iter = 10000
    no_improvement_limit = 200
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Compute all correlations across shift values
        corr_full = correlate(h_current, 1 - h_current, mode='full')
        # Extract the part corresponding to k in [-2, 2]
        # Full correlation length is 2*N - 1; center corresponds to k=0
        mid = len(corr_full) // 2
        corr_shifted = corr_full[mid - n_points:mid + n_points]
        k_indices = np.arange(-n_points + 1, n_points)
        
        # Find the shift k that maximizes the overlap (worst case)
        worst_k_index = np.argmax(corr_shifted)
        worst_k = k_indices[worst_k_index]
        worst_corr = corr_shifted[worst_k_index]
        
        # Compute gradient for this specific shift to decrease this value
        # ∂/∂h[i] [∫h(x)(1 - h(x + k)) dx] = 1 - 2*h[i+k] if i+k valid
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + worst_k
            if 0 <= j < n_points:
                gradient[i] = -2 * (1 - h_current[j])
        
        # Update h using gradient step
        h_new = h_current + learning_rate * gradient
        h_new = repair_h(h_new, n_points)
        
        # Update h if it leads to better C5 value
        new_value = evaluate_h(h_new, dx)
        if new_value < best_value:
            h_current = h_new.copy()
            best_value = new_value
            best_h = h_new.copy()
        else:
            # Add some random perturbation to escape local minima
            if iter_count % 50 == 0:
                noise = np.random.normal(0, 0.001, n_points)
                h_temp = h_current + noise
                h_temp = repair_h(h_temp, n_points)
                temp_value = evaluate_h(h_temp, dx)
                if temp_value < best_value:
                    h_current = h_temp.copy()
                    best_value = temp_value
                    best_h = h_temp.copy()
        
        # Adaptive learning rate
        if iter_count % 100 == 0 and iter_count > 0:
            learning_rate *= 0.95
    
    # Final repair
    best_h = repair_h(best_h, n_points)
    final_value = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_value, n_points)
