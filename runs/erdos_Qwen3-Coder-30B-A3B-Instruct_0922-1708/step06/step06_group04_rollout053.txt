import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # CVXPY approach
    h_cvx = cp.Variable(n_points, name="h")
    t = cp.Variable(name="t")
    
    # Constraints
    constraints = [
        h_cvx >= 0,
        h_cvx <= 1,
        cp.sum(h_cvx) == n_points / 2,
        t >= 0
    ]
    
    # Compute overlap constraints
    max_lag = n_points - 1
    for lag in range(-max_lag, max_lag + 1):
        # Compute indices for valid correlation
        left_idx = max(0, -lag)
        right_idx = min(n_points, n_points - lag)
        
        if left_idx < right_idx:
            # Corresponding indices in the shifted array
            shifts = np.arange(left_idx, right_idx)
            original = np.arange(left_idx, right_idx)
            if lag >= 0:
                corr_horizontal = h_cvx[original] * (1 - h_cvx[original + lag])
            else:
                corr_horizontal = h_cvx[original - lag] * (1 - h_cvx[original])
            overlap = cp.sum(corr_horizontal)
            constraints.append(overlap <= t * (n_points / 2))
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_final = h_cvx.value
            c5_bound = t.value * dx
        else:
            # Return the initial solution if solver fails
            h_final = h_init
            c5_bound = evaluate_h(h_final, dx)
    except Exception as e:
        # Fallback to initial solution
        h_final = h_init
        c5_bound = evaluate_h(h_final, dx)
    
    # Final verification and refinement
    if time.time() - start_time < budget_s - 10:
        h_refined = h_final.copy()
        current_c5 = evaluate_h(h_refined, dx)
        
        # Gradient-based refinement
        max_refine_iter = 1000
        lr = 0.001
        
        for _ in range(max_refine_iter):
            if time.time() - start_time > budget_s - 10:
                break
            grad = np.zeros(n_points)
            eps = 1e-6
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - current_c5) / eps
            
            h_candidate = h_refined - lr * grad
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            new_c5 = evaluate_h(h_candidate, dx)
            if new_c5 < current_c5:
                h_refined = h_candidate
                current_c5 = new_c5
            else:
                lr *= 0.99
                
        h_final = h_refined
        c5_bound = current_c5
    
    return (h_final.tolist(), c5_bound, n_points)
