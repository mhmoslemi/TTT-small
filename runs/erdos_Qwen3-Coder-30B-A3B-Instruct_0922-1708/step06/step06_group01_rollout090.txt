import numpy as np
import time
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def halton_sequence(n, base=2):
    """Generate n points using Halton sequence in [0,1)."""
    # For simplicity, use a 1D Halton sequence
    sequence = []
    for i in range(n):
        f = 1
        r = 0
        while i > 0:
            f /= base
            r += (i % base) * f
            i //= base
        sequence.append(r)
    return np.array(sequence)

        seeds_to_try = [seed + i for i in range(10)]
        
        for s in seeds_to_try:
            np.random.seed(s)
            # Use Halton sequence as base
            halton_seq = halton_sequence(n_points)
            # Convert to step function
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points),

            # Float to step function
            h = repair_h(h, n_points)

    # Gradient-based refinement with adaptive step size
    h_current = h_init.copy()

        # Compute gradient with finite differences

            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')

            h_minus = h_current.copy()
            h_minus[j] -= eps

            c5_minus = np.max(corr_minus) * dx

        # Adaptive step size
        step_size = 0.01 / (1.0 + i / 1000.0)
        h_new = h_current - step_size * grad
        h_new = repair_h(h_new, n_points)

        if c5_new < c5_current:

                c5_best = c5_current
