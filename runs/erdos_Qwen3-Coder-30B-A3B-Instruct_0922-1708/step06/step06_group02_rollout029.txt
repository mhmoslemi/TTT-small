import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def hierarchical_optimization(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    
    # Initial h

        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Hierarchical optimization with 4 levels
    levels = [10, 20, 40, 100]
    for level in levels:
        if time.time() - start_time > budget_s - 10:
            break
            
        if level < n_points:
            # Interpolate to coarser grid
            x_orig = np.linspace(0, 2, n_points)
            x_target = np.linspace(0, 2, level)
            h_coarse = np.interp(x_target, x_orig, h)
        else:
            h_coarse = h
            
        # Optimize the coarse grid
        def objective_coarse(h_coarse_flat):
            # Interpolate back to full resolution
            if level < n_points:
                x_orig = np.linspace(0, 2, level)
                x_target = np.linspace(0, 2, n_points)
                h_full = np.interp(x_target, x_orig, h_coarse_flat)
            else:
                h_full = h_coarse_flat
                
            h_full = repair_h(h_full, n_points)
            return evaluate_h(h_full, dx)
        
        # Gradient computation for coarse optimization
        def gradient_coarse(h_coarse_flat):
            # Interpolate back to full resolution
            if level < n_points:
                x_orig = np.linspace(0, 2, level)
                x_target = np.linspace(0, 2, n_points)
                h_full = np.interp(x_target, x_orig, h_coarse_flat)
            else:
                h_full = h_coarse_flat
            
            h_full = repair_h(h_full, n_points)
            
            # Numerical gradient using finite differences
            eps = 1e-6
            grad = np.zeros_like(h_coarse_flat)
            for j in range(len(h_coarse_flat)):
                h_plus = h_coarse_flat.copy()
                h_plus[j] += eps
                
                # Interpolate to full resolution
                if level < n_points:
                    x_orig = np.linspace(0, 2, level)
                    x_target = np.linspace(0, 2, n_points)
                    h_full_plus = np.interp(x_target, x_orig, h_plus)
                else:
                    h_full_plus = h_plus
                    
                h_full_plus = repair_h(h_full_plus, n_points)
                corr_plus = correlate(h_full_plus, 1 - h_full_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr_current = correlate(h_full, 1 - h_full, mode='full')
                c5_current = np.max(corr_current) * dx
                
                grad[j] = (c5_plus - c5_current) / eps
                
            return grad

            # Repair in case we violate bounds
            h = repair_h(h, n_points)
            
        except:
            pass

    def objective(h_flat):
        h_full = repair_h(h_flat, n_points)
        return evaluate_h(h_full, dx)

        h_full = repair_h(h_flat, n_points)
        eps = 1e-6

            c5_current = np.max(corr_current) * dx

                      jac=gradient,
                      options={'maxiter': 1000, 'gtol': 1e-8})
        
        if res.success:
            h = res.x
            h = repair_h(h, n_points)
    except:
        pass
    
    final_value = evaluate_h(h, dx)
    
    return (h.tolist(), final_value, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return hierarchical_optimization(seed, budget_s)
