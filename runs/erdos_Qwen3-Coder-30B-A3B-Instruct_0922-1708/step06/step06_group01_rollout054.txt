import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    h = np.clip(h, 0, 1)

        return h

    h = h * target_sum / current_sum
    
    # Now, clip
    h = np.clip(h, 0, 1)
    
    # Adjust again if necessary - iterate to balance
    # Instead, use quadratic programming directly for better accuracy
    return project_qph(h, target_sum)
