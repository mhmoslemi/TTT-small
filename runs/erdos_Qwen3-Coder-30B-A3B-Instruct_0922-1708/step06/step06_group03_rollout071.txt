import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Adaptive algorithm parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    max_iter = 100000
    progress_count = 0
    max_progress = 200
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap and find maximum lag
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient of maximum overlap with respect to h
        gradient = np.zeros(n_points)
        for j in range(n_points):
            # Compute gradient of overlap at k_max with respect to h[j]
            grad_val = 0
            for k in range(n_points):
                indicator = 1 if k == k_max else 0
                # For convolution overlap calculation
                idx = (j - k) % n_points
                grad_val += indicator * (1 - current_h[idx])
            gradient[j] = grad_val * dx
            
        # Apply gradient descent
        new_h = current_h - learning_rate * gradient
        
        # Perturb to escape local optima
        if i - last_improvement > 100 and np.random.rand() < 0.3:
            # Perturb indices with low gradient magnitude
            grad_magnitude = np.abs(gradient)
            low_grad_indices = np.where(grad_magnitude < np.percentile(grad_magnitude, 30))[0]
            if len(low_grad_indices) > 0:
                idx_to_perturb = np.random.choice(low_grad_indices)
                new_h[idx_to_perturb] += np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
            
        # Repair and validate
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject with simulated annealing style acceptance
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                last_improvement = i
                progress_count = 0
            else:
                progress_count += 1
        else:
            # Accept with probability to escape local minima
            if np.random.rand() < 0.01:
                current_h = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    last_improvement = i
                    progress_count = 0

                    progress_count += 1

        if i % 100 == 0:

        if i % 500 == 0:

        # Adaptive cooling
        if progress_count > max_progress:
            learning_rate *= 0.9
            perturbation_magnitude *= 0.95
            progress_count = 0
    
    # Final refinement
    h_refined = best_h.copy()
    best_c5_refined = best_c5

        h_refined = h_refined - 0.0001 * grad

        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_c5_refined = np.max(corr) * dx
        if best_c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = best_c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
