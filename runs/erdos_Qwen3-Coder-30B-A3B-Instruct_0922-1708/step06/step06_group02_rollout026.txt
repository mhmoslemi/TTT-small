import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize starting point
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5  # Start with uniform distribution
    
    # Create a more sophisticated optimization approach
    best_h = h_init.copy()
    best_value = evaluate_h(best_h, dx)
    
    # Hierarchical optimization with increasing resolution
    resolutions = [10, 20, 40, 100]
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Create a coarse version for this level
        if res == 100:
            h_coarse = best_h
        else:
            # Interpolate to coarser resolution
            coarse_points = np.linspace(0, 2, res)
            fine_points = np.linspace(0, 2, n_points)
            h_coarse = np.interp(coarse_points, fine_points, best_h)
        
        h_coarse = repair_h(h_coarse, res)
        
        # Optimize the coarse version
        def objective(h_c):
            # Interpolate back to fine resolution
            fine_points = np.linspace(0, 2, n_points)
            coarse_points = np.linspace(0, 2, res)
            h_fine = np.interp(fine_points, coarse_points, h_c)
            h_fine = repair_h(h_fine, n_points)
            return evaluate_h(h_fine, dx)
        
        def constraint(h_c):
            return np.sum(h_c) * (2.0 / res) - 1.0
            
        # Use scipy minimize for optimization
        try:
            # Optimize with bounds and constraints
            bounds = [(0, 1) for _ in range(res)]
            cons = {'type': 'eq', 'fun': constraint}
            # Initial guess (use previous best if available)
            if res == 100:
                x0 = h_coarse
            else:
                x0 = h_coarse
            
            # Use L-BFGS-B for better convergence
            result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                            constraints=cons, options={'maxiter': 1000})
            
            if result.success:
                h_opt = result.x
                # Interpolate back to full resolution
                fine_points = np.linspace(0, 2, n_points)
                coarse_points = np.linspace(0, 2, res)
                h_refined = np.interp(fine_points, coarse_points, h_opt)
                h_refined = repair_h(h_refined, n_points)
                
                value = evaluate_h(h_refined, dx)
                if value < best_value:
                    best_value = value
                    best_h = h_refined.copy()
                    
        except Exception:
            pass
        
        # Optimization at this resolution is complete
    
    # Local refinement with gradients
    if best_h is not None:
        h_refined = best_h.copy()

        for _ in range(500):

            eps = 1e-6
            grad = np.zeros(n_points)
            
            for j in range(n_points):

                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
            
            h_refined = h_refined - 0.001 * grad
            h_refined = repair_h(h_refined, n_points)
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            current_value = np.max(corr) * dx
            
            if current_value < best_value:
                best_value = current_value
                best_h = h_refined.copy()
    
    return (best_h.tolist(), best_value, n_points)
