import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_gradient(h, dx):
    """Compute the gradient of C5 with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-6
    
    for j in range(n):
        h_plus = h.copy()
        h_plus[j] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        
        h_minus = h.copy()
        h_minus[j] -= eps
        h_minus = repair_h(h_minus, n)
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        c5_minus = np.max(corr_minus) * dx
        
        grad[j] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def backtracking_line_search(h, grad, dx, alpha=0.5, beta=0.8, max_iter=10):
    """Perform backtracking line search."""
    t = 1.0
    current_c5 = evaluate_h(h, dx)
    
    for _ in range(max_iter):
        h_new = h - t * grad
        h_new = repair_h(h_new, len(h))
        new_c5 = evaluate_h(h_new, dx)
        
        if new_c5 <= current_c5 - alpha * t * np.dot(grad, grad):
            return h_new, t
        
        t *= beta
    
    return h, t

    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Multi-resolution approach:
    # Start with smaller resolution, then interpolate to full resolution
    resolutions = [10, 20, 40, 100]
    max_iter_per_res = 200
