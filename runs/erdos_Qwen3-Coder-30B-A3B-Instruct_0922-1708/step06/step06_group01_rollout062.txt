import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise linear function
        x = np.linspace(0, 2, n_points)
        # Place a triangular pulse in the middle
        midpoint = 1.0
        width = 0.5
        for i in range(n_points):
            xi = x[i]
            if xi < midpoint - width/2:
                h_init[i] = 0
            elif xi < midpoint + width/2:
                h_init[i] = 2*(xi - (midpoint - width/2)) / width
            else:
                h_init[i] = 0
        # Scale to target sum
        h_init = repair_h(h_init, n_points)
    
    # Coordinate descent with projection
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient estimation parameters
    epsilon = 1e-6
    learning_rate = 0.01
    max_iter = 50000
    last_improvement = 0
    patience = 100
    plateau_count = 0
    adaptive_lr = learning_rate
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic coordinate-wise updates
        for j in range(n_points):
            # Evaluate gradient using finite difference
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            h_plus = repair_h(h_plus, n_points)
            h_minus = repair_h(h_minus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            grad_j = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update
            h_current[j] = h_current[j] - adaptive_lr * grad_j
            h_current[j] = max(0, min(1, h_current[j]))
        
        # Project to feasible set
        h_current = repair_h(h_current, n_points)
        
        # Evaluate
        c5_current = evaluate_h(h_current, dx)
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            plateau_count = 0
        else:
            plateau_count += 1
        
        # Adaptive learning rate
        if plateau_count > patience:
            adaptive_lr *= 0.9
            plateau_count = 0
            
        # Occasionally perturb solution to escape local minima
        if i % 500 == 0 and i > 0:
            for _ in range(5):
                idx = np.random.randint(0, n_points)
                h_current[idx] += np.random.normal(0, 0.01)
            h_current = repair_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
    
    return (h_best.tolist(), c5_best, n_points)
