import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a sieve-like pattern in frequency domain
    freqs = np.arange(n_points)
    # High frequencies have decreasing amplitudes to suppress large overlaps
    spectrum = np.exp(-0.5 * (freqs / (n_points/4))**2)
    # Ensure DC component is zero to avoid bias
    spectrum[0] = 0
    # Inverse FFT to get real-space pattern
    h = np.fft.ifft(spectrum).real
    # Make it positive and normalize
    h = np.abs(h)
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = fourier_initialization(n_points)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    max_iter = 2000
    perturb_std = 0.2
    stagnation_limit = 50
    last_improvement = 0
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_overlap(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    population = [current_h.copy()]
    fitness = [current_c5]
    
    # Tabu search
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break

        # Create candidate by perturbing current best
        h_new = current_h + np.random.normal(0, perturb_std, n_points)
        h_new = project_h(h_new, n_points)
        
        # Check tabu
        move_key = tuple((h_new > 0.5).astype(int))
        if move_key in tabu_list:
            # Accept with small probability
            if np.random.random() > 0.1:
                continue
                
        # Evaluate
        c5_new = evaluate_overlap(h_new, dx)
        population.append(h_new)
        fitness.append(c5_new)
        
        # Update best
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iter_count
            
        # Update tabu list
        tabu_list.append(move_key)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
            
        # Update current
        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
            
        # Diversity injection
        if iter_count - last_improvement > stagnation_limit:
            h_diverse = fourier_initialization(n_points)
            population.append(h_diverse)
            fitness.append(evaluate_overlap(h_diverse, dx))
            last_improvement = iter_count
            
        # Keep best population
        pop_indices = np.argsort(fitness)[:len(population)//2]
        population = [population[i] for i in pop_indices]
        fitness = [fitness[i] for i in pop_indices]
        if len(population) > 0:
            current_h = population[0]
            current_c5 = fitness[0]
            
    # Gradient refinement of best solution
    h_refined = best_h.copy()
    prev_c5 = best_c5
    lr = 0.0001

        eps = 1e-6
        grad = np.zeros(n_points)

            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_c5) / eps
            
        h_refined = h_refined - lr * grad
        h_refined = project_h(h_refined, n_points)
        prev_c5 = evaluate_overlap(h_refined, dx)
        
        if prev_c5 < best_c5:
            best_h = h_refined.copy()
            best_c5 = prev_c5
            
    return (best_h.tolist(), best_c5, n_points)
