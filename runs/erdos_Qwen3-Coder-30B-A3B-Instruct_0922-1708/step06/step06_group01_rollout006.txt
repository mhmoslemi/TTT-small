import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1, sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a simple construction
        h_init = np.ones(n_points) * 0.5

    # Frank-Wolfe optimization
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Frank-Wolfe parameters
    max_iter = 20000
    patience = 1000
    last_improvement = 0
    eta_max = 1.0
    eps = 1e-6
    
    # Objective for tracking quality
    def objective(h):
        return evaluate_h(h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = objective(h_plus)
            grad[j] = (c5_plus - c5_current) / eps
        
        # Solve linear subproblem: min_d <grad, d> s.t. 0 <= d[i] <= 1, sum(d) = 0
        # This is a Euclidean projection onto the simplex with additional bounds
        # We simplify approach: find descent direction
        
        # Simple line search in negative gradient direction
        eta = eta_max
        d = -grad
        
        # Ensure d has zero sum
        d = d - np.mean(d)
        
        # Clamp d to feasible region (we're doing simple descent, so we just bound it)
        d = np.clip(d, -1.0, 1.0)
        
        # Line search
        while eta > 1e-10:
            h_new = h_current + eta * d
            h_new = project_h(h_new, n_points)
            c5_new = objective(h_new)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
                break
            else:
                eta *= 0.5
        
        # Adaptive step size
        if i - last_improvement > patience:
            eta_max *= 0.99
            last_improvement = i
    
    # Final refinement with local search
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient more carefully here
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_best.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = objective(h_plus)
            grad[j] = (c5_plus - c5_best) / eps
            
        # Update with gradient step
        h_best = h_best - 0.001 * grad
        h_best = project_h(h_best, n_points)
        c5_best = objective(h_best)
        
        if c5_best < c5_current:
            h_current = h_best.copy()
            c5_current = c5_best

    return (h_best.tolist(), c5_best, n_points)
