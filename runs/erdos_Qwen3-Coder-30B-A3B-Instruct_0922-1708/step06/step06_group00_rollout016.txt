```python
import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Variables
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1
    ]
    
    # Objective
    objective = cp.Minimize(t)
    
    # Prepare to build the problem
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Iteratively optimize
    max_iter = 50
    mu = 1.0
    step_size = 0.1
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Temporary fix - don't build full correlation; use direct method
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Barrier method with scaled model
        # Solve with a new problem per iteration
        h_var.value = h.copy()
        t.value = c5  # Initialize t
        
        # Construct overlapping constraints
        # j: lag; for lag k, sum (h[i] * (1 - h[i+k]))
        # - compute overlap for all lags
        # - for k=0, jump by 1 due to indexing differences
        
        overlaps = []
        for k in range(1, n_points):
            # For each lag k (0, 1, ..., n_points-1)
            # compute: sum_{i=0}^{n_points-k-1} h[i] * (1 - h[i+k])
            acc = 0
            for i in range(n_points - k):
                acc += h[i] * (1 - h[i+k])
            overlaps.append(acc * dx)
        
        # Add constraints t >= overlap for all k when k>0
        for i, overlap_value in enumerate(overlaps):
            if i < n_points - 1:
                constraints.append(t >= overlap_value)
        
        # Problem formulation
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, warm_start=True, verbose=False, max_iters=1000)
            new_h = h_var.value
            if new_h is not None:
                # Clip and rescale
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                h = new_h
        except:
            # Fall back to gradient or simple update
            pass
        
        # Update mu slightly (similar to barrier method)
        mu *= 1.05
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx

    return (best_h.tolist(), final_c5, n_points)
```
