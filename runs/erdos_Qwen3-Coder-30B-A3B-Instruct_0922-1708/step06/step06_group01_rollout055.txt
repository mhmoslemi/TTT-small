import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fw_subproblem(gradient, n_points):
    """Solve the Frank-Wolfe linear subproblem."""
    # Find the direction that minimizes gradient * d subject to sum(d) = 0
    # and each component in [0,1]
    d = np.zeros(n_points)
    
    # For minimizing linear term with sum constraint,
    # we can take d = argmin_i <gradient, e_i> where e_i is unit vector
    # Perhaps simpler: we want the direction of maximum decrease in gradient.
    # But since we need sum(d) = 0, we want something that decreases gradient.
    
    # Take d so that we decrease gradient
    i_min = np.argmin(gradient)
    i_max = np.argmax(gradient)
    
    # Direction: essentially make i_max more negative, i_min more positive
    d[i_min] = 1
    d[i_max] = -1
    
    # Project d to meet sum(d) = 0
    # Actually, we want d such that sum(d) = 0 and d in [0,1]^n 
    # and we want to minimize <grad, d> = sum(grad[i] * d[i])
    # This is hard to do directly in Frank-Wolfe framework.
    # So we'll do a crude version: maybe move one at a time
    
    # Simpler concrete approach: 
    # let v = gradient
    # we want to choose d such that sum(d) = 0, 0<=d[i]<=1, and v^T * d is minimized.
    # since the sum of d is 0, if we set d[i] = 1 for one element and -1 for another,
    # this trades one contribution to gradient against another.
    
    # More simply: for the direction d, we want to maximize <grad, d> under constraints
    # The best we can do is:
    # For d[i] = -1 (try to decrease gradient in that direction)
    # and  d[j] = 1
    
    # so we take d = e_i - e_j where i has max gradient and j has min gradient
    # but we need to make sure that we don't exceed the domain [0,1] bounds.
    # so we take an extreme but feasible direction.
    
    # So we compute that:
    # if d = e_i - e_j, then the gradient term is v_i - v_j
    # we want to maximize v_i - v_j subject to d_i >= 0 and d_j <= 0
    # i.e. -1 <= d_i <= 1, but we want to make sure the resulting d 
    # is feasible for the domain constraint.
    
    # This is essentially:
    # If we can make every d_i from 0 to 1, with sum=0 constraint,
    # the Key is finding a d which makes gradient decrease.
    
    # Naive choice for direction that decreases gradient most:
    # Pick one point to increase (positive d) and one to decrease (negative d)
    # to minimize v^T d, we want to pick i with min v_i and j with max v_j
    d = np.zeros(n_points)
    i = np.argmin(gradient)
    j = np.argmax(gradient)
    d[i] = 1.0
    d[j] = -1.0
    
    # Ensure that we do not exceed boundaries in d
    # we keep d as a unit step with sum=0 in the feasible direction
    
    # Rescale if necessary
    # d = d / max(1e-8, np.sum(np.abs(d)))
    
    return d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a structured pattern, e.g., a triangular ramp from 0 to 1 over x=0 to 2
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.tanh((x - 1) * 3))  # ramp from 0 to 1 smoothly
        h_init = repair_h(h_init, n_points)
    
    # Initialize current and best solutions
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    h_best = h_current.copy()
    c5_best = c5_current
    
    patience = 1000
    last_improvement = 0
    max_iter = 200000  # rough estimate
    alpha = 0.8
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_current) / eps
        
        # Solve linear subproblem for Frank-Wolfe step
        # We want min d: <grad, d> such that sum(d) = 0 and 0 <= d[i] <= 1
        d = fw_subproblem(grad, n_points)
        
        # Line search to find step size
        eta_max = 1.0
        # Try to find the best step size
        best_eta = 0.0
        best_c5 = c5_current
        
        eta = eta_max
        while eta > 1e-8:
            h_new = h_current + eta * d
            h_new = repair_h(h_new, n_points)
            corr = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr) * dx
            
            # Accept if better
            if c5_new < best_c5:
                best_eta = eta
                best_c5 = c5_new
            eta *= alpha
            
        # Update with best eta found
        if best_eta > 0:
            h_current = h_current + best_eta * d
            h_current = repair_h(h_current, n_points)
            c5_current = best_c5
            
            # Update best solution
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            # If no progress, try a new random move with small step
            # Heuristic: mutate one variable by a small amount and reproject
            h_new = h_current.copy()
            idx = np.random.randint(0, n_points)
            h_new[idx] += np.random.uniform(-0.02, 0.02)
            h_new = repair_h(h_new, n_points)
            corr = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr) * dx
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i

        if i - last_improvement > patience:
            # Increase patience and step size
            patience = max(500, patience // 2)
            alpha = min(0.99, alpha + 0.001)
            last_improvement = i
            
    return (h_best.tolist(), c5_best, n_points)
