import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Start with initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.full(n_points, 0.5)
    
    # Add smooth noise to initial
    noise = np.random.normal(0, 0.05, n_points)
    h_init = h_init + noise
    h_init = repair_h(h_init, n_points)
    
    best_h = h_init.copy()
    best_value = evaluate_h(h_init, dx)
    
    # Set up parameters
    max_iter = 10000
    learning_rate = 0.001
    stagnation_threshold = 100
    last_improvement = 0
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradients
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - best_value) / eps
        
        # Update h
        h_new = best_h - learning_rate * grad
        h_new = repair_h(h_new, n_points)
        
        new_c5 = evaluate_h(h_new, dx)
        
        if new_c5 < best_value:
            best_h = h_new
            best_value = new_c5
            last_improvement = iter_count
        else:
            # Check for stagnation
            if iter_count - last_improvement > stagnation_threshold:
                # Inject diversity
                h_new = np.random.rand(n_points)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                new_c5 = evaluate_h(h_new, dx)
                if new_c5 < best_value:
                    best_h = h_new
                    best_value = new_c5
                    last_improvement = iter_count
        
        # Adaptive learning rate
        if iter_count > 0 and iter_count % 100 == 0:
            learning_rate *= 0.95
    
    # Final refinement
    for i in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - best_value) / eps
        
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        best_value = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_value, n_points)
