import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Adagrad-style adaptive learning rate
    learning_rate = 5e-4
    momentum = 0.9
    grad_sum = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Dual variables for each lag
    dual_lags = np.zeros(2*n_points - 1)
    dual_temp = 0.1
    
    # Main optimization loop
    iteration = 0
    stagnation_count = 0
    max_stagnation = 1000
    
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        
        # Optimizer: compute all overlaps via convolution
        overlaps = correlate(current_h, 1 - current_h, mode='full')
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]
        
        # Update dual variables with variational inference
        if iteration % 10 == 0:
            dual_weights = np.exp(-overlaps / dual_temp)
            dual_weights /= np.sum(dual_weights)
            dual_lags = dual_weights
        
        # Compute gradient using dual variables
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(-n_points+1, n_points):
                if i + k >= 0 and i + k < n_points:
                    grad[i] += dual_lags[k + n_points - 1] * (1 - current_h[i + k])
        
        # Adagrad-style gradient descent
        grad_sum += grad ** 2
        adjusted_lr = learning_rate / (np.sqrt(grad_sum) + 1e-7)
        v = momentum * v - adjusted_lr * grad
        current_h += v
        
        # Repair to feasible solution
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)

        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1

        # Update dual temperature
        dual_temp *= 0.999
        
        # Decrease learning rate
        if iteration % 500 == 0:
            learning_rate *= 0.99
            
        # Reset if stuck
        if stagnation_count > max_stagnation:
            current_h = h_init.copy()
            best_c5 = evaluate_h(current_h, dx)
            best_h = current_h.copy()
            stagnation_count = 0
            learning_rate = 5e-4
            
    return (best_h.tolist(), best_c5, n_points)
