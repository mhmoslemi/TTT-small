import numpy as np
import time

def evaluate_h(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h0))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h0)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Fill remaining population
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # DE parameters
    F = 1.0
    CR = 0.9
    stagnation_count = 0
    best_c5 = float('inf')
    best_h = None
    
    # Evolution loop
    gen = 0
    max_gens = 1000
    while gen < max_gens and time.time() - start_time < budget_s - 5:
        new_population = []
        
        for i in range(pop_size):
            # Select three distinct vectors
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            v = a + F * (b - c)
            v = np.clip(v, 0, 1)
            
            # Normalize
            v = repair_h(v, n_points)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = v[j]
            
            # Evaluate
            c5 = evaluate_h(trial, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = trial.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
                
            new_population.append(trial)
            
        population = new_population
        gen += 1
        
        # Adaptive parameter updates
        if stagnation_count > 50:
            F = min(1.5, F + 0.1)
            stagnation_count = 0
    
    # Final refinement of best solution
    if best_h is not None:
        h = best_h.copy()
        for _ in range(500):
            if time.time() - start_time > budget_s - 2:
                break
            eps = 1e-6
            grad = np.zeros(n_points)
            c5_eval = evaluate_h(h, dx)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                grad[i] = (c5_plus - c5_eval) / eps
            h = h - 0.0002 * grad
            h = repair_h(h, n_points)
            new_c5 = evaluate_h(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
