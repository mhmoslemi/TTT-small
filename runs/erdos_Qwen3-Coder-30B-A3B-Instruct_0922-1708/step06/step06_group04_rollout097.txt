import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a sieve-like pattern in frequency domain
    freqs = np.fft.fftfreq(n_points, 1.0/n_points)
    # High frequencies get higher amplitudes
    spectrum = np.exp(-0.1 * np.abs(freqs)**1.5)
    spectrum[0] = 0  # DC component set to 0
    # Inverse FFT to get real signal
    h = np.fft.ifft(spectrum).real
    # Rescale to meet constraints
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Initialize using either provided construction or sieve pattern
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        h_init = fourier_initialization(n_points)

    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    memory_decay = 0.95
    max_iter = 2000
    perturb_scale = 0.2
    default_learning_rate = 0.0001
    diversity_threshold = 50

    # Best solution tracking
    current_h = h_init.copy()
    current_c5 = evaluate_overlap(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5

    iter_count = 0
    last_improvement = 0

    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate neighbor by perturbing current best
        h_new = current_h.copy()
        # Random perturbation
        perturb = np.random.normal(0, perturb_scale, n_points)
        h_new += perturb
        
        # Project to feasible region
        h_new = project_h(h_new, n_points)
        
        # Check tabu list
        move_key = tuple((h_new[i] - current_h[i]) for i in range(n_points) if abs(h_new[i] - current_h[i]) > 1e-6)
        if move_key in tabu_list:
            # Accept with low probability
            if np.random.rand() > 0.2:
                iter_count += 1
                continue
        
        # Evaluate
        new_c5 = evaluate_overlap(h_new, dx)
        
        if new_c5 < current_c5 or np.random.rand() < 0.05:
            # Accept new solution
            current_h = h_new.copy()
            current_c5 = new_c5
            
            # Update best if improved
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                last_improvement = iter_count
                
            # Update tabu list
            tabu_list.append(move_key)
            if len(tabu_list) > tabu_size:
                # Remove oldest entry
                tabu_list.pop(0)
        
        # Decay tabu list entries (soft forgetting)
        if len(tabu_list) > 0:
            for i in range(len(tabu_list)):
                if np.random.rand() < 1 - memory_decay:
                    tabu_list.pop(i)
                    break
        
        # Inject diversity if stuck
        if iter_count - last_improvement > diversity_threshold:
            h_new = fourier_initialization(n_points)
            current_h = h_new
            current_c5 = evaluate_overlap(current_h, dx)
            last_improvement = iter_count
            
            # Clear tabu list
            tabu_list = []

        iter_count += 1

    # Gradient descent refinement
    h_refined = best_h.copy()
    prev_c5 = best_c5
    learning_rate = default_learning_rate
    for iteration in range(500):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient numerically
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_c5) / eps

        # Update
        h_refined = h_refined - learning_rate * grad
        h_refined = project_h(h_refined, n_points)
        new_c5 = evaluate_overlap(h_refined, dx)
        
        # Line search for better step
        if new_c5 < prev_c5:
            prev_c5 = new_c5
            best_h = h_refined.copy()
            best_c5 = new_c5
        else:
            learning_rate *= 0.9
            
    return (best_h.tolist(), best_c5, n_points)
