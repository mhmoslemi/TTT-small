import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Barrier method optimization
    mu = 1.0
    max_iter = 1000
    convergence_threshold = 1e-6
    mu_factor = 1.1
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current C5
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Formulate convex optimization problem
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Bound constraints
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Overlap constraints
        # For each k in valid range, we add a constraint t >= overlap_k
        for k in range(n_points):
            # Compute overlap for lag k
            if k == 0:
                overlap = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
            else:
                overlap = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
            constraints.append(t >= overlap * dx)
        
        # Barrier terms for bounds
        # -ln(h) - ln(1-h) for h in (0,1)
        barrier_terms = -mu * (cp.sum(cp.entr(h_var)) + cp.sum(cp.entr(1 - h_var)))
        objective = cp.Minimize(t + barrier_terms)
        
        # Create and solve problem
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                h_new = np.clip(h_new, 1e-8, 1 - 1e-8)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
                # Increase barrier parameter
                mu *= mu_factor
            else:
                # If optimization fails, break
                break
        except:
            # If optimization fails, break
            break
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
