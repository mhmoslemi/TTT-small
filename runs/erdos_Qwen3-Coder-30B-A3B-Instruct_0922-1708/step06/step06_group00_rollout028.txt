import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50000
    initial_temp = 0.1
    cooling_rate = 0.999
    tabu_size = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create initial binary vector with roughly half 1s
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    # Binary representation (0 or 1)
    h_binary = (h > 0.5).astype(float)
    
    # Track tabu list for recent moves
    tabu_list = []
    
    # Initialize best solution
    best_h = h_binary.copy()
    corr = correlate(h_binary, 1 - h_binary, mode='full')
    best_c5 = np.max(corr) * dx
    current_c5 = best_c5
    
    # Simulated annealing with tabu search
    temp = initial_temp
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate candidate solution by flipping a random bit
        candidate_h = h_binary.copy()
        flip_idx = np.random.randint(0, n_points)
        candidate_h[flip_idx] = 1 - candidate_h[flip_idx]
        
        # Check tabu list
        if (flip_idx, h_binary[flip_idx]) in tabu_list:
            continue
            
        # Ensure sum constraint (need to adjust another bit)
        ones_count = np.sum(candidate_h)
        if ones_count > n_points / 2:
            # Need to flip a 1 to 0
            idx_to_flip = np.random.choice(np.where(candidate_h == 1)[0])
            candidate_h[idx_to_flip] = 0
        elif ones_count < n_points / 2:
            # Need to flip a 0 to 1
            idx_to_flip = np.random.choice(np.where(candidate_h == 0)[0])
            candidate_h[idx_to_flip] = 1
            
        # Compute overlap for candidate
        corr = correlate(candidate_h, 1 - candidate_h, mode='full')
        candidate_c5 = np.max(corr) * dx
        
        # Accept or reject based on simulated annealing
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h_binary = candidate_h.copy()
            current_c5 = candidate_c5
            
            # Update tabu list
            tabu_list.append((flip_idx, h_binary[flip_idx]))
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
                
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_binary.copy()
        
        # Cool down temperature
        temp *= cooling_rate
    
    # Convert final result to h values preserving sum constraint
    h_final = best_h.copy()
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Final validation
    corr = correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_final.tolist(), final_c5, n_points)
