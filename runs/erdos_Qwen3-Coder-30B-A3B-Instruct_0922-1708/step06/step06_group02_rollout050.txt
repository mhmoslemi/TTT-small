import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        h = h_init.copy()
    else:
        h = np.ones(n_points) * 0.5
        h = repair_h(h, n_points)
    
    # Multi-resolution optimization
    resolutions = [10, 20, 40, 100]
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Coarsen to current resolution
        if res < n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_coarse = np.interp(x_new, x_orig, h)
            h_coarse = repair_h(h_coarse, res)
        else:
            h_coarse = h.copy()
            
        # Gradient based optimization on coarse h
        def obj_and_grad(h_coarse_flat):
            h_coarse = h_coarse_flat.copy()
            h_coarse = repair_h(h_coarse, len(h_coarse))
            # Interpolate back to full resolution
            x_orig = np.linspace(0, 2, len(h_coarse))
            x_full = np.linspace(0, 2, n_points)
            h_full = np.interp(x_full, x_orig, h_coarse)
            h_full = repair_h(h_full, n_points)
            # Evaluate
            corr = correlate(h_full, 1 - h_full, mode='full')
            max_corr = np.max(corr)
            c5 = max_corr * dx
            
            # Gradient estimation
            eps = 1e-5
            grad = np.zeros_like(h_coarse_flat)
            for i in range(len(h_coarse_flat)):
                h_plus = h_coarse_flat.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, len(h_plus))
                x_orig = np.linspace(0, 2, len(h_plus))
                x_full = np.linspace(0, 2, n_points)
                h_full_plus = np.interp(x_full, x_orig, h_plus)
                h_full_plus = repair_h(h_full_plus, n_points)
                corr_plus = correlate(h_full_plus, 1 - h_full_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            
            return c5, grad
        
        # Optimize coarse h
        try:
            res_coarse = minimize(
                lambda h_flat: obj_and_grad(h_flat)[0], 
                h_coarse.flatten(), 
                method='L-BFGS-B',
                jac=lambda h_flat: obj_and_grad(h_flat)[1],
                bounds=[(0, 1)] * len(h_coarse),
                options={'maxiter': 100}
            )
            
            if res_coarse.success:
                h_coarse = res_coarse.x.copy()
                h_coarse = repair_h(h_coarse, len(h_coarse))
                x_orig = np.linspace(0, 2, len(h_coarse))
                x_full = np.linspace(0, 2, n_points)
                h = np.interp(x_full, x_orig, h_coarse)
                h = repair_h(h, n_points)
                
                current_value = evaluate_h(h, dx)
                if current_value < best_value:
                    best_value = current_value
                    best_h = h.copy()
                    
        except Exception:
            pass
        
        h = best_h.copy()
    
    # Final intensive gradient descent
    def final_obj_and_grad(h_flat):
        h_full = h_flat.copy()
        h_full = repair_h(h_full, n_points)
        corr = correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        c5 = max_corr * dx
        
        # Gradient estimation
        eps = 1e-6
        grad = np.zeros_like(h_flat)
        for i in range(len(h_flat)):
            h_plus = h_flat.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
            
        return c5, grad
    
    # Refine final solution
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        try:
            res = minimize(
                lambda h_flat: final_obj_and_grad(h_flat)[0],
                best_h.flatten(),
                method='L-BFGS-B',
                jac=lambda h_flat: final_obj_and_grad(h_flat)[1],
                bounds=[(0, 1)] * n_points,
                options={'maxiter': 10}
            )
            if res.success:
                best_h = res.x.copy()
                best_h = repair_h(best_h, n_points)
                best_value = evaluate_h(best_h, dx)
            else:
                break
        except:
            break
    
    return (best_h.tolist(), best_value, n_points)
