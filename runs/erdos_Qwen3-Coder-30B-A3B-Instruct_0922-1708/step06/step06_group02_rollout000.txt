import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    h = repair_h(h, n_points)
    
    # Store best solution
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Hierarchical optimization
    levels = [10, 20, 40, 100]
    for level in levels:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Downsample to current level
        if level < n_points:
            h_coarse = np.array([np.mean(h[i:i*level//n_points]) 
                               for i in range(level)])
            h_coarse = repair_h(h_coarse, level)
        else:
            h_coarse = h.copy()
            level = n_points
            
        # Optimization on coarse grid
        def objective(h_coarse):
            h_full = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(h_coarse)), h_coarse)
            h_full = repair_h(h_full, n_points)
            return evaluate_h(h_full, dx)
            
        def constraint_func(h_coarse):
            return np.sum(h_coarse) * (2.0 / level) - 1.0
            
        # Create constraint dictionary
        cons = {'type': 'eq', 'fun': constraint_func}
        
        # Optimize with bounds
        bounds = [(0, 1) for _ in range(len(h_coarse))]
        result = minimize(objective, h_coarse, method='SLSQP', 
                         bounds=bounds, constraints=cons, 
                         options={'maxiter': 100})
        
        if result.success:
            h_coarse = result.x
        else:
            h_coarse = h_coarse
            
        # Upsample to next level or final
        if level < n_points:
            h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                                  np.linspace(0, 2, len(h_coarse)), h_coarse)
            h = repair_h(h_upsampled, n_points)
        else:
            h = repair_h(h_coarse, n_points)
            
        value = evaluate_h(h, dx)
        if value < best_value:
            best_value = value
            best_h = h.copy()

    # Fine tuning with gradient descent
    def objective_grad(h):
        return evaluate_h(h, dx)
    
    # Compute gradient and improve
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - objective_grad(h)) / eps
            
        h = h - 0.0001 * grad
        h = repair_h(h, n_points)
        value = evaluate_h(h, dx)
        if value < best_value:
            best_value = value
            best_h = h.copy()
            
    return (best_h.tolist(), best_value, n_points)
