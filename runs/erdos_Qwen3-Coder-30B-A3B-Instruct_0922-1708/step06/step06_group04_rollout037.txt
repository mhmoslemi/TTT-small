import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum=50):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to 50
    h = project_h(h, 50)
    
    # Parameters
    lambda_reg = 0.01
    alpha = 0.1
    eps = 1e-6
    max_iter = 1000
    patience = 100
    best_value = evaluate_h(h, dx)
    best_h = h.copy()
    no_improve_count = 0
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current objective value
        current_value = evaluate_h(h, dx)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, 50)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_value) / eps
        
        # Compute L1 regularization gradient
        l1_grad = lambda_reg * np.sign(h)
        
        # Update with regularization
        h_new = h - alpha * (grad + l1_grad)
        h_new = project_h(h_new, 50)
        
        # Evaluate new value
        new_value = evaluate_h(h_new, dx)
        
        # Backtracking line search
        if new_value <= current_value:
            h = h_new
            if new_value < best_value:
                best_h = h.copy()
                best_value = new_value
                no_improve_count = 0
            else:
                no_improve_count += 1
            # Reduce learning rate
            alpha = max(alpha * 0.8, 1e-4)
        else:
            # Increase learning rate on rejection
            alpha = min(alpha * 1.2, 1.0)
            no_improve_count += 1
            
        # Check for stagnation
        if no_improve_count >= patience:
            # Inject perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_h(h, 50)
            no_improve_count = 0
            
    return (best_h.tolist(), best_value, n_points)
