import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialization with parent or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Convex relaxation stage (short)
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Try CVXPY relaxation for starting point
    try:
        h_cvx = cp.Variable(n_points)
        # Define objective that approximates max with squared terms
        # This is a heuristic proxy to initial guidance
        # For simplicity, we'll just project our current solution
        h_repaired = repair_h(h, n_points)
        h = h_repaired.copy()
        current_c5 = evaluate_h(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    except Exception:
        pass  # Continue with current h if CVXPY fails
    
    # Refinement stage: adaptive gradient descent
    max_iter = 1000
    lr_base = 0.01
    beta = 0.9
    stagnation_limit = 50
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
            
        # Adaptive learning rate
        lr = lr_base / (1.0 + iteration ** beta)
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        # Evaluate and update best
        c5 = evaluate_h(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Occasionally perturb
        if stagnation_count > stagnation_limit or np.random.rand() < 0.02:
            h += np.random.normal(0, 0.005, n_points)
            h = repair_h(h, n_points)
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
    
    # Final optimization with refined approach
    for iteration in range(200):
        if time.time() - start_time > budget_s - 2:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
            
        # Smaller step size for final stage
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
    return (best_h.tolist(), best_c5, n_points)
