import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum=50.0):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def compute_c5(h, dx):
    """Compute C5 value for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to meet constraint
    h = project_h(h, n_points / 2)
    
    # Add small noise to break symmetry
    h += np.random.normal(0, 0.01, n_points)
    h = project_h(h, n_points / 2)
    
    # Parameters
    learning_rate = 0.1
    lambda_reg = 0.01
    max_iter = 1000
    improvement_threshold = 1e-4
    stagnation_limit = 100
    
    # Optimization loop
    last_improvement = 0
    prev_c5 = compute_c5(h, dx)
    best_c5 = prev_c5
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient numerically
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points / 2)
            c5_plus = compute_c5(h_plus, dx)
            grad[i] = (c5_plus - prev_c5) / eps
        
        # Add L1 regularization gradient
        l1_grad = lambda_reg * np.sign(h)
        
        # Update step
        h_new = h - learning_rate * (grad + l1_grad)
        
        # Project to feasible set
        h_new = project_h(h_new, n_points / 2)
        
        # Backtracking line search
        c5_new = compute_c5(h_new, dx)
        
        # Accept or reject
        if c5_new < prev_c5:
            h = h_new
            prev_c5 = c5_new
            
            if prev_c5 < best_c5:
                best_h = h.copy()
                best_c5 = prev_c5
                last_improvement = iteration
                
            # Reduce learning rate if we're making progress
            learning_rate = max(1e-4, learning_rate * 0.95)
        else:
            # Increase learning rate if we're not improving
            learning_rate = min(1.0, learning_rate * 1.1)
            
        # Stagnation detection
        if iteration - last_improvement > stagnation_limit:
            # Try to break stagnation by reinitializing
            h = np.random.rand(n_points) * 0.5
            h = project_h(h, n_points / 2)
            prev_c5 = compute_c5(h, dx)
            last_improvement = iteration
            
    return (best_h.tolist(), best_c5, n_points)
