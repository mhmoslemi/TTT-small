import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h = (h_init > 0.5).astype(int)
        current_sum = np.sum(h)
        target_sum = n_points // 2
        if current_sum != target_sum:
            # Adjust to get exactly target_sum ones
            diff = target_sum - current_sum
            if diff > 0:
                # Find zeros to flip to ones
                zero_indices = np.where(h == 0)[0]
                for i in range(min(diff, len(zero_indices))):
                    h[zero_indices[i]] = 1
            elif diff < 0:
                # Find ones to flip to zeros
                one_indices = np.where(h == 1)[0]
                for i in range(min(-diff, len(one_indices))):
                    h[one_indices[i]] = 0
    else:
        # Default initialization
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # MIP solver setup
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points // 2]
    
    # For each shift k, create overlap constraint
    # Note: correlation is symmetric, but we'll ensure correctness
    for k in range(-n_points + 1, n_points):
        # Find valid indices for this k
        valid_indices = []
        for i in range(n_points):
            if 0 <= i + k < n_points:
                valid_indices.append((i, i + k))
        
        if len(valid_indices) > 0:
            # Sum of h[i] * (1 - h[j]) over valid pairs
            overlap_expr = cp.sum([h_var[i] * (1 - h_var[j]) for i, j in valid_indices])
            # Ensure the overlap doesn't exceed t * (n_points / 2)
            constraints.append(overlap_expr <= t * (n_points / 2))
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solver parameters
    max_time = budget_s - 10
    if max_time <= 0:
        # If not enough time, just return initial h
        final_c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
        return (h.tolist(), final_c5, n_points)
    
    # Solve
    try:
        prob.solve(solver=cp.CBC, time_limit=max_time, gap_limit=0.01)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = np.array(h_var.value).astype(int)
            if np.sum(h_opt) != n_points // 2:
                # Adjust if sum is incorrect, flip some values
                current_sum = np.sum(h_opt)
                target_sum = n_points // 2
                diff = target_sum - current_sum
                if diff > 0:
                    # Flip zeros to ones
                    zero_indices = np.where(h_opt == 0)[0]
                    for i in range(min(diff, len(zero_indices))):
                        h_opt[zero_indices[i]] = 1
                elif diff < 0:
                    # Flip ones to zeros
                    one_indices = np.where(h_opt == 1)[0]
                    for i in range(min(-diff, len(one_indices))):
                        h_opt[one_indices[i]] = 0
            h = h_opt
        else:
            # Return best known h in case of solver failure
            pass
    except Exception as e:
        # Fall back to the original h
        pass

    # Final computation
    corr = correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h.tolist(), final_c5, n_points)
