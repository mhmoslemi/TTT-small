import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def create_initial_h(n_points):
    """Create an initial step function."""
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        return h_init
    else:
        return np.ones(n_points) * 0.5

def evolve_step(h, mutation_scale, n_points):
    """Apply evolutionary mutation."""
    noise = np.random.normal(0, mutation_scale, n_points)
    h_mut = h + noise
    h_mut = repair_h(h_mut, n_points)
    return h_mut

def crossover_h(h1, h2, crossover_rate=0.3):
    """Perform crossover between two solutions."""
    mask = np.random.rand(len(h1)) < crossover_rate
    h_child = np.where(mask, h1, h2)
    return h_child

def optimize_resolution(h_start, n_points, max_iter, dx):
    """Optimize at given resolution."""
    population_size = 50
    mutation_scale = 0.1 / np.sqrt(n_points)
    survival_rate = 0.2
    
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = evolve_step(h_start, mutation_scale, n_points)
        population.append(h)
    
    best_h = None
    best_value = float('inf')
    
    for iter_count in range(max_iter):
        # Evaluate population
        fitness = []
        for h in population:
            value = evaluate_h(h, dx)
            fitness.append(value)
            if value < best_value:
                best_value = value
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:int(population_size * survival_rate)]
        population = [population[i] for i in top_indices]
        
        # Create new population
        new_population = population.copy()
        while len(new_population) < population_size:
            # Selection with replacement
            p1 = population[np.random.randint(len(population))]
            p2 = population[np.random.randint(len(population))]
            child = crossover_h(p1, p2)
            child = evolve_step(child, mutation_scale, n_points)
            new_population.append(child)
        
        population = new_population
        
        if iter_count % 20 == 0:
            best_value = min(fitness)
        
        if best_value < 0.3807:
            break
    
    return best_h, best_value

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Coarse resolution optimization
    coarse_points = [10, 20, 40, 100]
    best_h = None
    best_value = float('inf')
    
    # Initialize
    prev_h = create_initial_h(coarse_points[0])
    dx = 2.0 / coarse_points[0]
    
    for i, n_points in enumerate(coarse_points):
        if time.time() - start_time > budget_s - 10:
            break
            
        dx = 2.0 / n_points
        
        if i == 0:
            init_h = prev_h.copy()
        else:
            init_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(prev_h)), prev_h)
            init_h = repair_h(init_h, n_points)
            
        # Optimize
        n_iter = 100 if i < 3 else 200
        h_opt, value = optimize_resolution(init_h, n_points, n_iter, dx)
        
        if value < best_value:
            best_h = h_opt.copy()
            best_value = value
            
        prev_h = h_opt.copy()
        
        # Early termination if already good enough
        if best_value < 0.3807:
            break
    
    n_points = 100
    
    # Intensive gradient descent on best solution
    if best_h is not None:
        h_refined = best_h.copy()
        current_value = best_value
        learning_rate = 0.001
        eps = 1e-6
        
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
                
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
                
            h_refined = h_refined - learning_rate * grad
            h_refined = repair_h(h_refined, n_points)
            
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            current_value = np.max(corr) * dx
            
            if current_value < best_value:
                best_value = current_value
                best_h = h_refined.copy()
    
    return (best_h.tolist(), best_value, n_points)
