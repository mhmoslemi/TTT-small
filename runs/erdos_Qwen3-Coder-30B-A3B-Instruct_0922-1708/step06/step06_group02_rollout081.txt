import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def create_initial_h(n_points):
    """Create initial h values"""
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        return h_init
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        return h_init

def mutate_h(h, mutation_scale):
    """Mutate h with Gaussian noise"""
    noise = np.random.normal(0, mutation_scale, len(h))
    mutated = h + noise
    return repair_h(mutated, len(h))

def crossover_h(h1, h2, crossover_rate=0.3):
    """Crossover two h vectors"""
    mask = np.random.rand(len(h1)) < crossover_rate
    child = np.where(mask, h1, h2)
    return child

def optimize_stage(h_initial, n_points, generations, mutation_scale):
    """Optimize a stage of the evolutionary process"""
    population = [h_initial]
    for _ in range(49):  # Create 50 total individuals
        h = mutate_h(h_initial, mutation_scale)
        population.append(h)
    
    for _ in range(generations):
        # Evaluate population
        fitness = [evaluate_h(h, 2.0 / n_points) for h in population]
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        
        # Keep top 20% (10 individuals)
        top_individuals = population[:10]
        
        # Create new generation
        new_population = top_individuals.copy()
        while len(new_population) < 50:
            # Tournament selection
            idx1, idx2 = np.random.choice(10, 2, replace=False)
            parent1 = top_individuals[idx1]
            parent2 = top_individuals[idx2]
            
            # Crossover
            child = crossover_h(parent1, parent2)
            
            # Mutate
            child = mutate_h(child, mutation_scale)
            
            new_population.append(child)
            
        population = new_population
        
    # Return best individual
    best_fitness = [evaluate_h(h, 2.0 / n_points) for h in population]
    best_idx = np.argmin(best_fitness)
    return population[best_idx], best_fitness[best_idx]

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize coarse resolution
    coarse_points = 10
    h_coarse = create_initial_h(coarse_points)
    dx_coarse = 2.0 / coarse_points
    
    # Optimize at coarse resolution
    t0 = time.time()
    h_best_coarse, _ = optimize_stage(h_coarse, coarse_points, 100, 0.1)
    elapsed = time.time() - t0
    
    # If we're running out of time, return result now
    if elapsed > budget_s - 2:
        return (h_best_coarse.tolist(), evaluate_h(h_best_coarse, dx_coarse), coarse_points)
        
    # Refinement stages
    resolutions = [20, 40, 100]
    prev_h = h_best_coarse
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Interpolate to higher resolution
        x_coarse = np.linspace(0, 2, len(prev_h))
        x_fine = np.linspace(0, 2, n_points)
        h_fine = np.interp(x_fine, x_coarse, prev_h)
        
        # Repair to sum constraint
        h_fine = repair_h(h_fine, n_points)
        
        # Add some random variants
        additional_h = []
        for _ in range(10):
            h_random = np.random.rand(n_points)
            h_random = h_random * (n_points / 2) / np.sum(h_random)
            additional_h.append(h_random)
            
        # Optimize at this resolution
        h_best, _ = optimize_stage(h_fine, n_points, 50, 0.1 / np.sqrt(n_points))
        prev_h = h_best
        
    # Final local refinement
    dx_fine = 2.0 / len(prev_h)
    h_refined = prev_h.copy()
    best_value = evaluate_h(h_refined, dx_fine)
    
    for i in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(len(h_refined))
        for j in range(len(h_refined)):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, len(h_refined))
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx_fine
            grad[j] = (c5_plus - best_value) / eps
            
        h_refined = h_refined - 0.001 * grad
        h_refined = repair_h(h_refined, len(h_refined))
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_value = np.max(corr) * dx_fine
        
    return (h_refined.tolist(), best_value, len(h_refined))
