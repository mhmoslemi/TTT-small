import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum=50.0):
    """Project h to feasible region: [0,1] and sum=h_target"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def compute_c5(h, dx):
    """Compute C5 value from h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to target sum
    h = project_h(h, target_sum)
    
    # Regularization parameter
    lambda_reg = 0.01
    
    # Initial learning rate
    alpha = 0.1
    
    # Optimization parameters
    max_iter = 1000
    patience = 100
    min_improvement = 1e-4
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current c5 and gradient
        current_c5 = compute_c5(h, dx)
        
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Add L1 regularization gradient
        reg_grad = lambda_reg * np.sign(h)
        grad += reg_grad
        
        # Update
        h_new = h - alpha * grad
        h_new = project_h(h_new, target_sum)
        
        # Compute new c5
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                last_improvement = iteration
            alpha *= 0.8  # Decrease learning rate
        else:
            alpha *= 1.2  # Increase learning rate
        
        # Bound alpha
        alpha = np.clip(alpha, 1e-4, 1.0)
        
        # Early stopping check
        if iteration - last_improvement > patience:
            break

    return (best_h.tolist(), best_c5, n_points)
