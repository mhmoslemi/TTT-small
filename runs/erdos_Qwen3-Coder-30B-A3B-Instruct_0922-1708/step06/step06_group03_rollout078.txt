import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradients(h, dx, max_k):
    """Compute gradients for the maximum overlap term."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute all correlations first
    corr = correlate(h, 1 - h, mode='full')
    max_corr_idx = np.argmax(corr)
    max_corr_val = corr[max_corr_idx]
    
    # For finite differences
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_corr_idx] - max_corr_val) / eps
    
    return grad
