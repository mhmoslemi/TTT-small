import numpy as np
import time
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    # Use scipy.signal.correlate for efficient computation
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Rescale to meet integral constraint exactly
        h = h * target_sum / current_sum
    return h

    h = h * (n_points / 2) / np.sum(h)

    best_c5 = evaluate_h(h, dx)
