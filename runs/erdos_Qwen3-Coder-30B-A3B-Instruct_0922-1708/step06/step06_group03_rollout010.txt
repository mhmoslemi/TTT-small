import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent with gradient descent
    dual_vars = np.ones(2 * n_points - 1)  # λ_k for k in [-n+1, n-1]
    dual_vars /= np.sum(dual_vars)  # Normalize
    learning_rate = 5e-4
    temp = 0.1
    stagnation_count = 0
    max_stagnation = 50
    
    for iteration in range(30000):  # Up to 30k iterations
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all overlaps
        corr = correlate(current_h, 1 - current_h, mode='full')
        overlaps = corr * dx  # Normalize overlaps
        max_overlap_idx = np.argmax(overlaps)
        
        # Update dual variables using variational inference
        # λ_k ∝ exp(-overlap_k / temp)
        dual_vars = np.exp(-overlaps / temp)
        dual_vars /= np.sum(dual_vars)
        
        # Compute gradient of Lagrangian w.r.t. h
        grad = np.zeros(n_points)
        # For each position in h
        for i in range(n_points):
            for k in range(-n_points + 1, n_points):
                # k is the shift
                # overlap_k = ∫ h(x)(1 - h(x+k)) dx
                # Compute ∂/∂h[i] of overlap_k term
                k_idx = k + n_points - 1
                if k_idx < 0 or k_idx >= len(overlaps):
                    continue
                # contribution to gradient at i
                if i + k >= 0 and i + k < n_points:
                    # h[i+k] contributes to overlap_k
                    grad[i] += dual_vars[k_idx] * (1 - current_h[i+k])
        
        # Update h
        current_h = current_h - learning_rate * grad
        current_h = repair_h(current_h, n_points)
        
        # Update parameters
        current_c5 = evaluate_h(current_h, dx)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive updates
        learning_rate *= 0.99995
        temp *= 0.9999
        
        if stagnation_count > max_stagnation:
            # Fallback to initial
            current_h = h_init.copy()
            current_c5 = evaluate_h(current_h, dx)
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
