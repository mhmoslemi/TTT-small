import numpy as np
import time
import cvxpy as cp
from scipy import signal

def evaluate_h(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    learning_rate = 0.001
    max_iter = 5000
    stagnation_limit = 50
    stagnation_count = 0
    
    # Quantized adaptive learning rates
    lr_schedule = [0.001, 0.0005, 0.0002]
    lr_idx = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient approximation using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        
        # Update with adaptive learning
        h = h - lr_schedule[lr_idx] * grad
        h = repair_h(h, n_points)
        
        # Evaluate new solution
        c5 = evaluate_h(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > stagnation_limit:
            if lr_idx < len(lr_schedule) - 1:
                lr_idx += 1
                stagnation_count = 0
            else:
                # Add noise to escape local minima
                noise = np.random.normal(0, 0.001, n_points)
                h += noise
                h = repair_h(h, n_points)
                stagnation_count = 0

    # Final refinement with lower learning rate
    for _ in range(1000):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-5
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5

    return (best_h.tolist(), best_c5, n_points)
