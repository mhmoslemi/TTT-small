import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Gradient-based optimization parameters
    learning_rate = 1e-4
    perturb_mag = 0.05
    max_iter = 100000
    progress_count = 0
    max_progress = 100
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient for the maximum overlap term
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient for current_h
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Gradient is based on contribution of h[j] to max overlap
            # Work backward from k_max to see which indices contribute
            grad[j] = 1 - current_h[(j - k_max) % n_points]
        
        # Update with gradient descent
        new_h = current_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject the update
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                progress_count = 0
            else:
                progress_count += 1
        else:
            progress_count += 1
            
        # Periodic perturbations to escape local minima
        if i % 500 == 499 and progress_count > 50:
            perturb_h = current_h.copy()
            idx = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            for j in idx:
                delta = np.random.uniform(-perturb_mag, perturb_mag)
                perturb_h[j] += delta
            perturb_h = repair_h(perturb_h, n_points)
            perturb_c5 = evaluate_h(perturb_h, dx)
            
            if perturb_c5 < current_c5:
                current_h = perturb_h
                current_c5 = perturb_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    progress_count = 0
                else:
                    progress_count += 1
            
            # Decay perturbation magnitude
            perturb_mag *= 0.999
            
        # Decrease learning rate and perturb magnitude
        if i % 100 == 99:
            learning_rate *= 0.99
            perturb_mag *= 0.999
            
        # Adaptive cooling
        if progress_count > max_progress:
            learning_rate = max(learning_rate * 0.9, 1e-8)
            progress_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
