import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def project_h(h, n_points):
    """Project h to feasible set."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.full_like(h, 1.0 / n_points)
    return h

        # Coordinate-wise gradient descent
        h_new = h_current.copy()

            h_plus = h_current.copy()
            h_minus = h_current.copy()
            
            # Perturb j-th element
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            
            # Project to feasible set
            h_plus = project_h(h_plus, n_points)
            h_minus = project_h(h_minus, n_points)

            c5_plus = evaluate_h(h_plus, dx)

            # Update coordinate

        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)

        if c5_new < c5_current:

            c5_current = c5_new
