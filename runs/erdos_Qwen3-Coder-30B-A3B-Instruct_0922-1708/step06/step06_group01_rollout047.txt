import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute finite difference gradient of C5."""
    n = len(h)
    grad = np.zeros(n)
    h_base = h.copy()
    c5_base = evaluate_h(h_base, dx)
    
    for i in range(n):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

def trust_region_step(h, dx, grad, trust_radius):
    """Compute step using projected trust region approach."""
    n = len(h)
    
    # Project gradient onto feasible set and sum constraint
    # We solve: min ||d|| s.t. 0 <= h+d <= 1, sum(d) = 0
    # This is a quadratic program; could use more sophisticated solver
    # Here, we use a simple projected gradient method
    
    d = -grad
    # Project onto feasibility constraints
    d_proj = np.zeros(n)
    for i in range(n):
        if h[i] + d[i] < 0:
            d_proj[i] = -h[i]
        elif h[i] + d[i] > 1:
            d_proj[i] = 1 - h[i]
        else:
            d_proj[i] = d[i]
    
    # Project onto sum=0 constraint
    d_sum = np.sum(d_proj)
    d_proj = d_proj - d_sum / n
    
    # Trim to trust region
    norm_d = np.linalg.norm(d_proj)
    if norm_d > trust_radius and norm_d > 1e-10:
        d_proj = d_proj * trust_radius / norm_d
    
    return d_proj
