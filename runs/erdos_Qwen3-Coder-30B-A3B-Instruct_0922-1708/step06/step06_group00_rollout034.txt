        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Start with binary construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = float('inf')

    # Cutting plane method
    max_iter = 100
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlaps for all shifts
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        k_max = max_lag_idx - (len(h) - 1)
        
        # Use integer programming approach to solve for upper bound
        # Minimize t such that: t >= sum(h[i] * (1 - h[i+k])) for all valid k
        # This becomes: t >= sum(h[i] - h[i]*h[i+k]) = sum(h[i]) - sum(h[i]*h[i+k])
        # Thus: sum(h[i]*h[i+k]) >= sum(h[i]) - t
        # Apply quadratic constraints via CVXPY
        
        # Skip cutting plane optimization for now, and just try local search
        
        # Try to improve with gradient-based method on maximum overlap
        c5_prev = np.max(corr) * dx
        if c5_prev < best_c5:
            best_c5 = c5_prev
            best_h = h.copy()
        
        # Local optimization
        # Estimate gradient by finite differences
        grad = np.zeros_like(h)
        epsilon = 1e-5
        
        for i in range(len(h)):

            h_plus[i] += epsilon

        step_size = 0.01
        h = h - step_size * grad

        h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
