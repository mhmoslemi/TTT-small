import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, mutation_rate=0.1, noise_std=0.1):
    """Apply mutation to a candidate solution."""
    if np.random.rand() < mutation_rate:
        h = h.copy()
        # Only mutate non-zero elements to preserve structure
        nonzero_indices = np.where(h > 1e-6)[0]
        if len(nonzero_indices) > 0:
            num_mutations = max(1, int(len(nonzero_indices) * 0.1))
            indices = np.random.choice(nonzero_indices, size=num_mutations, replace=False)
            noise = np.random.normal(0, noise_std, len(indices))
            h[indices] += noise
            return repair_h(h, len(h))
    return h

def crossover_h(parent1, parent2, crossover_rate=0.5):
    """Perform crossover between two parents."""
    if np.random.rand() < crossover_rate:
        alpha = np.random.rand()
        child = alpha * parent1 + (1 - alpha) * parent2
        return repair_h(child, len(parent1))
    return parent1

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    elite_size = int(population_size * 0.2)
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init.copy()]
    else:
        population = []
    
    # Fill population with diverse random solutions
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        
    # Initialize best solution
    best_h = None
    best_value = float('inf')
    
    for gen in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all solutions
        fitness = []
        for h in population:
            value = evaluate_h(h, dx)
            fitness.append(value)
            if value < best_value:
                best_value = value
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select parents
            parents = np.random.choice(elite_size, size=2, replace=False)
            parent1 = population[parents[0]]
            parent2 = population[parents[1]]
            
            # Crossover
            child = crossover_h(parent1, parent2)
            
            # Mutation
            child = mutate_h(child)
            
            new_population.append(child)
            
        population = new_population
        
        # Gradient-based refinement on best individual
        if best_h is not None:
            h_refined = best_h.copy()
            current_value = best_value
            
            # Simple gradient descent step
            learning_rate = 0.001
            for _ in range(50):
                if time.time() - start_time > budget_s - 10:
                    break
                eps = 1e-6
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_value) / eps
                    
                h_refined = h_refined - learning_rate * grad
                h_refined = repair_h(h_refined, n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                current_value = np.max(corr) * dx
                
                if current_value < best_value:
                    best_value = current_value
                    best_h = h_refined.copy()
    
    return (best_h.tolist(), best_value, n_points)
