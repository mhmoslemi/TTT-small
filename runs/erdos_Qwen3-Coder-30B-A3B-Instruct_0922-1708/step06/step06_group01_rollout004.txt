import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints using quadratic programming."""
    # Project onto box constraints [0,1]
    h = np.clip(h, 0, 1)
    
    # Project onto the constraint sum(h) = n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) > 1e-8:
        # Use quadratic programming for precise projection
        # minimize ||h_new - h||^2 s.t. sum(h_new) = target_sum, 0 <= h_new <= 1
        # This is a simple quadratic program
        # All variables are equal, so h_new[i] = (target_sum - sum(h) + h[i] * n_points) / n_points
        # If h[i] >= 1, then h_new[i] should be adjusted
        h_new = h + (target_sum - current_sum) / n_points
        h_new = np.clip(h_new, 0, 1)
        
        # Final correction
        final_sum = np.sum(h_new)
        diff = target_sum - final_sum
        if abs(diff) > 1e-8:
            # Spread the error as uniformly as possible
            indices = np.argsort(h_new)
            for i in range(min(int(abs(diff) * 100), len(indices))):
                idx = indices[i] if diff < 0 else indices[-(i+1)]
                if diff < 0 and h_new[idx] < 1:
                    add = min(1 - h_new[idx], -diff)
                    h_new[idx] += add
                    diff += add
                elif diff > 0 and h_new[idx] > 0:
                    subtract = min(h_new[idx], diff)
                    h_new[idx] -= subtract
                    diff -= subtract
                if abs(diff) < 1e-8:
                    break
                
        h = h_new
    
    return h

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
    
    # Start with the initial solution
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    last_improvement = 0
    patience = 1000
    max_iter = 20000
    adaptive_rate = 0.05
    adaptive_step = 0.01
    
    # Gradient computation function for local search
    def compute_gradient(h, dx, epsilon=1e-6):
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = repair_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            grad[i] = (c_plus - c5_current) / epsilon
        return grad
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient-based update
        grad = compute_gradient(h_current, dx)
        h_new = h_current - adaptive_step * grad
        h_new = repair_h(h_new, n_points)
        
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i

        # Efficiency: Reduce adaptive_rate and adaptive_step if no improvement
        # and reverse it after some time without improvement
        if i - last_improvement > patience:
            adaptive_rate *= 0.9
            adaptive_step *= 0.9
            last_improvement = i
            if adaptive_rate < 1e-6:
                adaptive_rate = 1e-6
            if adaptive_step < 1e-6:
                adaptive_step = 1e-6
                
        # Perturbation with randomness
        noise_strength = adaptive_rate * np.std(h_current) + 1e-6
        h_pert = h_current + np.random.normal(0, noise_strength, n_points)
        h_new = repair_h(h_pert, n_points)
        c5_pert = evaluate_h(h_new, dx)
        
        if c5_pert < c5_current:
            h_current = h_new
            c5_current = c5_pert
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
    return (h_best.tolist(), c5_best, n_points)
