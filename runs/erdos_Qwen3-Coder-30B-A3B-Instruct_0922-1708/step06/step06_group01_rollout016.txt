import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def halton_sequence(n, base=2):
    """Generate n points of 1D Halton sequence."""
    result = np.zeros(n)
    for i in range(n):
        f = 1.0
        result[i] = 0.0
        while i > 0:
            f /= base
            result[i] = (i % base) * f
            i //= base
    return result

def sobol_sequence(n):
    """Generate n points of 1D Sobol sequence."""
    # Using a simple variant for demonstration
    # For production use, would use a full Sobol generator
    return np.linspace(0, 1, n, endpoint=False)

def initialize_h(n_points):
    """Initialize h using low-discrepancy sequences."""
    # Try Halton first
    seq = halton_sequence(n_points)
    h = np.interp(seq, [0, 1], [0, 2])
    
    # Map to step function with uniform density
    h = np.array([0.5] * n_points)  # Simple initialization
    
    # Better to interpolate a Halton-style function
    h = np.sin(np.pi * seq)  # Non-trivial but smooth function
    
    # Still need to make it step-like and feasible
    # Use a smooth step function based on sequence
    h = np.interp(seq, [0, 1], [0, 1])**2  # Squared for more non-linearity
    
    # Project to feasible set
    h = repair_h(h, n_points)
    return h
