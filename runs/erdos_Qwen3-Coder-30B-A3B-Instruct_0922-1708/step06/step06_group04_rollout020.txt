import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create sieve pattern in frequency domain
    freq = np.arange(n_points)
    # Inverse sieve pattern - higher frequencies get lower amplitudes
    spectrum = 1.0 / (1.0 + freq * 0.1)
    # Ensure DC component is 0
    spectrum[0] = 0
    # Make spectrum real and symmetric for real IFFT
    spectrum = spectrum[:n_points//2+1]
    # Create symmetric spectrum for real IFFT
    spectrum_full = np.concatenate([spectrum, spectrum[-2:0:-1]])
    # Inverse FFT to get real signal
    h = np.fft.irfft(spectrum_full)
    h = np.real(h)
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    # Ensure bounds
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from parent or create sieve pattern
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = fourier_initialization(n_points)
    
    # Initial tabu search parameters
    tabu_list = []
    tabu_size = 50
    max_tabu_age = 100
    tabu_ages = np.zeros(n_points)
    
    # Tabu search
    current_h = h_init.copy()
    current_c5 = evaluate_overlap(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    last_improvement = 0
    iter_count = 0
    
    max_iter = 2000
    perturb_magnitude = 0.2
    learning_rate = 0.0001
    patience = 50
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 30:
        if time.time() - start_time > budget_s - 30:
            break
            
        # Generate new candidate by perturbing best solution
        h_try = current_h.copy()
        # Randomly perturb some indices
        indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
        for i in indices:
            h_try[i] += np.random.normal(0, perturb_magnitude)
        h_try = project_h(h_try, n_points)
        
        # Check if this move is tabu
        move_key = tuple(sorted(indices))
        if move_key in tabu_list or (iter_count - last_improvement) > patience:
            # Small random perturbation to escape taboo
            h_try = project_h(current_h + np.random.normal(0, 0.01, n_points), n_points)
            if (iter_count - last_improvement) > patience:
                # Reset tabu list if stuck
                tabu_list = []
        else:
            # Apply tabu constraint and try to add to tabu
            tabu_list.append(move_key)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        
        c5_try = evaluate_overlap(h_try, dx)
        
        if c5_try < current_c5:
            current_h = h_try.copy()
            current_c5 = c5_try
            if c5_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c5_try
                last_improvement = iter_count
        
        # Update tabu ages
        tabu_ages += 1
        if len(tabu_list) > 0:
            tabu_ages[list(tabu_list[-1])] = 0
        iter_count += 1
    
    # Gradient descent refinement on best solution found
    h_refined = best_h.copy()
    prev_value = best_c5
    iter_grad = 0
    max_grad_iter = 500
    
    while iter_grad < max_grad_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        h_refined = h_refined - learning_rate * grad
        h_refined = project_h(h_refined, n_points)
        prev_value = evaluate_overlap(h_refined, dx)
        if prev_value < best_c5:
            best_h = h_refined.copy()
            best_c5 = prev_value
        iter_grad += 1
    
    return (best_h.tolist(), best_c5, n_points)
