import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Barrier method parameters
    mu = 1.0
    mu_increase = 1.1
    max_iter = 500
    min_improvement = 1e-6
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Solve convex optimization problem with barrier method
        # Variables
        h_vars = cp.Variable(n_points)
        t = cp.Variable()
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Bound constraints
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Overlap constraints with barrier
        for k in range(n_points):
            # Compute overlap term efficiently using correlation
            if k == 0:
                overlap = 0
                for i in range(n_points):
                    overlap += h_vars[i] * (1 - h_vars[i])
            else:
                overlap = 0
                for i in range(n_points - k):
                    overlap += h_vars[i] * (1 - h_vars[i + k])
            overlap *= dx
            constraints.append(t >= overlap + mu * (-cp.log(h_vars[k]) - cp.log(1 - h_vars[k]) if k < n_points - 1 else 0))
        
        # Solve problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            if prob.status == 'optimal':
                h_new = h_vars.value
                h_new = np.clip(h_new, 0, 1)
                h_new_sum = np.sum(h_new)
                if abs(h_new_sum - n_points / 2) > 1e-8:
                    h_new = h_new * (n_points / 2) / h_new_sum
                h = h_new
                
                # Evaluate c5
                corr = correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                
                # Update mu
                if mu < 1000:  # Prevent overflow
                    mu *= mu_increase
                else:
                    break  # Stop if barrier becomes too large
            else:
                # Fallback to gradient method
                break
        except:
            # Fallback to gradient method if CVXPY fails
            break
            
    # Final gradient-based refinement step
    if best_c5 == float('inf'):
        # Use gradient descent directly
        h = best_h.copy()
        learning_rate = 0.001
        for i in range(1000):
            if (time.time() - start_time) > budget_s - 5:
                break
                
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            max_lag_idx = np.argmax(corr)
            k = max_lag_idx - (len(h) - 1)
            
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                if h_plus[i] > 1:
                    h_plus[i] = 1
                h_plus_sum = np.sum(h_plus)
                if abs(h_plus_sum - n_points / 2) > 1e-8:
                    h_plus = h_plus * (n_points / 2) / h_plus_sum
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / 1e-6
            
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-8:
                h = h * (n_points / 2) / h_sum
            
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final validation
    corr = correlate(best_h, 1 - best_h, mode='full')
    c5_final = np.max(corr) * dx
    
    return (best_h.tolist(), c5_final, n_points)
