import numpy as np
import time
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Stage 1: Convex relaxation to get good initial solution
    print("Starting convex relaxation stage...")
    try:
        # Use cvxpy to solve relaxation
        h_var = cp.Variable(n_points)
        # Minimize max correlation
        max_corr = cp.Variable()
        
        # Formulate correlation constraints
        h_vec = h_var
        ones_vec = np.ones(n_points)
        
        # For each shift k, we define a constraint
        # Though the full correlation matrix is complex, we use a surrogate approach:
        # minimize (1-alpha) * max_corr + alpha * sum of squares of correlations
        # This is heuristic but effective
        
        # Build part of the problem
        correlations = []
        for k in range(2 * n_points - 1):
            # This is a technical simplification - think of approximating
            corr_val = 0.0
            # Consider the actual correlation for this shift
            # The full computation is expensive, so we use approximation
            if k < n_points:
                corr_val = cp.sum(h_vec[:n_points-k] * (1 - h_vec[k:]))
            else:
                corr_val = cp.sum(h_vec[k-n_points+1:] * (1 - h_vec[:n_points-(k-n_points+1)]))
            correlations.append(corr_val)
            
        # Take max directly in CVXPY
        max_corr_var = cp.max(cp.hstack(correlations))
        
        # Objective: weighted sum of squared correlations and max
        # Instead directly optimize GC_tilde (simplified opinion to reduce complexity)
        obj = cp.Minimize(max_corr_var)
        
        # Build constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_init = h_var.value
            h = repair_h(h_init, n_points)
            c5_new = evaluate_h(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    except Exception as e:
        print("Convex relaxation failed:", e)
    
    print("Starting gradient-based refinement stage...")
    # Stage 2: Refinement with adaptive gradient descent
    # More direct gradient computation using efficient methods
    
    # Adaptive learning rates
    eta = 0.01
    beta = 0.9
    stagnation_count = 0
    stagnation_limit = 200
    history = []
    
    # Initialize gradient term
    last_c5 = best_c5
    
    # Gradient-based refinement
    for k in range(20000):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute numerical gradients using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
            
        # Update using adaptive learning rate
        eta_k = eta / (1 + k ** beta)
        h_new = best_h - eta_k * grad
        h_new = repair_h(h_new, n_points)
        
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new

            history = []

            stagnation_count += 1
            history.append(c5_eval)
            if len(history) > 100 and np.all(np.array(history[-100:]) - c5_eval > -1e-6):
                break  # Stagnation detected

        # Occasionally add noise to escape local minima
        if np.random.random() < 0.01:
            noise = np.random.normal(0, 0.001, n_points)
            h_new = best_h + noise
            h_new = repair_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
    return (best_h.tolist(), best_c5, n_points)
