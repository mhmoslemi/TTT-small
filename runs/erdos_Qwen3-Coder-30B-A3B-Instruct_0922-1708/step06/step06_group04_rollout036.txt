import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum=50):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    h = project_h(h, 50)

    lambda_reg = 0.01

    alpha = 0.1
