import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, k_max, n_points, dx):
    """Compute gradient of C5 with respect to h."""
    grad = np.zeros(n_points)
    # Compute gradient for the maximum overlap term
    for i in range(n_points):
        j = (i - k_max) % n_points
        grad[i] = (1 - h[j]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Gradient descent parameters
    lr = 1e-4
    alpha = 0.99
    eps = 1e-6
    max_iter = 100000
    progress_count = 0
    max_progress = 500
    perturbation_interval = 500
    perturbation_magnitude = 0.05
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap for all lags and find maximum
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient of the maximum overlap
        grad = compute_gradient(current_h, k_max, n_points, dx)
        
        # Apply gradient descent update
        new_h = current_h - lr * grad
        
        # Apply random perturbations periodically
        if i % perturbation_interval == 0 and i > 0:
            idxs = np.random.choice(n_points, size=n_points//10, replace=False)
            for idx in idxs:
                new_h[idx] += np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
        
        # Repair the new solution
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Update if better or accept with probability
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                progress_count = 0
            else:
                progress_count += 1
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / (lr * 0.1)):
                current_h = new_h
                current_c5 = new_c5
            else:
                progress_count += 1
        
        # Learning rate decay
        if i % 100 == 0:
            lr *= alpha
            
        # Perturbation magnitude decay
        if i % 500 == 0 and i > 0:
            perturbation_magnitude *= 0.999
            
        # Reset progress counter if improvement occurs
        if progress_count >= max_progress:
            perturbation_magnitude = min(perturbation_magnitude * 1.1, 0.1)
            progress_count = 0
            
    return (best_h.tolist(), best_c5, n_points)
