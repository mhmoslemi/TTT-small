import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a sine-like pattern with frequency components
    # designed to create a low-overlap initial solution
    f = np.arange(n_points)
    # Create a "sieve" pattern: high frequencies have low amplitudes
    # This is intended to avoid strong correlations in overlapping regions
    amplitudes = 1.0 / (1.0 + f * 0.1)
    # Add some randomness but keep the sieve structure
    amplitudes = amplitudes * (0.8 + 0.4 * np.random.rand(n_points))
    # Use real inverse FFT to create a real-valued signal
    # This may have negative parts, so we'll process carefully
    h = np.fft.irfft(amplitudes)
    # Normalize to sum to 1
    h = h - np.min(h)
    h = h * (n_points / 2) / np.sum(h)
    return h

    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Try the frequency domain approach
        h_init = fourier_initialization(n_points)
    
    h_init = project_h(h_init, n_points)
    
    # Tabu search part
    tabu_list = []
    tabu_size = 50
    memory_decay = 0.95
    max_iter = 2000
    perturb_scale = 0.2
    pop_size = 100
    
    # Initialize population
    population = [h_init]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points) * 0.5  # Start with a moderate level
        h = project_h(h, n_points)
        population.append(h)
    
    fitness = [evaluate_overlap(h, dx) for h in population]
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate new solutions via perturbation
        new_population = []
        new_fitness = []
        
        for i in range(len(population)):
            h = population[i].copy()
            
            # Create a perturbation
            perturb = np.random.randn(n_points) * perturb_scale
            h_new = h + perturb
            
            # Repair
            h_new = project_h(h_new, n_points)
            new_population.append(h_new)
            new_fitness.append(evaluate_overlap(h_new, dx))
        
        # Combine with existing population
        combined_population = population + new_population
        combined_fitness = fitness + new_fitness
        
        # Keep best half
        indices = np.argsort(combined_fitness)[:pop_size]
        population = [combined_population[i] for i in indices]
        fitness = [combined_fitness[i] for i in indices]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_value:
            best_h = population[current_best_idx].copy()
            best_value = fitness[current_best_idx]
            last_improvement = iter_count
            
        # Update tabu list
        tabu_list.append(iter_count)

            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
            
        # Update
        h_refined = h_refined - 0.0001 * grad
        h_refined = project_h(h_refined, n_points)
        prev_value = evaluate_overlap(h_refined, dx)
        
        if prev_value < best_value:
            best_h = h_refined.copy()
            best_value = prev_value

    return (best_h.tolist(), best_value, n_points)
