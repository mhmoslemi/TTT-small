import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.full(n_points, 0.5)
        # Add small random perturbations
        perturbation = np.random.normal(0, 0.1, n_points)
        h_init = h_init + perturbation
        h_init = repair_h(h_init, n_points)
    
    best_h = h_init.copy()
    best_value = evaluate_h(h_init, dx)
    
    # Gradient-based optimization with adaptive learning rate
    max_iter = 10000
    learning_rate = 0.0001
    tolerance = 1e-6
    last_improvement = 0
    
    h_current = h_init.copy()
    prev_value = best_value
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient numerically
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        # Update
        h_new = h_current - learning_rate * grad
        h_new = repair_h(h_new, n_points)
        
        new_value = evaluate_h(h_new, dx)
        
        if new_value < prev_value:
            h_current = h_new
            prev_value = new_value
            if new_value < best_value:
                best_h = h_current.copy()
                best_value = new_value
                last_improvement = i
        else:
            # Reduce learning rate if no improvement
            learning_rate *= 0.995
            if learning_rate < 1e-8:
                learning_rate = 1e-8
                
        # Inject diversity if stuck
        if i - last_improvement > 100:
            h_current = np.random.rand(n_points)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            last_improvement = i
            
        # Early stopping criteria
        if abs(prev_value - best_value) < tolerance:
            break
    
    # Final refinement of best solution
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        best_h = best_h - 0.00005 * grad
        best_h = repair_h(best_h, n_points)
        best_value = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_value, n_points)
