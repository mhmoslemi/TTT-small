import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_gradient(h, k, n_points, dx):
    """Compute gradient of overlap integral with respect to h at shift k."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        # For k-th shift, contribution to overlap is h[i] * (1 - h[i+k])
        # But we consider the full correlation: we look at h[i] * (1 - h[i+k]) 
        # so the derivative with respect to h[i] would be (1 - h[i+k]) - h[i+k]
        # Actually we just need contribution from indices that overlap
        # See how correlate computes it
        # This is the impact of h[i] on C(k) = sum(h[j] * (1 - h[j+k]))
        if i + k >= 0 and i + k < n_points:
            grad[i] = -2 * (h[i + k] - 0.5)  # Simplification
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization with sum = n_points/2
        h = np.random.rand(n_points)
    
    h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate initial solution
    current_c5 = evaluate_h(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Shift grid for approximation
    n_shifts = 100
    shifts = np.linspace(-2, 2, n_shifts)
    
    # Parameters
    learning_rate = 0.01
    max_iter = 5000
    patience = 100
    no_improvement = 0
    
    # Main optimization loop
    iter_count = 0
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 5:
        # Compute all correlations to find worst shift
        corr = correlate(h, 1-h, mode='full')
        c5_values = corr * dx
        
        # Find k that maximizes the overlap (worst case)
        worst_k_idx = np.argmax(c5_values)
        worst_k = worst_k_idx - (len(corr) - 1) // 2
        
        # Since we computed all shifts, just compute gradient at worst_k
        # with k in the range for usual correlation logic
        # Alternative: we compute gradient directly for C5 at worst_k
        # Taking derivative of correlation at worst k wrt h
        # If C(k) = sum_j h[j] * (1 - h[j+k]) 
        # Then dC(k)/dh[i] = (1 - h[i+k]) - h[i+k] = 1 - 2*h[i+k] if i+k is valid
        # We will now do this properly by reversing compute on corr to find derivative
        
        # Get forward indices where k-shifted element in h exists
        # indices of h[j] term contribute to correlation at shift k: when sum is
        # over j the h[j] * h[j-k] term contributes
        # But we're computing sum(h[j] * (1 - h[j-k]))
        # So partial derivative w.r.t. h[i] = (1 - h[i-k])
        # Actually this is tradition from the correlation equation 
        
        # For indices proper
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Term h[i] contributes to correlation when h[j] occurs with 1-h[j-k]
            # So for h[i] to contribute, we look at contributions from h[j] where
            # i = j - k, so j = i + k  
            j = i + worst_k
            if 0 <= j < n_points:
                # In the convolution sense, for z[l] = sum_{m} x[m] * y[l-m]
                # the derivative of z[l] wrt x[m] = y[l-m]
                # Here, our overlap at shift k is sum_i h[i] (1 - h[i+k]) 
                # So the derivative wrt h[i] = (1 - h[i+k]) - h[i+k]
                # Actually it's more like: if correlation = sum h[i] h[i+k]
                # then derivative is h[i+k]
                # But our correlation is: sum h[i] (1-h[i+k])
                # d/dh[i] (sum_h[i] (1-h[i+k]) ) = (1 - h[i+k])
                
                # More precisely, the j gets shifted by -k, and they x[j] * y[j+k] = x[j] * (1 - x[j+k]) 
                # d/dx[i] of that is: when we have x[i] that multiplies (1 - x[i+k])  
                # So derivative = (1 - 0) if i = j or 0 
                # The gradient is concentrated on j where h[j] is the term resolved.
                
                # So for correlation C(k) = sum_i h[i] (1 - h[i+k]) 
                # The variation of C(k) wrt h[i] = (1 - h[i+k]) 
                # h[i] in the term x[i] * (1 - x[i+k]) = (1 - h[i+k]) 
                
                # But again this should match those direct indices the h[i] occurs
                # The coefficient is when x[i] appears (is the x on the left) 
                # That contributes to the term with x[i+k] on right, i-k 
                grad[i] = 1 - h[j]
                
        # update h with gradient
        h_new = h - learning_rate * grad
        h_new = repair_h(h_new, n_points)
        
        # check new C5
        new_c5 = evaluate_h(h_new, dx)
        
        # acceptance
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            # Revert to old h to try another move
            pass
            
        # Periodic perturbation to escape local optima
        if iter_count % 50 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = repair_h(h, n_points)
            
        # Adaptive learning rate
        if iter_count % 100 == 0:
            if no_improvement > 50:
                learning_rate *= 0.5
                
        if no_improvement > patience:
            break
            
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
