import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h = h_init.copy()
    else:
        h = np.ones(n_points) * 0.5
        
    h = repair_h(h, n_points)
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Optimization loop
    max_iter = 2000
    learning_rate = 0.001
    patience = 100
    no_improvement = 0
    last_best = best_value
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:

        # Compute gradients
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        # Update h
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        
        # Evaluate new solution
        current_value = evaluate_h(h, dx)
        
        # Update best solution
        if current_value < best_value:
            best_value = current_value
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Early stopping
        if no_improvement > patience:
            break
            
        # Reduce learning rate if stuck
        if no_improvement % 50 == 0 and no_improvement > 0:
            learning_rate *= 0.5
            
    return (best_h.tolist(), best_value, n_points)
