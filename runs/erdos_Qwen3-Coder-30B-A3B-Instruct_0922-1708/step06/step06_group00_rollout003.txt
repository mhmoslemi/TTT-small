import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Binary approximation
    h_binary = (h > 0.5).astype(float)
    h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.999
    max_iter = 50000
    tabu_size = 50
    tabu_list = []
    
    # Current best solution
    best_h = h_binary.copy()
    curr_h = h_binary.copy()
    
    # Compute initial overlaps
    corr = correlate(curr_h, 1 - curr_h, mode='full')
    best_c5 = np.max(corr) * dx
    curr_c5 = best_c5
    
    # Main SA loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping one bit
        candidate_h = curr_h.copy()
        flip_idx = np.random.randint(0, n_points)
        
        # Flip the bit at position flip_idx
        candidate_h[flip_idx] = 1 - candidate_h[flip_idx]
        
        # Adjust to maintain sum constraint
        current_sum = np.sum(curr_h)
        candidate_sum = np.sum(candidate_h)
        
        # If we flip a 1 to 0, we need to adjust other points
        if curr_h[flip_idx] == 1:
            # Find another 0 to flip to 1
            zero_indices = np.where(candidate_h == 0)[0]
            if len(zero_indices) > 0:
                adjust_idx = np.random.choice(zero_indices)
                candidate_h[adjust_idx] = 1
        else:
            # Find another 1 to flip to 0
            one_indices = np.where(curr_h == 1)[0]
            if len(one_indices) > 0:
                adjust_idx = np.random.choice(one_indices)
                candidate_h[adjust_idx] = 0
                
        # Ensure we maintain the sum constraint exactly
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Compute overlap
        corr = correlate(candidate_h, 1 - candidate_h, mode='full')
        candidate_c5 = np.max(corr) * dx
        
        # Accept or reject based on SA criterion
        delta = candidate_c5 - curr_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            curr_h = candidate_h
            curr_c5 = candidate_c5
            
            # Check if this improves the global best
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = curr_h.copy()
                
        # Update temperature
        temp = max(1e-8, temp * cooling_rate)
        
        # Update tabu list
        if flip_idx not in tabu_list:
            tabu_list.append(flip_idx)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Ensure the result is valid according to constraints
    # Normalize so that sum = n_points/2 exactly
    h_sum = np.sum(best_h)
    best_h = best_h * (n_points / 2) / h_sum
    
    return (best_h.tolist(), final_c5, n_points)
