import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h = (h_init > 0.5).astype(int)
        h_sum = np.sum(h)
        if h_sum != n_points // 2:
            # Adjust to target sum by flipping some bits
            target = n_points // 2
            if h_sum > target:
                # Need to flip some 1s to 0s
                indices = np.where(h == 1)[0]
                flip_indices = np.random.choice(indices, size=int(h_sum - target), replace=False)
                h[flip_indices] = 0
            else:
                # Need to flip some 0s to 1s
                indices = np.where(h == 0)[0]
                flip_indices = np.random.choice(indices, size=int(target - h_sum), replace=False)
                h[flip_indices] = 1
    else:
        # Create uniform binary initial solution with exactly n_points/2 ones
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    # MIP formulation
    # Variables
    h_vars = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_vars) == n_points // 2]
    
    # For each possible shift k, add overlap constraint
    # valid shifts for k: -n_points+1 to n_points-1
    for k in range(-n_points + 1, n_points):
        # For shift k, compute valid i indices such that i+k is in [0, n_points-1]
        valid_indices = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_indices.append((i, j))
        
        if valid_indices:
            # Sum of h[i] * (1 - h[j]) for all valid pairs
            overlap_expr = sum(h_vars[i] * (1 - h_vars[j]) for i, j in valid_indices)
            # Divide by n_points to match the integral scaling
            constraints.append(overlap_expr <= t * (n_points / 2))
    
    # Problem setup
    prob = cp.Problem(objective, constraints)
    
    # Set solver and solve
    try:
        if (time.time() - start_time) > budget_s - 10:
            # Not enough time, return current solution
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return (h.tolist(), c5, n_points)
        
        # Solve with time limit
        prob.solve(solver=cp.CBC, time_limit=budget_s - 10, gap=0.01)
        
        # Extract solution
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = np.round(h_vars.value).astype(int)
            h_opt = np.clip(h_opt, 0, 1)
            
            # Ensure exactly the right number of 1s
            h_sum = np.sum(h_opt)
            if h_sum != n_points // 2:
                # Adjust if needed
                if h_sum > n_points // 2:
                    indices = np.where(h_opt == 1)[0]
                    flip_indices = np.random.choice(indices, size=int(h_sum - n_points // 2), replace=False)
                    h_opt[flip_indices] = 0
                else:
                    indices = np.where(h_opt == 0)[0]
                    flip_indices = np.random.choice(indices, size=int(n_points // 2 - h_sum), replace=False)
                    h_opt[flip_indices] = 1
            
            h = h_opt
            
    except Exception:
        # Fall back to gradient approach if MIP fails
        # Continue with gradient optimization approach, but with better restart
        h = h.astype(float)  # convert for gradient steps
        learning_rate = 0.01
        max_iter = 5000
        
        for iteration in range(max_iter):
            if (time.time() - start_time) > budget_s - 5:
                break
                
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Gradient estimation
            grad = np.zeros_like(h)
            epsilon = 1e-4
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += epsilon
                h_plus = np.clip(h_plus, 0, 1)
                h_sum = np.sum(h_plus)
                if abs(h_sum - n_points / 2) > 1e-8:
                    h_plus = h_plus * (n_points / 2) / h_sum
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / epsilon
            
            # Update
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-8:
                h = h * (n_points / 2) / h_sum
            
            # Binary approximation for discrete solution
            bin_h = (h > 0.5).astype(int)
            bin_sum = np.sum(bin_h)
            
            # Adjust number of 1s if needed
            if bin_sum < n_points // 2:
                diff = n_points // 2 - bin_sum
                indices = np.where(bin_h == 0)[0]
                if len(indices) >= diff:
                    flip_indices = np.random.choice(indices, diff, replace=False)
                    bin_h[flip_indices] = 1
            elif bin_sum > n_points // 2:
                diff = bin_sum - n_points // 2
                indices = np.where(bin_h == 1)[0]
                if len(indices) >= diff:
                    flip_indices = np.random.choice(indices, diff, replace=False)
                    bin_h[flip_indices] = 0
            
            h = bin_h.astype(float)
    
    # Final computation
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    
    return (h.tolist(), c5, n_points)
