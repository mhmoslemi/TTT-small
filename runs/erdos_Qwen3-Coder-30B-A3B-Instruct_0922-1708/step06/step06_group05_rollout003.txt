import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Define convex optimization problem for relaxation
    def optimize_convex(h_init):
        h_var = cp.Variable(n_points)
        # Objective: minimize sum of squared correlations (relaxed)
        # This is a rough approximation of the true objective
        expr = 0
        for k in range(2 * n_points - 1):
            shift = np.zeros(2 * n_points - 1)
            shift[k] = 1
            cross = cp.convolve(h_var, 1 - h_var, 'full')
            expr += cross[k]**2
        obj = cp.Minimize(expr)
        
        # Constraints
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h_var.value
        except:
            pass
        return h_init
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Optimizer state
    learning_rate = 0.001
    adams = [np.zeros(n_points), np.zeros(n_points)]
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    t = 0
    
    for iteration in range(20000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Adaptive gradients using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        h_eval = evaluate_h(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - h_eval) / eps
            
        # Adam optimizer
        t += 1
        adams[0] = beta1 * adams[0] + (1 - beta1) * grad
        adams[1] = beta2 * adams[1] + (1 - beta2) * grad**2
        m_hat = adams[0] / (1 - beta1**t)
        v_hat = adams[1] / (1 - beta2**t)
        h_update = m_hat / (np.sqrt(v_hat) + epsilon)
        
        h = h - learning_rate * h_update
        h = repair_h(h, n_points)
        
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Occasionally solve convex relaxed problem if not stale
        if iteration % 500 == 0:
            h_convex = optimize_convex(h)
            h_convex = repair_h(h_convex, n_points)
            c5_convex = evaluate_h(h_convex, dx)
            if c5_convex < best_c5:
                best_c5 = c5_convex
                best_h = h_convex.copy()
                
        # Occasionally perturb
        if iteration % 1000 == 0 and iteration > 0:
            perturb_indices = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
            h[perturb_indices] += np.random.normal(0, 0.005, len(perturb_indices))
            h = repair_h(h, n_points)
            
    return (best_h.tolist(), best_c5, n_points)
