import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_to_feasible(h_init, n_points)
    else:
        # Start with a triangular function
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = project_to_feasible(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Frank-Wolfe parameters
    max_iter = 10000
    patience = 1000
    last_improvement = 0
    eta_max = 1.0
    eps = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_current) / eps
        
        # Solve linear subproblem: min_d ∇C₅ • d subject to d in feasible set
        # The feasible set is {d : 0 <= d_i <= 1, sum d_i = 0}
        # We're looking to move in the direction of -grad
        d = -grad
        
        # Project d to satisfy sum = 0
        d = d - np.mean(d)
        
        # Clip to [0,1] to satisfy box constraints
        d = np.clip(d, -h_current, 1 - h_current)
        
        # Line search for optimal step size
        eta = eta_max
        h_test = h_current + eta * d
        h_test = project_to_feasible(h_test, n_points)
        c5_test = evaluate_h(h_test, dx)
        
        while c5_test >= c5_current and eta > 1e-10:
            eta /= 2
            h_test = h_current + eta * d
            h_test = project_to_feasible(h_test, n_points)
            c5_test = evaluate_h(h_test, dx)
        
        # Update
        h_current = h_test
        c5_current = c5_test
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        if i - last_improvement > patience:
            break
            
    # Final refinement with local search
    h_final = h_best.copy()
    c5_final = c5_best
    
    # Try to refine with gradient descent
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_final) / eps
            
        h_final = h_final - 0.01 * grad
        h_final = project_to_feasible(h_final, n_points)
        corr = correlate(h_final, 1 - h_final, mode='full')
        c5_final = np.max(corr) * dx
        
        if c5_final < c5_best:
            h_best = h_final.copy()
            c5_best = c5_final
    
    return (h_best.tolist(), c5_best, n_points)
