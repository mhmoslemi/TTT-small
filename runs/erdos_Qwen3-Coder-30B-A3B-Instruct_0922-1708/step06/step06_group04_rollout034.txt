import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # PSO parameters
    num_particles = 100
    max_iter = 1000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    inertia_decay = 0.99
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    for i in range(num_particles):
        if i == 0:
            h = h_init.copy()
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_h(h, dx))
    
    # Initialize global best
    best_idx = np.argmin(pbest_vals)
    gbest = pbest[best_idx].copy()
    gbest_val = pbest_vals[best_idx]
    
    iter_count = 0
    no_improvement_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        for i in range(num_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * v + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)
            
            # Update position
            h_new = h + v
            
            # Apply repair
            h_new = repair_h(h_new, n_points)
            
            # Evaluate fitness
            fitness = evaluate_h(h_new, dx)
            
            # Update personal best
            if fitness < pbest_vals[i]:
                pbest[i] = h_new.copy()
                pbest_vals[i] = fitness
                
                # Update global best
                if fitness < gbest_val:
                    gbest = h_new.copy()
                    gbest_val = fitness
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            
            # Update particle
            particles[i] = h_new
            velocities[i] = v
            
        # Adjust inertia weight
        w = w * inertia_decay
        
        # Check for stagnation and inject diversity
        if no_improvement_count > 50:
            for _ in range(5):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles.append(h)
                velocities.append(np.random.uniform(-0.5, 0.5, n_points))
                pbest.append(h.copy())
                pbest_vals.append(evaluate_h(h, dx))
                
            # Keep only the best ones
            indices = np.argsort(pbest_vals)[:num_particles]
            particles = [particles[i] for i in indices]
            velocities = [velocities[i] for i in indices]
            pbest = [pbest[i] for i in indices]
            pbest_vals = [pbest_vals[i] for i in indices]
            best_idx = np.argmin(pbest_vals)
            gbest = pbest[best_idx].copy()
            gbest_val = pbest_vals[best_idx]
            no_improvement_count = 0
            
        iter_count += 1
    
    # Gradient descent refinement of best solution
    h_refined = gbest.copy()
    prev_value = gbest_val
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        h_refined = h_refined - 0.0001 * grad
        h_refined = repair_h(h_refined, n_points)
        prev_value = evaluate_h(h_refined, dx)
        
        if prev_value < gbest_val:
            gbest = h_refined.copy()
            gbest_val = prev_value
    
    return (gbest.tolist(), gbest_val, n_points)
