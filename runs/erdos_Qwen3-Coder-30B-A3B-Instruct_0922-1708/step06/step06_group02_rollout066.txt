import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def crossover(parent1, parent2):
    """Create child from two parents using uniform crossover."""
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1):
    """Mutate h by adding Gaussian noise."""
    noise = np.random.normal(0, 0.1, len(h))
    mask = np.random.rand(len(h)) < mutation_rate
    h = h + noise * mask
    return h

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population with diverse solutions
    population_size = 50
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    else:
        # Generate a uniform distribution
        h_init = np.ones(n_points) * 0.5
        population.append(h_init.copy())
    
    # Fill remaining population with random feasible solutions
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Initialize best solution
    best_h = None
    best_value = float('inf')
    
    for p in population:
        value = evaluate_h(p, dx)
        if value < best_value:
            best_value = value
            best_h = p.copy()
    
    # Evolution parameters
    generations = 1000
    elite_size = int(population_size * 0.2)
    mutation_rate = 0.1
    
    # Main evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate population
        fitness = []
        for p in population:
            value = evaluate_h(p, dx)
            fitness.append(value)
            if value < best_value:
                best_value = value
                best_h = p.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Create new population
        new_population = population[:elite_size].copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1_idx = np.random.randint(0, elite_size)
            parent2_idx = np.random.randint(0, elite_size)
            if parent1_idx != parent2_idx:
                child = crossover(population[parent1_idx], population[parent2_idx])
                child = repair_h(child, n_points)
                if np.random.rand() < 0.5:  # 50% chance to mutate
                    child = mutate(child, mutation_rate)
                    child = repair_h(child, n_points)
                new_population.append(child)
        
        population = new_population
        
        # Adaptive mutation rate
        if gen % 20 == 0 and gen > 0:
            current_best = evaluate_h(population[0], dx)
            if current_best < best_value:
                mutation_rate *= 0.95  # Reduce mutation rate if improvement seen
            else:

            h_refined = repair_h(h_refined, n_points)
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            current_value = np.max(corr) * dx
            
            if current_value < best_value:
                best_value = current_value
                best_h = h_refined.copy()
    
    return (best_h.tolist(), best_value, n_points)
