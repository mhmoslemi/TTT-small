import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution from parent or random
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points) * 0.5
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Optimization loop
    iter_count = 0
    max_iter = 5000
    learning_rate = 0.0001
    stagnation_count = 0
    max_stagnation = 200
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient numerically
        eps = 1e-6
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(h_best, dx)
        
        for i in range(n_points):
            h_plus = h_best.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update with gradient descent
        h_new = h_best - learning_rate * grad
        h_new = repair_h(h_new, n_points)
        
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if better
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Inject diversity if stalled
        if stagnation_count > max_stagnation:
            h_new = np.random.rand(n_points) * 0.5
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_h(h_new, dx)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
            stagnation_count = 0
            
        iter_count += 1

    # Final gradient refinement
    for i in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(h_best, dx)
        
        for j in range(n_points):
            h_plus = h_best.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - current_c5) / eps
            
        h_best = h_best - 0.00001 * grad
        h_best = repair_h(h_best, n_points)
        c5_best = evaluate_h(h_best, dx)

    return (h_best.tolist(), c5_best, n_points)
