def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def halton_sequence(n, base=2):
    """Generate Halton sequence for given base."""
    sequence = []
    for i in range(n):
        value = 0
        f = 1.0 / base
        k = i
        while k > 0:
            value += (k % base) * f
            k //= base
            f /= base
        sequence.append(value)
    return np.array(sequence)

def sobol_sequence(n):
    """Generate Sobol sequence."""
    # This is a basic implementation of Sobol sequence generation
    # In a real implementation we'd use a library like sobol_seq
    # For now, use a basic construction
    seq = np.zeros(n)
    for i in range(n):
        seq[i] = i / (n - 1) if n > 1 else 0.5
    return seq

def create_step_function(x_values, y_values, n_points, dx):
    """Create a step function from x and y values."""
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        # Find the right segment
        idx = np.searchsorted(x_values, x, side='right') - 1
        if idx < 0:
            idx = 0
        elif idx >= len(y_values):
            idx = len(y_values) - 1
        h[i] = y_values[idx]
    return h
