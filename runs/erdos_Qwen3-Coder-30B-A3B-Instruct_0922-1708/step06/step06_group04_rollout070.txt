import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create sieve-like pattern based on inverse FFT
    freqs = np.fft.fftfreq(n_points, 1.0 / n_points)
    # High frequencies get lower amplitudes to suppress overlap
    ampls = np.exp(-0.1 * np.abs(freqs)**1.5)
    ampls[0] = 0  # DC component to zero
    # Create complex spectrum with random phases
    phases = np.random.rand(n_points) * 2 * np.pi
    spectrum = ampls * np.exp(1j * phases)
    # Make it real
    h = np.fft.ifft(spectrum).real
    # Normalize and project
    h = project_h(h, n_points)
    return h

    else:

    best_h = h.copy()

    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    memory_decay = 0.95
    population_size = 100
    population = [h.copy()]
    fitness = [best_value]
    
    # Add diverse random solutions
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = project_h(h_rand, n_points)
        population.append(h_rand)
        fitness.append(evaluate_overlap(h_rand, dx))
    
    # Best from initial population
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]

    iter_count = 0
    stagnation_counter = 0

        if time.time() - start_time > budget_s - 10:

        for i in range(len(population)):

            perturb = np.random.randn(n_points) * 0.2

            h_new = project_h(h_new, n_points)
            
            new_candidates.append(h_new)
            
        # Evaluate
        new_fitness = [evaluate_overlap(h, dx) for h in new_candidates]
        
        # Combine with population
        all_population = population + new_candidates
        all_fitness = fitness + new_fitness
        
        # Select best solutions
        indices = np.argsort(all_fitness)[:population_size]
        population = [all_population[i] for i in indices]
        fitness = [all_fitness[i] for i in indices]
        
        # Update overall best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_value:

            best_h = h_refined.copy()
            best_value = current_value
