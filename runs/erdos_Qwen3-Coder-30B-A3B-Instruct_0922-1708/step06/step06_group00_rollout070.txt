import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Proximal gradient with Nesterov acceleration
    gamma = 0.01  # Step size
    beta = 0.9    # Momentum
    max_iter = 50000
    
    # For Nesterov acceleration
    v = np.zeros(n_points)
    y = h.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap for all shifts
        corr = correlate(y, 1 - y, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = y.copy()
        
        # Find the maximizing lag
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(y) - 1)
        
        # Compute gradient at the max lag
        grad = np.zeros_like(y)
        if k >= 0:
            grad[:-k] = -(1 - y[k:])
        else:
            grad[-k:] = -(1 - y[:k])
        
        # Proximal gradient update with Nesterov acceleration
        y_new = y - gamma * grad
        # Project to [0,1]
        y_new = np.clip(y_new, 0, 1)
        # Enforce sum constraint
        y_new = y_new * (n_points / 2) / np.sum(y_new)
        
        # Nesterov acceleration
        v = beta * v + (y_new - y)
        y = y_new + beta * v
        
        # Project again after acceleration
        y = np.clip(y, 0, 1)
        y = y * (n_points / 2) / np.sum(y)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
