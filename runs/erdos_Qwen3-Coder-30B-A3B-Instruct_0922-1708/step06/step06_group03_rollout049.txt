import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Adaptive parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    patience = 0
    max_patience = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        
        # Compute current overlap values
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Compute gradient w.r.t. h
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Use finite difference to compute gradient
            h_plus = current_h.copy()
            h_plus[i] += 1e-6
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = corr_plus[k_max]
            gradient[i] = (max_plus - max_overlap) / 1e-6
            
        # Apply gradient descent
        current_h = current_h - learning_rate * gradient

        if iteration % 100 == 0:

            low_grad_indices = np.where(np.abs(gradient) < 0.01)[0]
            if len(low_grad_indices) > 0:
                perturb_indices = np.random.choice(
                    low_grad_indices, 
                    min(10, len(low_grad_indices)), 
                    replace=False
                )
                for idx in perturb_indices:
                    current_h[idx] += np.random.normal(0, perturbation_magnitude)
        
        # Repair h to ensure constraints
        current_h = repair_h(current_h, n_points)
        
        # Evaluate new solution
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            
        # Adapt learning rate and perturbation magnitude
        if patience > 50:
            if learning_rate > 1e-7:
                learning_rate *= 0.99
            if perturbation_magnitude > 1e-4:
                perturbation_magnitude *= 0.999
            patience = 0

        if iteration % 1000 == 0:
            current_h = repair_h(current_h + np.random.normal(0, 0.01, n_points), n_points)
