import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_grad(h, lambda_vals, dx):
    """Compute gradient of the Lagrangian."""
    n = len(h)
    grad = np.zeros(n)
    
    # For each position i, compute the derivative
    for i in range(n):
        for k in range(2 * n - 1):
            if k < n:
                j = i - k
                if j >= 0:
                    grad[i] += lambda_vals[k] * (1 - h[j])
            else:
                j = i + (2 * n - 1) - k
                if j < n:
                    grad[i] += lambda_vals[k] * (1 - h[j])
    
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h with the initial_h_values if available, else random
    if 'initial_h_values' in globals() and 'initial_h_values' is not None:
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Use a more structured initialization
        h_init = np.zeros(n_points)
        # Place a few spikes
        spike_positions = [10, 30, 50, 70, 90]
        for i in spike_positions:
            if i < n_points:
                h_init[i] += 0.3
        # Normalize
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    h = h_init.copy()
    h = repair_h(h, n_points)
    current_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Dual variables for Lagrangian
    lambda_vals = np.ones(2 * n_points - 1) * 0.01
    
    # Parameters
    learning_rate = 5e-4
    dual_learning_rate = 1e-3
    max_iter = 50000
    cooldown = 0
    max_cooldown = 100
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap correlation and find maximum
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Update dual variables
        lambda_vals += dual_learning_rate * (corr - max_val)
        lambda_vals = np.maximum(lambda_vals, 0)  # Keep non-negative
        
        # Gradient descent step
        for _ in range(5):
            grad = compute_grad(h, lambda_vals, dx)
            h = h - learning_rate * grad
            h = repair_h(h, n_points)
            
        # Evaluate new solution
        c5 = evaluate_h(h, dx)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            cooldown = 0
        else:
            cooldown += 1
            
        # Cooling schedule for learning rate
        if iteration % 500 == 0:
            learning_rate *= 0.99
            
        # Early stopping if no improvement
        if cooldown > max_cooldown:
            break
    
    return (best_h.tolist(), best_c5, n_points)
