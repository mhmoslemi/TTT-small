import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_h(current_h, dx)
    current_c5 = best_c5
    
    # Optimizer hyperparameters
    learning_rate = 5e-4
    temperature = 0.1
    decay_rate = 0.999
    dual_update_freq = 10
    max_stagnation = 200
    stagnation_count = 0
    
    # Initialize dual variables
    lambda_k = np.ones(2 * n_points - 1)  # All dual variables
    lambda_k = lambda_k / np.sum(lambda_k)  # Normalize to sum to 1
    
    for iteration in range(100000):  # Sufficient iterations
        if time.time() - start_time > budget_s - 5:
            break
            
        if iteration % dual_update_freq == 0:
            # Compute all overlaps via convolution
            corr = correlate(current_h, 1 - current_h, mode='full')
            overlaps = corr * dx
            
            # Update dual variables using variational inference
            # λ_k ∝ exp(-overlap_k / temperature)
            lambda_k = np.exp(-overlaps / temperature)
            lambda_k = lambda_k / np.sum(lambda_k)  # Normalize
            
            # Update temperature
            temperature = max(1e-6, temperature * decay_rate)
        
        # Compute gradients for h
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Calculate gradient contribution from each lag
            for k in range(-n_points + 1, n_points):
                if 0 <= i - k < n_points:
                    grad[i] += lambda_k[k + n_points - 1] * (1 - current_h[i - k])
                elif i - k < 0:
                    # This corresponds to negative lag case; not important here
                    pass
        
        # Update h using gradient descent
        current_h = current_h - learning_rate * grad
        
        # Repair h to maintain constraints
        current_h = repair_h(current_h, n_points)
        
        # Evaluate current solution
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adapt learning rate based on stagnation
        if stagnation_count > max_stagnation:
            learning_rate *= 0.5
            stagnation_count = 0
            if learning_rate < 1e-8:
                break
    
    # Final verification and refinement
    final_h = repair_h(best_h, n_points)
    final_c5 = evaluate_h(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
