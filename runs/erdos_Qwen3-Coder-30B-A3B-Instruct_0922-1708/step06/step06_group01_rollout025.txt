import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and h[i] in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_to_feasible(h_init, n_points)
    else:
        h_init = np.random.rand(n_points)
        h_init = project_to_feasible(h_init, n_points)

    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current

    # Optimization setup
    max_iter = 10000
    step_size = 0.01
    perturb_rate = 0.05
    patience = 500
    last_improvement = 0

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break

        # Gradient approximation using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - c5_current) / eps

        # Update using gradient
        h_new = h_current - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            # Random perturbation with adaptive rate
            num_perturb = max(1, int(perturb_rate * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                delta = np.random.uniform(-0.05, 0.05)
                h_new[idx] = h_current[idx] + delta
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i

        # Adapt step size and perturbation rate
        if i - last_improvement > patience:
            step_size *= 0.9
            perturb_rate *= 1.1
            last_improvement = i

    # Final refinement with more aggressive local search
    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:

        grad = np.zeros(n_points)

            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - c5_best) / eps
            
        h_best = h_best - 0.005 * grad
        h_best = project_to_feasible(h_best, n_points)
        corr = correlate(h_best, 1 - h_best, mode='full')
        c5_best = np.max(corr) * dx

    return (h_best.tolist(), c5_best, n_points)
