import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Proximal gradient with Nesterov acceleration
    gamma = 0.01  # step size
    beta = 0.9    # momentum
    max_iter = 10000
    
    # For Nesterov acceleration
    v = np.zeros_like(h)
    h_acc = h.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation (overlap)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Identify the maximum lag
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h) - 1)
        
        # Compute gradient for the worst-case shift
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + k >= 0 and i + k < len(h):
                grad[i] = -dx * (1 - h[i + k])
        
        # Nesterov acceleration
        h_acc = h + beta * (h - best_h)
        
        # Proximal gradient step
        h_new = h_acc - gamma * grad
        
        # Proximal operator: clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure sum constraint
        h_new_sum = np.sum(h_new)
        if abs(h_new_sum - n_points/2) > 1e-8:
            h_new = h_new * (n_points/2) / h_new_sum
            
        # Update for next iteration
        v = beta * v + (h_new - h)
        h = h_new + beta * v
        
        # Ensure again the sum constraint
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2) > 1e-8:
            h = h * (n_points/2) / h_sum
        
        # Restart strategy if stuck
        if iteration > 0 and iteration % 1000 == 0 and best_c5 < 0.381:
            gamma *= 0.9
            
    # Final evaluation with best solution
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
