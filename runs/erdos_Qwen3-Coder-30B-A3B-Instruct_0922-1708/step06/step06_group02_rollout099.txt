import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Multi-resolution optimization
    resolutions = [10, 20, 40, 100]
    
    # Augmented Lagrangian parameters
    lambda_lag = 0.0
    rho = 1.0
    max_iter_per_res = 100
    tol = 1e-6
    
    for res in resolutions:
        # Interpolate to current resolution
        if res < n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_target = np.linspace(0, 2, res)
            h_interp = np.interp(x_target, x_orig, h)
            h = h_interp
            n_points = res
            dx = 2.0 / n_points
        else:
            if res > n_points:
                # Downsample if needed
                h = h[::(res//n_points)] if res//n_points > 1 else h
                n_points = res
                dx = 2.0 / n_points
        
        # Augmented Lagrangian optimization
        for iter_count in range(max_iter_per_res):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute functional and gradients
            # For augmented Lagrangian, we need C_5 and the constraint violation
            current_C5 = evaluate_h(h, dx)
            
            # Constraint check
            constraint_violation = abs(np.sum(h) * dx - 1.0)
            
            # Gradients by finite differences
            grad = np.zeros_like(h)
            eps = 1e-6
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_C5) / eps
            
            # Update h with augmented Lagrangian term
            lagrangian_grad = grad - lambda_lag * dx
            h_new = h - 0.001 * lagrangian_grad
            h_new = repair_h(h_new, n_points)
            
            # Update multiplier
            lambda_lag += rho * (np.sum(h_new) * dx - 1.0)
            
            # Update penalty if needed
            if constraint_violation > 1e-4:
                rho *= 2.0
                
            h = h_new
            
            # Check for improvement
            new_C5 = evaluate_h(h, dx)
            if new_C5 < best_value:
                best_value = new_C5
                best_h = h.copy()
            
            # Stagnation check
            if abs(new_C5 - current_C5) < tol:
                break
    
    # Further refinement with gradient descent on finest resolution
    h_refined = best_h.copy()
    current_value = best_value
    
    refine_iter = 0
    max_refine_iter = 500
    while refine_iter < max_refine_iter and time.time() - start_time < budget_s - 5:
        eps = 1e-6
        grad = np.zeros_like(h_refined)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_value) / eps
            
        h_refined = h_refined - 0.0001 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        current_value = np.max(corr) * dx
        
        if current_value < best_value:
            best_value = current_value
            best_h = h_refined.copy()
        
        refine_iter += 1
    
    return (best_h.tolist(), best_value, n_points)
