import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Store best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization using scipy
    def objective(h_vec):
        # Reshape to 1D
        h = h_vec.reshape(-1)
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def constraint_sum(h_vec):
        h = h_vec.reshape(-1)
        return np.sum(h) - n_points / 2
    
    # Set up constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization loop
    iter_count = 0
    max_iter = 5000
    
    while (time.time() - start_time) < budget_s - 5 and iter_count < max_iter:
        # Random restart to escape local minima
        if iter_count % 10 == 0:
            # Apply small random perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            # Re-scale
            h_sum = np.sum(h_perturbed)
            h = h_perturbed * (n_points / 2) / h_sum
        
        # Optimize with bounds and constraint
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 20}, callback=lambda x: None)
        
        if result.success:
            h = result.x.reshape(-1)
            # Normalize to correct sum
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-8:
                h = h * (n_points / 2) / h_sum
                
            # Evaluate objective
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        iter_count += 1
        
        # Occasionally attempt major perturbation
        if iter_count % 50 == 0:
            # Uniform random step in constrained space
            u = np.random.uniform(0, 1, n_points)
            h = u * (n_points / 2) / np.sum(u)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
