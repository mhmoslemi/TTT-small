import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1):
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, 0.1, len(h))
    h_mut = h + mask * noise
    return h_mut

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided seed if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init.copy()]
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init.copy()]
    
    # Add diversity through random sampling
    while len(population) < 50:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = None
    best_value = float('inf')
    
    # Evaluate initial population
    for h in population:
        value = evaluate_h(h, dx)
        if value < best_value:
            best_value = value
            best_h = h.copy()
    
    # Parameters
    population_size = 50
    generations = 1000
    elite_size = 10
    crossover_rate = 0.5
    mutation_rate = 0.1
    learning_rate = 0.001
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate current population
        fitness = [evaluate_h(h, dx) for h in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update global best
        if fitness[0] < best_value:
            best_value = fitness[0]
            best_h = population[0].copy()
        
        # Crossover and mutation to generate new population
        new_population = population[:elite_size]  # Keep best individuals
        
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate:
                # Tournament selection
                idx1 = np.random.randint(0, elite_size)
                idx2 = np.random.randint(0, elite_size)
                parent1 = population[idx1]
                parent2 = population[idx2]
                child = crossover(parent1, parent2)
            else:
                # Mutation only
                idx = np.random.randint(0, elite_size)
                child = mutate(population[idx])
            
            # Repair the child
            child = repair_h(child, n_points)
            
            new_population.append(child)
        
        population = new_population
        
        # Gradient refinement on top individual
        top_h = population[0]
        top_value = fitness[0]
        
        # Compute gradient
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = top_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - top_value) / eps
        
        # Update with gradient
        top_h = top_h - learning_rate * grad
        top_h = repair_h(top_h, n_points)
        
        # Evaluate refined version
        new_value = evaluate_h(top_h, dx)
        if new_value < top_value:
            population[0] = top_h
            if new_value < best_value:
                best_value = new_value
                best_h = top_h.copy()
        
        # Adjust learning rate if stagnation
        if gen % 100 == 0 and gen > 0:
            learning_rate *= 0.5
    
    return (best_h.tolist(), best_value, n_points)
