import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def interpolate_h(h_coarse, n_points):
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_points)
    return np.interp(x_fine, x_coarse, h_coarse)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution stages: 10, 20, 40, 100
    res_levels = [10, 20, 40, 100]
    best_h = None
    best_value = float('inf')
    
    # Initialize coarsest resolution
    n_points = res_levels[0]
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        coarse_h = h_init.copy()
    else:
        coarse_h = np.random.rand(n_points)
        coarse_h = coarse_h * (n_points / 2) / np.sum(coarse_h)
    
    # ES parameters for coarse level
    pop_size = 50
    mutation_std = 0.1
    selection_pressure = 0.2
    crossover_rate = 0.3
    
    # Coarse optimization
    population = [coarse_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    for gen in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        # Evaluate
        values = [evaluate_h(h, dx) for h in population]
        
        # Selection
        sorted_indices = np.argsort(values)
        top_indices = sorted_indices[:int(pop_size * selection_pressure)]
        top_population = [population[i] for i in top_indices]
        
        # Update best
        min_val = min(values)
        if min_val < best_value:
            best_value = min_val
            best_h = population[sorted_indices[0]].copy()
        
        # Create new population
        new_population = top_population.copy()
        
        # Crossover
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate:
                i1, i2 = np.random.choice(len(top_population), 2, replace=False)
                h1, h2 = top_population[i1], top_population[i2]
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, h1, h2)
            else:
                child = top_population[np.random.randint(len(top_population))].copy()
            
            # Mutation
            noise = np.random.normal(0, mutation_std, n_points)
            child = child + noise
            child = repair_h(child, n_points)
            new_population.append(child)
        
        population = new_population[:pop_size]
    
    # Refinement stages
    for i, n_points in enumerate(res_levels[1:], 1):
        if time.time() - start_time > budget_s - 10:
            break
            
        dx = 2.0 / n_points
        # Get top solutions from previous stage
        top_solutions = []
        if i == 1:
            top_solutions = [best_h.copy()]
        else:
            # Create solutions from previous finer resolution
            top_solutions = [interpolate_h(best_h, n_points)]
        
        # Add fresh random solutions to keep diversity
        while len(top_solutions) < pop_size:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            top_solutions.append(h)
        
        # Mutation std decreases with resolution
        mutation_std = 0.1 / np.sqrt(n_points)
        
        population = top_solutions.copy()
        
        for gen in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            values = [evaluate_h(h, dx) for h in population]
            
            # Selection
            sorted_indices = np.argsort(values)
            top_indices = sorted_indices[:int(pop_size * selection_pressure)]
            top_population = [population[i] for i in top_indices]
            
            # Update best
            min_val = min(values)
            if min_val < best_value:
                best_value = min_val
                best_h = population[sorted_indices[0]].copy()
            
            # Create new population
            new_population = top_population.copy()
            
            # Crossover
            while len(new_population) < pop_size:
                if np.random.rand() < crossover_rate:
                    i1, i2 = np.random.choice(len(top_population), 2, replace=False)
                    h1, h2 = top_population[i1], top_population[i2]
                    mask = np.random.rand(n_points) < 0.5
                    child = np.where(mask, h1, h2)
                else:
                    child = top_population[np.random.randint(len(top_population))].copy()
                
                # Mutation
                noise = np.random.normal(0, mutation_std, n_points)
                child = child + noise
                child = repair_h(child, n_points)
                new_population.append(child)
            
            population = new_population[:pop_size]
    
    # Final gradient descent refinement
    if best_h is not None:
        h_refined = best_h.copy()
        current_value = best_value
        
        # Gradient descent with adaptive learning rate
        learning_rate = 0.001
        for step in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            eps = 1e-6
            grad = np.zeros_like(h_refined)
            for j in range(len(h_refined)):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, len(h_refined))
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
                
            h_refined = h_refined - learning_rate * grad
            h_refined = repair_h(h_refined, len(h_refined))
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            current_value = np.max(corr) * dx
            
            if current_value < best_value:
                best_value = current_value
                best_h = h_refined.copy()
                
            # Decrease learning rate every 100 steps if no improvement
            if step > 0 and step % 100 == 0:
                learning_rate *= 0.5
                
    return (best_h.tolist(), best_value, len(best_h))
