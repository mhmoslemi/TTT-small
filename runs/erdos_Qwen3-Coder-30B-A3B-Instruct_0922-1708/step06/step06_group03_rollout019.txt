import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max, n_points):
    """Compute gradient of C5 with respect to h."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        # For each position i, the contribution to C5 is from the maximum overlap
        # The gradient is the derivative of max_k (h ⋆ (1 - h))[k] wrt h[i]
        # Only the term corresponding to k_max contributes
        j = (i - k_max) % n_points
        grad[i] = (1 - h[j]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Gradient descent parameters
    lr = 1e-4
    decay_rate = 0.99
    perturb_freq = 500
    perturb_mag = 0.05
    decay_perturb = 0.999
    patience = 100
    no_improve_count = 0
    
    for i in range(100000):  # Up to 100k steps (budget ~600s at ~1ms/step)
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap for all lags
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Compute gradient
        grad = compute_gradient(current_h, dx, k_max, n_points)
        
        # Update with gradient descent
        current_h = current_h - lr * grad
        
        # Apply perturbations periodically to escape local minima
        if i % perturb_freq == 0 and i > 0:
            perturb_mag *= decay_perturb
            indices = np.random.choice(n_points, max(1, n_points // 10), replace=False)
            for idx in indices:
                current_h[idx] += np.random.uniform(-perturb_mag, perturb_mag)
        
        # Repair solution
        current_h = repair_h(current_h, n_points)
        
        # Evaluate new C5
        current_c5 = evaluate_h(current_h, dx)
        
        # Track best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Adaptive learning rate
        if no_improve_count > patience:
            lr *= decay_rate
            no_improve_count = 0
        
        # Ensure non-negative learning rate
        lr = max(lr, 1e-8)
    
    # Final refinement step
    for j in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        corr = correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        grad = compute_gradient(best_h, dx, k_max, n_points)
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        best_c5 = evaluate_h(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
