import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Hyperparameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    alpha = 0.99
    beta = 0.999
    patience = 200
    patience_counter = 0
    
    for i in range(100000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Adaptive learning rate and perturbation
        if i % 100 == 0:
            learning_rate *= alpha
        if i % 500 == 0:
            perturbation_magnitude *= beta
            
        # Compute gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        corr_current = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr_current)
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        # Apply gradient descent with perturbation
        h_new = current_h - learning_rate * grad
        
        # Apply structured perturbation to escape local optima
        if pat_counter > patience:
            # Perturb indices with low gradient contribution
            low_grad_idx = np.where(np.abs(grad) < np.percentile(np.abs(grad), 20))[0]
            for idx in np.random.choice(low_grad_idx, max(1, len(low_grad_idx)//2), replace=False):
                delta = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
                h_new[idx] += delta
            pat_counter = 0
        else:
            pat_counter += 1
            
        # Repair
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                patience_counter = 0
            else:
                patience_counter += 1
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            # Recovery mechanism: reinitialize to best solution
            current_h = best_h.copy()
            current_c5 = best_c5
            patience_counter = 0
    
    # Final refinement
    h_final = best_h.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        corr_current = correlate(h_final, 1 - h_final, mode='full')
        k_max = np.argmax(corr_current)
        
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        h_final = h_final - 0.0001 * grad
        h_final = repair_h(h_final, n_points)
        corr = correlate(h_final, 1 - h_final, mode='full')
        best_c5 = np.max(corr) * dx
        
        if best_c5 < best_c5:
            best_h = h_final.copy()
            best_c5 = best_c5
    
    return (best_h.tolist(), best_c5, n_points)
