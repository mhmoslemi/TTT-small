import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, target_sum=50.0):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = project_h(h, target_sum=n_points/2)
    else:
        h = np.full(n_points, 0.5)
        h = project_h(h, target_sum=n_points/2)
    
    # Sparsity-promoting gradient descent
    lambda_reg = 0.01
    alpha = 0.1
    max_iter = 1000
    patience = 100
    best_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    prev_c5 = best_c5
    patience_counter = 0
    
    # Gradient computation
    def compute_gradient(h, eps=1e-6):
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, target_sum=n_points/2)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - prev_c5) / eps
        return grad
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute current gradient
        grad = compute_gradient(h)
        
        # L1 regularization gradient
        reg_grad = lambda_reg * np.sign(h)
        
        # Update
        h_new = h - alpha * (grad + reg_grad)
        h_new = project_h(h_new, target_sum=n_points/2)
        
        # Evaluate new C5
        new_c5 = evaluate_h(h_new, dx)
        
        # Backtracking line search
        step_size = 1.0
        while step_size > 1e-6:
            h_try = h - step_size * (grad + reg_grad)
            h_try = project_h(h_try, target_sum=n_points/2)
            try_c5 = evaluate_h(h_try, dx)
            if try_c5 < new_c5:
                h_new = h_try
                new_c5 = try_c5
                break
            step_size *= 0.5
        
        # Update
        h = h_new
        prev_c5 = new_c5
        
        # Check for improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience_counter = 0
            alpha = max(1e-4, alpha * 0.95)  # Adaptive decay
        else:
            patience_counter += 1
            if patience_counter > patience:
                # Reset with sparser solution
                h = project_h(np.random.rand(n_points) * 0.1, target_sum=n_points/2)
                patience_counter = 0
                alpha = 0.1
            else:
                alpha = min(1.0, alpha * 1.2)  # Increase learning rate
    
    return (best_h.tolist(), best_c5, n_points)
