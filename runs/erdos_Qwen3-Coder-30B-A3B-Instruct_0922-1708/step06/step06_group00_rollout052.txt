import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50000
    initial_temp = 0.1
    cooling_rate = 0.999
    tabu_size = 50
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    h_binary = (h > 0.5).astype(float)
    
    # Track best solution
    best_h = h_binary.copy()
    best_c5 = float('inf')
    
    # Tabu list to avoid revisiting same solutions
    tabu_list = []
    
    # Simulated annealing parameters
    temperature = initial_temp
    current_h = h_binary.copy()
    current_c5 = np.max(correlate(current_h, 1 - current_h, mode='full')) * dx
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Main optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping a random bit
        candidate_h = current_h.copy()
        flip_idx = np.random.randint(0, n_points)
        candidate_h[flip_idx] = 1 - candidate_h[flip_idx]
        
        # Maintain sum constraint by adjusting another bit
        current_sum = np.sum(current_h)
        if abs(current_sum - n_points/2) > 1e-8:
            # Adjust another bit to maintain constraint
            adjust_idx = np.random.randint(0, n_points)
            if adjust_idx != flip_idx:
                candidate_h[adjust_idx] = 1 - candidate_h[adjust_idx]
        
        # Ensure values stay in [0,1]
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Ensure constraint ∫h(x)dx = 1
        candidate_sum = np.sum(candidate_h)
        if abs(candidate_sum - n_points/2) > 1e-8:
            # Rescale to maintain constraint
            candidate_h = candidate_h * (n_points / 2) / candidate_sum
        
        # Compute C5 for candidate
        candidate_c5 = np.max(correlate(candidate_h, 1 - candidate_h, mode='full')) * dx
        
        # Check tabu list
        tabu_key = tuple(candidate_h.astype(int))
        if tabu_key in tabu_list and candidate_c5 >= current_c5:
            # Skip if in tabu list and not improving
            continue
        
        # Accept criteria for simulated annealing
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        # Update tabu list
        tabu_list.append(tabu_key)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Cooling schedule
        temperature *= cooling_rate
    
    # Final evaluation with high precision
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
