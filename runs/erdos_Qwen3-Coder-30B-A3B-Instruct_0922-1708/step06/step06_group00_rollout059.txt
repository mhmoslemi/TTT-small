import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure initial feasibility
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(h_sum - target_sum) > 1e-8:
        h = h * target_sum / h_sum

    best_h = h.copy()
    best_c5 = float('inf')
    
    max_iter = 100
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Identify the maximum lag
        max_lag_idx = np.argmax(corr)
        k_max = max_lag_idx - (len(h) - 1)
        
        # Formulate a convex subproblem to minimize the overlap at k_max
        h_var = cp.Variable(n_points)
        t = cp.Variable(1)
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        
        # Constraints
        # sum(h) = n_points/2
        constraints = [cp.sum(h_var) == n_points / 2]
        
        # h in [0,1]^n
        constraints += [h_var >= 0, h_var <= 1]
        
        # overlap_k_max <= t
        # overlap_k_max = 1 - (2/n_points) * sum(h_i * h_{i+k_max}) 
        # For simplicity, access shifted version of h
        if k_max >= 0:
            # h_{i+k_max} shifts into later positions
            h_shifted = h_var[k_max:]
            h_original = h_var[:len(h_shifted)]
        else:
            # h_{i+k_max} shifts into earlier positions 
            h_shifted = h_var[:k_max]
            h_original = h_var[-k_max:]
            
        # Need to make sure the overlap constraint is computed correctly for the shifted arrays
        # We'll handle k_max-specific case with a loop approach
        # Better: directly define the lagged correlation with explicit handling
        
        if k_max == 0:
            # overlap_0 = (2/n_points) * sum(h_i (1 - h_i)) = (2/n_points) * (sum(h_i) - sum(h_i^2))
            overlap_constraint = 1 - (2/n_points)*cp.sum(cp.multiply(h_var, h_var)) <= t
        else:
            # Consider all valid pairings in correlation
            # Form a matrix representation for efficiency
            # Compute the required dot product for the lag k_max
            
            # We want to compute:
            # sum_{i=0}^{n-k_max-1} h[i] * h[i+k_max] 
            # For positive k_max: h[i+k_max] is valid where i+k_max < n
            # i.e., i < n - k_max
            # So right range is from k_max to n-1 for component j, i.e., range [0, n-k_max)
            
            h_left = h_var[0:n-k_max] if k_max > 0 else h_var[-k_max:]
            h_right = h_var[k_max:] if k_max > 0 else h_var[:n+k_max]
            # Also need to negate: 1 - h[i+k_max] in total
            # So overlap = sum(h[i] * (1 - h[i+k_max])) 
            # == sum(h[i] - h[i]*h[i+k_max])
            # == sum(h[i]) - sum(h[i]*h[i+k_max]) 
            # == sum(h[i]) - dot(h_left, h_right)
            h_sum_term = cp.sum(h_var)
            dot_product = cp.sum(cp.multiply(h_left, h_right))  
            overlap_term = h_sum_term - dot_product
            overlap_constraint = (2/n_points)*overlap_term - 1 <= t 
                
        constraints.append(overlap_constraint)
        
        # Solve the convex subproblem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                # Ensure feasibility
                new_h = np.clip(new_h, 0, 1)
                h_sum_new = np.sum(new_h)
                if abs(h_sum_new - target_sum) > 1e-8:
                    new_h = new_h * target_sum / h_sum_new
                h = new_h
            else:
                # Fallback: gradient-like update
                # Simple update by computing approximate gradient and changing 
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
        except:
            # fallback update
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
