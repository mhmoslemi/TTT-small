import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    start_time = time.time()

    else:
        # Random initialization
        n_points = 20
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, 2.0 / n_points)
    
    # Simulated Annealing parameters
    T = 1.0
    T_min = 1e-6
    alpha = 0.99
    step_size = 0.1
    
    # Optimization parameters
    lr = 0.01
    decay_factor = 0.1
    decay_interval = 1000
    
    # Multi-scale loop
    scales = [20, 40, 80, 200]
    scale_idx = 0
    
    iteration = 0
    while T > T_min and (time.time() - start_time) < budget_s - 5:
        # Current resolution
        current_n = scales[scale_idx] if scale_idx < len(scales) else scales[-1]
        dx = 2.0 / current_n
        
        for _ in range(500):
            if (time.time() - start_time) > budget_s - 5:
                break
            
            # Perturb h with Gaussian noise scaled by current temperature
            h_new = h + np.random.normal(0, T * step_size, n_points)
            h_new = repair_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept or reject based on Metropolis criterion
            if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):
                h = h_new.copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            
            # Decay learning rate
            if iteration % decay_interval == 0 and iteration > 0:
                lr *= decay_factor
                
            iteration += 1
        
        # Increase resolution systematically
        if scale_idx < len(scales) - 1:
            # Interpolate to higher resolution
            x_high = np.linspace(0, 2, scales[scale_idx + 1])
            x_low = np.linspace(0, 2, current_n)
            h = np.interp(x_high, x_low, h)
            n_points = scales[scale_idx + 1]
            scale_idx += 1
            # Ensure integral constraint after interpolation
            h = h * (n_points / 2) / np.sum(h)
            
            # Gradient refinement for new resolution
            for _ in range(200):

                eps = 1e-6
                grad = np.zeros(n_points)
                c5_eval = evaluate_h(h, dx)

                    h_plus[i] += eps
