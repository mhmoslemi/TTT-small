import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def halton_sequence(n, base=2):
    """Generate n points from Halton sequence with given base."""
    result = np.zeros(n)
    for i in range(n):
        f = 1.0
        result[i] = 0.0
        while i > 0:
            f /= base
            result[i] = (result[i] + (i % base) * f)
            i //= base
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    
    # Use multiple initializations to avoid local optima
    num_initializations = 10
    for init_seed in range(num_initializations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate structured initial point using Halton sequence
        if 'initial_h_values' in globals() and init_seed == 0:
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_target = np.linspace(0, 2, n_points)
                h_init = np.interp(x_target, x_orig, h_init)
        else:
            # Generate a structured sequence
            halton_points = halton_sequence(n_points, 2)
            h_init = np.sin(2 * np.pi * halton_points) * 0.5 + 0.5
            
        # Make the sequence more structured
        h_init = repair_h(h_init, n_points)
        
        # Gradient descent optimization
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            
        # Optimization loop
        max_iter = 5000
        step_size = 0.01
        for iter_num in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient via finite differences
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_current.copy()
                h_minus[j] -= eps
                h_minus = repair_h(h_minus, n_points)
                c5_minus = evaluate_h(h_minus, dx)
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update step
            h_current = h_current - step_size * grad
            h_current = repair_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            
            # Update best
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
                
    return (best_h.tolist(), best_c5, n_points)
