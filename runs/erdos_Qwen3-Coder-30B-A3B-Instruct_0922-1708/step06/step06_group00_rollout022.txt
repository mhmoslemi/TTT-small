import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize step function
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Proximal gradient with Nesterov acceleration
    gamma = 0.01
    beta = 0.9
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Precompute correlation indices
    full_corr_len = 2 * n_points - 1
    
    for iteration in range(10000):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the shift that maximizes overlap
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (n_points - 1)
        
        # Compute gradient w.r.t. this shift
        grad = np.zeros(n_points)
        if k >= 0:
            # h[i] affects overlap with h[i+k] for i = 0..(n_points-1-k)
            for i in range(n_points - k):
                grad[i] -= (1 - h[i + k]) * dx
        else:
            # h[i] affects overlap with h[i+k] for i = -k..(n_points-1)
            for i in range(-k, n_points):
                grad[i] -= (1 - h[i + k]) * dx
        
        # Proximal gradient update with Nesterov acceleration
        v = beta * v + grad
        h_new = h - gamma * (grad + beta * v)
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        h_new_sum = np.sum(h_new)
        if abs(h_new_sum) > 1e-10:
            h_new = h_new * (n_points / 2) / h_new_sum
            
        h = h_new
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
