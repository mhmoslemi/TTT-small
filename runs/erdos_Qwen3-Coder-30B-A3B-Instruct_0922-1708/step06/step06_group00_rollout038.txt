import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h = (h_init > 0.5).astype(int)
        h_sum = np.sum(h)
        if h_sum != n_points // 2:
            # Adjust to have exactly n_points/2 ones
            num_ones_needed = n_points // 2
            num_ones_current = h_sum
            if num_ones_current < num_ones_needed:
                # Need to set more zeros to ones
                zero_indices = np.where(h == 0)[0]
                num_to_set = num_ones_needed - num_ones_current
                for i in range(num_to_set):
                    h[zero_indices[i]] = 1
            else:
                # Need to set some ones to zeros
                one_indices = np.where(h == 1)[0]
                num_to_unset = num_ones_current - num_ones_needed
                for i in range(num_to_unset):
                    h[one_indices[i]] = 0
    else:
        # Default uniform binary solution
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # Binary vector for MIP
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points // 2]
    
    # Compute overlap constraints for all possible shifts
    # Shift k ranges from -(n_points-1) to (n_points-1)
    for k in range(-n_points + 1, n_points):
        # Valid indices where both h[i] and h[i+k] are in bounds
        valid_indices = []
        for i in range(n_points):
            if 0 <= i + k < n_points:
                valid_indices.append((i, i + k))
        
        if valid_indices:
            # Contribution to overlap for this shift k
            overlap_expr = cp.sum([h_var[i] * (1 - h_var[j]) for i, j in valid_indices])
            constraints.append(overlap_expr <= t * (n_points / 2))
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with time limit
    try:
        # Time limit for solver
        time_limit = budget_s - 10
        prob.solve(solver=cp.CBC, time_limit=time_limit, verbose=False)
        
        # Extract solution
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.round(h_var.value).astype(int)
        else:
            # Fallback to previous best
            h_sol = h
    except Exception as e:
        # Fallback to initial h
        h_sol = h
    
    # Final validation
    h_final = h_sol
    corr = correlate(h_final, 1 - h_final, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (h_final.tolist(), c5_bound, n_points)
