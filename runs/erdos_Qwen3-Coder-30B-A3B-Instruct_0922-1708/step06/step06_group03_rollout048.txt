import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimization parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    max_iter = 100000
    patience = 200
    patience_counter = 0
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap values
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient (simplified version)
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Gradient contribution from the maximum overlap term
            # d/dh[j] (h[j] * (1 - h[j-k_max])) for the k_max term
            k = k_max - j
            if k < 0:
                k += n_points
            if k < n_points:
                # Approximate gradient for current term
                grad[j] += (1 - current_h[k]) * dx
        
        # Update with gradient descent
        new_h = current_h - learning_rate * grad
        
        # Apply structured perturbations occasionally
        if i % 100 == 0:
            # Find indices with small gradient magnitude
            indices = np.where(np.abs(grad) < 0.1)[0]
            if len(indices) > 0:
                # Apply small random perturbations to those indices
                perturb_indices = np.random.choice(indices, 
                                                   max(1, len(indices) // 5), 
                                                   replace=False)
                for idx in perturb_indices:
                    new_h[idx] += np.random.uniform(-perturbation_magnitude, 
                                                   perturbation_magnitude)
        
        # Repair and update
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
                patience_counter = 0
                last_improvement = i
            else:
                patience_counter += 1
        else:
            # Accept with probability based on simulated annealing
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / (0.1 * current_c5 if current_c5 > 0 else 0.01)):
                current_h = new_h
                current_c5 = new_c5
        
        # Adaptive scheduling
        if i % 100 == 0:
            learning_rate *= 0.99
            perturbation_magnitude *= 0.999
            
        # Reset learning rate if too long since improvement
        if i - last_improvement > 500:
            learning_rate = 1e-4
            last_improvement = i
    
    # Final refinement
    for _ in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        
        # Compute gradient more accurately
        corr = correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
        
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    # Final check
    best_h = repair_h(best_h, n_points)
    corr = correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
