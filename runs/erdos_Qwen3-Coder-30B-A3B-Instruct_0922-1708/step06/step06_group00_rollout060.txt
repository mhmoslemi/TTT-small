import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h = (h_init > 0.5).astype(int)
        h_sum = np.sum(h)
        if h_sum != n_points // 2:
            # Adjust to correct sum
            diff = n_points // 2 - h_sum
            if diff > 0:
                # Need to set more zeros to ones
                zero_indices = np.where(h == 0)[0]
                for i in range(min(diff, len(zero_indices))):
                    h[zero_indices[i]] = 1
            else:
                # Need to set more ones to zeros
                one_indices = np.where(h == 1)[0]
                for i in range(min(-diff, len(one_indices))):
                    h[one_indices[i]] = 0
        else:
            # Already correct sum, just make binary
            pass
    else:
        # Initialize with alternating pattern
        h = np.zeros(n_points, dtype=int)
        h[::2] = 1
        if np.sum(h) != n_points // 2:
            # Adjust if necessary
            if np.sum(h) < n_points // 2:
                # Set some zeros to ones
                zero_indices = np.where(h == 0)[0]
                for i in range(n_points // 2 - np.sum(h)):
                    h[zero_indices[i]] = 1
            else:
                # Set some ones to zeros 
                one_indices = np.where(h == 1)[0]
                for i in range(np.sum(h) - n_points // 2):
                    h[one_indices[i]] = 0
    
    # MIP formulation
    # Variables
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points // 2]
    
    # For each shift k, add constraint:
    # sum_{i} h[i] * (1 - h[i+k]) <= t * (n_points / 2)
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # h[i] * (1 - h[i + k]) for i = 0 to n_points-1-k
            valid_indices = list(range(n_points - k))
            expr = cp.sum(cp.multiply(h_var[valid_indices], 1 - h_var[valid_indices + k]))
        else:
            # h[i] * (1 - h[i + k]) for i = -k to n_points - 1
            valid_indices = list(range(-k, n_points))
            expr = cp.sum(cp.multiply(h_var[valid_indices], 1 - h_var[valid_indices - k]))
        constraints.append(expr <= t * (n_points / 2))

    # Solve
    prob = cp.Problem(objective, constraints)
    
    # Use CBC solver with time limit
    solve_time = min(100, budget_s - 10)
    try:
        prob.solve(solver=cp.CBC, time_limit=solve_time, log_freq=100)
        
        # If solved successfully, extract solution
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = np.round(h_var.value).astype(int)
        else:
            # If infeasible or not solved well, use current
            h_solution = h
    except Exception:
        h_solution = h
    
    # Ensure the final h is feasible
    h_solution = np.array(h_solution)
    if np.sum(h_solution) != n_points // 2:
        # Fix the sum
        diff = n_points // 2 - np.sum(h_solution)
        if diff > 0:
            zero_indices = np.where(h_solution == 0)[0]
            for i in range(min(diff, len(zero_indices))):
                h_solution[zero_indices[i]] = 1
        else:
            one_indices = np.where(h_solution == 1)[0]
            for i in range(min(-diff, len(one_indices))):
                h_solution[one_indices[i]] = 0
    
    # Compute final metric
    corr = correlate(h_solution, 1 - h_solution, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_solution.tolist(), final_c5, n_points)
