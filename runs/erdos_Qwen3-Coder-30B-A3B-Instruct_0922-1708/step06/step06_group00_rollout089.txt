import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = (h_init > 0.5).astype(int)
    else:
        # Default initialization: uniform binary pattern 
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # Ensure sum(h) = n_points/2
    current_sum = np.sum(h)
    if current_sum != n_points // 2:
        # Adjust by flipping some bits
        num_to_flip = abs(current_sum - n_points // 2)
        indices = np.where(h == 0)[0][:num_to_flip] if current_sum < n_points // 2 else np.where(h == 1)[0][:num_to_flip]
        h[indices] = 1 - h[indices]
    
    # Create binary optimization variables
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective: minimize maximum overlap
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points // 2]
    
    # For each shift k, ensure overlap constraint
    for k in range(-n_points + 1, n_points):
        # Compute valid indices for shift k
        valid_i = []
        valid_j = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_i.append(i)
                valid_j.append(j)
        
        if valid_i:
            # Add constraint for this shift
            overlap_expr = cp.sum(cp.multiply(h_var[valid_i], 1 - h_var[valid_j]))
            constraints.append(overlap_expr <= t * (n_points / 2))
    
    # Formulate the problem
    prob = cp.Problem(objective, constraints)
    
    # Set solver parameters
    solver = cp.CBC
    time_limit = budget_s - 10
    
    try:
        # Solve with optimization
        prob.solve(solver=solver, time_limit=time_limit, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Extract solution h from integer variables
            h_solution = h_var.value
            h_solution = np.round(h_solution).astype(int)
            h_solution = np.clip(h_solution, 0, 1)
            
            # Ensure sum constraint
            current_sum = np.sum(h_solution)
            if current_sum != n_points // 2:
                num_to_flip = abs(current_sum - n_points // 2)
                indices = np.where(h_solution == 0)[0][:num_to_flip] if current_sum < n_points // 2 else np.where(h_solution == 1)[0][:num_to_flip]
                h_solution[indices] = 1 - h_solution[indices]
            
            h = h_solution
            
        # Fallback: stick with previous best if solution is not found or invalid
        else:
            pass
            
    except Exception as e:
        pass
    
    # Final evaluation using the best h
    corr = correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr) * dx
    
    # Ensure h is step function (0 or 1)
    h = h.astype(float)
    
    return (h.tolist(), c5_bound, n_points)
