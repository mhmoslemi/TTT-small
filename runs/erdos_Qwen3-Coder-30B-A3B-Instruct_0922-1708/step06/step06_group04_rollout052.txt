import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Build constraint matrix for overlaps
    # For each lag k in range [-n_points+1, n_points-1], collect valid indices
    # Compute all valid overlaps as linear constraints
    h_vars = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints: each overlap <= t
    constraints = []
    
    # Need to compute valid lag indices and constraints
    # For each lag k, we compute the sum h[i] * (1 - h[i-k]) where valid
    # That is, for i in [0, n_points-1] and i-k in [0, n_points-1]
    # So k must be in [i-n_points+1, i]
    
    # Omit the circular convolution for accuracy
    
    # Create overlap constraints for non-circular convolution
    max_lag = n_points - 1
    total_overlaps = 2 * max_lag + 1  # k from -max_lag to max_lag
    
    # Classify valid overlaps by lag
    k_range = range(-max_lag, max_lag + 1)
    
    # Generate constraints
    for k in k_range:
        # Compute indices where overlap is valid
        i_indices = np.arange(n_points)
        j_indices = i_indices - k
        
        # Keep only valid transitions within [0, n_points-1]
        valid_mask = (j_indices >= 0) & (j_indices < n_points)
        i_valid = i_indices[valid_mask]
        j_valid = j_indices[valid_mask]
        
        # Create constraint: sum h[i] * (1 - h[j]) <= t * dx * total_overlap
        # which scales to sum h[i] * (1 - h[j]) <= t * dx * n_points / 2
        # Scale by dx to match it to the continuous version
        # since the integral is scaled by dx, use t * n_points / 2
        # but we want to minimize t vs. t * dx, so scale limit to like:
        # using the overlapping part (this is a bit tricky)
        # simplify to just hard thresholding using actual Hamming-like sums
        # In discrete, overlapping is sum_{i=0}^{n_points-1} h[i] * (1 - h[(i+k)%n_points])
        
        if len(i_valid) > 0:
            # Multiply h[i] * (1 - h[j]) for valid pairs
            left = h_vars[i_valid]
            right = 1 - h_vars[j_valid]
            if len(i_valid) == 1:
                overlap = left[0] * right[0]
            else:
                overlap = cp.sum(cp.multiply(left, right))
            
            # C5 should be in terms of normalized maximum, so align units by scaling
            # overlap = sum(h[i] * (1 - h[i+k])) * dx for k fixed
            # dt = float(n_points)  # To normalize
            constraints.append(overlap <= t * (n_points / 2))
        
    # Add box constraints and sum constraint
    constraints.append(h_vars >= 0)
    constraints.append(h_vars <= 1)
    constraints.append(cp.sum(h_vars) == n_points / 2)
    
    # minimize t
    problem = cp.Problem(cp.Minimize(t), constraints)
    
    # try to solve with available solvers
    try:
        # Use high-precision solver unless unavailable
        problem.solve(solver=cp.MOSEK, verbose=False)
        if problem.status in ['optimal', 'optimal_inaccurate']:
            h_result = h_vars.value
            c5_bound = t.value * dx
        else:
            # fallback
            h_result = h_init
            c5_bound = evaluate_erdos_solution(h_result, dx) 
    except:
        # fallback to simple gradient descent on the best initial
        h_result = h_init.copy()
        c5_bound = evaluate_erdos_solution(h_result, dx)
    
    # refine with simple gradient ascent on h to reduce bound
    best_h = h_result.copy()
    best_c5 = c5_bound
    
    def eval_h(h):
        return evaluate_erdos_solution(h, dx)
    
    # Limited BFGS-like update; start with a small perturbation
    for _ in range(200):
        if time.time() - start_time > budget_s - 10:
            break
            
        eps = 1e-6
        grad = np.zeros_like(best_h)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = eval_h(h_plus)
            grad[j] = (c5_plus - best_c5) / eps
            
        # update with step size 1e-4
        new_h = best_h - 1e-4 * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        new_c5 = eval_h(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    # replicate your previous evaluation logic
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
