import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1 - np.abs(x - 1)
        h_init = repair_h(h_init, n_points)
    
    # Coordinate descent with projection
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    learning_rate = 0.01
    epsilon = 1e-6
    max_iter = 100000
    plateau_window = 100
    plateau_count = 0
    best_c5 = c5_current
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Coordinate-wise updates
        for j in range(n_points):
            # Compute finite difference gradient
            h_plus = h_current.copy()
            h_plus[j] += epsilon
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[j] -= epsilon
            h_minus = repair_h(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update
            h_current[j] -= learning_rate * grad
            h_current[j] = np.clip(h_current[j], 0, 1)
            
        # Project to feasible set
        h_current = repair_h(h_current, n_points)
        
        # Evaluate
        c5_current = evaluate_h(h_current, dx)
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            best_c5 = c5_current
            plateau_count = 0
        else:
            plateau_count += 1
            
        # Adaptive learning rate
        if plateau_count > plateau_window:
            learning_rate *= 0.9
            plateau_count = 0
            
        # Perturb occasionally
        if i % 1000 == 0 and i > 0:
            perturb_indices = np.random.choice(n_points, int(0.05 * n_points))
            for idx in perturb_indices:
                h_current[idx] += np.random.normal(0, 0.001)
            h_current = repair_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current

    return (h_best.tolist(), c5_best, n_points)
