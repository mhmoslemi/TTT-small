import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set: [0,1]^n with sum = n/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def gradient_approx(h, dx, eps=1e-6):
    """Approximate gradient of C5 with finite differences."""
    n_points = len(h)
    grad = np.zeros(n_points)
    c5_base = evaluate_h(h, dx)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n_points)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

def solve_linear_subproblem(grad, n_points):
    """Solve linear subproblem for Frank-Wolfe: min_d grad^T d s.t. d in feasible set."""
    # Simple approach: take direction that minimizes the gradient dot product
    # subject to constraints. This is a simplified L1 projection.
    d = -grad
    # Enforce sum(d) = 0 constraint
    d = d - np.mean(d)
    # Ensure sum to 0
    return d

def line_search(h, d, dx, max_eta=1.0):
    """Perform backtracking line search to find suitable eta."""
    eta = max_eta
    c5_base = evaluate_h(h, dx)
    while eta > 1e-8:
        h_new = h + eta * d
        h_new = project_h(h_new, len(h))
        c5_new = evaluate_h(h_new, dx)
        if c5_new <= c5_base:
            return eta
        eta *= 0.5
    return eta

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured construction
        h_init = np.ones(n_points) * 0.5
        # Concentrate mass in a small interval
        center = n_points // 2
        width = n_points // 4
        for i in range(max(0, center - width), min(n_points, center + width)):
            h_init[i] = 1.0
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    patience = 0
    max_patience = 1000
    
    # Initialize Frank-Wolfe
    eta_max = 1.0
    
    iteration = 0
    while time.time() - start_time < budget_s - 5 and patience < max_patience:
        iteration += 1
        
        # Compute gradient
        grad = gradient_approx(h_current, dx)
        
        # Solve linear subproblem
        d = solve_linear_subproblem(grad, n_points)
        
        # Perform line search
        eta = line_search(h_current, d, dx, eta_max)
        if eta <= 1e-6:
            eta = 1e-2  # fallback
        eta_max = eta
        
        # Update
        h_new = h_current + eta * d
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            
        # Occasionally recompute gradient after large step
        if iteration % 50 == 0:
            c5_current = evaluate_h(h_current, dx)
    
    return (h_best.tolist(), c5_best, n_points)
