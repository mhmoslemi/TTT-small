import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    start_time = time.time()
    
    # Setup CVXPY optimization
    h = cp.Variable(n_points, name='h')
    t = cp.Variable(name='t')
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        t >= 0
    ]
    
    # Add overlap constraints
    max_lag = n_points - 1  # max lag we can compute
    for lag in range(-max_lag, max_lag + 1):
        if lag < 0:
            # Forward overlap
            indices = np.arange(-lag, n_points)
            lag_indices = np.arange(0, n_points + lag)
        elif lag > 0:
            # Backward overlap
            indices = np.arange(0, n_points - lag)
            lag_indices = np.arange(lag, n_points)
        else:
            # Self overlap
            indices = np.arange(0, n_points)
            lag_indices = np.arange(0, n_points)
        
        if len(indices) == 0:
            continue
            
        # Compute overlap for this lag
        overlap_expr = cp.sum(cp.multiply(h[indices], 1 - h[lag_indices]))
        constraints.append(overlap_expr <= t * (n_points / 2))
    
    # Objective
    objective = cp.Minimize(t)
    
    # Formulate problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - 10)
        
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_solution = h.value
        else:
            # Fallback to initial solution
            h_solution = h_init
            
    except:
        # Fallback to initial solution
        h_solution = h_init
    
    # Ensure final solution satisfies constraints
    h_solution = np.clip(h_solution, 0, 1)
    h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
    
    # Final evaluation
    final_c5 = evaluate_h(h_solution, dx)
    
    return (h_solution.tolist(), final_c5, n_points)
