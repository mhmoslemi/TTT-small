import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to the feasible set: 0 <= h[i] <= 1 and sum(h) = target_sum"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to target sum
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, target_sum / len(h))
    return h

def gradient_step(h, dx, epsilon=1e-6):
    """Compute gradient of C5 with finite differences."""
    n = len(h)
    grad = np.zeros(n)
    c5_base = evaluate_h(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project_to_feasible(h_plus, n/2)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / epsilon
    return grad

def frank_wolfe_step(h, dx, max_iter=100):
    """Frank-Wolfe step to minimize C5"""
    n = len(h)
    # Linear subproblem: minimize <grad, d> subject to d in feasible set
    # The feasible set here is: 0 <= d[i] <= 1, sum(d) = 0
    # Since sum(d) = 0, we're looking for the point that minimizes gradient dot product
    # This is effectively: minimize sum_i grad[i] * d[i] subject to constraints
    
    # Simple solution: take minimum over indices
    # But for Frank-Wolfe in constrained simplex case,
    # better approach: use active set methods or simple projected gradient
    
    # For now: use projected gradient descent as a proxy to ensure good step
    grad = gradient_step(h, dx)
    
    # Frank-Wolfe direction: solve min_d <grad, d> subject to d feasible
    # This is equivalent to taking the most negative gradient direction
    # and way to project onto feasible set with sum=0
    # since we're in the sum=0 constraint case, can do better
    
    # Simple approximate step
    # Create a step that enforces sum(d) = 0, and 0 <= d[i] <= 1
    # and minimizes gradient*step
    step_size = 1.0
    d = -grad
    d = project_to_feasible(d, 0)  # This is issue, not exactly sum=0?
    
    # Better: since we can't easily solve linear program here, we'll do a
    # line search variant.
    # However let me use basic approach of fixed step and grad descent variant.
    return d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = project_to_feasible(h, n_points / 2)
    else:
        # Start with a triangular shape
        x = np.linspace(0, 2, n_points)
        h = 1.0 - np.abs(x - 1.0)  # Triangle centered at 1
        h = project_to_feasible(h, n_points / 2)
    
    # Frank-Wolfe style optimization with some heuristics
    h_current = h.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    patience = 0
    max_patience = 1000
    
    step_size = 1.0
    
    for iteration in range(20000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        grad = gradient_step(h_current, dx)
        
        # Perform Frank-Wolfe step (implicitly approximated)
        # Take a step in gradient direction, scaled appropriately
        d = -grad
        # Normalize to reasonable scale
        norm_d = np.linalg.norm(d)
        if norm_d > 0:
            d = d / norm_d
        # Take adaptive step size
        step = step_size * d
        
        # Try different step sizes to find a descent
        best_step_size = 0.0
        best_h_new = h_current.copy()
        best_c5 = c5_current
        
        try_candidates = [0.001, 0.005, 0.01, 0.05, 0.1, 0.2]
        for candidate in try_candidates:
            h_proposed = h_current + candidate * step
            h_proposed = project_to_feasible(h_proposed, n_points / 2)
            c5_proposed = evaluate_h(h_proposed, dx)
            if c5_proposed < best_c5:
                best_c5 = c5_proposed
                best_h_new = h_proposed.copy()
                best_step_size = candidate
        
        if best_c5 < c5_current:
            h_current = best_h_new
            c5_current = best_c5
            if c5_current < c5_best:
                c5_best = c5_current
                h_best = h_current.copy()
                patience = 0
            else:
                patience += 1
            # Decrease step size when improving
            step_size = max(0.0001, step_size * 0.99)
        else:
            patience += 1
            step_size = max(0.0001, step_size * 0.95)

        if patience >= max_patience:
            # Restart with scaled new random
            h_current = project_to_feasible(np.random.rand(n_points), n_points / 2)
            patience = 0
            step_size = 0.1

    # Final check
    h_final = project_to_feasible(h_best, n_points / 2)
    c5_final = evaluate_h(h_final, dx)
    
    return (h_final.tolist(), c5_final, n_points)
