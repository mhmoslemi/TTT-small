import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to feasible set: [0,1]^n and sum(h) = n/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Project to sum = n/2 using quadratic programming
    # Minimize ||h_new - h||^2 subject to sum(h_new) = n/2 and 0 <= h_new <= 1
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Simple projection: adjust by a constant
    adjustment = (target_sum - current_sum) / n_points
    h = h + adjustment
    
    # Clip again after adjustment
    h = np.clip(h, 0, 1)
    return h

    else:

        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)

            h_plus = h_current.copy()

            h_plus = project_to_feasible(h_plus, n_points)

            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_current) / eps
        
        # Update with gradient
        h_new = h_current - lr * grad
        h_new = project_to_feasible(h_new, n_points)

        # Adaptive learning rate
        if i - last_improvement > patience:
            lr *= 0.9
            last_improvement = i

    return (h_best.tolist(), c5_best, n_points)
