import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fw_subproblem(gradient, n_points):
    """Solve the Frank-Wolfe linear subproblem."""
    # We want to minimize <gradient, d> subject to:
    # 0 <= d[i] <= 1, sum d = 0
    # This is equivalent to finding the direction that most decreases the objective
    # The constraint d[i] in [0,1], sum d = 0, with sum d_i = 0
    
    # Approximate solution:
    # Find minimum in direction of gradient, respecting constraints
    d = np.zeros(n_points)
    
    # Compute signs for gradient
    sign = np.sign(gradient)
    
    # Initialize with sign of gradient
    # For sum constraint: keep sign of gradient to maximize negative dot product
    # This should work for approximate solution to linear subproblem
    d = -sign
    
    # Project onto sum=0 
    d = d - np.mean(d)
    
    # Clip to [0,1] and ensure sum=0
    d = np.clip(d, 0, 1)
    d = d - np.mean(d)
    
    # Normalize step: make sure we don't go too far
    norm_d = np.linalg.norm(d) 
    if norm_d > 1e-10:
        d = d / norm_d
        
    return d

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a structured, reasonably good solution
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.tanh(2.5 * (0.5 - x)))
        h_init = repair_h(h_init, n_points)
    
    # Frank-Wolfe parameters
    max_iter = 10000
    epsilon = 1e-6
    eta_max = 1.0
    
    # Current solution
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Track last improvement and patience
    last_improvement = 0
    patience = 1000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        gradient = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += epsilon
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            gradient[j] = (c5_plus - c5_current) / epsilon
        
        # Solve linear subproblem to get descent direction
        d = fw_subproblem(gradient, n_points)
        
        # Line search to find optimal eta
        eta = eta_max
        h_new = h_current + eta * d
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Reduce step size if no improvement
        while c5_new >= c5_current and eta > 1e-8:
            eta /= 2.0
            h_new = h_current + eta * d
            h_new = repair_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
        # Update current solution
        h_current = h_new
        c5_current = c5_new
        
        # Update best solution if improved
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Increase patience if no beneficial improvement
        if i - last_improvement > patience:
            break

    # Final validation and return
    h_final = h_best.copy()
    c5_final = c5_best
    
    return (h_final.tolist(), c5_final, n_points)
