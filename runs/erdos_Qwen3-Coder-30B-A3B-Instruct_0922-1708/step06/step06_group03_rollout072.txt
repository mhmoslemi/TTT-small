import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimization parameters
    learning_rate = 1e-4
    perturbation_mag = 0.05
    max_iter = 100000
    stagnation_count = 0
    max_stagnation = 200
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all overlaps to find the maximum
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Compute gradient for the maximum overlap term
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Gradient contribution for overlap at lag k_max
            # This is a simplified estimation; full gradient requires
            # partial derivatives of the convolution
            g = 0
            for k in range(len(corr)):
                if k == k_max:
                    # The gradient of h[j] in this term
                    # The term is h[j] * (1 - h[(j - k) % n_points])
                    # If we're calculating the gradient w.r.r. h[j]:
                    # This term contributes (1 - h[(j - k) % n_points]) to the gradient
                    idx = (j - k) % n_points
                    g += 1 - current_h[idx]
            grad[j] = g * dx
            
        # Apply gradient descent step
        h_candidate = current_h - learning_rate * grad

        # Apply perturbations to escape local optima
        if i % 100 == 0 and i > 0:
            # Perturb indices with low gradient magnitude
            abs_grad = np.abs(grad)
            low_grad_indices = np.where(abs_grad < 0.1)[0]
            if len(low_grad_indices) > 0:
                num_perturb = min(len(low_grad_indices), n_points // 10)
                indices = np.random.choice(low_grad_indices, num_perturb, replace=False)
                for idx in indices:
                    perturbation = np.random.uniform(-perturbation_mag, perturbation_mag)
                    h_candidate[idx] += perturbation
        
        # Repair to maintain feasibility
        h_candidate = repair_h(h_candidate, n_points)
        candidate_c5 = evaluate_h(h_candidate, dx)
        
        # Accept or reject the candidate
        if candidate_c5 < current_c5:
            current_h = h_candidate
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Update parameters
        if i % 100 == 0:
            learning_rate *= 0.99
            perturbation_mag *= 0.999
            
        # Adaptive cooling if no improvement
        if stagnation_count > max_stagnation:
            learning_rate = max(1e-6, learning_rate * 0.9)
            stagnation_count = 0

    # Final refinement
    h_refined = best_h.copy()
    best_c5_refined = best_c5
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        
        # Compute gradient via finite difference
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5_refined) / eps
            
        h_refined = h_refined - 0.0001 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_c5_refined = np.max(corr) * dx
        
        if best_c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = best_c5_refined
    
    # Final verification
    test_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(test_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
