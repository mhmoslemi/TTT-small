import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    """Create a child from two parents using average crossover."""
    alpha = np.random.random()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1):
    """Mutate h by adding Gaussian noise."""
    noise = np.random.normal(0, 0.1, len(h))
    mask = np.random.random(len(h)) < mutation_rate
    h = h + noise * mask
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    elite_size = pop_size // 5
    mutation_rate = 0.1
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init.copy()]
    else:
        population = []
    
    # Fill population with random feasible solutions
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(p, dx) for p in population]
    
    # Find best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    # Evolution loop
    generations = 1000
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_population = population[:elite_size]
        
        # Create offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = sorted_indices[np.random.randint(elite_size)]
            parent2_idx = sorted_indices[np.random.randint(elite_size)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            if np.random.random() < 0.5:
                child = crossover(parent1, parent2)
                child = repair_h(child, n_points)
            else:
                child = population[parent1_idx].copy()
                child = mutate(child, mutation_rate)
                child = repair_h(child, n_points)
                
            new_population.append(child)
            
        population = new_population
        
        # Evaluate new population
        fitness = [evaluate_h(p, dx) for p in population]
        
        # Update best solution
        min_fitness = min(fitness)
        if min_fitness < best_value:
            best_value = min_fitness
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
        
        # Gradient refinement on best solution
        if gen % 10 == 0:
            h_refined = best_h.copy()
            current_value = best_value
            learning_rate = 0.001
            
            for _ in range(50):

                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_value) / eps
                    
                h_refined = h_refined - learning_rate * grad
                h_refined = repair_h(h_refined, n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                current_value = np.max(corr) * dx
                
                if current_value < best_value:
                    best_value = current_value
                    best_h = h_refined.copy()
    
    return (best_h.tolist(), best_value, n_points)
