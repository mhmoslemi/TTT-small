import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_to_feasible(h_init, n_points)
    else:
        # Initialize with structured pattern
        h_init = np.zeros(n_points)
        # Triangular waveform
        x_vals = np.linspace(0, 2, n_points)
        for i, x in enumerate(x_vals):
            if 0.5 <= x <= 1.5:
                h_init[i] = 1.0 - abs(x - 1.0)
            else:
                h_init[i] = 0.0
        h_init = project_to_feasible(h_init, n_points)

    # Frank-Wolfe optimization with aggressive projection
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    h_best = h_current.copy()
    c5_best = c5_current

    # Adaptive line search parameters
    max_line_search = 10
    step_size = 1.0
    patience = 1000
    last_improvement = 0

    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        if iter_count > patience:
            break

        # Compute gradient approximately using finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_current) / eps

        # Linearized subproblem: minimize grad^T d subject to d in feasible set
        # This is a simplified approach: we find the direction that minimizes
        # the linear approximation, with constraints
        d = -grad  # Negative gradient (descent direction)
        
        # Simple simplex projection (ensure sum(d) = 0)
        d = d - np.mean(d)
        
        # Optimize step size
        best_step = 0.0
        best_c5 = c5_current
        current_step = step_size
        for _ in range(max_line_search):
            h_new = h_current + current_step * d
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_step = current_step
            else:
                break
            current_step *= 0.5

        h_current = h_current + best_step * d
        h_current = project_to_feasible(h_current, n_points)
        c5_current = best_c5

        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = iter_count

    return (h_best.tolist(), c5_best, n_points)
