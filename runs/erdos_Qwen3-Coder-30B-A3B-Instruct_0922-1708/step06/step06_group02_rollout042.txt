import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.1):
    mask = np.random.rand(len(h)) < mutation_rate
    if np.any(mask):
        noise = np.random.normal(0, noise_std, len(h))
        h = h + mask * noise
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial construct if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        
    # Initialize population
    pop_size = 50
    population = [repair_h(h_init.copy(), n_points)]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
        
    best_h = None
    best_value = float('inf')
    
    # Evaluate initial population
    fitness = []
    for h in population:
        val = evaluate_h(h, dx)
        fitness.append(val)
        if val < best_value:
            best_value = val
            best_h = h.copy()
    
    # Generation loop
    max_generations = 1000
    no_improvement = 0
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort population by fitness
        idxs = np.argsort(fitness)
        elites = [population[idx] for idx in idxs[:pop_size // 5]]
        
        # Select parents
        parents = []
        for i in range(pop_size // 2):
            p1 = population[idxs[np.random.randint(0, pop_size // 2)]]
            p2 = population[idxs[np.random.randint(0, pop_size // 2)]]
            parents.append(crossover(p1, p2))
            
        # Mutate
        mutated = []
        for h in parents:
            h = mutate(h, 0.1, 0.1)
            h = repair_h(h, n_points)
            mutated.append(h)
            
        # Replace worst with new individuals
        new_pop = elites + mutated
        new_fitness = []
        for h in new_pop:
            val = evaluate_h(h, dx)
            new_fitness.append(val)
            if val < best_value:
                best_value = val
                best_h = h.copy()
                
        # Update population
        idxs_new = np.argsort(new_fitness)
        population = [new_pop[i] for i in idxs_new[:pop_size]]
        fitness = [new_fitness[i] for i in idxs_new[:pop_size]]
        
        if gen % 50 == 0:
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > 100:
            break
            
    # Final refinement with gradient
    h_refined = best_h.copy()
    current_value = best_value
    learning_rate = 0.001
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_value) / eps
            
        h_refined = h_refined - learning_rate * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        current_value = np.max(corr) * dx
        
        if current_value < best_value:
            best_value = current_value
            best_h = h_refined.copy()
            
    return (best_h.tolist(), best_value, n_points)
