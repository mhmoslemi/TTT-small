import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    else:
        h = np.random.rand(n_points)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Parameters
    pop_size = 50
    F = 1.0
    CR = 0.9
    max_generations = 1000
    stagnation_limit = 100
    stagnation_count = 0
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            pop_h = h.copy()
            pop_h += np.random.normal(0, 0.01, n_points)
        else:
            pop_h = np.random.rand(n_points)
        pop_h = repair_h(pop_h, n_points)
        population.append(pop_h)
    
    if 'initial_h_values' in globals():
        population[0] = h.copy()
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        new_population = []
        fitnesses = []
        
        for i in range(pop_size):
            # Mutation and crossover
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = repair_h(mutant, n_points)

            trial = np.where(np.random.rand(n_points) < CR, mutant, population[i])
            trial = repair_h(trial, n_points)
            
            # Evaluate
            c5 = evaluate_h(trial, dx)
            fitnesses.append(c5)
            new_population.append(trial)
        
        # Selection
        indices = np.argsort(fitnesses)
        population = [new_population[i] for i in indices[:pop_size//2]]
        
        # Keep best
        for i in range(len(indices)):
            if fitnesses[indices[i]] < best_c5:
                best_c5 = fitnesses[indices[i]]
                best_h = new_population[indices[i]].copy()
                stagnation_count = 0
            if stagnation_count > stagnation_limit:
                break
                
        stagnation_count += 1
        
        # Adaptive F
        if generation % 50 == 0 and generation > 0:
            F = min(1.5, F + 0.1)
    
    # Refinement at higher resolution
    if time.time() - start_time < budget_s - 10:
        n_points_high = 200
        dx_high = 2.0 / n_points_high
        x_target = np.linspace(0, 2, n_points_high)
