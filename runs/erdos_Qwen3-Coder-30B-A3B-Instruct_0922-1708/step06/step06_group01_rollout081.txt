import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def project_to_simplex(y, target_sum):
    """Project y to the simplex with sum = target_sum"""
    # Sort in descending order
    u = np.sort(y)[::-1]
    # Compute the cumulative sums
    cu = np.cumsum(u)
    # Find the threshold
    rho = np.where(u > (target_sum - cu + u) / np.arange(1, len(u) + 1))[0]
    if len(rho) == 0:
        # All elements are negative
        return np.zeros_like(y)
    rho = rho[-1]  # Last element that satisfies the condition
    # Compute the threshold
    theta = (cu[rho] - target_sum) / (rho + 1)
    # Project
    x = np.maximum(y - theta, 0)
    return x

def fw_subproblem(gradient, n_points):
    """Solve the FW linear subproblem"""
    # We want to minimize <gradient, d> s.t. d in [0,1]^n and sum(d) = 0
    # Adjust gradient for the constraints
    # Since sum(d) = 0, let's find the direction that maximizes <gradient, d>
    # subject to d in [0,1]^n and sum(d) = 0
    # The direction is sign(-gradient)* indicator of which bound is hit
    # However, to be careful, let's solve this minimum.
    # direction h = argmin_{||h||_1 <= 1} <gradient, h> s.t. sum(h) = 0
    
    # Start with -gradient, then project to feasible set
    direction = -gradient
    # Make sum zero
    direction = direction - np.mean(direction)
    # Project to [0,1]^n
    direction = np.clip(direction, 0, 1)
    # Ensure sum = 0
    direction = direction - np.mean(direction)
    return direction
