import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 30
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Clip and rescale
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the worst shift
        k_max = np.argmax(corr) - (len(h) - 1)
        
        # Formulate convex subproblem
        h_var = cp.Variable(n_points)
        t_var = cp.Variable(1)
        
        # Objective: minimize t
        objective = cp.Minimize(t_var)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Bound constraints
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Worst-case overlap constraint
        # For shift k_max, overlap = sum h[i] * (1 - h[i + k_max])
        # This is: sum h[i] - sum h[i] * h[i + k_max]
        # Or equivalently: 1 - sum h[i] * h[i + k_max]
        # Which we constrain as <= t
        
        if k_max >= 0:
            # h[i + k_max] refers to elements at indices [k_max, ..., -1]
            # h[i] refers to elements at indices [0, ..., -1-k_max]
            h_i = h_var[:n_points-k_max]
            h_i_k = h_var[k_max:]
            corr_term = cp.sum(cp.multiply(h_i, h_i_k))
        else:
            # k_max < 0, so h[i + k_max] refers to elements at indices [0, ..., -1+k_max]
            # h[i] refers to elements at indices [-k_max, ..., -1]
            h_i = h_var[-k_max:]
            h_i_k = h_var[:len(h_i)]
            corr_term = cp.sum(cp.multiply(h_i, h_i_k))
        
        overlap_constraint = 1 - corr_term <= t_var
        constraints.append(overlap_constraint)
        
        # Form and solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            # Try different solvers in case one fails
            prob.solve(solver=cp.ECOS, max_iters=1000)
            # prob.solve(solver=cp.MOSEK, verbose=False)
        except:
            try:
                prob.solve(solver=cp.SCS, max_iters=5000)
            except:
                # Fallback: simple gradient step with clipping
                # Compute gradient to reduce overlap for the k_max
                grad = np.zeros_like(h)
                if k_max >= 0:
                    # h[i + k_max] refers to elements [k_max, ..., -1]
                    # Gradient: negative partial derivatives
                    for i in range(n_points - k_max):
                        grad[i] -= h[i + k_max]
                    for i in range(k_max, n_points):
                        grad[i] -= h[i - k_max]
                else:
                    # k_max < 0
                    for i in range(-k_max):
                        grad[i] -= h[i - k_max]
                    for i in range(n_points + k_max):
                        grad[i] -= h[i + k_max]
                grad = 2 * grad / n_points  # Normalize
                
                # Gradient descent step
                h = h - 0.01 * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                continue

        if prob.status == "optimal":
            h_new = h_var.value
            if h_new is not None:
                h = np.clip(h_new, 0, 1)
                h_sum = np.sum(h)
                target_sum = n_points / 2
                h = h * target_sum / h_sum
            else:
                # If no solution, fallback to gradient descent with subgradient
                grad = np.zeros_like(h)
                if k_max >= 0:
                    for i in range(n_points - k_max):
                        grad[i] -= h[i + k_max]
                    for i in range(k_max, n_points):
                        grad[i] -= h[i - k_max]
                else:
                    for i in range(-k_max):
                        grad[i] -= h[i - k_max]
                    for i in range(n_points + k_max):
                        grad[i] -= h[i + k_max]
                grad = 2 * grad / n_points
                h = h - 0.01 * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        else:
            # Fall back if optimization fails
            grad = np.zeros_like(h)
            if k_max >= 0:
                for i in range(n_points - k_max):
                    grad[i] -= h[i + k_max]
                for i in range(k_max, n_points):
                    grad[i] -= h[i - k_max]
            else:
                for i in range(-k_max):
                    grad[i] -= h[i - k_max]
                for i in range(n_points + k_max):
                    grad[i] -= h[i + k_max]
            grad = 2 * grad / n_points
            h = h - 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
                
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
