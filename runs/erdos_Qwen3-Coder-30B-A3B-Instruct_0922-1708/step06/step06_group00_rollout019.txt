import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Define barrier optimization parameters
    max_iter = 500
    mu = 1.0
    mu_factor = 1.1
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap with current h
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Construct convex optimization problem
        h_var = cp.Variable(n_points)
        t_var = cp.Variable()
        
        # Objective: minimize t
        objective = cp.Minimize(t_var)
        
        # Constraints
        constraints = []
        
        # Constraint on sum
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Constraint on bounds
        for i in range(n_points):
            constraints.append(h_var[i] >= 0)
            constraints.append(h_var[i] <= 1)
        
        # Overlap constraints: t >= overlap_k for all k
        for k in range(2 * n_points - 1):
            # compute overlap for lag k
            overlap_k = 0.0
            if k < n_points:
                # h[i] * (1 - h[i+k]) where i+k < n_points
                for i in range(n_points - k):
                    overlap_k += h_var[i] * (1 - h_var[i + k])
            else:
                # For negative lags, we are looking at h[i+k] * (1 - h[i])
                for i in range(k - n_points + 1, n_points):
                    overlap_k += h_var[i + k - n_points + 1] * (1 - h_var[i])
            
            # Add constraint
            constraints.append(t_var >= overlap_k * dx)
        
        # Barrier term: add penalty for boundary approaches
        barrier_term = 0
        for i in range(n_points):
            barrier_term += -cp.log(h_var[i] + 1e-12) - cp.log(1 - h_var[i] + 1e-12)
        
        # Add barrier to objective
        objective = cp.Minimize(t_var + mu * barrier_term)
        
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
        
        try:
            # Try solving with different solvers based on availability
            prob.solve(solver=cp.ECOS, max_iters=1000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_var.value
                if np.all(h_new >= 0) and np.all(h_new <= 1):
                    h = h_new.copy()
                    # Normalize
                    h_sum = np.sum(h)
                    if abs(h_sum - n_points / 2) > 1e-6:
                        h = h * (n_points / 2) / h_sum
                else:
                    # Fallback if not valid
                    h = np.clip(h_new, 0, 1)
                    h_sum = np.sum(h)
                    h = h * (n_points / 2) / h_sum
            else:
                # Fallback if infeasible, continue with previous h
                pass
        except Exception:
            # If CVXPY throws error, fall back
            pass
            
        # Update mu
        mu *= mu_factor
        
        # Project h to [0,1] and rescale
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / h_sum
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
