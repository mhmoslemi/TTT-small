import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def interpolate_h(h_coarse, n_points):
    """Interpolate coarse h to new resolution."""
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_points)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine

def mutate_h(h, std):
    """Add Gaussian noise to h with given standard deviation."""
    noise = np.random.normal(0, std, len(h))
    return h + noise

def crossover_h(h1, h2, prob=0.5):
    """Crossover two h vectors."""
    mask = np.random.rand(len(h1)) < prob
    h_new = np.where(mask, h1, h2)
    return h_new

def evolution_strategy_run(h_coarse, n_points, generations, pop_size, mutation_std, selection_rate, crossover_rate):
    """Run evolutionary strategy optimization."""
    pop = [h_coarse.copy() for _ in range(pop_size)]
    best_h = None
    best_value = float('inf')
    
    for gen in range(generations):
        # Evaluate population
        fitness = []
        for h in pop:
            h_repaired = repair_h(h, n_points)
            value = evaluate_h(h_repaired, 2.0 / n_points)
            fitness.append(value)
            if value < best_value:
                best_value = value
                best_h = h_repaired.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        pop = [pop[i] for i in sorted_indices]
        
        # Selection
        num_selected = max(1, int(pop_size * selection_rate))
        selected = pop[:num_selected]
        
        # Generate new population
        new_pop = selected.copy()
        
        # Crossover
        while len(new_pop) < pop_size:
            if np.random.rand() < crossover_rate:
                i1, i2 = np.random.choice(len(selected), 2, replace=False)
                h_new = crossover_h(selected[i1], selected[i2])
            else:
                h_new = selected[np.random.choice(len(selected))].copy()
            
            # Mutation
            h_new = mutate_h(h_new, mutation_std)
            h_new = repair_h(h_new, n_points)
            new_pop.append(h_new)
        
        pop = new_pop[:pop_size]
    
    return best_h, best_value

    stages = [10, 20, 40, 100]
    best_h = None
    best_value = float('inf')
    current_resolution = 10

        if len(h_init) != current_resolution:

            x_target = np.linspace(0, 2, current_resolution)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (current_resolution / 2) / np.sum(h_init)
        coarse_h = h_init.copy()
    else:
        coarse_h = np.ones(current_resolution) * 0.5
    
    # Multi-resolution optimization
    for stage_idx, target_points in enumerate(stages):
        if time.time() - start_time > budget_s - 10:
            break
            
        if stage_idx > 0:
            # Interpolate from previous resolution
            coarse_h = interpolate_h(coarse_h, target_points)
            current_resolution = target_points
            
        # Adjust parameters for current resolution
        mutation_std = 0.1 / np.sqrt(target_points)
        pop_size = 50
        num_generations = 50
        selection_rate = 0.2
        crossover_rate = 0.3
        
        # Run evolution strategy
        new_h, new_value = evolution_strategy_run(
            coarse_h, target_points, num_generations, pop_size, 
            mutation_std, selection_rate, crossover_rate
        )
        
        if new_value < best_value:
            best_value = new_value
            best_h = new_h.copy()
            
        coarse_h = new_h.copy()
        
    # Local refinement with gradient descent
    if best_h is not None:
        h_final = best_h.copy()
        current_value = best_value
        dx = 2.0 / len(h_final)
        
        # Gradient descent
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            eps = 1e-6
            grad = np.zeros(len(h_final))
            for j in range(len(h_final)):
                h_plus = h_final.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, len(h_final))
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
                
            h_final = h_final - 0.001 * grad
            h_final = repair_h(h_final, len(h_final))
            corr = correlate(h_final, 1 - h_final, mode='full')
            current_value = np.max(corr) * dx
            
            if current_value < best_value:
                best_value = current_value
                best_h = h_final.copy()
    
    return (best_h.tolist(), best_value, len(best_h))
