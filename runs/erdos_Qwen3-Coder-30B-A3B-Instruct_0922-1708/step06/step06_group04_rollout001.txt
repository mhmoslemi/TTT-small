import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h based on existing construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Create CVXPY variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        t >= 0
    ]
    
    # Compute overlap constraints
    max_lag = n_points - 1
    for k in range(-max_lag, max_lag + 1):
        # Compute valid indices
        valid_indices = []
        for i in range(n_points):
            j = i - k
            if 0 <= j < n_points:
                valid_indices.append((i, j))
        
        if valid_indices:
            # Construct overlap constraint
            indices1 = [idx[0] for idx in valid_indices]
            indices2 = [idx[1] for idx in valid_indices]
            overlap = cp.sum(cp.multiply(h[indices1], 1 - h[indices2]))
            # We must divide by dx to match the discrete integral definition
            constraints.append(overlap <= t * (n_points / 2))
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_bound = t.value * dx
        else:
            # Fallback to initial solution
            h_opt = h_init
            c5_bound = evaluate_h(h_init, dx)
    except Exception:
        # Fallback to initial solution if solver fails
        h_opt = h_init
        c5_bound = evaluate_h(h_init, dx)
    
    # Final verification
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    final_c5 = evaluate_h(h_opt, dx)
    
    return (h_opt.tolist(), final_c5, n_points)
