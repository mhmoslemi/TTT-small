import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h = h_init.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Hierarchical refinement approach
    scales = [10, 20, 40, 100]
    best_h = None
    best_value = float('inf')
    
    # Initially optimize on coarse grid
    h_coarse = h.copy()
    for scale in scales:
        n_coarse = scale
        if scale == 100:
            # Keep full resolution
            h_old = h_coarse.copy()
        else:
            # Interpolate to coarser grid
            x_orig = np.linspace(0, 2, len(h_coarse))
            x_new = np.linspace(0, 2, n_coarse)
            h_old = np.interp(x_new, x_orig, h_coarse)
            # Ensure sum constraint
            h_old = h_old * (n_coarse / 2) / np.sum(h_old)
            
        # Optimization using L-BFGS
        def objective(x):
            h_temp = x.copy()
            h_temp = h_temp * (n_coarse / 2) / np.sum(h_temp)
            h_temp = repair_h(h_temp, n_coarse)
            return evaluate_h(h_temp, 2.0 / n_coarse)
            
        def constraint(x):
            return np.sum(x) - n_coarse / 2
            
        # Initial optimization
        bounds = [(0, 1) for _ in range(n_coarse)]
        cons = {'type': 'eq', 'fun': constraint}
        
        try:
            res = minimize(objective, h_old, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
            if res.success:
                h_temp = res.x.copy()
                h_temp = h_temp * (n_coarse / 2) / np.sum(h_temp)
                h_temp = repair_h(h_temp, n_coarse)
                value = evaluate_h(h_temp, 2.0 / n_coarse)
                if value < best_value:
                    best_value = value
                    if scale == 100:
                        best_h = h_temp.copy()
                    else:
                        # Interpolate back to full resolution
                        x_new = np.linspace(0, 2, n_points)
                        x_orig = np.linspace(0, 2, n_coarse)
                        temp_full = np.interp(x_new, x_orig, h_temp)
                        best_h = temp_full.copy()
        except:
            pass
            
        h_coarse = best_h.copy() if best_h is not None else h_coarse
        
        if time.time() - start_time > budget_s - 5:
            break
    
    # Final refinement with full resolution gradient descent
    if best_h is None:
        best_h = h.copy()
        
    h_full = best_h.copy()
    
    for iter_count in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient numerically at current maximum
        grad = np.zeros(n_points)
        eps = 1e-6
        current_value = evaluate_h(h_full, dx)
        
        for j in range(n_points):
            h_plus = h_full.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_value) / eps
            
        h_full = h_full - 0.0001 * grad
        h_full = repair_h(h_full, n_points)
        current_value = evaluate_h(h_full, dx)
        
        if current_value < best_value:
            best_value = current_value
            best_h = h_full.copy()
    
    return (best_h.tolist(), best_value, n_points)
