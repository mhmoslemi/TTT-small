import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50000
    tabu_size = 50
    
    # Initialize from parent or random binary
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Convert to binary representation with exactly n_points/2 ones
    h_binary = (h > 0.5).astype(float)
    current_sum = np.sum(h_binary)
    if current_sum != n_points / 2:
        # Adjust to get exactly n_points/2 ones
        diff = int(n_points/2 - current_sum)
        if diff > 0:
            # Set zeros to ones
            zero_indices = np.where(h_binary == 0)[0]
            if len(zero_indices) > 0:
                indices_to_flip = np.random.choice(zero_indices, min(diff, len(zero_indices)), replace=False)
                for idx in indices_to_flip:
                    h_binary[idx] = 1
        else:
            # Set ones to zeros
            one_indices = np.where(h_binary == 1)[0]
            if len(one_indices) > 0:
                indices_to_flip = np.random.choice(one_indices, min(-diff, len(one_indices)), replace=False)
                for idx in indices_to_flip:
                    h_binary[idx] = 0

    # Initialize tabu list
    tabu_list = []
    
    # Start with original binary solution
    best_h = h_binary.copy()
    corr = correlate(h_binary, 1 - h_binary, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Simulated annealing parameters
    temp = 0.1
    cooling_rate = 0.999
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping a random bit
        candidate_h = h_binary.copy()
        flip_idx = np.random.randint(0, n_points)
        candidate_h[flip_idx] = 1 - candidate_h[flip_idx]
        
        # Ensure we maintain exactly n_points/2 ones
        ones_count = np.sum(candidate_h)
        if ones_count != n_points / 2:
            # Revert the flip and try again more carefully
            if ones_count > n_points / 2:
                # Too many ones, flip a one to zero
                one_indices = np.where(candidate_h == 1)[0]
                flip_one = np.random.choice(one_indices)
                candidate_h[flip_one] = 0
            else:
                # Too few ones, flip a zero to one
                zero_indices = np.where(candidate_h == 0)[0]
                flip_zero = np.random.choice(zero_indices)
                candidate_h[flip_zero] = 1
        
        # Compute overlap
        corr = correlate(candidate_h, 1 - candidate_h, mode='full')
        c5 = np.max(corr) * dx
        
        # Accept or reject based on simulated annealing
        if c5 < best_c5:
            h_binary = candidate_h.copy()
            corr = correlate(h_binary, 1 - h_binary, mode='full')
            best_c5 = np.max(corr) * dx
            best_h = h_binary.copy()
        else:
            # Energy difference
            delta = c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                h_binary = candidate_h.copy()
        
        # Update temperature
        temp *= cooling_rate
        
        # Add current solution to tabu list
        tabu_list.append(h_binary.copy())
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
    
    # Final refinement with optimization
    # Use gradient descent on the continuous version
    h_continuous = best_h.astype(float)
    
    # Refine the continuous version using simple gradient ascent
    learning_rate = 0.01
    for _ in range(100):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        corr = correlate(h_continuous, 1 - h_continuous, mode='full')
        c5 = np.max(corr) * dx
        
        # Compute gradient as finite differences
        grad = np.zeros_like(h_continuous)
        for i in range(len(h_continuous)):
            h_plus = h_continuous.copy()
            h_plus[i] += 1e-4
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            h_plus = h_plus * (n_points / 2) / h_plus_sum
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / 1e-4
        
        h_continuous = h_continuous - learning_rate * grad
        h_continuous = np.clip(h_continuous, 0, 1)
        h_continuous_sum = np.sum(h_continuous)
        h_continuous = h_continuous * (n_points / 2) / h_continuous_sum
    
    # Final evaluation using discrete version
    corr = correlate(h_continuous, 1 - h_continuous, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Return best solution
    return (h_continuous.tolist(), final_c5, n_points)
