import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_to_feasible(h_init, n_points)
    else:
        # Initialize with structured pattern
        x = np.linspace(0, 2, n_points)
        h_init = np.maximum(0, 1 - np.abs(x - 1))
        h_init = project_to_feasible(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    patience = 0
    max_patience = 1000
    
    # Frank-Wolfe parameters
    max_iter = 10000
    epsilon = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += epsilon
            h_plus = project_to_feasible(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_current) / epsilon
            
        # Solve linear subproblem: min_d <grad, d> s.t. d in feasible set
        # The feasible set is {d | 0 <= d <= 1, sum(d) = 0}
        # This is a constrained optimization problem that can be solved
        # by setting d to be the negative gradient with appropriate scaling
        # to satisfy sum(d) = 0
        
        # Direct approach:
        # We want to minimize <grad, d> subject to sum(d) = 0 and 0 <= d_i <= 1
        # Set components of d corresponding to negative grad to 0
        # Set components of d corresponding to positive grad to 1
        # Adjust to satisfy sum(d) = 0
        d = np.zeros(n_points)
        indices_neg = np.where(grad < 0)[0]
        indices_pos = np.where(grad > 0)[0]
        
        # Set d to values that try to minimize the inner product
        d[indices_neg] = 0.0
        d[indices_pos] = 1.0
        
        # Adjust d to satisfy sum = 0
        current_sum = np.sum(d)
        if abs(current_sum) > 1e-10:
            delta = -current_sum / n_points
            d = d + delta
        
        # Project d to be feasible:
        # Ensure 0 <= d_i <= 1
        d = np.clip(d, 0, 1)
        
        # Adjust again to meet sum = 0 constraint
        current_sum = np.sum(d)
        if abs(current_sum) > 1e-10:
            delta = -current_sum / n_points
            d = d + delta
            d = np.clip(d, 0, 1)
        
        # Line search to find good step size
        eta_max = 1.0
        eta = eta_max
        improved = False
        
        while eta > 1e-8:
            h_new = h_current + eta * d
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new <= c5_current:
                h_current = h_new
                c5_current = c5_new
                improved = True
                break
            eta /= 2.0
            
        if not improved:
            # If not improved, try a random direction
            h_new = h_current.copy()
            # Perturb by a small amount in a random direction and project
            noise = np.random.normal(0, 0.01, n_points)
            h_new += noise
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
        
        # Update the best if improved
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            # Reset if stuck
            patience = 0
            h_current = h_init.copy()
            c5_current = evaluate_h(h_current, dx)
            h_best = h_current.copy()
            c5_best = c5_current
    
    # Final evaluation
    h_final = project_to_feasible(h_best, n_points)
    c5_final = evaluate_h(h_final, dx)
    
    return (h_final.tolist(), c5_final, n_points)
