import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimization parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    max_iter = 100000
    progress_count = 0
    max_progress = 200
    adaptive_lr_factor = 0.99
    adaptive_perturb_factor = 0.999
    
    # Main loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        corr_vals = correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr_vals)
        max_overlap = corr_vals[max_idx]
        
        # Compute gradient for maximum overlap
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        # Apply gradient descent update
        update = -learning_rate * grad
        
        # Add structured perturbation to escape local optima
        if i % 100 == 0 and i > 0:
            # Identify indices with low gradient values (high contribution to maximum)
            abs_grad = np.abs(grad)
            low_grad_indices = np.where(abs_grad < 0.1 * np.max(abs_grad))[0]
            if len(low_grad_indices) > 0:
                num_perturb = min(10, len(low_grad_indices))
                indices_to_perturb = np.random.choice(
                    low_grad_indices, num_perturb, replace=False
                )
                for idx in indices_to_perturb:
                    delta = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
                    update[idx] += delta
        
        new_h = current_h + update
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept the new point if it's better or with some probability
        if new_c5 < current_c5 or np.random.rand() < 0.01:
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                progress_count = 0
            else:
                progress_count += 1
        else:
            progress_count += 1
            
        # Adaptive adjustments
        if progress_count > max_progress:
            learning_rate *= adaptive_lr_factor
            perturbation_magnitude *= adaptive_perturb_factor
            progress_count = 0
        
        # Ensure minimum learning rate and perturbation
        learning_rate = max(1e-8, learning_rate)
        perturbation_magnitude = max(1e-6, perturbation_magnitude)
    
    # Final refinement
    h_refined = best_h.copy()
    best_c5_refined = best_c5
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        
        # Forward pass to compute gradient
        corr_vals = correlate(h_refined, 1 - h_refined, mode='full')

        h_refined = h_refined - 0.0001 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_c5_refined = np.max(corr) * dx
        
        if best_c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = best_c5_refined
    
    # Final validation
    final_h = repair_h(best_h, n_points)
    final_c5 = evaluate_h(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
