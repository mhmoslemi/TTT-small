import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h = h_init.copy()
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Heuristic: use a hierarchical approach with different grid sizes
    grid_sizes = [10, 20, 40, 100]
    
    for i, num_intervals in enumerate(grid_sizes):
        # Create a coarse representation
        if num_intervals == 100:
            h_coarse = h.copy()
        else:
            coarse_x = np.linspace(0, 2, num_intervals)
            fine_x = np.linspace(0, 2, n_points)
            h_coarse = np.interp(fine_x, coarse_x, h)
        
        # Gradient descent on the coarse h
        def objective(h_coarse_flat):
            h_full = h_coarse_flat.copy()
            h_full = repair_h(h_full, num_intervals)
            return evaluate_h(h_full, dx)
        
        # Optimize with constraints
        bounds = [(0, 1) for _ in range(num_intervals)]
        x0 = h_coarse[:num_intervals]  # Take first num_intervals values
        # Use L-BFGS with bounds
        try:
            res = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 500})
            if res.success:
                h_coarse = res.x.copy()
            h_coarse = repair_h(h_coarse, num_intervals)
            
            # Refine to next grid size
            if num_intervals < 100:
                fine_x = np.linspace(0, 2, n_points)
                coarse_x = np.linspace(0, 2, num_intervals)
                h = np.interp(fine_x, coarse_x, h_coarse)
                h = repair_h(h, n_points)
            else:
                h = h_coarse.copy()
        except Exception:
            pass
        
        current_value = evaluate_h(h, dx)
        if current_value < best_value:
            best_value = current_value
            best_h = h.copy()
        
        if time.time() - start_time > budget_s - 10:
            break
    
    # Intensive local optimization on the final h
    def grad_eval(h_flat):
        h_full = h_flat.copy()
        h_full = repair_h(h_full, n_points)
        corr = correlate(h_full, 1 - h_full, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Compute gradient at the maximum correlation point
        grad = np.zeros(n_points)
        eps = 1e-6
        for j in range(n_points):
            h_plus = h_full.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[j] = (corr_plus[max_idx] - max_val) / eps
        return grad
    
    # Gradient descent refinement
    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = grad_eval(best_h)
        best_h = best_h - 0.001 * grad
        best_h = repair_h(best_h, n_points)
        current_value = evaluate_h(best_h, dx)
        if current_value < best_value:
            best_value = current_value
    
    return (best_h.tolist(), best_value, n_points)
