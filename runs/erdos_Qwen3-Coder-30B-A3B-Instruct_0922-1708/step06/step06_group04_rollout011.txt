import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, target_sum=50):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = project_h(h, target_sum=n_points/2)
    
    # Regularization parameters
    lambda_reg = 0.01
    alpha = 0.1
    max_iter = 1000
    patience = 100
    best_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient computation
        eps = 1e-6
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, target_sum=n_points/2)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
        
        # L1 regularization gradient
        l1_grad = lambda_reg * np.sign(h)
        
        # Update
        h_new = h - alpha * (grad + l1_grad)
        h_new = project_h(h_new, target_sum=n_points/2)
        
        # Backtracking line search
        c5_new = evaluate_h(h_new, dx)
        if c5_new < current_c5:
            h = h_new
            alpha *= 0.9  # Reduce learning rate on improvement
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            alpha *= 1.1  # Increase learning rate on no improvement
            no_improvement_count += 1
        
        # Adjust learning rate bounds
        alpha = np.clip(alpha, 1e-4, 1.0)
        
        # Early stopping
        if no_improvement_count >= patience:
            # Try increased regularization or restart
            lambda_reg *= 1.1
            h = project_h(best_h, target_sum=n_points/2)
            no_improvement_count = 0
            alpha = 0.1
    
    return (best_h.tolist(), best_c5, n_points)
