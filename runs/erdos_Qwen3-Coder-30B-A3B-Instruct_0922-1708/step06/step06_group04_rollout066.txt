import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Start with a symmetric construction, perturbed
        h = np.full(n_points, 0.5)
        # Add small Gaussian noise to break symmetry
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = repair_h(h, n_points)

    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Gradient-based optimization with adaptive learning rate
    max_iter = 2000
    learning_rate = 0.0005
    patience = 50
    
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute finite difference gradient
        grad = np.zeros(n_points)
        eps = 1e-6
        current_c5 = evaluate_h(h, dx)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        # Update parameters
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        new_c5 = evaluate_h(h, dx)
        
        if new_c5 < best_value:
            best_h = h.copy()
            best_value = new_c5
            last_improvement = iter_count
            if new_c5 < 0.3807:
                break
        
        # Adaptive learning rate
        if iter_count > 0 and iter_count % 100 == 0:
            learning_rate *= 0.95
        
        # Inject diversity if no improvement for a while
        if iter_count - last_improvement > patience:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iter_count
        
        iter_count += 1
    
    # Final gradient refinement of best solution
    h_final = best_h.copy()
    prev_value = best_value
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        grad = np.zeros(n_points)
        eps = 1e-6
        current_c5 = evaluate_h(h_final, dx)
        
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        h_final = h_final - 0.0001 * grad
        h_final = repair_h(h_final, n_points)
        new_c5 = evaluate_h(h_final, dx)
        
        if new_c5 < prev_value:
            prev_value = new_c5
            best_h = h_final.copy()
    
    return (best_h.tolist(), prev_value, n_points)
