import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Design QP problem formulation
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        t >= 0
    ]
    
    # Compute overlap constraints for all valid lags
    # For lag j in [-floor((n_points-1)/2), floor((n_points-1)/2)]
    max_lag = (n_points - 1) // 2
    for j in range(-max_lag, max_lag + 1):
        # Build overlap expression
        expr = 0
        for i in range(n_points):
            idx = (i - j) % n_points
            expr += h[i] * (1 - h[idx])
        # Add constraint: overlap <= t * (n_points / 2) 
        # (scaled to match discrete formulation)
        constraints.append(expr <= t * (n_points / 2))
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_API_ERROR]:
            h_opt = h.value
            c5_bound = t.value * dx
        else:
            # Fallback to initial if not optimal
            h_opt = h_init
            c5_bound = evaluate_h(h_init, dx)
    except:
        # Fallback to initial if CVXPY fails
        h_opt = h_init
        c5_bound = evaluate_h(h_init, dx)
    
    # Ensure bound is valid
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    final_c5 = evaluate_h(h_opt, dx)
    
    return (h_opt.tolist(), final_c5, n_points)
