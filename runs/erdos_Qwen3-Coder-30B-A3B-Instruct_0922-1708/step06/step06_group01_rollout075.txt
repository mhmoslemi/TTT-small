import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints using quadratic programming."""
    # Project to [0,1]^n and then to the hyperplane sum(h) = n/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
    
    # Store best solution
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Gradient-based optimization with adaptive learning rate
    step_size = 0.01
    patience = 0
    max_patience = 1000
    last_improvement = 0
    
    # Adaptive perturbation rate
    perturb_rate = 0.05

    c5_current = c5_best

        if time.time() - start_time > budget_s - 5:
            break
            
        # Create neighbor by perturbing some elements with adaptive rate
        num_perturb = max(1, int(perturb_rate * n_points))
        h_new = h_current.copy()
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            delta = np.random.uniform(-0.05, 0.05)
            h_new[idx] += delta

        h_new = repair_h(h_new, n_points)

        # Adaptive cooling
        if i - last_improvement > max_patience:
            patience += 1
            if patience > 5:
                step_size *= 0.95
                perturb_rate *= 1.1
                patience = 0
                if step_size < 1e-6:
                    break
            last_improvement = i
    
    # Final polishing with gradient-based refinement
    h_final = h_best.copy()
