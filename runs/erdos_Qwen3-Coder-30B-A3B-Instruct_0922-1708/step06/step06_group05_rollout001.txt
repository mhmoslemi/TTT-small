import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = 200
        x_orig = np.linspace(0, 2, len(h0))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h0)
    else:
        h = np.random.rand(200)
    
    # Ensure integral constraint
    h = h * (200 / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_h(h, 2.0 / 200)
    
    # Try CVXPY approach
    def solve_relaxed_convex(h_init, n_points):
        h = cp.Variable(n_points)
        # Objective: minimize max of convolutions (quadratic approx)
        # For simplicity, use squared sum of all correlations
        objective = cp.Minimize(cp.sum_squares(cp.convolution(h, 1 - h, 'full')))
        # Constraints
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.CVXOPT, eps=1e-6, verbose=False)
            if problem.status == cp.OPTIMAL:
                return h.value
        except:
            pass
        return h_init
    
    # Primary refined approach
    lr = 0.001
    n_points = 200
    dx = 2.0 / n_points
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient descent with adaptive learning
        eps = 1e-6
        grad = np.zeros(n_points)
        h_eval = evaluate_h(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - h_eval) / eps
        
        # Update with adaptive learning rate
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        # Evaluate new
        c5 = evaluate_h(h, dx)
        
        # Track best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Adaptive decrement of learning rate
        if iteration > 0 and iteration % 100 == 0:
            lr *= 0.95
            if lr < 1e-6:
                lr = 1e-6
                
        # Occasionally solve relaxed convex problem
        if iteration % 500 == 0:
            try:
                h_cvx = solve_relaxed_convex(h, n_points)
                h_cvx = repair_h(h_cvx, n_points)
                c5_cvx = evaluate_h(h_cvx, dx)
                if c5_cvx < best_c5:
                    best_c5 = c5_cvx
                    best_h = h_cvx.copy()
            except:
                pass
        
    # Final refinement
    for i in range(5000):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - c5_eval) / eps
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
