import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum=50):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from existing solution or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = project_h(h, target_sum=n_points/2)
        # Add small perturbation to escape local minima
        h += 0.1 * np.random.randn(n_points)
        h = project_h(h, target_sum=n_points/2)
    else:
        h = np.ones(n_points) * 0.5
        h = project_h(h, target_sum=n_points/2)
        
    # Regularization parameter
    lambda_reg = 0.01
    # Learning rate
    alpha = 0.1
    # Minimum learning rate
    alpha_min = 1e-4
    # Maximum learning rate
    alpha_max = 1.0
    # Convergence tolerance
    tol = 1e-4
    # Iteration counter
    iter_count = 0
    # Best solution
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    # Previous function value
    prev_c5 = best_c5
    # Iterations without improvement
    stagnation = 0
    # Max stagnation before restarting
    max_stagnation = 100
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 10 and iter_count < 10000:
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, target_sum=n_points/2)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - prev_c5) / eps
            
        # Compute L1 regularization gradient
        reg_grad = lambda_reg * np.sign(h)
        
        # Update step
        h_new = h - alpha * (grad + reg_grad)
        
        # Project to feasible region
        h_new = project_h(h_new, target_sum=n_points/2)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept the new solution if it improves the objective
        if c5_new < prev_c5:
            h = h_new.copy()
            prev_c5 = c5_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
            # Reduce learning rate
            alpha = max(alpha_min, alpha * 0.95)
            stagnation = 0
        else:
            # Increase learning rate on failure
            alpha = min(alpha_max, alpha * 1.05)
            stagnation += 1
            
        # Check for stagnation
        if stagnation >= max_stagnation:
            # Perturb solution and reset
            h += 0.01 * np.random.randn(n_points)
            h = project_h(h, target_sum=n_points/2)
            stagnation = 0
            alpha = 0.1
            
        iter_count += 1
        
        # Convergence check
        if abs(prev_c5 - best_c5) < tol:
            break
    
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
