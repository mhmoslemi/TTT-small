import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and 0 <= h[i] <= 1."""
    # First project to box constraints [0,1]
    h = np.clip(h, 0, 1)
    
    # Then project to sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Use quadratic programming: min ||x - h||^2 s.t. sum(x) = target_sum, 0 <= x[i] <= 1
    # This is a simple projected gradient step
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values. Allocate proportionally to current values.
        # But ensure we don't exceed 1.
        increase = diff / n_points
        h = np.minimum(h + increase, 1.0)
        # Recalculate and adjust
        current_sum = np.sum(h)
        if current_sum < target_sum:
            # We're at the boundary, adjust remaining with available space
            available_space = n_points - current_sum
            if available_space > 0:
                increase = available_space / n_points
                h = np.minimum(h + increase, 1.0)
    else:
        # Need to decrease values. Distribute excess
        decrease = -diff / n_points
        h = np.maximum(h - decrease, 0.0)
        # Recalculate and adjust
        current_sum = np.sum(h)
        if current_sum > target_sum:
            # We're at the lower boundary, adjust remaining with available space
            available_space = current_sum - target_sum
            if available_space > 0:
                decrease = available_space / n_points
                h = np.maximum(h - decrease, 0.0)
    
    # Final correction to exactly match        
    current_sum = np.sum(h)
    h = h + (target_sum - current_sum) / n_points
    h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Approximate gradient of C5 w.r.t. h using finite differences."""
    n = len(h)
    grad = np.zeros(n)
    base_c5 = evaluate_h(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

    c5_current = evaluate_h(h_current, dx)
