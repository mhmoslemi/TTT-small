import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy constraints using quadratic programming."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Use projection to maintain feasibility
        # h_new = h - (sum(h) - target_sum) / n_points * ones
        h = h + (target_sum - current_sum) / n_points
        h = np.clip(h, 0, 1)
    return h
