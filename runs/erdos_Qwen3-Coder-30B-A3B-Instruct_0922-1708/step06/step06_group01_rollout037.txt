import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1 - np.abs(x - 1)
        h_init = project_h(h_init, n_points, dx)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Parameters
    eta = 0.01
    epsilon = 1e-6
    plateau_window = 100
    patience = 50
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        if iteration % plateau_window == 0:
            # Perform periodic perturbation to escape local minima
            idx = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            for i in idx:
                h_current[i] += np.random.uniform(-0.05, 0.05)
            h_current = project_h(h_current, n_points, dx)
            c5_current = evaluate_h(h_current, dx)
            
        # Coordinate-wise updates
        for i in range(n_points):
            # Compute gradient using finite differences
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            grad_i = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update
            h_current[i] = h_current[i] - eta * grad_i
            h_current[i] = max(0, min(1, h_current[i]))
            
        # Project to feasible set
        h_current = project_h(h_current, n_points, dx)
        
        # Evaluate
        c5_current = evaluate_h(h_current, dx)
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = iteration
            
        # Adaptive step size
        if iteration - last_improvement > patience:
            eta *= 0.95
            if eta < 1e-6:
                break
                
        iteration += 1
        
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_best.copy()
            h_plus[i] += epsilon
            h_plus = project_h(h_plus, n_points, dx)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_best) / epsilon
            
        # Update
        h_best = h_best - 0.001 * grad
        h_best = project_h(h_best, n_points, dx)
        c5_best = evaluate_h(h_best, dx)
        
    return (h_best.tolist(), c5_best, n_points)
