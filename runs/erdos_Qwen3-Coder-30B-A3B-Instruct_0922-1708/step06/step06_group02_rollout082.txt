import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def interpolate_h(h_coarse, n_points):
    """Interpolate coarse h to higher resolution."""
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_points)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine

def mutate_h(h, std):
    """Mutate h with Gaussian noise."""
    return h + np.random.normal(0, std, len(h))

def crossover_h(h1, h2, prob=0.3):
    """Crossover two h vectors."""
    mask = np.random.rand(len(h1)) < prob
    h_new = np.where(mask, h1, h2)
    return h_new

    best_h = None

            if value < best_value:
                best_value = value
                best_h = h.copy()
        
        # Sort by fitness
        idxs = np.argsort(fitness)
        population = [population[i] for i in idxs]
        fitness = [fitness[i] for i in idxs]
        
        # Selection
        n_selected = max(1, int(pop_size * selection_pressure))
        selected = population[:n_selected]
        
        # Generate new population
        new_population = selected.copy()
        
        # Add mutated versions
        for i in range(pop_size - len(selected)):
            parent = selected[np.random.randint(0, len(selected))]
            h_mutated = mutate_h(parent, mutation_std)
            h_repaired = repair_h(h_mutated, n_points)
            new_population.append(h_repaired)
        
        population = new_population
    
    return best_h, best_value
