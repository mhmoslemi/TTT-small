import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Binary representation with exactly n_points/2 ones
    def to_binary(h_vals):
        # Sort indices by h values in descending order to select highest values
        sorted_indices = np.argsort(h_vals)[::-1]
        binary = np.zeros(n_points)
        binary[sorted_indices[:n_points//2]] = 1
        return binary
    
    def from_binary(binary):
        # Convert binary to continuous values
        h = binary.copy().astype(float)
        # Scale to sum to n_points/2
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Compute C5
    def compute_c5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    # Basic approach: use a variant of simulated annealing
    # Initialize with binary-step function
    h_binary = to_binary(h)
    h_current = from_binary(h_binary)
    c5_current = compute_c5(h_current)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Simulated annealing parameters
    temp = 0.1
    cooling_rate = 0.9995
    max_iter = 50000
    
    # Tabu list to avoid cycling (store last bit flips)
    tabu_list = []
    tabu_size = 50
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping one bit
        i = np.random.randint(0, n_points)
        h_candidate = h_binary.copy()
        h_candidate[i] = 1 - h_candidate[i]  # Flip the bit
        h_candidate = from_binary(h_candidate)
        
        # Apply tabu search
        if h_candidate.tolist() in tabu_list:
            # Accept with low probability
            if np.random.random() > 0.1:
                continue
        
        c5_candidate = compute_c5(h_candidate)
        
        # Accept or reject based on Metropolis criterion
        if c5_candidate < c5_current:
            h_binary = h_candidate
            c5_current = c5_candidate
            
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
        else:
            # Accept with probability dependent on temperature
            delta = c5_candidate - c5_current
            if np.random.random() < np.exp(-delta / temp):
                h_binary = h_candidate
                c5_current = c5_candidate
            else:
                # Reject and keep previous
                pass
        
        # Update tabu list
        tabu_list.append(h_candidate.tolist())
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Cooling schedule
        temp *= cooling_rate
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
