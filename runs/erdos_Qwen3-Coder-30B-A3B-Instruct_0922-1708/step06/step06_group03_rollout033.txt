import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent with projected gradient descent
    lambda_k = np.zeros(2 * n_points - 1)
    learning_rate = 5e-4
    patience = 0
    max_patience = 100
    
    for iteration in range(100000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap correlations
        corr = correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Update lambda_k using gradient ascent
        lambda_k = np.maximum(0, lambda_k + learning_rate * (corr - max_val))
        
        # Project lambda_k to reasonable bounds
        lambda_k = np.minimum(1000, np.maximum(0, lambda_k))
        
        # Compute gradient of Lagrangian w.r.t h
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(2 * n_points - 1):
                if k < n_points and i >= k:
                    grad[i] += lambda_k[k] * (1 - current_h[i - k])
        
        # Perform gradient step
        current_h = current_h - learning_rate * grad
        current_h = repair_h(current_h, n_points)
        
        # Update learning rate
        if iteration % 500 == 0 and iteration > 0:
            learning_rate *= 0.99
            
        # Evaluate the new solution
        current_c5 = evaluate_h(current_h, dx)
        
        # Check for improvement
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
