import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create structured h using sieve pattern in frequency domain
    freqs = np.fft.fftfreq(n_points, 1.0/n_points)
    # High frequency components suppressed to minimize overlaps
    amplitudes = np.exp(-freqs**2 / (2*10**2))  # Gaussian envelope
    amplitudes[0] = 0  # DC component
    # Create complex spectrum with random phases
    phases = np.random.uniform(0, 2*np.pi, n_points)
    spectrum = amplitudes * np.exp(1j * phases)
    # Make spectrum real for real h
    spectrum = np.concatenate([spectrum[:n_points//2], 
                               spectrum[n_points//2:][::-1].conj()])
    # Inverse fft
    h = np.fft.ifft(spectrum).real
    # Ensure positive part and correct sum
    h = np.abs(h)
    h = project_h(h, n_points)
    return h

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        h_init = fourier_initialization(n_points)
    
    # Tabu search parameters
    population_size = 100
    tabu_list_size = 50
    tabu_list = []
    memory_decay = 0.95
    
    # Initialize population
    population = [h_init.copy()]
    fitness = [evaluate_overlap(h_init, dx)]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        population.append(h)
        fitness.append(evaluate_overlap(h, dx))
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    # Extra iterations to spend time efficiently
    max_iter = 2000
    iter_count = 0
    stagnation_counter = 0
    perturbation_size = 0.2
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 20:
        if time.time() - start_time > budget_s - 20:
            break
            
        # Select a random individual for modification
        parent_idx = np.random.randint(len(population))
        parent_h = population[parent_idx].copy()
        
        # Perturb with Gaussian noise
        perturb = np.random.randn(n_points) * perturbation_size
        h_new = parent_h + perturb
        h_new = project_h(h_new, n_points)
        
        # Check tabu
        if len(tabu_list) >= tabu_list_size:
            tabu_list = tabu_list[-tabu_list_size:]
            
        # Get hash of the perturbation
        pert_hash = hash(tuple(np.round(h_new - parent_h, decimals=6)))
        if pert_hash in tabu_list:
            h_new = parent_h.copy()  # Skip this move
            continue
            
        new_c5 = evaluate_overlap(h_new, dx)
        # Update tabu list
        tabu_list.append(pert_hash)
        if len(tabu_list) > tabu_list_size:
            tabu_list.pop(0)
            
        # Keep in population
        population.append(h_new)
        fitness.append(new_c5)
        
        # Keep top population_size
        indices = np.argsort(fitness)[:population_size]
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        
        # Track best
        if new_c5 < best_value:
            best_h = h_new.copy()
            best_value = new_c5
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Inject diversity periodically
        if stagnation_counter > 50:
            for _ in range(10):
                h = np.random.rand(n_points)
                h = project_h(h, n_points)
                population.append(h)
                fitness.append(evaluate_overlap(h, dx))
            # Keep best
            indices = np.argsort(fitness)[:population_size]
            population = [population[i] for i in indices]
            fitness = [fitness[i] for i in indices]
            stagnation_counter = 0
            
        iter_count += 1
    
    # Final gradient refinement on best solution
    h_refined = best_h.copy()
    prev_value = best_value
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        h_refined = h_refined - 0.0001 * grad
        h_refined = project_h(h_refined, n_points)
        prev_value = evaluate_overlap(h_refined, dx)
        
        if prev_value < best_value:
            best_h = h_refined.copy()
            best_value = prev_value
    
    return (best_h.tolist(), best_value, n_points)
