import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Initialize dual variables
    lambda_dual = np.ones(2 * n_points - 1)  # for all lags
    lambda_dual /= np.sum(lambda_dual)  # normalize
    tau = 0.1  # temperature for dual update
    
    # Hyperparameters
    learning_rate = 5e-4
    decay_rate = 0.99
    dual_update_freq = 10
    max_stagnation = 1000
    stagnation_count = 0
    
    for iteration in range(100000):  # Large iteration count
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all overlaps
        corr = correlate(current_h, 1 - current_h, mode='full')
        overlaps = corr * dx
        
        # Update dual variables using variational inference
        if iteration % dual_update_freq == 0:
            lambda_dual = np.exp(-overlaps / tau)
            lambda_dual /= np.sum(lambda_dual)
            tau *= 0.999
            
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(-n_points + 1, n_points):
                if 0 <= i - k < n_points:
                    grad[i] += lambda_dual[k + n_points - 1] * (1 - current_h[i - k])
        
        # Update h
        current_h = current_h - learning_rate * grad
        current_h = repair_h(current_h, n_points)
        
        # Evaluate new objective
        current_c5 = evaluate_h(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive learning rate
        if stagnation_count > 500:
            learning_rate *= decay_rate
            stagnation_count = 0
            
        # Fallback to initial solution if stuck
        if stagnation_count > max_stagnation:
            current_h = h_init.copy()
            current_c5 = evaluate_h(current_h, dx)
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
