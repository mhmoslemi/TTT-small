import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, max_k):
    """Compute gradient of the maximum overlap with respect to h."""
    n_points = len(h)
    grad = np.zeros(n_points)
    
    # Compute the correlation and find the maximum
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    
    # For each position, compute the derivative
    for i in range(n_points):
        # Approximate gradient by finite difference
        h_plus = h.copy()
        h_plus[i] += 1e-6
        h_plus = repair_h(h_plus, n_points)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_idx] - max_val) / 1e-6
    
    return grad

    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)

        # Compute gradient
        grad = compute_gradient(current_h, dx, n_points - 1)
        
        # Take gradient step
        new_h = current_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)

        if new_c5 < current_c5:

                best_h = current_h.copy()
                best_c5 = current_c5
                progress_counter = 0
            else:
                progress_counter += 1
        else:
            progress_counter += 1
        
        # Adjust learning rate if no progress
        if progress_counter > max_progress:
            learning_rate *= decay_factor
            progress_counter = 0
            
        # Reduce learning rate periodically
        if i % decay_every == 0 and i > 0:
            learning_rate *= decay_factor
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:

        refined_c5 = evaluate_h(refined_h, dx)
        
        if refined_c5 < best_c5:
