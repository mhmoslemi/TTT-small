import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent approach
    # Initialize dual variables for each lag k
    lambda_k = np.ones(2 * n_points - 1)  # For k in [-n+1, n-1]
    lambda_k = lambda_k / np.sum(lambda_k)  # Normalize
    learning_rate = 5e-4
    temp = 0.1
    decay_rate = 0.999
    dual_freq = 10
    stagnation_count = 0
    max_stagnation = 1000
    
    for iter_num in range(30000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all overlaps in one convolution
        overlaps = correlate(current_h, 1 - current_h, mode='full')
        overlaps = overlaps * dx  # Scale by dx
        
        # Update dual variables using softmax
        if iter_num % dual_freq == 0:
            # Variational update: exponential of negative overlaps, with temperature
            exp_overlaps = np.exp(-overlaps / temp)
            lambda_k = exp_overlaps / np.sum(exp_overlaps)
            
            # Cool down temperature
            temp = max(temp * decay_rate, 1e-6)
        
        # Compute gradient w.r.t h
        grad_h = np.zeros(n_points)
        for i in range(n_points):
            for k in range(-n_points + 1, n_points):
                if 0 <= i - k < n_points:
                    grad_h[i] += lambda_k[k + n_points - 1] * (1 - current_h[i - k])
        
        # Apply gradient step
        current_h = current_h - learning_rate * grad_h
        
        # Repair to maintain constraints
        current_h = repair_h(current_h, n_points)
        
        # Evaluate new solution
        current_c5 = evaluate_h(current_h, dx)
        
        # Store the best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive learning rate
        if stagnation_count > 200:
            learning_rate *= 0.5
            stagnation_count = 0
            
        if learning_rate < 1e-8:
            break
    
    return (best_h.tolist(), best_c5, n_points)
