import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Barrier method parameters
    mu = 1.0
    max_iter = 1000
    min_improvement = 1e-6
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation to check current state
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # CVXPY optimization
        # Variables
        h_vars = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2, t >= 0]
        
        # Add overlap constraints
        max_lag = len(h) - 1
        for k in range(-max_lag, max_lag + 1):
            if k >= 0:
                overlap = cp.sum(cp.multiply(h_vars[:len(h)-k], 1 - h_vars[k:]))
            else:
                overlap = cp.sum(cp.multiply(h_vars[-k:], 1 - h_vars[:len(h)+k]))
            constraints.append(t >= overlap * dx)
        
        # Objective
        objective = cp.Minimize(t - mu * cp.sum(cp.log(h_vars) + cp.log(1 - h_vars)))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h = np.array(h_vars.value).reshape(-1)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to gradient descent if optimization fails
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback to gradient descent
            h = h * (n_points / 2) / np.sum(h)
            
        # Update barrier parameter
        mu *= 1.1
        
        # Safety check
        if np.isnan(h).any() or np.isinf(h).any():
            h = best_h.copy()
            mu = 1.0
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
