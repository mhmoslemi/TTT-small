import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of C5 with respect to h."""
    n_points = len(h)
    grad = np.zeros(n_points)
    
    # The gradient computation: for each index j,
    # grad[j] = dx * sum_k (indicator(k == k_max) * (1 - h[(j - k) mod n_points]))
    for j in range(n_points):
        # For the specific k_max, add contribution from lag k_max
        # This is the contribution to overlap when shifting by k_max
        for k in range(n_points):
            if k == k_max:
                # Get index in cyclic fashion
                idx = (j - k) % n_points
                grad[j] += (1 - h[idx])
    
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Parameters for gradient descent
    learning_rate = 1e-4
    decay_rate = 0.99
    patience = 100
    patience_counter = 0
    
    # Adaptive perturbation strategy
    perturbation_magnitude = 0.05
    perturbation_decay = 0.999
    perturbation_interval = 500
    perturbation_counter = 0
    
    for i in range(1000000):  # Large number, but bounded by time
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient
        grad = compute_gradient(current_h, dx, k_max)
        
        # Update with gradient descent
        new_h = current_h - learning_rate * grad
        
        # Apply perturbation periodically
        perturbation_counter += 1
        if perturbation_counter >= perturbation_interval:
            # Add small random perturbation
            noise = np.random.uniform(-perturbation_magnitude, perturbation_magnitude, n_points)
            new_h = new_h + noise
            perturbation_counter = 0
            perturbation_magnitude *= perturbation_decay
            
        # Repair and evaluate
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            patience_counter = 0
            # Reduce learning rate periodically
            learning_rate *= decay_rate
            
        # Early termination if no progress
        if patience_counter > patience:
            break
    
    # Final refinement with lower learning rate
    last_h = best_h.copy()
    last_c5 = best_c5
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap and gradient with new learning rate
        corr = correlate(last_h, 1 - last_h, mode='full')
        k_max = np.argmax(corr)
        
        grad = compute_gradient(last_h, dx, k_max)
        
        # Small learning rate for final refinement
        new_h = last_h - 1e-5 * grad
        
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        if new_c5 < last_c5:
            last_h = new_h
            last_c5 = new_c5
            
    return (last_h.tolist(), last_c5, n_points)
