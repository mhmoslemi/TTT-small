import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.full(n_points, 0.5)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.0001
    max_iter = 10000
    patience = 100
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient
        grad = np.zeros(n_points)
        eps = 1e-6
        current_c5 = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        
        # Check improvement
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_value:
            best_h = h.copy()
            best_value = new_c5
            last_improvement = iteration
        
        # Early stopping if no improvement recently
        if iteration - last_improvement > patience:
            break
    
    # Final refinement
    h_refined = best_h.copy()
    prev_value = best_value
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:

        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        h_refined = h_refined - 0.00005 * grad
        h_refined = repair_h(h_refined, n_points)
        prev_value = evaluate_h(h_refined, dx)
        
        if prev_value < best_value:
            best_h = h_refined.copy()
            best_value = prev_value
    
    return (best_h.tolist(), best_value, n_points)
