import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, target_sum=50):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.full(n_points, 0.5)
    
    # Perturb and project
    h += 0.1 * np.random.randn(n_points)
    h = project_h(h)
    
    # Parameters
    lambda_reg = 0.01
    alpha = 0.1
    max_iter = 1000
    min_improvement = 1e-4
    stagnation_limit = 100
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    last_improvement = 0
    prev_value = best_value
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient via finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - prev_value) / eps
        
        # L1 regularization gradient
        l1_grad = lambda_reg * np.sign(h)
        
        # Update with regularization
        h_new = h - alpha * (grad + l1_grad)
        h_new = project_h(h_new)
        
        # Evaluate new value
        new_value = evaluate_h(h_new, dx)
        
        # Accept or reject
        if new_value < prev_value:
            h = h_new
            prev_value = new_value
            if new_value < best_value:
                best_h = h.copy()
                best_value = new_value
                last_improvement = iteration
        else:
            # Backtracking line search
            alpha *= 1.2
            if alpha > 1:
                alpha = 0.1  # Reset if too large
        
        # Adaptive learning rate
        alpha = max(1e-4, min(1, alpha * 0.9999))
        
        # Check stagnation
        if iteration - last_improvement > stagnation_limit:
            # Add noise to escape local minima
            h += 0.01 * np.random.randn(n_points)
            h = project_h(h)
            last_improvement = iteration
            
    return (best_h.tolist(), best_value, n_points)
