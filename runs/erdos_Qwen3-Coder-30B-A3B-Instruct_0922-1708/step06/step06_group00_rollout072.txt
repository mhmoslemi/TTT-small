import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Nesterov accelerated proximal gradient with adaptive step size
    gamma = 0.05
    beta = 0.9
    max_iter = 10000
    last_improvement = 0
    patience = 500
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Nesterov momentum
    v = np.zeros_like(h)
    h_acc = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
        
        # Identify the maximum lag
        k = max_lag_idx - (len(h) - 1)
        
        # Compute gradient for the maximum shift
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + k < len(h) and i + k >= 0:
                grad[i] -= (1 - h[i + k]) * dx

        # Proximal gradient step
        h_acc = h + gamma * grad
        h_proj = np.clip(h_acc, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        
        # Nesterov acceleration
        v = beta * v + (h_proj - h)
        h = h_proj + beta * v
        
        # Ensure sum constraint
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / h_sum
        
        # Adaptive step size
        if iteration > 0 and iteration % 100 == 0:
            gamma *= 0.95
            
        # Check if stagnated
        if iteration - last_improvement > patience:
            break
    
    # Final re-evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
