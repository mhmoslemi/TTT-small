import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent parameters
    learning_rate = 5e-4
    dual_step = 0.1
    max_iter = 30000
    progress_count = 0
    max_progress = 1000
    dual_vars = np.zeros(2 * n_points - 1)
    
    # Gradient descent settings per dual update
    grad_steps_per_dual = 5
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlations with current h
        corr_current = correlate(current_h, 1 - current_h, mode='full')
        corr_max_idx = np.argmax(corr_current)
        corr_max_val = corr_current[corr_max_idx]
        
        # Update dual variables using projected gradient ascent
        # Projected gradient ascent on dual variables
        dual_grad = corr_current - dual_vars
        dual_vars += dual_step * dual_grad
        
        # Project to non-negative
        dual_vars = np.maximum(dual_vars, 0)
        
        # Gradient descent for h with fixed duals
        for _ in range(grad_steps_per_dual):
            # Compute gradient with respect to current h and current duals
            grad = np.zeros(n_points)
            for j in range(n_points):
                # Compute contribution to gradient from the specific lag
                for k in range(2 * n_points - 1):
                    if k < n_points and j >= k:
                        grad[j] += dual_vars[k] * (1 - current_h[j - k])
                    elif j + k < 2 * n_points - 1:
                        grad[j] += dual_vars[k] * (1 - current_h[j + k - (n_points - 1)])
            
            # Update h with gradient
            current_h -= learning_rate * grad
            current_h = repair_h(current_h, n_points)
            
        # Update learning rate adaptively
        if i % 500 == 0:
            learning_rate *= 0.99
            
        # Evaluate new C5
        current_c5 = evaluate_h(current_h, dx)

        if current_c5 < best_c5:

            progress_count = 0
