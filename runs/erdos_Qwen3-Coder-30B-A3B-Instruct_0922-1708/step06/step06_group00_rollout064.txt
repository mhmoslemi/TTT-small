import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h = (h_init > 0.5).astype(int)
        h_sum = np.sum(h)
        if h_sum != n_points // 2:
            # Adjust to have exactly n_points/2 ones
            diff = n_points // 2 - h_sum
            if diff > 0:
                # Find zeros and turn them to ones
                zero_indices = np.where(h == 0)[0]
                for i in range(diff):
                    if i < len(zero_indices):
                        h[zero_indices[i]] = 1
            else:
                # Find ones and turn them to zeros
                one_indices = np.where(h == 1)[0]
                for i in range(-diff):
                    if i < len(one_indices):
                        h[one_indices[i]] = 0
    else:
        # Initialize with uniform binary pattern
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # MIP formulation
    # Variable: h binary vector, t continuous
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points // 2]
    
    # Add overlap constraints for all shifts
    for k in range(-n_points+1, n_points):
        if k == 0:
            continue
        # Find valid indices
        valid_i = []
        valid_j = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_i.append(i)
                valid_j.append(j)
        if valid_i:
            # Sum h[i] * (1 - h[j]) for valid pairs
            overlap_expr = cp.sum(cp.multiply(h_var[valid_i], 1 - h_var[valid_j]))
            constraints.append(overlap_expr <= t * (n_points / 2))
    
    # Objective
    objective = cp.Minimize(t)
    
    # MIP problem
    prob = cp.Problem(objective, constraints)
    
    # Solve MIP using CBC
    try:
        # Time limit
        time_limit = budget_s - 10
        prob.solve(solver=cp.CBC, time_limit=time_limit, gap_limit=0.01)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = np.round(h_var.value).astype(int)
        else:
            # Fallback to best known solution
            h_opt = h
    except:
        # Fallback to simple method
        h_opt = h
    
    # Final refinement
    # Use a refinement technique by simulating gradient ascent
    h_refined = h_opt.astype(float)
    max_iter_refine = 200
    learning_rate = 0.01
    for _ in range(max_iter_refine):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        max_overlap = np.max(corr) * dx
        
        # Estimate gradient by finite difference
        grad = np.zeros_like(h_refined)
        epsilon = 1e-5
        
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            sum_h = np.sum(h_plus)
            h_plus = h_plus * (n_points / 2) / sum_h
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            max_overlap_plus = np.max(corr_plus) * dx
            
            grad[i] = (max_overlap_plus - max_overlap) / epsilon
        
        # Update
        h_refined = h_refined - learning_rate * grad
        h_refined = np.clip(h_refined, 0, 1)
        sum_h = np.sum(h_refined)
        h_refined = h_refined * (n_points / 2) / sum_h
    
    # Final evaluation
    corr_final = correlate(h_refined, 1 - h_refined, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
