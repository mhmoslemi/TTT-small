import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = (h_init > 0.5).astype(int)
    else:
        # Create a uniform binary distribution with exactly 50 ones
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # Ensure h has exactly n_points/2 ones
    current_sum = np.sum(h)
    if current_sum != n_points // 2:
        # Adjust to get exactly n_points/2 ones
        diff = n_points // 2 - current_sum
        if diff > 0:
            # Find zeros and turn them to ones
            zero_indices = np.where(h == 0)[0]
            for i in range(diff):
                if i < len(zero_indices):
                    h[zero_indices[i]] = 1
        elif diff < 0:
            # Find ones and turn them to zeros
            one_indices = np.where(h == 1)[0]
            for i in range(-diff):
                if i < len(one_indices):
                    h[one_indices[i]] = 0
    
    # MIP formulation
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points // 2]
    
    # For each shift k, add constraint on overlap
    max_shift = n_points - 1
    for k in range(-max_shift, max_shift + 1):
        # Get valid indices for the current shift
        if k >= 0:
            valid_indices = list(zip(range(n_points - k), range(k, n_points)))
        else:
            valid_indices = list(zip(range(-k, n_points), range(n_points + k)))
        
        # Build constraint sum(h[i] * (1 - h[i + k])) <= t * (n_points / 2)
        overlap_sum = cp.sum([h_var[i] * (1 - h_var[j]) for i, j in valid_indices])
        constraints.append(overlap_sum <= t * (n_points / 2))
        
        if (time.time() - start_time) > budget_s - 10:
            break
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    try:
        # Use CBC solver for binary problem
        prob.solve(solver=cp.CBC, time_limit=budget_s - 10, gap_tol=0.01)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value.round().astype(int)
            if np.sum(h_opt) != n_points // 2:
                # Fallback to a simple local search
                h_opt = h.copy()
        else:
            # Fallback to current best if optimization failed
            h_opt = h.copy()
    except Exception:
        # Fallback to current h if solver fails
        h_opt = h.copy()
    
    # Final refinement of the solution if needed
    best_c5 = float('inf')
    best_h = h_opt
    
    # Evaluate current solution
    corr = correlate(h_opt, 1 - h_opt, mode='full')
    c5 = np.max(corr) * dx
    
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_opt.copy()
    
    # Do a quick local search (basically hill climbing)
    for _ in range(5000):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Try flipping one bit and check if it improves
        i = np.random.randint(0, n_points)
        h_trial = h_opt.copy()
        h_trial[i] = 1 - h_trial[i]
        
        # Ensure the sum constraint
        if np.sum(h_trial) == n_points // 2:
            corr_trial = correlate(h_trial, 1 - h_trial, mode='full')
            c5_trial = np.max(corr_trial) * dx
            
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h_opt = h_trial.copy()
    
    # Final evaluation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
