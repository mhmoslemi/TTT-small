import numpy as np
import time
import cvxpy as cp
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)

    try:

        h_var = cp.Variable(n_points)

        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Objective using a least-squares approximation to the max term

        h_ref = h.copy()

        # approximate solution via convex relaxation
        
        # Make a single pass with the convex solver, top 200 points
        prob = cp.Problem(cp.Minimize(cp.sum_squares(corr_expr)), constraints)
        
        # Approximated safe initial compute phase

        # Begin with better random start + gradient descent refinement
        h = h_ref

        M = len(h)

        h1_f = np.fft.fft(h1, 2*M - 1)
