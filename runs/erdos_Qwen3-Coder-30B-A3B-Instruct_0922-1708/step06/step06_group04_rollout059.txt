import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize swarm
    population_size = 100
    particles = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Create initial population
    for i in range(population_size):
        if i == 0:
            h = h_init.copy()
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        fitness = evaluate_h(h, dx)
        pbest.append(h.copy())
        pbest_fitness.append(fitness)
    
    # Global best
    best_idx = np.argmin(pbest_fitness)
    gbest = pbest[best_idx].copy()
    gbest_fitness = pbest_fitness[best_idx]
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    stagnation_limit = 50
    last_improvement = 0
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate fitness
            fitness = evaluate_h(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = fitness
                    last_improvement = iter_count
        
        # Check for stagnation
        if iter_count - last_improvement > stagnation_limit:
            # Inject diversity
            for _ in range(10):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles.append(h)
                velocities.append(np.random.uniform(-0.5, 0.5, n_points))
                fitness = evaluate_h(h, dx)
                pbest.append(h.copy())
                pbest_fitness.append(fitness)
                
                if fitness < gbest_fitness:
                    gbest = h.copy()
                    gbest_fitness = fitness
                    last_improvement = iter_count
                    
            # Keep best population_size
            indices = np.argsort(pbest_fitness)[:population_size]
            particles = [particles[i] for i in indices]
            velocities = [velocities[i] for i in indices]
            pbest = [pbest[i] for i in indices]
            pbest_fitness = [pbest_fitness[i] for i in indices]
            best_idx = np.argmin(pbest_fitness)
            gbest = pbest[best_idx].copy()
            gbest_fitness = pbest_fitness[best_idx]
            last_improvement = iter_count
        
        iter_count += 1
    
    # Gradient descent refinement of best solution
    h_refined = gbest.copy()
    prev_value = gbest_fitness
    learning_rate = 0.0001
    max_grad_iter = 500
    
    for _ in range(max_grad_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        h_refined = h_refined - learning_rate * grad
        h_refined = repair_h(h_refined, n_points)
        prev_value = evaluate_h(h_refined, dx)
        
        if prev_value < gbest_fitness:
            gbest = h_refined.copy()
            gbest_fitness = prev_value
    
    return (gbest.tolist(), gbest_fitness, n_points)
