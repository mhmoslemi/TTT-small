import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, lambda_vals, dx, n_points):
    """Compute the gradient with respect to h."""
    grad = np.zeros(n_points)
    # For each position i, compute partial derivative
    for i in range(n_points):
        # Contribution from all valid lags k (where h[i] contributes to (h ⋆ (1-h))[k])
        for k in range(2 * n_points - 1):
            if 0 <= i - k < n_points:
                grad[i] += lambda_vals[k] * (1 - h[i - k]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a smooth step-like function
        h_init = np.zeros(n_points)
        h_init[n_points//4:3*n_points//4] = 1.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent parameters
    lambda_vals = np.zeros(2 * n_points - 1)  # dual variables for each lag
    learning_rate = 5e-4
    max_iter = 10000
    
    # Adaptive learning rate
    lr_decay = 0.99
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute current correlations for the max
        corr = correlate(current_h, 1 - current_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_corr = corr[max_lag_idx]
        
        # Update dual variable for the maximum lag
        lambda_vals[max_lag_idx] += learning_rate * (max_corr - best_c5)
        
        # Gradient descent for h
        for _ in range(5):  # Take several gradient steps for each dual update
            if time.time() - start_time > budget_s - 5:
                break
            grad = compute_gradient(current_h, lambda_vals, dx, n_points)
            current_h -= learning_rate * grad
            current_h = repair_h(current_h, n_points)
            
        # Update learning rate
        learning_rate *= lr_decay
            
        # Evaluate and update best solution
        current_c5 = evaluate_h(current_h, dx)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)
