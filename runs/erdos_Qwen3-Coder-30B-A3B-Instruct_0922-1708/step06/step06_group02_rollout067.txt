import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def crossover(parent1, parent2):
    """Create a child from two parents via uniform crossover."""
    child = np.zeros_like(parent1)
    alpha = np.random.rand()
    for j in range(len(parent1)):
        child[j] = alpha * parent1[j] + (1 - alpha) * parent2[j]
    return child

def mutate(h, noise_std):
    """Mutate h by adding Gaussian noise to non-zero elements."""
    h_mut = h.copy()
    for j in range(len(h)):
        if h[j] > 1e-6:
            h_mut[j] += np.random.normal(0, noise_std)
    return h_mut

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    top_candidates = int(population_size * 0.2)
    max_generations = 1000
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init.copy()]
    else:
        population = []
    
    # Fill remaining population with random feasible solutions
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        
    # Initialize best solution
    best_h = None
    best_value = float('inf')
    
    # Initial evaluation
    fitness = []
    for p in population:
        value = evaluate_h(p, dx)
        fitness.append(value)
        if value < best_value:
            best_value = value
            best_h = p.copy()
    
    # Set mutation parameters
    noise_std = 0.1
    learning_rate = 0.001
    
    # Main loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Select top parents
        parents = population[:top_candidates]
        
        # Create new population
        new_population = parents.copy()
        
        # Generate offspring
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < 0.5 and len(parents) > 1:
                p1, p2 = np.random.choice(parents, 2, replace=False)
                child = crossover(p1, p2)
            else:
                # Mutation
                p = np.random.choice(parents)
                child = mutate(p, noise_std)
            
            child = repair_h(child, n_points)
            new_population.append(child)
            
        # Evaluate new population
        new_fitness = []
        for p in new_population:
            value = evaluate_h(p, dx)
            new_fitness.append(value)
            if value < best_value:
                best_value = value
                best_h = p.copy()
        
        # Replace population with new generation
        population = new_population
        fitness = new_fitness
        
        # Adaptive mutation
        if generation % 100 == 0:
            noise_std *= 0.9
        
        # Gradient refinement for best solution
        if best_h is not None:
            h_refined = best_h.copy()
            current_value = best_value
            
            for _ in range(100):
                if time.time() - start_time > budget_s - 10:
                    break
                eps = 1e-6
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_value) / eps
                    
                h_refined = h_refined - learning_rate * grad
                h_refined = repair_h(h_refined, n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                current_value = np.max(corr) * dx
                
                if current_value < best_value:
                    best_value = current_value
                    best_h = h_refined.copy()
    
    return (best_h.tolist(), best_value, n_points)
