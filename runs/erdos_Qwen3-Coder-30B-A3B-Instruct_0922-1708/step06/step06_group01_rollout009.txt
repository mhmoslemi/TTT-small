import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute approximate gradient of C5 with finite differences."""
    n_points = len(h)
    grad = np.zeros(n_points)
    c5_base = evaluate_h(h, dx)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n_points)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

def solve_subproblem(grad, n_points):
    """Solve the linear subproblem: min_d <grad, d> s.t. d in feasible set."""
    # Simple Frank-Wolfe step with maximum descent direction
    # Choose direction that minimizes gradient inner product
    d = np.zeros(n_points)
    # At each point, go in the direction of negative gradient
    for i in range(n_points):
        d[i] = -grad[i]
    
    # Ensure sum(d) = 0
    d_mean = np.mean(d)
    d = d - d_mean
    
    # Clip to [0,1], then scale to make sum=0
    d_clipped = np.clip(d, -1, 1)
    # Project to sum=0
    d_clipped -= np.mean(d_clipped)
    
    return d_clipped

def line_search(h, d, dx, max_steps=10):
    """Find best step size by line search."""
    c5_base = evaluate_h(h, dx)
    eta_max = 1.0
    for i in range(max_steps):
        eta = eta_max * (0.5 ** i)
        h_new = h + eta * d
        h_new = project_h(h_new, len(h))
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_base:
            return eta, h_new
    return 0.0, h

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured pattern (simple pattern to start)
        h_init = np.ones(n_points) * 0.5
        h_init = project_h(h_init, n_points)
    
    # Start Frank-Wolfe
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    h_best = h_current.copy()
    c5_best = c5_current
    
    patience = 1000
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 5 and iteration - last_improvement < patience:
        grad = compute_gradient(h_current, dx)
        d = solve_subproblem(grad, n_points)
        
        step_size, h_new = line_search(h_current, d, dx)
        
        if step_size > 0:
            h_current = h_new
            c5_current = evaluate_h(h_current, dx)
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = iteration
        
        iteration += 1
    
    # Final check
    c5_final = evaluate_h(h_best, dx)
    return (h_best.tolist(), c5_final, n_points)
