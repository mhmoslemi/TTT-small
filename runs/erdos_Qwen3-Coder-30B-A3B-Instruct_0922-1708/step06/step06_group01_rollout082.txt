import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a triangular function that is 0 at ends
        x = np.linspace(0, 2, n_points)
        h_init = 1 - np.abs(x - 1)
        h_init = project_h(h_init, n_points)
    
    # Frank-Wolfe-like approach: find descent direction by finite differences
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    patience = 0
    max_patience = 1000
    
    # Adaptive step sizes
    eta = 1.0
    min_eta = 1e-6
    eta_decay = 0.95
    
    # Frank-Wolfe iterations
    iteration = 0
    while time.time() - start_time < budget_s - 5 and patience < max_patience:
        iteration += 1
        
        # Estimate gradient via finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_current) / eps

        # Solve linear subproblem: find direction d that minimizes grad^T d
        # subject to 0 <= d[i] <= 1, sum(d) = 0
        # Use gradient descent with projected steps
        d = -grad
        # Project d to satisfy sum(d) = 0
        d = d - np.mean(d)
        # Clip d to stay in feasible region (0,1) - not fully correct, but heuristic
        d = np.clip(d, -1, 1) # simple bound
        d = d - np.mean(d)  # Make sum zero

        # Line search for optimal step size
        best_eta = 0
        best_c5 = c5_current
        temp_eta = eta
        
        while temp_eta >= min_eta:
            h_new = h_current + temp_eta * d
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_eta = temp_eta
            temp_eta *= 0.5
        
        if best_eta > 0:
            h_current = h_current + best_eta * d
            h_current = project_h(h_current, n_points)
            c5_current = best_c5
            patience = 0
        else:
            patience += 1
            eta *= eta_decay
        
        # Update the best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            patience = 0
        
        if iteration % 100 == 0:
            eta = max(min_eta, eta * 0.99)
    
    # Final check
    c5_final = evaluate_h(h_best, dx)
    return (h_best.tolist(), c5_final, n_points)
