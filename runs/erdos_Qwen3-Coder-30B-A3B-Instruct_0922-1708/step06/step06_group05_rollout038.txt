import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-scale annealing approach
    n_points_list = [20, 40, 80, 200]
    current_n = 20
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, current_n)
        h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(current_n)
    h = repair_h(h, current_n)
    dx = 2.0 / current_n
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Annealing parameters
    T0 = 1.0
    alpha = 0.99
    T = T0
    min_T = 1e-6
    iteration = 0
    current_scale_idx = 0
    scale_change_interval = 500
    
    while time.time() - start_time < budget_s - 2:
        if T < min_T:
            break
            
        # Perturb h with Gaussian noise scaled by T
        h_perturbed = h + np.random.normal(0, T, current_n)
        h_perturbed = repair_h(h_perturbed, current_n)
        
        # Evaluate
        c5_new = evaluate_h(h_perturbed, dx)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):
            h = h_perturbed
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        
        # Update temperature
        T *= alpha
        
        # Increase resolution periodically
        iteration += 1
        if iteration % scale_change_interval == 0 and current_scale_idx < len(n_points_list) - 1:
            current_scale_idx += 1
            new_n = n_points_list[current_scale_idx]
            # Interpolate best h to new resolution
            x_old = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, new_n)
            h = np.interp(x_new, x_old, best_h)  
            current_n = new_n
            dx = 2.0 / current_n
            h = repair_h(h, current_n)
            # Refine at new resolution
            for _ in range(200):
                if time.time() - start_time > budget_s - 2:
                    break
                eps = 1e-6
                grad = np.zeros(current_n)
                c5_eval = evaluate_h(h, dx)
                for i in range(current_n):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = repair_h(h_plus, current_n)
                    c5_plus = evaluate_h(h_plus, dx)
                    grad[i] = (c5_plus - c5_eval) / eps
                h = h - 0.01 * grad
                h = repair_h(h, current_n)
                new_c5 = evaluate_h(h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            # Reset T for new scale
            T = T0
            
    # Final polishing with gradient descent
    for _ in range(500):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(current_n)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(current_n):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, current_n)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        best_h = best_h - 0.001 * grad
        best_h = repair_h(best_h, current_n)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
    return (best_h.tolist(), best_c5, current_n)
