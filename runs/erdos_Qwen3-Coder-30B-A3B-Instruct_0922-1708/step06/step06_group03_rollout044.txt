import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max, n_points):
    """Compute the gradient of the maximum overlap term."""
    grad = np.zeros(n_points)
    for j in range(n_points):
        # The gradient contribution from the k_max term
        # d/dh_j (h[j] * (1 - h[(j-k_max) % n_points])) = 1 - h[(j-k_max) % n_points] 
        # if j-k_max >= 0, else 1 - h[(j-k_max + n_points) % n_points]
        k_index = (j - k_max) % n_points
        grad[j] = 1 - h[k_index]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Gradient-based optimization with adaptive perturbations
    learning_rate = 1e-4
    cooldown_rate = 0.99
    perturbation_magnitude = 0.05
    perturbation_decay = 0.999
    
    # Track progress
    last_improvement = 0
    last_c5 = current_c5
    
    for i in range(1000000):  # High iteration limit, will break on time
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlation (overlap) for all lags
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient
        grad = compute_gradient(current_h, dx, k_max, n_points)
        
        # Update with gradient descent
        new_h = current_h - learning_rate * grad
        
        # Apply perturbations periodically or when stuck
        if i % 500 == 0 or (i - last_improvement > 1000 and np.random.rand() < 0.1):
            perturbation_magnitude = max(1e-6, perturbation_magnitude * perturbation_decay)
            # Apply small random perturbations
            noise = np.random.uniform(-perturbation_magnitude, perturbation_magnitude, n_points)
            new_h = new_h + noise
            
        # Repair to feasible region
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < 0.05:  # Allow some uphill moves
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                last_improvement = i
                last_c5 = current_c5
        
        # Adjust learning rate
        if i % 100 == 0:
            learning_rate *= cooldown_rate
            
    return (best_h.tolist(), best_c5, n_points)
