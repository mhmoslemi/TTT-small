import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_coarse_h(n_points):
    """Initialize coarse h from initial_h_values or uniform distribution."""
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        h_init = np.ones(n_points) * 0.5
    return h_init

def mutate(h, std):
    """Add Gaussian noise to h with specified standard deviation."""
    return h + np.random.normal(0, std, len(h))

def crossover(p1, p2, prob=0.5):
    """Create offspring by averaging two parents with probability prob."""
    mask = np.random.rand(len(p1)) < prob
    return np.where(mask, p1, p2)

def es_optimize(population, fitness, mutation_std, selection_pressure, crossover_rate, generations, n_points):
    """Evolutionary Strategy optimization loop."""
    pop_size = len(population)
    selection_count = int(pop_size * selection_pressure)
    
    for gen in range(generations):
        # Evaluate fitness
        fitness_values = [fitness(p) for p in population]
        # Rank individuals
        indices = np.argsort(fitness_values)
        population = [population[i] for i in indices]
        fitness_values = [fitness_values[i] for i in indices]
        # Keep top individuals
        selected = population[:selection_count]
        # Generate new population
        new_population = selected.copy()
        while len(new_population) < pop_size:
            # Select parents
            p1, p2 = np.random.choice(selected, 2, replace=False)
            # Crossover
            if np.random.rand() < crossover_rate:
                offspring = crossover(p1, p2)
            else:
                offspring = p1.copy()
            # Mutate
            offspring = mutate(offspring, mutation_std)
            # Repair
            offspring = repair_h(offspring, n_points)
            new_population.append(offspring)
        population = new_population[:pop_size]
        
    return population
