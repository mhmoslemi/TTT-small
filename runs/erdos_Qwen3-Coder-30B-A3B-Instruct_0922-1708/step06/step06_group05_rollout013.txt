import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    best_h = h.copy()
    best_c5 = evaluate_h(h, 2.0 / n_points)
    
    # Annealing parameters
    T0 = 1.0
    alpha = 0.99
    T = T0
    min_T = 1e-6
    
    # Optimization parameters
    lr = 0.01
    decay_every = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s - 5 and T > min_T:
        # Perturb h
        perturbation = np.random.normal(0, T, n_points)
        h_new = h + perturbation
        
        # Repair new h
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new C5
        c5_new = evaluate_h(h_new, 2.0 / n_points)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Update temperature
        T *= alpha
        
        # Periodically increase resolution
        if iteration % 500 == 0 and n_points < 200:

                    h_plus[i] += eps
                    h_plus = repair_h(h_plus, n_points)
                    c5_plus = evaluate_h(h_plus, 2.0 / n_points)
                    grad[i] = (c5_plus - c5_eval) / eps
                h = h - lr * grad
                h = repair_h(h, n_points)
                c5_new = evaluate_h(h, 2.0 / n_points)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    lr *= 0.99  # decay learning rate
