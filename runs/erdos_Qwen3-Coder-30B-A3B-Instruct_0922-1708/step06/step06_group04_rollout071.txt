import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    """Generate initial h using inverse Fourier transform of sieve pattern."""
    # Create frequency domain with decreasing amplitudes
    freq = np.arange(n_points)
    # Higher frequencies are suppressed to reduce overlap
    amp = 1.0 / (1.0 + freq * 0.1)
    # Ensure DC component is zero to maintain sum constraint
    amp[0] = 0
    # Create a complex signal with random phases to add structure
    phase = np.random.rand(n_points) * 2 * np.pi
    freq_signal = amp * np.exp(1j * phase)
    # Take IFFT to get real signal
    h = np.real(np.fft.ifft(freq_signal))
    # Shift to make it positive and normalized
    h = h - np.min(h)
    h = h / np.max(h) * 0.5 if np.max(h) > 0 else h
    return project_h(h, n_points)

    else:

    # Tabu search parameters
    max_iter = 5000
    tabu_list_size = 50
    tabu_list = []
    best_h = h_init.copy()
    best_value = evaluate_overlap(best_h, dx)
    stagnation_count = 0
    
    # Perturbation and learning rates
    perturb_rate = 0.2
    lr = 0.0001

        if time.time() - start_time > budget_s - 10:

        h_neigh = best_h.copy()
        # Randomly change a few (10%) indices
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for idx in indices:
            # Add a random change
            delta = np.random.normal(0, perturb_rate)
            h_neigh[idx] += delta
        h_neigh = project_h(h_neigh, n_points)
        
        # Check if move is tabu
        move = tuple(sorted(indices))
        if move in tabu_list:
            # Accept with low probability if tabu
            if np.random.rand() > 0.7:

            else:
                continue
        else:
            # Add move to tabu list
            tabu_list.append(move)
            if len(tabu_list) > tabu_list_size:

        neighbor_value = evaluate_overlap(h_neigh, dx)
