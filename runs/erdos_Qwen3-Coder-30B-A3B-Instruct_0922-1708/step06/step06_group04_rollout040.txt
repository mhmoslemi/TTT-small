import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to correct sum
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure it's in valid range
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize
    def objective(h):
        return evaluate_h(h, dx)
    
    # Gradient descent with adaptive step size
    best_h = h.copy()
    best_value = objective(h)
    
    # Parameters
    max_iter = 10000
    learning_rate = 0.001
    tolerance = 1e-6
    patience = 100
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute numerical gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        h_current = h.copy()
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = objective(h_plus)
            grad[i] = (c5_plus - objective(h_current)) / eps
            
        # Update
        h = h - learning_rate * grad
        
        # Repair and normalize
        h = repair_h(h, n_points)
        
        # Evaluate
        current_value = objective(h)
        
        if current_value < best_value:
            best_value = current_value
            best_h = h.copy()
            last_improvement = iteration
            
        # Adaptive learning rate
        if iteration > 0 and iteration % 10 == 0:
            if abs(current_value - objective(h_current)) < tolerance:
                learning_rate *= 0.5
            if iteration - last_improvement > patience:
                # Inject small random perturbations to escape local minima
                h = h + np.random.normal(0, 0.01, n_points)
                h = repair_h(h, n_points)
                last_improvement = iteration
    
    # Final gradient descent refinement on best solution
    h_final = best_h.copy()
    prev_value = best_value
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - prev_value) / eps
        
        h_final = h_final - 0.0001 * grad
        h_final = repair_h(h_final, n_points)
        prev_value = evaluate_h(h_final, dx)
        
        if prev_value < best_value:
            best_h = h_final.copy()
            best_value = prev_value
    
    return (best_h.tolist(), best_value, n_points)
