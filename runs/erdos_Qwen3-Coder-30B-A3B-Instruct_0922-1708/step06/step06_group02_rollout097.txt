import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_current = h_init.copy()
    else:
        h_current = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Optimization parameters
    lr = 0.001
    max_iter = 10000
    patience = 100
    best_value = float('inf')
    best_h = None
    no_improvement = 0
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        base_value = evaluate_h(h_current, dx)
        
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - base_value) / eps
        
        # Update step
        h_new = h_current - lr * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        new_value = evaluate_h(h_new, dx)
        
        # Accept or reject update
        if new_value < base_value:
            h_current = h_new
            if new_value < best_value:
                best_value = new_value
                best_h = h_current.copy()
                no_improvement = 0
            else:
                no_improvement += 1
        else:

        # Adjust learning rate
        if no_improvement > 50:
            lr *= 0.5
            no_improvement = 0
            if lr < 1e-6:
                break
                
        # Early stopping if no improvement
        if no_improvement > patience:

    if best_h is not None:
