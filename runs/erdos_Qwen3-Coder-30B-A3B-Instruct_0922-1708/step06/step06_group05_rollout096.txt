import numpy as np
import time
from scipy.signal import fftconvolve

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using FFT-based convolution."""
    # Use FFT-based convolution for efficiency
    corr = fftconvolve(h, 1 - h, mode='full')
    # Take the maximum value and scale by dx
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Rescale to meet integral constraint exactly
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, max_shift_idx):
    """Compute gradient with respect to h for the maximum overlap."""
    eps = 1e-6
    grad = np.zeros_like(h)
    h_eval = evaluate_h(h, dx)
    
    # Only compute gradient for the most critical shifts
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, len(h))
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - h_eval) / eps
    
    return grad
