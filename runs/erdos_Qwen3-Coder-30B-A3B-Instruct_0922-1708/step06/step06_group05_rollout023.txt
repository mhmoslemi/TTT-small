import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using FFT-based convolution."""
    # Use FFT for efficient convolution
    corr = signal.fftconvolve(h, 1 - h, mode='full')
    # Only consider the valid overlaps (shifts from -1 to 1)
    # The full convolution has length 2*n_points - 1
    # The valid range is [n_points-1, 2*n_points-2] for shifts from -1 to 1
    start_idx = 0
    end_idx = 2 * len(h) - 1
    # For overlap computation, we need the shifts in [-1, 1], 
    # which corresponds to the middle section of the correlation
    center = len(h) - 1
    # Take the middle portion excluding the invalid ones
    valid_corr = corr[center : center + len(h)]
    # The maximum overlap and multiply by dx
    return np.max(valid_corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Rescale to meet integral constraint exactly
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points exactly
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # When no parent construction, use a smooth step function
        h = np.ones(n_points)
        h[:n_points//4] = 0.5
        h[n_points//4:3*n_points//4] = 1.0
        h[3*n_points//4:] = 0.5
        
    # Guarantee the constraint
    h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Algorithm parameters
    lr = 0.001
    perturb_rate = 0.05
    stagnation_limit = 100
    stagnation_count = 0
    max_iter = 10000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences at max_overlap region
        eps = 1e-6
        h_eval = evaluate_h(h, dx)
        
        max_shift_idx = np.argmax(signal.fftconvolve(h, 1 - h, mode='full')[len(h)-1:2*len(h)-1])
        max_shift = max_shift_idx - (len(h) - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Compute gradient component at index i for shift max_shift
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            
            # Only evaluate what is needed for gradient at max_shift
            # We can do this without recomputing full convolution
            corr_plus = signal.fftconvolve(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus[len(h)-1:2*len(h)-1]) * dx
            
            grad[i] = (c5_plus - h_eval) / eps
            
        # Update using gradient
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        # Evaluate the new solution
        c5 = evaluate_h(h, dx)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Occasionally stochastic perturbation
        if np.random.rand() < perturb_rate or stagnation_count > stagnation_limit:
            perturb_indices = np.random.choice(n_points, size=int(n_points * perturb_rate), replace=False)
            h[perturb_indices] += np.random.normal(0, 0.005, len(perturb_indices))
            h = repair_h(h, n_points)
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
                
        # Coarse restart every 500 iterations
        if iteration % 500 == 0 and iteration > 0:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            lr /= 2  # Reduce learning rate
            
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = signal.fftconvolve(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus[len(h_plus)-1:2*len(h_plus)-1]) * dx
            grad[i] = (c5_plus - c5_eval) / eps
            
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
    return (best_h.tolist(), best_c5, n_points)
