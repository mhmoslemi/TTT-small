import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = (h_init > 0.5).astype(int)
    else:
        # Initialize with uniform binary pattern
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # Ensure sum is exactly n_points/2
    current_sum = np.sum(h)
    if current_sum > n_points // 2:
        # Turn some 1s to 0s
        zeros_indices = np.where(h == 1)[0]
        for i in range(current_sum - n_points // 2):
            h[zeros_indices[i]] = 0
    elif current_sum < n_points // 2:
        # Turn some 0s to 1s
        ones_indices = np.where(h == 0)[0]
        for i in range(n_points // 2 - current_sum):
            h[ones_indices[i]] = 1
    
    # Define MIP
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Constraint: sum of h equals n_points/2
    constraints = [cp.sum(h_var) == n_points // 2]
    
    # Add overlap constraints
    max_shift = n_points - 1
    for k in range(-max_shift, max_shift + 1):
        # Find valid indices
        valid_indices = []
        if k >= 0:
            # h[i] * (1 - h[i + k]) for i in range(0, n_points - k)
            valid_indices = [(i, i + k) for i in range(n_points - k)]
        else:
            # h[i] * (1 - h[i + k]) for i in range(-k, n_points)
            valid_indices = [(i, i + k) for i in range(-k, n_points)]
        
        if valid_indices:
            # Constraint sum_{valid i} h[i] * (1 - h[i + k]) <= t * (n_points / 2)
            constraint_expr = sum(h_var[i] * (1 - h_var[j]) for i, j in valid_indices)
            constraints.append(constraint_expr <= t * (n_points / 2))
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        # Set a reasonable time limit
        prob.solve(solver=cp.CBC, timeout=budget_s - 10, warm_start=True)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = np.round(h_var.value).astype(int)
            if np.sum(h_solution) == n_points // 2:
                h = h_solution
        else:
            # Fallback to the initial solution
            pass
    except Exception:
        # Use the initial solution if we fail to solve
        pass
    
    # Calculate final C5 value
    corr = correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr) * dx
    
    # Final check
    if abs(np.sum(h) - n_points / 2) > 1e-6:
        # Re-adjust to ensure sum = n_points/2
        current_sum = np.sum(h)
        diff = int(current_sum - n_points // 2)
        if diff > 0:
            # Turn some 1s to 0s
            ones_indices = np.where(h == 1)[0]
            for i in range(diff):
                h[ones_indices[i]] = 0
        elif diff < 0:
            # Turn some 0s to 1s
            zeros_indices = np.where(h == 0)[0]
            for i in range(-diff):
                h[zeros_indices[i]] = 1
    
    # Final evaluation
    corr = correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h.tolist(), final_c5, n_points)
