import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    def to_binary(h_vals):
        return (h_vals > 0.5).astype(float)
    
    def from_binary(b_vals):
        h_new = b_vals.copy()
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h_new = h_new * target_sum / h_sum
        return h_new
    
    # Compute C5 value from h
    def compute_c5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    # Tabu search with simulated annealing
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5
    
    # Tabu list to track recent moves
    tabu_list = []
    tabu_size = 50
    
    # Annealing parameters
    initial_temp = 0.1
    cooling_rate = 0.999
    max_iter = 50000
    
    # Track accepted moves for temperature adjustment
    accepted_moves = 0
    total_moves = 0
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate candidate by flipping one bit
        candidate_h = current_h.copy()
        flip_idx = np.random.randint(0, n_points)
        
        # Keep candidate in binary form
        temp_bin = to_binary(candidate_h)
        temp_bin[flip_idx] = 1 - temp_bin[flip_idx]
        candidate_h = from_binary(temp_bin)
        
        # Skip if this move is in tabu list
        move = (flip_idx,)
        if move in tabu_list:
            continue
        
        # Compute overlap
        candidate_c5 = compute_c5(candidate_h)
        
        # Accept or reject based on SA criteria
        if candidate_c5 < current_c5:
            accept = True
        else:
            # Simulated Annealing acceptance probability
            temp = initial_temp * (cooling_rate ** iteration)
            if temp > 1e-10:
                prob = np.exp(-(candidate_c5 - current_c5) / temp)
                accept = np.random.rand() < prob
            else:
                accept = False
        
        if accept:
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            accepted_moves += 1
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        # Update tabu list
        tabu_list.append(move)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
            
        total_moves += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
