import numpy as np
import time
import cvxpy as cp
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def solve_convex_relaxation(h, n_points, dx):
    """Solve convex relaxation for better initialization."""
    # Formulate with CVXPY
    h_var = cp.Variable(n_points)
    # Set up the convex relaxation problem minimizing a quadratic approximation
    # of the max term
    obj = cp.Minimize(cp.sum_squares(h_var - h) + 
                      cp.sum(cp.multiply(h_var, 1-h_var)))
    constraints = [cp.sum(h_var) == n_points / 2,
                   h_var >= 0,
                   h_var <= 1]
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        return np.array(h_var.value)
    except:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Stage 1: Convex relaxation - use part of budget
    if time.time() - start_time < budget_s * 0.3:
        h = solve_convex_relaxation(h, n_points, dx)
        h = repair_h(h, n_points)
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Stage 2: Gradient-based refinement with adaptive learning rate
    lr_schedule = lambda k: 0.01 / (1 + 0.005 * k)**0.9
    max_iter = 10000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute finite difference gradient more efficiently
        grad = np.zeros(n_points)
        eps = 1e-6
        c5_eval = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps

        # Apply adaptive learning rate
        lr = lr_schedule(iteration)
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        # Evaluate
        c5 = evaluate_h(h, dx)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Occasionally enforce integral constraint via projection
        if iteration % 100 == 0:
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
