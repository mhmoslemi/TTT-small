import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h = (h_init > 0.5).astype(int)
        # Adjust to have exactly n_points/2 ones
        current_sum = np.sum(h)
        target_sum = n_points // 2
        if current_sum != target_sum:
            # Find indices to flip
            zero_indices = np.where(h == 0)[0]
            one_indices = np.where(h == 1)[0]
            if current_sum > target_sum:
                # Need to flip some 1s to 0s
                flip_count = current_sum - target_sum
                indices_to_flip = one_indices[:flip_count]
                h[indices_to_flip] = 0
            else:
                # Need to flip some 0s to 1s
                flip_count = target_sum - current_sum
                indices_to_flip = zero_indices[:flip_count]
                h[indices_to_flip] = 1
    else:
        # Create binary vector with exactly n_points/2 ones
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # MIP formulation
    # Variables
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points // 2]
    
    # For each valid shift k, add overlap constraint
    for k in range(-n_points+1, n_points):
        valid_pairs = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_pairs.append((i, j))
        
        if valid_pairs:
            overlap_expr = cp.sum([h_var[i] * (1 - h_var[j]) for i, j in valid_pairs])
            constraints.append(overlap_expr <= t * (n_points / 2))
    
    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with CBC solver, with a time limit
    try:
        prob.solve(solver=cp.CBC, time_limit=budget_s - 10, log=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.round(h_var.value).astype(int)
        else:
            # If solver fails, use current h as solution
            h_sol = h
    except Exception:
        # Fallback behavior if solver fails
        h_sol = h
    
    # Ensure the final h satisfies constraints
    h_sol = np.clip(h_sol, 0, 1)
    h_sol = (h_sol > 0.5).astype(int)
    current_sum = np.sum(h_sol)
    target_sum = n_points // 2
    if current_sum != target_sum:
        # Adjust to have right sum
        zero_indices = np.where(h_sol == 0)[0]
        one_indices = np.where(h_sol == 1)[0]
        if current_sum > target_sum:
            flip_count = current_sum - target_sum
            indices_to_flip = one_indices[:flip_count]
            h_sol[indices_to_flip] = 0
        else:
            flip_count = target_sum - current_sum
            indices_to_flip = zero_indices[:flip_count]
            h_sol[indices_to_flip] = 1
    
    # Final evaluation
    corr = correlate(h_sol, 1 - h_sol, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_sol.tolist(), final_c5, n_points)
