import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_current = h_init.copy()
    else:
        h_current = np.ones(n_points) * 0.5
        h_current = repair_h(h_current, n_points)
    
    best_h = h_current.copy()
    best_value = evaluate_h(h_current, dx)
    
    # Hierarchical optimization
    levels = [10, 20, 40, 100]
    
    for i, level in enumerate(levels):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Create a coarse version of h
        if level < n_points:
            h_coarse = np.array([np.mean(h_current[j*2:(j+1)*2]) for j in range(level)])
            # Interpolate back to full resolution
            x_coarse = np.linspace(0, 2, level)
            x_full = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_full, x_coarse, h_coarse)
            h_current = repair_h(h_interp, n_points)
        else:
            h_interp = h_current
            
        # Gradient-based optimization
        def objective(h):
            if time.time() - start_time > budget_s - 5:
                return float('inf')
            h = repair_h(h, n_points)
            return evaluate_h(h, dx)
            
        def gradient(h):
            h = repair_h(h, n_points)
            eps = 1e-6
            grad = np.zeros_like(h)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - objective(h)) / eps
            return grad
            
        # Try different optimization approaches
        try:
            # Try L-BFGS optimization
            res = minimize(objective, h_current, method='L-BFGS-B', 
                          bounds=[(0, 1) for _ in range(n_points)],
                          jac=gradient, options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_current = res.x
                h_current = repair_h(h_current, n_points)
            else:
                # Fall back to coordinate descent
                for _ in range(50):
                    if time.time() - start_time > budget_s - 5:
                        break
                    for j in range(n_points):
                        h_plus = h_current.copy()
                        h_plus[j] += 1e-4
                        h_plus = repair_h(h_plus, n_points)
                        val_plus = evaluate_h(h_plus, dx)
                        
                        h_minus = h_current.copy()
                        h_minus[j] -= 1e-4
                        h_minus = repair_h(h_minus, n_points)
                        val_minus = evaluate_h(h_minus, dx)

                        else:

            pass

        current_value = evaluate_h(h_current, dx)

            h = repair_h(h, n_points)

                h_plus[j] += eps

                grad[j] = (c5_plus - objective(h)) / eps
            return grad

            # Grid-based local optimization
            for _ in range(500):
                if time.time() - start_time > budget_s - 5:
                    break
                updates = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = best_h.copy()
                    h_plus[j] += 1e-5
                    h_plus = repair_h(h_plus, n_points)
                    c5_plus = evaluate_h(h_plus, dx) 
                    
                    h_minus = best_h.copy()
                    h_minus[j] -= 1e-5
                    h_minus = repair_h(h_minus, n_points)
                    c5_minus = evaluate_h(h_minus, dx)
                    
                    updates[j] = (c5_minus - c5_plus) / (2e-5)
                    
                delta = 1e-6 * updates
                new_h = best_h - delta
                new_h = repair_h(new_h, n_points)
                if evaluate_h(new_h, dx) < best_value:
                    best_h = new_h
                    best_value = evaluate_h(best_h, dx)
        except Exception as e:
            pass
            
    return (best_h.tolist(), best_value, n_points)
