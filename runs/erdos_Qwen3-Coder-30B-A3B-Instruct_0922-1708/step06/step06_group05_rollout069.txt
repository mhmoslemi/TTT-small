import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    population = []
    
    if 'initial_h_values' in globals():

        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Fill population with random samples
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
        
    # Evolution parameters
    F = 1.0
    CR = 0.9
    stagnation_limit = 100
    stagnation_count = 0
    
    # Keep track of best solution
    best_h = population[0].copy()
    best_c5 = evaluate_h(best_h, dx)
    
    generation = 0
    max_generations = 1000
    
    while generation < max_generations and time.time() - start_time < budget_s - 10:
        new_population = []
        for h in population:
            # DE mutation and crossover
            a, b, c = population[np.random.randint(0, pop_size)], \
                      population[np.random.randint(0, pop_size)], \
                      population[np.random.randint(0, pop_size)]
            v = a + F * (b - c)
            v = np.clip(v, 0, 1)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, v, h)
            trial = repair_h(trial, n_points)
            
            # Evaluate
            c5 = evaluate_h(trial, dx)
            
            # Selection
            if c5 < evaluate_h(h, dx):
                new_population.append(trial)
            else:
                new_population.append(h)
                
        population = new_population
        
        # Track best
        for h in population:
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        if stagnation_count > stagnation_limit:
            F = min(2.0, F + 0.1)
            stagnation_count = 0
            
        generation += 1
        
    # Final refinement with gradient-based method
    for i in range(500):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - c5_eval) / eps

        best_h = best_h - 0.0002 * grad
