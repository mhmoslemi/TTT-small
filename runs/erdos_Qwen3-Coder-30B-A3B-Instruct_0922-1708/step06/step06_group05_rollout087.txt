import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    temp = 1.0

    lr_decay = 0.1

        # Interpolate to current resolution if needed

            h = np.interp(x_new, x_old, h)

            if c5 < best_c5:
