import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Hyperparameters
    n_points = 100
    pop_size = 50
    F = 1.0
    CR = 0.9
    max_generations = 2000
    stagnation_limit = 100
    elite_fraction = 0.5
    
    # Create population
    population = []
    
    # Try to initialize from existing values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        # Interpolate to n_points
        x_orig = np.linspace(0, 2, len(h0))
        x_target = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_target, x_orig, h0)
        population.append(h_interp)
    else:
        population.append(np.random.rand(n_points))
    
    # Fill the rest of the population
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        population.append(h)
    
    # Normalize all in population
    for i in range(len(population)):
        population[i] = repair_h(population[i], n_points)
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_h(best_h, 2.0/n_points)
    
    # Stagnation counter
    stagnation_count = 0
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate new population
        new_population = []
        fitnesses = []
        
        for i in range(pop_size):
            # Select three distinct individuals
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            v = a + F * (b - c)
            v = np.clip(v, 0, 1)
            
            # Normalize to meet integral constraint
            v = repair_h(v, n_points)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, v, population[i])
            
            # Evaluate
            c5 = evaluate_h(trial, 2.0/n_points)
            fitnesses.append(c5)
            new_population.append(trial.copy())
        
        # Tournament selection: keep top 50% (elite)
        indices = np.argsort(fitnesses)[:int(pop_size * elite_fraction)]
        new_population = [new_population[i] for i in indices]
        fitnesses = [fitnesses[i] for i in indices]
        
        # Update population and best solution
        population = new_population
        
        # Update best solution
        for i in range(len(population)):
            c5 = fitnesses[i]
            if c5 < best_c5:
                best_c5 = c5
                best_h = population[i].copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Adapt F if stagnating
        if stagnation_count > stagnation_limit and F < 2.0:
            F += 0.1
            stagnation_count = 0
            
        # Occasionally add random mutations
        if np.random.rand() < 0.01:
            idx = np.random.randint(0, pop_size)
            population[idx] = np.random.rand(n_points)
            population[idx] = repair_h(population[idx], n_points)
            c5 = evaluate_h(population[idx], 2.0/n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = population[idx].copy()
                stagnation_count = 0
    
    # Final refinement using gradient descent on best solution
    h = best_h.copy()
    dx = 2.0 / n_points
    lr = 0.001
    for _ in range(500):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        h = h - lr * grad
        h = repair_h(h, n_points)
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
