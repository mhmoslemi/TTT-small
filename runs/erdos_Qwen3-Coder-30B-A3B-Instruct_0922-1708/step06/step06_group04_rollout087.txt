import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum=50):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    start_time = time.time()
    
    # Initialize from initial_h_values or create starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h += 0.1 * np.random.randn(n_points)
        h = project_h(h, target_sum)
    else:
        h = np.full(n_points, 0.5)
        h = project_h(h, target_sum)
    
    # Parameters
    lambda_reg = 0.01  # Sparsity regularization
    alpha = 0.1        # Initial learning rate
    max_iter = 1000
    min_improvement = 1e-4
    stagnation_limit = 100
    
    current_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient numerically
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, target_sum)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - current_c5) / eps
            
        # Add L1 regularization gradient (sign term)
        reg_grad = lambda_reg * np.sign(h)
        
        # Update
        h_new = h - alpha * (grad + reg_grad)
        
        # Apply constraints
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new function value
        new_c5 = evaluate_h(h_new, dx)
        
        # Backtracking line search
        if new_c5 < current_c5 or iteration < 10:
            h = h_new
            current_c5 = new_c5
            
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            # Reduce learning rate
            alpha *= 0.8
            stagnation_count += 1
            
        # Adapt learning rate
        if new_c5 < current_c5:
            alpha *= 1.2
        else:
            alpha *= 0.8
            
        # Bound learning rate
        alpha = np.clip(alpha, 1e-4, 1.0)
        
        # Check for stagnation
        if stagnation_count > stagnation_limit:
            # Restart with different initialization
            h = np.random.rand(n_points)
            h = project_h(h, target_sum)
            stagnation_count = 0
            alpha = 0.1
            
    return (best_h.tolist(), best_c5, n_points)
