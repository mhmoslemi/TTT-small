import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a sieve-like pattern in frequency domain
    freq = np.fft.fftfreq(n_points, 1/n_points)
    # Design for low-frequency components to minimize overlap
    # This construction encourages a more uniform distribution of h
    amp = np.zeros(n_points)
    # Apply a decreasing envelope to higher frequencies
    # to decrease overlap at larger shifts
    for i in range(1, n_points):
        if i < n_points // 2:
            amp[i] = 1.0 / (1 + i * 0.1)
        else:
            amp[i] = 1.0 / (1 + (n_points - i) * 0.1)
    # Set DC component to achieve correct mean
    amp[0] = 0.5
    # Use inverse FFT
    h = np.real(np.fft.ifft(amp))
    # Add a small offset to ensure all values are positive
    h = h + 0.01
    h = project_h(h, n_points)
    return h
