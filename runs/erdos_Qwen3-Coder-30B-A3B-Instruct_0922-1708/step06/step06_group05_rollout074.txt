import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using FFT-based correlation."""
    # Use FFT-based convolution for efficiency
    corr = signal.fftconvolve(h, 1 - h, mode='full')
    # Only consider the valid range (shifts from -1 to 1)
    mid = len(corr) // 2
    valid_corr = corr[mid-1:mid+1]  # We only care about shifts k in [-1,1]
    # Full convolution includes all shifts from -n+1 to n-1, but we need just the overlap
    # For interval [0,2], shifts in [-1,1] are valid
    shift_indices = np.arange(-len(h)+1, len(h))
    valid_shifts = shift_indices[np.abs(shift_indices) <= 1]
    # Get maximum value over the valid range
    max_val = np.max(valid_corr)
    return max_val * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h
