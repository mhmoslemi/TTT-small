import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient with respect to h for the maximum overlap term."""
    n_points = len(h)
    grad = np.zeros(n_points)
    for j in range(n_points):
        # Gradient contribution from the k_max overlap term
        # d/dh[j] (h[j] * (1 - h[(j-k_max) % n_points])) 
        # = 1 - h[(j-k_max) % n_points] if j is the first term, 0 otherwise
        # Actually, we need the convolution's contribution.
        # For a fixed k_max, let's say the maximum overlap is at lag k_max
        # Then the sum is over i: h[i] * (1 - h[(i - k_max) % n_points])
        # The gradient respect to h[j] of this term is [1 - h[(j - k_max) % n_points]] if
        # there is a term with i = j, i.e., when j = (i - k_max) % n_points,
        # which means when i = (j + k_max) % n_points. This happens exactly once.
        # But also consider how every term contributes to the sum.
        grad[j] = 1 - h[(j - k_max) % n_points]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    early_stopping_count = 0
    max_early_stopping = 1000
    
    for i in range(100000):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Compute overlap for all lags to find k_max
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient at k_max
        grad = compute_gradient(current_h, dx, k_max)
        
        # Update with gradient descent
        new_h = current_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                early_stopping_count = 0
        else:
            early_stopping_count += 1
        
        # Adaptive perturbation
        if early_stopping_count > 500 or (i > 0 and i % 500 == 0):
            noise = np.random.uniform(-perturbation_magnitude, perturbation_magnitude, n_points)
            current_h = current_h + noise
            current_h = repair_h(current_h, n_points)
            current_c5 = evaluate_h(current_h, dx)
            early_stopping_count = 0
            perturbation_magnitude *= 0.999
            learning_rate *= 0.99
            
        # Early stopping check based on progress
        if early_stopping_count > max_early_stopping:
            break
    
    # Final refinement with small steps if time remains
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        corr = correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        grad = compute_gradient(best_h, dx, k_max)
        best_h -= 0.00001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
