import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute gradient of C5 with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    h_base = h.copy()
    c5_base = evaluate_h(h_base, dx)
    
    for i in range(n):
        h_plus = h_base.copy()
        h_minus = h_base.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = repair_h(h_plus, n)
        h_minus = repair_h(h_minus, n)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        c5_minus = np.max(corr_minus) * dx
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

def generate_low_discrepancy_h(n_points, seed=0):
    """Generate a low-discrepancy step function."""
    np.random.seed(seed)
    # Use Halton sequence for better distribution
    indices = np.arange(n_points)
    # Generate Halton sequence (simplified version)
    halton = np.zeros(n_points)
    for i in range(n_points):
        # Simple implementation of base-2 Halton sequence
        index = i + 1
        value = 0
        base = 2
        factor = 0.5
        while index > 0:
            value += (index % base) * factor
            index //= base
            factor /= base
        halton[i] = value
    
    # Scale to [0,1] and create step function
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points), halton)
    return h
