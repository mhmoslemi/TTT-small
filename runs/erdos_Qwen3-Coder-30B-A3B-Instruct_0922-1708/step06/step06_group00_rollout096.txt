import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize momentum
    v = np.zeros_like(h)
    beta = 0.9
    gamma = 0.01
    best_h = h.copy()
    best_c5 = float('inf')
    
    # For tracking improvement
    last_improvement_iter = 0
    no_improvement_count = 0
    max_no_improvement = 500
    
    for iteration in range(10000):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement_iter = iteration
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Check for stagnation and adjust learning rate
        if no_improvement_count > max_no_improvement:
            gamma *= 0.5
            no_improvement_count = 0
            
        # Find the maximum overlap index
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h) - 1)
        
        # Compute gradient for this specific shift
        gradient = np.zeros_like(h)
        if k >= 0:
            gradient[k:] = -dx * (1 - h[:-k])
        else:
            gradient[:k] = -dx * (1 - h[-k:])
        
        # Update using Nesterov accelerated proximal gradient
        # Compute candidate step
        v = beta * v + gradient
        h_candidate = h - gamma * v
        
        # Proximal projection to [0,1] and rescale to sum to n_points/2
        h_candidate = np.clip(h_candidate, 0, 1)
        h_sum = np.sum(h_candidate)
        if abs(h_sum) > 1e-10:  # Avoid division by zero
            h_candidate = h_candidate * (n_points / 2) / h_sum
        
        # Apply Nesterov acceleration
        h = h_candidate + beta * (h_candidate - h)
        
        # Ensure final projection
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / h_sum
        
        # Perturb occasionally to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            perturb_mask = np.random.random(n_points) < 0.001
            h[perturb_mask] += np.random.normal(0, 0.01, np.sum(perturb_mask))
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-8:
                h = h * (n_points / 2) / h_sum
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
