import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def halton_sequence(n, base=2):
    """Generate n elements of a Halton sequence with given base."""
    result = []
    for i in range(n):
        f = 1.0
        result_i = 0.0
        while i > 0:
            f /= base
            result_i += (i % base) * f
            i //= base
        result.append(result_i)
    return np.array(result)

def sobol_sequence(n):
    """Generate n elements of a Sobol sequence."""
    # Simplified Sobol sequence generation
    result = []
    for i in range(n):
        # This is a very simplified version for demonstration
        if i == 0:
            result.append(0.0)
        else:
            result.append((i * 0.5555555555555555) % 1.0)
    return np.array(result)

    np.random.seed(seed)

    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    
    # Try 10 different seeds for initialization
    best_h = None
    best_c5 = float('inf')

        h_init = get_initial_h(n_points, seed_i)
        if h_init is None:
            continue
            
        # Gradient-based optimization 
        def objective(h):
            h = repair_h(h, n_points)
            return evaluate_h(h, dx)
        
        # Optimize with L-BFGS
        try:
            result = minimize(objective, h_init, method='L-BFGS-B', bounds=[(0, 1)]*n_points, 
                              options={'maxiter': 1000, 'ftol': 1e-8})
            h_opt = result.x
            h_opt = repair_h(h_opt, n_points)
            c5_opt = evaluate_h(h_opt, dx)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

            # Fall back to simpler approach if optimization fails
            h_opt = h_init
            c5_opt = evaluate_h(h_opt, dx)
            if c5_opt < best_c5:

        best_h = h
        best_c5 = evaluate_h(h, dx)
    
    # Final optimization with local search
    h_current = best_h.copy()
    c5_current = best_c5
    
    # Simple hill climbing
    for i in range(5000):
