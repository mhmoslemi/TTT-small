import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def optimize_single_step(h, dx, max_iter=100):
    """Perform gradient-based optimization on h."""
    def objective(h_flat):
        h_vec = h_flat.reshape(-1)
        h_vec = repair_h(h_vec, len(h_vec))
        return evaluate_h(h_vec, dx)
    
    def gradient(h_flat):
        h_vec = h_flat.reshape(-1)
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, len(h_plus))
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - objective(h_flat)) / eps
        return grad.flatten()
    
    h_flat = h.flatten()
    result = minimize(objective, h_flat, method='L-BFGS-B', jac=gradient, 
                      bounds=[(0, 1) for _ in range(len(h_flat))],
                      options={'maxiter': max_iter})
    return result.x.reshape(h.shape)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization through hierarchy
    resolutions = [10, 20, 40, 100]
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
        
        if res == 100:
            # Full resolution: do intensive optimization
            h = optimize_single_step(h, dx, max_iter=500)
        else:
            # Coarse resolution: interpolate and optimize
            coarse_h = np.linspace(0, 2, res)
            fine_h = np.linspace(0, 2, n_points)
            h_coarse = np.interp(fine_h, coarse_h, np.linspace(0, 2, res))
            h_coarse = h_coarse * (n_points / 2) / np.sum(h_coarse)
            h = optimize_single_step(h_coarse, dx, max_iter=100)
            
        h = repair_h(h, n_points)
    
    # Final refinement with gradient descent
    best_h = h.copy()
    best_value = evaluate_h(best_h, dx)
    
    # Gradient descent to improve further
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        new_value = np.max(corr) * dx
        if new_value < best_value:
            best_value = new_value
            
    return (best_h.tolist(), best_value, n_points)
