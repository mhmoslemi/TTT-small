import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def augmented_lagrangian_optimize(h_init, n_points, dx, max_iter=1000):
    """Optimize using augmented Lagrangian method."""
    h = h_init.copy()
    lambda_ = 0.0
    rho = 1.0
    tolerance = 1e-6
    last_improvement = 0
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    for iteration in range(max_iter):
        if iteration - last_improvement > 100:
            break
            
        # Compute constraint violation
        constraint_violation = np.sum(h) - n_points / 2
        
        # Compute gradient of objective + penalty term
        grad = np.zeros(n_points)
        for j in range(n_points):
            eps = 1e-6
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - evaluate_h(h, dx)) / eps
            
        # Augmented Lagrangian gradient
        penalty_grad = rho * constraint_violation
        aug_grad = grad + lambda_ * penalty_grad
        
        # Update h with gradient descent and feasibility repair
        step_size = 0.001
        h_new = h - step_size * aug_grad
        h_new = repair_h(h_new, n_points)
        
        # Backtracking line search
        c5_new = evaluate_h(h_new, dx)
        c5_old = evaluate_h(h, dx)
        
        if c5_new < c5_old:
            h = h_new
            constraint_violation = np.sum(h) - n_points / 2
            if abs(constraint_violation) > 1e-4:
                rho *= 2  # Increase penalty
            lambda_ += rho * constraint_violation  # Update Lagrange multiplier
            
            if c5_new < best_value:
                best_value = c5_new
                best_h = h.copy()
                last_improvement = iteration
        else:
            step_size *= 0.5
            if step_size < 1e-8:
                break
                
    return best_h, best_value

        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.full(n_points, 0.5)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
    h = h_init.copy()
    best_value = evaluate_h(h, dx)
    best_h = h.copy()
    
    # Multi-resolution coarse-to-fine optimization
    resolutions = [10, 20, 40, 100]
    last_h = h_init
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Interpolate to current resolution
        if res < n_points:
            x_target = np.linspace(0, 2, res)
            x_orig = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_target, x_orig, last_h)
            h_interp = h_interp * (res / 2) / np.sum(h_interp)
        else:
            h_interp = last_h

        if res < n_points:
            x_target = np.linspace(0, 2, n_points)
            x_orig = np.linspace(0, 2, res)
