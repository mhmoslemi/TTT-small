import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h.""" 
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    h_current = h_init.copy()
    h_current = repair_h(h_current, n_points)
    c5_current = evaluate_h(h_current, dx)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Dual ascent parameters
    dual_vars = np.ones(2 * n_points - 1)  # For all lags
    dual_vars /= np.sum(dual_vars)  # Normalize
    dual_lr = 0.1
    temp = 0.1
    temp_decay = 0.999
    
    # Optimization parameters
    learning_rate = 5e-4
    lr_decay = 0.99
    decay_interval = 500
    dual_update_freq = 10
    stagnation_patience = 1000
    patience_counter = 0
    
    # Gradient descent iterations
    max_iter = 100000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Once in a while, update the dual variables
        if i % dual_update_freq == 0:
            # Compute overlaps for all lags
            corr_full = correlate(h_current, 1 - h_current, mode='full')
            corr_full = corr_full * dx  # Convert to actual overlaps
            
            # Update dual vars based on current overlaps
            # Exploit idea of variational inference: higher overlap = more importance
            exp_weights = np.exp(-corr_full / temp)
            dual_vars = exp_weights / np.sum(exp_weights)
            temp *= temp_decay
        
        # Compute gradient w.r.t. h
        grad = np.zeros(n_points)
        # Use the dual vars to construct gradient
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                # h[i] contributes to overlap at lag k with h[i-k]
                for i in range(k, n_points):
                    grad[i] += dual_vars[k + n_points - 1] * (1 - h_current[i-k])
            else:
                # h[i] contributes to overlap at lag k with h[i-k] where i-k >= 0
                for i in range(-k, n_points):
                    grad[i] += dual_vars[k + n_points - 1] * (1 - h_current[i+k])
        
        # Update h
        h_current = h_current - learning_rate * grad
        h_current = repair_h(h_current, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_current, dx)
        
        if c5_new < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_new
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if i % decay_interval == 0 and i > 0:
            learning_rate *= lr_decay
        
        # Fallback if no progress
        if patience_counter > stagnation_patience:
            h_current = h_init.copy()
            h_current = repair_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            patience_counter = 0

    return (best_h.tolist(), best_c5, n_points)
