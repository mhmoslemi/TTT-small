import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h = h_init.copy()
    else:
        h = np.ones(n_points) * 0.5
    
    # Hierarchical optimization
    resolutions = [10, 20, 40, 100]
    prev_h = None
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Create reduced resolution h
        if res == 100:
            curr_h = h.copy()
        else:
            # Create coarser h
            coarse_h = np.zeros(res)
            for i in range(res):
                start_idx = int(i * n_points / res)
                end_idx = int((i + 1) * n_points / res)
                coarse_h[i] = np.mean(h[start_idx:end_idx])
            curr_h = coarse_h.copy()
        
        # Gradient-based optimization
        def objective(h_vec):
            h_repaired = repair_h(h_vec, len(h_vec))
            return evaluate_h(h_repaired, dx)
        
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(len(curr_h))]
        
        # Optimize using L-BFGS-B
        try:
            result = minimize(objective, curr_h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                              options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6})
            
            if result.success:
                curr_h = result.x
                curr_h = repair_h(curr_h, len(curr_h))
        except:
            pass
        
        # Interpolate back to higher resolution if needed
        if res < 100:
            # Interpolate fine resolution h from coarse
            fine_h = np.zeros(n_points)
            for i in range(n_points):
                idx = int(i * res / n_points)
                idx = min(idx, res - 1)
                fine_h[i] = curr_h[idx]
            h = fine_h.copy()
        else:
            h = curr_h.copy()
    
    # Final gradient descent refinement
    def objective(h_vec):
        h_repaired = repair_h(h_vec, len(h_vec))
        return evaluate_h(h_repaired, dx)
    
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(len(h))]
    
    # Use L-BFGS-B with more iterations
    try:
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                          options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-8})
        if result.success:
            h = result.x
        h = repair_h(h, len(h))
    except:
        pass
    
    final_c5 = evaluate_h(h, dx)
    return (h.tolist(), final_c5, n_points)
