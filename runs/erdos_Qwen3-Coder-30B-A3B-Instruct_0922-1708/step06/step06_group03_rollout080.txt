import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent optimization
    lambda_vals = np.zeros(2 * n_points - 1)  # dual variables for each lag
    learning_rate = 5e-4
    dual_step = 0.05
    max_iter = 30000
    patience = 200
    no_improvement_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute current correlations
        corr = correlate(current_h, 1 - current_h, mode='full')
        
        # Update dual variables (projected gradient ascent)
        lambda_vals += dual_step * (corr - np.max(corr))
        lambda_vals = np.maximum(lambda_vals, 0)  # non-negative constraint
        
        # Compute gradient of Lagrangian with respect to h
        grad = np.zeros(n_points)
        for j in range(n_points):
            temp = np.zeros(2 * n_points - 1)
            temp[j] = 1.0
            convolved = correlate(temp, 1 - current_h, mode='full')
            grad[j] = np.dot(lambda_vals, convolved)
        
        # Gradient descent step
        current_h -= learning_rate * grad
        current_h = repair_h(current_h, n_points)
        
        # Update best solution
        current_c5 = evaluate_h(current_h, dx)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Learning rate decay
        if i % 500 == 0:
            learning_rate *= 0.99
            
        # Early stopping if no improvement
        if no_improvement_count > patience:
            break
    
    # Final refinement using gradient descent with adaptive step size
    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        current_h = current_h - 0.0001 * grad
        current_h = repair_h(current_h, n_points)
        corr = correlate(current_h, 1 - current_h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
