import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum=50.0):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to meet sum constraint
    h = project_h(h, target_sum=n_points/2)
    
    # Regularization parameter for sparsity
    lambda_reg = 0.01
    
    # Initial evaluation
    current_c5 = evaluate_h(h, dx)
    
    # Gradient step size and adaptation
    alpha = 0.1
    min_alpha = 1e-4
    max_alpha = 1.0
    
    # Early stopping
    best_c5 = current_c5
    best_h = h.copy()
    stagnation_count = 0
    
    # Main optimization loop
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, target_sum=n_points/2)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Add L1 regularization gradient
        reg_grad = lambda_reg * np.sign(h)
        total_grad = grad + reg_grad
        
        # Update with backtracking line search
        h_new = h - alpha * total_grad
        h_new = project_h(h_new, target_sum=n_points/2)
        new_c5 = evaluate_h(h_new, dx)
        
        # Backtracking: if no improvement, reduce step size
        backtrack_count = 0
        while new_c5 >= current_c5 and backtrack_count < 10:
            alpha *= 0.5
            alpha = max(alpha, min_alpha)
            h_new = h - alpha * total_grad
            h_new = project_h(h_new, target_sum=n_points/2)
            new_c5 = evaluate_h(h_new, dx)
            backtrack_count += 1
        
        # Accept move if improvement
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            alpha *= 1.2
            alpha = min(alpha, max_alpha)
            stagnation_count = 0
        else:
            alpha *= 0.8
            alpha = max(alpha, min_alpha)
            stagnation_count += 1
            
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early stopping if no improvement
        if stagnation_count > 100:
            break
    
    return (best_h.tolist(), best_c5, n_points)
