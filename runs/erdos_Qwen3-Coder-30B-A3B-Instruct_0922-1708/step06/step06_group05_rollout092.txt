import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    pop_size = 50
    F = 1.0  # Mutation scaling factor
    CR = 0.9  # Crossover probability
    max_generations = 1000
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        # Interpolate initial_h_values to match n_points
        x_orig = np.linspace(0, 2, len(initial_h_values))
        x_target = np.linspace(0, 2, n_points)
        h0 = np.interp(x_target, x_orig, initial_h_values)
        h0 = repair_h(h0, n_points)
        population.append(h0)
    
    # Fill remaining population
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(h, 2.0/n_points) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    stagnation_count = 0
    generation = 0
    
    while generation < max_generations and (time.time() - start_time) < budget_s - 5:
        new_population = []
        new_fitness = []
        
        for i in range(pop_size):
            # Select three distinct individuals
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            v = a + F * (b - c)
            
            # Boundary handling (clip to [0,1])
            v = np.clip(v, 0, 1)
            
            # Normalize to satisfy integral constraint
            v = repair_h(v, n_points)
            
            # Crossover
            trial = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = v[j]
                else:
                    trial[j] = population[i][j]
            
            # Evaluate
            c5 = evaluate_h(trial, 2.0/n_points)
            
            new_population.append(trial)
            new_fitness.append(c5)
            
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = trial.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Selection: keep best half
        combined = list(zip(new_population, new_fitness))
        combined.sort(key=lambda x: x[1])
        population = [x[0] for x in combined[:pop_size]]
        fitness = [x[1] for x in combined[:pop_size]]
        
        # Adaptive F
        if stagnation_count > 50:
            F = min(2.0, F + 0.1)
            stagnation_count = 0
            
        generation += 1
    
    # Final refinement
    for _ in range(500):
        if (time.time() - start_time) > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, 2.0/n_points)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, 2.0/n_points)
            grad[i] = (c5_plus - c5_eval) / eps
        best_h = best_h - 0.0002 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, 2.0/n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
