import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    population = []
    fitness = []
    
    # Initial solution from parent if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize swarm
    for _ in range(population_size):
        if np.random.random() < 0.3:  # 30% use initial solution
            h = h_init.copy()
            # Add small noise
            h += np.random.normal(0, 0.01, n_points)
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_h(h, dx))
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize particle bests
    pbest = population.copy()
    pbest_fitness = fitness.copy()
    
    # Find global best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    # Initialize velocities
    velocities = [np.random.uniform(-0.5, 0.5, n_points) for _ in range(population_size)]
    
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - population[i]) + 
                           c2 * r2 * (best_h - population[i]))
            
            # Update position
            population[i] = population[i] + velocities[i]
            
            # Apply constraints
            population[i] = repair_h(population[i], n_points)
            
            # Update fitness
            new_fitness = evaluate_h(population[i], dx)
            fitness[i] = new_fitness
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = population[i].copy()
                pbest_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < best_value:
                    best_h = population[i].copy()
                    best_value = new_fitness
                    last_improvement = iter_count
        
        # Check for stagnation
        if iter_count - last_improvement > 50:
            # Introduce diversity
            for _ in range(10):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                population.append(h)
                fitness.append(evaluate_h(h, dx))
                
                # Keep best population_size solutions
                indices = np.argsort(fitness)[:population_size]
                population = [population[i] for i in indices]
                fitness = [fitness[i] for i in indices]
                
                # Update pbest too
                pbest = [population[i] for i in indices]
                pbest_fitness = [fitness[i] for i in indices]
                
                # Update global best
                new_best_idx = np.argmin(fitness)
                if fitness[new_best_idx] < best_value:
                    best_h = population[new_best_idx].copy()
                    best_value = fitness[new_best_idx]
                    last_improvement = iter_count
        
        iter_count += 1
    
    # Gradient descent refinement
    h_refined = best_h.copy()
    prev_value = best_value
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        h_refined = h_refined - 0.0001 * grad
        h_refined = repair_h(h_refined, n_points)
        prev_value = evaluate_h(h_refined, dx)
        
        if prev_value < best_value:
            best_h = h_refined.copy()
            best_value = prev_value
    
    return (best_h.tolist(), best_value, n_points)
