import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Use CVXPY to solve the convex optimization
    # Variable for step function h
    h = cp.Variable(n_points)
    # Variable for the maximum overlap
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        t >= 0
    ]
    
    # Compute overlaps for all valid lags (non-circular)
    # For each lag j, compute sum h[i] * (1 - h[i-j]) over valid indices
    # Lag indices span from -(n_points-1) to (n_points-1)
    # But we only consider valid (non-circular) cases where i-j is in [0, n_points-1]
    # The valid lags are from 0 to n_points-1 for forward shifts and 1 to n_points-1 for backward shifts.
    # We can reformulate this as checking lag j from 0 to 2*n_points-2 and adjusting indexing.
    # To avoid circular convolution in our formulations for non-circular overlaps:
    # For each valid lag shift k (where k goes from 0 to n_points-1):
    # We compute sum_{i=0}^{n_points-1} h[i] * (1 - h[(i-k)%n]) for a valid forward shift
    # or we consider length-2n_points-1 full overlap where it wraps around.
    # Our actual problem is not circular: we compute overlaps for the standard
    # cross-correlation in first n_points-1 positions of the full cross-correlation (which is maximum of 2*n_points-1)
    # The valid range is non-circular and we skip direct use of circular index shifts.
    # Let's reformulate: for each lag k in [0, 2*n_points-2], compute:
    # sum_{i=max(0,k), min(n_points-1,n_points-1+k)} h[i] * (1 - h[i-k]) if k >= 0 and i-k >= 0
    
    # Create constraints for all lags in non-circular manner
    # For this task we'll compute overlap correlation for full-length, then extract relevant parts
    # Formulate: h[i] * (1 - h[i - shift]) for valid indices only
    
    max_lag = 2 * n_points - 2
    summary_size = n_points - 1
    # Valid lags range from -summary_size to +summary_size
    
    for k in range(-summary_size, summary_size + 1):
        # For lag k, index i-k must be valid (0 to n_points-1)
        # i ranges from [max(0, k), min(n_points-1, n_points-1+k)] = [max(0, k), min(n_points-1, n_points-1+k)]
        # That is:
        # i from max(0, k) to min(n_points-1, n_points-1+k)
        # This means:
        # if k >= 0: i from k to n_points-1, so valid range is [k, n_points-1]
        # if k < 0: i from 0 to n_points-1+k, so valid range is [0, n_points-1+k]
        if k >= 0:
            indices_i = np.arange(k, n_points)
            indices_j = np.arange(0, n_points - k)
        else:
            indices_i = np.arange(0, n_points + k)
            indices_j = np.arange(-k, n_points)
            
        # overlap = sum(h[i] * (1 - h[j]))
        # This is the part of cross-correlation non-circularly valid.
        # The constraint is sum(h[i] * (1 - h[j])) <= t * (n_points/2) / dx in a scaled form
        # We make it proportionally equal to dx terms to match the overlap integral concept
        # Overlay and use a efficient way to compute dot products for valid indices
        
        # Dot product equivalent:
        # coefficient matrix of terms involving h[i]
        # For j mod n_points = i-k, so j = (i-k) mod n_points
        # For different lag k, we manually construct constraints:
        # Using multilayer indexing and sums
        # Overlap sum (non-circular):
        # computed as corr[k + summary_size] where k in [-summary_size, summary_size]
        # we want sum(h[i] * (1 - h[i-k])) <= t for each valid k
        # Perhaps more directly, let's reconstruct loop differently

        # We reinterpret and create all constraints
        # Let us compute one forward pass:
        # i from max(0, k) to min(n_points-1, n_points-1+k), j = i-k, j from i-k = [max(0, k) - k, min(n_points-1, n_points-1+k) - k]
        # which simplifies to: j from max(-k, 0) to min(n_points-1-k, n_points-1)
        # i from max(0, k) to min(n_points-1, n_points-1+k)
        # for lag k, we compute dot product of index sets:
        # i=0 to n_points-1, but at positions: 0 to max(0,k) inbackward and backward eliminated, and ...
        # Let's just do the direct computation with manual constraint assembling.
        # Build: Dot product h[element] * (1 - h[element+k]) for lag k valid on valid indices:
        # Without repeating code structure, compute overlaps between indices
        if k < 0:
            indices_i = np.arange(0, n_points + k)
            indices_j = np.arange(-k, n_points)
        else:
            indices_i = np.arange(k, n_points)
            indices_j = np.arange(0, n_points - k)
            
        # Determine the corresponding constraint:
        overlap_expr = 0
        for i, j in zip(indices_i, indices_j):
            overlap_expr += h[i] * (1 - h[j])
        constraints.append(overlap_expr <= t * dx * (n_points / 2) )

    # Optimize
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Set a timeout
    # Check if we have enough time to solve exactly
    if time.time() - start_time > budget_s - 50:
        # Use good starting point and solve in a simplified way
        h_solution = h_init.copy()
        h_solution = np.clip(h_solution, 0, 1)
        h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
        c5_bound = evaluate_h(h_solution, dx)
        return (h_solution.tolist(), c5_bound, n_points)
    
    # Use CVXOPT solver
    try:
        # Note: although MOSEK might be better, it's not always easily available
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
        # The original passed realistic optimization even though this could also just be
        # lda optimizer    
    except Exception as e:
        # fallback, if solver failed
        h_solution = h_init.copy()
        h_solution = np.clip(h_solution, 0, 1)
        h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
        c5_bound = evaluate_h(h_solution, dx)
        return (h_solution.tolist(), c5_bound, n_points)
    
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_solution = h.value
    else:
        # Fallback to high-quality starting point
        h_solution = h_init.copy() 
        h_solution = np.clip(h_solution, 0, 1)
        # If not absolutely optimal, improve with local optimization
        h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
        c5_bound = evaluate_h(h_solution, dx)
        # Refine with gradient descent (simpler)
        learning_rate = 0.0001
        eps = 1e-6
        for step in range(200):
            if time.time() - start_time > budget_s - 10:
                break
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_solution.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - c5_bound) / eps

            h_solution = h_solution - learning_rate * grad
            h_solution = np.clip(h_solution, 0, 1)
            h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
            c5_bound = evaluate_h(h_solution, dx)

        return (h_solution.tolist(), c5_bound, n_points)

    # Reassemble and return
    h_solution = np.clip(h_solution, 0, 1)
    h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
    c5_bound = evaluate_h(h_solution, dx)
    return (h_solution.tolist(), c5_bound, n_points)
