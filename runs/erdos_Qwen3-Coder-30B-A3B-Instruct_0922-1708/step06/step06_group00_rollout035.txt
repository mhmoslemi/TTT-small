import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Interpolate if needed
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        # Convert to binary
        h = (h_init > 0.5).astype(int)
        # Adjust sum to be exactly n_points/2
        current_sum = np.sum(h)
        target_sum = n_points // 2
        if current_sum != target_sum:
            # Adjust by flipping bits
            flip_count = int(abs(current_sum - target_sum))
            if current_sum > target_sum:
                # Need to flip some 1's to 0's
                indices = np.where(h == 1)[0]
                flip_indices = indices[:flip_count]
                h[flip_indices] = 0
            else:
                # Need to flip some 0's to 1's
                indices = np.where(h == 0)[0]
                flip_indices = indices[:flip_count]
                h[flip_indices] = 1
    else:
        # Start with uniform binary pattern
        h = np.ones(n_points, dtype=int)
        h[n_points//2:] = 0
    
    # MIP model using CVXPY
    # Variables
    h_vars = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_vars) == n_points // 2]
    
    # For each possible shift k, add constraint on overlap
    # k ranges from -n_points+1 to n_points-1
    for k in range(-n_points + 1, n_points):
        # Find valid indices for this k
        if k >= 0:
            valid_i = list(range(n_points - k))
            valid_j = list(range(k, n_points))
        else:
            valid_i = list(range(-k, n_points))
            valid_j = list(range(n_points + k))
        
        # Create constraints
        overlap_sum = cp.sum(cp.multiply(h_vars[valid_i], 1 - h_vars[valid_j]))
        constraints.append(overlap_sum <= t * (n_points / 2))
    
    # Formulate and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve using CBC solver with timeout
    prob.solve(solver=cp.CBC, time_limit=budget_s - 10, gap_limit=0.01)
    
    # If solution exists, extract optimal h
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h_vars.value
        if h_opt is not None:
            h_opt = np.round(h_opt).astype(int)
    else:
        # If no good solution, return the original h
        h_opt = h.copy()
    
    # Ensure sum of h_opt is exactly n_points/2
    if np.sum(h_opt) != n_points // 2:
        # Adjust by flipping bits to reach target sum
        actual_sum = np.sum(h_opt)
        target_sum = n_points // 2
        if actual_sum > target_sum:
            # Flip some 1's to 0's
            indices = np.where(h_opt == 1)[0]
            flip_count = actual_sum - target_sum
            h_opt[indices[:flip_count]] = 0
        else:
            # Flip some 0's to 1's
            indices = np.where(h_opt == 0)[0]
            flip_count = target_sum - actual_sum
            h_opt[indices[:flip_count]] = 1
    
    # Final evaluation
    corr = correlate(h_opt, 1 - h_opt, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_opt.tolist(), final_c5, n_points)
