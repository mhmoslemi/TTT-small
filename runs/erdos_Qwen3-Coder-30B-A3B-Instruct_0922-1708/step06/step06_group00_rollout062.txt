import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = (h_init > 0.5).astype(int)
    else:
        h = np.ones(n_points, dtype=int)
        h[n_points//2:] = 0
    
    # Ensure sum equals n_points/2
    current_sum = np.sum(h)
    target_sum = n_points // 2
    if current_sum > target_sum:
        # Turn on some 1s to 0s
        positions = np.where(h == 1)[0]
        for i in range(current_sum - target_sum):
            h[positions[i]] = 0
    elif current_sum < target_sum:
        # Turn off some 0s to 1s
        positions = np.where(h == 0)[0]
        for i in range(target_sum - current_sum):
            h[positions[i]] = 1
    
    # MIP formulation
    # Variable: h as binary vector
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_var) == target_sum]
    
    # For each shift k, compute overlap constraint
    # k ranges from -n_points+1 to n_points-1
    for k in range(-n_points + 1, n_points):
        # Compute valid indices for shift k
        valid_indices = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_indices.append((i, j))
        
        if valid_indices:
            # For each pair (i, j), add constraint h[i] * (1 - h[j])
            # This is a linearization for the binary variables
            # We need to ensure sum of h[i] * (1-h[j]) <= t * (n_points / 2)
            # which is equivalent to sum of h[i] - h[i] * h[j] <= t * (n_points / 2)
            # For the MIP, we use a direct formulation.
            term = 0
            for i, j in valid_indices:
                term += h_var[i] * (1 - h_var[j])
            constraints.append(term <= t * (n_points / 2))
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with CBC (default MIP solver)
    try:
        # Set timeout
        prob.solve(solver=cp.CBC, time_limit=budget_s - 10, gap=0.01)
    except Exception:
        pass
        
    # Extract the best solution formed so far
    best_h = h.copy()
    corr = correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # If we have solution from CVXPY solver, use it
    if h_var.value is not None:
        h_sol = np.round(h_var.value).astype(int)
        if np.sum(h_sol) == target_sum:
            corr = correlate(h_sol, 1 - h_sol, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_h = h_sol
                best_c5 = c5
                
    return (best_h.tolist(), best_c5, n_points)
