import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent parameters
    learning_rate = 5e-4
    decay_factor = 0.99
    max_iter = 50000
    max_no_progress = 1000
    no_progress_count = 0
    last_improvement = 0
    
    # Lagrange multipliers
    lambda_k = np.zeros(2 * n_points - 1)
    step_size = 0.05
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute current overlaps
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        k_max_val = corr[k_max]
        
        # Update Lagrange multipliers
        lambda_k += step_size * (corr - k_max_val)
        lambda_k = np.maximum(lambda_k, 0)  # Keep non-negative
        
        # Gradient descent on h
        for _ in range(5):
            # Compute gradient
            grad = np.zeros(n_points)
            for j in range(n_points):
                # Forward convolve lambda with 1-h
                for k in range(2 * n_points - 1):
                    if 0 <= j - (k - n_points + 1) < n_points:
                        grad[j] += lambda_k[k] * (1 - current_h[j - (k - n_points + 1)])
            
            # Update h
            current_h = current_h - learning_rate * grad
            current_h = repair_h(current_h, n_points)
            
        # Re-evaluate C5 after gradient steps
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            no_progress_count = 0
            last_improvement = i
        else:
            no_progress_count += 1
            
        # Adaptive learning rate
        if i % 500 == 0 and i > 0:
            learning_rate *= decay_factor
            
        # Early stopping if no progress
        if no_progress_count > max_no_progress:
            break
            
    # Final refinement using gradient descent if needed
    h_refined = best_h.copy()
    best_c5_refined = best_c5
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5_refined) / eps
            
        h_refined = h_refined - 0.0001 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_c5_refined = np.max(corr) * dx
        
        if best_c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = best_c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
