import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Proximal gradient with Nesterov acceleration
    gamma = 0.1
    beta = 0.9
    max_iter = 10000
    
    best_h = h.copy()
    best_c5 = float('inf')
    h_k = h.copy()
    v_k = np.zeros_like(h)
    h_k_plus_1 = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap for all shifts
        corr = correlate(h_k, 1 - h_k, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_k.copy()
        
        # Find the shift that achieves maximum overlap
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h_k) - 1)
        
        # Compute gradient w.r.t. h_k for the worst case shift
        grad = np.zeros_like(h_k)
        if k >= 0:
            # h[i] contributes to overlap when h[i+k] is considered
            # For h_k, contribution to overlap under shift k is -1 * (1-h_k[i+k])
            # So gradient is -1 if i+k < n_points
            for i in range(len(h_k)):
                if i + k < len(h_k):
                    grad[i] = -1 * (1 - h_k[i + k])
        else:
            # k < 0: h[i+k] contributes to overlap at position i
            for i in range(len(h_k)):
                if i + k >= 0:
                    grad[i] = -1 * (1 - h_k[i + k])
        
        # Apply proximal gradient step
        h_candidate = h_k - gamma * grad
        
        # Project to [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Enforce sum constraint
        h_sum = np.sum(h_candidate)
        if abs(h_sum) > 1e-10:
            h_candidate = h_candidate * (n_points / 2) / h_sum
        
        # Nesterov acceleration
        v_k = beta * v_k + (h_candidate - h_k)
        h_k_plus_1 = h_candidate + beta * v_k
        
        # Ensure feasibility
        h_k_plus_1 = np.clip(h_k_plus_1, 0, 1)
        h_sum = np.sum(h_k_plus_1)
        if abs(h_sum) > 1e-10:
            h_k_plus_1 = h_k_plus_1 * (n_points / 2) / h_sum
        
        h_k = h_k_plus_1.copy()
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
