import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Gradient descent parameters
    lr = 1e-4
    decay = 0.99
    perturbation_freq = 500
    perturbation_mag = 0.05
    perturbation_decay = 0.999
    no_improvement_count = 0
    max_no_improvement = 1000
    
    for i in range(1000000):  # Large iteration count
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap for all lags
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient: dC5/dh_j
        grad = np.zeros(n_points)
        for j in range(n_points):
            # For the maximum overlap term, compute contribution to gradient
            # This approximates: derivative of max_k (h ⋆ (1 - h))[k] w.r.t h[j]
            # Expand integral term: contribution to (h ⋆ (1 - h))[k_max] 
            # due to h[j] is (1 - h[(j - k_max) % n_points])
            for k in range(n_points):
                if k == k_max:
                    l = (j - k) % n_points
                    grad[j] += (1 - current_h[l]) * dx
                    
        # Update with gradient descent
        new_h = current_h - lr * grad
        
        # Apply perturbation occasionally
        if i % perturbation_freq == 0:
            # Perturb some elements to escape local minima
            num_perturb = max(1, n_points // 10)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                delta = np.random.uniform(-perturbation_mag, perturbation_mag)
                new_h[idx] += delta
        
        # Repair to maintain constraints
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept if better or probabilistically accept worse
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            best_h = current_h.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate and perturbation
        if no_improvement_count > 50:
            lr *= decay
            if lr < 1e-8:
                lr = 1e-8
            no_improvement_count = 0
            
        if i % 500 == 0:
            perturbation_mag *= perturbation_decay
            if perturbation_mag < 1e-6:
                perturbation_mag = 1e-6
                
        # Early termination if no improvement for many steps
        if no_improvement_count > max_no_improvement:
            break

    # Final evaluation to make sure we have correct C5
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
