```python
import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def project_to_feasible(h, n_points):
    """Project h to feasible set: [0,1]^n and sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_to_feasible(h_init, n_points)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = project_to_feasible(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    h_best = h_current.copy()
    c5_best = c5_current
    
    patience = 0
    max_patience = 1000
    step_size = 0.01
    perturb_rate = 0.05
    
    iteration = 0
    max_iter = 20000
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_current) / eps
        
        # Update with gradient
        h_new = h_current - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        c5_new = evaluate_h(h_new, dx)
        
        # Accept new solution if better or with some probability
        if c5_new < c5_current or np.random.rand() < 0.01:
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            
        # Adaptive perturbation
        if np.random.rand() < perturb_rate:
            num_perturb = max(1, int(perturb_rate * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                # Apply small random perturbation
                delta = np.random.uniform(-0.05, 0.05)
                h_current[idx] += delta
            h_current = project_to_feasible(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            
        # Adapt step size
        if patience > 50:
            step_size *= 0.95
            if patience > max_patience:
                break
            
        iteration += 1
    
    return (h_best.tolist(), c5_best, n_points)
```
