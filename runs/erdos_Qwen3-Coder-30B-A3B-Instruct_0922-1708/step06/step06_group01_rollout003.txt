import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) > 1e-8:
        # Use quadratic programming to project
        # min 0.5 * ||x - h||^2
        # s.t. sum(x) = target_sum, 0 <= x <= 1
        # This can be solved analytically or via scipy.optimize
        
        # Simple projection: adjust all values equally
        adjustment = (target_sum - current_sum) / n_points
        h += adjustment
        
        # Clip to bounds again
        h = np.clip(h, 0, 1)
        
        # Correct sum exactly
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-10:
            # Find largest element and decrease it
            # or find smallest and increase it
            idx = np.argmax(h)
            if diff > 0:
                # Should increase
                h[idx] = min(1, h[idx] + diff)
            else:
                # Should decrease
                h[idx] = max(0, h[idx] + diff)
    
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute gradient of max correlation w.r.t. h using finite differences."""
    grad = np.zeros_like(h)
    base_c5 = evaluate_h(h, dx)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    
    return grad

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

        if len(h_current) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_current = np.interp(x_target, x_orig, h_current)
        h_current = project_to_feasible(h_current, n_points)
    else:
        # Initialize with a random feasible solution
        h_current = np.random.rand(n_points)
        h_current = project_to_feasible(h_current, n_points)
    
    best_h = h_current.copy()
    best_c5 = evaluate_h(h_current, dx)
    print(f"Initial C5: {best_c5}")
    
    # Parameters
    max_iter = 10000
    step_size = 0.01
    perturb_rate = 0.05
    patience = 1000
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient descent step
        grad = compute_gradient(h_current, dx)
        h_new = h_current - step_size * grad
        
        # Apply perturbation at a fraction of elements
        num_perturb = max(1, int(perturb_rate * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            # Random small perturbation
            perturbation = np.random.normal(0, 0.01)
            h_new[idx] += perturbation
            
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept or reject based on improvement
        if new_c5 < best_c5 or np.random.rand() < 0.1:
            h_current = h_new
            current_c5 = new_c5

                best_c5 = current_c5
                last_improvement = i

        # Update step size and perturb rate
        if i > 0 and i % 100 == 0:
            if i - last_improvement > patience:
                step_size *= 0.9
                perturb_rate *= 1.1
                last_improvement = i
    
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
