import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    population = []
    
    # Use initial_h_values if available

        h = np.interp(x_target, x_orig, h)
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    # Fill population with random solutions
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    def de_mutation(a, b, c, F):
        v = a + F * (b - c)
        return np.clip(v, 0, 1)
        
    def de_crossover(parent, trial, CR):
        mask = np.random.rand(n_points) < CR
        return np.where(mask, trial, parent)
        
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    F = 1.0
    CR = 0.9
    
    iteration = 0
    while time.time() - start_time < budget_s - 5 and iteration < 10000:
        if iteration % 100 == 0 and iteration > 0:
            # Adaptive F based on progress
            if F < 1.8:
                F += 0.05
                
        new_population = []
        for i in range(pop_size):
            # Selection of three distinct individuals
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = de_mutation(a, b, c, F)
            mutant = repair_h(mutant, n_points)
            
            # Crossover
            trial = de_crossover(population[i], mutant, CR)
            
            # Evaluate trial
            c5_trial = evaluate_h(trial, dx)
            
            # If better, keep trial
            if c5_trial < evaluate_h(population[i], dx):
                new_population.append(trial)
            else:
                new_population.append(population[i])
                
        # Selection: keep best POP_SIZE/2
        population_scores = [evaluate_h(h, dx) for h in new_population]
        sorted_indices = np.argsort(population_scores)
        population = [new_population[i] for i in sorted_indices[:pop_size]]
        
        # Update best
        current_best = min(population_scores)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[sorted_indices[0]].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Reset F if stagnation
        if stagnation_count > 100:
            F = max(0.5, F - 0.1)
            stagnation_count = 0
            
        iteration += 1
        
    # Final refinement
    if best_h is not None:
        # Gradient-based refinement
        lr = 0.0001

                h_plus = repair_h(h_plus, n_points)

            best_h = best_h - lr * grad

            new_c5 = evaluate_h(best_h, dx)
            if new_c5 < best_c5:

    # Ensure full precision
    best_h = repair_h(best_h, n_points)
    best_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
