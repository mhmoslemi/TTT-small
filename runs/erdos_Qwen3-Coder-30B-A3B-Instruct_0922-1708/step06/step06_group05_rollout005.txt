import numpy as np
import time
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use dense sampling
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
        # Adjust to meet integral constraint
        h = h * (n_points / 2) / np.sum(h)
    
    # Convex relaxation stage
    # Solve a relaxed version to get a good starting point
    h_relaxed = h.copy()
    
    # Use CVXPY with a dual approach
    h_var = cp.Variable(n_points)
    objective = cp.Minimize(cp.sum_squares(cp.convolution(h_var, 1 - h_var, 'full') * dx))
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1
    ]
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_relaxed = h_var.value
            h_relaxed = np.clip(h_relaxed, 0, 1)
            h_relaxed = h_relaxed * (n_points / 2) / np.sum(h_relaxed)
    except:
        pass
    
    # Refine with projected gradient descent
    h = h_relaxed.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Gradient-free optimized parameters
    max_iter = 5000
    
    # Keep track of recent improvements
    recent_improvements = []
    stagnation_limit = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
        
        # Compute finite difference gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        h_eval = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - h_eval) / eps
        
        # Adaptive step size
        learning_rate = 0.01 / (1.0 + iteration / 100.0)
        h_new = h - learning_rate * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate the new value
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            recent_improvements = []
        else:
            recent_improvements.append(iteration)
            if len(recent_improvements) > stagnation_limit:
                # Random perturbation to escape local minima
                indices = np.random.choice(n_points, size=n_points // 20, replace=False)
                h_new[indices] += np.random.normal(0, 0.01, len(indices))
                h_new = repair_h(h_new, n_points)
                c5_new = evaluate_h(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    recent_improvements = []
        
        h = h_new
    
    return (best_h.tolist(), best_c5, n_points)
