import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_current = h_init.copy()
    else:
        h_current = np.random.rand(n_points)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
    best_h = h_current.copy()
    best_value = evaluate_h(h_current, dx)
    
    # Hierarchical optimization
    levels = [10, 20, 40, 100]
    for level in levels:
        if time.time() - start_time > budget_s - 10:
            break
            
        if level == 100:
            h_sampled = h_current
        else:
            # Downsample to level
            x_orig = np.linspace(0, 2, n_points)
            x_target = np.linspace(0, 2, level)
            h_sampled = np.interp(x_target, x_orig, h_current)
            
        # Optimize on this level
        def optimize_function(h_sampled_flat):
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, len(h_sampled)), h_sampled_flat)
            h_interp = repair_h(h_interp, n_points)
            return evaluate_h(h_interp, dx)
            
        # Flatten the optimization vector
        h_flat = h_sampled.copy()
        
        # Use COBYLA for constraint handling
        constraints = []
        bounds = [(0, 1) for _ in range(len(h_flat))]
        
        # Set up initial solution and optimization
        opts = {'maxiter': 200, 'disp': False}
        try:
            result = minimize(optimize_function, h_flat, method='COBYLA', 
                            constraints=constraints, bounds=bounds, options=opts)
            if result.success:
                h_new = result.x
                h_interp = np.interp(np.linspace(0, 2, n_points), 
                                   np.linspace(0, 2, len(h_new)), h_new)
                h_interp = repair_h(h_interp, n_points)
                current_value = evaluate_h(h_interp, dx)
                if current_value < best_value:
                    best_value = current_value
                    best_h = h_interp.copy()
                h_current = h_interp.copy()
        except:
            pass
        
        # Add some noise for exploration
        if level < 100:
            h_current = repair_h(h_current + np.random.normal(0, 0.01, n_points), n_points)
    
    # Final gradient-based refinement if time allows
    if time.time() - start_time < budget_s - 10:
        def loss_grad(h_params):
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, len(h_params)), h_params)
            h_interp = repair_h(h_interp, n_points)
            corr = correlate(h_interp, 1 - h_interp, mode='full')
            max_corr = np.max(corr)
            c5 = max_corr * dx            
            # Approximate gradient using finite differences
            grad = np.zeros_like(h_params)
            eps = 1e-6
            for i in range(len(h_params)):
                h_plus = h_params.copy()
                h_plus[i] += eps
                h_interp_plus = np.interp(np.linspace(0, 2, n_points), 
                                       np.linspace(0, 2, len(h_plus)), h_plus)
                h_interp_plus = repair_h(h_interp_plus, n_points)
                corr_plus = correlate(h_interp_plus, 1 - h_interp_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            return c5, grad
        
        # Use scipy minimize with L-BFGS-B for better local refinement
        h_flat = best_h.copy()
        optimizer = minimize(lambda x: loss_grad(x)[0], h_flat, method='L-BFGS-B',
                            jac=lambda x: loss_grad(x)[1], bounds=[(0,1) for _ in range(len(h_flat))],
                            options={'maxiter': 200, 'disp': False})
        
        if optimizer.success:
            h_final = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(optimizer.x)), optimizer.x)
            h_final = repair_h(h_final, n_points)
            final_value = evaluate_h(h_final, dx)
            if final_value < best_value:
                best_h = h_final
                best_value = final_value
    
    return (best_h.tolist(), best_value, n_points)
