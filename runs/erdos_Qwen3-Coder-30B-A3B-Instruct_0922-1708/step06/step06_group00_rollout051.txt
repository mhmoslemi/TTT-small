import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50000
    tabu_size = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5

    # Convert to binary representation for tabu search
    def to_binary(h_vals):
        return (h_vals > 0.5).astype(float)
    
    def from_binary(b_vals):
        return b_vals * (n_points / 2) / np.sum(b_vals)
    
    # Ensure constraint sum(h) = n_points/2
    def project_to_constraint(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_sum = np.sum(h_vals)
        if abs(h_sum) < 1e-10:
            h_vals = np.ones_like(h_vals) * 0.5
        else:
            h_vals = h_vals * (n_points / 2) / h_sum
        return h_vals
    
    # Compute C5
    def compute_c5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    # Tabu list to store recently flipped indices
    tabu_list = []
    
    # Initial binary representation
    h_bin = to_binary(h)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    initial_temp = 0.1
    cooling_rate = 0.999
    temperature = initial_temp
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping a random bit
        flip_idx = np.random.randint(0, n_points)
        h_candidate_bin = h_bin.copy()
        h_candidate_bin[flip_idx] = 1 - h_candidate_bin[flip_idx]
        
        # Check tabu list
        if flip_idx in tabu_list:
            # Accept with low probability based on temperature
            if np.random.rand() > np.exp(-1.0 / temperature):
                continue
        
        # Convert to continuous real-valued representation
        h_candidate = from_binary(h_candidate_bin)
        h_candidate = project_to_constraint(h_candidate)
        
        # Compute new C5
        c5_candidate = compute_c5(h_candidate)
        
        # Accept or reject based on simulated annealing
        if c5_candidate < current_c5 or np.random.rand() < np.exp(-(c5_candidate - current_c5) / temperature):
            current_c5 = c5_candidate
            h = h_candidate
            h_bin = h_candidate_bin
            
            # Update tabu list
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(flip_idx)
            
            # Update best
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
        
        # Update temperature
        temperature *= cooling_rate
    
    return (best_h.tolist(), best_c5, n_points)
