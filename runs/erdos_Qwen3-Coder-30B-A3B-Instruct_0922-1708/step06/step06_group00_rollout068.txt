import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Barrier method parameters
    mu = 1.0
    mu_factor = 1.1
    max_iter = 50
    solver = cp.ECOS
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Define convex optimization problem
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Bound constraints
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Overlap constraints
        for k in range(n_points):
            overlap = 0
            for i in range(n_points - k):
                overlap += h_var[i] * (1 - h_var[i + k])
            overlap *= dx
            constraints.append(t >= overlap)
        
        # Barrier term for constraints
        barrier_term = mu * cp.sum(cp.log(h_var) + cp.log(1 - h_var))
        objective = cp.Minimize(t - barrier_term)
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h = np.array(h_var.value).reshape(-1)
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if abs(h_sum - n_points / 2) > 1e-8:
                    h = h * (n_points / 2) / h_sum
            else:
                # Fallback to gradient step if solver fails
                h = h - 0.01 * (h - 0.5)
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if abs(h_sum - n_points / 2) > 1e-8:
                    h = h * (n_points / 2) / h_sum
        except Exception:
            # Fallback to simpler update if solver fails
            h = h - 0.01 * (h - 0.5)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-8:
                h = h * (n_points / 2) / h_sum
                
        # Increase barrier parameter
        mu *= mu_factor
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
