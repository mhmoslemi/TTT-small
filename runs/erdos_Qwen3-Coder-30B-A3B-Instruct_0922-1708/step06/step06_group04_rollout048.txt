import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a structured pattern in frequency domain
    # Design: inverse sieve pattern with high frequencies having higher amplitudes
    freqs = np.arange(n_points)
    # Create a weight vector that decreases with frequency (sieve-like)
    weights = 1.0 / (1.0 + freqs * 0.1)
    # Add some noise to avoid perfect symmetry
    weights += np.random.normal(0, 0.01, n_points)
    # Ensure non-negative
    weights = np.maximum(weights, 0)
    
    # Create frequency domain signal
    # Even part for real signal
    real_part = np.zeros(n_points)
    real_part[:n_points//2] = weights[:n_points//2]
    real_part[n_points//2:] = weights[:n_points//2][::-1]
    
    # Transform back
    h = np.fft.ifft(real_part).real
    # Shift to make more localized
    h = np.roll(h, n_points // 4)
    return project_h(h, n_points)
