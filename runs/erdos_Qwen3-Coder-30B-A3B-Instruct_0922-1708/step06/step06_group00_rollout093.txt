```python
import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Use CVXPY for convex optimization approach
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Barrier method parameters
    mu = 1.0
    mu_factor = 1.1
    max_iter = 500
    
    # Set up the optimization variables
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1]
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Build overlap constraints
    for k in range(n_points):
        # Compute overlap for shift k
        overlap_expr = 0
        for i in range(n_points - k):
            overlap_expr += h_var[i] * (1 - h_var[i+k])
        overlap_expr *= dx
        constraints.append(t >= overlap_expr)
    
    # Barrier function - minimize t + mu * sum(-log(h_var[i]) - log(1-h_var[i]))
    barrier_term = 0
    for i in range(n_points):
        barrier_term += -cp.log(h_var[i] + 1e-12) - cp.log(1 - h_var[i] + 1e-12)
    
    objective = cp.Minimize(t + mu * barrier_term)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Update h in the problem
        h_var.value = h.copy()
        
        # Solve the convex problem
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                # Check if we have a better solution
                corr = correlate(h_new, 1 - h_new, mode='full')
                c5 = np.max(corr) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                
                h = h_new.copy()
                mu *= mu_factor
            else:
                # Fall back to naive approach if problem fails
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                target_sum = n_points / 2
                if abs(h_sum - target_sum) > 1e-8:
                    h = h * target_sum / h_sum
        except:
            # If solving fails, fallback to rescaled update
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            target_sum = n_points / 2
            if abs(h_sum - target_sum) > 1e-8:
                h = h * target_sum / h_sum

    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
```
