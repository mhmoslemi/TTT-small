import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Proximal gradient with Nesterov acceleration
    gamma = 0.05
    beta = 0.9
    max_iter = 10000
    
    h_prev = h.copy()
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Identify the maximum lag
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h) - 1)
        
        # Compute gradient for overlap at lag k
        grad = np.zeros_like(h)
        if k >= 0:
            grad[:len(h)-k] = -(1 - h[k:])
        else:
            grad[-k:] = -(1 - h[:len(h)+k])
        
        # Proximal gradient step
        h_candidate = h - gamma * grad
        h_proj = np.clip(h_candidate, 0, 1)
        
        # Rescale to satisfy sum constraint
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        
        # Nesterov acceleration
        v = beta * v + (h_proj - h_prev)
        h = h_proj + beta * v
        
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        h_prev = h.copy()
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
