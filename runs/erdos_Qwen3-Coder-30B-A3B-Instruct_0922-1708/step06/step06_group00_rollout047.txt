import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Optimization parameters
    max_iter = 10000
    gamma = 0.1
    beta = 0.9
    best_h = h.copy()
    best_c5 = float('inf')
    
    # For Nesterov acceleration
    v = np.zeros_like(h)
    h_acc = h.copy()
    
    # Store best solution found so far
    best_solution = (h.copy(), float('inf'), n_points)
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlations for all shifts
        corr = correlate(h, 1-h, mode='full')
        c5 = np.max(corr) * dx
        
        # Record best solution so far
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_solution = (best_h.copy(), best_c5, n_points)
        
        # Identify maximum overlap lag
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h) - 1)
        
        # Compute gradient for the worst case shift
        grad = np.zeros_like(h)
        if k >= 0:
            # h[i] affects overlap with h[i + k]
            for i in range(len(h) - k):
                grad[i] -= (1 - h[i + k]) * dx
        else:
            # h[i + |k|] affects overlap with h[i]
            for i in range(-k, len(h)):
                grad[i - (-k)] -= (1 - h[i]) * dx
        
        # Nesterov accelerated update
        v = beta * v + grad
        h_acc = h - gamma * v
        
        # Proximal projection onto [0,1] and sum constraint
        h_acc = np.clip(h_acc, 0, 1)
        h_sum = np.sum(h_acc)
        if abs(h_sum) > 1e-10:
            h_acc = h_acc * (n_points / 2) / h_sum
        
        # Update h with Nesterov acceleration
        h = h_acc + beta * (h_acc - h)
        
        # Final projection
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - (n_points / 2)) > 1e-8:
            h = h * (n_points / 2) / h_sum
    
    # Evaluate final best solution again to be sure
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
