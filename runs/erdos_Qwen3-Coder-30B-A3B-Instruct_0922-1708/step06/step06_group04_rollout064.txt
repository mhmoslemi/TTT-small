import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, target_sum=50):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Add small perturbation
    h += np.random.normal(0, 0.05, n_points)
    h = project_h(h)
    
    # Optimizer parameters
    alpha = 0.1
    lambda_reg = 0.01  # L1 regularization weight
    min_alpha = 1e-4
    max_alpha = 1.0
    max_iter = 10000
    patience = 100
    best_value = float('inf')
    best_h = h.copy()
    
    iter_count = 0
    last_improvement = 0
    prev_c5 = evaluate_h(h, dx)
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_c5) / eps
        
        # Add L1 regularization gradient
        reg_grad = lambda_reg * np.sign(h)
        
        # Update step
        h_new = h - alpha * (grad + reg_grad)
        h_new = project_h(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_h(h_new, dx)
        
        # Backtracking line search
        if new_c5 < prev_c5:
            h = h_new
            prev_c5 = new_c5
            alpha *= 0.9  # Decrease learning rate
            alpha = max(min_alpha, alpha)
            
            if new_c5 < best_value:
                best_value = new_c5
                best_h = h.copy()
                last_improvement = iter_count
        else:
            alpha *= 1.1  # Increase learning rate
            alpha = min(max_alpha, alpha)
        
        iter_count += 1
        
        # Early stopping if no improvement
        if iter_count - last_improvement > patience:
            break
            
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
