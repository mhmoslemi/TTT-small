import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints using quadratic projection."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Project onto the feasible set
        # Solve min ||x - h||^2 s.t. sum(x) = target_sum, 0 <= x <= 1
        # Using Lagrange multipliers: x* = h - lambda * ones
        # Need to satisfy sum(x*) = target_sum
        # sum(h) - lambda * n = target_sum => lambda = (sum(h) - target_sum) / n
        lambda_val = (current_sum - target_sum) / n_points
        h_proj = h - lambda_val
        # Then clip to [0,1]
        h = np.clip(h_proj, 0, 1)
        # Re-adjust to exactly target_sum
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        # Distribute the difference
        if abs(diff) > 1e-8:
            indices = np.argsort(h)
            # Add diff to the smallest elements
            for i in range(int(np.ceil(abs(diff)))):
                if diff > 0:
                    h[indices[i]] = min(1.0, h[indices[i]] + diff)
                else:
                    h[indices[-(i+1)]] = max(0.0, h[indices[-(i+1)]] + diff)
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute finite difference gradient of C5 w.r.t. h."""
    n_points = len(h)
    grad = np.zeros(n_points)
    h_base = h.copy()
    c5_base = evaluate_h(h_base, dx)
    for i in range(n_points):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n_points)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial_h_values if available

            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.random.rand(n_points)
    
    # Repair initial solution
    h_init = repair_h(h_init, n_points)
    
    # Optimizer parameters
    max_iter = 20000
    step_size = 0.01
    patience = 1000
    last_improvement = 0
    best_h = h_init.copy()
    best_c5 = evaluate_h(h_init, dx)
    
    h_current = h_init.copy()
    c5_current = best_c5
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        grad = compute_gradient(h_current, dx)
        
        # Update with gradient
        h_new = h_current - step_size * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                last_improvement = i
        else:
            # Accept with probability
            prob = np.exp(-(c5_new - c5_current) / max(1e-8, np.linalg.norm(grad)))
            if np.random.rand() < prob:
                h_current = h_new
                c5_current = c5_new
                
        # Adaptive step size
        if i % 100 == 0 and i > 0:
            step_size *= 0.99
        
        # Early stopping
        if i - last_improvement > patience:
            break
            
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = compute_gradient(best_h, dx)
        best_h = best_h - 0.001 * grad
        best_h = repair_h(best_h, n_points)
        c5_final = evaluate_h(best_h, dx)
        if c5_final < best_c5:
            best_c5 = c5_final
