import numpy as np
import time
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_halton_sequence(n_points, seed=42):
    """Generate a Halton sequence for initialization."""
    sampler = qmc.Halton(d=1, seed=seed)
    points = sampler.random(n_points)
    return points.flatten()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Try multiple seeds for different low-discrepancy sequences
    for init_seed in range(10):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate low-discrepancy sequence
        try:
            halton_vals = generate_halton_sequence(n_points, init_seed)
            # Map to [0, 2] interval
            halton_vals = halton_vals * 2
            
            # Create step function: interpolate and then construct step values
            x_vals = np.linspace(0, 2, n_points)
            # Interpolate low-discrepancy values onto grid
            h_init = np.interp(x_vals, np.linspace(0, 2, n_points), halton_vals)
            
            # Slightly randomize to avoid exact degeneracies
            h_init += np.random.normal(0, 0.01, n_points)
            h_init = repair_h(h_init, n_points)
        except:
            # Fallback to random initialization if something goes wrong
            h_init = np.random.rand(n_points)
            h_init = repair_h(h_init, n_points)

        # Refinement using gradient descent
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Gradient-based refinement
        learning_rate = 0.01
        max_iter_refinement = 5000
        
        for ref_iter in range(max_iter_refinement):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient using finite differences
            eps = 1e-6
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_current.copy()
                h_minus[i] -= eps
                h_minus = repair_h(h_minus, n_points)
                c5_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with gradient
            h_new = h_current - learning_rate * grad
            h_new = repair_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
        
        # Periodic reinitialization
        if ref_iter % 1000 == 0 and ref_iter > 0:
            h_init = np.random.rand(n_points)
            h_init = repair_h(h_init, n_points)
            h_current = h_init
    
    # Final verification of best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to initial_h_values or random if nothing found
        if 'initial_h_values' in globals():
            h_fallback = np.array(initial_h_values)
            if len(h_fallback) != n_points:
                x_orig = np.linspace(0, 2, len(h_fallback))
                x_target = np.linspace(0, 2, n_points)
                h_fallback = np.interp(x_target, x_orig, h_fallback)
            h_fallback = repair_h(h_fallback, n_points)
            c5_fallback = evaluate_h(h_fallback, dx)
            return (h_fallback.tolist(), c5_fallback, n_points)
        else:
            h_fallback = np.random.rand(n_points)
            h_fallback = repair_h(h_fallback, n_points)
            c5_fallback = evaluate_h(h_fallback, dx)
            return (h_fallback.tolist(), c5_fallback, n_points)
