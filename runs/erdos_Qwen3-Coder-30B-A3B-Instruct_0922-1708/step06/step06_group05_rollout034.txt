import numpy as np
import time
from scipy import signal
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points

        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Convex relaxation stage
    if time.time() - start_time < budget_s - 30:
        # Use CVXPY for convex relaxation
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Approximate the maximum by minimizing a quadratic
        # We'll minimize sum((h * (1 - h))[k]^2) for all shifts k
        h_ref = h.copy()
        b = np.correlate(h_ref, 1 - h_ref, mode='full')
        b = b * dx
        
        # Minimize a weighted sum of squared correlations
        # We'll solve an approximating problem
        # Convolution can be computed via FFT  
        def correlation_square_sum(h_vals):
            corr = np.correlate(h_vals, 1 - h_vals, mode='full')
            return np.sum(corr**2) * dx**2
        
        # Simple alternating proximal gradient approach
        max_iter_convex = 100
        lr = 0.01
        for _ in range(max_iter_convex):
            if time.time() - start_time > budget_s - 20:
                break
            # Use projected gradient
            h = h - lr * (2 * dx * (np.correlate(h, 1-h, mode='full')[2:] - b[2:]))
            h = repair_h(h, n_points)
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Refinement stage
    if time.time() - start_time < budget_s - 5:
        lr0 = 0.01
        beta = 0.9
        min_lr = 1e-6
        stagnation_limit = 100
        stagnation_count = 0
        last_improvement = 0
        
        for iteration in range(1000):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Gradient estimation using finite differences
            grad = np.zeros(n_points)
            base_c5 = evaluate_h(h, dx)
            
            # Estimate the gradient at each point
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                grad[i] = (c5_plus - base_c5) / eps
            
            # adaptive learning rate
            eta = lr0 / (1 + (iteration ** beta))
            eta = max(eta, min_lr)
            
            h = h - eta * grad
            h = repair_h(h, n_points)
            c5 = evaluate_h(h, dx)
            
            # Check progress
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
                last_improvement = iteration
            else:
                stagnation_count += 1
                
            if stagnation_count > stagnation_limit:
                break
                
    return (best_h.tolist(), best_c5, n_points)
