import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction or random binary
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Convert to binary if not already
    h_binary = (h > 0.5).astype(float)
    h = h_binary.copy()
    
    # Start with the constraint that sum(h) = n_points / 2
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-8:
        # Adjust by scaling or flipping bits
        h = h * (n_points / 2) / current_sum
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    max_iter = 50000
    T = 0.1
    cooling_rate = 0.999
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping one bit
        idx = np.random.randint(0, n_points)
        h_candidate = h.copy()
        h_candidate[idx] = 1 - h_candidate[idx]
        
        # Ensure sum constraint
        diff = np.sum(h_candidate) - (n_points / 2)
        if abs(diff) > 1e-8:
            # Adjust another bit to compensate
            idx2 = np.random.randint(0, n_points)
            while idx2 == idx:
                idx2 = np.random.randint(0, n_points)
            h_candidate[idx2] = 1 - h_candidate[idx2]
        
        # Compute overlap
        corr = correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr) * dx
        
        # If better, accept
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
        
        # Accept with probability if not better
        if c5_candidate < best_c5 or np.random.rand() < np.exp(-(c5_candidate - best_c5) / T):
            h = h_candidate.copy()
            
        # Keep tabu list
        tabu_list.append(idx)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
            
        # Cool down
        T *= cooling_rate
    
    # Final evaluation with best solution
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
