import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def interpolate_h(h_coarse, n_points):
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_points)
    return np.interp(x_fine, x_coarse, h_coarse)

def mutation(h, std):
    noise = np.random.normal(0, std, len(h))
    h_mut = h + noise
    return repair_h(h_mut, len(h))

def crossover(h1, h2):
    mask = np.random.random(len(h1)) < 0.5
    h_child = np.where(mask, h1, h2)
    return h_child

def es_optimize(pop_size, generations, h_coarse, n_points, dx):
    current_population = [h_coarse.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        current_population.append(h)
    
    for gen in range(generations):
        # Evaluate all individuals
        fitness = []
        for h in current_population:
            fit = evaluate_h(h, dx)
            fitness.append(fit)
        
        # Sort and select top 20%
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:max(1, int(pop_size * 0.2))]
        
        # Create new population
        new_population = [current_population[i] for i in top_indices]
        
        # Retain top individuals and add new ones
        while len(new_population) < pop_size:
            # Selection via tournament
            idx1, idx2 = np.random.choice(top_indices, 2, replace=False)
            parent1 = current_population[idx1]
            parent2 = current_population[idx2]
            
            # Crossover
            if np.random.random() < 0.3:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
                
            # Mutation
            std = 0.1 / (n_points**0.5)
            child = mutation(child, std)
            new_population.append(child)
        
        current_population = new_population
        
    # Return best individual
    fitness = [evaluate_h(h, dx) for h in current_population]
    best_idx = np.argmin(fitness)
    return current_population[best_idx], fitness[best_idx]

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial coarse resolution
    coarse_resolution = 10
    dx_coarse = 2.0 / coarse_resolution
    
    # Initialize coarse h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != coarse_resolution:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, coarse_resolution)
            h_coarse = np.interp(x_target, x_orig, h_init)
        else:
            h_coarse = h_init.copy()
    else:
        # Initialize with uniform distribution
        h_coarse = np.random.rand(coarse_resolution)
    
    # Adjust to sum constraint
    h_coarse = h_coarse * (coarse_resolution / 2) / np.sum(h_coarse)
    h_coarse = repair_h(h_coarse, coarse_resolution)
    
    # Coarse optimization
    h_best_coarse, c5_coarse = es_optimize(
        pop_size=50,
        generations=100,
        h_coarse=h_coarse,
        n_points=coarse_resolution,
        dx=dx_coarse
    )
    best_h = h_best_coarse
    best_c5 = c5_coarse
    
    # Gradual refinement
    resolutions = [20, 40, 100]
    for res in resolutions:
        if time.time() - start_time > budget_s - 30:
            break
            
        # Interpolate to target resolution
        h_fine = interpolate_h(h_best_coarse, res)
        
        # Add some diversity to population
        current_population = [h_fine.copy()]
        for _ in range(49):
            h = np.random.rand(res)
            h = repair_h(h, res)
            current_population.append(h)
        
        # Optimize at this resolution
        h_best, c5_best = es_optimize(
            pop_size=50,
            generations=50,
            h_coarse=h_fine,
            n_points=res,
            dx=2.0 / res
        )
        
        # Update best solution if improved
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best
        
        # Update coarse solution for next iteration
        h_best_coarse = h_best
        # Use same dx for next step (should work for next resolution)
        
    # Final refinement with gradient descent
    dx_final = 2.0 / len(best_h)
    h_refined = best_h.copy()
    current_c5 = best_c5
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(len(h_refined))
        for i in range(len(h_refined)):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, len(h_refined))
            corr = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr) * dx_final
            grad[i] = (c5_plus - current_c5) / eps
            
        h_refined = h_refined - 0.001 * grad
        h_refined = repair_h(h_refined, len(h_refined))
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        current_c5 = np.max(corr) * dx_final
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
    
    return (best_h.tolist(), best_c5, len(best_h))
