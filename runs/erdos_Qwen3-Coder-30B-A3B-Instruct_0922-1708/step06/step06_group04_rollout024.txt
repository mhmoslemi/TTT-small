import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a sieve-like pattern in frequency domain
    # High frequencies get lower amplitudes to reduce overlap
    freqs = np.arange(n_points)
    # Inverse sieve: higher frequencies have lower amplitudes
    amplitudes = 1.0 / (1.0 + freqs**2 / (n_points/4)**2)
    # Add some randomness to break symmetry
    amplitudes *= (1 + 0.1 * np.random.rand(n_points))
    # Create complex signal with these amplitudes and random phases
    phases = np.random.rand(n_points) * 2 * np.pi
    signal = amplitudes * np.exp(1j * phases)
    # Inverse FFT to get real signal
    h = np.real(np.fft.ifft(signal))
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    return h

        new_value = evaluate_overlap(h_new, dx)
