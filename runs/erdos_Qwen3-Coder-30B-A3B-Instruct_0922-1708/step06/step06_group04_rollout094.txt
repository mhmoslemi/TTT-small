import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from the parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a symmetric function
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.1 * np.sin(np.pi * x)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure initial bounds and sum
    h = repair_h(h, n_points)
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Gradient descent with adaptive learning rate
    max_iter = 10000
    learning_rate = 0.0001
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute numerical gradient
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
        
        # Update step
        h_new = h - learning_rate * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        new_value = evaluate_h(h_new, dx)
        
        if new_value < best_value:
            h = h_new
            best_value = new_value
            last_improvement = iter_count
            if best_value < 0.3807:
                break
        else:
            # Reduce learning rate if no improvement
            learning_rate *= 0.995
            
        # Inject diversity if no improvement for a while
        if iter_count - last_improvement > 100:
            # Random perturbation with controlled magnitude
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + perturbation
            h = repair_h(h, n_points)
            last_improvement = iter_count
            
        iter_count += 1
    
    # Final refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        grad = np.zeros(n_points)
        eps = 1e-6
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
        
        h = h - 0.00005 * grad
        h = repair_h(h, n_points)
        new_value = evaluate_h(h, dx)
        
        if new_value < best_value:
            best_h = h.copy()
            best_value = new_value
            
    return (best_h.tolist(), best_value, n_points)
