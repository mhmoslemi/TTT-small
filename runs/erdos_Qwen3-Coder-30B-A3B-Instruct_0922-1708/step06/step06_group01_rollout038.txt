import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = np.maximum(0, 1 - np.abs(x - 1))
        h_init = repair_h(h_init, n_points)
    
    # Coordinate descent with projection
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient computation parameters
    epsilon = 1e-6
    learning_rate = 0.01
    
    # Track last improvement and plateau detection
    last_improvement = 0
    plateau_count = 0
    max_plateau = 100
    
    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        improved = False
        
        # Shuffle indices for better exploration
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for i in indices:
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute finite difference gradient approximation
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            
            # Perturb current point
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            
            # Ensure feasibility after perturbation
            h_plus = repair_h(h_plus, n_points)
            h_minus = repair_h(h_minus, n_points)
            
            # Evaluate
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Approximate gradient
            grad_i = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update gradient descent step
            h_current[i] = h_current[i] - learning_rate * grad_i
            
            # Project back to feasible set
            h_current = repair_h(h_current, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_current, dx)
            
            if c5_new < c5_current:
                c5_current = c5_new
                improved = True
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = iter_count
        
        # Adaptive learning rate adjustments
        if improved:
            plateau_count = 0
            learning_rate = min(0.1, learning_rate * 1.01)
        else:
            plateau_count += 1
            if plateau_count > max_plateau:
                # Perturb some coordinates to escape local minima
                num_perturb = max(1, int(0.05 * n_points))
                perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
                for idx in perturb_indices:
                    h_current[idx] += np.random.uniform(-0.01, 0.01)
                    h_current[idx] = np.clip(h_current[idx], 0, 1)
                
                # Re-project
                h_current = repair_h(h_current, n_points)
                c5_current = evaluate_h(h_current, dx)
                plateau_count = 0
                learning_rate = max(1e-6, learning_rate * 0.8)

    return (h_best.tolist(), c5_best, n_points)
