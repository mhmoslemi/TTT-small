import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    """Create an initial h using inverse Fourier transform with sieve pattern"""
    # Create frequency components with inverse sieve pattern
    freq = np.fft.fftfreq(n_points, 1.0/n_points)
    # Higher frequencies get lower amplitudes (sieve pattern)
    amp = 1.0 / (1 + np.abs(freq)**1.5) 
    # Add some complex phase to break symmetry
    phase = np.random.uniform(0, 2*np.pi, n_points)
    comp = amp * np.exp(1j * phase)
    
    # Ensure real output by making spectrum conjugate symmetric
    if n_points % 2 == 0:
        comp[0] = np.real(comp[0])
        comp[n_points//2] = np.real(comp[n_points//2])
        comp[1:n_points//2] = comp[1:n_points//2] * 2
        comp[n_points//2+1:] = np.conj(comp[n_points//2-1:0:-1])
    else:
        comp[0] = np.real(comp[0])
        comp[1:(n_points+1)//2] = comp[1:(n_points+1)//2] * 2
        comp[(n_points+1)//2:] = np.conj(comp[(n_points-1)//2:0:-1])
        
    h = np.fft.ifft(comp).real
    # Scale to target sum
    h = project_h(h, n_points)
    return h

    else:
        # Initialize using sieve pattern
        h = fourier_initialization(n_points)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    tabu_decay = 0.95
    max_tabu_age = 100
    perturb_scale = 0.2
    max_stagnation = 50
    best_h = h.copy()
    best_value = evaluate_overlap(h, dx)
    
    # Exploration phase
    iter_count = 0
    stagnation = 0
    
    while time.time() - start_time < budget_s - 10 and iter_count < 2000:
        # Generate neighbor by perturbing with Gaussian noise
        h_new = h + np.random.normal(0, perturb_scale, n_points)
        h_new = project_h(h_new, n_points)
        
        # Check tabu list
        move_key = tuple(np.round(h_new, decimals=4))
        if move_key in tabu_list:
            # Accept with small probability or generate different neighbor
            if np.random.random() > 0.1:
                iter_count += 1
                continue
        
        # Evaluate
        new_value = evaluate_overlap(h_new, dx)
        
        # Update tabu list
        if len(tabu_list) >= tabu_size:
            tabu_list.pop(0)
        tabu_list.append(move_key)

        if new_value < best_value:

            best_value = new_value
            stagnation = 0
