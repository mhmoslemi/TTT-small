import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Gradient descent parameters
    learning_rate = 1e-4
    max_iter = 100000
    progress_count = 0
    max_progress = 200
    perturbation_magnitude = 0.05
    perturbation_frequency = 100
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient with respect to the maximum overlap
        # First find the index k_max where overlap is maximum
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        
        # Gradient computation for that specific k_max
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Approximate gradient: derivative of correlation term
            grad[j] = (1 - current_h[(j - k_max) % n_points]) * dx
        
        # Apply gradient descent
        h_candidate = current_h - learning_rate * grad
        
        # Apply Structured Perturbations to escape local minima
        if i % perturbation_frequency == 0 and i > 0:
            # Pick indices with small gradient to perturb
            abs_grad = np.abs(grad)
            # Pick indices where gradient is below threshold
            threshold = np.percentile(abs_grad, 30) if np.any(abs_grad > 0) else 1e-6
            weak_indices = np.where(abs_grad < threshold)[0]
            
            # Perturb selected indices
            if len(weak_indices) > 0:
                num_perturb = min(len(weak_indices), max(1, n_points // 10))
                indices_to_perturb = np.random.choice(weak_indices, num_perturb, replace=False)
                for idx in indices_to_perturb:
                    delta = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
                    h_candidate[idx] += delta
        
        # Repair the candidate
        h_candidate = repair_h(h_candidate, n_points)
        candidate_c5 = evaluate_h(h_candidate, dx)
        
        # Accept or reject the candidate
        if candidate_c5 < current_c5:
            current_h = h_candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                progress_count = 0
            else:
                progress_count += 1
        else:
            progress_count += 1
            
        # Adaptive adjustments
        if progress_count > max_progress:
            learning_rate *= 0.99
            perturbation_magnitude *= 0.999
            progress_count = 0
            
        # Decrease learning rate and perturbation
        if i % 100 == 0 and i > 0:
            learning_rate *= 0.99
            perturbation_magnitude *= 0.999
            
    return (best_h.tolist(), best_c5, n_points)
