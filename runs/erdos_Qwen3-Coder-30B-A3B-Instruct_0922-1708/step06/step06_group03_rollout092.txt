import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of C5 with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        # j = (i - k_max) mod n
        j = (i - k_max) % n
        grad[i] = 1 - h[j]
    return grad * dx

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    max_steps_without_improvement = 500
    step_count = 0
    steps_without_improvement = 0
    
    while time.time() - start_time < budget_s - 5:
        # Compute overlap values and find maximum
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient
        grad = compute_gradient(current_h, dx, k_max)
        
        # Update with gradient descent
        next_h = current_h - learning_rate * grad
        next_h = repair_h(next_h, n_points)
        
        # Evaluate new solution
        next_c5 = evaluate_h(next_h, dx)
        
        # Accept or reject the update
        if next_c5 < current_c5 or np.random.rand() < 0.1:
            current_h = next_h
            current_c5 = next_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                steps_without_improvement = 0
            else:
                steps_without_improvement += 1
        else:
            steps_without_improvement += 1
        
        # Apply perturbation occasionally
        if steps_without_improvement > max_steps_without_improvement:
            perturbation_magnitude *= 0.999
            perturb_indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
            for i in perturb_indices:
                delta = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
                current_h[i] += delta
            current_h = repair_h(current_h, n_points)
            current_c5 = evaluate_h(current_h, dx)
            steps_without_improvement = 0
            # Reset learning rate occasionally
            if step_count % 1000 == 0:
                learning_rate *= 0.99
        
        step_count += 1
        # Reduce learning rate over time
        if step_count % 100 == 0:
            learning_rate *= 0.99
        # Reduce perturbation magnitude
        if step_count % 500 == 0:
            perturbation_magnitude *= 0.999
    
    return (best_h.tolist(), best_c5, n_points)
