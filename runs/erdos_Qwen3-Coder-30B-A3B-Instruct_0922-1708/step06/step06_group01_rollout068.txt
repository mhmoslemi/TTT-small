def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):

    n = len(h)

        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = repair_h(h_plus, n)
        h_minus = repair_h(h_minus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        c5_minus = np.max(corr_minus) * dx
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def halton_sequence(n, base=2):
    """Generate Halton sequence."""
    sequence = []
    for i in range(n):
        x = 0
        f = 1.0 / base
        j = i
        while j > 0:
            x += (j % base) * f
            j //= base
            f /= base
        sequence.append(x)
    return np.array(sequence)

def sobol_sequence(n):
    """Generate Sobol sequence."""
    # Simplified version - not a true Sobol
    # In practice, more complex generation would be used
    return np.random.rand(n)

    np.random.seed(seed)

        seq = halton_sequence(n_points, base=2)
    elif method == 'sobol':
        seq = sobol_sequence(n_points)
    else:
        seq = np.random.rand(n_points)
    
    # Scale to [0,1] and then to satisfy integral constraint
    h = seq / np.max(seq) if np.max(seq) > 0 else np.ones(n_points)
    h = repair_h(h, n_points)
    return h
