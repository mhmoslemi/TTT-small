import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a sieve pattern in frequency domain
    freq = np.fft.fftfreq(n_points, 1.0/n_points)
    # High frequencies get lower amplitudes (like sieve)
    amp = np.exp(-0.1 * np.abs(freq)**1.5)
    amp[0] = 0  # DC component
    # Create complex signal with random phases
    phase = np.random.uniform(0, 2*np.pi, n_points)
    signal = amp * np.exp(1j * phase)
    # Inverse transform
    h = np.real(np.fft.ifft(signal))
    # Ensure real-valued and scale to unit sum
    h = h - h.min()
    h = h / np.sum(h) * (n_points / 2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialization
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = fourier_initialization(n_points)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    max_iter = 2000
    perturb_scale = 0.2
    diversity_threshold = 50
    
    # Start with initial solution
    current_h = h_init.copy()
    current_h = project_h(current_h, n_points)
    current_c5 = evaluate_overlap(current_h, dx)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Generate candidate by perturbing current solution
        candidate_h = current_h + np.random.normal(0, perturb_scale, n_points)
        candidate_h = project_h(candidate_h, n_points)
        
        # Check tabu list
        move = tuple(np.argsort(np.abs(candidate_h - current_h))[:n_points//4])
        if move in tabu_list:
            # Accept with low probability anyway
            if np.random.rand() < 0.1:
                pass
            else:
                # Try a different perturbation
                candidate_h = current_h + np.random.normal(0, perturb_scale, n_points) * 0.1
                candidate_h = project_h(candidate_h, n_points)
        
        # Evaluate
        candidate_c5 = evaluate_overlap(candidate_h, dx)
        
        # Update tabu list
        if len(tabu_list) >= tabu_size:
            tabu_list.pop(0)
        tabu_list.append(move)
        
        # Accept if better or accept worse with some probability
        if candidate_c5 < current_c5:
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                last_improvement = iter_count
        elif np.random.rand() < 0.01:
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
        
        # Inject diversity if stagnated
        if iter_count - last_improvement > diversity_threshold:
            current_h = fourier_initialization(n_points)
            current_h = project_h(current_h, n_points)
            current_c5 = evaluate_overlap(current_h, dx)
            last_improvement = iter_count
            # Reset tabu list
            tabu_list = []
        
        iter_count += 1
    
    # Gradient descent refinement
    h_refined = best_h.copy()
    prev_c5 = best_c5
    
    # For fine-tuning, we do fewer iterations
    for i in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_c5) / eps
        
        h_refined = h_refined - 0.0001 * grad
        h_refined = project_h(h_refined, n_points)
        prev_c5 = evaluate_overlap(h_refined, dx)
        
        if prev_c5 < best_c5:
            best_h = h_refined.copy()
            best_c5 = prev_c5
    
    return (best_h.tolist(), best_c5, n_points)
