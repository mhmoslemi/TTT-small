import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Cutting-plane method
    max_iter = 200
    solver = 'ECOS'  # Efficient for small problems
    
    # For each major iteration, we solve a subproblem 
    # to minimize the maximum overlap with a constraint
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 10:
            break
            
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        t = cp.Variable(1)
        
        # Create constraints for all shifts
        constraints = [cp.sum(h_var) == n_points / 2,
                       h_var >= 0,
                       h_var <= 1,
                       t >= 0]
        
        # The overlap constraint for k_max
        # This recalculates overlap_k_max using matrix approach 
        # For shift k_max, the overlap is sum of h[i] * (1 - h[i-k_max])
        # We make this quadratic in t
        
        # We'll manually set a quadratic constraint ensuring  
        # overlap_k_max <= t
        k = k_max - (n_points - 1)
        # Extract appropriate indices for shifted correlation
        shift = k
        if k >= 0:
            h_forward = h_var[k:]
            h_backward = h[:-k]
        else:  # k < 0
            h_forward = h_var[:k]
            h_backward = h[-k:]
        
        # The overlap for this shift (approximation)
        # Since we are using correlation (i.e., how much h[i] penalizes h[i+k])
        # Create an explicit overlap expression
        if k >= 0:
            h_shift = np.concatenate([np.zeros(k), h[:-k]])
        else:
            h_shift = np.concatenate([h[-k:], np.zeros(-k)])
        
        # Compute overlap with current h and 1-h shift
        corr_expr = h_var @ (1 - h_shift)  # scalar prod of h and 1-h_shift
        # This corrects for dx (already done in original)
        constraints.append(corr_expr <= t * n_points / 2)
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver, verbose=False, max_iters=300)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                if new_h is not None:
                    h = np.clip(new_h, 0, 1)
                    # Rescale to correct sum
                    h = h * (n_points / 2) / np.sum(h)
                else:
                    # Resort to random perturbation
                    h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            else:
                # If infeasible, attempt to fix some variables
                # Reinitialize from previous best
                h = np.clip(best_h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # In case of solver failure, basic update
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
