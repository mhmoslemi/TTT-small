import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set: [0,1] with sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a triangular pattern
        x = np.linspace(0, 2, n_points)
        h_init = 1 - np.abs(x - 1)
        h_init = project_h(h_init, n_points)
    
    # Coordinate-wise gradient descent with projection
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Parameters
    eta = 0.01
    epsilon = 1e-6
    max_iter = 100000
    patience = 100
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic coordinate updates
        for j in range(n_points):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Finite difference gradient estimation
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            
            h_plus = project_h(h_plus, n_points)
            h_minus = project_h(h_minus, n_points)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update
            h_current[j] = h_current[j] - eta * grad_j
            h_current[j] = max(0, min(1, h_current[j]))
            
        # Apply projection to maintain feasibility
        h_current = project_h(h_current, n_points)
        
        # Evaluate current solution
        c5_current = evaluate_h(h_current, dx)
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            eta *= 0.9
            last_improvement = i
            
    return (h_best.tolist(), c5_best, n_points)
