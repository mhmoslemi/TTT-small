import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured pattern
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Use a triangular pattern as starting point
        for i in range(n_points):
            h_init[i] = max(0, 1 - abs(x[i] - 1))
        h_init = project_h(h_init, n_points)

    # Frank-Wolfe optimization
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    patience = 0
    max_patience = 1000

    for iteration in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient for current solution using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5_current) / eps

        # Solve linear subproblem: min <grad, d> s.t. d in feasible set
        # The feasible set is: 0 <= d_i <= 1, sum(d) = 0
        # So d is optimal when d_i = -grad_i if that doesn't violate constraints
        # But we also need sum(d) = 0. Let's use a combination of gradient and constraint.

        # Initialize candidate direction
        d = -grad
        # Ensure sum(d) = 0 by adjusting it
        d = d - np.mean(d)
        # Clip to [0,1]
        d = np.clip(d, -h_current, 1-h_current)
        # Make sure we don't exceed bounds in the directions
        # Project if necessary to stay feasible
        d_sum = np.sum(d)
        if abs(d_sum) > 1e-10:
            # Adjust d to satisfy sum(d) = 0
            d = d - d_sum / n_points

        # Perform line search for step size eta
        eta_max = 1.0
        eta = eta_max
        while eta > 1e-8:
            h_new = h_current + eta * d
            h_new = project_h(h_new, n_points)
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            if c5_new < c5_current:
                break
            eta *= 0.5

        # If step improves, accept it
        if eta > 1e-8:
            h_current = h_current + eta * d
            h_current = project_h(h_current, n_points)
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                patience = 0
            else:
                patience += 1
        else:
            patience += 1

        if patience > max_patience:
            break

    # Final ensure feasibility and compute actual C5
    h_final = project_h(h_best, n_points)
    corr_final = correlate(h_final, 1 - h_final, mode='full')
    c5_final = np.max(corr_final) * dx

    return (h_final.tolist(), c5_final, n_points)
