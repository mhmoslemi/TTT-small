import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Optimization parameters
    max_iter = 10000
    gamma = 0.01
    beta = 0.9
    tolerance = 1e-6
    
    # For Nesterov acceleration
    v = np.zeros(n_points)
    h_acc = h.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Identify worst shift
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h) - 1)
        
        # Compute gradient for worst shift
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if 0 <= i + k < len(h):
                grad[i] = -dx * (1 - h[i + k])
        
        # Proximal gradient step
        h_new = h - gamma * grad
        
        # Project to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy integral constraint
        h_new_sum = np.sum(h_new)
        if abs(h_new_sum) > 1e-10:
            h_new = h_new * (n_points / 2) / h_new_sum
        
        # Nesterov acceleration
        v = beta * v + (h_new - h)
        h_acc = h_new + beta * v
        
        # Ensure constraint
        h_acc_sum = np.sum(h_acc)
        if abs(h_acc_sum) > 1e-10:
            h = h_acc * (n_points / 2) / h_acc_sum
        else:
            h = h_new
        
        # Early stopping
        if iteration > 100 and abs(c5 - best_c5) < tolerance:
            break
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
