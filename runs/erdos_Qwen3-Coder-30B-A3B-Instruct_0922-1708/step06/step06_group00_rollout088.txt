import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = (h_init > 0.5).astype(int)
    else:
        # Create a uniform binary initialization with exactly n_points/2 ones
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # Ensure h has exactly n_points/2 ones
    current_sum = np.sum(h)
    if current_sum != n_points // 2:
        # Move some 1s to 0s or vice versa to match target
        diff = current_sum - n_points // 2
        if diff > 0:
            # Need to convert some 1s to 0s
            ones_indices = np.where(h == 1)[0]
            for i in range(diff):
                h[ones_indices[i]] = 0
        else:
            # Need to convert some 0s to 1s
            zeros_indices = np.where(h == 0)[0]
            for i in range(-diff):
                h[zeros_indices[i]] = 1
    
    # Define MIP problem
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points // 2]
    
    # Compute overlaps for all possible shifts
    max_shift = n_points - 1
    for k in range(-max_shift, max_shift + 1):
        # Find valid indices
        if k >= 0:
            valid_i = list(range(n_points - k))
            valid_j = list(range(k, n_points))
        else:
            valid_i = list(range(-k, n_points))
            valid_j = list(range(n_points + k))
        
        # Ensure valid indices are within bounds
        valid_i = [i for i in valid_i if 0 <= i < n_points]
        valid_j = [j for j in valid_j if 0 <= j < n_points]
        if len(valid_i) == 0 or len(valid_j) == 0:
            continue
            
        # Sum of h[i] * (1 - h[j]) for valid pairs
        overlap_sum = cp.sum(cp.multiply(h_var[valid_i], 1 - h_var[valid_j]))
        constraints.append(overlap_sum <= t * (n_points / 2))
    
    # Formulate problem
    prob = cp.Problem(objective, constraints)
    
    # Set solver and solve
    try:
        prob.solve(solver=cp.CBC, time_limit=budget_s - 10)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value.astype(int)
        else:
            # Fall back to previous best if optimization failed
            h_opt = h
    except Exception:
        # If solver fails, use initial construction
        h_opt = h
    
    # Check final solution
    h_final = h_opt.copy()
    h_final = (h_final > 0.5).astype(int)
    
    # Adjust to ensure sum is exactly n_points/2
    current_sum = np.sum(h_final)
    if current_sum != n_points // 2:
        diff = current_sum - n_points // 2
        if diff > 0:
            # Convert 1s to 0s
            ones_indices = np.where(h_final == 1)[0]
            for i in range(diff):
                h_final[ones_indices[i]] = 0
        else:
            # Convert 0s to 1s
            zeros_indices = np.where(h_final == 0)[0]
            for i in range(-diff):
                h_final[zeros_indices[i]] = 1
    
    # Evaluate final result
    corr = correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_final.tolist(), final_c5, n_points)
