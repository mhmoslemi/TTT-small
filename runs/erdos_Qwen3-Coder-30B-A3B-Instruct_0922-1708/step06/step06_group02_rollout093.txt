import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    """Create offspring via average crossover."""
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1):
    """Apply Gaussian noise mutation."""
    mutated = h.copy()
    indices = np.where(mutated > 1e-6)[0]
    if len(indices) > 0:
        noise_std = 0.1
        noise = np.random.normal(0, noise_std, len(indices))
        mutated[indices] += noise
        mutated = np.clip(mutated, 0, 1)
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    initial_candidates = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        initial_candidates.append(h_init.copy())
    
    while len(initial_candidates) < population_size:
        h = np.random.rand(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        initial_candidates.append(h)
    
    population = [repair_h(h, n_points) for h in initial_candidates]
    
    # Initialize best solution
    best_h = None
    best_value = float('inf')
    
    for h in population:
        value = evaluate_h(h, dx)
        if value < best_value:
            best_value = value
            best_h = h.copy()
    
    # Evolution parameters
    generations = 1000
    elite_size = population_size // 5
    mutation_rate = 0.1
    learning_rate = 0.001
    no_improvement_count = 0
    last_best = best_value
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            value = evaluate_h(h, dx)
            fitness.append(value)
        
        # Sort by fitness and select elite
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Generate new population
        new_population = elite.copy()
        
        # Fill rest with crossover and mutation
        while len(new_population) < population_size:
            # Select two parents randomly from elite
            parent1_idx = np.random.randint(0, elite_size)
            parent2_idx = np.random.randint(0, elite_size)
            
            if parent1_idx != parent2_idx:
                child = crossover(elite[parent1_idx], elite[parent2_idx])
                if np.random.rand() < 0.5:
                    child = mutate(child, mutation_rate)
                child = repair_h(child, n_points)
                new_population.append(child)
            else:
                # If same parent, just mutate
                child = mutate(elite[parent1_idx], mutation_rate)
                child = repair_h(child, n_points)
                new_population.append(child)
        
        population = new_population
        
        # Update best solution
        for h in population:
            value = evaluate_h(h, dx)
            if value < best_value:
                best_value = value
                best_h = h.copy()
        
        # Gradient refinement of top candidate
        top_candidate = population[sorted_indices[0]]
        if np.random.rand() < 0.3:  # 30% chance to refine
            h_refined = top_candidate.copy()
            current_value = fitness[sorted_indices[0]]
            
            # Gradient descent with finite differences
            for _ in range(50):
                if time.time() - start_time > budget_s - 10:
                    break
                eps = 1e-6
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_value) / eps
                
                h_refined = h_refined - learning_rate * grad
                h_refined = repair_h(h_refined, n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                current_value = np.max(corr) * dx
                
                if current_value < best_value:
                    best_value = current_value
                    best_h = h_refined.copy()
        
        # Check for stagnation
        if abs(best_value - last_best) < 1e-8:
            no_improvement_count += 1
            if no_improvement_count > 50:
                break
        else:
            no_improvement_count = 0
        last_best = best_value
    
    return (best_h.tolist(), best_value, n_points)
