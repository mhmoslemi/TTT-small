import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    h = repair_h(h, n_points)
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Hierarchical optimization
    resolutions = [10, 20, 40, 100]
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Interpolate to lower resolution
        if res < n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_coarse = np.interp(x_new, x_orig, h)
        else:
            h_coarse = h
            
        # Optimize coarse levels using gradient descent
        def objective_coarse(h_coarse_flat):
            # Reshape to full
            h_full = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_coarse)), h_coarse_flat)
            h_full = repair_h(h_full, n_points)
            return evaluate_h(h_full, dx)
            
        def gradient_coarse(h_coarse_flat):
            h_full = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_coarse)), h_coarse_flat)
            h_full = repair_h(h_full, n_points)
            eps = 1e-6
            grad = np.zeros_like(h_coarse_flat)
            for i in range(len(h_coarse_flat)):
                h_plus = h_coarse_flat.copy()
                h_plus[i] += eps
                grad[i] = (objective_coarse(h_plus) - objective_coarse(h_coarse_flat)) / eps
            return grad
            
        # Run optimization
        h_coarse_flat = h_coarse.copy()
        bounds = [(0, 1) for _ in range(len(h_coarse_flat))]
        try:
            res = minimize(objective_coarse, h_coarse_flat, method='L-BFGS-B', 
                          bounds=bounds, jac=gradient_coarse, options={'maxiter': 500})
            if res.success:
                h_coarse_flat = res.x
                h_full = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_coarse)), h_coarse_flat)
                h = repair_h(h_full, n_points)
                
                # Check new value
                current_val = evaluate_h(h, dx)
                if current_val < best_value:
                    best_value = current_val
                    best_h = h.copy()
        except:
            pass
            
        h = best_h.copy()
        
    # Final gradient descent
    def objective(h):
        h = repair_h(h, n_points)
        return evaluate_h(h, dx)
        
    def gradient(h):
        h = repair_h(h, n_points)
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            grad[i] = (objective(h_plus) - objective(h)) / eps
        return grad
        
    try:
        res = minimize(objective, best_h, method='L-BFGS-B', 
                      jac=gradient, options={'maxiter': 1000})
        if res.success:
            best_h = res.x
    except:
        pass
        
    best_h = repair_h(best_h, n_points)
    final_value = evaluate_h(best_h, dx)
    
    # Ensure we still meet constraints
    assert abs(np.sum(best_h) - n_points/2) < 1e-6, "Sum constraint violated"
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Bounds constraint violated"
    
    return (best_h.tolist(), final_value, n_points)
