import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = np.maximum(0, 1 - np.abs(x - 1))
        h_init = repair_h(h_init, n_points)
    
    # Initialize optimization parameters
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    eta = 0.01
    epsilon = 1e-6
    max_iter = 100000
    patience = 100
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic coordinate-wise updates
        for j in range(n_points):
            # Compute gradient using finite differences
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            
            h_plus = repair_h(h_plus, n_points)
            h_minus = repair_h(h_minus, n_points)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update coordinate
            h_current[j] = h_current[j] - eta * grad
            h_current[j] = max(0, min(1, h_current[j]))
        
        # Apply feasibility projection 
        h_current = repair_h(h_current, n_points)
        
        # Evaluate new solution
        c5_current = evaluate_h(h_current, dx)
        
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
        
        # Adaptive step size
        if i - last_improvement > patience:
            eta *= 0.95
            if eta < 1e-6:
                eta = 1e-6
            last_improvement = i
            
        # Occasionally add small perturbations
        if i % 500 == 0:
            perturb_rate = 0.05
            num_perturb = max(1, int(perturb_rate * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                delta = np.random.uniform(-0.02, 0.02)
                h_current[idx] += delta
                h_current[idx] = max(0, min(1, h_current[idx]))

    return (h_best.tolist(), c5_best, n_points)
