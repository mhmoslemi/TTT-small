import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def finite_diff_gradient(h, dx, eps=1e-6):
    n = len(h)
    grad = np.zeros(n)
    h_base = h.copy()
    for i in range(n):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        c5_plus = evaluate_h(h_plus, dx)
        
        h_minus = h_base.copy()
        h_minus[i] -= eps
        h_minus = repair_h(h_minus, n)
        c5_minus = evaluate_h(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def trust_region_step(h, dx, grad, H, trust_radius):
    n = len(h)
    
    def tr_subproblem(d):
        return grad.dot(d) + 0.5 * d.dot(H.dot(d))
    
    # Simple trust region solver: just find the projected step
    d = -grad
    d_norm = np.linalg.norm(d)
    if d_norm > trust_radius:
        d = d * trust_radius / d_norm
        
    # Project to feasible set
    h_new = h + d
    h_new = repair_h(h_new, n)
    
    return d, h_new
