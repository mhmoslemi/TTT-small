import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a simple construction
        h = np.ones(n_points) * 0.5
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Gradient descent parameters
    learning_rate = 5e-4
    temperature = 0.1
    dual_update_freq = 10
    stagnation_count = 0
    max_stagnation = 500
    
    for iteration in range(30000):  # Roughly 30k iterations in 600s
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlaps for each lag
        corr = correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        
        # Update dual variables using soft-max
        if iteration % dual_update_freq == 0:
            temp = 1.0 / (temperature + 1e-10)
            dual_weights = np.exp(-overlaps * temp)
            dual_weights = dual_weights / np.sum(dual_weights)
        
        # Compute gradient of Lagrangian w.r.t h
        gradient = np.zeros(n_points)
        # For each point h[i], compute contribution to gradient from all lags
        for i in range(n_points):
            grad_sum = 0.0
            for k in range(-n_points + 1, n_points):  # k from -n+1 to n-1
                if i + k >= 0 and i + k < n_points:
                    grad_sum += dual_weights[k + n_points - 1] * (1 - h[i + k])
            gradient[i] = grad_sum
        
        # Update h
        h = h - learning_rate * gradient
        
        # Repair h to satisfy constraints
        h = repair_h(h, n_points)
        
        # Check current solution
        current_c5 = evaluate_h(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive learning rate and temperature
        if iteration % 500 == 0:
            learning_rate *= 0.99
            temperature *= 0.999
            
        # Fallback to initial if stagnating
        if stagnation_count > max_stagnation:
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                if len(h) != n_points:
                    x_orig = np.linspace(0, 2, len(h))
                    x_target = np.linspace(0, 2, n_points)
                    h = np.interp(x_target, x_orig, h)
                h = h * (n_points / 2) / np.sum(h)
                stagnation_count = 0
            else:
                h = np.ones(n_points) * 0.5
                h = repair_h(h, n_points)
                stagnation_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
