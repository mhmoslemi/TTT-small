import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_gradient(h, k, dx, n_points):
    """Compute gradient of overlap at shift k."""
    grad = np.zeros(n_points)
    # The gradient of h(x) * (1 - h(x + k)) w.r.t. h(x) is:
    # 1 - h(x + k) when x is valid
    # -h(x - k) when x is valid
    
    # For each position calculate the contribution to gradient
    for i in range(n_points):
        # Forward difference: h(x) * (1 - h(x + k)) 
        # Gradient w.r.t. h(x) is 1 - h(x + k)
        if i + k < n_points and i + k >= 0:
            grad[i] -= (1 - h[i + k]) * dx
            
        # Backward difference: h(x) * (1 - h(x - k)) 
        # Gradient w.r.t. h(x) is -h(x - k)
        if i - k >= 0 and i - k < n_points:
            grad[i] -= h[i - k] * dx
            
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5  # Uniform initialization
        
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Optimization parameters
    learning_rate = 0.01
    max_iter = 5000
    patience = 100
    no_improvement_count = 0
    
    iter_count = 0
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 10:
        # Compute all correlations on a coarse grid to find worst k
        shifts = np.linspace(-2, 2, 100)
        
        max_overlap = 0
        worst_k_idx = 0
        
        # Compute overlaps for all shifts
        for k_idx, k in enumerate(shifts):
            # Cross-correlation properly 
            k_int = int(np.round(k / dx))
            if k_int >= 0:
                h_shifted = np.concatenate([h[k_int:], np.zeros(k_int)])
            else:
                h_shifted = np.concatenate([np.zeros(-k_int), h[:k_int]])
                
            corr_val = np.sum(h * (1 - h_shifted)) * dx
            if corr_val > max_overlap:
                max_overlap = corr_val
                worst_k_idx = k_idx
                
        # Get the worst k closest to actual shift
        worst_k = shifts[worst_k_idx]
        k_int = int(np.round(worst_k / dx))
        
        # Compute gradient at worst shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_int < n_points and i + k_int >= 0:
                grad[i] -= (1 - h[i + k_int]) * dx
            if i - k_int >= 0 and i - k_int < n_points:
                grad[i] -= h[i - k_int] * dx
                
        # Update h
        h += learning_rate * grad
        
        # Repair h to satisfy constraints
        h = repair_h(h, n_points) 
        
        # Check for improvement
        current_value = evaluate_h(h, dx)
        if current_value < best_value:
            best_value = current_value
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adapt learning rate
        if no_improvement_count % 50 == 0 and no_improvement_count > 0:
            learning_rate *= 0.95
            
        # Perturbation to escape local optima
        if no_improvement_count % 100 == 0 and no_improvement_count > 0:
            noise = np.random.normal(0, 0.001, n_points)
            h += noise
            h = repair_h(h, n_points)
            current_value = evaluate_h(h, dx)  # Evaluate again
            if current_value < best_value:
                best_value = current_value
                best_h = h.copy()
                
        iter_count += 1
        
        # Early stopping
        if no_improvement_count > patience:
            break
            
    return (best_h.tolist(), best_value, n_points)
