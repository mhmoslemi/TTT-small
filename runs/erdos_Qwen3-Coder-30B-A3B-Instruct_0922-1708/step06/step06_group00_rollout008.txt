import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
        
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-8:
        h = h * (n_points / 2) / h_sum
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Cutting plane approach
    max_iter = 500
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Identify maximum overlap lag
        max_lag_idx = np.argmax(corr)
        k_max = max_lag_idx - (len(h) - 1)
        
        # Formulate and solve convex subproblem
        h_var = cp.Variable(n_points)
        t = cp.Variable(1)
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        
        # Constraints:
        # 1. sum(h) = n_points / 2
        sum_constraint = cp.Equality(cp.sum(h_var), n_points / 2)
        
        # 2. h in [0,1]
        bound_constraints = [h_var >= 0, h_var <= 1]
        
        # 3. t >= overlap_k_max
        # Overlap at shift k_max = 1 - (2/n_points) * sum(h[i] * h[i+k_max])
        if k_max > 0:
            # h[i+k_max] exists for i in [0, n_points-k_max-1]
            overlap_expr = 1 - (2 / n_points) * cp.sum(cp.multiply(h_var[:n_points-k_max], h_var[k_max:]))
        elif k_max < 0:
            # h[i] exists for i in [0, n_points+k_max-1]
            overlap_expr = 1 - (2 / n_points) * cp.sum(cp.multiply(h_var[-k_max:], h_var[:n_points+k_max]))
        else:
            # k_max == 0 case
            overlap_expr = 1 - (2 / n_points) * cp.sum(cp.multiply(h_var, h_var))
        
        overlap_constraint = cp.Constraint(overlap_expr <= t)
        
        # Form the problem and solve
        constraints = [sum_constraint] + bound_constraints + [overlap_constraint]
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new_sum = np.sum(h_new)
                if abs(h_new_sum - n_points / 2) > 1e-8:
                    h_new = h_new * (n_points / 2) / h_new_sum
                h = h_new
            else:
                # If problem failed, try a gradient step
                proj_h = h.copy()
                proj_h = np.clip(proj_h, 0, 1)
                proj_h_sum = np.sum(proj_h)
                if abs(proj_h_sum - n_points / 2) > 1e-8:
                    proj_h = proj_h * (n_points / 2) / proj_h_sum
                h = proj_h
        except:
            # Fall back to simple gradient step
            h = h.copy()
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-8:
                h = h * (n_points / 2) / h_sum
                
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
