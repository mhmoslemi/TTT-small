import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialization with the provided construction or a symmetric variant
    if 'initial_h_values' in globals():

            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create symmetric initialization
        h = np.ones(n_points) * 0.5
        # Add a mild random perturbation
        h += np.random.normal(0, 0.05, n_points)
        h = repair_h(h, n_points)
    
    # Optimization with gradient descent
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.0001
    max_iter = 10000
    patience = 100
    patience_counter = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute numerical gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        base_value = evaluate_h(h, dx)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - base_value) / eps
        
        # Update step with adaptive learning rate
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        
        # Evaluate and update best solution
        current_value = evaluate_h(h, dx)
        if current_value < best_value:
            best_h = h.copy()
            best_value = current_value
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Reduce learning rate if no improvement for a while
        if patience_counter > patience:
            learning_rate *= 0.8
            patience_counter = 0
            
    # Final refinement of the best solution
    h_refined = best_h.copy()
    prev_value = best_value
    
    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        base_value = evaluate_h(h_refined, dx)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - base_value) / eps
        
        h_refined = h_refined - 0.00005 * grad
        h_refined = repair_h(h_refined, n_points)

            best_value = new_value
