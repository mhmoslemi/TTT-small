import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution from parent if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.full(n_points, 0.5)
        h_init[0] = 0.1
        h_init[1] = 0.9
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Add symmetry and disturbance
    h_init = (h_init + h_init[::-1]) / 2
    h_init = h_init + np.random.normal(0, 0.05, n_points)
    h_init = repair_h(h_init, n_points)
    
    best_h = h_init.copy()
    best_value = evaluate_h(h_init, dx)
    
    # Gradient-based optimization
    max_iter = 2000
    lr = 0.0001
    eps = 1e-6
    stagnation_count = 0
    max_stagnation = 100
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient numerically
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
        
        # Update step
        new_h = best_h - lr * grad
        new_h = repair_h(new_h, n_points)
        new_value = evaluate_h(new_h, dx)
        
        if new_value < best_value:
            best_h = new_h.copy()
            best_value = new_value
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Inject diversity if stuck
        if stagnation_count > max_stagnation:
            new_h = np.random.rand(n_points)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_h = (new_h + new_h[::-1]) / 2
            new_h = repair_h(new_h, n_points)
            new_value = evaluate_h(new_h, dx)
            if new_value < best_value:
                best_h = new_h.copy()
                best_value = new_value
            stagnation_count = 0
            
        # Decrease learning rate if necessary
        if iter_count % 200 == 0 and iter_count > 0:
            lr *= 0.9
    
    # Final refinement
    h_refined = best_h.copy()
    prev_value = best_value
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        h_refined = h_refined - 0.0001 * grad
        h_refined = repair_h(h_refined, n_points)
        prev_value = evaluate_h(h_refined, dx)
        
        if prev_value < best_value:
            best_h = h_refined.copy()
            best_value = prev_value
    
    return (best_h.tolist(), best_value, n_points)
