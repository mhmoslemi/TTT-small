import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    # CVXPY Optimization
    h_var = cp.Variable(n_points)
    t = cp.Variable()

    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2,
        t >= 0
    ]
    
    # Lag indices for non-circular convolution
    max_lag = n_points - 1
    for lag in range(-max_lag, max_lag + 1):
        # Compute valid indices
        if lag >= 0:
            valid_indices = np.arange(lag, n_points)
            h_shifted = h_var[:n_points-lag]
        else:
            valid_indices = np.arange(-lag, n_points)
            h_shifted = h_var[-lag:]
        
        if len(valid_indices) > 0:
            # Compute overlap for this lag
            overlap = cp.sum(cp.multiply(h_var[valid_indices], 1 - h_shifted))
            constraints.append(overlap <= t * (n_points / 2))
    
    # Objective
    objective = cp.Minimize(t)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            c5_bound = t.value * dx
        else:
            h_opt = h_init
            c5_bound = evaluate_h(h_init, dx)
    except:
        h_opt = h_init
        c5_bound = evaluate_h(h_init, dx)
    
    # Post-processing: Ensure constraints are satisfied
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    
    # Final evaluation
    final_c5 = evaluate_h(h_opt, dx)
    
    return (h_opt.tolist(), final_c5, n_points)
