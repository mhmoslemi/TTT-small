import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create sieve pattern in frequency domain
    freq = np.fft.fftfreq(n_points, 1/n_points)
    # Higher frequencies should be suppressed to create smoother patterns
    # Initial: low frequencies dominate
    spectrum = np.zeros(n_points)
    # Apply sieve pattern: high frequencies suppressed
    for i in range(n_points):
        if i < n_points // 2:
            spectrum[i] = 1.0 / (1 + i * 0.1)
        else:
            spectrum[i] = 1.0 / (1 + (n_points - i) * 0.1)
    
    # Generate real signal from complex spectrum (Hermitian symmetry)
    spectrum = np.fft.fftshift(spectrum)
    h = np.fft.ifft(spectrum).real
    h = np.fft.fftshift(h)
    
    # Ensure feasibility
    h = project_h(h, n_points)
    return h
