import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():

        if len(h) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Hierarchical optimization
    resolutions = [10, 20, 40, 100]
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        if res < n_points:
            # Interpolate to coarser grid
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_coarse = np.interp(x_new, x_orig, h)
            h_coarse = h_coarse * (res / 2) / np.sum(h_coarse)
        else:
            h_coarse = h.copy()
            
        # Optimize on this resolution using L-BFGS
        def objective(h_coarse_flat):
            h_interp = h_coarse_flat.copy()
            h_interp = h_interp * (res / 2) / np.sum(h_interp)
            return evaluate_h(h_interp, 2.0 / res)
        
        def constraint_func(h_coarse_flat):
            return np.sum(h_coarse_flat) - res / 2
            
        # Flatten for optimization
        h_flat = h_coarse.copy()
        bounds = [(0, 1) for _ in range(len(h_flat))]
        
        # Minimize with bounds
        try:
            result = minimize(objective, h_flat, method='L-BFGS-B', 
                            bounds=bounds, options={'maxiter': 1000})
            if result.success:
                h_coarse = result.x * (res / 2) / np.sum(result.x)
        except:
            pass
            
        # Interpolate back to full resolution if needed
        if res < n_points:
            x_orig = np.linspace(0, 2, res)
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h_coarse)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = h_coarse.copy()
            
    # Intensive local optimization using projected gradient
    max_iter = 10000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        base_val = evaluate_h(h, dx)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            val_plus = evaluate_h(h_plus, dx)
            grad[j] = (val_plus - base_val) / eps
        
        # Update with gradient descent
        step_size = 1e-4
        h_new = h - step_size * grad
        h_new = repair_h(h_new, n_points)
        
        new_val = evaluate_h(h_new, dx)
        if new_val < base_val:
            h = h_new.copy()
        elif np.allclose(h_new, h, atol=1e-10):
            break
            
    # Final evaluation
    final_value = evaluate_h(h, dx)
    
    return (h.tolist(), final_value, n_points)
