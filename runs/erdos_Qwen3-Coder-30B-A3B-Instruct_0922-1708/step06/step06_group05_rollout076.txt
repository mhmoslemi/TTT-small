import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Convex relaxation parameters
    max_iter = 5000
    cvx_tolerance = 1e-6
    lr = 0.001
    
    def cvx_solve(h):
        try:
            h_var = cp.Variable(n_points)
            # Constraint definitions
            constraint1 = cp.constraints.NonNeg(h_var)
            constraint2 = cp.constraints_affine.Equality(h_var @ np.ones(n_points), n_points/2)
            # Convex relaxation - minimize max of all correlation values
            # But we'll do a quadratic proxy
            corr = np.zeros(2*n_points - 1)
            for k in range(2*n_points - 1):
                corr[k] = np.sum(h_var[:min(k+1,n_points)] * (1-h_var[max(0,n_points-k-1):n_points]))
            
            obj = cp.Minimize(cp.sum_squares(corr))
            prob = cp.Problem(obj, [constraint1, constraint2])
            prob.solve(solver=cp.CVXOPT, abstol=cvx_tolerance, reltol=cvx_tolerance)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h_var.value
        except:
            pass
        return None
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Try convex relaxation
        h_cvx = cvx_solve(h)
        if h_cvx is not None:
            h = h_cvx
        else:
            # Fallback to gradient descent with Adam-like updates
            eps = 1e-6
            grad = np.zeros(n_points)
            h_eval = evaluate_h(h, dx)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                grad[i] = (c5_plus - h_eval) / eps
            
            h = h - lr * grad
            h = repair_h(h, n_points)
        
        c5 = evaluate_h(h, dx)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Stochastic restarts
        if iteration % 300 == 0 and iteration > 0:
            if np.random.rand() < 0.3:
                h = np.random.rand(n_points)
                h = repair_h(h, n_points)
                
    # Final refinement using gradient
    for _ in range(500):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        best_h = best_h - 0.0002 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
    return (best_h.tolist(), best_c5, n_points)
