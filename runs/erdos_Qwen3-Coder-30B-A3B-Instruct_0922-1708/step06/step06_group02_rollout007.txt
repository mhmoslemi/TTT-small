import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def interpolate_h(h_coarse, n_points):
    """Interpolate coarse solution to higher resolution."""
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_points)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    return h_fine

def mutate_h(h, mu, sigma):
    """Apply Gaussian mutation to h."""
    noise = np.random.normal(mu, sigma, len(h))
    h_mut = h + noise
    return h_mut

def crossover_h(h1, h2, prob=0.5):
    """Perform uniform crossover between two h vectors."""
    mask = np.random.random(len(h1)) < prob
    h_child = np.where(mask, h1, h2)
    return h_child

def es_optimize(pop_size, n_points, generations, mutation_scale, elite_rate=0.2):
    """Evolutionary Strategy optimization."""
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init.copy()]
    else:
        population = []
    
    # Initialize population
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Evolution
    for gen in range(generations):
        # Evaluate fitness
        fitness = []
        for h in population:
            fit = evaluate_h(repair_h(h, n_points), 2.0 / n_points)
            fitness.append(fit)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Select elite
        elite_count = max(1, int(pop_size * elite_rate))
        elite = population[:elite_count]
        
        # Create new population
        new_population = elite.copy()
        
        # Add children
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = population[np.random.randint(0, pop_size//2)]
            parent2 = population[np.random.randint(0, pop_size//2)]
            
            # Crossover
            child = crossover_h(parent1, parent2)
            
            # Mutation
            child = mutate_h(child, 0, mutation_scale)
            
            # Repair
            child = repair_h(child, n_points)
            
            new_population.append(child)
        
        population = new_population
        
        if gen % 20 == 0:
            best_fit = fitness[0]
            if best_fit < 0.3807:
                break
    
    # Return best solution
    best_h = repair_h(population[0], n_points)
    return best_h

    start_time = time.time()

    resolutions = [10, 20, 40, 100]
    best_h = None
    best_value = float('inf')

    pop_size = 50
    generations = 100

        for _ in range(1000):

                break

            grad = np.zeros(resolution)
            for j in range(resolution):
                h_plus = current_h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, resolution)

    if best_h is None:
        best_h = current_h
        best_value = evaluate_h(best_h, dx)
        
    return (best_h.tolist(), best_value, len(best_h))
