import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints using quadratic programming."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Project onto the feasible set: sum = n_points/2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    # Simple projection: adjust linearly to meet sum constraint
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    
    return h

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
    
    # Gradient-based optimization with adaptive steps
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Parameters
    max_iter = 100000
    adaptive_step_size = 0.01
    perturb_rate = 0.05
    patience = 1000
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient approximation
        eps = 1e-5
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - c5_current) / eps
        
        # Update with gradient descent
        h_new = h_current - adaptive_step_size * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject based on improvement
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
        else:
            # With low probability, accept worse solutions
            if np.random.rand() < 0.01:
                h_current = h_new
                c5_current = c5_new
        
        # Random perturbation
        if np.random.rand() < perturb_rate:
            num_perturb = max(1, int(perturb_rate * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                h_current[idx] += np.random.uniform(-0.1, 0.1)
            h_current = repair_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            
        # Update best if improved
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive step size
        if i - last_improvement > patience:
            adaptive_step_size *= 0.9
            
    return (h_best.tolist(), c5_best, n_points)
