import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or random binary
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with binary pattern
        h = np.zeros(n_points)
        half_size = n_points // 2
        h[:half_size] = 1.0
    
    # Convert to binary and ensure sum constraint
    h = (h > 0.5).astype(float)
    h = h * (n_points / 2) / np.sum(h)
    
    # Tabu search parameters
    max_iter = 50000
    tabu_size = 50
    tabu_list = []
    
    # Annealing parameters
    T = 0.1
    cooling_rate = 0.999
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping a random bit
        candidate_h = h.copy()
        flip_idx = np.random.randint(0, n_points)
        candidate_h[flip_idx] = 1.0 - candidate_h[flip_idx]
        
        # Ensure sum constraint
        current_sum = np.sum(h)
        candidate_sum = current_sum - h[flip_idx] + candidate_h[flip_idx]
        candidate_h = candidate_h * (n_points / 2) / candidate_sum
        
        # Compute overlap
        corr = correlate(candidate_h, 1 - candidate_h, mode='full')
        c5 = np.max(corr) * dx
        
        # Accept or reject based on tabu and simulated annealing
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate_h.copy()
            
        # Tabu search logic
        if c5 < best_c5 or len(tabu_list) < tabu_size:
            h = candidate_h.copy()
            # Add to tabu list
            tabu_list.append(flip_idx)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        else:
            # Accept with probability based on temperature
            delta = c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = candidate_h.copy()
        
        # Update temperature
        T *= cooling_rate
    
    # Final evaluation using continuous approach to make sure we get the right answer
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
