import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Proximal gradient with Nesterov acceleration
    learning_rate = 0.01
    momentum = 0.9
    max_iter = 10000
    
    # For Nesterov acceleration
    v = np.zeros_like(h)
    y = h.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(y, 1 - y, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = y.copy()
        
        # Identify the maximum lag
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(y) - 1)
        
        # Compute gradient for the identified maximizer
        grad = np.zeros_like(y)
        
        # Gradient computation according to the overlap at the identified lag k
        if k >= 0:
            for i in range(len(y) - k):
                grad[i] -= (1 - y[i + k]) * dx
        else:
            for i in range(-k, len(y)):
                grad[i] -= (1 - y[i + k]) * dx
                
        # Update with proximal gradient + Nesterov acceleration
        # Proximal gradient step with learning rate
        y_new = y - learning_rate * grad
        
        # Projection onto [0,1]
        y_new = np.clip(y_new, 0, 1)
        
        # Enforce sum constraint
        h_sum = np.sum(y_new)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            y_new = y_new * target_sum / h_sum
            
        # Nesterov acceleration
        v = momentum * v + (y_new - y)
        y = y_new + momentum * v
        
        # Ensure final y satisfies constraints
        y = np.clip(y, 0, 1)
        h_sum = np.sum(y)
        if abs(h_sum - target_sum) > 1e-8:
            y = y * target_sum / h_sum
        
        # Update learning rate dynamically
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.95
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
