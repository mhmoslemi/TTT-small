import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_value = evaluate_h(best_h, dx)
    
    # Hierarchical optimization approach
    resolutions = [10, 20, 40, 100]
    
    for res in resolutions:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Interpolate to current resolution
        if res == 100:
            h_current = best_h.copy()
        else:
            x_orig = np.linspace(0, 2, len(best_h))
            x_target = np.linspace(0, 2, res)
            h_current = np.interp(x_target, x_orig, best_h)
        
        # Optimization with gradient descent
        def objective(h):
            # Reshape and repair
            h = np.clip(h, 0, 1)
            h = h * (res / 2) / np.sum(h)
            corr = correlate(h, 1 - h, mode='full')
            return -np.max(corr) * (2.0 / res)  # Negative for minimization
        
        def constraint_func(h):
            h = np.clip(h, 0, 1)
            h = h * (res / 2) / np.sum(h)
            return np.sum(h) - res / 2
        
        # Create bounds and constraints
        bounds = [(0, 1) for _ in range(res)]
        cons = {'type': 'eq', 'fun': constraint_func}
        
        # Optimize using scipy
        try:
            result = minimize(objective, h_current, method='SLSQP', bounds=bounds, constraints=cons, 
                            options={'maxiter': 1000, 'ftol': 1e-8})
            if result.success:
                h_current = result.x
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (res / 2) / np.sum(h_current)
        except:
            pass
        
        # Interpolate back to full resolution
        if res < 100:
            x_orig = np.linspace(0, 2, res)
            x_target = np.linspace(0, 2, 100)
            h_temp = np.interp(x_target, x_orig, h_current)
            h_current = h_temp
        
        # Repair and evaluate
        h_current = repair_h(h_current, 100)
        current_value = evaluate_h(h_current, dx)
        
        if current_value < best_value:
            best_value = current_value
            best_h = h_current.copy()

        best_h = best_h - 0.0001 * grad

        best_value = np.max(corr) * dx
    
    return (best_h.tolist(), best_value, 100)
