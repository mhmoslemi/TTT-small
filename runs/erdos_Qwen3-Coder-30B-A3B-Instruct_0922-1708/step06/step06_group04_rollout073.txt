import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a frequency domain pattern with decreasing amplitude
    freq = np.arange(n_points)
    # High frequencies have lower amplitudes (sieve pattern)
    amp = 1.0 / (1.0 + freq * 0.1)
    # Make it real and symmetric (for real IFFT)
    # Use a complex signal with conjugate symmetry for real output
    # High frequency components dominate the early part
    signal = np.zeros(n_points, dtype=complex)
    signal[1:n_points//2] = amp[1:n_points//2] * np.exp(1j * np.random.rand(n_points//2-1) * 2 * np.pi)
    signal[n_points//2+1:] = np.conj(signal[1:n_points//2][::-1])
    signal[0] = np.real(signal[0])
    signal[n_points//2] = np.real(signal[n_points//2])
    
    # Inverse FFT to get real signal
    h = np.real(np.fft.ifft(signal))
    # Normalize and scale
    h = h - np.min(h)
    h = h / np.max(h) if np.max(h) > 0 else h
    return h

    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize solution
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = fourier_initialization(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    best_h = h_init.copy()
    best_value = evaluate_overlap(best_h, dx)
    
    # Tabu search parameters
    max_iter = 2000
    tabu_list = []
    tabu_size = 50
    memory_decay = 0.95
    perturb_magnitude = 0.2
    diversity_threshold = 50
    
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Create new candidate by perturbing best solution
        h_new = best_h + np.random.normal(0, perturb_magnitude, n_points)
        h_new = project_h(h_new, n_points)
        
        # Exclude from tabu if current move was recently made
        move_hash = tuple(np.round(h_new, decimals=6))
        if move_hash in tabu_list:
            iter_count += 1
            continue
            
        # Calculate new value
        new_value = evaluate_overlap(h_new, dx)
        
        # Update best if improved
        if new_value < best_value:
            best_h = h_new.copy()
            best_value = new_value
            last_improvement = iter_count
            # Add to tabu list
            if len(tabu_list) >= tabu_size:
                tabu_list = tabu_list[1:]  # Remove oldest
            tabu_list.append(move_hash)
            
        # Apply memory decay
        if len(tabu_list) > 0:
            for i in range(len(tabu_list)):
                tabu_list[i] = tabu_list[i]  # Placeholder for decay logic
        
        # Inject diversity if stagnated
        if iter_count - last_improvement > diversity_threshold:
            h_new = fourier_initialization(n_points)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_value = evaluate_overlap(h_new, dx)
            
            if new_value < best_value:
                best_h = h_new.copy()
                best_value = new_value
                last_improvement = iter_count
                if len(tabu_list) >= tabu_size:
                    tabu_list = tabu_list[1:]
                tabu_list.append(move_hash)
        
        iter_count += 1
        
    # Final gradient refinement
    h_refined = best_h.copy()
    prev_value = best_value

        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)

            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        h_refined = h_refined - 0.0001 * grad
        h_refined = project_h(h_refined, n_points)
        prev_value = evaluate_overlap(h_refined, dx)
        
        if prev_value < best_value:
            best_h = h_refined.copy()
            best_value = prev_value
    
    return (best_h.tolist(), best_value, n_points)
