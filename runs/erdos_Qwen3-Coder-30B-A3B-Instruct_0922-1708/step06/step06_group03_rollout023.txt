import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a more structured approach
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimizer parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    decay_rate = 0.999
    perturbation_decay = 0.999
    max_iter = 100000
    
    # Track progress to adjust learning rate
    improvement_count = 0
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap and identify k_max
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute grad using finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - max_overlap) / eps
        
        # Apply perturbation to escape local minimum
        if i % 100 == 0 and improvement_count < 10:
            # Perturb indices with small gradient
            abs_grad = np.abs(grad)
            low_grad_indices = np.where(abs_grad < np.percentile(abs_grad, 30))[0]
            if len(low_grad_indices) > 0:
                num_perturb = max(1, len(low_grad_indices) // 10)
                indices = np.random.choice(low_grad_indices, num_perturb, replace=False)
                for idx in indices:
                    current_h[idx] += np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
        
        # Gradient descent step
        current_h = current_h - learning_rate * grad
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            improvement_count = 0
        else:
            improvement_count += 1
            
        # Adjust learning rate
        if improvement_count > 50:
            learning_rate *= 0.99
            improvement_count = 0
        
        # Update perturbation magnitude
        if i % 1000 == 0:
            perturbation_magnitude *= perturbation_decay
    
    # Final refinement with sharper learning rate
    for _ in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - max_overlap) / eps
        best_h = best_h - 1e-5 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
        
        if time.time() - start_time > budget_s - 5:
            break
    
    return (best_h.tolist(), best_c5, n_points)
