import numpy as np
import time

def evaluate_h(h, dx):
    # Compute the maximum overlap integral
    n = len(h)
    max_overlap = 0
    for k in range(n):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        max_overlap = max(max_overlap, overlap)
    return max_overlap * dx

def project_h(h, target_sum):
    # Project h to [0,1] and scale to target sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    # Compute gradient using finite differences
    n = len(h)
    grad = np.zeros(n)
    base_c5 = evaluate_h(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, 50.0)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = 50.0  # since sum(h) * dx = 1, so sum(h) = 1/dx = 50
    
    start_time = time.time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        # Rescale to target sum (important!)
        h = h * (target_sum / np.sum(h))
    else:
        # Reasonable starting point
        h = np.ones(n_points) * 0.5
        h = project_h(h, target_sum)
    
    # Sparsity regularization parameter (small to maintain density, but not too small)
    lambda_reg = 0.01
    
    # Initialize hyperparameters
    alpha = 0.1
    max_iter = 1000
    adaptive_alpha = True
    convergence_count = 0
    last_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    best_c5 = last_c5
    
    # Optimization loop
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Compute gradient
        grad = compute_gradient(h, dx)
        # Add L1 regularization gradient
        l1_grad = lambda_reg * np.sign(h)
        total_grad = grad + l1_grad
        
        # Update with learning rate
        h_new = h - alpha * total_grad
        
        # Project to feasible region
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept if better, or perform backtracking line search
        if new_c5 < last_c5:
            h = h_new
            last_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
            convergence_count = 0
        else:
            # Adaptive learning rate
            if adaptive_alpha:
                if last_c5 == 0 or abs(new_c5 - last_c5) < 1e-10:
                    alpha = min(alpha * 1.2, 1.0)
                else:
                    alpha = max(alpha * 0.8, 1e-4)
            convergence_count += 1
        
        # Early stopping if no improvement
        if convergence_count > 100:
            break
    
    return (best_h.tolist(), best_c5, n_points)
