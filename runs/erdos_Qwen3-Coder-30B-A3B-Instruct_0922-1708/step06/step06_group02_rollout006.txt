import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def create_population(n_points, n_pop):
    """Create initial population."""
    pop = []
    for _ in range(n_pop):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        pop.append(h)
    return pop

def mutate(h, std, n_points):
    """Mutate a solution."""
    noise = np.random.normal(0, std, n_points)
    mutated = h + noise
    return repair_h(mutated, n_points)

def crossover(h1, h2, crossover_rate=0.3):
    """Crossover two solutions."""
    if np.random.rand() > crossover_rate:
        return h1.copy()
    mask = np.random.rand(len(h1)) < 0.5
    child = np.where(mask, h1, h2)
    return child

def select_pop(pop, values, n_select):
    """Select top individuals."""
    indices = np.argsort(values)[:n_select]
    return [pop[i] for i in indices]

def optimize_stage(h_init, n_points, n_gen, mut_std):
    """Optimize a stage with given parameters."""
    population = [h_init.copy()]
    population.extend(create_population(n_points, 49))
    
    best_h = None
    best_value = float('inf')
    
    for gen in range(n_gen):
        values = [evaluate_h(h, 2.0 / n_points) for h in population]
        
        for i, v in enumerate(values):
            if v < best_value:
                best_value = v
                best_h = population[i].copy()
        
        # Selection
        selected = select_pop(population, values, 10)
        
        # Create new population
        new_pop = selected.copy()
        while len(new_pop) < 50:
            # Crossover
            parent1 = selected[np.random.randint(len(selected))]
            parent2 = selected[np.random.randint(len(selected))]
            child = crossover(parent1, parent2)
            
            # Mutate
            child = mutate(child, mut_std, n_points)
            new_pop.append(child)
            
        population = new_pop
    
    return best_h, best_value

    else:
        n_points = 100

        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
    # Stage 1: Coarse optimization (10 points)
    n_coarse = 10
    coarse_h = np.array([np.mean(h_init[i:i + 10]) for i in range(0, 100, 10)])
    coarse_h = coarse_h * (n_coarse / 2) / np.sum(coarse_h)
    
    coarse_h, coarse_value = optimize_stage(coarse_h, n_coarse, 100, 0.1)
    
    # Stage 2: Refinement (20 points)
    h20 = np.interp(np.linspace(0, 2, 20), np.linspace(0, 2, n_coarse), coarse_h)

    best_h = h100.copy()

    eps = 1e-6

            break
