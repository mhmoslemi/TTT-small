import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial guess
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Build CVXPY optimization problem
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2, t >= 0]
    
    # Compute lag indices for valid overlaps
    summary_size = n_points - 1
    for k in range(-summary_size, summary_size + 1):
        # The convolution overlap is sum(h[i] * (1 - h[i-k]) for valid i)
        # We handle the indexing manually (non-circular)
        expr = 0
        for i in range(n_points):
            j = i - k
            if 0 <= j < n_points:
                expr += h[i] * (1 - h[j])
        # Add constraint: expr <= t * dx
        constraints.append(expr <= t * dx)
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Try using a specific solver
    solver = cp.MOSEK
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        # Fallback to ECOS if MOSEK is not available
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            # If all fails, return heuristic solution
            return (h_init.tolist(), evaluate_h(h_init, dx), n_points)
    
    # Get solution
    h_opt = h.value
    c5_bound = t.value * dx
    
    # If DCP error, let's return the best heuristic
    if h_opt is None or np.isnan(c5_bound):
        h_opt = h_init
        c5_bound = evaluate_h(h_init, dx)
    
    return (h_opt.tolist(), c5_bound, n_points)
