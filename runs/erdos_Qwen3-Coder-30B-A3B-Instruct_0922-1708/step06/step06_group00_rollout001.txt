import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50000
    tabu_size = 50
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure the constraint ∫h(x)dx = 1 by rescaling
    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize binary representation and tabu list
    h_binary = (h > 0.5).astype(float)
    tabu_list = []
    
    # Initial evaluation
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    best_h_binary = h_binary.copy()
    
    # Annealing parameters
    temperature = 0.1
    cooling_rate = 0.999
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
        
        # Generate candidate by flipping one bit
        candidate_h_binary = h_binary.copy()
        flip_idx = np.random.randint(0, n_points)
        candidate_h_binary[flip_idx] = 1 - candidate_h_binary[flip_idx]
        
        # Ensure the binary sum constraint
        current_sum = np.sum(h_binary)
        candidate_sum = current_sum - h_binary[flip_idx] + candidate_h_binary[flip_idx]
        
        # Only accept moves that keep sum close to n/2
        if abs(candidate_sum - n_points/2) > 1e-6:
            continue
        
        # Apply the binary updates to continuous h
        candidate_h = candidate_h_binary.astype(float)
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Check if in tabu list
        if tuple(candidate_h_binary) in tabu_list:
            continue
        
        # Evaluate candidate
        corr = correlate(candidate_h, 1 - candidate_h, mode='full')
        candidate_c5 = np.max(corr) * dx
        
        # Acceptance criterion
        if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / temperature):
            h_binary = candidate_h_binary
            h = candidate_h
            current_c5 = candidate_c5
            
            # Update best
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                best_h_binary = h_binary.copy()
        
        # Update tabu list
        tabu_list.append(tuple(h_binary))
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Update temperature
        temperature *= cooling_rate
    
    # Final evaluation with best solution
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
