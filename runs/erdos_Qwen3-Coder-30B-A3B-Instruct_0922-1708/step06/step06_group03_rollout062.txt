import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a more structured approach
        h_init = np.zeros(n_points)
        # Use a smoother pattern to start
        x = np.linspace(0, 2, n_points)
        # Some smooth ramp
        h_init = 0.5 + 0.3 * np.sin(np.pi * x - np.pi/2)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Initialize dual variables for each lag
    dual_vars = np.ones(2 * n_points - 1)  # For lags from -n+1 to n-1
    dual_vars /= np.sum(dual_vars)  # Normalize
    
    # Parameters
    lr = 5e-4
    temp = 0.1
    decay_rate = 0.999
    dual_update_freq = 10
    stagnation_patience = 1000
    patience_counter = 0
    
    for iteration in range(100000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all overlaps
        corr = correlate(current_h, 1 - current_h, mode='full')
        overlaps = corr * dx  # Each overlap value
        
        # Update dual variables using soft-max strategy
        if iteration % dual_update_freq == 0:
            # Compute gradient of dual variables
            # λ_k ∝ exp(-overlaps_k / temp)
            softmax_arg = -overlaps / temp
            # Clamp to avoid overflow
            softmax_arg = np.clip(softmax_arg, -50, 50)
            new_dual = np.exp(softmax_arg)
            # Keep dual_vars normalized
            dual_vars = new_dual / np.sum(new_dual)
            
            # Decrease temperature
            temp *= decay_rate

        # Compute gradient ∂L/∂h[i]
        # Note: Our Lagrangian is:
        # L = sum(λ_k * overlap_k) + η*(sum(h) - n_points/2)
        # So ∂L/∂h[i] = sum_k λ_k * (1 - h[i-k]) if k <= i else 0
        # And also add constraint gradient: η = (sum(h) - n_points/2) / n_points
        
        # Compute delta for (1-h[i-k]) part
        grad = np.zeros(n_points)
        # We need d/dh[i] of each overlap: overlap_k = ∫ h(x)(1-h(x+k))dx
        # Ignore the boundary terms for now
        
        # For each position i
        # Use the fact that due to convolution structure:
        # ∂overlap_k/∂h[i] = (1-h[i-k]) if k <= i and i-k >= 0 else 0
        for i in range(n_points):
            for k in range(-n_points + 1, n_points):
                # index of h[i-k] in the full range of overlap
                idx = i - k + n_points - 1  # Convert lag k to full index
                if 0 <= idx < n_points:
                    # Assuming we have a backward shift model
                    # For the integral h(x)*(1-h(x+k)) dx, the contribution to ∂h[i]/∂h[i]
                    # depends on where the convolution neighbor comes from
                    # This is a simplified approach - perform direct backward computation
                    pass
                # The real gradient computation from convolution is:
                # If we consider overlap_k = ∫ h(x)(1 - h(x+k))dx
                # Then ∂overlap_k/∂h[i] = ∫ 1_{x=i} (1-h(x+k)) dx = (1-h[i+k]) if i+k is valid

        # Re-compute overlaps to get exact gradient approximation through finite difference
        grad = np.zeros_like(current_h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = current_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update h
        current_h -= lr * grad
        current_h = repair_h(current_h, n_points)
        
        # Recompute C5 for current h
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Decrease learning rate occasionally
        if iteration % 500 == 0 and iteration > 0:
            lr *= 0.99
        
        # Early stopping on stagnation
        if patience_counter > stagnation_patience:
            # Fallback to initial solution if no progress
            current_h = h_init.copy()
            current_c5 = evaluate_h(current_h, dx)
            best_h = h_init.copy()
            best_c5 = evaluate_h(best_h, dx)
            patience_counter = 0
            # Reset dual vars
            dual_vars = np.ones(2 * n_points - 1) / (2 * n_points - 1)
    
    # Final refinement in case needed
    h_final = best_h.copy()
    c5_final = best_c5
    
    # Apply more gradient steps to improve the final solution
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
            
        # finite difference for gradient estimation
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5_final) / eps
            
        h_final = h_final - 0.001 * grad
        h_final = repair_h(h_final, n_points)
        corr = correlate(h_final, 1 - h_final, mode='full')
        c5_final = np.max(corr) * dx
        
        if c5_final < c5_final:
            h_final = best_h.copy()
            c5_final = best_c5

    return (h_final.tolist(), c5_final, n_points)
