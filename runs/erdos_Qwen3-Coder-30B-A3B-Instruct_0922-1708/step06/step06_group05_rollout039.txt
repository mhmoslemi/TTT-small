import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    best_h = h.copy()
    best_c5 = evaluate_h(h, 2.0 / n_points)
    
    # Annealing parameters
    T0 = 1.0
    alpha = 0.99
    T = T0
    max_iter = 2000
    iter_per_scale = 500

    lr = 0.01
    lr_decay = 0.1
    decay_steps = 1000

    for iteration in range(max_iter):

        # Cooling schedule
        if iteration > 0 and iteration % 100 == 0:
            T *= alpha
            
        # Perturb using simulated annealing
        h_perturbed = h.copy()
        noise_scale = T
        h_perturbed += np.random.normal(0, noise_scale, n_points)
        h_perturbed = repair_h(h_perturbed, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_perturbed, 2.0 / n_points)
        
        # Accept or reject
