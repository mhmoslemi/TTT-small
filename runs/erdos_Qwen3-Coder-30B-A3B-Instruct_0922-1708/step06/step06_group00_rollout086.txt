import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Convert to binary
        h = (h_init > 0.5).astype(int)
        # Adjust to have sum = n_points / 2
        current_sum = np.sum(h)
        target_sum = n_points // 2
        if current_sum > target_sum:
            # Zero out some 1s
            indices = np.where(h == 1)[0]
            zeros_to_flip = int(current_sum - target_sum)
            for i in range(zeros_to_flip):
                h[indices[i]] = 0
        elif current_sum < target_sum:
            # Flip some 0s to 1s
            indices = np.where(h == 0)[0]
            ones_to_flip = int(target_sum - current_sum)
            for i in range(ones_to_flip):
                h[indices[i]] = 1
    else:
        # Initialize with uniform binary pattern
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # MIP formulation
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points // 2]
    
    # For each shift, add constraint on overlap
    for k in range(-n_points + 1, n_points):
        # Determine valid indices for shift k
        valid_i = []
        valid_j = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_i.append(i)
                valid_j.append(j)
        
        if valid_i:
            # Sum of h[i] * (1 - h[j]) over valid indices
            overlap_expr = cp.sum(cp.multiply(h_var[valid_i], 1 - h_var[valid_j]))
            # Constraint that overlap <= t * (n_points / 2)
            # Since we're using continuous relaxation and the number of terms is fixed,
            # we need to adjust the constraint to account for the scaling properly.
            # The key is to scale the overlap by dx so that it matches the final integral.
            constraints.append(overlap_expr <= t * n_points / 2)
    
    # Construct problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with CBC
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=budget_s - 10)
        # Check if problem was solved successfully
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = np.round(h_var.value).astype(int)
        else:
            # Fallback to original h if optimization failed
            h_solution = h
    except Exception:
        # Fallback to original h if optimization failed
        h_solution = h
    
    # Check that h_solution satisfies constraints
    h_solution = np.clip(h_solution, 0, 1)
    current_sum = np.sum(h_solution)
    target_sum = n_points // 2
    if abs(current_sum - target_sum) > 1e-6:
        # Adjust if necessary
        diff = int(current_sum - target_sum)
        if diff < 0: # we need more ones
            indices = np.where(h_solution == 0)[0]
            for i in range(-diff):
                h_solution[indices[i]] = 1
        else: # we need fewer ones
            indices = np.where(h_solution == 1)[0]
            for i in range(diff):
                h_solution[indices[i]] = 0
    
    # Final computation
    corr = np.correlate(h_solution, 1 - h_solution, mode='full')
    c5 = np.max(corr) * dx
    
    return (h_solution.tolist(), c5, n_points)
