import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize swarm
    pop_size = 100
    swarm = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    for i in range(pop_size):
        if i == 0:
            h = h_init.copy()
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate_h(h, dx))
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Global best tracking
    gb_idx = np.argmin(pbest_values)
    gbest = pbest[gb_idx].copy()
    gbest_value = pbest_values[gb_idx]
    
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - swarm[i]) + 
                            c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Repair constraints
            swarm[i] = repair_h(swarm[i], n_points)
            
            # Evaluate and update pbest
            current_value = evaluate_h(swarm[i], dx)
            if current_value < pbest_values[i]:
                pbest[i] = swarm[i].copy()
                pbest_values[i] = current_value
                
                if current_value < gbest_value:
                    gbest = swarm[i].copy()
                    gbest_value = current_value
                    last_improvement = iter_count
        
        # Check for stagnation
        if iter_count - last_improvement > 50:
            # Inject random particles
            for _ in range(10):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                swarm.append(h)
                velocities.append(np.random.uniform(-0.5, 0.5, n_points))
                pbest.append(h.copy())
                pbest_values.append(evaluate_h(h, dx))
            
            # Keep best particles
            indices = np.argsort(pbest_values)[:pop_size]
            swarm = [swarm[i] for i in indices]
            velocities = [velocities[i] for i in indices]
            pbest = [pbest[i] for i in indices]
            pbest_values = [pbest_values[i] for i in indices]
            
            # Update gbest
            gb_idx = np.argmin(pbest_values)
            gbest = pbest[gb_idx].copy()
            gbest_value = pbest_values[gb_idx]
            
            last_improvement = iter_count
            
        iter_count += 1
    
    # Final gradient refinement
    h_refined = gbest.copy()
    prev_value = gbest_value
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        h_refined = h_refined - 0.0001 * grad
        h_refined = repair_h(h_refined, n_points)
        prev_value = evaluate_h(h_refined, dx)
        
        if prev_value < gbest_value:
            gbest = h_refined.copy()
            gbest_value = prev_value
    
    return (gbest.tolist(), gbest_value, n_points)
