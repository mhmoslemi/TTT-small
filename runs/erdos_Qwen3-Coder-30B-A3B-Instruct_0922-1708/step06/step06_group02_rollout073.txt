import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

    resolutions = [10, 20, 40, 100]
    current_h = h_init.copy()
    
    # Initialize penalty parameters
    lambda_lag = 0.0
    rho = 1.0
    max_iter = 500
    
    # Optimization loop over resolutions
    for res in resolutions:
        if time.time() - start_time > budget_s - 20:

        # Interpolate to current resolution
        if res < len(current_h):
            x_orig = np.linspace(0, 2, len(current_h))
            x_target = np.linspace(0, 2, res)
            h_current = np.interp(x_target, x_orig, current_h)
            h_current = h_current * (res / 2) / np.sum(h_current)
        else:
            h_current = current_h.copy()

        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 20:

            h = h_current.copy()
            value = evaluate_h(h, dx)
            
            # Gradient estimation
            eps = 1e-6
            grad = np.zeros_like(h)
            
            for j in range(len(h)):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, len(h))
                neg_value = evaluate_h(h_plus, dx)
                grad[j] = (neg_value - value) / eps
            
            # Update h using gradient descent
            h_new = h - 0.001 * grad
            h_new = repair_h(h_new, len(h))
            
            # Augmented Lagrangian update
            sum_h = np.sum(h_new)
            constraint_violation = abs(sum_h - (len(h_new) / 2))
            
            if constraint_violation > 1e-4:
                lambda_lag += rho * (sum_h - len(h_new) / 2)
                rho *= 2.0
                
            # Apply Lagrange multiplier adjustment
            lagrangian_grad = grad + lambda_lag
            h_new = h - 0.001 * lagrangian_grad
            h_new = repair_h(h_new, len(h))
            
            h_current = h_new.copy()
            current_value = evaluate_h(h_current, dx)
