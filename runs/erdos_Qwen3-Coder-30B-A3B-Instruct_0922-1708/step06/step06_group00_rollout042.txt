import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Barrier method parameters
    mu = 1.0
    max_iter = 500
    penalty_factor = 1.1
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Define optimization problem with barrier
        h_var = Variable(n_points)
        t = Variable()
        
        # Objective: minimize t with barrier term
        # Project h to feasible region
        h_clipped = np.clip(h, 1e-6, 1 - 1e-6)
        barrier =_mu * sum(-log(h_var[i]) - log(1 - h_var[i]) for i in range(n_points))
        
        # Constraint: sum h = n_points/2
        constraints = [sum(h_var) == n_points / 2]
        
        # Constraint: t >= overlap_k for all k
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h) - 1)
        
        # Capture all overlaps efficiently
        for k_offset in range(1 - n_points, n_points):
            kcorr = correlate(h, 1 - h, mode="full")
            overlap = np.max(kcorr[k_offset + len(h) - 1])
            constraints.append(t >= overlap * dx)
        
        # Add the main constraint
        constraints.append(t >= c5)
        
        # Objective function
        objective = Minimize(t + barrier)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, max_iters=100)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                h = np.clip(h_sol, 0, 1)
                h_sum = np.sum(h)
                if abs(h_sum - n_points / 2) > 1e-6:
                    h = h * (n_points / 2) / h_sum
            else:
                # Fallback if optimization fails
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if abs(h_sum - n_points / 2) > 1e-6:
                    h = h * (n_points / 2) / h_sum
        except:
            # Keep current h if optimization fails
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-6:
                h = h * (n_points / 2) / h_sum
                
        # Increase penalty parameter
        mu *= penalty_factor
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
