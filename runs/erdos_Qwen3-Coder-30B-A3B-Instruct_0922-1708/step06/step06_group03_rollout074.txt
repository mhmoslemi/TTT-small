import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Set up parameters for optimization
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    max_iter = 100000
    improvement_count = 0
    max_no_improvement = 200
    perturbation_frequency = 100
    perturbation_decay = 0.999
    lr_decay = 0.99
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        h = current_h.copy()
        corr = correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        n = len(h)
        
        # Compute gradient components
        grad = np.zeros(n)
        for j in range(n):
            # This corresponds to derivative of the max term w.r.t. h[j]
            # Find all terms in the convolution that involve h[j]
            for k in range(n):
                if (j - k) % n == max_k:
                    grad[j] += (1 - h[k])
        
        # Apply gradient descent with learning rate
        h_new = h - learning_rate * grad
        
        # Apply perturbations periodically to escape local minima
        if i % perturbation_frequency == 0:
            indices_to_perturb = np.where(np.abs(grad) < np.percentile(np.abs(grad), 30))[0]
            if len(indices_to_perturb) > 0:
                for idx in indices_to_perturb:
                    h_new[idx] += np.random.normal(0, perturbation_magnitude)
                perturbation_magnitude *= perturbation_decay
        
        # Repair to satisfy constraints
        h_new = repair_h(h_new, n_points)
        
        # Compute new objective
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept or reject
        if new_c5 <= current_c5:
            current_h = h_new
            current_c5 = new_c5
            improvement_count = 0
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            improvement_count += 1
            
        # Adaptive learning rate and perturbation
        if improvement_count > max_no_improvement:
            learning_rate *= lr_decay
            improvement_count = 0
        learning_rate = max(1e-7, learning_rate)
        
    return (best_h.tolist(), best_c5, n_points)
