import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default initialization
        h = np.full(n_points, 0.5)
        h += np.random.normal(0, 0.1, n_points)
        h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Adaptive gradient descent
    max_iter = 10000
    lr = 0.0001
    tolerance = 1e-6
    patience = 50
    no_improvement_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(h, dx)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        # Update
        h = h - lr * grad
        h = repair_h(h, n_points)

            no_improvement_count = 0
            lr = min(lr * 1.05, 0.001)  # Increase learning rate
        else:
            no_improvement_count += 1
            if no_improvement_count > patience:
                lr *= 0.5  # Reduce learning rate
                no_improvement_count = 0
        
        # Early stopping if target achieved
        if best_value < 0.3807:
            break
        
        # Diversity injection
        if no_improvement_count > 100:
            h = best_h.copy()
            h += np.random.normal(0, 0.05, n_points)
            h = repair_h(h, n_points)
            no_improvement_count = 0
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(best_h, dx)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps

        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
