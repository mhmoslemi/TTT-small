import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max, n_points):
    """Compute gradient of C5 with respect to h."""
    grad = np.zeros(n_points)
    for j in range(n_points):
        # For the maximum overlap term at lag k_max
        # d/dh_j [h_j * (1 - h_{j-k_max})] = 1 - h_{j-k_max} if j-k_max >= 0, else 0
        # But in convolution, it's h_j * (1 - h_{j-k_max}) where k_max is the max lag
        # So we consider the contribution to the maximum overlap term
        k = k_max
        # j-k should be valid index (mod n_points)
        idx = (j - k) % n_points
        grad[j] = 1 - h[idx]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimizer parameters
    lr = 1e-4
    lr_decay = 0.99
    perturb_prob = 0.1
    perturb_mag = 0.05
    perturb_decay = 0.999
    max_iter = 100000
    patience = 100
    best_iter = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Recompute the overlap for all lags
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        current_c5 = max_overlap * dx
        
        # Compute gradient 
        grad = compute_gradient(current_h, dx, k_max, n_points)
        
        # Perturb with given probability
        if np.random.rand() < perturb_prob:
            # Perturb a small amount of indices
            num_perturb = max(1, n_points // 10)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                current_h[idx] += np.random.uniform(-perturb_mag, perturb_mag)
        
        # Update step (gradient descent)  
        current_h = current_h - lr * grad
        
        # Repair solution to satisfy constraints
        current_h = repair_h(current_h, n_points)
        
        # Recompute C5 
        corr = correlate(current_h, 1 - current_h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            best_iter = i
            # Reset patience
            patience = 100
        
        # Reduce learning rate and perturbation magnitude
        if i % 100 == 0:
            lr *= lr_decay
        if i % 500 == 0:
            perturb_mag *= perturb_decay
            
        # Early stopping for no improvement
        if i - best_iter > patience:
            break
    
    # Final refinement step with targeted perturbations
    h_refined = best_h.copy()
    best_c5_refined = best_c5
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb around current solution
        h_perturbed = h_refined.copy()
        perturb_indices = np.random.choice(n_points, n_points//10, replace=False) 
        for idx in perturb_indices:
            h_perturbed[idx] += np.random.normal(0, 0.01)
        h_perturbed = repair_h(h_perturbed, n_points)
        
        corr = correlate(h_perturbed, 1 - h_perturbed, mode='full')
        c5_perturbed = np.max(corr) * dx
        
        if c5_perturbed < best_c5_refined:
            h_refined = h_perturbed.copy() 
            best_c5_refined = c5_perturbed
        
    return (h_refined.tolist(), best_c5_refined, n_points)
