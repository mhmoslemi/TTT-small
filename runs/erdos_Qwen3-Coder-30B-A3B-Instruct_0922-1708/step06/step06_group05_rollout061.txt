import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    # Use scipy.signal.correlate for efficient computation
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Rescale to meet integral constraint exactly
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():

        # Interpolate to match n_points=20
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, 20)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(20)
    
    # Adjust to meet the integral constraint
    h = h * (20 / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, 2.0/20)
    
    # Multi-scale SA parameters
    n_points_list = [20, 40, 80, 200]
    current_scale = 0
    n_points = n_points_list[current_scale]
    dx = 2.0 / n_points
    
    # Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-6
    sa_steps_per_scale = 500
    
    # Learning rate for gradient descent
    lr = 0.01
    lr_decay = 0.1
    lr_decay_interval = 1000
    
    step_count = 0
    
    while time.time() - start_time < budget_s - 2:
        # Perturb h
        h_perturbed = h + np.random.normal(0, T, len(h))
        h_perturbed = repair_h(h_perturbed, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_perturbed, dx)
        
        # Accept with probability
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):
            h = h_perturbed
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                print(f"New best: {best_c5:.6f}")
        
        # Gradient descent refinement periodically
        if step_count % 50 == 0 and n_points >= 40:
            # Compute finite difference gradient
            eps = 1e-5
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                grad[i] = (c5_plus - best_c5) / eps
            h = h - lr * grad
            h = repair_h(h, n_points)
            
            # Update best if improved
            c5_test = evaluate_h(h, dx)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h.copy()
        
        # Adjust learning rate
        if step_count % lr_decay_interval == 0 and step_count > 0:
            lr *= lr_decay
        
        step_count += 1
        
        # Increase resolution every sa_steps_per_scale steps
        if step_count % sa_steps_per_scale == 0 and current_scale < len(n_points_list) - 1:
            current_scale += 1
            n_points = n_points_list[current_scale]
            dx = 2.0 / n_points
            # Interpolate best solution to new grid
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_points)

            h = repair_h(h, n_points)
            best_h = h.copy()
            best_c5 = evaluate_h(h, dx)
            print(f"Resized to {n_points} points, C5 = {best_c5:.6f}")
            
            # Reset temperature
            T = 1.0
            
        # Lower temperature
        T *= alpha
        if T < min_T:
            T = min_T
    
    # Final refinement at highest resolution only
    x_old = np.linspace(0, 2, len(best_h))
    x_new = np.linspace(0, 2, 200)

        h = repair_h(h, 200)
        new_c5 = evaluate_h(h, 2.0/200)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, 200)
