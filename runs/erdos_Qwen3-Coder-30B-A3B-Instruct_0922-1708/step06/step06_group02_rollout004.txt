import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Start with initial values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        h = h_init.copy()
    else:
        # Initialize with a feasible random solution
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Hierarchical optimization: coarse to fine
    coarse_levels = [10, 20, 40, 100]
    for level in coarse_levels:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Create coarse grid
        if level < n_points:
            coarse_h = np.array([np.mean(h[i::n_points//level]) for i in range(n_points//level)])
            # Interpolate to full resolution
            x_coarse = np.linspace(0, 2, len(coarse_h))
            x_full = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_full, x_coarse, coarse_h)
            h = repair_h(h_interp, n_points)
            
        # Local gradient-based optimization
        def objective(h_flat):
            h = h_flat.copy()
            h = h.reshape(-1)
            h = repair_h(h, n_points)
            return evaluate_h(h, dx)
            
        def gradient(h_flat):
            h = h_flat.copy()
            h = h.reshape(-1)
            h = repair_h(h, n_points)
            eps = 1e-6
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - objective(h_flat)) / eps
            return grad.flatten()
            
        # Optimize using L-BFGS
        res = minimize(objective, h.flatten(), method='L-BFGS-B', 
                       jac=gradient, bounds=[(0, 1)]*len(h), 
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            h = res.x.reshape(-1)
            h = repair_h(h, n_points)
            c5_val = evaluate_h(h, dx)
            if c5_val < best_value:
                best_value = c5_val
                best_h = h.copy()
    
    # Final intensive optimization
    def objective_full(h_flat):
        h = h_flat.copy()
        h = h.reshape(-1)
        h = repair_h(h, n_points)
        return evaluate_h(h, dx)
        
    def gradient_full(h_flat):
        h = h_flat.copy()
        h = h.reshape(-1)
        h = repair_h(h, n_points)
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - objective_full(h_flat)) / eps
        return grad.flatten()
    
    # Run more intensive optimization with gradient descent
    h_current = best_h.copy()
    for _ in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = gradient_full(h_current.flatten())
        h_current = h_current - 0.001 * grad.reshape(h_current.shape)
        h_current = repair_h(h_current, n_points)
        c5_val = evaluate_h(h_current, dx)
        if c5_val < best_value:
            best_value = c5_val
            best_h = h_current.copy()
    
    return (best_h.tolist(), best_value, n_points)
