import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.full(n_points, 0.5)
    
    # Add small random perturbation
    h_init += np.random.normal(0, 0.01, n_points)
    h_init = repair_h(h_init, n_points)
    
    best_h = h_init.copy()
    best_value = evaluate_h(h_init, dx)
    
    # Gradient descent parameters
    learning_rate = 0.0005
    max_iter = 10000
    patience = 100
    
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(best_h, dx)
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        # Update step
        new_h = best_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        if new_c5 < best_value:
            best_h = new_h.copy()
            best_value = new_c5
            last_improvement = iter_count
            
        iter_count += 1
        
        # Early stopping if no improvement for long time
        if iter_count - last_improvement > patience:
            # Inject small noise to escape local minima
            noise_level = 0.0001 * (1.0 + 0.01 * np.random.random())
            best_h += np.random.normal(0, noise_level, n_points)
            best_h = repair_h(best_h, n_points)
            best_value = evaluate_h(best_h, dx)
            last_improvement = iter_count
            
        # Adjust learning rate dynamically
        if iter_count % 500 == 0:
            learning_rate *= 0.95
    
    # Final refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(best_h, dx)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        best_h = best_h - 0.00005 * grad
        best_h = repair_h(best_h, n_points)
        best_value = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_value, n_points)
