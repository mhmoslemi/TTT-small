import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set: clip to [0,1] and scale to sum=1/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured pattern: triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = np.maximum(0, 1 - np.abs(x - 1))
        h_init = project_h(h_init, n_points)
    
    # Frank-Wolfe parameters
    max_iter = 20000
    patience = 1000
    last_improvement = 0
    
    # Frank-Wolfe algorithm
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Line search parameters
    eps = 1e-6
    max_line_search_iter = 10
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - c5_current) / eps
        
        # Linear subproblem: min_d <grad, d> s.t. d in feasible set
        # Since feasible set is {d: 0<=d<=1, sum(d)=0}, 
        # we compute a descent direction by taking negative gradient and projecting
        d = -grad
        d = project_h(d, n_points)  # This makes sum(d) = 0, which is a constraint in the subproblem
        # However, we should go further - d should be in P = {d | 0<=d<=1, sum(d)=0}
        # This will do a projection onto the simplex with simple bounds, 
        # ensuring both bounds and the sum constraint
        # Here we do a simpler first-order approach that respects average(sum(d)=0)
        d = d - np.mean(d)  # This makes sum(d) = 0
        d = np.clip(d, -1, 1)  # Clip to [-1,1] to maintain feasibility

        # Line search
        eta = 1.0
        for _ in range(max_line_search_iter):
            h_trial = h_current + eta * d
            h_trial = project_h(h_trial, n_points)
            c5_trial = evaluate_h(h_trial, dx)
            if c5_trial < c5_current:
                h_current = h_trial
                c5_current = c5_trial
                break
            eta *= 0.5
            
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        if i - last_improvement > patience:
            break
    
    # Final verification and return
    final_h = project_h(h_best, n_points)
    final_c5 = evaluate_h(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
