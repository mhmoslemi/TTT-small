import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_to_feasible(h_init, n_points)
    else:
        # Initialize with a structured pattern - triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1 - np.abs(x - 1)  # Triangle centered at 1
        h_init = project_to_feasible(h_init, n_points)
    
    # Frank-Wolfe optimization
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    patience = 0
    max_patience = 1000
    max_iter = 5000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        h_base = h_best.copy()
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_best) / eps
        
        # Linear subproblem: min_d <grad, d> s.t. 0 <= d[i] <= 1, sum(d) = 0
        # We'll use a simple approximate approach:
        # Take direction that decreases gradient as much as possible while maintaining sum=0
        d = -grad
        d = d - np.mean(d)  # Ensure sum of d is 0
        
        # Project d to feasible range [0,1] (this is a simplification)
        d = np.clip(d, -1, 1)
        
        # Perform line search to find good step size
        eta_max = 1.0
        eta = eta_max
        decay = 0.5
        success = False
        while eta > 1e-6:
            h_new = h_best + eta * d
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
                success = True
                patience = 0
                break
            else:
                eta *= decay
        
        if not success:
            patience += 1
        
        # Early termination if we timed out or are not improving
        if patience > max_patience:
            break
            
    # Final evaluation
    c5_final = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), c5_final, n_points)
