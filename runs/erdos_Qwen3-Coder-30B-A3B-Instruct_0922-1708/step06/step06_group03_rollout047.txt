import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimization parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    max_iter = 100000
    patience = 200
    progress_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        
        # Find maximum overlap lag
        corr = correlate(current_h, 1 - current_h, mode='full')
        max_lag = np.argmax(corr)
        
        # Compute gradient component by component
        for j in range(n_points):
            # Small perturbation
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        # Apply gradient descent
        next_h = current_h - learning_rate * grad
        
        # Periodic perturbations for exploration
        if i % 100 == 0 and i > 0:
            # Find indices with small gradient
            low_grad_idx = np.where(np.abs(grad) < 0.01)[0]
            if len(low_grad_idx) > 0:
                perturb_indices = np.random.choice(low_grad_idx, 
                                                  min(len(low_grad_idx), 5),
                                                  replace=False)
                for idx in perturb_indices:
                    pert = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
                    next_h[idx] += pert
        
        next_h = repair_h(next_h, n_points)
        next_c5 = evaluate_h(next_h, dx)
        
        # Adaptive update
        if next_c5 < current_c5:
            current_h = next_h
            current_c5 = next_c5
            progress_count = 0
        else:
            progress_count += 1
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            progress_count = 0
            
        # Adaptive learning rate and perturbation
        if progress_count > patience:
            learning_rate *= 0.99
            perturbation_magnitude *= 0.999
            progress_count = 0
            
        # Further cooling if still no progress
        if i % 2000 == 0 and i > 0:
            if progress_count > 100:
                learning_rate *= 0.95
                perturbation_magnitude *= 0.995

    best_c5_final = best_c5

            h_plus = repair_h(h_plus, n_points)

            best_c5 = best_c5_final
