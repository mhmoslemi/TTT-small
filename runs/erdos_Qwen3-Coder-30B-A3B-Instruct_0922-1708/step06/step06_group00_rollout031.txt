import time

        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Binary initial construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0

    max_iter = 100
    best_c5 = float('inf')

    stagnation_count = 0

        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        k_max = max_lag_idx - (len(h) - 1)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Stop if no significant improvement
        if stagnation_count > 5:
            break
            
        # Formulate and solve subproblem
        # Minimize t such that:
        # 1. overlap_k_max <= t
        # 2. sum(h) = n_points/2
        # 3. h[i] in [0,1]
        
        # The overlap_k can be written as:
        # overlap_k = 1 - (2/n_points) * sum(h[i] * h[i+k]) for valid i
        
        # We fix all entries except for one to do a more focused optimization
        # Use a quadratic optimization approach with CVXPY
        
        # We solve a simpler relaxation
        h_old = h.copy()
        
        # Set up variables for optimization
        h_var = cp.Variable(n_points)

        # Constraints:
        # overlap_k_max <= t
        # sum(h) = n_points / 2
        # 0 <= h[i] <= 1
        
        # For this optimization, we will approximate by using a particular linearization
        # In principle we want to minimize the max overlap at shift k_max
        
        # More direct approach: we use an augmented lagrangian style approach
        # Similar to a local optimization around the worst case
        # We'll now try just adding gradient-descent-based update on h
        # This is a bit simpler and faster
        
        # Generate the gradient based on maximizing overlap at k_max
        nueric_grad = np.zeros_like(h)
        dx_h = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += dx_h
            if h_plus[i] > 1:
                h_plus[i] = 1.0
                
            # Correct the sum
            h_plus_sum = np.sum(h_plus)
            target_sum = n_points / 2
            h_plus = h_plus * target_sum / h_plus_sum
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            nueric_grad[i] = (c5_plus - c5) / dx_h
        
        # Gradient descent step
        step_size = 0.01
        h = h - step_size * nueric_grad
        
        # Project h back to valid region
        h = np.clip(h, 0, 1)
        # Rescale to correct sum
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2) > 1e-10:
            h = h * (n_points / 2) / h_sum
            
        # Occasionally enforce exact constraints
        if iteration % 10 == 0:

            h_sum = np.sum(h)
            if abs(h_sum - n_points/2) > 1e-8:
                h = h * (n_points / 2) / h_sum
                
    # Final evaluation of the best
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
