import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, target_sum=50.0):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from existing solution or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Add small Gaussian noise to break symmetry
    h += np.random.randn(n_points) * 0.05
    h = project_h(h)
    
    # Sparse optimization parameters
    lambda_reg = 0.01
    alpha = 0.1
    max_iter = 1000
    patience = 100
    best_value = evaluate_h(h, dx)
    best_h = h.copy()
    stagnation_counter = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient numerically
        eps = 1e-6
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Add L1 regularization gradient
        l1_grad = lambda_reg * np.sign(h)
        grad += l1_grad
        
        # Update with backtracking line search
        h_new = h - alpha * grad
        h_new = project_h(h_new)
        new_c5 = evaluate_h(h_new, dx)
        
        # Backtracking
        backtracking_alpha = alpha
        while new_c5 >= current_c5 and backtracking_alpha > 1e-6:
            backtracking_alpha *= 0.5
            h_new = h - backtracking_alpha * grad
            h_new = project_h(h_new)
            new_c5 = evaluate_h(h_new, dx)
        
        # Accept improvement
        if new_c5 < current_c5:
            h = h_new
            alpha *= 0.99  # Adapt learning rate
        else:
            alpha *= 1.1  # Increase learning rate
        
        alpha = np.clip(alpha, 1e-4, 1.0) 
        
        if new_c5 < best_value:
            best_h = h.copy()
            best_value = new_c5
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        if stagnation_counter > patience:
            # Inject perturbation
            h += np.random.randn(n_points) * 0.02
            h = project_h(h)
            stagnation_counter = 0
    
    return (best_h.tolist(), best_value, n_points)
