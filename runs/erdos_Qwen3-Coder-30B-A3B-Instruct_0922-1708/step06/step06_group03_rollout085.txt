import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent parameters
    lambda_vals = np.ones(2 * n_points - 1)
    lambda_vals /= np.sum(lambda_vals)
    tau = 0.1
    learning_rate = 5e-4
    dual_update_freq = 10
    stagnation_count = 0
    max_stagnation = 100
    
    for i in range(100000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all overlaps
        corr = correlate(current_h, 1 - current_h, mode='full')
        overlaps = corr * dx
        
        # Update dual variables via variational inference
        if i % dual_update_freq == 0:
            lambda_vals = np.exp(-overlaps / tau)
            lambda_vals /= np.sum(lambda_vals)
            tau *= 0.999
            
        # Compute gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            for k in range(2 * n_points - 1):
                if k < n_points:
                    idx = j - k
                    if 0 <= idx < n_points:
                        grad[j] += lambda_vals[k] * (1 - current_h[idx])
                else:
                    idx = j - (k - (n_points - 1))
                    if 0 <= idx < n_points:
                        grad[j] += lambda_vals[k] * (1 - current_h[idx])
        
        # Update h
        current_h -= learning_rate * grad
        current_h = repair_h(current_h, n_points)
        
        # Evaluate new solution
        current_c5 = evaluate_h(current_h, dx)
        
        # Track best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Decrease learning rate over time
        if i % 500 == 0:
            learning_rate *= 0.99
            
        # Early exit if stagnation
        if stagnation_count > max_stagnation:
            break

    return (best_h.tolist(), best_c5, n_points)
