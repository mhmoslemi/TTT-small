import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    dx = 2.0 / n_points

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Augmented Lagrangian parameters
    lambda_ = 0.0
    rho = 1.0
    max_iter = 10000
    tolerance = 1e-8
    patience = 100
    
    # Coarse optimization levels
    resolutions = [10, 20, 40, 100]
    current_res = 0
    
    # Coarse optimization
    while current_res < len(resolutions) and time.time() - start_time < budget_s - 10:
        res = resolutions[current_res]
        dx_coarse = 2.0 / res
        
        # Initialize coarse grid
        if res == 10:
            h_coarse = np.ones(res) * 0.5
        else:
            # Interpolate from finer grid to coarser
            x_fine = np.linspace(0, 2, n_points)
            x_coarse = np.linspace(0, 2, res)
            h_coarse = np.interp(x_coarse, x_fine, h)
        
        # Ensure sum constraint
        h_coarse = h_coarse * (res / 2) / np.sum(h_coarse)
        
        # Local optimization on coarse grid
        for iter_count in range(500):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Evaluate constraint and gradient
            current_sum = np.sum(h_coarse)
            constraint_violation = abs(current_sum - res / 2)
            
            # Update Lagrange multiplier
            lambda_ += rho * (current_sum - res / 2)
            
            # Compute gradient of objective with respect to h
            grad = np.zeros(res)
            
            # Compute each component of gradient
            for i in range(res):
                h_plus = h_coarse.copy()
                h_plus[i] += 1e-6
                h_plus = repair_h(h_plus, res)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx_coarse
                
                h_minus = h_coarse.copy()
                h_minus[i] -= 1e-6
                h_minus = repair_h(h_minus, res)
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx_coarse
                
                grad[i] = (c5_plus - c5_minus) / (2e-6)
            
            # Lagrangian gradient update
            lagrangian_grad = grad + lambda_
            
            # Update step size with backtracking
            alpha = 0.01
            while True:
                h_new = h_coarse - alpha * lagrangian_grad
                h_new = repair_h(h_new, res)
                corr_new = correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr_new) * dx_coarse

        h = h - 0.0001 * grad
        h = repair_h(h, n_points)
        
        # Evaluate new value
        current_value = evaluate_h(h, dx)
        
        if current_value < best_value:
            best_h = h.copy()
            best_value = current_value
            patience_count = 0
        else:
            patience_count += 1
            
        if patience_count > 100:
            break
            
        # Early exit if happened recently
        if abs(current_value - last_best) < 1e-8:
            patience_count += 1
        else:
            patience_count = 0
        last_best = current_value
    
    return (best_h.tolist(), best_value, n_points)
