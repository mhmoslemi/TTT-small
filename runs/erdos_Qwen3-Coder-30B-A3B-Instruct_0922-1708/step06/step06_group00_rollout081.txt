import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Enforce constraint sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Iterative cutting plane approach
    max_iter = 500
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the maximum overlap lag
        max_lag_idx = np.argmax(corr)
        k_max = max_lag_idx - (len(h) - 1)
        
        # Define convex subproblem
        h_vars = cp.Variable(n_points)
        t = cp.Variable(1)
        
        # Objective
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add constraint for worst shift
        # The overlap at k_max is: 1 - (2/n_points) * sum(h[i] * h[i+k_max])
        if k_max < 0:
            left = h_vars[:-k_max]
            right = h_vars[-k_max:]
            overlap_constraint = 1 - (2 / n_points) * cp.sum(cp.multiply(left, right)) <= t
        elif k_max == 0:
            overlap_constraint = 1 - (2 / n_points) * cp.sum(cp.multiply(h_vars, h_vars)) <= t
        else:
            left = h_vars[k_max:]
            right = h_vars[:-k_max]
            overlap_constraint = 1 - (2 / n_points) * cp.sum(cp.multiply(left, right)) <= t
        
        constraints.append(overlap_constraint)
        
        # Formulate and solve problem
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                h = np.clip(h_new, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # If solving fails, try gradient step with finite differences
                step_size = 0.001
                grad = np.zeros_like(h)
                for i in range(len(h)):
                    h_plus = h.copy()
                    h_plus[i] += 1e-5
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5) / 1e-5
                
                h = h - step_size * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
        except:
            # If optimization fails, fallback to gradient descent
            step_size = 0.001
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / 1e-5
            
            h = h - step_size * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
