import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def solve_convex_relaxation(h, n_points, dx):
    # Use convex relaxation to get a better starting point
    try:
        # Formulate as convex optimization
        h_var = cp.Variable(n_points)
        # Minimize sum of squared correlations (relaxed version)
        # Note: We still want to minimize max correlation, so we use a proxy
        # We'll minimize the sum of squares of all correlations
        corr_vars = []
        for k in range(2 * n_points - 1):
            corr_val = cp.sum(cp.multiply(h_var, cp.roll(1 - h_var, k - n_points + 1)))
            corr_vars.append(corr_val)
        
        # Minimize a proxy for max correlation via sum of squares
        objective = cp.Minimize(cp.sum(cp.square(corr_vars)))
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solver settings
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-6, reltol=1e-6, abstol=1e-6)
        
        if prob.status == 'optimal':
            return np.array(h_var.value)
    except Exception:
        pass
    return h

def evaluate_h(h, dx):
    # Fast evaluate using correlation
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    # Repair to satisfy constraints
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-8:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use larger resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Try convex relaxation first
    if time.time() - start_time < budget_s - 10:
        refined_h = solve_convex_relaxation(h, n_points, dx)
        if refined_h is not None:
            refined_h = repair_h(refined_h, n_points)
            c5_refined = evaluate_h(refined_h, dx)
            if c5_refined < best_c5:
                best_h = refined_h
                best_c5 = c5_refined
    
    # Main optimization loop
    lr = 0.001
    iter_count = 0
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        
        # Use Adam-like approach with gradient descent
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        
        # Update with gradient
        best_h = best_h - lr * grad
        best_h = repair_h(best_h, n_points)
        
        # Check convergence
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adaptive learning rate
        if stagnation_count > max_stagnation // 2:
            lr *= 0.95
            stagnation_count = 0
            
        # Occasionally perturb
        if iter_count % 50 == 0 and time.time() - start_time < budget_s - 10:
            # Add small noise to some elements
            indices = np.random.choice(n_points, size=max(5, n_points // 20), replace=False)
            best_h[indices] += np.random.normal(0, 0.005, len(indices))
            best_h = repair_h(best_h, n_points)
            new_c5 = evaluate_h(best_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
        
        # Project integral constraint every 200 iterations
        if iter_count % 200 == 0:
            best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final fine-tuning
    for _ in range(500):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
