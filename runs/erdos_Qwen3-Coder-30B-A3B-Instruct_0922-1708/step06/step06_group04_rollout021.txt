import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a sieve-like pattern in frequency domain
    freq = np.arange(n_points)
    # Inverse sieve: emphasize high frequencies
    spectrum = 1.0 / (1.0 + freq * 0.1)
    # Ensure DC component is 0 for net zero mean
    spectrum[0] = 0
    # Apply inverse FFT to get real-valued signal
    h = np.real(np.fft.ifft(spectrum))
    # Normalize and project
    h = project_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        h_init = fourier_initialization(n_points)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    tabu_decay = 0.95
    population_size = 100
    max_iter = 2000
    perturb_size = 0.2
    diversity_inject = 50
    
    # Initialize population
    population = [h_init.copy()]
    fitness = [evaluate_overlap(h_init, dx)]
    
    for _ in range(population_size - 1):
        h = h_init + np.random.normal(0, perturb_size, n_points)
        h = project_h(h, n_points)
        population.append(h)
        fitness.append(evaluate_overlap(h, dx))
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Generate new solutions from existing ones
        new_population = []
        new_fitness = []
        
        for i in range(len(population)):
            h = population[i].copy()
            
            # Apply perturbation - slightly different from previous
            perturb = np.random.normal(0, perturb_size, n_points)
            h_new = h + perturb
            
            # Make sure we don't violate constraint too severely
            h_new = project_h(h_new, n_points)
            
            # Avoid local search if we have a tabu move
            if not any(np.allclose(h_new, tabu_h, atol=1e-3) for tabu_h in tabu_list):
                new_population.append(h_new)
                new_fitness.append(evaluate_overlap(h_new, dx))
        
        # Combine with existing population
        combined_population = population + new_population
        combined_fitness = fitness + new_fitness
        
        # Keep the best ones
        indices = np.argsort(combined_fitness)[:population_size]
        population = [combined_population[i] for i in indices]
        fitness = [combined_fitness[i] for i in indices]
        
        # Update best solution found
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_value:
            best_h = population[current_best_idx].copy()
            best_value = fitness[current_best_idx]
            last_improvement = iter_count
        
        # Update tabu list
        if len(tabu_list) >= tabu_size:
            # Remove first element and decay
            tabu_list = tabu_list[1:]
            tabu_list = [h * tabu_decay for h in tabu_list]

        if iter_count - last_improvement > 50:
            for _ in range(diversity_inject):
                h = fourier_initialization(n_points)
                population.append(h)
                fitness.append(evaluate_overlap(h, dx))
            
            # Keep best ones
            indices = np.argsort(fitness)[:population_size]
            population = [population[i] for i in indices]
            fitness = [fitness[i] for i in indices]
            last_improvement = iter_count
        
        iter_count += 1
    
    # Gradient descent refinement
    h_refined = best_h.copy()
    prev_value = best_value
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        h_refined = h_refined - 0.0001 * grad
        h_refined = project_h(h_refined, n_points)
        prev_value = evaluate_overlap(h_refined, dx)
        
        if prev_value < best_value:
            best_h = h_refined.copy()
            best_value = prev_value
    
    return (best_h.tolist(), best_value, n_points)
