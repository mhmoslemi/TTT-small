import numpy as np
import time
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def halton_sequence(n, base=2):
    """Generate Halton sequence for low-discrepancy sampling."""
    return qmc.Halton(d=1, base=base).random(n)[:,0]

def sobol_sequence(n):
    """Generate Sobol sequence for low-discrepancy sampling."""
    return qmc.Sobol(d=1, scramble=True).random(n)[:,0]

def get_initial_h(n_points, seed=42):
    """Generate initial h using low-discrepancy sequences."""
    np.random.seed(seed)
    
    # Try different sequences
    sequences = [
        halton_sequence(n_points, 2),
        halton_sequence(n_points, 3),
        sobol_sequence(n_points),
        np.linspace(0, 1, n_points)  # Uniform for fallback
    ]
    
    best_h = None
    best_c5 = float('inf')
    
    # For each sequence, construct a step function and evaluate
    for i, seq in enumerate(sequences):
        # Map to [0, 2]
        x = seq * 2
        
        # Create step function from sampled points
        h = np.zeros(n_points)
        for j in range(n_points):
            # Assign to nearest grid point
            idx = int(np.floor(x[j] / (2.0 / n_points)))
            if 0 <= idx < n_points:
                h[idx] += 1.0
                
        # Normalize
        h = repair_h(h, n_points)
        
        # Evaluate
        c5 = evaluate_h(h, 2.0 / n_points)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    if best_h is None:
        # Fallback to random feasible solution
        h = np.random.rand(n_points)
        best_h = repair_h(h, n_points)
        
    return best_h

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        h_init = get_initial_h(n_points, seed)
    
    # Use multiple seeds to avoid local optima
    best_h = h_init.copy()
    best_c5 = evaluate_h(h_init, dx)
    
    # Run multiple explorations with different initializations
    n_seeds = 10
    for seed_i in range(n_seeds):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new initialization
        h_new = get_initial_h(n_points, seed + seed_i)
        
        # Gradient-based refinement
        h_refined = h_new.copy()
        c5_refined = evaluate_h(h_refined, dx)
        
        # Optimization loop
        for iter_i in range(2000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient via centered finite differences
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_refined.copy()
                h_minus[j] -= eps
                h_minus = repair_h(h_minus, n_points)
                c5_minus = evaluate_h(h_minus, dx)
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
                
            # Update with gradient step
            step_size = 0.01 / (1.0 + iter_i / 1000.0)
            h_refined = h_refined - step_size * grad 
            h_refined = repair_h(h_refined, n_points)
            c5_refined = evaluate_h(h_refined, dx)
            
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
                
    return (best_h.tolist(), best_c5, n_points)
