import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, max_k):
    """Compute gradient of the maximum overlap w.r.t. h."""
    n_points = len(h)
    grad = np.zeros(n_points)
    
    # Compute convolution of h and (1-h)
    conv = correlate(h, 1 - h, mode='full')
    
    # Find the index of maximum overlap
    max_idx = np.argmax(conv)
    
    # The maximum overlap corresponds to some k = max_idx - (n_points-1)
    # Instead of computing full gradient for all entries, we discretize 
    # the gradient contribution from the max overlap term.
    # For each h[i], compute how much it contributes to the max overlap
    
    # Very simplified hand-coded derivation approach:
    # Consider the correlation computation.
    # We'll only compute gradient for h[max_k] in the max overlap term.
    # This also assumes we can ignore other indexes for a locally optimizing step,
    # which is reasonable for our update scheme.
    
    # We compute approximate gradient by finite differences.
    # This prevents full gradient computation which is O(n^2) and too expensive.
    
    # Simplified approach:
    # Just get the effect of varying individual h[i] on maximum correlation
    # We do one-sided finite difference approximation for directionality
    eps = 1e-6
    h_base = h.copy()
    max_val = np.max(conv)
    for i in range(n_points):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n_points)
        corr_plus = correlate(h_plus, 1-h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        grad[i] = (c5_plus - max_val) / eps
        
    return grad

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Main optimization loop
    learning_rate = 1e-4
    max_iter = 50000  # Enough steps for convergence
    iterations_without_improvement = 0
    max_no_improvement = 500
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient to decrease the maximum overlap
        grad = compute_gradient(current_h, dx, n_points)
        
        # Take a gradient step
        new_h = current_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        
        # Evaluate the new h
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept new solution if better (or sometimes worse based on learning rate)
        if new_c5 < current_c5 or np.random.rand() < 0.01:
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                iterations_without_improvement = 0
            else:
                iterations_without_improvement += 1
        else:
            iterations_without_improvement += 1
            
        # Adaptive learning rate
        if iterations_without_improvement > 100:
            learning_rate *= 0.99
            iterations_without_improvement = 0
            
        # Early stopping if no improvement for a long time
        if iterations_without_improvement > max_no_improvement:
            break
            
    # Final verification and refinement
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Perform local refinement: gradient decent on the best solution
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Another gradient step
        grad = compute_gradient(current_h, dx, n_points)
        current_h = current_h - 0.0001 * grad
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
    return (best_h.tolist(), best_c5, n_points)
