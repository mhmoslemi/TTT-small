```python
import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # CVXPY optimization
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2,
        t >= 0
    ]
    
    # Compute all overlap constraints
    max_lag = n_points - 1
    for lag in range(-max_lag, max_lag + 1):
        if lag >= 0:
            # Forward overlaps
            indices = np.arange(lag, n_points)
            other_indices = np.arange(0, n_points - lag)
        else:
            # Backward overlaps
            indices = np.arange(0, n_points + lag)
            other_indices = np.arange(-lag, n_points)
        
        if len(indices) > 0 and len(other_indices) > 0:
            # Compute the constraint: sum_{i} h[i] * (1 - h[i - lag]) <= t * dx * n_points / 2
            # But we use larger-scale approach: the constraint is at the level of t
            # So the constraint becomes: sum_{i} h[i] * (1 - h[i+lag]) <= t * (n_points / 2)
            # This is actually an overestimate. For correctness, we generate the correct 
            # convolution several steps ahead.
            
            # The proper bounds:
            # overlap[j] = sum over i where valid(0 to n_points - 1),
            # h[i] * (1 - h[(i - j) % n_points])
            # But since we're using fixed lag and want non-circular version,
            # we compute lag-based overlaps with appropriate index handling.
            
            # For j = 0, 2 sum(h[i] * (1 - h[i]))  -> sum(h) * 1 - sum(h^2)
            # For j=1, sum(h[i] * h[i-1]) -> careful with boundary
            
            # Use full overlap and in-line with numpy use:
            h_lagged = np.roll(h_var, -lag)
            if lag >= 0:
                overlap_sum = cp.sum(cp.multiply(h_var[:n_points-lag], 1 - h_lagged[:n_points-lag]))
            else:
                overlap_sum = cp.sum(cp.multiply(h_var[-lag:], 1 - h_lagged[-lag:]))
            constraints.append(overlap_sum <= t * (n_points / 2))
    
    # Objective function
    objective = cp.Minimize(t)
    
    # Solve problem
    prob = cp.Problem(objective, constraints)
    solver = cp.MOSEK  # or CP.SCS if MOSEK is not available
    
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = h_var.value
            c5_bound = t.value * dx
        else:
            # Fall back to heuristic, might be infeasible
            h_solution = h_init
            c5_bound = evaluate_h(h_solution, dx)
    except Exception:
        # Fallback to initial heuristics
        h_solution = h_init
        c5_bound = evaluate_h(h_solution, dx)
    
    # Final refinement with local optimization
    def eval_func(x):
        return evaluate_h(x, dx)
    
    h_final = h_solution.copy()
    current_c5 = c5_bound
    
    # Try a gradient-based local search
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = eval_func(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
        
        step_size = 1e-3
        h_new = h_final - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        new_c5 = eval_func(h_new)
        
        if new_c5 < current_c5:
            h_final = h_new
            current_c5 = new_c5
        else:
            break
    
    # Verify final solution
    current_c5 = evaluate_h(h_final, dx)
    
    return (h_final.tolist(), current_c5, n_points)
```
