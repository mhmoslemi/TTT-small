import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute gradient of C5 with respect to h using finite differences."""
    n = len(h)
    grad = np.zeros(n)
    base_value = evaluate_h(h, dx)
    
    for j in range(n):
        h_plus = h.copy()
        h_plus[j] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        grad[j] = (c5_plus - base_value) / eps
    return grad

def backtracking_line_search(h, grad, dx, alpha=0.001, beta=0.5, max_iter=10):
    """Perform backtracking line search."""
    h_new = h - alpha * grad
    h_new = repair_h(h_new, len(h))
    new_value = evaluate_h(h_new, dx)
    
    for _ in range(max_iter):
        if new_value < evaluate_h(h, dx):
            return h_new
        alpha *= beta
        h_new = h - alpha * grad
        h_new = repair_h(h_new, len(h))
        new_value = evaluate_h(h_new, dx)
    
    return h_new
