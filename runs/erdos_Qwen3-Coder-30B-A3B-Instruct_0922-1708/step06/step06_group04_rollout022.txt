import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a sieve-like pattern in frequency domain
    freq_domain = np.zeros(n_points)
    # High frequencies with decreasing amplitudes
    for i in range(1, n_points//2):
        freq_domain[i] = 1.0 / (1 + i * 0.5)
        freq_domain[n_points - i] = 1.0 / (1 + i * 0.5)
    # DC component
    freq_domain[0] = 0.5
    # Inverse FFT
    h = np.fft.ifft(freq_domain).real
    # Shift to make smoother
    h = np.roll(h, n_points//4)
    # Ensure it's positive and scaled correctly
    h = project_h(h, n_points)
    return h

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = fourier_initialization(n_points)
    
    # Tabu search
    population_size = 100
    tabu_list_size = 50
    tabu_list = []
    memory_decay = 0.95
    
    # Initialize population
    population = []
    fitness = []
    
    # Add initial solution and random ones
    population.append(h_init.copy())
    fitness.append(evaluate_overlap(h_init, dx))
    
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        population.append(h)
        fitness.append(evaluate_overlap(h, dx))
    
    # Find best in initial population
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    # Tabu search loop
    max_iter = 2000
    perturb_magnitude = 0.2
    
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Collect new candidates
        new_population = []
        new_fitness = []
        
        # Selector: pick top performers
        top_k = min(20, len(population) // 2)
        indices = np.argsort(fitness)[:top_k]
        
        for i in indices:
            h = population[i].copy()
            
            # Perturb with noise
            perturb = np.random.normal(0, perturb_magnitude, n_points)
            h_new = h + perturb
            
            # Repair
            h_new = project_h(h_new, n_points)
            
            # Add to new population
            new_population.append(h_new)
            new_fitness.append(evaluate_overlap(h_new, dx))
        
        # Combine with existing
        combined_population = population + new_population
        combined_fitness = fitness + new_fitness
        
        # Keep best
        indices = np.argsort(combined_fitness)[:population_size]
        population = [combined_population[i] for i in indices]
        fitness = [combined_fitness[i] for i in indices]

        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_value:

    h_refined = best_h.copy()

        eps = 1e-6
        grad = np.zeros(n_points)

            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        # Update
        h_refined = h_refined - learning_rate * grad
        h_refined = project_h(h_refined, n_points)
        prev_value = evaluate_overlap(h_refined, dx)
        
        if prev_value < best_value:
            best_h = h_refined.copy()
            best_value = prev_value
    
    return (best_h.tolist(), best_value, n_points)
