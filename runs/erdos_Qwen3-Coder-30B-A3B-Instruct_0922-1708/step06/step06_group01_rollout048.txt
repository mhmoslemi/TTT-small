import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute finite difference gradient."""
    n_points = len(h)
    grad = np.zeros(n_points)
    h_base = h.copy()
    for i in range(n_points):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n_points)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        
        h_minus = h_base.copy()
        h_minus[i] -= eps
        h_minus = repair_h(h_minus, n_points)
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        c5_minus = np.max(corr_minus) * dx
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def trust_region_step(h, grad, dx, trust_radius=1.0):
    """Find a step within trust region."""
    n_points = len(h)
    # Simple projected gradient descent with trust region
    step = -grad
    step_norm = np.linalg.norm(step)
    if step_norm > trust_radius:
        step = step * trust_radius / step_norm

    # Check bounds and feasibility
    h_new = h + step
    h_new = repair_h(h_new, n_points)
    return h_new

        h_init = repair_h(h_init, n_points)

    # Trust region parameters
    trust_radius = 1.0
    max_iter = 2000
    perturb_every = 250
    min_trust_radius = 1e-6
    max_trust_radius = 1.0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:

        grad = compute_gradient(h_current, dx)

        h_new = trust_region_step(h_current, grad, dx, trust_radius)
