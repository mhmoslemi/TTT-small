import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    if 'initial_h_values' in globals():

        x_orig = np.linspace(0, 2, len(h))

        h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Try convex relaxation first
    try:
        # Define the optimization variables
        h_var = cp.Variable(n_points)
        
        # Objective: minimize the maximum correlation
        # This is a convex relaxation; a better method would be to minimize
        # a quadratic approximation of this objective
        # For now, we'll try to approximate the objective using a least-squares fitting approach
        
        # We will treat the problem as minimizing the Frobenius norm of the shifted correlations
        # This is not the true objective but a reasonable relaxation
        corr_matrix = np.zeros((n_points, n_points))
        for k in range(n_points):
            # Alternatively using FFT for efficiency
            corr_matrix[k, k:] = 1 - h[:n_points-k] if k < n_points else np.zeros(0)
        
        # Just direct optimization through Adam
        pass
    except Exception:
        pass  # Fall back to gradient descent
    
    # Adam optimizer with adaptive learning rate
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.001
    t = 0
    
    # Iterative improvement using Adam
    for iteration in range(50000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Use finite differences for gradient estimation
        eps_val = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps_val
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps_val
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h -= lr * m_hat / (np.sqrt(v_hat) + eps)
        h = repair_h(h, n_points)
        
        # Evaluate new solution
        c5 = evaluate_h(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if iteration % 1000 == 0:
            lr *= 0.995  # Gradual decay
            
    # Refinement with direct gradient descent for final tightening
    for _ in range(10000):
        if time.time() - start_time > budget_s - 2:
            break
        eps_val = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)

            h_plus[i] += eps_val
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps_val
        best_h -= 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
    return (best_h.tolist(), best_c5, n_points)
