import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h = h_init.copy()
    else:
        h = np.ones(n_points) * 0.5
        
    h = repair_h(h, n_points)
    
    # Augmented Lagrangian parameters
    lambda_mult = 0.0
    rho = 1.0
    
    def augmented_lagrangian(h, lambda_mult, rho):
        # Functional value (C5)
        c5_val = evaluate_h(h, dx)
        # Constraint violation
        constraint_violation = np.sum(h) - n_points / 2
        # Augmented Lagrangian value
        al_value = c5_val + lambda_mult * constraint_violation + 0.5 * rho * constraint_violation**2
        return al_value
    
    def compute_gradient(h, dx, lambda_mult, rho):
        # Compute finite difference gradient for each component
        eps = 1e-6
        grad = np.zeros_like(h)
        h_base = h.copy()
        constraint_violation = np.sum(h) - n_points / 2
        
        for i in range(len(h)):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - evaluate_h(h_base, dx)) / eps
            
        # Add constraint gradient: lambda_mult + rho * (sum(h) - 1)
        constraint_grad = np.ones_like(h) * (lambda_mult + rho * constraint_violation)
        
        return grad + constraint_grad
    
    # Coarse to fine optimization loop
    resolutions = [10, 20, 40, 100]
    learning_rate = 0.001
    
    for res in resolutions:
        # Interpolate to current resolution
        if res < n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h = np.interp(x_new, x_orig, h)
            h = repair_h(h, res)
            n_points = res
            dx = 2.0 / n_points
        else:
            # Already at highest resolution
            pass
            
        # Optimization loop for current resolution
        last_c5 = float('inf')
        no_improvement_count = 0
        iter_count = 0
        max_iter = 500
        
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Gradient descent step
            grad = compute_gradient(h, dx, lambda_mult, rho)
            h = h - learning_rate * grad
            h = repair_h(h, n_points)
            
            # Update Lagrange multiplier
            constraint_violation = np.sum(h) - n_points / 2
            lambda_mult = lambda_mult + rho * constraint_violation
            
            # Update penalty parameter if needed
            if abs(constraint_violation) > 1e-4:
                rho = min(rho * 2, 1e5)  # Cap penalty
            else:
                # Slow penalty update when close
                rho = max(rho * 0.99, 1e-5)
                
            # Evaluate function
            current_c5 = evaluate_h(h, dx)
            
            if current_c5 < last_c5:
                last_c5 = current_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            if no_improvement_count > 100:
                break
                
        # Retain the current h as starting point for next resolution
        h = repair_h(h, n_points)
        learning_rate *= 0.8  # Reduce step size for later refinements
    
    # Final intensive refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 2:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - evaluate_h(h, dx)) / eps
            
        h = h - 0.0001 * grad
        h = repair_h(h, n_points)
    
    final_c5 = evaluate_h(h, dx)
    return (h.tolist(), final_c5, n_points)
