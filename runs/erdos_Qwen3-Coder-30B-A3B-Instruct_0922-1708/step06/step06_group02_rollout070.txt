import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_gradient(h, dx):
    """Compute gradient of C5 with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-6
    base_value = evaluate_h(h, dx)
    
    for j in range(n):
        h_plus = h.copy()
        h_plus[j] += eps
        h_plus = repair_h(h_plus, len(h))
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        grad[j] = (c5_plus - base_value) / eps
    
    return grad

def backtrack_line_search(h, grad, dx, alpha=0.5, beta=0.8):
    """Simple backtracking line search."""
    t = 1.0
    current_val = evaluate_h(h, dx)
    
    while True:
        h_new = h - t * grad
        h_new = repair_h(h_new, len(h))
        new_val = evaluate_h(h_new, dx)
        
        if new_val <= current_val - alpha * t * np.dot(grad, grad):
            return h_new, t
        t *= beta
        if t < 1e-10:
            return h, t

    best_h = None

    for res_idx, res in enumerate(resolutions):
        dx = 2.0 / res
        h_current = h.copy()
        
        if res < n_points:
            # Interpolate to coarser resolution
            x_orig = np.linspace(0, 2, len(h_current))
            x_target = np.linspace(0, 2, res)
            h_current = np.interp(x_target, x_orig, h_current)
            
        # Augmented Lagrangian optimization
        lambda_ = 0.0
        rho = 1.0
        
        for iter_count in range(500):
            if time.time() - start_time > budget_s - 5:

            # Evaluate current state
            current_value = evaluate_h(h_current, dx)
            
            # Compute constraint violation
            constraint_violation = np.abs(np.sum(h_current) - res / 2)
            
            # Update best if improved
            if current_value < best_value:
                best_value = current_value
                best_h = h_current.copy()
            
            # Compute gradient
            grad = compute_gradient(h_current, dx)
            
            # Gradient descent step with line search
            h_current, _ = backtrack_line_search(h_current, grad, dx)
            
            # Feasibility repair
            h_current = repair_h(h_current, res)

            if constraint_violation > 1e-4:

        if res_idx < len(resolutions) - 1:
