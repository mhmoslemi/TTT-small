import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum=50.0):
    """Project h to satisfy the sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Scale to satisfy sum constraint
    h = project_h(h, n_points / 2)
    
    # Add some noise for diversity
    noise_strength = 0.1
    h += np.random.normal(0, noise_strength, n_points)
    h = project_h(h, n_points / 2)
    
    # Optimization parameters
    lambda_reg = 0.01
    alpha = 0.1
    max_iter = 1000
    patience = 100

        eps = 1e-6
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(h, dx)

            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points / 2)

            # Reduce step size if improving

        else:
            # Increase step size if no improvement
            alpha = min(1.0, alpha * 1.2)
            no_improvement_count += 1
            
        # Early stopping if no improvement for too long
        if no_improvement_count > patience:
            break
            
    # Final evaluation of best solution
    final_c5 = evaluate_h(best_h, dx)
