import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints using quadratic programming."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Use quadratic programming to project onto feasible set
        # minimize ||h_new - h||^2 subject to h_new[i] in [0,1] and sum(h_new) = target_sum
        # We can solve this analytically.
        diff = target_sum - current_sum
        if diff > 0:
            # Need to increase sum, distribute among 0s
            # For now, we'll use a simple approach
            h = h + diff / n_points
        else:
            # Need to decrease sum, distribute among 1s
            h = h + diff / n_points
        h = np.clip(h, 0, 1)
    return h
