import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, target_sum=50):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.full(n_points, 0.5)
        h = project_h(h, target_sum=n_points / 2)
    
    # Add small random perturbation to break symmetry
    h += 0.1 * np.random.randn(n_points)
    h = project_h(h, target_sum=n_points / 2)
    
    # Regularization and optimization parameters
    lambda_reg = 0.01  # Sparsity promoting coefficient
    alpha = 0.1        # Initial learning rate
    max_iter = 1000
    min_improvement = 1e-4
    stagnation_limit = 100
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    last_improvement = 0
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate current objective and gradient
        current_c5 = evaluate_h(h, dx)
        
        # Compute gradient via finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, target_sum=n_points / 2)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        # Apply L1 regularization gradient
        reg_grad = lambda_reg * np.sign(h)
        total_grad = grad + reg_grad
        
        # Update step
        h_new = h - alpha * total_grad
        h_new = project_h(h_new, target_sum=n_points / 2)
        
        # Backtracking line search
        new_c5 = evaluate_h(h_new, dx)
        if new_c5 < current_c5:
            h = h_new
            alpha *= 0.8  # Reduce learning rate on improvement
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                last_improvement = i
        else:
            alpha *= 1.2  # Increase learning rate if no improvement
            # If no improvement for a while, slightly randomize
            if i - last_improvement > stagnation_limit:
                h = project_h(h + 0.01 * np.random.randn(n_points), target_sum=n_points / 2)
                last_improvement = i
        
        # Clamp learning rate
        alpha = max(1e-4, min(1.0, alpha))
    
    # Final refinement
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(h, dx)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, target_sum=n_points / 2)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        reg_grad = lambda_reg * np.sign(h)
        total_grad = grad + reg_grad
        
        h_new = h - 0.01 * total_grad
        h_new = project_h(h_new, target_sum=n_points / 2)
        new_c5 = evaluate_h(h_new, dx)
        
        if new_c5 < evaluate_h(h, dx):
            h = h_new
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
