import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def interpolate_to_resolution(h_coarse, new_n_points):
    """Interpolate coarse h to new resolution."""
    x_old = np.linspace(0, 2, len(h_coarse))
    x_new = np.linspace(0, 2, new_n_points)
    h_new = np.interp(x_new, x_old, h_coarse)
    return h_new

def genetic_evolution_step(particles, fitnesses, n_points, mutation_std, crossover_rate=0.3):
    """Perform one generation of evolutionary step."""
    # Sort by fitness
    sorted_indices = np.argsort(fitnesses)
    top_indices = sorted_indices[:len(particles) // 5]
    
    # Selection
    selected = [particles[i] for i in top_indices]
    
    # Generate new population
    new_population = selected.copy()
    
    # Crossover
    while len(new_population) < len(particles):
        parent1, parent2 = np.random.choice(selected, 2, replace=False)
        child = parent1.copy()
        mask = np.random.random(len(parent1)) < crossover_rate
        child[mask] = parent2[mask]
        new_population.append(child)
    
    # Mutation + repair
    mutated = []
    for p in new_population:
        noise = np.random.normal(0, mutation_std, n_points)
        mutated_p = p + noise
        mutated_p = repair_h(mutated_p, n_points)
        mutated.append(mutated_p)
        
    return mutated

        dx = 2.0 / n_points
        
        if i == 0:
            # Initial population
