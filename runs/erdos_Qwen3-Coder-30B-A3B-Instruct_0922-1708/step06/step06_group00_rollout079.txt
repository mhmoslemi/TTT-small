import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50000
    initial_temp = 0.1
    cooling_rate = 0.999
    tabu_size = 50
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        if len(h_current) != n_points:
            x_orig = np.linspace(0, 2, len(h_current))
            x_target = np.linspace(0, 2, n_points)
            h_current = np.interp(x_target, x_orig, h_current)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    else:
        # Start with a simple construction
        h_current = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    h_binary = (h_current > 0.5).astype(float)
    
    # Enforce sum constraint
    target_sum = n_points / 2
    current_sum = np.sum(h_binary)
    if current_sum != target_sum:
        # Adjust to get exact target sum
        indices = np.argsort(h_binary)[::-1]  # sort in descending order
        for i in indices:
            if current_sum == target_sum:
                break
            if h_binary[i] < 1:
                h_binary[i] = 1
                current_sum += 1
            else:
                h_binary[i] = 0
                current_sum -= 1
    
    # Track best solution
    best_h = h_binary.copy()
    best_c5 = float('inf')
    
    # Tabu list to avoid revisiting recent changes
    tabu_list = []
    
    # Simulated Annealing parameters
    current_temp = initial_temp
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        # Make a random change to the binary representation
        candidate_h = h_binary.copy()
        flip_idx = np.random.randint(0, n_points)
        candidate_h[flip_idx] = 1 - candidate_h[flip_idx]
        
        # Enforce target sum constraint
        current_sum = np.sum(candidate_h)
        if current_sum != target_sum:
            # Flip another bit to maintain sum
            candidates_to_flip = np.where(candidate_h != 1 - candidate_h[flip_idx])[0]
            if len(candidates_to_flip) > 0:
                flip_idx2 = np.random.choice(candidates_to_flip)
                candidate_h[flip_idx2] = 1 - candidate_h[flip_idx2]
        
        # Compute overlap for candidate
        corr = correlate(candidate_h, 1 - candidate_h, mode='full')
        c5_candidate = np.max(corr) * dx
        
        # Accept or reject based on simulated annealing rules
        if c5_candidate < best_c5:
            # Immediate acceptance if improving
            h_binary = candidate_h
            best_h = candidate_h.copy()
            best_c5 = c5_candidate
        else:
            # Accept with probability based on temperature
            delta = c5_candidate - best_c5
            if delta > 0:
                accept_prob = np.exp(-delta / current_temp)
                if np.random.rand() < accept_prob:
                    h_binary = candidate_h
            else:
                h_binary = candidate_h

        # Update temperature
        current_temp *= cooling_rate
        
        # Use tabu search to avoid local minimum
        # Store recent changes in tabu list
        change_tuple = tuple(np.where(h_binary != best_h)[0])
        if change_tuple in tabu_list:
            # If in tabu list, we might ignore if it's a significant improvement
            pass
        else:
            tabu_list.append(change_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        
        iteration += 1
    
    # Final validation and refinement
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Convert best_h back to continuous space
    h_continuous = best_h.copy()
    if final_c5 < 0.38085:  # If we found a good solution, refine a little
        # Simple gradient step to exploit local structure
        for _ in range(100):
            eps = 1e-6
            grad = np.zeros_like(h_continuous)
            for i in range(n_points):
                h_plus = h_continuous.copy()
                h_plus[i] += eps
                if h_plus[i] > 1:
                    h_plus[i] = 1
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - final_c5) / eps
            
            # Update with gradient descent
            h_continuous = h_continuous - 0.01 * grad
            h_continuous = np.clip(h_continuous, 0, 1)
            # Rescale to maintain constraint
            h_continuous = h_continuous * target_sum / np.sum(h_continuous)
            # Reconvert to binary for assessment
            temp_binary = (h_continuous > 0.5).astype(float)
            temp_sum = np.sum(temp_binary)
            if temp_sum != target_sum:
                # Adjust to preserve sum
                indices = np.argsort(temp_binary)[::-1]
                for i in indices:
                    if temp_sum == target_sum:
                        break
                    if temp_binary[i] < 1:
                        temp_binary[i] = 1
                        temp_sum += 1
                    else:
                        temp_binary[i] = 0
                        temp_sum -= 1
            corr = correlate(temp_binary, 1 - temp_binary, mode='full')
            temp_c5 = np.max(corr) * dx
            if temp_c5 < final_c5:
                final_c5 = temp_c5
                h_continuous = temp_binary.copy()
    
    return (h_continuous.tolist(), final_c5, n_points)
