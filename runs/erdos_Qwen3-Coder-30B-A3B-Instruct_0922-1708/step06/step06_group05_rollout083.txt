import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Hyperparameters for gradient descent
    lr = 0.01
    decay_factor = 0.99
    min_lr = 1e-6
    stagnation_limit = 50
    stagnation_count = 0
    last_improvement = 0
    
    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        
        # Compute FFT-based gradients
        # h * (1-h) correlation: use FFT for efficiency
        h_fft = np.fft.fft(h)
        ones_fft = np.fft.fft(np.ones_like(h))
        neg_h_fft = np.fft.fft(1 - h)
        corr_fft = h_fft * np.conj(neg_h_fft)
        corr = np.fft.ifft(corr_fft).real
        
        # Gradient w.r.t. to each h[i]
        grad = np.zeros(n_points)
        
        # This is a rough approximation; exact gradient is convolution of h with (h-1) on reversed indices (definitely works)
        # A more precise way is to compute:
        # For each k, find index of max correlation
        max_k = np.argmax(corr)
        
        # small perturbation of h
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - best_c5) / eps

        # Apply gradient
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        c5 = evaluate_h(h, dx)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
            last_improvement = iter_count
        else:
            stagnation_count += 1
            
        # Adjust learning rate
        if stagnation_count > stagnation_limit and lr > min_lr:
            lr *= decay_factor
            stagnation_count = 0
            
        # Early exit if no improvement for long time
        if iter_count - last_improvement > 1000:
            break

    return (best_h.tolist(), best_c5, n_points)
