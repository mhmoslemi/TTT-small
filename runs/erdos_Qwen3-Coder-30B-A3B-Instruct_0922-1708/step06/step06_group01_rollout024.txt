import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, epsilon=1e-6):
    n_points = len(h)
    grad = np.zeros(n_points)
    h_base = h.copy()
    for i in range(n_points):
        h_plus = h_base.copy()
        h_plus[i] += epsilon
        h_plus = repair_h(h_plus, n_points)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        
        h_minus = h_base.copy()
        h_minus[i] -= epsilon
        h_minus = repair_h(h_minus, n_points)
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        c5_minus = np.max(corr_minus) * dx
        
        grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
    return grad

def trust_region_subproblem(h, grad, hessian, trust_radius, n_points):
    # Simple quadratic model: min 1/2 * d^T H d + grad^T d
    # subject to: 0 <= h + d <= 1, sum(d) = 0
    
    # Simplified approach: gradient descent with trust region
    d = -grad
    norm_d = np.linalg.norm(d)
    if norm_d > trust_radius:
        d = d * trust_radius / norm_d
    
    # Project to feasible set
    h_new = h + d
    h_new = np.clip(h_new, 0, 1)
    
    # Adjust to meet sum constraint
    current_sum = np.sum(h_new)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    if abs(diff) > 1e-8:
        # Distribute adjustment evenly
        h_new += diff / n_points
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Use a low-discrepancy sequence (Halton-like approach)
        h_init = np.zeros(n_points)
        for i in range(n_points):
            h_init[i] = (i + 0.5) / n_points
        
        # Add some variation
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = repair_h(h_init, n_points)
    
    # Track best solution
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Trust region parameters
    trust_radius = 1.0
    trust_radius_min = 1e-6
    trust_radius_max = 0.5
    reduction_factor = 0.5
    expansion_factor = 2.0
    
    # Optimization parameters
    max_iter = 2000
    perturb_freq = 250
    perturb_magnitude = 0.1
    
    h_current = h_init.copy()
    c5_current = c5_best
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        grad = compute_gradient(h_current, dx)
        
        # Get candidate solution
        h_candidate = trust_region_subproblem(h_current, grad, None, trust_radius, n_points)
        h_candidate = repair_h(h_candidate, n_points)
        c5_candidate = evaluate_h(h_candidate, dx)
        
        # Accept or reject
        if c5_candidate < c5_current:
            h_current = h_candidate
            c5_current = c5_candidate
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                trust_radius = min(trust_radius_max, trust_radius * expansion_factor)
        else:
            trust_radius = max(trust_radius_min, trust_radius * reduction_factor)
            
        # Periodic perturbation
        if i % perturb_freq == 0 and i > 0:
            h_current = h_current + np.random.uniform(-perturb_magnitude, perturb_magnitude, n_points)
            h_current = repair_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                
    # Final local refinement with L-BFGS
    def objective(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-8:
            h = h * target_sum / current_sum
        corr = correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx

    def constraint(h):
        return np.sum(h) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = minimize(objective, h_best, method='SLSQP', bounds=bounds, constraints=cons, 
                        options={'maxiter': 500, 'ftol': 1e-8})
        if res.success:
            h_final = repair_h(res.x, n_points)
            c5_final = evaluate_h(h_final, dx)
            if c5_final < c5_best:
                h_best = h_final
                c5_best = c5_final
    except:
        pass
    
    return (h_best.tolist(), c5_best, n_points)
