import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Define optimization variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        t >= 0
    ]
    
    # Compute overlaps for all valid lags
    # For a step function h(x), overlap at shift k is
    # ∫ h(x) * (1 - h(x + k)) dx
    # Discrete version: sum h[i] * (1 - h[(i+k)%n]) * dx
    # We compute this for all valid shifts
    max_lag = n_points - 1  # Maximum lag for non-circular overlaps
    
    # For each lag k in [-max_lag, max_lag], 
    # we compute the contribution: sum over i of h[i] * (1 - h[i-k]) * dx
    
    # Construct the overlap constraints
    # For positive lag k: overlap_k = sum_{i} h[i] * (1 - h[i-k])
    # For negative lag k: overlap_k = sum_{i} h[i] * (1 - h[i-k])
    # Since we concatenate the kernel with itself for non-circular, we compute valid entries
    
    # Generate all lags from -max_lag to max_lag that are valid (symmetric case)
    # The symmetric solution means we only consider unique effective lags
    # But to be exhaustive, we go over all overlaps in the valid range
    
    c5_bound = np.inf
    
    # Index-based computation
    # We'll iterate lags in a way that matches the convolution code
    # Specifically, correlate with reverse of h to get the overlap
    
    # To compute it more precisely in QP:
    # overlap_k = sum_{i} h[i] * (1 - h[i-k]) for valid i,k
    # For h longer than 1, and the lag from -n+1 to n-1 (in discrete terms)
    # Let's compute the actual number of constraints needed
    
    # The effective overlaps for aligning h with (1-h) for all shifts
    # We go through indices for lag k from -(n-1) to (n-1)
    # for each k, we consider positions where the sum is valid
    
    # Optimization approach:
    # We compute the valid overlaps efficiently and use them
    
    # Build all overlap constraint entries
    # Each constraint: sum(h[i] * (1 - h[i-k])) <= t*dx
    
    # It's better to work with full convolution
    # For q1 = h, q2 = 1-h, overlap = q1 conv q2
    # We can do this once symmetrically
    
    valid_lags = list(range(-max_lag, max_lag + 1))
    
    # For the more precise optimization approach
    # Build overlap by summing over valid combinations
    for k in valid_lags:
        # Build the convolution-like constraint basing on k
        if k >= 0:
            # Left side shifted by k, right side shifted by 0, so valid overlaps
            # h[0:n-k] * (1 - h[k:n])
            indices_i = np.arange(0, n_points - k)
            indices_j = np.arange(k, n_points)
            if len(indices_i) == 0:
                continue
        else:
            # k < 0, right side shifted by k
            # h[0:n+k] * (1 - h[-k:n])
            indices_i = np.arange(-k, n_points)
            indices_j = np.arange(0, n_points + k)
            if len(indices_i) == 0:
                continue
        
        # Build term sum h[i] * (1 - h[j]) = sum_{i} h[i] * 1 - sum_{i} h[i] * h[j[i]]  
        # But we must map indices correctly
                
        # Total overlap term for lag k
        # sum(h[i] * (1 - h[j])) = sum(h[i]) - sum(h[i] * h[j])
        # But when both are arrays we need:
        # sum(h[indices_i] * (1 - h[indices_j]))
        # which is sum_i h[indices_i[i]] * (1 - h[indices_j[i]])
        # so we do: sum(h[indices_i] * (1 - h[indices_j]))
        
        # Let's make this properly with CPX
        # Use a piece-wise sum instead
        # Index-based constraint:
        # sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k]) <= t * dx
        
        if indices_i.size > 0:
            # Build the constraint involving this offset
            # Sum over indices_i of h[i] * (1 - h[i+k]) <= t * dx
            # => sum(h[i]) - sum(h[i]*h[i+k]) <= t*dx
            # We express this using disciplined convex programming
            # But we cannot just do the inner product direcly in CT
            # So we impose approximate constraint on certain terms using
            # direct CP expression
            
            # Methodology:
            # Broadcast indices for assembling the constraint  
            # Since i goes from 0 to n-k-1, and j = i+k, where k is fixed:
            # sum using index operations
            # For all valid i, h[i] * (1 - h[i+k]) <= t * dx  
            indices_i = indices_i.astype(int)
            indices_j = indices_j.astype(int)
            
            # Sum h[i] * (1 - h[i+k]) over all valid i
            overlap_expr = cp.sum(cp.multiply(
                h[indices_i], 
                1 - h[indices_j]
            ))
            constraints.append(overlap_expr <= t * dx)
    
    # Set objective
    objective = cp.Minimize(t)
    
    # Construct and solve the problem  
    prob = cp.Problem(objective, constraints)
    
    try:
        # Use well-suited solver for QP
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            opt_solution = h.value
            opt_c5 = t.value * dx
            c5_bound = opt_c5
        else:
            # Fallback to the original patch
            opt_solution = h_init.copy()
            c5_bound = evaluate_h(opt_solution, dx) 
    except:
        # Fallback
        opt_solution = h_init.copy() 
        c5_bound = evaluate_h(opt_solution, dx)
        
    # Final verification with the full evaluation function
    opt_solution = np.clip(opt_solution, 0, 1)
    opt_sum = np.sum(opt_solution)
    if abs(opt_sum - n_points / 2) > 1e-5:
        opt_solution = opt_solution * (n_points / 2) / opt_sum
    
    # Compute final c5 with full evaluation
    final_eval = evaluate_h(opt_solution, dx)
    
    return (opt_solution.tolist(), final_eval, n_points)

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
