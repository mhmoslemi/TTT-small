import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using FFT-based convolution."""
    # Using fftconvolve for efficiency
    corr = signal.fftconvolve(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Better initialization: try to make it more structured
        h = np.zeros(n_points)
        # Create a piecewise constant function that's reasonably balanced
        # Use ~5 segments to avoid too much complexity
        segments = 5
        seg_length = n_points // segments
        for i in range(segments):
            start = i * seg_length
            end = (i + 1) * seg_length if i < segments - 1 else n_points
            # Alternate between high and low values
            val = 0.6 if i % 2 == 0 else 0.2
            h[start:end] = val
            
    # Ensure integral constraint is met by rescaling
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Parameters
    lr = 0.001
    perturb_rate = 0.05
    max_iter = 10000
    stagnation_limit = 200
    stagnation_count = 0
    lr_decay = 0.95
    decay_every = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute the current overlap values
        corr = signal.fftconvolve(h, 1 - h, mode='full')
        
        # Find the shift where overlap is maximum
        max_idx = np.argmax(corr)
        
        # Compute gradient with respect to h at the maximum overlap shift
        # This is an approximation: finite difference along the maximum direction
        h_eval = evaluate_h(h, dx)
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - h_eval) / eps
        
        # Update step: focus on reducing the maximum overlap
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        # Evaluate
        c5 = evaluate_h(h, dx)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Random perturbation occasionally
        if np.random.rand() < perturb_rate or stagnation_count > stagnation_limit:
            # Perturb a fraction of elements
            indices = np.random.choice(n_points, size=int(n_points * perturb_rate), replace=False)
            h[indices] += np.random.normal(0, 0.01, len(indices))
            h = repair_h(h, n_points)
            # Reevaluate
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
                
        # Occasionally enforce integral constraint via projection
        if iteration % 100 == 0:
            h = h * (n_points / 2) / np.sum(h)
            
        # Learning rate decay
        if iteration > 0 and iteration % decay_every == 0:
            lr *= lr_decay
            
    # Final intensive refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 2:
            break
        corr = signal.fftconvolve(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
    return (best_h.tolist(), best_c5, n_points)
