import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Starting with a simple binary pattern
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Fast iteration: try alternating local search and cuts
    max_iterations = 300
    cut_freq = 10
    local_iter = 0
    
    for iteration in range(max_iterations):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Periodically solve a relaxed problem using cutting planes
        if iteration % cut_freq == 0:
            # Save current best
            corr = correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Formulate and solve the convex optimization subproblem
            try:
                # Solve the relaxed problem with a new cut
                t = cp.Variable()
                h_var = cp.Variable(n_points)
                
                # Constraint sum(h) = n_points / 2
                sum_constraint = cp.sum(h_var) == n_points / 2
                
                # Constraints h in [0,1]
                bound_constraints = [h_var >= 0, h_var <= 1]
                
                # Compute maximum overlap shift (selecting worst-case)
                corr_all = correlate(h, 1 - h, mode='full')
                max_lag_idx = np.argmax(corr_all)
                k_max = max_lag_idx - (len(h) - 1)
                
                # Overlap is expressed as: 
                # overlap_k = dx * sum(h[i] * (1 - h[i + k]))
                # Which = dx * (sum(h) - sum(h[i] * h[i + k]))
                # We compute this over all valid indices
                indices_i = np.arange(len(h))
                shifted_i = indices_i + k_max
                valid_mask = (shifted_i >= 0) & (shifted_i < len(h))
                i_valid = indices_i[valid_mask]
                j_valid = shifted_i[valid_mask]
                
                if len(i_valid) > 0:
                    # Constraint: overlapping integral <= t
                    # sum(h[i] * (1 - h[j])) <= t / dx 
                    # This is equivalent to: sum(h[i] * h[j]) >= sum(h) - t / dx
                    prod_sum = cp.sum(cp.multiply(h_var[i_valid], h_var[j_valid]))
                    cut_constraint = sum(h_var) - prod_sum <= t / dx
                    # Re-parameterized version of overlap_k constraint
                    # This is the main constraint being honored to avoid local maxima
                    
                else:
                    cut_constraint = t >= 0  # fallback
                    
                # Convex problem
                obj = cp.Minimize(t)
                constraints = [sum_constraint] + bound_constraints + [cut_constraint]
                prob = cp.Problem(obj, constraints)
                
                # Try different solvers
                solvers = ['ECOS', 'SCS']
                for solver in solvers:
                    try:
                        prob.solve(solver=solver, verbose=False)
                        if prob.status in ["optimal", "optimal_inaccurate"]:
                            h_new = h_var.value
                            if h_new is not None:
                                h = np.clip(h_new, 0, 1)
                                h_sum = np.sum(h)
                                if abs(h_sum - n_points / 2) > 1e-6:
                                    h = h * (n_points / 2) / h_sum
                                break
                    except:
                        continue
                else:
                    # If all solvers fail, just use current h
                    pass
                    
        else:
            # Local optimization to refine the solution
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Simple gradient-based update
            learning_rate = 0.001
            max_lag_idx = np.argmax(corr)
            k = max_lag_idx - (len(h) - 1)
            grad = np.zeros_like(h)
            
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                if h_plus[i] > 1:
                    h_plus[i] = 1
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / 1e-6
                
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-8:
                h = h * (n_points / 2) / h_sum
            
        local_iter += 1
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
