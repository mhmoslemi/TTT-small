import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Barrier method parameters
    mu = 1.0
    barrier_factor = 1.1
    max_iter = 30
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Define variables for optimization
    h_var = cp.Variable(n_points)
    t = cp.Variable()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Create constraints for overlaps
        constraints = []
        # Objective: minimize t
        objective = cp.Minimize(t)
        
        # Add constraints for all possible shifts
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        
        for k in range(1, n_points):
            # For each shift k, we want t >= overlap_k
            # overlap_k = sum h[i] * (1 - h[i+k]) over valid indices
            shift_sum = 0
            for i in range(n_points - k):
                shift_sum += h[i] * (1 - h[i + k])
            
            # Add constraint: t >= (shift_sum * dx)
            # But we directly use the precomputed overlap for this specific
            # shift since we want to use the actual values calculated by the
            # original nonlinear optimization framework
            # However for exact solution with current issue, we need to limit
            # how fast we rely on consistency and instead focus on the process
            # So recompute manually
            # Negative barrier term
            barrier_term = 0
            for i in range(n_points):
                barrier_term -= mu * (np.log(h_var[i]) + np.log(1 - h_var[i]))
            
            constraints.append(t >= shift_sum * dx)
            
        # Add sum constraint
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Add bounds: 0 <= h_var[i] <= 1
        for i in range(n_points):
            constraints.append(h_var[i] >= 0)
            constraints.append(h_var[i] <= 1)
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_var.value
                # Ensure h_new is feasible
                h_new = np.clip(h_new, 0, 1)
                h_new_sum = np.sum(h_new)
                if abs(h_new_sum) > 1e-10:
                    h_new = h_new * (n_points / 2) / h_new_sum
                else:
                    h_new = np.ones(n_points) * 0.5
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
                # Compute new C5
                corr = correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        except Exception:
            # If optimization fails, continue with simple update
            pass
            
        # Update barrier
        mu *= barrier_factor
        
        # Simple heuristic: slightly perturb h based on recent improvement
        if iteration % 3 == 0 and iteration > 0:
            h = h * 0.9 + np.random.rand(n_points) * 0.1
            h_sum = np.sum(h)
            if abs(h_sum) > 1e-10:
                h = h * (n_points / 2) / h_sum
    
    # Final estimate
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
