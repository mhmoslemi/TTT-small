import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5  # Uniform initial guess
        
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Shift grid setup
    shift_grid = np.linspace(-2, 2, 100)
    
    # Optimization parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 100
    no_improvement_count = 0
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        # Extract valid correlations
        mid = len(corr) // 2
        corr_values = corr[mid - n_points: mid + n_points]
        
        # Find worst shift
        worst_k_idx = np.argmax(corr_values)
        worst_k = shift_grid[worst_k_idx]
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            # We compute the derivative of integral h(x)(1 - h(x + k))
            # with respect to h[i]
            idx_shift = i + int(worst_k / dx)
            if 0 <= idx_shift < n_points:
                grad[i] -= 1.0  # derivative of -h(x+k) term
            idx_shift = i - int(worst_k / dx)
            if 0 <= idx_shift < n_points:
                grad[i] += 1.0  # derivative of h(x) term
        
        # Update h
        h += learning_rate * grad
        h = repair_h(h, n_points)
        
        # Evaluate current solution
        current_value = evaluate_h(h, dx)
        if current_value < best_value:
            best_value = current_value
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Periodic perturbation to escape local minima
        if it % 50 == 0 and it > 0:
            h += np.random.normal(0, 0.001, n_points)
            h = repair_h(h, n_points)
            
        # Adaptive learning rate
        if no_improvement_count > 100:
            learning_rate *= 0.95
            no_improvement_count = 0
            
        # Early stopping
        if no_improvement_count > patience:
            break
            
    return (best_h.tolist(), best_value, n_points)
