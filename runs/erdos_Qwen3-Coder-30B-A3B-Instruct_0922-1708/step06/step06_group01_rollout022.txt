import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.optimize import Bounds

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute gradient using finite differences"""
    n_points = len(h)
    grad = np.zeros(n_points)
    h_base = h.copy()
    for i in range(n_points):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n_points)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        
        h_minus = h_base.copy()
        h_minus[i] -= eps
        h_minus = repair_h(h_minus, n_points)
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        c5_minus = np.max(corr_minus) * dx
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def trust_region_step(h, dx, trust_radius=1.0, max_iterations=100):
    """Perform a trust region step"""
    n_points = len(h)
    h_current = h.copy()
    c5_current = evaluate_h(h_current, dx)
    
    for _ in range(max_iterations):
        # Compute gradient and Hessian approximation
        grad = compute_gradient(h_current, dx)
        
        # Trust region subproblem
        # min 0.5 * d^T * H * d + grad^T * d 
        # s.t. ||d|| <= trust_radius
        #      sum(d) = 0
        #      0 <= h_current + d <= 1
        
        # Simple approximate trust region step
        # Using gradient descent with step size bounded by trust radius
        step_size = min(trust_radius, 0.1)
        d = -step_size * grad
        
        # Enforce sum constraint
        d_sum = np.sum(d)
        d = d - d_sum / n_points
        
        # Clip to feasibility
        h_new = h_current + d
        h_new = np.clip(h_new, 0, 1)
        
        # Repair to sum constraint
        h_new = repair_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
        else:
            break
            
    return h_current, c5_current
