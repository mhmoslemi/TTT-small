import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = 100
    pop_size = 50
    F = 1.0
    CR = 0.9
    max_generations = 1000
    stagnation_limit = 100
    
    # Initialize population
    population = []

        # Interpolate to match n_points

        h = np.interp(x_target, x_orig, h)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Fill population with random solutions
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    population = np.array(population)
    
    # Track best solution
    best_idx = np.argmin([evaluate_h(h, 2.0/n_points) for h in population])
    best_h = population[best_idx].copy()
    best_c5 = evaluate_h(best_h, 2.0/n_points)
    stagnation_count = 0
    
    # Evolution loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # DE operations
        new_population = []
        
        for i in range(pop_size):
            # Select three distinct individuals
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs]
            
            # Mutation
            v = a + F * (b - c)
            v = np.clip(v, 0, 1)
            v = repair_h(v, n_points)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, v, population[i])
            
            # Ensure constraints hold
            trial = repair_h(trial, n_points)
            
            # Evaluate
            c5_trial = evaluate_h(trial, 2.0/n_points)
            
            # Selection
            if c5_trial < evaluate_h(population[i], 2.0/n_points):
                new_population.append(trial)
            else:
                new_population.append(population[i])
                
        population = np.array(new_population)
        
        # Update best
        current_best_idx = np.argmin([evaluate_h(h, 2.0/n_points) for h in population])
        current_best_c5 = evaluate_h(population[current_best_idx], 2.0/n_points)
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive parameters
        if stagnation_count > stagnation_limit:
            F = min(2.0, F + 0.1)
            stagnation_count = 0
    
    # Final refinement with gradient descent
    dx = 2.0 / n_points
    for _ in range(500):
        if time.time() - start_time > budget_s - 2:

        best_h = best_h - 0.0002 * grad

        if new_c5 < best_c5:
            best_c5 = new_c5
            
    return (best_h.tolist(), best_c5, n_points)
