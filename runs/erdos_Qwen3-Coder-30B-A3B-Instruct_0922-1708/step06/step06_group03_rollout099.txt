import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of C5 w.r.t h at k_max."""
    n = len(h)
    grad = np.zeros(n)
    
    # Gradient computation for the specific k_max
    for i in range(n):
        # Since the max is at k_max, we use the derivative of the convolution
        # at that lag
        grad[i] = (1 - h[(i - k_max) % n]) * dx
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    max_iter = 100000
    improvement_count = 0
    max_improvement = 500
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all overlaps to find max
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Compute gradient
        grad = compute_gradient(current_h, dx, k_max)
        
        # Apply gradient descent
        h_new = current_h - learning_rate * grad
        
        # Add structured perturbations
        if i % 100 == 0:
            # Identify indices with low gradient contribution
            abs_grad = np.abs(grad)
            # Perturb indices with gradient less than median
            low_grad_indices = np.where(abs_grad < np.median(abs_grad))[0]
            if len(low_grad_indices) > 0:
                # Perturb a few low-gradient indices
                perturb_indices = np.random.choice(

                for idx in perturb_indices:
                    h_new[idx] += np.random.uniform(
                        -perturbation_magnitude,
                        perturbation_magnitude)
        
        # Repair the new solution
        h_new = repair_h(h_new, n_points)
        new_c5 = evaluate_h(h_new, dx)

        if new_c5 < current_c5:
            current_h = h_new
            current_c5 = new_c5
            improvement_count = 0

                best_c5 = current_c5
