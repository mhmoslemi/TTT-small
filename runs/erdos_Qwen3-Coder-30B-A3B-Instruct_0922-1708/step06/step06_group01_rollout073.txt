import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute approximate gradient of C5."""
    n = len(h)
    grad = np.zeros(n)
    c5_base = evaluate_h(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    
    return grad

def trust_region_step(h, dx, grad, H, trust_radius):
    """Solve the trust region subproblem to find step."""
    n = len(h)
    # Simple projected gradient method for trust region step
    d = -grad
    d_norm = np.linalg.norm(d)
    if d_norm > trust_radius:
        d = d * trust_radius / d_norm
    
    # Ensure feasibility
    h_new = h + d
    h_new = repair_h(h_new, n)
    
    return h_new - h

    dx = 2.0 / n_points
    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a low-discrepancy sequence (Halton-like)
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
    
    # Start optimization
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    trust_radius = 1.0
    max_iter = 2000
    perturbation_frequency = 250
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Recompute gradient
        grad = compute_gradient(h_current, dx)
        
        # Approximate Hessian using finite differences
        H = np.zeros((n_points, n_points))
        eps_h = 1e-4
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps_h
            h_plus = repair_h(h_plus, n_points)
            grad_plus = compute_gradient(h_plus, dx)
            
            H[:, j] = (grad_plus - grad) / eps_h
        
        # Compute trust region step
        d = trust_region_step(h_current, dx, grad, H, trust_radius)
        
        # Try the step
        h_new = h_current + d
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Accept the step if it improves
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new

            alpha = 1.0
            while alpha > 1e-8:

                h_try = repair_h(h_try, n_points)
                c5_try = evaluate_h(h_try, dx)
                if c5_try < c5_current:
                    h_current = h_try
                    c5_current = c5_try
                    if c5_current < c5_best:

                        c5_best = c5_try
