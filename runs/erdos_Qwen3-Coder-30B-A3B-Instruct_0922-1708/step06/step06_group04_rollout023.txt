import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a sieve-like pattern in frequency domain
    freq = np.fft.fftfreq(n_points, 1/n_points)
    # Attract high frequencies to suppress occasional overlaps
    freq_resp = np.exp(-0.5 * (freq / (n_points/20))**2)
    # Apply inverse FFT to create spatial pattern
    h = np.fft.ifft(freq_resp).real
    h = np.abs(h)  # Ensure positivity
    return project_h(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialization
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        h_init = fourier_initialization(n_points)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 50
    max_iter = 2000  # Split iterations between tabu and gradient
    perturb_strength = 0.2
    diversity_injection_interval = 50
    stagnation_threshold = 50
    
    # Initialize tabu search
    population = []
    fitness = []
    population.append(h_init.copy())
    fitness.append(evaluate_overlap(h_init, dx))
    
    # Create multiple diverse starting points
    for _ in range(9):
        h = fourier_initialization(n_points) 
        population.append(h)
        fitness.append(evaluate_overlap(h, dx))
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate new candidate using perturbation
        h = population[best_idx].copy()
        # Perturb with a small random step
        perturb = np.random.normal(0, perturb_strength, n_points)
        h_new = h + perturb
        
        # Repair and project to feasible set
        h_new = project_h(h_new, n_points)
        
        # Avoid tabu moves by checking previously explored areas
        if len(tabu_list) > 0:
            # Simple move representation: percentage change
            move_repr = np.abs((h_new - h) / (h + 1e-8))
            if np.all(move_repr < 0.01):  # Ignore negligible changes
                pass
            else:
                # Check if this move has been made recently
                recent = False
                for tabu in tabu_list:
                    if np.allclose(h_new, tabu, atol=1e-4):
                        recent = True
                        break
                if recent:
                    # Apply a stronger perturbation
                    h_new = h + np.random.normal(0, perturb_strength*2, n_points)
                    h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        new_value = evaluate_overlap(h_new, dx)
        population.append(h_new)
        fitness.append(new_value)
        
        # Keep only top solutions
        indices = np.argsort(fitness)[:10]
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        best_idx = np.argmin(fitness)
        new_best_value = fitness[best_idx]
        
        # Update best solution
        if new_best_value < best_value:
            best_h = population[best_idx].copy()
            best_value = new_best_value
            last_improvement = iter_count
        
        # Update tabu list
        tabu_list.append(h_new)
        if len(tabu_list) > tabu_size:
            tabu_list = tabu_list[1:]  # Remove oldest
        iter_count += 1
        
        # Inject diversity periodically to avoid stagnation
        if iter_count % diversity_injection_interval == 0:
            h = fourier_initialization(n_points)
            population.append(h)
            fitness.append(evaluate_overlap(h, dx))
            
            if len(population) > 10:
                indices = np.argsort(fitness)[:10]
                population = [population[i] for i in indices]
                fitness = [fitness[i] for i in indices]
                best_idx = np.argmin(fitness)
                best_value = fitness[best_idx]
    
    # Gradient descent refinement of best solution
    h_refined = best_h.copy()
    prev_value = best_value
    learning_rate = 0.0001
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        h_refined = h_refined - learning_rate * grad
        h_refined = project_h(h_refined, n_points)
        prev_value = evaluate_overlap(h_refined, dx)
        
        if prev_value < best_value:
            best_h = h_refined.copy()
            best_value = prev_value
    
    return (best_h.tolist(), best_value, n_points)
