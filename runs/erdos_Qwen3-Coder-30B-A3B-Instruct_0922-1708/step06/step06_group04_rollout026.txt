import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Define optimization variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2,
        t >= 0
    ]
    
    # Generate all valid lag indices for non-circular convolution
    max_lag = n_points - 1
    valid_lags = list(range(-max_lag, max_lag + 1))
    
    # For each shift, add overlap constraint
    for lag in valid_lags:
        if lag >= 0:
            # Forward shift: h[i] * (1 - h[i-lag]) for i >= lag
            indices_i = np.arange(lag, n_points)
            indices_j = np.arange(0, n_points - lag)
            if len(indices_i) > 0:
                overlap_expr = cp.sum(cp.multiply(h[indices_i], 1 - h[indices_j]))
                constraints.append(overlap_expr <= t * (n_points / 2))
        else:
            # Backward shift: h[i] * (1 - h[i-lag]) for i < n_points + lag  
            indices_i = np.arange(0, n_points + lag)
            indices_j = np.arange(-lag, n_points)
            if len(indices_i) > 0:
                overlap_expr = cp.sum(cp.multiply(h[indices_i], 1 - h[indices_j]))
                constraints.append(overlap_expr <= t * (n_points / 2))
    
    # Objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.MOSEK, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_result = h.value
            c5_result = t.value * dx
        else:
            # Fall back to local search if optimization fails
            h_result = h_init.copy()
            c5_result = evaluate_h(h_result, dx)
            # Simple local search
            best_h = h_result.copy()
            best_c5 = c5_result
            for _ in range(5000):
                if time.time() - start_time > budget_s - 10:
                    break
                direction = np.random.randn(n_points)
                step_size = 1e-4
                new_h = h_result + step_size * direction
                new_h = np.clip(new_h, 0, 1)
                new_sum = np.sum(new_h)
                new_h = new_h * (n_points / 2) / new_sum
                new_c5 = evaluate_h(new_h, dx)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                h_result = new_h.copy()
            h_result = best_h.copy()
            c5_result = best_c5
    except:
        # Fallback
        h_result = h_init.copy()
        c5_result = evaluate_h(h_result, dx)
    
    return (h_result.tolist(), c5_result, n_points)
