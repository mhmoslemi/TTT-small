import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50000
    tabu_size = 50
    initial_temp = 0.1
    cooling_rate = 0.999
    
    # Initialize h from initial_h_values or create binary
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Convert to binary representation with exact sum constraint
    def to_binary(h_vec):
        # Create binary vector with sum = n_points/2
        h_copy = h_vec.copy()
        h_copy = np.clip(h_copy, 0, 1)
        h_sorted = np.sort(h_copy)[::-1]
        h_binary = np.zeros_like(h_copy)
        # Set first n_points//2 positions to 1
        h_binary[np.argsort(h_copy)[::-1][:n_points//2]] = 1
        return h_binary
    
    # Convert to binary and back to continuous for optimization
    h_binary = to_binary(h)
    
    # Track tabu list
    tabu_list = []
    
    # Initialize best values
    best_h = h_binary.copy()
    best_c5 = float('inf')
    current_h = h_binary.copy()
    
    # Simulated annealing
    temperature = initial_temp
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Randomly flip bits to generate new candidate
        candidate_h = current_h.copy()
        flip_positions = np.random.choice(n_points, size=5, replace=False)
        for pos in flip_positions:
            candidate_h[pos] = 1 - candidate_h[pos]
        
        # Ensure sum constraint
        current_sum = np.sum(current_h)
        candidate_sum = np.sum(candidate_h)
        
        # Correct the candidate to maintain the sum constraint
        if abs(candidate_sum - n_points/2) > 1e-10:
            diff = int(n_points/2 - candidate_sum)
            # If we need to add more ones
            if diff > 0:
                # Pick zero positions and turn them into ones
                zero_positions = np.where(candidate_h == 0)[0]
                if len(zero_positions) >= diff:
                    add_positions = np.random.choice(zero_positions, size=diff, replace=False)
                    for pos in add_positions:
                        candidate_h[pos] = 1
                else:
                    # If not enough zeros, remove some ones to match target
                    one_positions = np.where(candidate_h == 1)[0]
                    remove_positions = np.random.choice(one_positions, size=abs(diff), replace=False)
                    for pos in remove_positions:
                        candidate_h[pos] = 0
            # If we need to subtract some ones
            elif diff < 0:
                # Pick one positions and turn them into zeros
                one_positions = np.where(candidate_h == 1)[0]
                if len(one_positions) >= -diff:
                    remove_positions = np.random.choice(one_positions, size=-diff, replace=False)
                    for pos in remove_positions:
                        candidate_h[pos] = 0
                else:
                    # If not enough ones, add some zeros to match target
                    zero_positions = np.where(candidate_h == 0)[0]
                    add_positions = np.random.choice(zero_positions, size=abs(diff), replace=False)
                    for pos in add_positions:
                        candidate_h[pos] = 1
            
        # Avoid forbidden moves - this is a simple tabu check
        move_key = tuple(sorted(flip_positions))
        if move_key in tabu_list:
            # Pick a different move that's not tabu
            candidate_h = current_h.copy()
            # Just try flipping a few positions differently
            flip_positions = np.random.choice(n_points, size=3, replace=False)
            for pos in flip_positions:
                candidate_h[pos] = 1 - candidate_h[pos]
            # Apply same correction logic
            candidate_sum = np.sum(candidate_h)
            if abs(candidate_sum - n_points/2) > 1e-10:
                diff = int(n_points/2 - candidate_sum)
                if diff > 0:
                    zero_positions = np.where(candidate_h == 0)[0]
                    if len(zero_positions) >= diff:
                        add_positions = np.random.choice(zero_positions, size=diff, replace=False)
                        for pos in add_positions:
                            candidate_h[pos] = 1
                    else:
                        one_positions = np.where(candidate_h == 1)[0]
                        remove_positions = np.random.choice(one_positions, size=abs(diff), replace=False)
                        for pos in remove_positions:
                            candidate_h[pos] = 0
                elif diff < 0:
                    one_positions = np.where(candidate_h == 1)[0]
                    if len(one_positions) >= -diff:
                        remove_positions = np.random.choice(one_positions, size=-diff, replace=False)
                        for pos in remove_positions:
                            candidate_h[pos] = 0
                    else:
                        zero_positions = np.where(candidate_h == 0)[0]
                        add_positions = np.random.choice(zero_positions, size=abs(diff), replace=False)
                        for pos in add_positions:
                            candidate_h[pos] = 1
        
        # Compute correlation for candidate
        corr = correlate(candidate_h, 1 - candidate_h, mode='full')
        c5_candidate = np.max(corr) * dx
        
        # Compute correlation for current
        corr_current = correlate(current_h, 1 - current_h, mode='full')
        c5_current = np.max(corr_current) * dx
        
        # Accept if better or with probability exp(-Δ/T)
        delta = c5_candidate - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            current_h = candidate_h
            # Update tabu list
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(move_key)
        
        # Update best solution
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = current_h.copy()
        
        # Update temperature
        temperature *= cooling_rate
    
    # Final refinement via gradient-step algorithm
    # Average the binary values to get a continuous approximation
    h_continuous = best_h.astype(float)
    
    # Refine using projected gradient descent with finite difference
    learning_rate = 0.01
    max_inner_iter = 100
    
    for inner_iter in range(max_inner_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap for current h
        corr = correlate(h_continuous, 1 - h_continuous, mode='full')
        c5_current = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h_continuous) - 1)
        
        # Estimate gradient via finite differences
        grad = np.zeros_like(h_continuous)
        epsilon = 1e-5
        for i in range(len(h_continuous)):
            h_plus = h_continuous.copy()
            h_plus[i] += epsilon
            # Clip to [0,1]
            h_plus = np.clip(h_plus, 0, 1)
            # Rescale to maintain sum
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5_current) / epsilon
        
        # Gradient descent step
        h_continuous = h_continuous - learning_rate * grad
        
        # Project to [0,1] and rescale
        h_continuous = np.clip(h_continuous, 0, 1)
        h_sum = np.sum(h_continuous)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h_continuous = h_continuous * target_sum / h_sum
    
    # Evaluate final C5
    corr_final = correlate(h_continuous, 1 - h_continuous, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (h_continuous.tolist(), final_c5, n_points)
