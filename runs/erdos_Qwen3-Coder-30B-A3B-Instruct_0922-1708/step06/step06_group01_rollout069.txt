import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def halton_sequence(n, base=2):
    """Generate Halton sequence for [0,1]"""
    result = []
    for i in range(n):
        x = 0
        f = 1.0 / base
        k = i
        while k > 0:
            x += (k % base) * f
            k //= base
            f /= base
        result.append(x)
    return np.array(result)

def sobol_sequence(n):
    """Generate Sobol sequence using primitive polynomials"""
    # Simple 1D Sobol using base 2
    result = []
    for i in range(n):
        x = 0
        k = i
        bit = 1
        while k > 0:
            if k & 1:
                x += bit
            k >>= 1
            bit *= 0.5
        result.append(x)
    return np.array(result)

def init_h_from_sequence(n_points, sequence_func, seed):
    """Initialize h from a sequence function"""
    np.random.seed(seed)
    seq = sequence_func(n_points)
    h = seq.copy()
    # Smooth and normalize
    h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_points), h)
    h = (h - h.min()) / (h.max() - h.min()) if h.max() > h.min() else np.ones(n_points) * 0.5
    h = repair_h(h, n_points)
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute finite difference gradient"""
    grad = np.zeros_like(h)
    c5_current = evaluate_h(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = repair_h(h_plus, len(h))
        h_minus = repair_h(h_minus, len(h))
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        c5_minus = np.max(corr_minus) * dx
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    
    # Best solution tracking
    h_best = None
    c5_best = float('inf')

    seeds = [seed + i for i in range(10)]
    init_methods = [halton_sequence, sobol_sequence]
    
    for seed in seeds:
        # Initialize

                h = np.interp(x_target, x_orig, h)
            h = repair_h(h, n_points)
        else:
            method = init_methods[(seed + 1) % len(init_methods)]
            h = init_h_from_sequence(n_points, method, seed)
        
        # Gradient descent optimization
        max_iter = 5000
        step_size = 0.01
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
            
            # Compute gradient
            grad = compute_gradient(h, dx)
            
            # Update
            h = h - step_size * grad
            h = repair_h(h, n_points)
            
            # Evaluate
            c5 = evaluate_h(h, dx)
