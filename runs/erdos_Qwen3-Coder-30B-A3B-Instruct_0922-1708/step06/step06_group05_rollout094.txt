import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    start_time = time.time()

    n_points = 100
    pop_size = 50
    max_generations = 1000
    F = 1.0
    CR = 0.9
    stagnation_limit = 100
    
    # Initialize population

        h = repair_h(h, n_points)
        population.append(h)
    else:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Fill the rest of population with random individuals
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(h, 2.0/n_points) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    stagnation_count = 0
    generation = 0
    
    while generation < max_generations and time.time() - start_time < budget_s - 5:
        new_population = []
        new_fitness = []
        
        for i in range(pop_size):
            # Mutation
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            v = a + F * (b - c)
            v = np.clip(v, 0, 1)
            v = repair_h(v, n_points)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, v, population[i])
            
            # Evaluate trial
            c5 = evaluate_h(trial, 2.0/n_points)
            
            # Selection
            if c5 < fitness[i]:
                new_population.append(trial)
                new_fitness.append(c5)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
                
        # Update population and fitness
        population = new_population
        fitness = new_fitness
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive parameter control
        if stagnation_count > stagnation_limit:
            F = min(F + 0.1, 1.5)
            stagnation_count = 0
            
        generation += 1
    
    # Improve the final solution using gradient-based refinement
    for _ in range(200):

            best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
