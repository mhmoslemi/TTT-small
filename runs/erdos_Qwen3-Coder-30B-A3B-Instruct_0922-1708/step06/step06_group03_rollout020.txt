import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of C5 for the given h and k_max."""
    n_points = len(h)
    grad = np.zeros(n_points)
    for i in range(n_points):
        # For overlap at k_max, contributions come from h[i] * (1 - h[(i-k_max)%n_points])
        # We need to compute d/dh[j] of this contribution
        # For j == i, we get -1 * (1 - h[(i-k_max)%n_points])
        # For j == (i-k_max)%n_points, we get h[i]
        # All others give 0
        if (i - k_max) % n_points < n_points:
            j = (i - k_max) % n_points
            grad[j] += h[i]
        grad[i] -= (1 - h[(i - k_max) % n_points])
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    adaptive_step = 0
    no_improvement_steps = 0
    max_no_improvement = 500
    
    for iteration in range(100000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all overlaps
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        c5_current = np.max(corr) * dx
        
        # Compute gradient for the max overlap
        grad = compute_gradient(current_h, dx, k_max)
        
        # Update with gradient descent
        new_h = current_h - learning_rate * grad
        
        # Apply structured perturbations occasionally to escape local optima
        if iteration % 100 == 0 and no_improvement_steps > 100:
            # Find indices of low gradient
            low_grad_indices = np.where(np.abs(grad) < 0.5 * np.max(np.abs(grad)))[0]
            num_perturb = max(1, len(low_grad_indices) // 3)
            perturb_indices = np.random.choice(low_grad_indices, num_perturb, replace=False)
            for idx in perturb_indices:
                perturbation = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
                new_h[idx] += perturbation
                
            # Reduce perturbation magnitude over time
            perturbation_magnitude *= 0.999
            
        # Repair to satisfy constraints
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Update if better
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            no_improvement_steps = 0
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            no_improvement_steps += 1
            
        # Adaptive learning rate
        adaptive_step += 1
        if adaptive_step % 100 == 0:
            learning_rate *= 0.99
            
        # Reset learning rate if stuck
        if no_improvement_steps > max_no_improvement:
            learning_rate = max(1e-6, learning_rate * 0.5)
            no_improvement_steps = 0
    
    # Final verification
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), n_points)
