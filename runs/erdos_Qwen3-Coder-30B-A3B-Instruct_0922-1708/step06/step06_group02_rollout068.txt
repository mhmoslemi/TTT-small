import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    """Perform arithmetic crossover between two parents."""
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.1):
    """Apply Gaussian mutation to the solution."""
    mask = np.random.rand(len(h)) < mutation_rate
    if np.any(mask):
        noise = np.random.normal(0, noise_std, len(h))
        h = h + mask * noise
    return h
