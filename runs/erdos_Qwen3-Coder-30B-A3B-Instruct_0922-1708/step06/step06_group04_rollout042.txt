import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution from global variable if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.full(n_points, 0.5)
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = repair_h(h_init, n_points)
    
    best_h = h_init.copy()
    best_value = evaluate_h(best_h, dx)
    
    # Gradient-based refinement with adaptive learning rate
    max_iter = 10000
    learning_rate = 0.0001
    stagnation_count = 0
    max_stagnation = 200
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        grad = np.zeros(n_points)
        eps = 1e-6
        current_c5 = evaluate_h(best_h, dx)
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        # Update
        best_h = best_h - learning_rate * grad
        best_h = repair_h(best_h, n_points)
        new_value = evaluate_h(best_h, dx)
        
        if new_value < best_value:
            best_value = new_value
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > 10:
                learning_rate *= 0.95
            if stagnation_count > max_stagnation:
                # Inject diversity
                h_div = np.random.rand(n_points)
                h_div = h_div * (n_points / 2) / np.sum(h_div)
                if evaluate_h(h_div, dx) < best_value:
                    best_h = h_div
                    best_value = evaluate_h(best_h, dx)
                stagnation_count = 0
                learning_rate = max(1e-6, learning_rate * 0.8)
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros(n_points)
        eps = 1e-6
        current_c5 = evaluate_h(best_h, dx)
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        best_h = best_h - 0.00005 * grad
        best_h = repair_h(best_h, n_points)
        new_value = evaluate_h(best_h, dx)
        
        if new_value < best_value:
            best_value = new_value
    
    return (best_h.tolist(), best_value, n_points)
