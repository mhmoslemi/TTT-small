import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    pop_size = 50
    F = 1.0
    CR = 0.9
    max_generations = 1000
    stagnation_limit = 100
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
        population.append(h.copy())
    
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    dx = 2.0 / n_points
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
                
        # Selection and evolution
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:pop_size//2]]
        
        new_population = selected.copy()
        
        # DE mutation and crossover
        for i in range(len(selected)):
            # Select three different individuals
            a, b, c = np.random.choice(len(selected), 3, replace=False)
            a, b, c = selected[a], selected[b], selected[c]
            
            # Mutation
            v = a + F * (b - c)
            v = np.clip(v, 0, 1)
            v = v * (n_points / 2) / np.sum(v)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, v, selected[i])
            new_population.append(trial)
            
        population = new_population
        
        # Check for stagnation
        if stagnation_count > stagnation_limit:
            F = min(1.5, F + 0.1)
            stagnation_count = 0
            
        stagnation_count += 1
        
    # Refinement phase
    if best_h is not None:
        for _ in range(500):
            if time.time() - start_time > budget_s - 2:
                break
            eps = 1e-6
            grad = np.zeros(n_points)
            c5_eval = evaluate_h(best_h, dx)
            for i in range(n_points):
                h_plus = best_h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                grad[i] = (c5_plus - c5_eval) / eps
            best_h = best_h - 0.0002 * grad
            best_h = repair_h(best_h, n_points)
            new_c5 = evaluate_h(best_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                
    return (best_h.tolist(), best_c5, n_points)
