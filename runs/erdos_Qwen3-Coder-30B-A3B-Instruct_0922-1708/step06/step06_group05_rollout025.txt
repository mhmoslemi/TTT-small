import numpy as np
import time
import cvxpy as cp
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    # Use scipy.signal.correlate for efficient computation
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Rescale to meet integral constraint exactly
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Parameters
    lr = 0.001
    max_iter = 50000
    stagnation_limit = 500
    stagnation_count = 0
    decay_factor = 0.95
    
    # Try to use CVXPY for better optimization
    try:
        # Define CVXPY variables
        h_var = cp.Variable(n_points)
        # Objective: minimize sum of squared correlations
        corr = signal.correlate(h_var, 1 - h_var, mode='full')
        # Maximize the correlation (the full correlation is too complex to model directly)
        # Instead, minimize the maximum element of the correlation
        max_corr = cp.Variable()
        
        # The constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Create objective to minimize max of correlations
        # We'll solve a convex relaxation and iterate to improve
        objective = cp.Minimize(max_corr)
        cvx_prob = cp.Problem(objective, constraints)
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Solve the relaxed problem
            try:
                cvx_prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-6)
                if cvx_prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h = h_var.value
                    # Repair h
                    h = repair_h(h, n_points)
                    c5 = evaluate_h(h, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        stagnation_count = 0
                        # Reduce learning rate
                        lr *= decay_factor
                    else:
                        stagnation_count += 1
                else:
                    # Fall back to gradient descent if solver fails
                    break
            except Exception:
                break
                
            if stagnation_count > stagnation_limit:
                # Restart with new initialization
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                stagnation_count = 0
                
    except Exception:
        pass
    
    # Fall back to gradient descent if CVXPY fails
    if not 'cvx_prob' in locals() or cvx_prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        # Authors: Use Adam-like adaptive method
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8
        adam_lr = 0.001
        
        try:
            for iteration in range(max_iter):
                if time.time() - start_time > budget_s - 5:
                    break
                    
                # Compute finite difference gradient
                eps = 1e-5
                grad = np.zeros(n_points)
                h_eval = evaluate_h(h, dx)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = repair_h(h_plus, n_points)
                    c5_plus = evaluate_h(h_plus, dx)
                    grad[i] = (c5_plus - h_eval) / eps
                
                # Adam
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * grad**2
                m_hat = m / (1 - beta1**(iteration+1))
                v_hat = v / (1 - beta2**(iteration+1))
                
                h -= adam_lr * m_hat / (np.sqrt(v_hat) + epsilon)
                h = repair_h(h, n_points)
                
                # Evaluate
                c5 = evaluate_h(h, dx)
                
                # Update best
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                
                # Restart strategy
                if stagnation_count > stagnation_limit:
                    h = np.random.rand(n_points)
                    h = h * (n_points / 2) / np.sum(h)
                    m = np.zeros(n_points)
                    v = np.zeros(n_points)
                    stagnation_count = 0
                    
        except Exception:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
