import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize initial h to correct sum
    h_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(h_sum - target_sum) > 1e-8:
        h = h * target_sum / h_sum
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Iterative approach with cutting plane:
    max_iter = 100
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 5:
        # Compute current overlaps
        corr = correlate(h, 1 - h, mode='full')
        c5_current = np.max(corr) * dx
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        
        # Find worst shift
        max_lag_idx = np.argmax(corr)
        k_max = max_lag_idx - (len(h) - 1)
        
        # Formulate convex subproblem
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Overlap constraint for k_max
        # overlap_k_max = sum h[i] * (1 - h[i+k_max]) for all i
        # Which equals: n_points - sum h[i] * h[i+k_max]  
        # So: 1 - (2/n_points) * sum h[i] * h[i+k_max] <= t
        
        # Build the constraint
        # Need to handle the indexing properly for circular shifts
        constraint_expr = 0
        for i in range(n_points):
            if k_max >= 0:
                j = i + k_max
                if j < n_points:
                    constraint_expr += h_var[i] * h_var[j]
            else:
                j = i + k_max
                if j >= 0:
                    constraint_expr += h_var[i] * h_var[j]
        
        constraints = [
            t >= 1 - (2.0 / n_points) * constraint_expr,
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Solve using CVXPY with ECOS
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'optimal':
                h_new = h_var.value
                h = np.maximum(0, np.minimum(1, h_new))
                h = h * target_sum / np.sum(h)
            else:
                # if not optimal, we try a simple gradient step
                # This is a fallback in case convex solver fails
                pass
        except:
            # Fallback to simple gradient descent if solver fails
            pass
        
        iter_count += 1
        
        # Simple gradient step as fallback
        if iter_count % 5 == 0:
            # Compute current overlap function again
            corr = correlate(h, 1 - h, mode='full')
            max_lag_idx = np.argmax(corr)
            k = max_lag_idx - (len(h) - 1)
            
            # Gradient for a practical update
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = np.clip(h_plus, 0, 1)
                h_plus_sum = np.sum(h_plus)
                if abs(h_plus_sum - target_sum) > 1e-8:
                    h_plus = h_plus * target_sum / h_plus_sum
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5_current) / 1e-5
            
            h = h - 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
