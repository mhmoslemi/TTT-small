import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = (h_init > 0.5).astype(int)
    else:
        # Initialize binary h with equal 0s and 1s
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # Ensure sum(h) = n_points / 2
    current_sum = np.sum(h)
    if current_sum != n_points / 2:
        # Flip some 0s to 1s or 1s to 0s to balance the sum
        if current_sum < n_points / 2:
            # Need to flip more 0s to 1s
            positions = np.where(h == 0)[0]
            for i in range(int(n_points/2 - current_sum)):
                if i < len(positions):
                    h[positions[i]] = 1
        else:
            # Need to flip some 1s to 0s
            positions = np.where(h == 1)[0]
            for i in range(int(current_sum - n_points/2)):
                if i < len(positions):
                    h[positions[i]] = 0
    
    # MIP formulation
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points / 2]
    
    # For each shift k, add constraint
    for k in range(-n_points+1, n_points):
        # find valid indices
        valid_indices = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_indices.append((i, j))
        
        if valid_indices:
            # Build the sum of h[i] * (1 - h[j]) for valid pairs
            expr = sum(h_var[i] * (1 - h_var[j]) for i, j in valid_indices)
            constraints.append(expr <= t * (n_points / 2))
    
    # Objective
    objective = cp.Minimize(t)
    
    # Create the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with CBC solver
    try:
        prob.solve(solver=cp.CBC, max_seconds=budget_s - 10, gap_limit=0.01)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_result = h_var.value
            if h_result is not None:
                h_binary = np.round(h_result).astype(int)
                # Ensure exact sum
                current_sum = np.sum(h_binary)
                if current_sum != n_points / 2:
                    # Adjust the binary solution to fix the sum constraint
                    positions_0 = np.where(h_binary == 0)[0]
                    positions_1 = np.where(h_binary == 1)[0]
                    if current_sum < n_points / 2:
                        # Need to flip more 0s to 1s
                        num_flip = int(n_points/2 - current_sum)
                        for i in range(num_flip):
                            if i < len(positions_0):
                                h_binary[positions_0[i]] = 1
                    else:
                        # Need to flip some 1s to 0s
                        num_flip = int(current_sum - n_points/2)
                        for i in range(num_flip):
                            if i < len(positions_1):
                                h_binary[positions_1[i]] = 0
                h_final = h_binary
            else:
                h_final = h
        else:
            h_final = h
    except:
        h_final = h
    
    # Final verification
    corr = correlate(h_final, 1 - h_final, mode='full')
    c5_final = np.max(corr) * dx
    
    return (h_final.tolist(), c5_final, n_points)
