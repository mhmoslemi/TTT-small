import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):

    n = len(h)

        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = repair_h(h_plus, n)
        h_minus = repair_h(h_minus, n)
        c5_plus = evaluate_h(h_plus, dx)
        c5_minus = evaluate_h(h_minus, dx)
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def trust_region_step(h, dx, grad, H, trust_radius):
    """Solve trust-region subproblem to find step d."""
    n = len(h)
    # Simple trust region solver using projected gradient descent
    d = -grad
    # Project d to satisfy constraints
    d = np.clip(d, -h, 1-h)
    # Project d to satisfy sum constraint
    d_sum = np.sum(d)
    d = d - d_sum / n
    # Scale to trust region size
    d_norm = np.linalg.norm(d)
    if d_norm > trust_radius:
        d = d * trust_radius / d_norm
    return d
