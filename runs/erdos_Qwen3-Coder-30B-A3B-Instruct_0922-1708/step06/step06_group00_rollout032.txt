import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure feasibility
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-8:
        h = h * (n_points / 2) / h_sum
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Cutting plane approach
    max_iter = 300
    patience = 10
    no_improve_count = 0
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current overlaps
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > patience:
            break
            
        # Find the maximum overlap lag
        max_lag_idx = np.argmax(corr)
        k_max = max_lag_idx - (len(h) - 1)
        
        # For the subset of indices affected
        indices = np.arange(len(h))
        shift_indices = indices + k_max
        valid_indices = (shift_indices >= 0) & (shift_indices < len(h))
        
        # Construct linear program for this constraint
        # Use CVXPY for subproblem
        h_var = cp.Variable(len(h))
        t = cp.Variable()
        
        # Objective
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = []
        # Feasibility constraints
        constraints.append(cp.sum(h_var) == n_points / 2)
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # The key constraint from the worst overlap at k_max
        # overlap_k_max = sum h_i * (1 - h_{i+k_max}) 
        # = sum h_i - sum h_i * h_{i+k_max}
        # Set this less than or equal to t:
        # sum h_i - sum h_i * h_{i+k_max} <= t
        # sum h_i * h_{i+k_max} >= sum h_i - t
        
        # Build the quadratic constraint properly
        if k_max >= 0:
            # h_i * h_{i+k_max} where i+k_max < len
            xi = h_var[:len(h)-k_max]
            yi = h_var[k_max:]
        else:
            # h_i * h_{i+k_max} where i+k_max >= 0 (k_max < 0)
            xi = h_var[-k_max:]
            yi = h_var[:len(h)+k_max] 
        
        constraint_expr = cp.sum(cp.multiply(xi, yi))
        constraints.append(constraint_expr >= cp.sum(h_var) - t)
        
        # Formulate and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                new_h = h_var.value
                if new_h is not None:
                    h = np.array(new_h)
                    # Re-scale to meet sum constraint
                    h_sum = np.sum(h)
                    h = h * (n_points / 2) / h_sum
                    h = np.clip(h, 0, 1)
        except Exception:
            pass
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
