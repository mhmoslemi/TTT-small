import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of C5 w.r.t h."""
    n_points = len(h)
    grad = np.zeros(n_points)
    for i in range(n_points):
        grad[i] = (1 - h[(i - k_max) % n_points])
    return grad * dx

    dx = 2.0 / n_points

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    max_iter = 100000
    improvement_steps = 0
    max_improvement_steps = 200
    
    # Gradient descent with adaptive perturbations
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlation and find maximum overlap
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)

        grad = compute_gradient(current_h, dx, k_max)

                for idx in perturb_indices:

        current_h = current_h - learning_rate * grad
        
        # Repair to maintain constraints
        current_h = repair_h(current_h, n_points)
        
        current_c5 = evaluate_h(current_h, dx)

        if current_c5 < best_c5:

            improvement_steps = 0
