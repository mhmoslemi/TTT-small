import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5

    # Dual ascent approach with projected gradient descent
    lambda_k = np.zeros(2 * n_points - 1)  # dual variables
    learning_rate = 5e-4
    max_iter = 50000
    progress_count = 0
    max_progress = 500
    dual_iter = 0
    dual_update_freq = 10

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break

        # Dual update step
        if dual_iter % dual_update_freq == 0:
            # Compute correlations for all lags
            corr_full = correlate(current_h, 1 - current_h, mode='full')
            # Update dual variables to promote optimality
            lambda_k = np.maximum(lambda_k + learning_rate * (corr_full - np.max(corr_full)), 0)
            lambda_k = np.clip(lambda_k, 0, 1e6)  # prevent extreme values

        # Solve for optimal h with fixed lambda
        # Gradient of Lagrangian: (2/n_points) * (lambda * (1-h) convolved with 1)
        # gradient_h[i] = (2/n_points) * sum_k lambda_k * (1 - h[i-k] if k <= i else 0)
        grad = np.zeros(n_points)
        for k in range(2 * n_points - 1):
            if k < n_points:
                # k-th element of 1-h conv h
                if k < n_points:
                    grad[k] += lambda_k[k] * (1 - current_h[0])
                else:
                    grad[0] += lambda_k[k] * (1 - current_h[k - n_points + 1])
            else:
                # k-th element where k >= n_points
                if k >= n_points:
                    j = k - n_points + 1
                    if j < n_points:
                        grad[j] += lambda_k[k] * (1 - current_h[0])

        # Projected gradient descent
        current_h = current_h - learning_rate * grad
        current_h = repair_h(current_h, n_points)
        
        # Recompute current bound
        current_c5 = evaluate_h(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            progress_count = 0
        else:
            progress_count += 1

        # Adaptive learning rate
        if progress_count > max_progress:
            learning_rate *= 0.99
            progress_count = 0
        
        dual_iter += 1

    # Final refinement
    for refinement_step in range(100):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps

        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx

    return (best_h.tolist(), best_c5, n_points)
