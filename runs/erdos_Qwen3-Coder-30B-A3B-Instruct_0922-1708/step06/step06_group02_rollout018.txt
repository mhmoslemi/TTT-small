import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def crossover(parent1, parent2):
    """Perform crossover between two parents."""
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1):
    """Mutate a solution."""
    h = h.copy()
    mask = np.random.rand(len(h)) < mutation_rate
    if np.any(mask):
        noise = np.random.normal(0, 0.05, len(h))
        h[mask] += noise[mask]
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    elite_size = population_size // 5
    mutation_rate = 0.1
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init.copy()]
    else:
        population = []
    
    # Fill population with random feasible solutions
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Evaluate initial population
    fitness_scores = []
    for h in population:
        score = evaluate_h(h, dx)
        fitness_scores.append(score)
    
    # Find best solution
    best_idx = np.argmin(fitness_scores)
    best_h = population[best_idx].copy()
    best_value = fitness_scores[best_idx]
    
    # Evolution loop
    max_generations = 1000
    no_improvement_count = 0
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Keep elite
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        while len(new_population) < population_size:
            # Selection - tournament
            tournament_size = 5
            selected_indices = np.random.choice(population_size, tournament_size)
            selected_fitness = [fitness_scores[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitness)]
            
            # Crossover with random parent
            parent1 = population[winner_idx]
            parent2 = population[np.random.randint(0, population_size)]
            child = crossover(parent1, parent2)
            
            # Mutation
            child = mutate(child, mutation_rate)
            
            # Repair
            child = repair_h(child, n_points)
            
            new_population.append(child)
        
        # Replace population
        population = new_population
        
        # Evaluate new population
        fitness_scores = []
        for h in population:
            score = evaluate_h(h, dx)
            fitness_scores.append(score)
            
            if score < best_value:
                best_value = score
                best_h = h.copy()
        
        # Adaptive mutation rate
        if abs(best_value - fitness_scores[0]) < 1e-10:
            no_improvement_count += 1
            if no_improvement_count > 10:
                mutation_rate = min(mutation_rate * 1.1, 0.5)
        else:
            no_improvement_count = 0
            mutation_rate = max(mutation_rate * 0.95, 0.01)
            
        # Gradient refinement on best
        if gen % 10 == 0:
            refined_h = best_h.copy()
            current_value = best_value
            
            for _ in range(20):
                if time.time() - start_time > budget_s - 10:
                    break
                    
                eps = 1e-6
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = refined_h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_value) / eps
                
                refined_h = refined_h - 0.001 * grad
                refined_h = repair_h(refined_h, n_points)
                corr = correlate(refined_h, 1 - refined_h, mode='full')
                current_value = np.max(corr) * dx
                
                if current_value < best_value:
                    best_value = current_value
                    best_h = refined_h.copy()
    
    return (best_h.tolist(), best_value, n_points)
