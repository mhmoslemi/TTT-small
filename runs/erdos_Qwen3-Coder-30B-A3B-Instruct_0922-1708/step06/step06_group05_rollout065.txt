import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound using full correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    population = []
    
    if 'initial_h_values' in globals():

        x_orig = np.linspace(0, 2, len(h_init))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_init)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Populate the rest with random samples
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # DE parameters
    F = 1.0
    CR = 0.9
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    max_stagnation = 100
    
    generation = 0
    while time.time() - start_time < budget_s - 5 and stagnation_count < max_stagnation:
        generation += 1
        fitness = []
        new_population = []
        
        for i in range(pop_size):
            # Select three distinct individuals
            indices = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            v = a + F * (b - c)
            v = np.clip(v, 0, 1)
            v = repair_h(v, n_points)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, v, population[i])
            
            # Evaluate trial
            c5_trial = evaluate_h(trial, dx)
            fitness.append(c5_trial)
            
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = trial.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
                
            new_population.append(trial)
            
        # Selection step: keep best individuals
        fitness = np.array(fitness)
        selected_indices = np.argsort(fitness)[:pop_size]
        population = [new_population[i] for i in selected_indices]
        
        # Adaptive parameter adjustment
        if generation % 50 == 0 and F < 1.5:
            F += 0.1
    
    # Refined optimization
    if best_h is not None:
        for _ in range(100):
            if time.time() - start_time > budget_s - 2:
                break
            eps = 1e-6
            grad = np.zeros(n_points)
            c5_eval = evaluate_h(best_h, dx)
            for i in range(n_points):
                h_plus = best_h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                grad[i] = (c5_plus - c5_eval) / eps
            best_h = best_h - 0.0001 * grad
