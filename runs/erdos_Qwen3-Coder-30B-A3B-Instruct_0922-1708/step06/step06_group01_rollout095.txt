import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.linalg import norm

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    n = len(h)
    grad = np.zeros(n)
    h_base = h.copy()
    for i in range(n):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        
        h_minus = h_base.copy()
        h_minus[i] -= eps
        h_minus = repair_h(h_minus, n)
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        c5_minus = np.max(corr_minus) * dx
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def trust_region_step(h, dx, grad, trust_radius, max_iter=100):
    n = len(h)
    h_new = h.copy()
    
    # Project gradient to feasible set
    # Gradient direction within box constraints [0,1]^n
    # Gradient with ∑h = n/2 constraint
    # Use scaled gradient, not scaled projection
    
    # Simple fixed-size step
    step_size = min(trust_radius, 1.0)
    h_new = h - step_size * grad
    h_new = repair_h(h_new, n)
    
    return h_new

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Use Halton sequence as initial
        # Construct low-discrepancy sequence
        h_init = np.zeros(n_points)
        for i in range(n_points):
            h_init[i] = (i + 1) / n_points
        h_init = repair_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Trust region parameters
    trust_radius = 1.0
    alpha = 0.5  # shrink factor
    beta = 2.0   # expand factor
    max_iter = 2000
    perturb_freq = 250
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time < budget_s - 5):
        if iteration % perturb_freq == 0 and iteration > 0:
            # Add small random perturbation
            pert = np.random.normal(0, 0.01, n_points)
            h_current = h_current + pert
            h_current = repair_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current

        # Compute gradient
        grad = compute_gradient(h_current, dx)
        
        # Trust-region step
        h_new = trust_region_step(h_current, dx, grad, trust_radius)
        
        # Evaluate new point
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            trust_radius *= beta
        else:
            trust_radius *= alpha
            
        trust_radius = max(trust_radius, 1e-6)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            
        iteration += 1

    # Local refinement (simple gradient descent)
    h_final = h_best.copy()
    for _ in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = compute_gradient(h_final, dx)
        h_final = h_final - 0.001 * grad
        h_final = repair_h(h_final, n_points)
        corr = correlate(h_final, 1 - h_final, mode='full')
        c5_final = np.max(corr) * dx
        if c5_final < c5_best:
            h_best = h_final.copy()
            c5_best = c5_final

    return (h_best.tolist(), c5_best, n_points)
