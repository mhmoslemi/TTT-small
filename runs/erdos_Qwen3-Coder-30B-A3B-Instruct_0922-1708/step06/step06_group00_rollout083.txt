import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Iterative cutting-plane method
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the maximum overlap index
        max_lag_idx = np.argmax(corr)
        k_max = max_lag_idx - (len(h) - 1)
        
        # Create optimization variables
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Build constraints for maximum overlap
        # overlap_k = sum h[i] * (1 - h[i+k]) for all valid i
        # For numerical stability, we use the equivalent:
        # overlap_k = 1 - sum h[i] * h[i+k]
        # We want: overlap_k <= t for the max overlap shift k_max
        
        expr = 0
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                expr += h_var[i] * h_var[j]
        
        # Construct the constraint
        objective = cp.Minimize(t)
        constraints = [
            t >= 1 - expr,
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Solve the subproblem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_new = h_var.value
            if h_new is not None:
                h = h_new.copy()
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if abs(h_sum - n_points / 2) > 1e-8:
                    h = h * (n_points / 2) / h_sum
        except:
            # fallback
            pass
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
