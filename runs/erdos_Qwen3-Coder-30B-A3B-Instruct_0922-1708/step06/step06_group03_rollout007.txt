import numpy as np
import time
from scipy.signal import correlate

def repair_h(h, n_points):
    """Repair h to satisfy constraints: h[i] in [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_overlap(h, dx):
    """Compute the overlap integral."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    h = repair_h(h, n_points)
    
    # Dual variables for lag indices
    lambda_lag = np.zeros(2 * n_points - 1)
    
    # Parameters
    step_size = 5e-4
    decay_rate = 0.99
    max_iter = 50000
    patience = 100
    patience_counter = 0
    
    best_h = h.copy()
    best_c5 = compute_overlap(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        
        # Update dual variables based on violation
        lag_index = np.argmax(corr)
        lambda_lag[lag_index] += step_size * (corr[lag_index] - best_c5)
        lambda_lag[lag_index] = max(0, lambda_lag[lag_index])
        
        # Compute gradient of Lagrangian with respect to h
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(2 * n_points - 1):
                if 0 <= i - (k - n_points + 1) < n_points:
                    grad[i] += lambda_lag[k] * (1 - h[i - (k - n_points + 1)])
        
        # Update h
        h = h - step_size * grad
        h = repair_h(h, n_points)
        
        # Compute current C5
        c5 = compute_overlap(h, dx)
        
        # Update best solution
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adapt step size
        if iteration % 500 == 0:
            step_size *= decay_rate
            
        # Early stopping
        if patience_counter > patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
