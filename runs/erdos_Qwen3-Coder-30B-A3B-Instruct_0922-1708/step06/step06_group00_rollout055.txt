import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Ensure h is feasible
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Iterative cutting plane
    max_iterations = 500
    subproblem_timeout = 5.0  # seconds
    
    for iteration in range(max_iterations):
        if (time.time() - start_time) > budget_s - 10:
            break
        
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the k that maximizes overlap
        max_lag_idx = np.argmax(corr)
        k_max = max_lag_idx - (len(h) - 1)
        
        # Formulate and solve subproblem using CVXPY
        # Variables
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Overlap constraint for maximum k
        overlap_constraint = cp.sum(
            cp.multiply(h_var, cp.roll(1 - h_var, k_max))
        ) >= (1 - t) * n_points / 2
        
        constraints.append(overlap_constraint)
        
        # Objective
        objective = cp.Minimize(t)
        
        # Formulate and solve
        prob = cp.Problem(objective, constraints)
        
        solve_start = time.time()
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                h = np.clip(h_new, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            pass
            
        if time.time() - solve_start > subproblem_timeout:
            break
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
