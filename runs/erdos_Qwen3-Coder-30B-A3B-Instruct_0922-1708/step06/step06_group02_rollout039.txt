import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_gradient(h, k, dx, n_points):
    """Compute gradient of overlap at shift k."""
    grad = np.zeros(n_points)
    # For a given shift k, we look at the contribution to the integral
    # Note: to avoid boundary issues, we compute direct finite differences
    for i in range(n_points):
        if 0 <= i + k < n_points:
            grad[i] -= (1 - h[i + k])  # Chain rule
        if 0 <= i - k < n_points:
            grad[i] -= h[i - k]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # For gradient-based approach, we need to refine our idea
    # Use binary search to find critical shifts (worst overlaps)
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Grid for shifts
    num_shifts = 100
    shift_grid = np.linspace(-2, 2, num_shifts)
    
    # Optimization parameters
    learning_rate = 0.005
    min_lr = 1e-6
    patience = 100
    patience_counter = 0
    last_best = best_value
    
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        
        # Compute all correlations 
        corr_full = correlate(h, 1 - h, mode='full')
        # Compute correlation values at each shift in the grid
        corr_values = []
        center_idx = len(corr_full) // 2  # Corresponds to k=0
        for k in shift_grid:
            k_idx = int(k / (2.0 / n_points)) + center_idx
            if 0 <= k_idx < len(corr_full):
                corr_values.append(corr_full[k_idx])
            else:
                corr_values.append(0)
        corr_values = np.array(corr_values)
        
        # Find the worst shift
        worst_k_idx = np.argmax(corr_values)
        worst_k = shift_grid[worst_k_idx]
        
        # Compute gradient using finite differences
        # Small perturbation to find the gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_plus_val = corr_plus[worst_k_idx + center_idx] if 0 <= worst_k_idx + center_idx < len(corr_plus) else 0
            grad[i] = (corr_plus_val - corr_full[worst_k_idx + center_idx]) / eps
            
        # Update h
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        
        # Evaluate new solution
        new_value = evaluate_h(h, dx)
        
        if new_value < best_value:
            best_value = new_value
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Reduce learning rate if no improvement
        if patience_counter >= patience:
            learning_rate *= 0.5
            patience_counter = 0
            if learning_rate < min_lr:
                break
                
        # Add slight random perturbation occasionally to escape local minima
        if iteration % 50 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = h + noise
            h = repair_h(h, n_points) 
            
    return (best_h.tolist(), best_value, n_points)
