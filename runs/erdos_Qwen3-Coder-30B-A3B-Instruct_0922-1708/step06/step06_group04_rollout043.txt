import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.full(n_points, 0.5)
        # Add small perturbations to break symmetry
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = repair_h(h_init, n_points)
    
    # Start with the best known solution or initial
    h = h_init.copy()
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Gradient descent with adaptive learning rate
    max_iter = 10000
    learning_rate = 0.001
    eps = 1e-6
    last_improvement = 0
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute finite differences gradient
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(h, dx)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        # Update step
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        
        # Check if this is an improvement
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_value:
            best_h = h.copy()
            best_value = new_c5
            last_improvement = iter_count
            # Decrease learning rate on improvement
            learning_rate *= 0.995
        
        # Check for stagnation
        if iter_count - last_improvement > 200:
            # Inject small random perturbations
            h = h + np.random.normal(0, 0.001, n_points)
            h = repair_h(h, n_points)
            learning_rate = max(0.00001, learning_rate * 0.95)
            last_improvement = iter_count
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros(n_points)
        current_c5 = evaluate_h(best_h, dx)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_value:
            best_value = new_c5
    
    return (best_h.tolist(), best_value, n_points)
