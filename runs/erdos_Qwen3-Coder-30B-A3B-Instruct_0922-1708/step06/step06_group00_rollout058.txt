import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Random binary approximation with sum constraint
        h = np.random.rand(n_points) > 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Iterative cutting plane approach
    max_iterations = 500
    iter_count = 0
    
    while (time.time() - start_time) < budget_s - 5 and iter_count < max_iterations:
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        overlap_values = corr * dx
        max_overlap_idx = np.argmax(overlap_values)
        max_k = max_overlap_idx - (n_points - 1)
        max_overlap = overlap_values[max_overlap_idx]
        
        # Check if we've improved
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Use CVXPY to solve subproblem
        try:
            # Create variables
            h_var = cp.Variable(n_points)
            t_var = cp.Variable(1)
            
            # Objective
            objective = cp.Minimize(t_var)
            
            # Constraints
            constraints = []
            
            # Sum constraint
            constraints.append(cp.sum(h_var) == n_points / 2)
            
            # Bound constraints
            constraints.append(h_var >= 0)
            constraints.append(h_var <= 1)
            
            # Worst-case shift constraint
            # overlap_k = sum h[i] * (1 - h[i + k]) for all i
            # overlap_k = sum h[i] - sum h[i] * h[i + k] = 1 - sum h[i] * h[i + k]
            term = 0
            for i in range(n_points):
                i_k = (i + max_k) % n_points
                term += h_var[i] * h_var[i_k]
            constraints.append(1 - term <= t_var)
            
            # Form and solve problem
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h = h_var.value
                # Ensure feasibility
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                h = h * (n_points / 2) / h_sum
                
                # Apply small random perturbation to escape local minima
                if iter_count % 5 == 0:
                    noise = np.random.normal(0, 0.01, n_points)
                    h = h + noise
                    h = np.clip(h, 0, 1)
                    h_sum = np.sum(h)
                    h = h * (n_points / 2) / h_sum
        except:
            # Fallback: gradient-based update if optimization fails
            # Compute the gradient direction for the worst overlap
            # Use finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                if h_plus[i] > 1:
                    h_plus[i] = 1
                h_plus_sum = np.sum(h_plus)
                h_plus = h_plus * (n_points / 2) / h_plus_sum
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - max_overlap) / eps
            
            # Update
            h = h - 0.001 * grad
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            
        iter_count += 1
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
