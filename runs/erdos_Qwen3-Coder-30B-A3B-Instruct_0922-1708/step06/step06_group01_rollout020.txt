import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute numerical gradient of C5 with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    h_base = h.copy()
    c5_base = evaluate_h(h_base, dx)
    
    for i in range(n):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

def trust_region_step(h, dx, trust_radius=1.0, max_iter=100):
    """Compute a step using trust region method."""
    n = len(h)
    grad = compute_gradient(h, dx)
    
    # Try simple gradient descent within trust region
    step = -grad
    step_norm = np.linalg.norm(step)
    if step_norm > trust_radius:
        step = step * trust_radius / step_norm
    
    h_new = h + step
    h_new = repair_h(h_new, n)
    return h_new
