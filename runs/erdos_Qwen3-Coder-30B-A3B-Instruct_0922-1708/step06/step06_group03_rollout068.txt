import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, max_lag_idx):
    """Compute gradient of C5 with respect to h."""
    n_points = len(h)
    grad = np.zeros(n_points)
    
    # For each position j, compute contribution to gradient
    for j in range(n_points):
        # Find the maximum overlap lag that involves position j
        # In our case, that depends on k such that j = (k + lag) % n_points  
        # which means lag = (j - k) % n_points  
        # But max_lag_idx is the index in the full correlation array
        # The actual time lag is: (max_lag_idx - n_points + 1) mod (2*n_points - 1) - (n_points - 1)
        # Simplifying: (max_lag_idx - n_points + 1)
        # However, we can directly consider the contribution from specific k_max
        k_max = max_lag_idx - (n_points - 1)
        # Compute k_max corresponding position in array
        # dist from j to the other element in convolution (for maximum)
        # It's the element at (j - k_max) mod n_points
        # In the correlation, the entries are indexed by k with k = -n+1 to n-1  
        # So to make our arithmetic easier, let's work directly
        conv_pos = max_lag_idx
        lag_pos = conv_pos - (n_points - 1)  # Lag corresponds to positioning
        if lag_pos >= 0:
            # Lag is for forward shift
            # Write: C5 = sum_i h[i] * (1 - h[i - lag_pos]) if i >= lag_pos
            #        = sum_i h[i] * (1 - h[i + n_points - lag_pos]) if i < lag_pos  
            # But j is the current position where we compute gradient
            # hence we care about contribution to: h[j] from term involving h[j - lag_pos]
            prev_idx = (j - lag_pos) % n_points
            grad[j] = 1 - h[prev_idx]
        else:
            # Lag is backward shift
            # Term is h[j] * (1 - h[j + lag_pos]) where lag_pos < 0 
            prev_idx = (j + abs(lag_pos)) % n_points
            grad[j] = 1 - h[prev_idx]

    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Parameters
    learning_rate = 1e-4
    perturbation_rate = 0.05
    max_iter = 100000
    no_improvement_count = 0
    max_no_improvement = 2000
    perturb_freq = 500
    lr_decay = 0.99
    perturb_decay = 0.999
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlation with current h
        corr = correlate(current_h, 1 - current_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_corr_value = corr[max_lag_idx] * dx

        # Compute gradient of max value
        grad = compute_gradient(current_h, dx, max_lag_idx)
        
        # Update h using gradient descent
        h_new = current_h - learning_rate * grad
        
        # Perturb every perturb_freq steps if no improvement
        if i % perturb_freq == 0 and no_improvement_count > 50:
            # Perturb with small stochastic component
            perturbation = np.random.uniform(-perturbation_rate, perturbation_rate, n_points)
            h_new += perturbation
        
        # Repair new h
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = h_new
            current_c5 = new_c5
            no_improvement_count = 0
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            no_improvement_count += 1
            
        # Adaptive adjustment
        if no_improvement_count > max_no_improvement:
            # Reset learning rate and perturbation
            learning_rate = 1e-4
            perturbation_rate = 0.05
            no_improvement_count = 0
        
        # Decaying parameters
        if i % 100 == 0:
            learning_rate *= lr_decay
            perturbation_rate *= perturb_decay

    return (best_h.tolist(), best_c5, n_points)
