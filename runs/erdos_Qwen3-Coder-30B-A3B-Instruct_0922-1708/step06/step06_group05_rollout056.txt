import numpy as np
import time
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Convex relaxation stage
    if time.time() - start_time < budget_s / 3:
        h_cvx = cp.Variable(n_points)
        h_cvx.value = h
        
        # Solve surrogate convex optimization problem
        objective = cp.Minimize(
            cp.sum_squares(cp.convolve(h_cvx, 1 - h_cvx, 'full') * dx) +
            1e-3 * cp.max(cp.convolve(h_cvx, 1 - h_cvx, 'full') * dx)
        )
        constraints = [
            cp.sum(h_cvx) == n_points / 2,
            h_cvx >= 0,
            h_cvx <= 1
        ]
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if h_cvx.value is not None:
                h = np.array(h_cvx.value).flatten()
                h = repair_h(h, n_points)
                c5 = evaluate_h(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            pass

    # Gradient-based refinement
    lr = 0.01
    beta = 0.9
    max_iter = 5000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Adaptive learning rate
        lr_adapt = lr / (1 + (iteration ** beta))
        
        # Compute finite difference gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        h_eval = evaluate_h(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - h_eval) / eps
        
        # Update with adaptive LR and projection
        h = h - lr_adapt * grad
        h = repair_h(h, n_points)
        
        # Evaluate
        c5 = evaluate_h(h, dx)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final polishing with smaller learning rate
    for _ in range(500):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
