import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Proportional adjustment to meet target sum
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    top_candidates = 10
    max_generations = 1000
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init.copy()]
    else:
        population = []
    
    # Fill remaining population
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Track best solution
    best_h = None
    best_value = float('inf')
    
    # Evaluate initial population
    fitness = []
    for h in population:
        value = evaluate_h(h, dx)
        fitness.append(value)
        if value < best_value:
            best_value = value
            best_h = h.copy()
    
    # Genetic algorithm parameters
    crossover_rate = 0.5
    mutation_rate = 0.1
    mutation_std = 0.1
    
    # Main loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        idxs = np.argsort(fitness)
        sorted_pop = [population[i] for i in idxs]
        sorted_fitness = [fitness[i] for i in idxs]
        
        # Keep top performers
        new_pop = sorted_pop[:top_candidates].copy()
        
        # Generate offspring
        while len(new_pop) < population_size:
            if np.random.rand() < crossover_rate:
                # Crossover
                parent1_idx = np.random.randint(0, top_candidates)
                parent2_idx = np.random.randint(0, top_candidates)
                parent1 = sorted_pop[parent1_idx]
                parent2 = sorted_pop[parent2_idx]
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                # Mutation
                parent_idx = np.random.randint(0, top_candidates)
                parent = sorted_pop[parent_idx]
                child = parent.copy()
                # Add Gaussian noise to non-zero elements
                mask = child > 1e-8
                noise = np.random.normal(0, mutation_std, n_points)
                child[mask] += noise[mask]
                
            child = repair_h(child, n_points)
            new_pop.append(child)
            
        # Evaluate new population
        fitness = []
        for h in new_pop:
            value = evaluate_h(h, dx)
            fitness.append(value)
            if value < best_value:
                best_value = value
                best_h = h.copy()
                
        # Replace population
        population = new_pop
        
        # Gradient refinement of best solution
        if gen % 10 == 0:
            h_refined = best_h.copy()
            current_value = best_value
            
            # Local optimization
            for _ in range(50):
                if time.time() - start_time > budget_s - 10:
                    break
                eps = 1e-5
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_value) / eps
                    
                h_refined = h_refined - 0.0001 * grad
                h_refined = repair_h(h_refined, n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                current_value = np.max(corr) * dx
                
                if current_value < best_value:
                    best_value = current_value
                    best_h = h_refined.copy()
    
    # Final gradient refinement on best solution
    if best_h is not None:
        h_refined = best_h.copy()
        current_value = best_value
        
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
                
            h_refined = h_refined - 0.0001 * grad
            h_refined = repair_h(h_refined, n_points)
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            current_value = np.max(corr) * dx
            
            if current_value < best_value:
                best_value = current_value
                best_h = h_refined.copy()
    
    return (best_h.tolist(), best_value, n_points)
