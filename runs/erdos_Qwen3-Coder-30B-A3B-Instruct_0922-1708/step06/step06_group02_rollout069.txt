import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    """Create a child from two parents using uniform crossover."""
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, mutation_rate=0.1, noise_std=0.1):
    """Mutate a candidate by adding Gaussian noise."""
    h = h.copy()
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, noise_std, len(h))
    h[mask] += noise[mask]
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial population
    pop_size = 50
    elite_size = int(0.2 * pop_size)
    mutation_rate = 0.1
    noise_std = 0.1
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init.copy()]
    else:
        population = []
        
    # Fill initial population
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    # Genetic algorithm loop
    generations = 1000
    no_improvement = 0
    max_no_improvement = 100
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Early termination if no improvement
        if fitness[0] < best_value:
            best_h = population[0].copy()
            best_value = fitness[0]
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > max_no_improvement:
                break
            
        # Create new population
        new_population = population[:elite_size]  # Keep elite
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection (tournament)
            tournament_size = 5
            idx1 = np.random.choice(elite_size, tournament_size)
            idx2 = np.random.choice(elite_size, tournament_size)
            parent1 = population[np.argmin([fitness[i] for i in idx1])]
            parent2 = population[np.argmin([fitness[i] for i in idx2])]
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutation
            child = mutate(child, mutation_rate, noise_std)
            
            # Repair
            child = repair_h(child, n_points)
            
            new_population.append(child)
            
        population = new_population
        
        # Evaluate new population
        fitness = [evaluate_h(h, dx) for h in population]
        
        # Adaptive parameters
        if gen % 50 == 0 and gen > 0:
            improvement = np.mean(fitness[:elite_size]) - np.mean(fitness[elite_size:2*elite_size])
            if improvement < 0.0001:
                mutation_rate *= 1.2
                noise_std *= 1.2
            else:
                mutation_rate = max(0.01, mutation_rate * 0.9)
                noise_std = max(0.01, noise_std * 0.9)
    
    # Final gradient refinement on best solution
    h_refined = best_h.copy()
    current_value = best_value
    
    # Inner loop for gradient refinement
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-5
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - current_value) / eps
            
        h_refined = h_refined - 0.001 * grad
        h_refined = repair_h(h_refined, n_points)
        current_value = evaluate_h(h_refined, dx)
        
        if current_value < best_value:
            best_value = current_value
            best_h = h_refined.copy()
    
    return (best_h.tolist(), best_value, n_points)
