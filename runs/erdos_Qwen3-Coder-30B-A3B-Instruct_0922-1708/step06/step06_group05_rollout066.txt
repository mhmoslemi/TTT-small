import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Parameters for DE
    pop_size = 50
    F = 1.0
    CR = 0.9
    max_generations = 2000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h0))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h0)
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    # Fill remaining population
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    population = np.array(population)
    
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate current population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = population[sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Create new population
        new_population = [population[0]]  # Elitism
        
        for i in range(1, pop_size):
            # Select three different individuals
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Differential mutation
            v = a + F * (b - c)
            v = np.clip(v, 0, 1)
            v = repair_h(v, n_points)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = v[j]
            
            # Ensure constraint satisfaction
            trial = repair_h(trial, n_points)
            
            new_population.append(trial)
            
        population = np.array(new_population)
        stagnation_count += 1
        if stagnation_count > 100 and F < 1.5:
            F += 0.1
            stagnation_count = 0
        elif stagnation_count > 100:
            break
            
    # Final intensive refinement
    for _ in range(200):
        if time.time() - start_time > budget_s - 2:
            break
        # Gradient descent around best solution
        eps = 1e-6
        grad = np.zeros(n_points)
        h_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - h_eval) / eps
        best_h = best_h - 0.0002 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
    return (best_h.tolist(), best_c5, n_points)
