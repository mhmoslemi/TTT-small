import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def halton_sequence(n, base=2):
    """Generate Halton sequence for n points."""
    result = np.zeros(n)
    for i in range(n):
        f = 1.0
        r = 0.0
        while i > 0:
            f /= base
            r += (i % base) * f
            i //= base
        result[i] = r
    return result

def generate_initial_h(n_points, seed=42):
    """Generate a structured initial function using Halton sequence."""
    np.random.seed(seed)
    # Generate Halton sequence in [0, 2]
    halton = halton_sequence(n_points)
    h = np.interp(halton, [0, 1], [0, 2])
    # Map to [0,1] via sigmoid to make smooth
    h = 1 / (1 + np.exp(-10*(h-1)))
    return repair_h(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Try multiple initial seeds
    seeds = [seed + i for i in range(10)]
    initial_h_list = []
    
    # Generate initial solutions
    for s in seeds:
        if time.time() - start_time > budget_s - 10:
            break
        h = generate_initial_h(n_points, s)
        initial_h_list.append(h)
    
    # Optimization loop
    for h_init in initial_h_list:
        if time.time() - start_time > budget_s - 10:
            break
            
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
        
        # Gradient-based refinement
        iter_count = 0
        max_iter = 5000
        step_size = 0.01
        
        while iter_count < max_iter and time.time() - start_time < budget_s - 10:
            # Compute gradient using finite differences
            eps = 1e-6
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_current.copy()
                h_minus[i] -= eps
                h_minus = repair_h(h_minus, n_points)
                c5_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Gradient descent step
            h_new = h_current - step_size * grad
            h_new = repair_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept the step if it improves
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
            
            iter_count += 1
    
    # Final validation
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to initial construction if available
        if 'initial_h_values' in globals():
            h_fallback = np.array(initial_h_values)
            if len(h_fallback) != n_points:
                x_orig = np.linspace(0, 2, len(h_fallback))
                x_target = np.linspace(0, 2, n_points)
                h_fallback = np.interp(x_target, x_orig, h_fallback)
            h_fallback = repair_h(h_fallback, n_points)
            c5_fallback = evaluate_h(h_fallback, dx)
            return (h_fallback.tolist(), c5_fallback, n_points)
        else:
            # Create random initialization
            h_rand = np.random.rand(n_points)
            h_rand = repair_h(h_rand, n_points)
            c5_rand = evaluate_h(h_rand, dx)
            return (h_rand.tolist(), c5_rand, n_points)
