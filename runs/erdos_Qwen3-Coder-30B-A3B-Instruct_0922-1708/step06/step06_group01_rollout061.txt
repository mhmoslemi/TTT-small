import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a structured construction
        h_init = np.zeros(n_points)
        # Create a simple triangular-like wave with good properties for overlap
        x = np.linspace(0, 2, n_points)
        # triangular wave that satisfies integral constraint
        h_init = np.maximum(0, 1 - np.abs(x - 1))
        h_init = repair_h(h_init, n_points)
    
    # Coordinate-wise optimization with line search and projection
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    eta = 0.01
    epsilon = 1e-6
    patience = 100
    last_improvement = 0
    
    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        # Coordinate-wise updates
        improved = False
        for i in range(n_points):
            # Compute gradient using central difference
            h_plus = h_current.copy()
            h_plus[i] += epsilon
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[i] -= epsilon
            h_minus = repair_h(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)
            
            # Gradient estimate
            grad_i = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update coordinate
            h_new = h_current.copy()
            h_new[i] = h_current[i] - eta * grad_i
            h_new = repair_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept the update if it improves the objective
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                improved = True
                
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = iter_count

        # Adaptive step size and early stopping
        if improved:
            eta = min(0.1, eta * 1.05)  # Increase step size
        else:
            eta *= 0.95  # Decrease step size
            
        # Perturbation if no improvement for a while
        if iter_count - last_improvement > patience:
            # Add small random perturbation
            noise = np.random.normal(0, 0.001, n_points)
            h_current += noise
            h_current = repair_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            last_improvement = iter_count
            eta = max(1e-4, eta * 0.9)
                
    return (h_best.tolist(), c5_best, n_points)
