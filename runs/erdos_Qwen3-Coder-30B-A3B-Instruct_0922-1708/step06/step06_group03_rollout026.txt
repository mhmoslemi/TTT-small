import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, max_k_idx):
    """Compute gradient of the maximum overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # For the maximum overlap at index max_k_idx, compute gradient
    # We use finite differences to approximate the gradient
    
    eps = 1e-6
    max_overlap = evaluate_h(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        overlap_plus = evaluate_h(h_plus, dx)
        grad[i] = (overlap_plus - max_overlap) / eps
    
    return grad

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimization parameters
    learning_rate = 1e-4
    decay_rate = 0.99

    progress_count = 0
    max_progress = 100

        # Calculate gradient with respect to the max overlap
        grad = compute_gradient(current_h, dx, max_k_idx)
        
        # Update h using gradient descent
        new_h = current_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject update
        if new_c5 < current_c5:

            progress_count = 0
        else:

        if current_c5 < best_c5:
