import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create sieve pattern in frequency domain
    freq = np.fft.fftfreq(n_points)
    # High frequency components should be reduced to reduce overlap
    amp = np.exp(-freq**2 * n_points / 10)
    # Ensure DC component is zero
    amp[0] = 0
    # Create complex spectrum with proper symmetry
    phase = np.random.uniform(0, 2*np.pi, n_points)
    spectrum = amp * np.exp(1j * phase)
    # Make it real by exploiting symmetry
    spectrum[0] = 0  # DC - no constant term
    spectrum[n_points//2] = np.real(spectrum[n_points//2])  # Nyquist
    for i in range(1, n_points//2):
        spectrum[n_points-i] = np.conj(spectrum[i])
    
    # IFT and project
    h = np.fft.ifft(spectrum).real
    h = h - h.min()  # shift to start at 0
    h = h / h.max() * 0.5  # make peak at 0.5
    h = project_h(h, n_points)
    return h

    else:

    # Parameters
    max_iter = 2000

        if len(tabu_list) > 0 and h_new.tolist() in tabu_list:
