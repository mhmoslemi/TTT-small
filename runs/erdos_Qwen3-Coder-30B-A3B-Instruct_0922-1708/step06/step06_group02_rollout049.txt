import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with the best known construction or uniform if not available
    n_points = 100
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Uniform initialization
        h = np.ones(n_points) * 0.5
    
    h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Multi-resolution optimization
    resolutions = [10, 20, 40, 100]
    horizons = [100, 200, 300, 400]  # iterations at each resolution
    
    for res_idx, (resolution, horizon) in enumerate(zip(resolutions, horizons)):
        if time.time() - start_time > budget_s - 20:
            break
            
        # Resize h to current resolution
        if resolution < n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, resolution)
            h = np.interp(x_new, x_orig, h)
            h = h * (resolution / 2) / np.sum(h)
        else:
            h = h[:resolution]
        
        # Augmented Lagrangian optimization
        lambda_ = 0.0
        rho = 1.0
        max_iter = horizon
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute current constraint violation
            constraint_violation = np.sum(h) - (resolution / 2)
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            epsilon = 1e-6
            
            for j in range(len(h)):
                h_plus = h.copy()
                h_plus[j] += epsilon
                h_plus = repair_h(h_plus, resolution)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * (2.0 / resolution)
                
                grad[j] = (c5_plus - best_value) / epsilon
            
            # Update h using gradient descent with adaptive step size
            lr = 0.001
            h_new = h - lr * grad
            
            # Augmented Lagrangian update
            h_new = h_new - rho * constraint_violation
            
            # Repair feasibility
            h_new = repair_h(h_new, resolution)
            
            # Update best if improved
            value = evaluate_h(h_new, 2.0 / resolution)
            if value < best_value:
                best_value = value
                best_h = h_new.copy()
            
            # Update Lagrange multiplier and penalty
            lambda_ = lambda_ + rho * constraint_violation
            if abs(constraint_violation) > 1e-4:
                rho = min(1e6, rho * 2)
            
            h = h_new
    
    # Fine-tune with gradient descent on full resolution
    if 'best_h' in locals():
        h = best_h.copy()
    else:
        h = repair_h(h, n_points)
        
    for i in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        ## Compute gradient using finite differences
        grad = np.zeros_like(h)
        epsilon = 1e-6
        current_value = evaluate_h(h, dx)
        
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_value) / epsilon

        # Line search for step size
        lr = 0.001
        while True:
            h_new = h - lr * grad
            h_new = repair_h(h_new, n_points)
            value = evaluate_h(h_new, dx)
            if value < current_value:
                break
            lr *= 0.5
            if lr < 1e-8:
                break

        h = h_new
        if value < best_value:
            best_value = value
            best_h = h.copy()
    
    return (best_h.tolist(), best_value, n_points)
