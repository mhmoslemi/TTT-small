import numpy as np

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    """Compute finite difference gradient."""
    n = len(h)
    grad = np.zeros(n)
    h_base = h.copy()
    for i in range(n):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        
        h_minus = h_base.copy()
        h_minus[i] -= eps
        h_minus = repair_h(h_minus, n)
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        c5_minus = np.max(corr_minus) * dx
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

    # Construct constraint matrix for sum(d) = 0

    bounds = [(-delta, delta)] * n

    def objective(d):
        return grad.dot(d) + 0.5 * d.dot(d)
    
    def constraint_sum(d):
        return np.sum(d)
    
    # Use scipy minimize with trust-constr
    try:
        # Initialize with zero step
        d0 = np.zeros(n)
        result = minimize(objective, d0, method='trust-constr',

    except:
        pass
    
    # Fallback to linear approximation with feasible projected step
    d = -grad * 0.01

    norm_d = np.linalg.norm(d)
    if norm_d > delta and norm_d > 1e-10:
        d = d * delta / norm_d
    
    return d
