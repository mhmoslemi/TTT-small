import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_gradient(h, k, n_points, dx):
    """Compute gradient with respect to k shift."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        # h[i] contributes to overlap with h[i-k] and h[i+k]
        # when i-k >= 0 and i+k < n_points (or wraparound)
        if i - k >= 0:
            grad[i] -= 2 * (1 - h[i - k]) * dx
        if i + k < n_points:
            grad[i] -= 2 * (1 - h[i + k]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    learning_rate = 0.01
    max_iter = 10000
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Discretize shifts
    shift_grid_size = 100
    shifts = np.linspace(-2, 2, shift_grid_size)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlaps for all shifts
        overlaps = []
        for k in shifts:
            k_idx = int((k + 2) / 4 * (shift_grid_size - 1))
            if k_idx < 0 or k_idx >= shift_grid_size:
                overlaps.append(float('inf'))
            else:
                # Compute actual overlap:
                # For shift k, we're evaluating integral of h(x)(1 - h(x + k))
                # But with discrete discrete correlation approach:
                corr = correlate(h, 1 - h, mode='full')
                k_start = len(corr) // 2 - shift_grid_size // 2
                corr_vals = corr[k_start:k_start + shift_grid_size]
                overlap = np.max(corr_vals) * dx
                overlaps.append(overlap)
        
        # Find worst shift
        worst_idx = np.argmax(overlaps)
        worst_k = shifts[worst_idx]
        
        # Compute gradient (targeted approach)
        grad = compute_gradient(h, int((worst_k + 2) / 4 * (shift_grid_size - 1)), n_points, dx)
        
        # Update h using gradient step
        h = h - learning_rate * grad
        
        # Repair the solution
        h = repair_h(h, n_points)
        
        # Evaluate the new solution
        c5 = evaluate_h(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate
        if i % 100 == 0 and i > 0:
            if no_improvement_count > 50:
                learning_rate *= 0.5
                no_improvement_count = 0
                
        # Add some randomness to escape local minima occasionally
        if i % 50 == 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + perturbation
            h = repair_h(h, n_points)
            
        if no_improvement_count >= max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
