import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h)*dx = 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initial construction: triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1 - np.abs(x - 1)  # triangular in [0,2]
        h_init = project_h(h_init, n_points, dx)

    # Coordinate-wise gradient descent with projection
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    eta = 0.01
    eps = 1e-6
    patience = 100
    last_improvement = 0

    for iteration in range(100000):
        if time.time() - start_time > budget_s - 5:
            break

        # Update each coordinate in cyclic order
        for i in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            # Estimate gradient
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad_i = (c5_plus - c5_minus) / (2 * eps)
            
            # Update coordinate
            h_current[i] = h_current[i] - eta * grad_i
            
        # Project to feasible set
        h_current = project_h(h_current, n_points, dx)
        
        # Evaluate and update best
        c5_current = evaluate_h(h_current, dx)
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = iteration

        # Adapt learning rate
        if iteration - last_improvement > patience:
            eta *= 0.9
            if eta < 1e-6:
                eta = 1e-6
            last_improvement = iteration
            
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            perturb_mask = np.random.random(n_points) < 0.05
            h_current[perturb_mask] += np.random.normal(0, 0.01, np.sum(perturb_mask))
            h_current = project_h(h_current, n_points, dx)

    return (h_best.tolist(), c5_best, n_points)
