import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def optimize_convex_relaxation(h, n_points, dx):
    """Solve a relaxed convex problem."""
    try:
        # Formulate the problem in CVXPY
        h_var = cp.Variable(n_points)
        # Objective: minimize sum of squares of correlations
        # This is a convex relaxation
        objective = cp.Minimize(cp.sum_squares(cp.convolve(h_var, 1 - h_var, 'full')))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, feastol=1e-6, reltol=1e-6, abstol=1e-6)
        if prob.status == cp.OPTIMAL:
            return h_var.value
    except Exception:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Parameters
    lr = 0.001
    iters = 0
    
    while time.time() - start_time < budget_s - 2:
        # Try convex relaxation first
        try:
            relaxed_h = optimize_convex_relaxation(h, n_points, dx)
            if relaxed_h is not None:
                h = relaxed_h
        except:
            pass
            
        # Gradient descent
        eps = 1e-6
        grad = np.zeros(n_points)
        h_eval = evaluate_h(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - h_eval) / eps
        
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        # Evaluate
        c5 = evaluate_h(h, dx)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        iters += 1
        
        # Every 100 iterations, try random restart
        if iters % 100 == 0:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            lr = min(lr * 1.01, 0.01)  # Adaptive learning rate
            
        # Adaptive learning rate decay
        if iters % 1000 == 0:
            lr *= 0.95

    return (best_h.tolist(), best_c5, n_points)
