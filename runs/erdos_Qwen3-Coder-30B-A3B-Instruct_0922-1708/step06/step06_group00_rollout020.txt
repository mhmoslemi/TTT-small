import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Optimization parameters
    max_iter = 10000
    gamma = 0.01  # Step size
    beta = 0.9    # Nesterov momentum
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Nesterov acceleration
    v = np.zeros_like(h)
    h_accel = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the maximum lag
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h) - 1)
        
        # Compute gradient with respect to the worst lag
        grad = np.zeros_like(h)
        if k >= 0:
            for i in range(len(h) - k):
                grad[i] -= (1 - h[i + k]) * dx
        else:
            for i in range(-k, len(h)):
                grad[i] -= (1 - h[i + k]) * dx
        
        # Update with proximal gradient and Nesterov acceleration
        h_accel = h + beta * v
        h_new = h_accel - gamma * grad
        
        # Proximal projection onto [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet sum constraint
        h_new_sum = np.sum(h_new)
        target_sum = n_points / 2
        if abs(h_new_sum - target_sum) > 1e-8:
            h_new = h_new * target_sum / h_new_sum
            
        # Update momentum
        v = beta * v + (h_new - h)
        h = h_new
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
