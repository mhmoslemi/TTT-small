import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of C5 with respect to h."""
    n_points = len(h)
    grad = np.zeros(n_points)
    for j in range(n_points):
        # For each position, determine how much it contributes to the maximum overlap
        for k in range(n_points):
            if k == k_max:
                grad[j] += (1 - h[(j - k) % n_points])
    return grad * dx

def adaptive_gradient_descent(h_init, n_points, dx, budget_s):
    """Perform gradient-based optimization with adaptive perturbations."""
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimizer parameters
    lr = 1e-4
    perturbation_magnitude = 0.05
    no_improvement_steps = 0
    max_no_improvement = 1000
    perturbation_frequency = 500
    
    start_time = time.time()
    
    for iteration in range(1000000):  # Large number, will be bounded by time
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap for all lags
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Compute gradient
        grad = compute_gradient(current_h, dx, k_max)
        
        # Apply gradient descent step
        new_h = current_h - lr * grad
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Update if we improve
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            no_improvement_steps = 0
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            no_improvement_steps += 1
            
        # Apply adaptive perturbations
        if no_improvement_steps > 0 and no_improvement_steps % perturbation_frequency == 0:
            # Add noise in areas of low gradient contribution
            perturb_indices = np.where(np.abs(grad) < 0.1)[0]
            if len(perturb_indices) > 0:
                num_perturb = min(5, len(perturb_indices))
                indices = np.random.choice(perturb_indices, num_perturb, replace=False)
                for idx in indices:
                    delta = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
                    current_h[idx] += delta
                current_h = repair_h(current_h, n_points)
                current_c5 = evaluate_h(current_h, dx)
                
        # Adaptive learning rate
        if no_improvement_steps % 100 == 0 and no_improvement_steps > 0:
            lr *= 0.99
            
        # Adaptive perturbation magnitude
        if no_improvement_steps % 500 == 0 and no_improvement_steps > 0:
            perturbation_magnitude *= 0.999
            
    return best_h, best_c5
