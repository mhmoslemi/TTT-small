import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize initial h to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Iterative cutting plane approach
    max_iter = 100
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the worst case shift k_max
        max_lag_idx = np.argmax(corr)
        k_max = max_lag_idx - (len(h) - 1)
        
        # Build convex subproblem to reduce max overlap at k_max
        # Define variables
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Objective
        objective = cp.Minimize(t)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Bound constraints
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Max overlap constraint for shift k_max
        # overlap_k_max = sum(h[i] * (1 - h[i + k_max]) for i)
        # For i = 0 to n_points-1: h[i] * (1 - h[i + k_max])
        # Note that i + k_max can go out of bounds
        
        # Build the expression for overlap constraint
        # i ranges from 0 to min(n_points-1, n_points-1-k_max) if k_max >= 0
        # i ranges from max(0, -k_max) to n_points-1 if k_max < 0
        
        if k_max >= 0:
            # k_max >= 0 means we shift forward
            valid_range = range(n_points - k_max)
            expr = 0
            for i in valid_range:
                expr += h_var[i] * (1 - h_var[i + k_max])
        else:
            # k_max < 0 means we shift backward
            valid_range = range(-k_max, n_points)
            expr = 0
            for i in valid_range:
                expr += h_var[i] * (1 - h_var[i + k_max])
        
        # Add the constraint
        constraints.append(expr <= t)
        
        # Form the problem
        prob = cp.Problem(objective, constraints)
        
        # Solve with ECOS for faster solving
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h_var.value
                if h_new is not None:
                    # Adjust any values to stay within bounds
                    h_new = np.clip(h_new, 0, 1)
                    # Re-normalize to sum to n_points / 2
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    h = h_new
        except:
            # If optimization fails, fall back to simple gradient descent
            # To save time, just continue to next iter
            pass
            
    if best_h is None:
        # Fallback: original h
        corr = correlate(h, 1 - h, mode='full')
        best_c5 = np.max(corr) * dx
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
