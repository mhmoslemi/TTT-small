import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, k_max, n_points, dx):
    """Compute gradient for the maximum overlap."""
    grad = np.zeros(n_points)
    for j in range(n_points):
        # Compute contribution of h[j] to the gradient
        for k in range(n_points):
            if k == k_max:
                # The gradient of overlap[k] w.r.t. h[j] is (1 - h[(j-k) % n_points])
                i = (j - k) % n_points
                grad[j] += (1 - h[i]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Gradient-based optimization parameters
    learning_rate = 1e-4
    perturb_freq = 500
    perturb_magnitude = 0.05
    no_improvement_patience = 100
    iter_no_improvement = 0
    step = 0
    
    while time.time() - start_time < budget_s - 5:
        # Compute overlap for all lags
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient
        grad = compute_gradient(current_h, k_max, n_points, dx)
        
        # Update h using gradient descent
        current_h = current_h - learning_rate * grad
        current_h = repair_h(current_h, n_points)
        
        # Evaluate new C5
        current_c5 = evaluate_h(current_h, dx)
        
        # Apply perturbation periodically or if no improvement
        if step % perturb_freq == 0 or iter_no_improvement > no_improvement_patience:
            perturb_magnitude *= 0.999
            perturb_indices = np.random.choice(n_points, max(1, n_points // 20), replace=False)
            for idx in perturb_indices:
                current_h[idx] += np.random.uniform(-perturb_magnitude, perturb_magnitude)
            current_h = repair_h(current_h, n_points)
            current_c5 = evaluate_h(current_h, dx)
            iter_no_improvement = 0
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
        
        # Update learning rate
        if step % 100 == 0:
            learning_rate *= 0.99
        step += 1
    
    return (best_h.tolist(), best_c5, n_points)
