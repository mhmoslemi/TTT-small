import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    h = np.zeros(n_points, dtype=int)
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h = (h_init > 0.5).astype(int)
        # Adjust sum to be exactly n_points/2
        current_sum = np.sum(h)
        target_sum = n_points // 2
        if current_sum != target_sum:
            # Adjust h to make sum exactly target_sum
            # Find indices to flip
            flip_indices = np.where(h == 1)[0]
            if current_sum > target_sum:
                # Need to flip some 1s to 0s
                to_flip = current_sum - target_sum
                h[flip_indices[:to_flip]] = 0
            else:
                # Need to flip some 0s to 1s
                to_flip = target_sum - current_sum
                flip_indices = np.where(h == 0)[0]
                h[flip_indices[:to_flip]] = 1
    else:
        # Set up a uniform binary pattern
        h[:n_points//2] = 1
    
    # MIP formulation
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points // 2]
    
    # For each possible shift k
    for k in range(-n_points + 1, n_points):
        # Indices for valid correlation computation
        if k >= 0:
            # i from 0 to n_points - 1 - k
            valid_indices = list(zip(range(n_points - k), range(k, n_points)))
        else:
            # i from -k to n_points - 1
            valid_indices = list(zip(range(-k, n_points), range(n_points + k)))
            
        # Sum of h[i] * (1 - h[i + k]) for valid i
        overlap_term = cp.sum([h_var[i] * (1 - h_var[j]) for (i, j) in valid_indices])
        constraints.append(overlap_term <= t * (n_points / 2))
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Use a MIP solver
    try:
        # If solver is available, use it with a time limit
        prob.solve(solver=cp.CBC, time_limit=budget_s - 10, gap_rel=0.01)
        
        # Extract solution
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h_var.value.astype(int)
            # Calculate C5
            corr = np.correlate(h_sol, 1 - h_sol, mode='full')
            c5_bound = np.max(corr) * dx
        else:
            # Fallback to initial or heuristic
            h_sol = h
            corr = np.correlate(h_sol, 1 - h_sol, mode='full')
            c5_bound = np.max(corr) * dx
    except Exception as e:
        # Fallback
        h_sol = h
        corr = np.correlate(h_sol, 1 - h_sol, mode='full')
        c5_bound = np.max(corr) * dx
    
    # Final adjustment to ensure sum = n_points/2 for integer binary
    if np.sum(h_sol) != n_points // 2:
        # Make adjustments to get exact sum
        diff = np.sum(h_sol) - n_points // 2
        if diff > 0:
            # Need to flip some 1s to 0s
            ones_idx = np.where(h_sol == 1)[0]
            h_sol[ones_idx[:diff]] = 0
        else:
            # Need to flip some 0s to 1s
            zeros_idx = np.where(h_sol == 0)[0]
            h_sol[zeros_idx[:abs(diff)]] = 1
    
    assert np.sum(h_sol) == n_points // 2
    corr = np.correlate(h_sol, 1 - h_sol, mode='full')
    c5_final = np.max(corr) * dx
    
    return (h_sol.tolist(), c5_final, n_points)
