import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_gradient(h, worst_k, dx, n_points):
    """Compute gradient of C5 with respect to h at the worst shift."""
    grad = np.zeros(n_points)
    # For each position j, compute the derivative of the overlap at worst_k
    for j in range(n_points):
        # Approximate gradient using finite differences
        # Only consider valid indices for the convolution
        if worst_k >= 0:
            # h[j] contributes to integral over [j, j+worst_k] and [j+worst_k, j]
            forward_idx = (j + worst_k) % n_points
            backward_idx = (j - worst_k) % n_points
            grad[j] = -2 * dx * (h[forward_idx] + h[backward_idx])
        else:
            forward_idx = (j - worst_k) % n_points
            backward_idx = (j + worst_k) % n_points
            grad[j] = -2 * dx * (h[forward_idx] + h[backward_idx])
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():

        if len(h) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize best solution
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Gradient step parameters
    learning_rate = 0.01
    decay_factor = 0.95
    max_iter = 10000
    patience = 100
    last_improvement = 0
    
    # Main optimization loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all correlations to find worst case
        corr = correlate(h, 1 - h, mode='full')
        # Find the maximum correlation and k value
        max_idx = np.argmax(corr)
        # Compute the actual shift k from the max index
        k = max_idx - n_points + 1  # Since mode='full', ranges from -n+1 to n-1 
        c5_value = corr[max_idx] * dx 
        
        # Compute gradient
        grad = compute_gradient(h, k, dx, n_points)
        
        # Update h
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_h(h, dx)
        
        # Update best if improved
        if new_c5 < best_value:
            best_value = new_c5
            best_h = h.copy()
            last_improvement = iter_count
            # Adjust learning rate
            learning_rate = min(0.02, learning_rate * 1.02)
            
        # Reduce learning rate if no improvement for a while
        if iter_count - last_improvement > patience:
            learning_rate *= decay_factor
            last_improvement = iter_count
            
        # Add small random perturbations occasionally
        if iter_count % 50 == 0 and iter_count > 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + perturbation
            h = repair_h(h, n_points)
    
    # Final fine-tune
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros(n_points)
        # Use approximate gradient
        for j in range(n_points):
            eps = 1e-6
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
        
        h = h - 0.001 * grad
        h = repair_h(h, n_points)
        current_c5 = evaluate_h(h, dx)
        if current_c5 < best_value:
            best_value = current_c5
            best_h = h.copy()
    
    # Final check
    best_value = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_value, n_points)
