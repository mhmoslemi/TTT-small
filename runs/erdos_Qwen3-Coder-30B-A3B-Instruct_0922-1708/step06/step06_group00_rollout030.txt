import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 10
    
    while (time.time() - start_time) < budget_s - 5 and no_improvement_count < max_no_improvement:
        iter_count += 1
        
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Identify the maximum lag
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h) - 1)
        
        # Solve convex subproblem to reduce the overlap at k
        # Variables
        h_vars = cp.Variable(n_points)
        t = cp.Variable()
        
        # Objective
        objective = cp.Minimize(t)
        
        # Constraints
        # For lag k, constraint: sum h[i] * h[i+k] <= (1-t)*n_points/2
        # This ratio comes from: overlap_k = 1 - (2/n_points)*sum(h[i]*h[i+k]) <= t
        # => sum(h[i]*h[i+k]) >= (1-t)*n_points/2
        
        # We'll approximate this:
        # Help form constraint: sum h[i]*h[i+k] <= (1-t)*n_points/2
        
        # Instead, we solve: min_t s.t. overlap_k <= t
        # overlap_k = 1 - (2/n_points)*sum(h[i]*h[i+k]) <= t
        # => (2/n_points)*sum(h[i]*h[i+k]) >= 1-t
        # => sum(h[i]*h[i+k]) >= (1-t)*n_points/2
        # => sum(h[i]*h[i+k]) - (1-t)*n_points/2 >= 0
        
        # Constraint for specific lag k (using sparsity to reduce constraints)
        # Create constraints only for relevant positions
        sum_expr = 0
        for i in range(len(h)):
            j = i + k
            if 0 <= j < len(h):
                sum_expr += h_vars[i] * h_vars[j]
        
        constraint = [sum_expr >= (1 - t) * n_points / 2]
        
        # Other constraints
        constraint += [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Solve subproblem
        prob = cp.Problem(objective, constraint)
        
        try:
            # Try different solvers
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INaccurate]:
                h_new = h_vars.value
                if h_new is not None:
                    h = h_new.copy()
                    # Apply projection to stay in feasible region
                    h = np.clip(h, 0, 1)
                    h_sum = np.sum(h)
                    if abs(h_sum - n_points / 2) > 1e-8:
                        h = h * (n_points / 2) / h_sum
                else:
                    # Fall back to simpler update
                    h = np.clip(h, 0, 1)
            else:
                # Fall back to simpler update if optimization fails
                h = np.clip(h, 0, 1)
        except:
            # Fall back to a simple update
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-8:
                h = h * (n_points / 2) / h_sum
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
