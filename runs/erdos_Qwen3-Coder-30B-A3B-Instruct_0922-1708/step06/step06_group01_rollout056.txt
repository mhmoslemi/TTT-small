import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured pattern
        x = np.linspace(0, 2, n_points)
        h_init = np.maximum(0, 1 - np.abs(x - 1) / 0.5)
        h_init = project_h(h_init, n_points)
    
    # Frank-Wolfe parameters
    max_iter = 5000
    patience = 1000
    epsilon = 1e-6
    alpha = 0.9
    
    # Current and best solution
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    last_improvement = 0
    
    # Main Frank-Wolfe loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += epsilon
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_current) / epsilon
        
        # Linear subproblem: min_d <grad, d> s.t. 0 <= d[i] <= 1, sum(d) = 0
        # This is a constrained optimization; we approximate with a simple method
        # Choose d that minimizes <grad, d> with constraints
        d = -grad
        d = d - np.mean(d)  # Ensure sum(d) = 0
        d = np.clip(d, -1, 1)  # Clipping to [0,1] region
        
        # Line search to find optimal step size
        eta_max = 1.0
        eta = eta_max
        while eta > 1e-8:
            h_new = h_current + eta * d
            h_new = project_h(h_new, n_points)
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                break
            eta *= 0.5
        
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive step size
        if i - last_improvement > patience:
            alpha *= 0.9
            if alpha < 1e-6:
                break
    
    # Final refinement using gradient descent
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_best.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_best) / eps
        
        h_best = h_best - 0.001 * grad
        h_best = project_h(h_best, n_points)
        corr = correlate(h_best, 1 - h_best, mode='full')
        c5_best = np.max(corr) * dx
    
    # Final evaluation
    corr_final = correlate(h_best, 1 - h_best, mode='full')
    c5_final = np.max(corr_final) * dx
    
    return (h_best.tolist(), c5_final, n_points)
