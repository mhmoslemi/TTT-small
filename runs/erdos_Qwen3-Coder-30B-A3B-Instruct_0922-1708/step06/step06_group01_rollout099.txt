import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def finite_diff_grad(h, dx, eps=1e-6):
    """Compute finite difference gradient of C5."""
    n = len(h)
    grad = np.zeros(n)
    c5_base = evaluate_h(h, dx)
    h_plus = h.copy()
    for i in range(n):
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
        h_plus[i] -= eps
    return grad

def trust_region_step(h, dx, grad, trust_radius=1.0):
    """Solve trust-region subproblem for step direction."""
    n = len(h)
    # Simple approximate trust-region step using projected gradient descent
    step = -grad
    step = np.clip(step, -trust_radius, trust_radius)
    return step

def perturb_h(h, scale=0.05):
    """Perturb h to escape local optima."""
    n = len(h)
    h_new = h.copy()
    num_perturb = max(1, int(scale * n))
    indices = np.random.choice(n, num_perturb, replace=False)
    for idx in indices:
        delta = np.random.uniform(-0.1, 0.1)
        h_new[idx] += delta
    return h_new
