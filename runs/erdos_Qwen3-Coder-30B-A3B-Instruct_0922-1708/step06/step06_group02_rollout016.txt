import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def crossover(parent1, parent2):
    """Create child from two parents."""
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(h, rate=0.1):
    """Mutate a candidate solution."""
    mask = np.random.rand(len(h)) < rate
    if not np.any(mask):
        idx = np.random.randint(len(h))
        mask[idx] = True
    noise = np.random.normal(0, 0.1, len(h))
    h = h + noise * mask
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    generations = 1000
    elite_size = int(population_size * 0.2)
    mutation_rate = 0.1
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init.copy()]
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init.copy()]
    
    # Fill population
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Initialize best solution
    best_h = None
    best_value = float('inf')
    
    # Evaluate initial population
    fitness = []
    for i, h in enumerate(population):
        value = evaluate_h(h, dx)
        fitness.append(value)
        if value < best_value:
            best_value = value
            best_h = h.copy()
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_value:
            best_value = fitness[0]
            best_h = population[0].copy()
        
        # Create new population
        new_population = population[:elite_size]
        new_fitness = fitness[:elite_size]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection: tournament
            tournament_size = 5
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Mutation (10% chance)
            if np.random.rand() < 0.5:
                h = mutate(population[winner_idx], mutation_rate)
                h = repair_h(h, n_points)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Crossover with another individual
                another_idx = np.random.choice(len(population[:elite_size]))
                h = crossover(population[winner_idx], population[another_idx])
                h = repair_h(h, n_points)
            
            new_population.append(h)
            new_fitness.append(evaluate_h(h, dx))
        
        population = new_population
        fitness = new_fitness
    
    # Final refinement with gradient descent on best solution
    if best_h is not None:
        h_refined = best_h.copy()
        current_value = best_value
        learning_rate = 0.001
        
        for iter_refine in range(500):
            if time.time() - start_time > budget_s - 5:
                break
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
                
            h_refined = h_refined - learning_rate * grad
            h_refined = repair_h(h_refined, n_points)
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            current_value = np.max(corr) * dx
            
            if current_value < best_value:
                best_value = current_value
                best_h = h_refined.copy()
    
    return (best_h.tolist(), best_value, n_points)
