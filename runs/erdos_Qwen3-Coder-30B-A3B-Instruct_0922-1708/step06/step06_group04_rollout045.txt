import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_initialization(n_points):
    # Create a sieve-like pattern in frequency domain
    freqs = np.fft.fftfreq(n_points, 1.0/n_points)
    # High frequencies get smaller amplitudes to reduce overlap
    spectrum = np.exp(-0.5 * (freqs / (n_points/10))**2)
    # Ensure DC component is real
    spectrum[0] = np.abs(spectrum[0])
    # Get real signal via inverse FFT
    h = np.real(np.fft.ifft(spectrum))
    # Normalize
    h = h - np.min(h)
    h = h / np.max(h) if np.max(h) > 0 else h
    h = project_h(h, n_points)
    return h

    dx = 2.0 / n_points

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = fourier_initialization(n_points)

    best_h = h_init.copy()
    best_value = evaluate_overlap(best_h, dx)
    
    # Tabu search parameters
    population_size = 100
    tabu_list_size = 50
    tabu_list = []
    memory_decay = 0.95
    stagnation_count = 0
    max_stagnation = 50
    perturb_magnitude = 0.2
    iterations = 0
    max_iterations = 2000

    while iterations < max_iterations and time.time() - start_time < budget_s - 10:
        # Generate new candidates by perturbing best solution
        candidates = []
        for _ in range(population_size):
            h_perturbed = best_h.copy()
            # Perturb with random changes
            indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            for idx in indices:
                h_perturbed[idx] += np.random.normal(0, perturb_magnitude)
            h_perturbed = project_h(h_perturbed, n_points)
            candidates.append(h_perturbed)
        
        # Evaluate candidates
        candidate_values = [evaluate_overlap(h, dx) for h in candidates]
        
        # Update tabu list
        tabu_list = [t for t in tabu_list if np.random.random() < memory_decay]
        
        # Find best candidate not in tabu list
        best_candidate_idx = np.argmin(candidate_values)
        current_best_value = candidate_values[best_candidate_idx]
        current_best_h = candidates[best_candidate_idx]
        
        # Avoid tabu moves by penalizing
        # The tabu penalty is already implicitly handled due to limited search
        
        if current_best_value < best_value:
            best_h = current_best_h.copy()
            best_value = current_best_value
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Diversity injection
        if stagnation_count >= max_stagnation:
            for _ in range(5):
                h_rand = fourier_initialization(n_points)
                candidates.append(h_rand)
                candidate_values.append(evaluate_overlap(h_rand, dx))
                
            # Update best again if needed
            best_candidate_idx = np.argmin(candidate_values)
            current_best_value = candidate_values[best_candidate_idx]
            current_best_h = candidates[best_candidate_idx]

    h_refined = best_h.copy()

            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps

        h_refined = project_h(h_refined, n_points)
        prev_value = evaluate_overlap(h_refined, dx)

            best_h = h_refined.copy()
            best_value = prev_value

    return (best_h.tolist(), best_value, n_points)
