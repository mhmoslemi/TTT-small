import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def trust_region_step(h, dx, grad, hess, trust_radius):
    """Solve trust region subproblem using projected gradient."""
    n = len(h)
    # Simple trust region solver with projected gradient
    d = -grad
    d_norm = np.linalg.norm(d)
    if d_norm > trust_radius:
        d = d * trust_radius / d_norm

    # Ensure feasibility
    h_new = h + d
    h_new = np.clip(h_new, 0, 1)
    
    # Project to sum constraint
    current_sum = np.sum(h_new)
    target_sum = n / 2
    if abs(current_sum - target_sum) > 1e-8:
        adjustment = (target_sum - current_sum) / n
        h_new = h_new + adjustment
        h_new = np.clip(h_new, 0, 1)
        
    return h_new

def compute_gradient(h, dx, epsilon=1e-6):
    """Compute finite difference gradient of C5."""
    n = len(h)
    grad = np.zeros(n)
    c5_orig = evaluate_h(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = repair_h(h_plus, n)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - c5_orig) / epsilon
    
    return grad

def compute_hessian(h, dx, epsilon=1e-4):
    """Compute finite difference Hessian of C5."""
    n = len(h)
    H = np.zeros((n, n))
    grad_base = compute_gradient(h, dx, epsilon=epsilon/2)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = repair_h(h_plus, n)
        grad_plus = compute_gradient(h_plus, dx, epsilon=epsilon/2)
        H[:, i] = (grad_plus - grad_base) / epsilon
    
    return H
