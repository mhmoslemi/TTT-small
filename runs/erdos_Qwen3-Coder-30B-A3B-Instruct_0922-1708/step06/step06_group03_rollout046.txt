import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of C5 with respect to h at the maximizing lag k_max."""
    n_points = len(h)
    grad = np.zeros(n_points)
    for i in range(n_points):
        # For the overlap at lag k_max, we have contribution from h[i] to h[(i-k_max)%n_points]
        j = (i - k_max) % n_points
        grad[i] = (1 - h[j]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    learning_rate = 1e-4
    perturb_mag = 0.05
    
    # Adaptive parameters
    last_improvement = 0
    patience = 200
    
    for i in range(100000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlaps to find maximum
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient at that maximum
        grad = compute_gradient(current_h, dx, k_max)
        
        # Apply gradient descent
        h_candidate = current_h - learning_rate * grad
        
        # Add structured perturbation
        if i % 100 == 0 and i > 0:
            # Perturb indices with low gradient magnitude
            abs_grad = np.abs(grad)
            low_grad_indices = np.where(abs_grad < np.percentile(abs_grad, 10))[0]
            if len(low_grad_indices) > 0:
                indices = np.random.choice(low_grad_indices, 
                                         size=max(1, len(low_grad_indices) // 10), 
                                         replace=False)
                for idx in indices:
                    h_candidate[idx] += np.random.uniform(-perturb_mag, perturb_mag)
        
        # Repair candidate
        h_candidate = repair_h(h_candidate, n_points)
        candidate_c5 = evaluate_h(h_candidate, dx)
        
        # Accept if better or sometimes with low probability
        if candidate_c5 < current_c5 or np.random.rand() < 0.01:
            current_h = h_candidate
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                last_improvement = i
                
        # Adaptive updates
        if i > 0 and i % 100 == 0:
            if i - last_improvement > patience:
                learning_rate *= 0.95
                perturb_mag *= 0.99
                last_improvement = i
                
        if i % 500 == 0:
            learning_rate = max(1e-6, learning_rate * 0.99)
            perturb_mag = max(1e-6, perturb_mag * 0.999)
            
    # Final refinement with smaller learning rate
    h_final = best_h.copy()
    final_c5 = best_c5
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        corr = correlate(h_final, 1 - h_final, mode='full')
        k_max = np.argmax(corr)
        grad = compute_gradient(h_final, dx, k_max)
        
        h_final = h_final - 1e-6 * grad
        h_final = repair_h(h_final, n_points)
        
        new_c5 = evaluate_h(h_final, dx)
        if new_c5 < final_c5:
            final_c5 = new_c5
    
    return (final_c5, final_c5, n_points)
