import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h = (h_init > 0.5).astype(int)
        h_sum = np.sum(h)
        if h_sum != n_points / 2:
            # Adjust to make sum = n_points / 2
            diff = int(n_points / 2 - h_sum)
            if diff > 0:
                # Convert some 0s to 1s
                zero_indices = np.where(h == 0)[0]
                for i in range(min(diff, len(zero_indices))):
                    h[zero_indices[i]] = 1
            elif diff < 0:
                # Convert some 1s to 0s
                one_indices = np.where(h == 1)[0]
                for i in range(min(-diff, len(one_indices))):
                    h[one_indices[i]] = 0
    else:
        # Initialize with uniform distribution of 1s and 0s
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # Define MIP problem
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
    ]
    
    # Add overlap constraints for all shifts k
    for k in range(-n_points + 1, n_points):
        # For each k, define valid indices
        valid_pairs = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_pairs.append((i, j))
        
        if valid_pairs:
            # For shift k, sum h[i] * (1 - h[j]) <= t * (n_points / 2)
            overlap_expr = cp.sum([h_var[i] * (1 - h_var[j]) for i, j in valid_pairs])
            constraints.append(overlap_expr <= t * (n_points / 2))
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Use CBC solver with time limit
    try:
        prob.solve(solver=cp.CBC, max_seconds=budget_s - 10, gap=0.01)
    except:
        # If solver fails, use warm start with initialized h
        try:
            prob.solve(solver=cp.CBC, max_seconds=budget_s - 10, gap=0.01, warm_start=True)
        except:
            pass
    
    # Extract solution
    if h_var.value is not None:
        h_sol = np.round(h_var.value).astype(int)
    else:
        h_sol = h
    
    # Ensure constraint is satisfied
    h_sol = h_sol.copy()
    actual_sum = np.sum(h_sol)
    target_sum = n_points / 2
    diff = target_sum - actual_sum
    
    # Adjust if sum is not exactly n_points/2
    if abs(diff) > 1e-5:
        if diff > 0:
            # Need to convert 0s to 1s for diff positions
            zero_indices = np.where(h_sol == 0)[0]
            for i in range(int(diff)):
                if i < len(zero_indices):
                    h_sol[zero_indices[i]] = 1
        else:
            # Need to convert 1s to 0s for abs(diff) positions
            one_indices = np.where(h_sol == 1)[0]
            for i in range(int(-diff)):
                if i < len(one_indices):
                    h_sol[one_indices[i]] = 0
    
    # Compute the final overlap value
    corr = correlate(h_sol, 1 - h_sol, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_sol.tolist(), final_c5, n_points)
