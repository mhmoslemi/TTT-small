import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def crossover(parent1, parent2):
    """Perform crossover between two parents."""
    alpha = np.random.rand()
    child = alpha * parent1 + (1 - alpha) * parent2
    return child

def mutate(individual, mutation_rate=0.1, noise_std=0.1):
    """Perform mutation on an individual."""
    if np.random.rand() < mutation_rate:
        noise = np.random.normal(0, noise_std, len(individual))
        mutated = individual + noise
        return mutated
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    
    # Fill the rest of the population
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        
    # Initialize best solution
    best_h = None
    best_value = float('inf')
    
    # Evaluate initial population
    fitness = []
    for i, h in enumerate(population):
        value = evaluate_h(h, dx)
        fitness.append(value)
        if value < best_value:
            best_value = value
            best_h = h.copy()
    
    # Evolution loop
    max_generations = 1000
    generation = 0
    
    # Track improvement
    last_improvement = 0
    patience = 50
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 20% as parents
        num_parents = max(2, population_size // 5)
        parents = sorted_population[:num_parents]
        
        # Generate new population
        new_population = parents[:]
        
        # Create offspring through crossover and mutation
        while len(new_population) < population_size:
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutation
            noise_std = 0.1 * (1.0 - sorted_fitness[0] / best_value) if best_value > 0 else 0.1
            child = mutate(child, noise_std=noise_std)
            
            # Repair
            child = repair_h(child, n_points)
            new_population.append(child)
            
        # Replace population
        population = new_population
        
        # Evaluate new population
        fitness = []
        for i, h in enumerate(population):
            value = evaluate_h(h, dx)
            fitness.append(value)
            if value < best_value:
                best_value = value
                best_h = h.copy()
                
        # Gradient refinement on top candidate
        if generation % 10 == 0 and best_h is not None:
            h_refined = best_h.copy()
            current_value = best_value
            
            for _ in range(50):

                    break
                eps = 1e-6
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_value) / eps
                    
                h_refined = h_refined - 0.0001 * grad
                h_refined = repair_h(h_refined, n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                current_value = np.max(corr) * dx
                
                if current_value < best_value:
                    best_value = current_value
                    best_h = h_refined.copy()
        
        # Check for stagnation
        if generation - last_improvement > patience:
            break
            
    return (best_h.tolist(), best_value, n_points)
