import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convex optimization approach
    h = Variable(n_points)
    t = Variable()
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        t >= 0
    ]
    
    # Overlap constraints for each lag
    max_lag = n_points - 1
    for lag in range(-max_lag, max_lag + 1):
        # Compute valid indices
        if lag >= 0:
            left_idx = np.arange(n_points - lag)
            right_idx = np.arange(lag, n_points)
        else:
            left_idx = np.arange(-lag, n_points)
            right_idx = np.arange(n_points + lag)
            
        if len(left_idx) > 0:
            # Valid correlation
            overlap_expr = h[left_idx] @ (1 - h[right_idx]) * dx
            constraints.append(overlap_expr <= t)
    
    # Objective
    objective = Minimize(t)
    
    # Problem
    prob = Problem(objective, constraints)
    
    # Attempt to solve
    try:
        prob.solve(solver=MOSEK, verbose=False, time_limit=budget_s - 10)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_sol = h.value
            c5_bound = t.value * dx
        else:
            h_sol = h_init
            c5_bound = np.inf
    except:
        h_sol = h_init
        c5_bound = np.inf
    
    # Final validation
    if np.any(h_sol < 0) or np.any(h_sol > 1):
        h_sol = np.clip(h_sol, 0, 1)
    if abs(np.sum(h_sol) - n_points / 2) > 1e-6:
        h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
    
    # Ensure we return a valid solution even if solver fails
    if np.isinf(c5_bound):
        h_sol = h_init
        # Evaluate with function
        from scipy.signal import correlate
        corr = correlate(h_sol, 1 - h_sol, mode='full')
        c5_bound = np.max(corr) * dx
    
    # Final refinement by gradient
    def eval_h(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    h_refined = h_sol.copy()
    best_val = eval_h(h_refined)
    
    # Simple gradient descent with small steps
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c_plus = eval_h(h_plus)
            grad[i] = (c_plus - best_val) / eps
        h_new = h_refined - 1e-3 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        c_new = eval_h(h_new)
        if c_new < best_val:
            h_refined = h_new
            best_val = c_new
            
    return (h_refined.tolist(), best_val, n_points)
