import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Learning rate schedule
    lr = 5e-4
    decay_factor = 0.99
    decay_interval = 500
    
    # Temperature for dual updates
    tau = 0.1
    tau_decay = 0.999
    
    # Convolution kernel for gradient
    kernel = np.ones(n_points)
    
    # Main optimization loop
    iter_count = 0
    stagnation_count = 0
    max_stagnation = 1000
    
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        
        if iter_count % decay_interval == 0:
            lr *= decay_factor
            
        if iter_count % 10 == 0:
            tau *= tau_decay
            
        # Compute overlaps for all lags
        overlaps = correlate(current_h, 1 - current_h, mode='full')
        overlap_max = np.max(overlaps)
        overlap_indices = np.arange(len(overlaps)) - (n_points - 1)
        
        # Compute dual variables using softmax
        exp_overlaps = np.exp(-overlaps / tau)
        exp_overlaps = exp_overlaps / np.sum(exp_overlaps)
        dual_vars = exp_overlaps
        
        # Compute gradient
        grad = np.zeros(n_points)
        full_h = np.concatenate([current_h, [0] * (n_points - 1)])
        
        for i in range(n_points):
            # Compute dL/dh[i] = sum_k dual_vars[k] * (1 - h[i-k])
            grad[i] = 0
            for k in range(n_points):
                if i - k >= 0:
                    grad[i] += dual_vars[k + n_points - 1] * (-full_h[i - k])
                else:
                    grad[i] += dual_vars[k + n_points - 1] * 1.0
                    
        # Update h
        current_h = current_h - lr * grad
        current_h = repair_h(current_h, n_points)
        
        # Evaluate current solution
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1

            # Fallback to initial solution
            current_h = h_init.copy()
            current_c5 = evaluate_h(current_h, dx)
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
            
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
