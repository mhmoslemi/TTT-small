import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def gradient_step(h, dx, max_k):
    """Compute and apply gradient step to reduce maximum overlap."""
    n_points = len(h)
    # Ensure h is feasible
    h = repair_h(h, n_points)
    
    # Compute correlations
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    
    # Compute gradient of correlation with respect to h
    grad = np.zeros(n_points)
    
    # Gradient computation for maximum element
    # For each position, it's the derivative of max_{k} corr[k] = max_{k} sum h[i] (1-h[i+k])
    # But we focus on the k that achieves the maximum to help with optimization
    k_offset = max_idx - (n_points - 1)
    
    # In convolution, we're looking at how much h[i] contributes to corr[k]
    # In conv(h, 1-h)[k] = sum h[i] (1-h[i+k]), which is 1 - h[i+k] where i+k = k
    # Backtracking: i = k - (i+k)
    
    # Gradient w.r.t. element h[j] at position j
    # if i+k = j, then i = j-k
    # So d corr[k]/d h[j] = 1 - h[j-k] if 0<=j-k<n_points else 0
    # grad[k] = (1 - h[k-k]) = 1 - h[0] if k = 0, for example 
    # Actually: d/dh[j] \sum_i h[i] (1-h[i+k]) = 1 - h[i+k] for i = j
    # So gradient is 1 - h[j-k] when not out of bounds
    
    # Partial gradients contribution (simplified)
    for i in range(n_points):
        # Set contribution: derivative of conv(h, 1-h)[k] with respect to h[i]
        if i >= k_offset and i - k_offset < n_points:
            grad[i] += 1 - h[i - k_offset]
    
    # Normalize gradient
    grad = grad / (n_points * np.linalg.norm(grad) + 1e-12)
    
    # Perturb in negative gradient direction
    step_size = 0.001
    h_new = h - step_size * grad
    
    # Return step
    return h_new

    # Initialize
