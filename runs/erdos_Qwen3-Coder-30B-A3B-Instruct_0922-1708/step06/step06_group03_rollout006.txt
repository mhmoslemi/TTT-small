import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent approach
    lambda_k = np.zeros(2 * n_points - 1)
    learning_rate = 5e-4
    max_iter = 30000
    progress_count = 0
    max_progress = 500
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlations and find max
        corr = correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Update dual variables
        lambda_k += learning_rate * (corr - max_val)
        lambda_k = np.maximum(lambda_k, 0)  # Project to non-negative
        
        # Compute gradient wrt h
        grad = np.zeros(n_points)
        for k in range(2 * n_points - 1):
            if k < n_points:
                # h[i] appears in the correlation at position k with value (1-h[i-k])
                # shift by k
                temp = np.roll(1 - current_h, k)
                grad += lambda_k[k] * temp
            else:
                # h[i] appears in the correlation at position k with value (1-h[i+k-n_points])
                temp = np.roll(1 - current_h, k - (n_points - 1))
                grad += lambda_k[k] * temp
        grad = grad * 2 / n_points
        
        # Update h
        current_h = current_h - learning_rate * grad
        
        # Repair h
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)
        
        # Track best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            progress_count = 0
        else:
            progress_count += 1
            
        # Adaptive learning rate
        if progress_count > max_progress:
            learning_rate *= 0.99
            progress_count = 0
    
    # Final gradient descent refinement
    h_refined = best_h.copy()
    best_c5_refined = best_c5
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5_refined) / eps
            
        h_refined = h_refined - 0.00001 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_c5_refined = np.max(corr) * dx
        
        if best_c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = best_c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
