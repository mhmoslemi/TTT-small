import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h = (h_init > 0.5).astype(int)
        # Adjust to ensure exactly n_points/2 ones
        current_sum = np.sum(h)
        target_sum = n_points // 2
        
        if current_sum != target_sum:
            # Find indices to flip
            zero_indices = np.where(h == 0)[0]
            one_indices = np.where(h == 1)[0]
            
            if current_sum < target_sum:
                # Need to flip some 0s to 1s
                num_flip = target_sum - current_sum
                flip_indices = zero_indices[:num_flip]
                h[flip_indices] = 1
            else:
                # Need to flip some 1s to 0s
                num_flip = current_sum - target_sum
                flip_indices = one_indices[:num_flip]
                h[flip_indices] = 0
    else:
        # Default initialization
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # Create MIP model
    h_var = cp.Variable(n_points, boolean=True)
    t = cp.Variable()
    
    # Objective
    objective = cp.Minimize(t)
    
    # Constraints
    constraints = [cp.sum(h_var) == n_points // 2]
    
    # Maximum overlap constraint
    for k in range(-n_points + 1, n_points):
        # Valid indices for shift k
        valid_pairs = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_pairs.append((i, j))
        
        if valid_pairs:
            overlap_expr = sum(h_var[i] * (1 - h_var[j]) for i, j in valid_pairs)
            # Scale by dx to match amount (n_points/2) vs count
            constraints.append(overlap_expr <= t * (n_points / 2))
    
    # Formulate problem
    prob = cp.Problem(objective, constraints)
    
    # Set solver and solve
    try:
        prob.solve(solver=cp.CBC, max_seconds=budget_s - 10, gap Rel=0.01)
        if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_result = h
        else:
            h_result = h_var.value.astype(int)
            # Ensure sum constraint is still satisfied
            current_sum = np.sum(h_result)
            if current_sum != n_points // 2:
                # Rebalancing
                zero_indices = np.where(h_result == 0)[0]
                one_indices = np.where(h_result == 1)[0]
                if current_sum < n_points // 2:
                    for idx in zero_indices[:n_points//2 - current_sum]:
                        h_result[idx] = 1
                else:
                    for idx in one_indices[:current_sum - n_points//2]:
                        h_result[idx] = 0
            
    except Exception:
        h_result = h
        
    # Compute final C5 value
    h_final = np.array(h_result, dtype=float)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    # Reconstruct with correct scaling
    h_final = h_final.clip(0, 1)
    
    # Compute overlap for verification
    corr = np.correlate(h_final, 1 - h_final, mode='full')
    c5_bound = np.max(corr) * dx
    
    return (h_final.tolist(), c5_bound, n_points)
