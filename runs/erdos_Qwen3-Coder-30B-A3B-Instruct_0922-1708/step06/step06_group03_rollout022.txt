import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a more structured approach
        h_init = np.random.rand(n_points) * 0.5
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    max_iter = 100000
    improvement_threshold = 200
    progress_count = 0
    stagnation_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Periodic exploration
        if i % 100 == 0 and i > 0:
            # Perturb elements with low gradient contribution
            h_copy = current_h.copy()
            indices_to_perturb = np.where(np.abs(np.gradient(h_copy)) < 0.01)[0]
            if len(indices_to_perturb) > 0:
                num_perturb = min(10, len(indices_to_perturb))
                indices = np.random.choice(indices_to_perturb, num_perturb, replace=False)
                for idx in indices:
                    h_copy[idx] += np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
                h_copy = repair_h(h_copy, n_points)
                new_c5 = evaluate_h(h_copy, dx)
                if new_c5 < current_c5:
                    current_h = h_copy
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
                        progress_count = 0
                    else:
                        progress_count += 1
                else:
                    progress_count += 1
            
            # Decrease learning rate and perturbation
            learning_rate *= 0.99
            perturbation_magnitude *= 0.999
            
        # Compute gradients
        eps = 1e-6
        grad = np.zeros(n_points)
        
        # Compute the maximum overlap lag
        corr = correlate(current_h, 1 - current_h, mode='full')
        max_k = np.argmax(corr)
        
        # Approximate gradient using finite differences
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
            
        # Apply gradient descent step
        new_h = current_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject based on improvement
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()

                stagnation_count = 0
            else:
                progress_count += 1
                stagnation_count += 1
        else:
            progress_count += 1
            stagnation_count += 1
            
        # Occasionally add random perturbation to escape local minima
        if stagnation_count > 500 or progress_count > improvement_threshold:
            h_perturb = current_h.copy()
            indices = np.random.choice(n_points, max(1, n_points // 20), replace=False)
            for idx in indices:
                h_perturb[idx] += np.random.uniform(-0.01, 0.01)
            h_perturb = repair_h(h_perturb, n_points)
            perturb_c5 = evaluate_h(h_perturb, dx)
            if perturb_c5 < current_c5:

                current_c5 = perturb_c5
                if current_c5 < best_c5:

                    best_c5 = current_c5
                    progress_count = 0
                    stagnation_count = 0
            else:
                progress_count += 1
                stagnation_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
