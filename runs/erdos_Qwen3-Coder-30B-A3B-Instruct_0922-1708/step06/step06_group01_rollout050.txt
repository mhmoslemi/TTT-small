import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def repair_h(h, n_points):
    """Repair h to satisfy the constraints using quadratic programming."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Project onto the feasible set
        # Solve min ||h_new - h||^2 s.t. sum(h_new) = target_sum, 0 <= h_new <= 1
        # This is a QP problem - for simplicity use projection
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
        # Renormalize to ensure sum = target_sum
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-8:
            h = h * target_sum / current_sum
    
    return h

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
    
    # Gradient-based optimization approach
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Adaptive parameters
    lr = 0.01
    perturb_rate = 0.05
    max_iter = 20000
    patience = 1000
    last_improvement = 0
    
    # Gradient estimate function
    def estimate_gradient(h, eps=1e-6):
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_current) / eps
        return grad
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Estimate gradient
        grad = estimate_gradient(h_current)
        
        # Update with gradient descent
        h_new = h_current - lr * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate the new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject based on improvement
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            # Accept with probability to avoid local minima
            if np.random.rand() < 0.1:
                h_current = h_new
                c5_current = c5_new
        
        # Random perturbation
        if np.random.rand() < perturb_rate:
            indices = np.random.choice(n_points, max(1, int(perturb_rate * n_points)), replace=False)

                delta = np.random.uniform(-0.05, 0.05)

            if lr < 1e-6:
