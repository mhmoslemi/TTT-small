import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h = (h_init > 0.5).astype(int)
        h_sum = np.sum(h)
        if h_sum != n_points // 2:
            # Adjust to target sum by flipping some bits
            diff = n_points // 2 - h_sum
            if diff > 0:
                # Need to flip diff zeros to ones
                zero_indices = np.where(h == 0)[0]
                flip_indices = zero_indices[:diff]
                h[flip_indices] = 1
            else:
                # Need to flip -diff ones to zeros
                one_indices = np.where(h == 1)[0]
                flip_indices = one_indices[:(-diff)]
                h[flip_indices] = 0
    else:
        # Initialize with uniform binary pattern
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # MIP solver parameters
    time_limit = budget_s - 10
    max_gap = 0.01
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Solve using Mixed-Integer Programming
    try:
        # Define MIP variables
        h_var = cp.Variable(n_points, boolean=True)
        t = cp.Variable()
        
        # Constraints for sum
        constraints = [cp.sum(h_var) == n_points // 2]
        
        # For each shift k, add constraint on overlap
        for k in range(-n_points + 1, n_points):
            # Compute valid indices for this shift
            valid_indices = []
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    valid_indices.append((i, j))
            
            if valid_indices:
                # Add constraint: sum h[i](1 - h[j]) <= t * (n_points/2)
                overlap = cp.sum([h_var[i] * (1 - h_var[j]) for i, j in valid_indices])
                constraints.append(overlap <= t * (n_points / 2))
        
        # Objective: minimize t
        problem = cp.Problem(cp.Minimize(t), constraints)
        
        # Solve with timeout and gap
        problem.solve(solver=cp.CBC, max_seconds=time_limit, gap_limit=max_gap)
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = np.round(h_var.value).astype(int)
            best_h = h_sol
        else:
            # Fallback to evaluating current h
            pass
            
    except Exception:
        # If MIP fails, fall back to current h
        pass
    
    # Final evaluation of best solution
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
