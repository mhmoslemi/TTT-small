import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_shifted_correlations(h, dx, shift_grid):
    """Compute correlations for given shifts."""
    n_points = len(h)
    correlations = np.zeros(len(shift_grid))
    h_extended = np.concatenate([h, h, h])
    
    for i, k in enumerate(shift_grid):
        k_idx = int(k / dx) + n_points
        corr = np.sum(h * (1 - h_extended[k_idx:k_idx+n_points]))
        correlations[i] = corr * dx
    
    return correlations

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h = h_init.copy()
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize best solution
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    last_improvement = 0
    
    # Shift grid for correlation computation
    shift_grid = np.linspace(-2, 2, 100)
    
    # Optimization parameters
    learning_rate = 0.001
    max_iter = 10000
    perturbation_interval = 50
    refinement_interval = 25
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find the worst shift
        correlations = compute_shifted_correlations(h, dx, shift_grid)
        worst_k_idx = np.argmax(correlations)
        worst_k = shift_grid[worst_k_idx]
        
        # Compute gradient for the worst shift
        # For shift k, gradient of integral h(x)(1 - h(x + k)) is -h(x + k)
        # In discrete form, we look for derivation of correlation 
        # Approximate gradient by finite differences around current point
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            # Approximate how changing h[j] affects the worst correlation
            h_plus = h.copy()
            eps = 1e-6
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = compute_shifted_correlations(h_plus, dx, shift_grid)
            grad[j] = (corr_plus[worst_k_idx] - correlations[worst_k_idx]) / eps
        
        # Update h using gradient
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        
        # Evaluate new value and check improvement
        current_value = evaluate_h(h, dx)
        if current_value < best_value:
            best_value = current_value
            best_h = h.copy()
            last_improvement = iteration
        elif iteration - last_improvement > 200:
            # Reset learning rate if no improvement for a while
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                learning_rate = 1e-6
        
        # Add small random perturbation occasionally
        if iteration % perturbation_interval == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = repair_h(h, n_points)
            
        # Refine shift grid around worst shift if needed
        if iteration % refinement_interval == 0:
            refine_grid = np.linspace(worst_k - 0.1, worst_k + 0.1, 30)
            shift_grid = refine_grid
    
    # Final evaluation and cleanup
    final_value = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_value, n_points)
