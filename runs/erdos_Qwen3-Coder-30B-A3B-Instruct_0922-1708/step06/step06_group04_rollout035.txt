```python
import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    start_time = time.time()
    
    # Initialize h: use parent if available, otherwise random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to target sum
    h = project_h(h, target_sum)
    
    # Add small perturbation
    h += np.random.randn(n_points) * 0.1
    h = project_h(h, target_sum)
    
    # Regularization parameter
    lambda_reg = 0.01
    
    # Learning rate
    alpha = 0.1
    
    # Optimization parameters
    max_iter = 1000
    patience = 100
    best_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    stagnation_counter = 0
    last_c5 = best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current gradient
        grad = np.zeros(n_points)
        eps = 1e-6
        current_c5 = evaluate_h(h, dx)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, target_sum)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - current_c5) / eps
        
        # Add regularization gradient
        reg_grad = lambda_reg * np.sign(h)
        grad += reg_grad
        
        # Update step
        h_new = h - alpha * grad
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new value
        new_c5 = evaluate_h(h_new, dx)
        
        # Backtracking line search
        if new_c5 < current_c5:
            h = h_new
            last_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stagnation_counter = 0
            else:
                stagnation_counter += 1
            alpha *= 0.8  # Reduce learning rate on success
        else:
            alpha *= 1.2  # Increase learning rate on failure
            
        # Bound alpha
        alpha = np.clip(alpha, 1e-4, 1.0)
        
        # Check for stagnation
        if stagnation_counter > patience:
            # Restart with sparser solution
            h = np.clip(h, 0, 1)
            median_val = np.median(h)
            h = h * (h > median_val)
            h = project_h(h, target_sum)
            stagnation_counter = 0
            
        # Early termination
        if abs(last_c5 - best_c5) < 1e-4:
            stagnation_counter += 1
            if stagnation_counter > 50:
                break
            
    return (best_h.tolist(), best_c5, n_points)
```
