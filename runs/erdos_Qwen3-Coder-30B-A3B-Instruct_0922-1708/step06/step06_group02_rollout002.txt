import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

    coarse_h = coarse_h * (coarse_points / 2) / np.sum(coarse_h)

        h_full = np.repeat(h_coarse, 10)
        h_full = h_full * (n_points / 2) / np.sum(h_full)
        return evaluate_h(h_full, dx)
    
    def coarse_gradient(h_coarse):
        h_full = np.repeat(h_coarse, 10)
        h_full = h_full * (n_points / 2) / np.sum(h_full)
        eps = 1e-6
        grad = np.zeros_like(h_coarse)
        for i in range(len(h_coarse)):
            h_plus = h_coarse.copy()
            h_plus[i] += eps
            grad[i] = (coarse_objective(h_plus) - coarse_objective(h_coarse)) / eps
        return grad

        h_full = h_final * (n_points / 2) / np.sum(h_final)
