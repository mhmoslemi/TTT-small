import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints using quadratic programming."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) > 1e-8:
        # Project onto the feasible set
        diff = target_sum - current_sum
        # Simple uniform adjustment
        h += diff / n_points
        h = np.clip(h, 0, 1)
    
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
    
    # Parameters
    max_iter = 20000
    perturb_rate = 0.05
    learning_rate = 0.01
    patience = 1000
    
    # Current solution
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Track last improvement
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_current) / eps
        
        # Update with gradient
        h_new = h_current - learning_rate * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(c5_new - c5_current) / (c5_current + 1e-8)):
                h_current = h_new
                c5_current = c5_new
                
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive perturbation
        if i - last_improvement > patience:
            perturb_rate = min(0.2, perturb_rate + 0.01)
            learning_rate = max(1e-6, learning_rate * 0.9)
            
        # Random perturbations
        if np.random.rand() < perturb_rate:
            num_perturb = max(1, int(perturb_rate * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                h_current[idx] += np.random.normal(0, 0.01)
            h_current = repair_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)

            c5_best = c5_current
