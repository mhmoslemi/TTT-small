import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent approach with gradient steps
    h = current_h.copy()
    dual_vars = np.ones(2 * n_points - 1)  # One for each lag
    dual_vars /= np.sum(dual_vars)  # Normalize
    dual_temp = 0.1
    learning_rate = 5e-4
    stagnation_count = 0
    max_stagnation = 1000
    
    for iteration in range(100000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all overlaps
        overlaps = correlate(h, 1 - h, mode='full')
        
        # Update dual variables with variational inference
        # λ_k ∝ exp(-overlap_k / τ)
        dual_vars = np.exp(-overlaps / dual_temp)
        dual_vars /= np.sum(dual_vars)
        dual_temp *= 0.999
        
        # Compute gradient: ∂L/∂h[i] = sum(λ_k * gradient_wrt_h[i])
        # For h[i], the gradient comes from terms where h[i] appears in overlap_k
        gradient = np.zeros(n_points)
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                # h[i] contributes to overlap_k only for i in [0, n_points-k-1]
                for i in range(n_points - k):
                    gradient[i] += dual_vars[k + n_points - 1] * (1 - h[i + k])
            else:
                # h[i+k] contributes to overlap_k only for i in [-k, n_points-1]
                for i in range(-k, n_points):
                    gradient[i] += dual_vars[k + n_points - 1] * (1 - h[i + k])
        
        # Update h
        h = h - learning_rate * gradient
        h = repair_h(h, n_points)
        
        # Evaluate and track best
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adjust learning rate
        if iteration % 500 == 0:
            learning_rate *= 0.99
            
        # Reset if stagnating
        if stagnation_count > max_stagnation:
            h = best_h.copy()
            stagnation_count = 0
            if learning_rate > 1e-7:
                learning_rate *= 0.5

    return (best_h.tolist(), best_c5, n_points)
