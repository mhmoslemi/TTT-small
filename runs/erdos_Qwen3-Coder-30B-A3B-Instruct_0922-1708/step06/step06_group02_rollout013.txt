import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def compute_gradient(h, worst_k, dx, n_points):
    """Compute gradient for the worst shift."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        # Evaluate partial derivative
        j = i - worst_k
        if j >= 0 and j < n_points:
            grad[i] = -2 * (h[j] + 0.5) * dx
        j = i + worst_k
        if j >= 0 and j < n_points:
            grad[i] = grad[i] - 2 * (h[j] + 0.5) * dx
            
    return grad

            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        
    # Track the best solution
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Grid for k in -2 to 2
    k_grid = np.linspace(-2, 2, 100)
    k_indices = np.arange(-n_points, n_points + 1)
    
    # Main optimization loop
    learning_rate = 0.01
    no_improvement_count = 0
    last_best = best_value
    
    for iteration in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all correlations
        corr = correlate(h, 1 - h, mode='full')
        # Compute total overlaps for all k
        full_corr = corr * dx
        
        # Find maximum overlap
        max_idx = np.argmax(full_corr)
        worst_k = max_idx - n_points + 1  # convert to k index
        
        # Compute gradient for worst shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i - worst_k
            if 0 <= j < n_points:
                grad[i] -= 2 * dx * h[j]
            j = i + worst_k
            if 0 <= j < n_points:
                grad[i] -= 2 * dx * h[j]
        
        # Update with gradient step
        h = h - learning_rate * grad
        
        # Repair h to satisfy constraints
        h = repair_h(h, n_points)

        current_value = evaluate_h(h, dx)
        
        # Update best solution if improved
        if current_value < best_value:

            no_improvement_count = 0

        if no_improvement_count > 100:

        if abs(best_value - last_best) < 1e-10:
