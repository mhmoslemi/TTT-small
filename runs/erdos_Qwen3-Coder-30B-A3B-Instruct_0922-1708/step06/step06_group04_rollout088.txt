import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum=50.0):
    """Project h to feasible region: [0,1] and sum=h_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.ones(n_points) * 0.5
    
    # Add small random perturbation
    h += 0.1 * np.random.randn(n_points)
    h = project_h(h, target_sum)
    
    # Parameters
    lambda_reg = 0.01
    alpha = 0.1
    max_iter = 1000
    convergence_window = 100
    min_improvement = 1e-4
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    last_improvement = 0
    prev_c5 = best_c5
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, target_sum)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
        
        # Regularization gradient (L1)
        reg_grad = lambda_reg * np.sign(h)
        
        # Update
        h_new = h - alpha * (grad + reg_grad)
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept if better
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            last_improvement = iteration
            stagnation_count = 0
            if new_c5 < best_c5:
                best_h = h.copy()
        else:
            stagnation_count += 1
            # Adaptive learning rate
            if stagnation_count > 10:
                alpha *= 0.8
                stagnation_count = 0
            if stagnation_count > 50:
                # Reinitialize in case of stall
                h = np.random.rand(n_points)
                h = h * (target_sum / np.sum(h))
                alpha = 0.1
                last_improvement = iteration
                
        # Bound alpha
        alpha = np.clip(alpha, 1e-4, 1.0)
        
        # Convergence check
        if abs(prev_c5 - best_c5) < min_improvement:
            if iteration - last_improvement > convergence_window:
                break
        prev_c5 = best_c5
    
    return (best_h.tolist(), best_c5, n_points)
