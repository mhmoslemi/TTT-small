import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Proximal gradient parameters
    gamma = 0.05  # Step size
    beta = 0.9   # Nesterov momentum
    max_iter = 10000
    
    # Variables for Nesterov acceleration
    v = np.zeros(n_points)
    h_prev = h.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation for all lags
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Identify the maximum lag
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h) - 1)
        
        # Compute gradient w.r.t. this shift
        grad = np.zeros_like(h)
        # Gradient of h[i] w.r.t. overlap at shift k
        for i in range(len(h)):
            if i + k >= 0 and i + k < len(h):
                grad[i] -= (1 - h[i + k]) * dx
        
        # Nesterov acceleration
        v = beta * v + grad
        h_acc = h - gamma * v
        
        # Proximal projection onto constraints
        h_new = np.clip(h_acc, 0, 1)
        h_sum = np.sum(h_new)
        if abs(h_sum) > 1e-10:
            h_new = h_new * (n_points / 2) / h_sum
        
        # Update with momentum
        h = h_new
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
