import numpy as np
import time
from scipy.signal import correlate
from scipy.stats import.qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def halton_sequence(n, skip=0):
    """Generate Halton sequence for 1D."""
    sampler = qmc.Halton(d=1, scramble=False, seed=42)
    sample = sampler.random(n + skip)
    return sample[skip:][:, 0]

def sobol_sequence(n, skip=0):
    """Generate Sobol sequence for 1D."""
    sampler = qmc.Sobol(d=1, scramble=False, seed=42)
    sample = sampler.random(n + skip)
    return sample[skip:][:, 0]

def initialize_h(n_points, method='halton'):
    """Initialize h using low-discrepancy sequences."""
    if method == 'halton':
        seq = halton_sequence(n_points, skip=0)
    elif method == 'sobol':
        seq = sobol_sequence(n_points, skip=0)
    else:
        # Fallback to random
        seq = np.random.rand(n_points)
    
    # Normalize to [0, 2] and convert to step function
    x = seq * 2
    h = np.zeros(n_points)
    
    # Sort x and distribute the step function
    sorted_indices = np.argsort(x)
    x_sorted = x[sorted_indices]
    
    # Create step function by placing points in bins
    bin_width = 2.0 / n_points
    for i in range(n_points):
        x_val = x_sorted[i]
        bin_index = int(x_val / bin_width)
        if bin_index >= n_points:
            bin_index = n_points - 1
        h[bin_index] += 1
    
    # Normalize to sum = n_points / 2
    h = repair_h(h, n_points)
    
    return h

def finite_difference_gradient(h, dx, eps=1e-6):
    """Compute gradient of C5 with respect to h using finite differences."""
    n_points = len(h)
    grad = np.zeros(n_points)
    c5_current = evaluate_h(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n_points)
        c5_plus = evaluate_h(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = repair_h(h_minus, n_points)
        c5_minus = evaluate_h(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    return grad

    num_seeds = 10
    for seed_idx in range(num_seeds):
