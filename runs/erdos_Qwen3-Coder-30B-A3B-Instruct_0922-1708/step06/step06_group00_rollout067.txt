import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Optimizer using convex approach
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Barrier method parameters
    mu = 1.0
    max_iter = 500
    penalty_factor = 1.1
    min_mu = 1e-8
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # The objective is to minimize t such that 
        # t >= sum h[i] * (1 - h[i + k]) for all k
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Formulate a simple optimization to guide search
        # Using trust-region ideas: small gradient updates
        h = h.copy()
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_sum = np.sum(h_plus)
            if abs(h_sum) > 1e-10:
                h_plus = h_plus * (n_points / 2) / h_sum
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
            
        # Update with gradient descent
        learning_rate = 0.001
        h = h - learning_rate * grad
        
        # Project h to [0,1] and rescale
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum

        # Adapt mu for barrier if needed
        if mu > min_mu:
            mu *= penalty_factor

    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
