import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    pop_size = 50
    F = 1.0
    CR = 0.9
    dx = 2.0 / n_points
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():

        x_orig = np.linspace(0, 2, len(h_init))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_init)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Fill remaining population with random vectors
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    stagnation_count = 0
    max_stagnation = 100
    
    generation = 0
    while time.time() - start_time < budget_s - 5 and stagnation_count < max_stagnation:
        new_population = [best_h.copy()]
        new_fitness = [best_c5]
        
        for _ in range(pop_size - 1):
            # Select three distinct individuals
            indices = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            v = a + F * (b - c)
            v = np.clip(v, 0, 1)
            
            # Normalize to meet integral constraint
            v = repair_h(v, n_points)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, v, population[indices[0]])
            
            # Evaluate
            c5_trial = evaluate_h(trial, dx)
            new_population.append(trial)
            new_fitness.append(c5_trial)
        
        # Selection: keep top half
        sorted_indices = np.argsort(new_fitness)[:pop_size//2]
        population = [new_population[i] for i in sorted_indices]
        fitness = [new_fitness[i] for i in sorted_indices]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        generation += 1
        
        # Adaptive F
        if stagnation_count > 50:
            F = min(1.5, F + 0.1)
    
    # Final refinement
    for _ in range(100):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5

    return (best_h.tolist(), best_c5, n_points)
