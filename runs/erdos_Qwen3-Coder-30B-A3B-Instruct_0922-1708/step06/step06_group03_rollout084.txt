import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent with projected gradient descent
    # Initialize dual variables for each lag
    lambda_k = np.zeros(2 * n_points - 1)
    learning_rate = 5e-4
    max_iter = 100000
    patience = 100
    progress_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute current correlations for all lags
        corr_vals = correlate(current_h, 1 - current_h, mode='full')
        
        # Update dual variables for complementarity
        # Maximize the overlap, so we focus on the maximum
        max_idx = np.argmax(corr_vals)
        lambda_k += learning_rate * (corr_vals - np.max(corr_vals))
        lambda_k = np.maximum(lambda_k, 0)  # Non-negative
        
        # Gradient of Lagrangian with respect to h
        grad_h = np.zeros(n_points)
        for k in range(2 * n_points - 1):
            if k < n_points:
                # h[idx] contributes to corr_vals[k] for indices idx < n_points - k
                # (1 - h[idx + k]) 
                right = lambda_k[k]
                for j in range(n_points - k):
                    grad_h[j] += right * (-1) * (1 - current_h[j + k])
            else:
                # j + k >= n_points
                # h[idx] contributes for indices 0 <= j < k - n_points + 1
                left = lambda_k[k]
                for j in range(k - n_points + 1):
                    grad_h[j + (k - n_points + 1)] += left * (-1) * (1 - current_h[j])
        
        # Update h
        current_h -= learning_rate * grad_h
        current_h = repair_h(current_h, n_points)
        
        # Compute new C5 value
        new_c5 = evaluate_h(current_h, dx)
        
        if new_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = new_c5
            progress_count = 0
        else:
            progress_count += 1
            
        # Adaptive learning rate
        if progress_count >= patience:
            learning_rate *= 0.99
            progress_count = 0
            
        # Early termination on plateau
        if progress_count > 200:
            break
    
    # Final refinement with gradient descent
    h_refined = best_h.copy()
    best_c5_refined = best_c5
    
    for _ in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5_refined) / eps
            
        h_refined = h_refined - 0.00005 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_c5_refined = np.max(corr) * dx
        
        if best_c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = best_c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
