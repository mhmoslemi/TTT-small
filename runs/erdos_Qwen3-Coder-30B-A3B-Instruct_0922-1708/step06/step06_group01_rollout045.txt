import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.linalg import norm

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, eps=1e-6):
    n = len(h)
    grad = np.zeros(n)
    h_base = h.copy()
    for i in range(n):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        c5_plus = evaluate_h(h_plus, dx)
        
        h_minus = h_base.copy()
        h_minus[i] -= eps
        h_minus = repair_h(h_minus, n)
        c5_minus = evaluate_h(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    return grad

def trust_region_step(h, dx, grad, hessian, trust_radius):
    n = len(h)
    # Simplified trust region using projected gradient method
    # We use a simple approximation for the subproblem
    # This is an approximation; using full TR solves would be more precise
    step = -grad
    step_norm = norm(step)
    if step_norm > trust_radius:
        step = step * trust_radius / step_norm
    
    trial = h + step
    trial = repair_h(trial, n)
    return trial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a low discrepancy sequence
        h_init = np.array([np.sin(i * 2 * np.pi / n_points) * 0.5 + 0.5 for i in range(n_points)])
        h_init = repair_h(h_init, n_points)

    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current

    # Trust-region parameters
    trust_radius = 0.5
    trust_radius_max = 1.0
    trust_radius_min = 1e-6
    alpha = 0.25
    beta = 2.0
    max_iterations = 2000

    # Main trust region loop
    iterations = 0
    while iterations < max_iterations and time.time() - start_time < budget_s - 5:
        # Compute gradient
        grad = compute_gradient(h_current, dx)

        # Compute Hessian approximation
        # For simplicity, we use an identity matrix
        hessian = np.eye(n_points)

        # Take a trust region step
        h_new = trust_region_step(h_current, dx, grad, hessian, trust_radius)

        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)

        # Accept step or not
        if c5_new <= c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
            trust_radius = min(trust_radius_max, trust_radius * beta)
        else:
            trust_radius = max(trust_radius_min, trust_radius * alpha)

        # Periodic perturbation to escape local optima
        if iterations % 250 == 0:
            perturbation = np.random.normal(0, 0.02, n_points)
            h_current = h_current + perturbation
            h_current = repair_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)

        iterations += 1

    # Final local refinement using scipy minimize
    def objective(h):
        return evaluate_h(h, dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    # Initialize with best found solution
    refined_sol = minimize(objective, h_best, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
    if refined_sol.success:
        h_final = repair_h(refined_sol.x, n_points)
        c5_final = evaluate_h(h_final, dx)
        if c5_final < c5_best:
            h_best = h_final
            c5_best = c5_final

    return (h_best.tolist(), c5_best, n_points)
