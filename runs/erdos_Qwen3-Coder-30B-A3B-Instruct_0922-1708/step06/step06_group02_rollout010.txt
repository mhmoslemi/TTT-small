import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Repair to ensure correct sum
    h = repair_h(h, n_points)
    
    # Best solution tracking
    best_h = h.copy()
    best_value = evaluate_h(h, dx)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 10000
    no_improvement_limit = 200
    
    # Main optimization loop
    iter_count = 0
    no_improvement_count = 0
    last_best = best_value
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 5:
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        k_indices = np.arange(-n_points + 1, n_points)
        max_k_idx = np.argmax(corr)
        worst_k = k_indices[max_k_idx]
        
        # Compute gradient directly at the worst k
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Check indices for valid bounds
            if (i + worst_k) >= 0 and (i + worst_k) < n_points:
                gradient[i] = -dx * (h[i + worst_k] + h[i - worst_k] if i - worst_k >= 0 and i - worst_k < n_points else 0)
        
        # Update h
        h = h + learning_rate * gradient
        h = repair_h(h, n_points)
        
        # Evaluate current solution
        current_value = evaluate_h(h, dx)
        
        # Update best if improved
        if current_value < best_value:
            best_value = current_value
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Adaptive learning rate
        if no_improvement_count % 100 == 0 and no_improvement_count > 0:
            learning_rate *= 0.8
            
        # Perturb occasionally to escape local minima
        if iter_count % 50 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = h + noise
            h = repair_h(h, n_points)
            
        iter_count += 1
        
        # Check for convergence
        if no_improvement_count > no_improvement_limit:
            break
            
    return (best_h.tolist(), best_value, n_points)
