import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def halton_sequence(n, base=2):
    """Generate Halton sequence for given base."""
    result = np.zeros(n)
    for i in range(n):
        f = 1.0
        result[i] = 0.0
        while i > 0:
            f /= base
            result[i] = (i % base) * f
            i //= base
    return result

def initialize_h(n_points, seed=42):
    """Initialize h using low-discrepancy sequences."""
    np.random.seed(seed)
    # Try multiple methods
    methods = [
        lambda: np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_points), 
                          halton_sequence(n_points, 2)),
        lambda: np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, n_points), 
                          halton_sequence(n_points, 3)),
        lambda: np.random.rand(n_points),
        lambda: np.ones(n_points) * 0.5,
        lambda: np.linspace(0, 1, n_points) < 0.5,
        lambda: np.ones(n_points) * 0.25,
        lambda: np.ones(n_points) * 0.75,
        lambda: np.sin(np.linspace(0, 4*np.pi, n_points)) * 0.5 + 0.5,
    ]
    # Use the method that gives best feasible result
    best_h = None
    best_c5 = float('inf')
    for method in methods:
        h = method()
        h = repair_h(h, n_points)
        c5 = evaluate_h(h, 2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    return best_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        h_init = initialize_h(n_points, seed)

    # Track best solution
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Optimization parameters
    max_iter = 10000
    step_size = 0.001
    patience = 200
    
    # Gradient approximation
    eps = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient approximation
        grad = np.zeros(n_points)
        h_current = h_best.copy()
        c5_current = c5_best
        
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] = min(1.0, h_plus[j] + eps)
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] = max(0.0, h_minus[j] - eps)
            h_minus = repair_h(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with gradient step
        h_new = h_current - step_size * grad
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement
        if c5_new < c5_current:
            h_best = h_new
            c5_best = c5_new
            
        # Adaptive step size
        if i % 500 == 0 and i > 0:
            step_size *= 0.95
            
    return (h_best.tolist(), c5_best, n_points)
