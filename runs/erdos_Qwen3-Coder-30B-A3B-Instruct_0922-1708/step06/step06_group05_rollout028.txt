import numpy as np
import time
import cvxpy as cp
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Rescale to meet integral constraint exactly
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Parameters
    lr = 0.001
    max_iter = 5000
    stagnation_limit = 100
    stagnation_count = 0
    adaptive_lr = lr
    
    # Try convex relaxation first
    try:
        # Define CVXPY problem
        h_var = cp.Variable(n_points)
        # Convolution constraint matrix (approximate)
        # We'll use a simpler relaxation strategy
        objective = cp.Minimize(cp.sum_squares(h_var) + 1e-8 * cp.sum(cp.abs(h_var)))
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-6, reltol=1e-6)
        if prob.status == cp.OPTIMAL:
            h = h_var.value
            h = repair_h(h, n_points)
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    except:
        pass
    
    # Gradient descent loop  
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
            
        # Update
        h = h - adaptive_lr * grad
        h = repair_h(h, n_points)
        
        # Evaluate
        c5 = evaluate_h(h, dx)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
            # Adaptive learning rate increase
            adaptive_lr = min(adaptive_lr * 1.05, 0.01)
        else:
            stagnation_count += 1
            # Adaptive learning rate decrease
            adaptive_lr = max(adaptive_lr * 0.95, 1e-6)
            
        # Restart with random perturbation if stuck
        if stagnation_count > stagnation_limit:
            h = best_h.copy()
            # Add noise
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = repair_h(h, n_points)
            stagnation_count = 0
            
        # Occasionally enforce integral constraint via projection
        if iteration % 50 == 0:
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
