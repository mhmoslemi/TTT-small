import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    pop_size = 50
    F = 1.0
    CR = 0.9
    max_generations = 1000
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    # Fill population
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    population = np.array(population)
    
    best_h = None
    best_c5 = float('inf')
    
    stagnation_count = 0
    generation = 0
    
    dx = 2.0 / n_points
    
    while generation < max_generations and (time.time() - start_time) < budget_s - 5:
        fitness = np.array([evaluate_h(h, dx) for h in population])
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Evolution
        new_population = []
        for i in range(pop_size):
            # Select three distinct individuals
            candidates = list(range(pop_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            v = population[a] + F * (population[b] - population[c])
            v = np.clip(v, 0, 1)
            v = repair_h(v, n_points)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = v[j]
            
            new_population.append(trial)
        
        population = np.array(new_population)
        generation += 1
        
        # Adaptive adjustment of F
        if stagnation_count > 50:
            F = min(1.5, F + 0.1)
            stagnation_count = 0
    
    # Final refinement
    if best_h is not None:
        for _ in range(100):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Gradient-free optimization using random search around best
            new_h = best_h.copy() + np.random.normal(0, 0.001, n_points)
            new_h = repair_h(new_h, n_points)
            c5_new = evaluate_h(new_h, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = new_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
