import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5

    # Variational dual ascent approach
    dual_vars = np.ones(2 * n_points - 1)  # λ_k for k in [-n+1, n-1]
    dual_vars /= np.sum(dual_vars)  # Normalize to sum to 1
    dual_temp = 0.1

    # Learning rate schedule
    lr = 5e-4
    lr_decay = 0.99
    lr_decay_freq = 500

    # Gradient descent parameters
    max_iter = 30000
    stagnation_count = 0
    max_stagnation = 500

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break

        # Compute all overlaps
        corr = correlate(current_h, 1 - current_h, mode='full')
        overlaps = corr * dx

        # Update dual variables (variational inference)
        # λ_k ∝ exp(-overlap_k / dual_temp)
        dual_vars = np.exp(-overlaps / dual_temp)
        dual_vars /= np.sum(dual_vars)

        # Update learning rate every few iterations
        if i % lr_decay_freq == 0 and i > 0:
            lr *= lr_decay

        # Compute gradient and update h
        grad = np.zeros(n_points)
        for j in range(n_points):
            for k in range(-n_points + 1, n_points):
                idx = k + n_points - 1
                if 0 <= idx < len(dual_vars):
                    if k <= j:
                        grad[j] += dual_vars[idx] * (1 - current_h[j - k])

        # Gradient update
        current_h = current_h - lr * grad
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)

        # Track best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1

        # Temperature decay for dual update
        dual_temp *= 0.999

        # Adaptive learning rate if stagnating
        if stagnation_count > max_stagnation:
            # Reset to best solution
            current_h = best_h.copy()
            current_c5 = best_c5
            stagnation_count = 0

    return (best_h.tolist(), best_c5, n_points)
