import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of C5 with respect to h."""
    n_points = len(h)
    grad = np.zeros(n_points)
    # For each position j, compute contribution to gradient
    for j in range(n_points):
        # When k = k_max, we're looking at correlation at lag k_max
        # which involves h[i] * (1 - h[(i - k_max) % n_points])
        # So d/dh_j of this term is (1 - h[(j - k_max) % n_points]) 
        # if i = j, 0 otherwise
        # But since k_max is fixed, we consider how many terms involve h[j]:
        for k in range(n_points):
            if (j - k) % n_points == k_max:
                grad[j] += (1 - h[(j - k_max) % n_points])
    return grad * dx

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Parameters
    learning_rate = 1e-4
    perturb_mag = 0.05
    max_iter = 100000
    improvement_window = 100
    perturb_window = 500
    progress_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap for all lags to find k_max
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Compute gradient of the maximum overlap w.r.t h
        grad = compute_gradient(current_h, dx, k_max)
        
        # Update h using gradient descent
        new_h = current_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept the new solution
        if new_c5 < current_c5 or np.random.rand() < 0.1:
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                progress_count = 0
            else:
                progress_count += 1
        else:
            progress_count += 1

            indices = np.random.choice(n_points, max(1, n_points // 10), replace=False)
