import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
        
    # For Nesterov acceleration
    v = np.zeros_like(h)
    beta = 0.9
    
    # For tracking best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Step size and learning rate
    gamma = 0.05
    max_iter = 10000
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation for all lags
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the lag that maximizes overlap
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h) - 1)
        
        # Compute gradient for this specific lag k
        grad = np.zeros_like(h)
        
        # Compute the corresponding shift in indices
        if k >= 0:
            # h[i] * (1 - h[i+k])
            grad[:len(h)-k] -= (1 - h[k:])
        else:
            # h[i+k] * (1 - h[i]) when k < 0
            grad[-k:] -= (1 - h[:len(h)+k])
            
        # Gradient descent step with proximal operator
        h_new = h - gamma * grad
        
        # Proximal projection onto box constraints [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy sum constraint
        h_sum = np.sum(h_new)
        if abs(h_sum) > 1e-10:
            h_new = h_new * (n_points / 2) / h_sum
            
        # Nesterov acceleration
        v = beta * v + (h_new - h)
        h = h_new + beta * v
        
        # Ensure feasibility if we're exceeding bounds
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
