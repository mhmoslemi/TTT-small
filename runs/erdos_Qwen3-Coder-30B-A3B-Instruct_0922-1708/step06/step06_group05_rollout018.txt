import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    # Fill remaining population with random h vectors
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    population = np.array(population)
    
    # Parameters
    F = 1.0
    CR = 0.9
    best_c5 = float('inf')
    best_h = None
    stagnation_count = 0
    
    # Evolution loop
    generation = 0
    max_generations = 10000
    
    while generation < max_generations and time.time() - start_time < budget_s - 5:
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
        
        # Create new population
        new_population = []
        
        for i in range(pop_size):
            # Select three distinct individuals
            indices = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            v = a + F * (b - c)
            v = np.clip(v, 0, 1)
            v = repair_h(v, n_points)
            
            # Crossover
            trial = np.copy(population[i])
            trial_mask = np.random.rand(n_points) < CR
            trial[trial_mask] = v[trial_mask]
            
            # Ensure integral constraint
            trial = repair_h(trial, n_points)
            
            new_population.append(trial)
        
        # Selection (keep top individuals)
        fitness = np.array(fitness)
        sorted_indices = np.argsort(fitness)
        population = population[sorted_indices[:pop_size]]
        
        # Keep new individuals for selection
        new_population = np.array(new_population)
        all_populations = np.vstack([population, new_population])
        all_fitness = []
        for h in all_populations:
            all_fitness.append(evaluate_h(h, dx))
        all_fitness = np.array(all_fitness)
        sorted_indices = np.argsort(all_fitness)
        population = all_populations[sorted_indices[:pop_size]]
        
        # Update stagnation
        if len(fitness) > 0 and np.min(fitness) >= best_c5 - 1e-6:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        # Adapt F if stagnating
        if stagnation_count > 100:
            F = min(2.0, F + 0.1)
            stagnation_count = 0
            
        # Increment generation
        generation += 1
        
    # Fine-tune best solution
    if best_h is not None:
        h = best_h.copy()
        for _ in range(500):
            if time.time() - start_time > budget_s - 2:
                break
            eps = 1e-6
            grad = np.zeros(n_points)
            c5_eval = evaluate_h(h, dx)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                grad[i] = (c5_plus - c5_eval) / eps
            h = h - 0.0002 * grad
            h = repair_h(h, n_points)
            new_c5 = evaluate_h(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
