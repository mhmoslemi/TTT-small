import numpy as np
import time
import cvxpy as cp
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fft_correlation(h, reverse_h):
    """Efficiently compute correlation using FFT."""
    H = np.fft.fft(h, 2 * len(h) - 1)
    Rev = np.fft.fft(reverse_h, 2 * len(h) - 1)
    corr = np.fft.ifft(H * Rev).real
    return corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a larger point count for better resolution
    n_points = 200
    dx = 2.0 / n_points

    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
        
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)

    # Convex relaxation for good starting point
    try:
        # Optimization using CVXPY
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(
            cp.sum_squares(cp.convolution(h_var, 1 - h_var, 'full')) + 
            1e-6 * cp.sum(h_var)
        )
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = h_var.value
    except:
        pass

    h = repair_h(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)

    # Gradient-based refinement with adaptive learning rate
    max_iter = 10000
    lr = 0.01
    beta = 0.9
    decay_factor = 0.95

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute finite difference gradient using FFT for efficiency
        grad = np.zeros(n_points)
        h_eval = evaluate_h(h, dx)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - h_eval) / eps

        # Update with adaptive learning rate
        lr = lr * (decay_factor ** (iteration // 1000))
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        # Update best solution
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5

    return (best_h.tolist(), best_c5, n_points)
