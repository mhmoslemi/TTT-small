import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or random binary
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Start with a symmetric binary construction
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Convert to binary representation (for tabu search)
    h_binary = (h > 0.5).astype(float)
    
    # Tabu search parameters
    max_iter = 50000
    tabu_size = 50
    tabu_list = []
    
    # Annealing parameters
    T = 0.1
    cooling_rate = 0.999
    
    # Initialize
    best_h = h_binary.copy()
    best_c5 = float('inf')
    
    # Current solution
    current_h = h_binary.copy()
    current_c5 = float('inf')
    
    # Initial evaluation
    corr = correlate(current_h, 1 - current_h, mode='full')
    current_c5 = np.max(corr) * dx
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Main loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
        
        # Generate candidate solution by flipping one bit
        candidate_h = current_h.copy()
        flip_idx = np.random.randint(0, n_points)
        candidate_h[flip_idx] = 1 - candidate_h[flip_idx]
        
        # Ensure constraint sum(h) = n_points/2
        ones_count = np.sum(candidate_h)
        if ones_count > n_points / 2:
            # Find zeros to flip to ones
            zero_indices = np.where(candidate_h == 0)[0]
            if len(zero_indices) > 0:
                flip_zero = np.random.choice(zero_indices)
                candidate_h[flip_zero] = 1
                candidate_h[flip_idx] = 0
        elif ones_count < n_points / 2:
            # Find ones to flip to zeros
            one_indices = np.where(candidate_h == 1)[0]
            if len(one_indices) > 0:
                flip_one = np.random.choice(one_indices)
                candidate_h[flip_one] = 0
                candidate_h[flip_idx] = 1
        
        # Check if in tabu list
        if tuple(candidate_h.astype(int)) in tabu_list:
            # Accept with low probability
            accept_prob = 0.05
            if np.random.rand() > accept_prob:
                continue
        
        # Evaluate
        corr = correlate(candidate_h, 1 - candidate_h, mode='full')
        candidate_c5 = np.max(corr) * dx
        
        # Accept if better
        if candidate_c5 < current_c5:
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = candidate_h.copy()
        else:
            # Accept with probability based on simulated annealing
            delta = candidate_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
        
        # Update tabu list
        tabu_list.append(tuple(current_h.astype(int)))
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Cool down
        T *= cooling_rate
    
    # Final refinement - switch to continuous optimization for final step
    h_continuous = best_h.astype(float)
    # Project to [0,1] and rescale
    h_continuous = h_continuous * (n_points / 2) / np.sum(h_continuous)
    
    # Gradient step on the final solution
    for _ in range(100):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        corr = correlate(h_continuous, 1 - h_continuous, mode='full')
        c5 = np.max(corr) * dx
        
        # Simple gradient estimation
        grad = np.zeros_like(h_continuous)
        eps = 1e-6
        for i in range(len(h_continuous)):
            h_test = h_continuous.copy()
            h_test[i] += eps
            if h_test[i] > 1:
                h_test[i] = 1
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            corr_test = correlate(h_test, 1 - h_test, mode='full')
            c5_test = np.max(corr_test) * dx
            grad[i] = (c5_test - c5) / eps
            
        # Update
        h_continuous -= 0.001 * grad
        h_continuous = np.clip(h_continuous, 0, 1)
        
        # Rescale
        h_continuous = h_continuous * (n_points / 2) / np.sum(h_continuous)
    
    # Final evaluation
    corr = correlate(h_continuous, 1 - h_continuous, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_continuous.tolist(), final_c5, n_points)
