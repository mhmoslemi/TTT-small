import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    F = 1.0
    CR = 0.9

        h = repair_h(h, n_points)
        population.append(h)
    
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    
    for generation in range(max_generations):

        # Evaluate population
        fitness = []

                best_h = h.copy()
                stagnation_count = 0
                
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]

        new_population = [sorted_population[0]]  # Elitism

            # Mutation
            v = a + F * (b - c)

            trial = np.where(np.random.rand(n_points) < CR, v, sorted_population[i])

        # Adaptive F
