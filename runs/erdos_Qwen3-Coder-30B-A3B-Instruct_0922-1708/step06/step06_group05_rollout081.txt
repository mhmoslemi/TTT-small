import numpy as np
import time
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    # Use scipy.signal.correlate for efficient computation
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Rescale to meet integral constraint exactly
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Interpolate to match n_points
        x_orig = np.linspace(0, 2, len(h))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Convex relaxation stage with SCA
    max_convex_iter = 50
    for iter_convex in range(max_convex_iter):
        if time.time() - start_time > budget_s - 60:
            break
            
        # Formulate as convex optimization with one
        # convex constraint for the worst-case correlation
        h_cvx = cp.Variable(n_points)
        # Maximize sum of all correlations
        # but first compute a feasible h
        corr_cached = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr_cached)
        max_corr_val = corr_cached[max_corr_idx]
        
        # Create quadratic surrogate
        objective = cp.Minimize(
            cp.sum_squares(h_cvx - h) + 0.01 * cp.sum_squares(cp.conv(h_cvx, 1-h_cvx)[1:n_points+1]))
        
        constraints = [
            h_cvx >= 0,
            h_cvx <= 1,
            cp.sum(h_cvx) == n_points / 2
        ]
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h = h_cvx.value
                h = repair_h(h, n_points)
        except:
            pass
            
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Projected gradient refinement
    lr = 0.01
    beta = 0.9
    max_refine_iter = 500
    last_improvement = 0
    
    for iteration in range(max_refine_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient efficiently in frequency domain via FFT
        # Correlation derivative: 
        # d/dh_k (sum_i h_i (1 - h_{i + k})) ≈ 1 - 2 h_k - (1-h_{k-1} + 1-h_{k+1}) 
        # But we're using correlation with 1-h, i.e. (h * (1-h)), so
        # equiv. to computing: sum_i of  [ h_i * d/dh_k (1 - h_{i+k}) ]
        # = sum_i of [ h_i * (-delta_{i+k, k}) ] = -h_k
        # But want d/dh_k [sum_i h_i (1 - h_{i+k}) ] = 1 - h_{k+1} - h_{k-1}
        # For k = 0, we get 1 - h_1 - h_{-1} = 1 - h_1 (since h_{-1} = 0)
        # For k = n_points-1, we get 1 - h_{n_points-2} - h_{n_points}
        #                         = 1 - h_{n_points-2} (since h_{n_points} = 0)
        # But controlled by convolution with (1, -1, -1), shift and finite tail
        # Using full correlation derivative:
        h_r = h.copy()
        gradient = np.zeros(n_points)
        
        h_1 = 1 - h
        hcorr = np.correlate(h, h_1, mode='full')
        # idx shift from time domain convolution to correlation
        # we have h * (1-h)[k] = sum_i h[i] * (1-h[i+k])
        # Now note that for a fixed shift k,
        # d/dh_i [h[i] * (1-h[i+k])] = delta_i^k (1-h[i+k]) - h[i] delta_{i+k}^i
        #                                = (1-h[i+k]) delta_i^k - h[i] delta_{i+k}^i  
        # for i > 0, equiv. to (1-h[k]) delta_i^k - h[i-k] delta_k^i
        # Give the gradient vector in index space:
        # d/dh_0: 1-h[0] + h[1] + h[2] + ... 
        # d/dh_1: 1-h[1] + h[2] + h[3] + ... 
        # ... etc
        # It's i->0 to n_points, but we build correlation for shift k,
        # so compute: 1 - h * (1 - h) 
        # Derivative is (1 - (1-h) * h) = h * (1-h) + h - h = h * (1-h) + h(1-h) = 2 h * (1-h)
        # Create.
        # Compute convolution (h * (1 - h)) via fft.
        # First reverse the last, pad first 
        # Use the fact that df/dh = gradient around the max arg
        # Based on the structure of correlation derivative,
        # d/dh_k [sum_i h[i] * (1 - h[i+k])] = 1 - h_{k+1} - h_{k-1}  
        # This is convolution of h with [-1, 1, -1], not including endpoints (0 and 2)
        # Core idea: core nonderivativeness appears for edges
        # for k=0, because of wraparound in correlation, in reverse logic:
        # h[0] * (1 - h[0]), h[1] * (1 - h[1]), ...
        # last elements are contribs from first part
        # Final derivative z_k := d/dh_k [h * (1 - h)][k] = 
        # det.h._k coeff is 1 - h_{k+1} - h_{k-1}
        corr_h = np.correlate(h, h_1, mode='full')
        max_corr_idx = np.argmax(corr_h)
        # Gradient direction:
        G = np.zeros_like(h)
        for k in range(n_points):
            left = np.roll(h, k+1)
            right = np.roll(h, k-1)
            if k == 0:
                # pos shift k = 0: match to k = 0, index edge
                G[k] = 1 - left[k]  # -h[k-1] is 0 at k=0, h_1[k+1] 
            elif k == n_points-1:
                # k =n-1   
                G[k] = 1 - right[k]
            else:
                # Remove 2 away iff i = k+1 or k-1, add one extra 1-hk+1
                G[k] = 1 - left[k] - right[k] 
              
        # now because we want the dx scaling
        G = G * dx
        
        # Projected gradient based on FFT
                
        # Use central difference approximation based on finite difference
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_eval = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_eval) / eps
            
        # Adaptive learning rate  
        eta = lr / (1 + iteration ** beta)
        
        # Update steps
        h_new = h - eta * grad
        h_new = repair_h(h_new, n_points)
        
        c5_new = evaluate_h(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Take step
        h = h_new.copy()
        
        if last_improvement > 100:
            # Escalate with artificial hill climb
            h = h + np.random.normal(0, 0.001, n_points)
            h = repair_h(h, n_points)
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
    return (best_h.tolist(), best_c5, n_points)
