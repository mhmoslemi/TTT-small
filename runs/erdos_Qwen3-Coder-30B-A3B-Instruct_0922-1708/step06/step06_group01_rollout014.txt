import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = np.maximum(0, 1 - np.abs(x - 1))
        h_init = repair_h(h_init, n_points)
    
    # Coordinate-wise optimization with projection
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Parameters
    eta = 0.01
    eps = 1e-6
    max_iter = 50000
    patience = 100
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic coordinate updates
        for j in range(n_points):
            if time.time() - start_time > budget_s - 5:
                break
            # Compute gradient using finite differences
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = repair_h(h_plus, n_points)
            h_minus = repair_h(h_minus, n_points)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_current[j] = h_current[j] - eta * grad_j
            h_current[j] = max(0, min(1, h_current[j]))
            
        # Repair and evaluate
        h_current = repair_h(h_current, n_points)
        c5_current = evaluate_h(h_current, dx)
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
        
        # Adaptive step size
        if i - last_improvement > patience:
            eta *= 0.9
            last_improvement = i
            
        # Perturb sometimes to escape local minima
        if i % 1000 == 0:
            perturb_indices = np.random.choice(n_points, max(1, n_points // 100))
            for idx in perturb_indices:
                h_current[idx] += np.random.uniform(-0.01, 0.01)
            h_current = repair_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current

    return (h_best.tolist(), c5_best, n_points)
