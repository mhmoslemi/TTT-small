import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Store best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Barrier method parameters
    mu = 1.0
    mu_factor = 1.1
    max_iter = 100
    
    # Optimize
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate current solution
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Construct convex problem
        h_var = Variable(n_points)
        t = Variable()
        
        # Objective: minimize t
        objective = Minimize(t)
        
        # Constraints
        constraints = []
        
        # Constraint: sum of h = n_points / 2
        constraints.append(sum(h_var) == n_points / 2)
        
        # Constraint: h in [0,1]
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Constraint: t >= overlap_k for all k
        max_lag = len(h) - 1
        for i in range(2 * max_lag + 1):
            k = i - max_lag
            if k >= 0:
                overlap_expr = 0
                for j in range(len(h) - k):
                    overlap_expr += h_var[j] * (1 - h_var[j + k])
                overlap_expr *= dx
                constraints.append(t >= overlap_expr)
            else:
                overlap_expr = 0
                for j in range(len(h) + k):
                    overlap_expr += h_var[j] * (1 - h_var[j - k])
                overlap_expr *= dx
                constraints.append(t >= overlap_expr)
        
        # Add barrier term to objective
        barrier = 0
        for i in range(n_points):
            barrier += -mu * (np.log(h_var[i]) + np.log(1 - h_var[i]))
        
        # Reshape problem with barrier
        objective = Minimize(t + barrier)
        
        # Solve problem
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                h_new = np.array(h_var.value).flatten()
                h = np.clip(h_new, 0, 1)
                h_sum = np.sum(h)
                if abs(h_sum - n_points / 2) > 1e-6:
                    h = h * (n_points / 2) / h_sum
            else:
                # Fallback if optimization fails
                h = best_h.copy()
        except:
            # Fallback to best solution
            h = best_h.copy()
            
        # Adjust barrier parameter
        mu *= mu_factor

    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
