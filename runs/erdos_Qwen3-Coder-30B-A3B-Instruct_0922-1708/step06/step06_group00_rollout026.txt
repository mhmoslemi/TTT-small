import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50000
    initial_temp = 0.1
    cooling_rate = 0.999
    tabu_size = 50
    
    # Initialize from parent or random binary
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Convert to binary for tabu search
    h_binary = (h > 0.5).astype(float)
    
    # Track current best solution
    best_h = h_binary.copy()
    best_c5 = float('inf')
    
    # Tabu list to avoid revisiting recent moves
    tabu_list = []
    
    # Annealing parameters
    temperature = initial_temp
    
    # Initial C5 computation
    corr = correlate(h_binary, 1 - h_binary, mode='full')
    current_c5 = np.max(corr) * dx
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_binary.copy()
    
    # Simulated annealing with tabu search
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by flipping a random bit
        candidate_idx = np.random.randint(0, n_points)
        h_candidate = h_binary.copy()
        h_candidate[candidate_idx] = 1 - h_candidate[candidate_idx]  # Flip bit
        
        # Check tabu list
        if (candidate_idx, h_candidate[candidate_idx]) in tabu_list:
            continue
            
        # Compute overlap for candidate
        corr = correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(corr) * dx
        
        # Accept or reject based on simulated annealing
        if candidate_c5 < current_c5:
            # Accept improvement
            h_binary = h_candidate.copy()
            current_c5 = candidate_c5
            # Update tabu list
            tabu_list.append((candidate_idx, h_candidate[candidate_idx]))
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        else:
            # Accept with probability
            if np.random.random() < np.exp(-(candidate_c5 - current_c5) / temperature):
                h_binary = h_candidate.copy()
                current_c5 = candidate_c5
                # Update tabu list
                tabu_list.append((candidate_idx, h_candidate[candidate_idx]))
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_binary.copy()
        
        # Update temperature
        temperature *= cooling_rate
    
    # Refine best binary solution with gradient-based optimization
    h_refined = best_h.astype(float)
    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
    
    # Allow some local optimization using simple gradients
    for _ in range(1000):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current overlap and gradient
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (len(h_refined) - 1)
        
        # Simple gradient ascent on individual values
        step_size = 0.001
        for i in range(len(h_refined)):
            # Approximate gradient by finite difference
            h_plus = h_refined.copy()
            h_plus[i] += 1e-5
            if h_plus[i] > 1:
                h_plus[i] = 1.0
                
            h_plus_sum = np.sum(h_plus)
            h_plus = h_plus * (n_points / 2) / h_plus_sum
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad = (c5_plus - current_c5) / 1e-5
            h_refined[i] -= step_size * grad
            h_refined[i] = max(0, min(1, h_refined[i]))
        
        # Rescale
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Recompute current C5
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
    
    return (best_h.tolist(), best_c5, n_points)
