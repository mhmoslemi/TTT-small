import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimizer parameters
    learning_rate = 1e-4
    perturbation_magnitude = 0.05
    max_iter = 100000
    progress_count = 0
    max_progress = 500
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradients using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        current_overlap = evaluate_h(current_h, dx)
        
        # Compute gradient with respect to maximum overlap
        # This is a simplified approximation approach
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            overlap_plus = evaluate_h(h_plus, dx)
            grad[j] = (overlap_plus - current_overlap) / eps
            
        # Perform gradient descent step
        h_new = current_h - learning_rate * grad
        h_new = repair_h(h_new, n_points)
        new_c5 = evaluate_h(h_new, dx)
        
        # Apply perturbations periodically 
        if i % 100 == 0 and i > 0:
            # Select indices with low gradient magnitude to perturb
            abs_grad = np.abs(grad)
            low_grad_indices = np.where(abs_grad < 0.1)[0]
            if len(low_grad_indices) > 0:
                num_perturb = min(10, len(low_grad_indices))
                perturb_indices = np.random.choice(low_grad_indices, num_perturb, replace=False)
                for idx in perturb_indices:
                    perturbation = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
                    h_new[idx] += perturbation
                h_new = repair_h(h_new, n_points)
                new_c5 = evaluate_h(h_new, dx)
        
        # Accept new solution with simulated annealing-like probability
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / max(1e-8, current_c5)):
            current_h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                progress_count = 0
            else:
                progress_count += 1
        else:
            progress_count += 1
            
        # Adaptive parameters
        if i % 100 == 0:
            learning_rate *= 0.99
            perturbation_magnitude *= 0.999
            
        if progress_count > max_progress:
            learning_rate = max(1e-6, learning_rate * 0.95)
            perturbation_magnitude = max(1e-6, perturbation_magnitude * 0.95)
            progress_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
