import numpy as np
import time
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_halton_sequence(n_points, seed=42):
    # Generate Halton sequence in [0,1]
    sampler = qmc.Halton(d=1, seed=seed)
    points = sampler.random(n_points)
    return points.flatten()

def init_from_low_discrepancy(n_points, seed=42):
    # Generate a sequence and reshape to step function
    seq = generate_halton_sequence(n_points, seed)
    # Convert to step function with suitable resolution
    # Use the sequence to generate convex combination of step functions
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = seq[i]  # Simple linear arrangement
    # Normalize and clip
    h = repair_h(h, n_points)
    return h

                h_plus[j] += eps

                c5_plus = np.max(corr_plus) * dx
