import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1, sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    
    return h

def fw_step(h, dx, epsilon=1e-6):
    """Compute Frank-Wolfe step direction."""
    grad = np.zeros_like(h)
    h_base = h.copy()
    
    for i in range(len(h)):
        h_plus = h_base.copy()
        h_plus[i] += epsilon
        h_plus = project_h(h_plus, len(h))
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        grad[i] = (c5_plus - evaluate_h(h_base, dx)) / epsilon
    
    # Solve linear subproblem: min_d <grad, d> s.t. 0 <= d[i] <= 1, sum(d) = 0
    # This is a weighted median problem with zero sum constraint
    # Simple approach: take gradient direction but make it have zero sum
    
    # Compute the gradient projection onto the zero-sum space
    grad_mean = np.mean(grad)
    d = grad - grad_mean
    
    # Now we need to make sure it's feasible and zero sum
    # Let's ensure no components are outside [0,1] while maintaining zero sum
    d = project_h(d, len(h)) - project_h(np.zeros_like(h), len(h))  # This should effectively just adjust
    # A safer approach, follow the L1 gradient portion in line search:
    return d

def line_search(h, d, dx, max_iter=10):
    """Perform line search to find optimal step size."""
    c5_current = evaluate_h(h, dx)
    max_step = 1.0
    for i in range(max_iter):
        h_new = h + max_step * d
        h_new = project_h(h_new, len(h))
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_current:
            return max_step, h_new
        max_step /= 2.0
    return max_step, h

    dx = 2.0 / n_points
    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Initialize with some reasonable construction
        # For example, a piecewise linear function that increases then decreases
        x = np.linspace(0, 2, n_points)
        h_init = np.ones(n_points) * 0.5
        # Try a triangular shape
        h_init = np.where((x >= 0.5) & (x <= 1.5), 1.0 - abs(x - 1.0), 0.0)
        h_init = project_h(h_init, n_points)

    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    h_best = h_current.copy()
    c5_best = c5_current
    
    patience = 0
    patience_limit = 1000
    
    iter_count = 0
    while time.time() - start_time < budget_s - 5 and patience < patience_limit:
        iter_count += 1
        # Apply Frank-Wolfe direction
        d = fw_step(h_current, dx)
        
        # Line search
        step_size, h_proposed = line_search(h_current, d, dx)

        if c5_current < c5_best:

            c5_best = c5_current
            patience = 0
        else:
            patience += 1
