import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent parameters
    dual_weights = np.ones(2 * n_points - 1)  # for each lag
    dual_weights /= np.sum(dual_weights)
    dual_learning_rate = 0.01
    temp = 0.1
    dual_update_freq = 10
    
    # Gradient descent parameters
    lr = 5e-4
    lr_decay = 0.99
    decay_freq = 500
    stagnation_count = 0
    max_stagnation = 500
    
    iteration = 0
    while time.time() - start_time < budget_s - 5 and stagnation_count < max_stagnation:
        iteration += 1
        
        # Evaluate overlaps
        h = current_h
        corr = correlate(h, 1 - h, mode='full')
        overlaps = corr * dx
        
        # Update dual weights via variational inference
        if iteration % dual_update_freq == 0:
            # Exponentiate with temperature
            exp_overlaps = np.exp(-overlaps / temp)
            dual_weights = exp_overlaps / np.sum(exp_overlaps)
            temp *= 0.999
        
        # Compute gradient of the Lagrangian w.r.t. h
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(-n_points + 1, n_points):
                if 0 <= i - k < n_points:
                    grad[i] += dual_weights[k + n_points - 1] * (1 - h[i - k])
        
        # Update h
        h_new = h - lr * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new C5
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept or reject update
        if new_c5 < current_c5:
            current_h = h_new
            current_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Update learning rate
        if iteration % decay_freq == 0:
            lr *= lr_decay
        
        # Track best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)
