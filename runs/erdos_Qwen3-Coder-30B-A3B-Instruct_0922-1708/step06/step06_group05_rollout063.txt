import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    # Use scipy.signal.correlate for efficient computation
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Rescale to meet integral constraint exactly
        h = h * target_sum / current_sum
    return h

    start_time = time.time()
    
    # Start with low resolution
    n_points = 20
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available

        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
    
    # Adjust to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Simulated Annealing parameters
    T0 = 1.0
    T = T0
    alpha = 0.99
    lr = 0.01
    max_iter = 100000
    
    # Multi-scale parameters
    resolutions = [20, 40, 80, 200]
    scale_idx = 0
    scale_changes = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Occasionally update resolution
        if iteration > 0 and iteration % 500 == 0 and scale_idx < len(resolutions) - 1:
            scale_idx += 1
            old_n = n_points
            n_points = resolutions[scale_idx]
            dx = 2.0 / n_points
            
            # Interpolate best_h to new resolution
            x_old = np.linspace(0, 2, old_n)
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, best_h)
            h = repair_h(h, n_points)
            best_h = h.copy()
            best_c5 = evaluate_h(h, dx)
            scale_changes += 1
        
        # Perturbation with current temperature
        h_new = h + np.random.normal(0, T, n_points)
        h_new = repair_h(h_new, n_points)
        
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Update temperature
        T = T * alpha
        
        # Periodic gradient descent refinement
        if iteration % 100 == 0:
            # Compute gradient with finite differences
            grad = np.zeros(n_points)
            eps = 1e-6
            c5_eval = evaluate_h(h, dx)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                grad[i] = (c5_plus - c5_eval) / eps
            
            # Update with gradient
            h = h - lr * grad
            h = repair_h(h, n_points)
            
            # Adjust learning rate
            if iteration % 1000 == 0:

    for _ in range(1000):
