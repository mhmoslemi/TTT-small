import numpy as np
import time
import cvxpy as cp
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

            dummy_shifted = np.zeros(n_points * 2 - 1)

    # Main refinement via projected gradient descent with adaptive learning rate
    # Step 1: Use a relatively efficient norm for derivative estimation 
    # Step 2: Alternate between full gradient steps and trust region steps
    
    lr = 0.01
    decay_exponent = 0.9
    max_iter = 5000

    def compute_correlation(h_vals):
        """Returns the full correlation array, c[shift] = sum h[i] * (1 - h[i+shift])"""
        h_shift = np.pad(h_vals, (0, len(h_vals)-1), 'constant')
        corr = np.correlate(h_shift, 1 - h_vals, mode='valid')
        return corr
    
    def gradient_at_point(h_vals, i):
        """Compute gradient component at position i using finite difference"""
        eps = 1e-6
        h_plus = h_vals.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n_points)
        corr_plus = compute_correlation(h_plus)
        c5_plus = np.max(corr_plus) * dx
        corr = compute_correlation(h_vals)
        c5 = np.max(corr) * dx
        return (c5_plus - c5) / eps

    def adaptive_step(h_vals, trial_steps):
        """Choose the best small adaptation from a handful of test steps"""
        base_c5 = evaluate_h(h_vals, dx)
        best_h = h_vals
        best_c5 = base_c5

            new_h = repair_h(new_h, n_points)
            new_c5 = evaluate_h(new_h, dx)

    best_h = repair_h(best_h, n_points)
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
