import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Dual ascent parameters
    dual_lambda = np.ones(2*n_points - 1)  # For all lags
    dual_lambda /= np.sum(dual_lambda)  # Normalize to sum to 1
    dual_lr = 1.0
    dual_temp = 0.1
    dual_update_freq = 10
    
    # Gradient descent parameters
    lr = 5e-4
    lr_decay = 0.99
    decay_interval = 500
    
    # For stagnation detection
    patience = 500
    no_improvement_count = 0
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        
        # Compute overlaps for current h
        corr_full = correlate(current_h, 1 - current_h, mode='full')
        overlaps = corr_full
        
        # Update dual variables using variational inference
        if iteration % dual_update_freq == 0:
            # Compute new dual variables (softmax over overlaps with temperature)
            dual_new = np.exp(-overlaps / dual_temp)
            dual_new /= np.sum(dual_new)
            dual_lambda = dual_lambda * 0.9 + dual_new * 0.1
            dual_temp *= 0.999
        
        # Compute gradient of Lagrangian w.r.t. h
        # ∂L/∂h[i] = sum_k λ_k * (δ[i-k] - h[i-k] if k ≤ i else 0) - η
        grad = np.zeros(n_points)
        # Use the fact that ∫ h(x)(1 - h(x+k)) dx is a convolution,
        # so the gradient in h is also a convolution with 1-h and -h
        
        # The Lagrangian gradient per position i is:
        #   sum_k λ_k * (1 - h[i-k])  if k ≥ 0,
        #   sum_k λ_k * (0)            if k < 0.
        # So δ[i-k] is just 1 for k <= i, and -h[i-k] (which is -1 * h[i-k]) for k >= 0.
        # For the specific computation, i-indexing requires care of the convolution structure.
        # Compute ∇ (sum λ_k * overlap_k) w.r.t. h:
        # Each overlap_k = ∫ h(x) (1-h(x+k))dx, so d(overlap_k)/dh[i] = 1-h[i+k] for x=i.
        # This leads to:
        # ∇ overlap_k / ∂ h[i] = (1-h[i+k] if i+k >=0 else 0) - (h[i+k] if i+k >=0 else 0)
        # Different from a full convolution, but let's do it properly:
        # Let Z_k = ∫_0^2 h(x)(1 - h(x+k)) dx.
        # ∂Z_k/∂h[i] = integral over x where x+i = i, so x = i.
        # But we are taking the derivative of the integral with respect to h[x].
        # This is the kernel convolution.
        # ∂Z_k/∂h[x] = 1 - h[x+k]
        # So ∂(∑ λ_k Z_k)/∂h[x] = ∑ λ_k * (1 - h[x+k]) for valid k that puts x+k in the domain.
        
        # Compute ∑ λ_k * dZ_k/dh[x]
        # Start with a zero-gradient
        grad = np.zeros(n_points)
        # Each term: λ_k * (1 - h[i-k])  but must consider boundaries
        # This is a convolution, let's compute it properly
        for k in range(-n_points+1, n_points):
            i_start = max(0, -k)   # Shift so that i-k >= 0  
            i_end = min(n_points, n_points-k) # i-k < n_points
            if i_end > i_start:
                for i in range(i_start, i_end):
                    grad[i] += dual_lambda[(n_points-1)+k] * (1 - current_h[i-k])
                    
        # The Lagrangian also includes a penalty term, the summation constraint:
        # L = sum(λ_k * Z_k) - η (sum(h) - n_points/2)
        # The term in gradient due to constraint is -η
        # η is chosen such that ∑h = n_points/2, not needed to compute gradient directly
        # For now we assume the gradient w.r.t the sum is already taken into account
        # by fixing h with projection using min/max and rescaling
        
        # Update h using gradient descent
        current_h -= lr * grad
        current_h = repair_h(current_h, n_points)
        
        # Evaluate current solution
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best if better
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Adapt learning rate
        if iteration % decay_interval == 0:
            lr *= lr_decay
            
        # Adapt dual temperature
        dual_temp *= 0.999
        
        # Early termination on stagnation
        if no_improvement_count > patience:
            # Reset to best solution
            current_h = best_h.copy()
            no_improvement_count = 0
            
    return (best_h.tolist(), best_c5, n_points)
