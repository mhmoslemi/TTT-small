import numpy as np
import time
from scipy.signal import correlate

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.full(n_points, 0.5)
    
    # Add small noise to break symmetry
    h += np.random.normal(0, 0.05, n_points)
    h = project_h(h, target_sum)
    
    # Parameters
    lambda_reg = 0.01
    alpha = 0.1
    max_iter = 1000
    patience = 100
    best_value = float('inf')
    best_h = h.copy()
    stagnation_counter = 0
    prev_value = evaluate_h(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, target_sum)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        # Add L1 regularization gradient
        reg_grad = lambda_reg * np.sign(h)
        grad += reg_grad
        
        # Update with backtracking line search
        h_new = h - alpha * grad
        h_new = project_h(h_new, target_sum)
        new_value = evaluate_h(h_new, dx)
        
        # Backtracking
        while new_value >= prev_value and alpha > 1e-6:
            alpha *= 0.5
            h_new = h - alpha * grad
            h_new = project_h(h_new, target_sum)
            new_value = evaluate_h(h_new, dx)
        
        # Accept update
        h = h_new
        prev_value = new_value
        
        if prev_value < best_value:
            best_h = h.copy()
            best_value = prev_value
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        # Adjust learning rate
        if prev_value < new_value:
            alpha *= 0.8
        else:
            alpha *= 1.2
            
        alpha = np.clip(alpha, 1e-4, 1.0)
        
        # Check for stagnation
        if stagnation_counter > patience:
            # Increase regularization strength and reset
            lambda_reg *= 2
            alpha = 0.1
            stagnation_counter = 0
            
            # Perturb solution
            h += np.random.normal(0, 0.01, n_points)
            h = project_h(h, target_sum)
            prev_value = evaluate_h(h, dx)
    
    return (best_h.tolist(), best_value, n_points)
