import numpy as np
import time
import cvxpy as cp
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h using correlation."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy bounds [0,1] and integral constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():

        # Interpolate to match n_points

        h = np.interp(x_target, x_orig, h)
    else:
        # Random initialization with feasible sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Project to feasible region
    h = repair_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Convex relaxation stage (first ~30s)
    if time.time() - start_time < budget_s - 30:
        # Convex formulation
        h_var = cp.Variable(n_points)
        # Objective: minimize max correlation term using smooth approximation
        # Placeholder for SCA approach - simplified version
        try:
            # Solve convex relaxation using CVXPY
            prob = cp.Problem(
                cp.Minimize(cp.sum_squares(h_var)), 
                [
                    cp.sum(h_var) == n_points/2,
                    h_var >= 0,
                    h_var <= 1
                ]
            )
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h = h_var.value
                h = repair_h(h, n_points)
                current_c5 = evaluate_h(h, dx)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
        except:
            pass  # Skip if convex solve fails
    
    # Gradient-based refinement
    lr = 0.01
    beta = 0.9
    stagnation_count = 0
    stagnation_limit = 50
    
    # Efficient gradient computation using FFT
    def compute_grad_fft(h, dx):
        # h is the function values
        # compute correlation: h * (1-h)
        h1 = h
        h2 = 1 - h
        # Pad for full convolution via FFT
        nfft = 2 * n_points - 1
        h1p = np.pad(h1, (0, nfft - n_points), 'constant')
        h2p = np.pad(h2, (0, nfft - n_points), 'constant')
        corr_fft = np.fft.fft(h1p) * np.conj(np.fft.fft(h2p))
        corr = np.fft.ifft(corr_fft).real[:n_points]
        
        # Find maximum index
        max_idx = np.argmax(corr)
        
        # Gradient computation - only care about contributing terms
        grad = np.zeros_like(h)

            # Construct h_plus by shifting values around index
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus = repair_h(h_plus, n_points)
            # Approximate gradient
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - best_c5) / 1e-6
            
        return grad

    for i in range(10000):

                h_plus = repair_h(h_plus, n_points)

        # Update with adaptive step size
        adaptive_lr = lr / (1 + i ** beta)

            new_c5 = evaluate_h(updated_h, dx)
            if new_c5 < best_c5:
                best_h = updated_h.copy()
                best_c5 = new_c5
                stagnation_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
