import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    lr_initial = 0.001
    lr_decay = 0.999
    max_iter = 5000
    n_restarts = 5
    
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_current = h.copy()
        lr = lr_initial
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            iteration += 1
            
            # Compute the overlap integral
            correlation = np.correlate(h_current, 1 - h_current, mode='full')
            c5 = np.max(correlation) * dx
            active_k = np.argmax(correlation)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_current.copy()
            
            # Compute exact gradient for the active lag
            grad = np.zeros_like(h_current)
            n = len(h_current)
            for j in range(n):
                # Gradient w.r.t h[j] is 1 - h[(j-k)%n] - h[(j+k)%n] where k is the active lag
                k = active_k - n + 1  # Convert from correlation index to actual lag
                grad[j] = (1 - h_current[(j - k) % n]) - h_current[(j + k) % n]
            
            # Update h
            h_new = h_current - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_current = h_new
            
            # Adaptive learning rate
            lr *= lr_decay
            
            # Occasionally check for early termination
            if i % 100 == 0 and i > 0:
                if time.time() - start_time > budget_s - 5:
                    break
    
    return (best_h.tolist(), best_c5, n_points)
