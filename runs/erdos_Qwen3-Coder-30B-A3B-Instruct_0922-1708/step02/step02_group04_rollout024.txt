import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize population
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 45
        h0 = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize initial solution
    h0 = h0 * (n_points / 2) / np.sum(h0)
    h0 = np.clip(h0, 0, 1)
    
    # Parameters
    pop_size = min(50, max(10, n_points // 2))
    mutation_rate = 0.1
    crossover_rate = 0.8
    max_generations = 1000
    stagnation_limit = 100
    
    # Initialize population
    population = []
    population_fitness = []
    
    # Add the initial solution
    population.append(h0.copy())
    correlation = correlate(h0, 1 - h0, mode='full')
    fitness = np.max(correlation) * dx
    population_fitness.append(fitness)
    
    # Fill rest with random solutions
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
        correlation = correlate(h, 1 - h, mode='full')
        fitness = np.max(correlation) * dx
        population_fitness.append(fitness)
    
    # Track best solution
    best_idx = np.argmin(population_fitness)
    best_h = population[best_idx].copy()
    best_c5 = population_fitness[best_idx]
    
    # For tracking stagnation
    last_improvement = 0
    stagnation_count = 0
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and stagnation_count < stagnation_limit:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(population_fitness)
        population = [population[i] for i in sorted_indices]
        population_fitness = [population_fitness[i] for i in sorted_indices]
        
        # Update best if improved
        if population_fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = population_fitness[0]
            last_improvement = generation
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Selection (roulette wheel)
        fitness_array = np.array(population_fitness)
        fitness_array = np.maximum(1e-8, 1.0 / (fitness_array + 1e-10))
        fitness_sum = np.sum(fitness_array)
        probabilities = fitness_array / fitness_sum
        
        # Create new population
        new_population = [population[0]]  # Elitism: keep best
        new_fitness = [population_fitness[0]]
        
        while len(new_population) < pop_size:
            # Selection
            parents = np.random.choice(pop_size, size=2, p=probabilities)
            parent1 = population[parents[0]]
            parent2 = population[parents[1]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    # Random Gaussian mutation
                    for i in range(n_points):
                        if np.random.rand() < 0.1:
                            child[i] += np.random.normal(0, 0.05)
                    
                    # Clamp to [0,1]
                    child = np.clip(child, 0, 1)
                    
                    # Normalize
                    child = child * (n_points / 2) / np.sum(child)
                    
                    # Evaluate fitness
                    correlation = correlate(child, 1 - child, mode='full')
                    fitness = np.max(correlation) * dx
                    new_population.append(child)
                    new_fitness.append(fitness)
        
        # Truncate to population size
        population = new_population[:pop_size]
        population_fitness = new_fitness[:pop_size]
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
