import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation to get a good starting point
    if time.time() - start_time < budget_s * 0.6:
        # Create variables for convex optimization
        h_vars = cp.Variable(n_points)
        
        # Constraints for the convex relaxation
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Objective is to minimize an upper bound on overlap
        # We'll create a simple convex surrogate for the max overlap
        # By using the fact that h(i) * (1 - h(i+k)) is upper bounded by 0.25 when
        # both values are 0.5 (worst-case for this product), but we can do better
        # with proper optimization.
        try:
            # Simple convex approximation: minimize sum of all products with 1-h[i+k]
            # But note that this is still not exact. Let's do a direct convex approach
            # Actually, let's approximate more directly: minimize sum of 
            # overapproximations. But for a better approach, let's solve a simpler
            # primal LP with finite differences to get approximate structure
            h_relaxed = h.copy()
            
            # Try a simple heuristic to get a better solution - adjust it with some smoothness
            # And use the previous solution as a starting point if possible
            
            # Let's just start with a better smooth initialization
            x = np.linspace(0, 2, n_points)
            # Try to have some peak or step profile
            h_relaxed = np.exp(-((x - 1.0)**2) / 0.2)  # Gaussian centered at x=1
            h_relaxed = h_relaxed / np.sum(h_relaxed) * (n_points / 2)
            h_relaxed = np.clip(h_relaxed, 0, 1)
            
            # Ensure the constraint is satisfied exactly
            h = h_relaxed.copy()
            
        except:
            pass
    
    # Now refine using gradient descent with projected updates
    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < 2000:
        iterations += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
