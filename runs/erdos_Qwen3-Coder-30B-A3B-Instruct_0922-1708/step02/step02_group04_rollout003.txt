import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Start with a symmetric configuration
        h = np.ones(n_points) * 0.5
        # Slightly perturb to break symmetry
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Generate perturbation
        perturb = np.random.normal(0, 0.02, n_points)
        h_new = h + perturb
        h_new = np.clip(h_new, 0, 1)
        # Rescale and normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Occasionally do larger changes
        if np.random.rand() < 0.05:
            # Target a few elements for sharper changes
            indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
            for i in indices:
                if np.random.rand() < 0.5:
                    h_new[i] = np.random.rand()
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            
        h = h_new
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
