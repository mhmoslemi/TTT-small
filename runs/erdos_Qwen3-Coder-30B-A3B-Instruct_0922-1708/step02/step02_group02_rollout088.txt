import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        # Create variants
        for _ in range(population_size):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    mutation_rate = 0.2
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for stagnation
        if generation > 1 and fitness[0] == fitness[1]:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        if no_improvement_count >= 100:
            mutation_rate = 0.3
            
        # Selection: keep top 20%
        keep_count = max(1, population_size // 5)
        selected = population[:keep_count]
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Crossover
            if random.random() < 0.5 and len(selected) >= 2:
                parent1, parent2 = random.sample(selected, 2)
                crossover_point = random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            
            # Mutation
            else:
                parent = random.choice(selected)
                child = parent.copy()
                num_mutations = max(1, int(mutation_rate * n_points))
                mutation_indices = random.sample(range(n_points), num_mutations)
                for i in mutation_indices:
                    child[i] += random.gauss(0, 0.02)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
                
        population = new_population[:population_size]
        
        # Occasionally rescale to avoid numerical issues
        if generation % 50 == 0:
            for i in range(len(population)):
                population[i] = population[i] * (n_points / 2) / np.sum(population[i])
    
    # Final evaluation of best solution
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
    else:
        # Re-evaluate final population if no best was found
        final_fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            final_fitness.append(c5)
        best_idx = np.argmin(final_fitness)
        best_h = population[best_idx]
        final_c5 = final_fitness[best_idx]
        
    return (best_h.tolist(), final_c5, n_points)
