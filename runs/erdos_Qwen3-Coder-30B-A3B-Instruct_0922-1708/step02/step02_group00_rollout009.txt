```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with provided initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        # Use a heuristic starting point
        h = np.ones(n_points) * 0.5
    
    # Scale to meet constraint ∫h(x)dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.001
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute full correlation to find the maximum overlap shift
        corr = correlate(h, 1 - h, mode='full')
        # The overlap integral corresponds to the maximum correlation value
        #_scaled_max_corr = np.max(corr) * dx
        max_corr_idx = np.argmax(corr)
        
        # Compute C5 using the maximum correlation
        # There are (2*n_points - 1) elements in the full correlation
        # The maximum corresponds to shift k, where:
        # k = max_corr_idx - (n_points - 1)
        # But we can also directly compute C5 as max(corr) * dx
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient with respect to the maximum shift
        # Forward pass to calculate the gradient
        grad = np.zeros_like(h)
        max_shift = max_corr_idx - (n_points - 1)
        
        # The gradient for the j-th point of f(x) = sum_i h_i (1 - h_{i+k})
        # For shift k, it's ∂f/∂h_j = (1 - h_{j+k}) if j+k < n_points
        # And ∂f/∂h_j += -h_{j-k} if j-k ≥ 0
        
        for i in range(n_points):
            if i + max_shift < n_points and i + max_shift >= 0:
                grad[i] += (1 - h[i + max_shift])
            
            if i - max_shift >= 0:
                grad[i] += -h[i - max_shift]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        
        # Bias correction
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        
        # Update h
        h = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
```
