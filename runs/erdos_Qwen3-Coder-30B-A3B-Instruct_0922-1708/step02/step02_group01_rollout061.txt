import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Binary search bounds
    C_low = 0.0
    C_high = 0.5
    epsilon = 1e-4
    best_h = h.copy()
    best_C5 = float('inf')
    
    start_time = time.time()
    
    while (C_high - C_low > epsilon) and (time.time() - start_time < budget_s - 1):
        C_mid = (C_low + C_high) / 2.0
        
        # Formulate LP feasibility problem
        # Minimize 0 subject to:
        # h[i] >= 0, h[i] <= 1
        # sum(h) = n_points/2
        # For each shift k, sum(h[i] * (1 - h[(i+k) % n_points]) * dx) <= C_mid
        
        # Setup constraints
        A_eq = np.zeros((1, n_points))
        A_eq[0] = np.ones(n_points)
        b_eq = np.array([n_points / 2.0])
        
        A_ub = []
        b_ub = []
        
        # Generate inequality constraints
        for k in range(n_points):
            row = np.zeros(n_points)
            for i in range(n_points):
                j = (i + k) % n_points
                row[i] = dx * (1 - h[j])  # d_i = dx * (1 - h_j) where j = i+k
            A_ub.append(row)
            b_ub.append(C_mid)
            
        # Add bounds for h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Solve LP
        result = linprog(
            c=np.zeros(n_points),
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method='highs'
        )
        
        if result.success:
            # Found a feasible solution
            h_feasible = result.x
            # Compute actual C5 for this h
            correlation = np.correlate(h_feasible, 1 - h_feasible, mode='full')
            C5 = np.max(correlation) * dx
            
            if C5 < best_C5:
                best_C5 = C5
                best_h = h_feasible.copy()
                
            C_high = C_mid
        else:
            C_low = C_mid
            
    # Make sure final h is constrained properly
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Final calculation of C5
    correlation = np.correlate(h_final, 1 - h_final, mode='full')
    C5_final = np.max(correlation) * dx
    
    return (h_final.tolist(), C5_final, n_points)
