import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h."""
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.min(h), np.max(h) - 1
    
    # Wrapper function to pass constraints
    def obj_func(h):
        # Ensure feasible
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_h(h, dx)
    
    # Population-based DE approach
    start_time = time.time()
    popsize = 50
    maxiter = 500
    
    # Run with time limit
    # We use a simple approach that allows for reasonable time usage
    
    # For a more direct DE approach, we might improve via gradient-free optimization
    
    # Instead of complex DE, we use scipy's differential_evolution
    # which is robust to non-smooth objectives
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution with bounds
    # We'll tighter control on iterations, using time tracking
    # The problem is constrained: sum(h) = n_points / 2, and 0 <= h <= 1
    
    # Set up the inequality constraint
    def constraint_sum_eq(h):
        return np.sum(h) - n_points / 2  # This should be close to zero
    
    # Convert to scipy constraints format
    constraints = [{'type': 'eq', 'fun': constraint_sum_eq}]
    
    # Implement fixed iterations with early stopping
    
    generation = 0
    last_improvement = 0
    patience = 50  # number of generations without improvement before early stop
    
    # We loop manually by using differential evolution's callback
    # But do so with limited time and fixed generations
    
    # Due to time complexity concerns, we do not model full DE here, 
    # instead directly optimizing with the best tools
    
    try:
        # Simple attempt: run differential evolution
        result = differential_evolution(
            obj_func,
            bounds,
            constraints=constraints,
            seed=seed,
            maxiter=100,  # keep it manageable
            popsize=20,   # smaller pop to save time
            alpha=1,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.7,
            updating='immediate'
        )
        
        if not result.success:
            # fallback if DE fails
            h_final = h
        else:
            h_final = np.clip(result.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
        
        final_c5 = evaluate_h(h_final, dx)
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final
    
    except Exception as e:
        # if optimization fails, use starting point
        pass
        
    # If not improved enough, do local search near best solution
    if time.time() - start_time < budget_s - 10:
        # Gradient-based or coordinate descent approach
        best_h = best_h.copy()

        # Simple forward/backward gradient approximation
        for iter_count in range(500):

            current_c5 = evaluate_h(best_h, dx)
            
            # Compute approximate gradient
            grad = np.zeros_like(best_h)

                h_plus = best_h.copy()

                grad[i] = (c5_plus - current_c5) / eps
            
            # Update step
            best_h = best_h - learning_rate * grad
            
            # Project to feasible set

            # Check for improvement
            new_c5 = evaluate_h(best_h, dx)

                best_c5 = new_c5
            else:
                # decrease step size
                learning_rate *= 0.97
            
    return (best_h.tolist(), best_c5, n_points)
