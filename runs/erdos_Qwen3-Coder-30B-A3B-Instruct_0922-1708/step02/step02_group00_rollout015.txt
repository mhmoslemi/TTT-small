import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
        for _ in range(population_size - 1):
            h_mut = h + np.random.normal(0, 0.05, n_points)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            population.append(h_mut)
    else:
        # Generate random initial population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < 1000:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50%
        next_generation = population[:population_size // 2]
        
        # Generate new individuals
        while len(next_generation) < population_size:
            # Tournament selection
            a = population[np.random.randint(0, population_size // 2)]
            b = population[np.random.randint(0, population_size // 2)]
            
            # Crossover with 70% probability
            if np.random.rand() < 0.7:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([a[:crossover_point], b[crossover_point:]])
                child2 = np.concatenate([b[:crossover_point], a[crossover_point:]])
                
                # Mutate children
                for child in [child1, child2]:
                    if np.random.rand() < 0.15:  # 15% mutation rate
                        num_mutations = np.random.randint(1, n_points // 5)
                        indices = np.random.choice(n_points, num_mutations, replace=False)
                        for i in indices:
                            child[i] += np.random.normal(0, 0.1)
                        child = np.clip(child, 0, 1)
                        child = child * (n_points / 2) / np.sum(child)
                    next_generation.append(child)
            else:
                # Just mutate one of the parents
                parent = a if np.random.rand() < 0.5 else b
                child = parent.copy()
                if np.random.rand() < 0.15:
                    num_mutations = np.random.randint(1, n_points // 5)
                    indices = np.random.choice(n_points, num_mutations, replace=False)
                    for i in indices:
                        child[i] += np.random.normal(0, 0.1)
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                next_generation.append(child)
        
        # Keep population size
        population = next_generation[:population_size]
    
    return (best_h.tolist(), best_c5, n_points)
