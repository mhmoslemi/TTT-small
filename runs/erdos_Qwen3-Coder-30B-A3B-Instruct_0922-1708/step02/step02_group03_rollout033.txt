import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    learning_rate = 0.01
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    # Gradient-based optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the shift with maximum correlation
        max_shift = np.argmax(correlation)
        # Shift should be between 0 and 2*n_points-1
        shift_idx = max_shift - (n_points - 1)  # Convert to actual shift
        
        # Compute exact gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Compute derivative of max correlation w.r.t h[i]
            # For each shift k, d/dh[i] h[i] * (1 - h[i+k]) = 1 - h[i+k] if i+k < n_points, else 0
            # But we have shifting, and we want to differentiate the maximum over shifts
            # For simplicity: we assume that the maximum occurs for some fixed shift
            # and compute the derivative (linear approximation)
            k = shift_idx
            if 0 <= i + k < n_points:
                grad[i] = -dx * (1 - h[i + k])
            elif 0 <= i + k - n_points < n_points:  # Wrap around
                grad[i] = -dx * (1 - h[i + k - n_points])
            else:
                grad[i] = 0
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        # Bias correction
        m_corr = m / (1 - beta1**(iteration))
        v_corr = v / (1 - beta2**(iteration))
        
        # Update h
        h_new = h - learning_rate * m_corr / (np.sqrt(v_corr) + eps)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        if iteration % 50 == 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
