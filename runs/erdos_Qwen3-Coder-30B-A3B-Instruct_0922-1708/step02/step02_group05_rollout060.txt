import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure initial sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Convex relaxation for initialization (first 60% of time)
    start_time = time.time()
    remaining_time = budget_s * 0.6
    
    # Formulating a simpler convex relaxation
    h_relaxed = cp.Variable(n_points)
    objective = cp.Minimize(cp.sum_squares(h_relaxed - h))
    constraints = [
        h_relaxed >= 0,
        h_relaxed <= 1,
        cp.sum(h_relaxed) == n_points / 2
    ]
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = np.array(h_relaxed.value).reshape(-1)
    except:
        pass
    
    # Gradient descent refinement (remaining 40%)
    remaining_time = budget_s * 0.4
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    iteration = 0
    while time.time() - start_time < remaining_time - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate based on gradient magnitude
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Try to avoid derivative issues due to boundary effects
        if iteration % 10 == 0:
            # Occasionally enable an alternative update to escape local minima
            h_new = h_new * 0.95 + 0.05 * best_h
            
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
