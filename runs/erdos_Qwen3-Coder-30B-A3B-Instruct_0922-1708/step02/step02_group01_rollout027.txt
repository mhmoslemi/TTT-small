import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    n_particles = 50
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Particle positions and velocities
    positions = np.random.uniform(0, 1, (n_particles, n_points))
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    # Initialize personal bests
    personal_bests = positions.copy()
    personal_best_scores = np.full(n_particles, float('inf'))
    
    # Global best
    global_best = h_init.copy()
    global_best_score = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    inertia = w
    
    start_time = time.time()
    iteration = 0
    stagnation_counter = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = positions[i]
            # Ensure constraints
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            positions[i] = h
            
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_bests[i] = h.copy()
                
            if c5 < global_best_score:
                global_best_score = c5
                global_best = h.copy()
                stagnation_counter = 0
        
        # Update velocities and positions
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # PSO velocity update
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (personal_bests[i] - positions[i]) + 
                           c2 * r2 * (global_best - positions[i]))
            
            # Position update
            positions[i] += velocities[i]
            
            # Project onto feasible region
            h = positions[i]
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            positions[i] = h * scale
            
        # Restart stagnant particles
        if iteration % 100 == 0:
            stagnation_counter += 1
            if stagnation_counter > 50:
                for i in range(n_particles):
                    if np.random.rand() < 0.1:  # 10% chance
                        positions[i] = np.random.uniform(0, 1, n_points)
                        velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                        
        # Early termination
        if global_best_score < 0.3807:
            break
            
    return (global_best.tolist(), global_best_score, n_points)
