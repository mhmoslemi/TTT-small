import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with parent solution or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:

            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Multi-start optimization with small variations
    num_starts = 5
    for start in range(num_starts):
        # Initialize from parent or random
        if start == 0 and 'initial_h_values' in globals():
            h_current = h.copy()
        else:
            # Add noise around parent or uniform
            noise_level = 0.05
            if 'initial_h_values' in globals():
                h_current = h + np.random.normal(0, noise_level, n_points) * h
            else:
                h_current = np.ones(n_points) * 0.5 + np.random.normal(0, noise_level, n_points) * 0.5
            h_current = np.clip(h_current, 0, 1)
            
        # Rescale to sum to n_points/2
        h_current = clip_and_scale(h_current, n_points)
        
        # Simple gradient descent with momentum
        for iteration in range(500):
            if time.time() - start_time > budget_s - 1:

            c5 = compute_c5(h_current, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_current.copy()
                
            # Finite difference gradient
            grad = np.zeros_like(h_current)

            # Update with small learning rate
            lr = 0.01
            h_new = h_current - lr * grad
            h_current = clip_and_scale(h_new, n_points)
            
    return (best_h.tolist(), best_c5, n_points)
