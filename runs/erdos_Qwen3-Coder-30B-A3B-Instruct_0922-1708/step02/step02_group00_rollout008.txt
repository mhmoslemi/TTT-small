import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        # Create a reasonable starting point
        h = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.001
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < 1000000:
        iteration += 1
        
        # Compute the full correlation to find critical shift
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient for the critical shift only
        grad = np.zeros_like(h)
        if k_max < n_points:
            # Contribution from h_i * (1 - h_{i+k})
            for i in range(n_points):
                j = i + k_max
                if j < n_points:
                    grad[i] += 1 - h[j]
        else:
            # Contribution from (1 - h_i) * h_{i-k}
            k_shift = k_max - n_points + 1
            for i in range(n_points):
                j = i - k_shift
                if j >= 0:
                    grad[i] -= h[j]
        
        # Apply Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale to maintain constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        c5 = max_val * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Periodic perturbation to escape local optima
        if iteration % 500 == 0:
            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
