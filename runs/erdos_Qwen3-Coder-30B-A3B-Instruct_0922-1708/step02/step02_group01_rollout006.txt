import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Multi-start with 10 particles
    particles = []
    for i in range(10):
        if i == 0 and 'initial_h_values' in globals():
            # Use initial construction
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_new = np.linspace(0, 2, n_points)
                h_init = np.interp(x_new, x_orig, h_init)
            h_init = h_init / np.sum(h_init) * (n_points / 2)
        else:
            # Random initialization
            h_init = np.random.uniform(0, 1, n_points)
            h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init.copy())
    
    # Learning rate scheduling
    lrs = [0.01, 0.005, 0.002]
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update each particle
        for i, particle in enumerate(particles):
            # Compute C5
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
            
            # Compute gradient via finite differences
            eps = 1e-4
            grad = np.zeros_like(particle)
            for j in range(n_points):
                h_plus = particle.copy()
                h_plus[j] += eps
                # Ensure bounds and sum constraint
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - c5) / eps
            
            # Update step with learning rate and momentum
            lr = lrs[min(iteration // 500, len(lrs) - 1)]
            momentum = 0.9
            if iteration == 1:
                velocity = -lr * grad
            else:
                velocity = momentum * velocity - lr * grad
                
            particle_new = particle + velocity
            
            # Apply constraints
            particle_new = np.clip(particle_new, 0, 1)
            particle_new = particle_new * (n_points / 2) / np.sum(particle_new)
            
            # Periodic perturbation
            if np.random.rand() < 0.05 or iteration % 1000 == 0:
                noise = np.random.normal(0, 0.02, n_points)
                particle_new += noise * (n_points / 2) / np.sum(particle_new)
                particle_new = np.clip(particle_new, 0, 1)
                particle_new = particle_new * (n_points / 2) / np.sum(particle_new)
            
            particles[i] = particle_new
    
    return (best_h.tolist(), best_c5, n_points)
