import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral C5 for given h"""
    correlation = correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def repair_h(h, n_points):
    """Ensure h is feasible: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    return h

def initialize_population(n_points, n_pop, initial_h):
    """Create initial population"""
    population = []
    
    if initial_h is not None:
        h = np.array(initial_h)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Fill the rest with random individuals
    for _ in range(n_pop - len(population)):
        h = np.random.random(n_points)
        h = repair_h(h, n_points)
        population.append(h)
        
    return population

def selection(population, fitness, tournament_size=2):
    """Tournament selection"""
    selected = []
    n = len(population)
    
    for _ in range(n):
        # Tournament selection
        indices = np.random.choice(n, tournament_size)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        selected.append(population[best_idx].copy())
        
    return selected

def crossover(parent1, parent2, crossover_rate=0.8):
    """Single-point crossover"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = np.random.randint(1, n)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_magnitude=0.1):
    """Mutate with noise and repair constraints"""
    if np.random.random() > mutation_rate:
        return h.copy()
    
    h_new = h.copy()
    mask = np.random.random(len(h)) < mutation_rate
    
    noise = np.random.uniform(-noise_magnitude, noise_magnitude, len(h))
    h_new[mask] += noise[mask]
    
    return repair_h(h_new, len(h))

def genetic_algorithm(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    n_pop = 50
    crossover_rate = 0.8
    mutation_rate = 0.2
    disturbance_rate = 0.2
    
    # Get initial construction if available
    initial_h = globals().get('initial_h_values', None)
    
    # Initialize population
    population = initialize_population(n_points, n_pop, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
                
        no_improvement_count += 1
        
        # Early termination
        if best_c5 < 0.3807:
            break
            
        # Select
        selected = selection(population, fitness)
        
        # Create new population through crossover and mutation
        new_population = []
        
        for i in range(0, n_pop, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % n_pop]
            
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
            
        # Elitism: keep best from previous generation
        elite_indices = np.argsort(fitness)[:n_pop//2]
        elite = [population[i] for i in elite_indices]
        new_population[:n_pop//2] = elite
        
        # Disturbance to avoid stagnation
        if no_improvement_count > max_no_improvement:
            for i in range(int(n_pop * disturbance_rate)):
                idx = np.random.randint(0, n_pop)
                h = np.random.random(n_points)
                h = repair_h(h, n_points)
                new_population[idx] = h
            no_improvement_count = 0
            
        population = new_population
        
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s, **kwargs)
