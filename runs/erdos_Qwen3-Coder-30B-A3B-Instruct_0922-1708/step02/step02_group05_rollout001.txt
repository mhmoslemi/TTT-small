import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h, n_points):
    """Evaluate the C5 bound for a given h."""
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

        return np.sum(x) * dx - 1.0

    def objective(x):
        # Renormalize to satisfy integral constraint
        x = x * (n_points / 2) / np.sum(x)
        return evaluate_h(x, n_points)

    start_time = time.time()

    population_size = 50

            mutant = mutant * (n_points / 2) / np.sum(mutant)

            CR = 0.5

            trial = np.where(crossover_mask, mutant, pop[i])
            
            # Evaluate
