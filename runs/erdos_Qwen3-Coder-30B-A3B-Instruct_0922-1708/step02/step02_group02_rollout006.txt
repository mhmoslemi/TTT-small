from scipy.optimize import minimize
import time

        # Construct an initial binary comb pattern
        h = np.zeros(n_points)

        pulse_width = 1.0 / m
        spacing = 1.0 / (m - 1)
        total_width = m * pulse_width + (m - 1) * spacing
        
        # Normalize to integral = 1
        scale_factor = 1.0 / total_width
        pulse_width *= scale_factor
        spacing *= scale_factor
        
        # Create binary pulses
        for i in range(m):
            start = i * (pulse_width + spacing)
            end = start + pulse_width
            start_idx = int(start / 2.0 * n_points)
            end_idx = int(end / 2.0 * n_points)
            if end_idx > start_idx:
                h[start_idx:end_idx] = 1.0
                
        # Adjust to ensure integral = 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        
    best_h = h.copy()
    best_c5 = float('inf')

        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Use a simple local search with perturbations
        h_new = h.copy()

    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
