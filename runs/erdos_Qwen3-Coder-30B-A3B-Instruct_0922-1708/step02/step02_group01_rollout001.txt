import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    n_particles = 50
    dx = 2.0 / n_points
    max_iter = int(500000)
    
    # Initialize particles
    particles = np.random.rand(n_particles, n_points)
    velocities = np.random.randn(n_particles, n_points) * 0.01
    
    # Scale to correct sum if initial_h_values exists
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            x_orig = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, n_points)
            h0 = np.interp(x_new, x_orig, h0)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Initialize particles with h0 as one of them
    particles[0] = h0
    for i in range(1, n_particles):
        particles[i] = np.random.rand(n_points)
    
    # Normalize all particles
    for i in range(n_particles):
        scale = (n_points / 2) / np.sum(particles[i])
        particles[i] = np.clip(particles[i] * scale, 0, 1)
    
    # Initialize personal best
    personal_best = particles.copy()
    personal_best_values = np.full(n_particles, np.inf)
    
    # Initialize global best
    global_best_idx = 0
    global_best_value = np.inf
    
    start_time = time.time()
    last_improvement = 0
    iteration = 0
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < personal_best_values[i]:
                personal_best_values[i] = c5
                personal_best[i] = h.copy()
                
            if c5 < global_best_value:
                global_best_value = c5
                global_best_idx = i
                last_improvement = iteration
                
        # Update swarm
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocity = (w * velocities[i] + 
                        c1 * r1 * (personal_best[i] - particles[i]) + 
                        c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            
            particles[i] += velocity
            
            # Project onto feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            scale = (n_points / 2) / np.sum(particles[i])
            particles[i] = particles[i] * scale
            
            velocities[i] = velocity  # Update velocity for next iteration
    
        # Restart mechanism if no improvement for a while
        if iteration - last_improvement > 100:
            velocities = np.random.randn(n_particles, n_points) * 0.01
    
    # Return best solution
    best_h = personal_best[global_best_idx]
    return (best_h.tolist(), global_best_value, n_points)
