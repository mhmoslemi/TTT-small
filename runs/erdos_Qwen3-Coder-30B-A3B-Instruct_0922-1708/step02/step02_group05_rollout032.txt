import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Convex relaxation to get a good initialization
    start_time = time.time()
    
    # Set up convex optimization problem
    h_vars = cp.Variable(n_points)
    tau = cp.Variable()
    
    # Constraints
    constraints = []
    constraints.append(cp.sum(h_vars) == n_points / 2)
    constraints.append(h_vars >= 0)
    constraints.append(h_vars <= 1)
    
    # Add constraint for each shift k
    shift_range = range(n_points)
    for k in shift_range:
        # Compute the correlation: sum of h[i] * (1 - h[i + k])
        # Handle boundary by wrapping around
        corr_expr = 0
        for i in range(n_points):
            j = (i + k) % n_points
            corr_expr += h_vars[i] * (1 - h_vars[j])
        constraints.append(corr_expr <= tau * n_points)
    
    # Objective
    objective = cp.Minimize(tau)
    
    # Solve the convex problem
    prob = cp.Problem(objective, constraints)
    
    # Use ECOS solver for convex problem
    try:
        prob.solve(solver=cp.ECOS, time_limit=30)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_init = np.array(h_vars.value).reshape(-1)
            # Clip and rescale to enforce bounds
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            h = h_init.copy()
    except:
        pass  # If convex optimization fails, continue with original h
    
    # Refinement with gradient-based optimization
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization using gradient descent with adaptive steps
    while time.time() - start_time < budget_s - 1:
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate based on gradient magnitude
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
