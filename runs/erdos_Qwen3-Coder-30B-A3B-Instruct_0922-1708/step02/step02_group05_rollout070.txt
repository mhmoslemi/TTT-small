import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    swarm_size = 50
    max_iter = 500
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles and velocities
    particles = np.random.rand(swarm_size, n_points) * 0.5
    for i in range(swarm_size):
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    velocities = (np.random.rand(swarm_size, n_points) - 0.5) * 0.1
    pbest_positions = particles.copy()
    pbest_scores = np.full(swarm_size, float('inf'))
    
    # Global best tracking
    gbest_position = None
    gbest_score = float('inf')
    
    # PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        # Evaluate current swarm
        for i in range(swarm_size):
            particle = particles[i]
            
            # Compute correlation and C5
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_scores[i]:
                pbest_scores[i] = c5
                pbest_positions[i] = particle.copy()
                
            if c5 < gbest_score:
                gbest_score = c5
                gbest_position = particle.copy()
                
            # Update bests
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (gbest_position - particles[i]))
            
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
    return (best_h.tolist(), best_c5, n_points)
