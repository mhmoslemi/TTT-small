import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive step size based on progress
        if iteration < 100:
            step_size = 0.02
        else:
            step_size = 0.01
        
        # Random perturbations with adaptive step size
        perturb = np.random.normal(0, step_size, n_points)
        h_new = h + perturb
        
        # Apply neighborhood flip strategy for better exploration
        if np.random.rand() < 0.05 and iteration > 20:
            # Flip a small neighborhood of elements
            i = np.random.randint(n_points)
            neighborhood_size = max(1, n_points // 20)
            start_idx = max(0, i - neighborhood_size // 2)
            end_idx = min(n_points, i + neighborhood_size // 2)
            h_new[start_idx:end_idx] = 1 - h_new[start_idx:end_idx]
        
        h_new = np.clip(h_new, 0, 1)
        # Rescale and normalize to make integral = 1
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Occasionally perform global perturbation
        if np.random.rand() < 0.02 and iteration > 50:
            # Replace quarter of elements with random values
            num_replace = n_points // 4
            indices = np.random.choice(n_points, num_replace, replace=False)
            h_new[indices] = np.random.rand(num_replace)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
        
        h = h_new
    
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
