import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a binary comb structure
        h = np.zeros(n_points)
        m = 5  # number of pulses
        w = 0.2  # width of each pulse
        s = 0.25  # spacing between pulses
        total_width = m * w + (m - 1) * s
        scale_factor = 1.0 / total_width
        
        for i in range(m):
            start = i * (w + s)
            end = start + w
            start_idx = int(start / 2.0 * n_points)
            end_idx = int(end / 2.0 * n_points)
            if start_idx < n_points:
                h[start_idx:end_idx] = 1.0
                
        h = h * scale_factor * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Simple gradient-based optimization with adaptive learning rate
    iteration = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        h_eval = h.copy()
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update h using gradient descent
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 10 == 0:
            lr *= 0.95
            
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
