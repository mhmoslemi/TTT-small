import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Generate diverse initializations
    initial_hs = []
    
    # Use parent if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_new = np.linspace(0, 2, n_points)
            h_base = np.interp(x_new, x_orig, h_base)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        initial_hs.append(h_base.copy())
        
        # Add noisy variations
        for _ in range(4):
            h_noisy = h_base + np.random.normal(0, 0.02, n_points)
            h_noisy = clip_and_scale(h_noisy, n_points)
            initial_hs.append(h_noisy.copy())
    else:
        # Create random sparse patterns
        for _ in range(5):
            h = np.random.rand(n_points) * 0.5
            h = clip_and_scale(h, n_points)
            initial_hs.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Simulated Annealing style with multiple restarts
    max_restarts = 10
    restart_count = 0
    
    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        h = initial_hs[restart_count % len(initial_hs)].copy()
        
        # Adam-like optimizer with adaptive learning
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8
        lr = 0.001
        
        inner_iterations = 0
        while time.time() - start_time < budget_s - 1 and inner_iterations < 500:
            inner_iterations += 1
            
            # Compute current C5
            c5 = compute_c5(h, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Finite difference gradient approximation
            grad = np.zeros(n_points)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = compute_c5(h_plus, dx)
                grad[i] = (c5_plus - c5) / eps
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**(inner_iterations + 1))
            v_hat = v / (1 - beta2**(inner_iterations + 1))
            
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
            
            # Ensure constraints
            h = clip_and_scale(h_new, n_points)
            
        restart_count += 1
        
    if best_h is None:
        # Fallback to a simple solution if nothing worked
        h = np.ones(n_points) * 0.5
        h = clip_and_scale(h, n_points)
        best_c5 = compute_c5(h, dx)
        best_h = h
    
    return (best_h.tolist(), best_c5, n_points)
