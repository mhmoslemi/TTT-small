import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Multi-start with multiple particles
    n_particles = 10
    particles = []
    for i in range(n_particles):
        if i == 0 and 'initial_h_values' in globals():
            # Use the initial construction for first particle
            h0 = np.array(initial_h_values)
            if len(h0) != n_points:
                x_orig = np.linspace(0, 2, len(h0))
                x_new = np.linspace(0, 2, n_points)
                h0 = np.interp(x_new, x_orig, h0)
            h0 = h0 / np.sum(h0) * (n_points / 2)
        else:
            # Random initialization
            h0 = np.random.uniform(0, 1, n_points)
            h0 = h0 / np.sum(h0) * (n_points / 2)
        particles.append(h0)
    
    # Optimization parameters
    lr_base = 0.01
    momentum = 0.9
    noise_prob = 0.05
    noise_scale = 0.05
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update all particles
        for p_idx, h in enumerate(particles):
            # Compute current C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update with momentum
            lr = lr_base / (1.0 + 0.0001 * iteration)
            if 'velocity' not in locals() or p_idx == 0:
                velocity = -lr * grad
            else:
                velocity = momentum * velocity - lr * grad
                
            h_new = h + velocity
            
            # Add noise occasionally
            if np.random.random() < noise_prob:
                noise = np.random.uniform(-noise_scale, noise_scale, n_points)
                h_new += noise
                
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[p_idx] = h_new
        
        # Occasionally perturb particles
        if iteration % 50 == 0:
            for p_idx in range(n_particles):
                if np.random.random() < 0.3:
                    # Add some noise
                    noise = np.random.uniform(-0.1, 0.1, n_points)
                    particles[p_idx] += noise
                    particles[p_idx] = np.clip(particles[p_idx], 0, 1)
                    particles[p_idx] = particles[p_idx] * (n_points / 2) / np.sum(particles[p_idx])
    
    return (best_h.tolist(), best_c5, n_points)
