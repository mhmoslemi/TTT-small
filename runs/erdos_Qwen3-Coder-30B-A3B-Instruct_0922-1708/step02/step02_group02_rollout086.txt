from scipy.optimize import minimize
import time
import random

    random.seed(seed)

    dx = 2.0 / n_points

    population_size = 50
    population = []

            h_base = np.interp(x_target, x_orig, h_base)
        h_base = h_base / np.sum(h_base) * (n_points / 2)
    else:
        h_base = np.ones(n_points) * 0.5
        
    # Create initial population
    for i in range(population_size):
        if i == 0:
            h = h_base.copy()
        else:
            # Add random noise to base solution
            noise = np.random.normal(0, 0.05, n_points)
            h = h_base + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]

        elite_size = population_size // 5
        elite = population[:elite_size]

        new_population = elite.copy()

        while len(new_population) < population_size:

            mutation_rate = 0.2

                mutation_rate = 0.3

                noise = np.random.normal(0, 0.02)
                child[i] += noise

    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (h_opt.tolist(), final_c5, n_points)
