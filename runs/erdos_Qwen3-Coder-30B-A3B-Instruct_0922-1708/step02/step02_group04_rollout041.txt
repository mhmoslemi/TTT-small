        mid = n_points // 2

        h_init[mid:] = 0.25

    h = h_init.copy()
    
    # Normalize to integral = 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    last_improvement = 0
    iteration = 0
    
    # Use a more systematic optimization approach
    while time.time() - start_time < budget_s - 1 and iteration - last_improvement < 200:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Create perturbation
        perturb = np.random.normal(0, 0.02, n_points)
        h_new = h + perturb
        
        # Apply clipping and rescaling
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)

            trial = h + step * perturb

            trial = trial * (n_points / 2) / np.sum(trial)
            
            corr = np.correlate(trial, 1 - trial, mode='full')

                best_c5_trial = c5_trial
