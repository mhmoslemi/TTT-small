import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    swarm_size = 50
    w_start = 0.9
    w_end = 0.4
    c1 = c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            if len(particle) != n_points:
                particle = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(particle)), particle)
            particle = np.clip(particle, 0, 1)
        else:
            particle = np.random.rand(n_points)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        corr = np.correlate(particle, 1 - particle, mode='full')
        c5 = np.max(corr) * dx
        pbest.append(particle.copy())
        pbest_c5.append(c5)
    
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # PSO loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        w = w_start - (w_start - w_end) * iteration / 500
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (gbest - particles[i])
            )
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
            # Update global best
            if c5 < gbest_c5:
                gbest = particles[i].copy()
                gbest_c5 = c5
                
        # Check if we've reached the target
        if gbest_c5 <= 0.3807:
            break
    
    # Final evaluation
    if gbest_c5 < best_c5:
        best_c5 = gbest_c5
        best_h = gbest.copy()
        
    return (best_h.tolist(), best_c5, n_points)
