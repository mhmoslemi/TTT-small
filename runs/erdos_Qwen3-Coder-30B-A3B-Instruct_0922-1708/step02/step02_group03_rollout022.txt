import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Hyperparameters
    lr_initial = 0.001
    lr_decay = 0.999
    max_iter = 10000
    n_restarts = 5
    
    for restart in range(n_restarts):
        h = best_h.copy() if restart == 0 else np.random.rand(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
        lr = lr_initial
        
        last_improvement = 0
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(correlation)
            c5 = correlation[max_idx] * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = iteration
                
            # Compute exact gradient with respect to the active lag
            grad = np.zeros_like(h)
            active_lag = max_idx - (len(h) - 1)
            
            # Approximate gradient for all positions
            eps = 1e-6
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = corr_plus[max_idx] * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update h
            h_new = h - lr * grad
            
            # Clip and rescale to maintain constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
            # Adaptive learning rate
            lr *= lr_decay
            
            # Early stopping
            if iteration - last_improvement > 500:
                break
                
    return (best_h.tolist(), best_c5, n_points)
