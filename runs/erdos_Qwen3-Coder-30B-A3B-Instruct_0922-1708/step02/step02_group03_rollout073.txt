import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    learning_rate = 0.001
    lr_decay = 0.999
    eps = 1e-6
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient more carefully - get the index of maximum correlation
        max_idx = np.argmax(correlation)
        max_k = max_idx - (len(h) - 1)  # lag
        
        # Compute exact gradient for the active lag
        grad = np.zeros_like(h)
        for i in range(n_points):
            # For lag k, derivative w.r.t. h[i]
            # h[i] contributes to correlation at positions 
            # i-k (or i+k if we consider circular)
            if (i - max_k) % n_points < n_points and (i - max_k) >= 0:
                grad[i] += (1 - h[(i - max_k) % n_points])
            if (i + max_k) % n_points < n_points and (i + max_k) >= 0:
                grad[i] -= h[(i + max_k) % n_points]
        
        # Update h
        h_new = h - learning_rate * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Adaptive learning rate
        learning_rate *= lr_decay
        
        # Occasionally check for early termination
        if iteration % 100 == 0 and iteration > 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    return (best_h.tolist(), final_c5, n_points)
