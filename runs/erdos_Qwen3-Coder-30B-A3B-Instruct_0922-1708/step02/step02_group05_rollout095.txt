import numpy as np
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    swarm_size = 50
    w_start = 0.7
    w_end = 0.4
    c1 = c2 = 2.0
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            pos = h.copy()
            pos += np.random.normal(0, 0.01, n_points)
        else:
            pos = np.random.rand(n_points) * 0.5
        pos = np.clip(pos, 0, 1)
        pos = pos * (n_points / 2) / np.sum(pos)
        particles.append(pos.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        pbest_positions.append(pos.copy())
        pbest_values.append(evaluate_c5(pos, dx))
    
    gbest_position = pbest_positions[np.argmin(pbest_values)]
    gbest_value = min(pbest_values)
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iteration / 500)
        
        # PSO update
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (gbest_position - particles[i]))
            
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_value = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest_positions[i] = particles[i].copy()
            
            # Update global best
            if current_value < gbest_value:
                gbest_value = current_value
                gbest_position = particles[i].copy()
    
    return (gbest_position.tolist(), gbest_value, n_points)
