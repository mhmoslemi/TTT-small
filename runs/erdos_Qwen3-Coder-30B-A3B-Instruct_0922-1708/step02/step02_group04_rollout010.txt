        n_points = finest_grid
        grid_sizes = [20, 40, 80, 160, 200]

    if n_points not in grid_sizes:

            # Interpolate to finer grid
            old_h = h.copy()

        for _ in range(50):

            correlation = np.correlate(h, 1 - h, mode='full')

            if c5 < best_c5:
                best_c5 = c5

            # Compute approximate gradient for each h[i]
            grad = np.zeros_like(h)

            for k in range(len(h)):

                if k < len(h):
                    # h[i] contributes to overlap(k) term if (i+k) exists in valid range

                        grad[i] += (1 - h[j])  # derivative of h[i] * (1 - h[j]) wrt h[i]

            alpha = 0.01
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)

    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
