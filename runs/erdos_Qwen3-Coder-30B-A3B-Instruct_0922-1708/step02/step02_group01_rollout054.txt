import numpy as np
import time

def project_h(h, n_points):
    """Project h to feasible region: clip to [0,1] and rescale to sum to n_points/2"""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    return h

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral C5"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution for better accuracy
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.random.rand(n_particles, n_points) * 0.5
    velocities = np.zeros((n_particles, n_points))
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        particles[0] = h_init / np.sum(h_init) * (n_points / 2)
    
    # Initialize personal bests
    personal_bests = particles.copy()
    personal_best_values = np.full(n_particles, np.inf)
    
    # Initialize global best
    global_best = particles[0].copy()
    global_best_value = np.inf
    
    # PSO parameters
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    best_found = np.inf
    best_h = None
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            c5 = evaluate_c5(h, dx)
            
            if c5 < personal_best_values[i]:
                personal_best_values[i] = c5
                personal_bests[i] = h.copy()
            
            if c5 < global_best_value:
                global_best_value = c5
                global_best = h.copy()
                best_found = c5
                
                if best_found < 0.3807:
                    # Early exit if target already achieved
                    break
        
        # Update global best tracking
        if global_best_value < best_found:
            best_found = global_best_value
            best_h = global_best.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_bests[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_h(particles[i], n_points)
        
        # Restart velocities if stagnation detected
        if iteration % 100 == 0:
            stagnation_count += 1
            if stagnation_count > 5:
                velocities = np.random.rand(n_particles, n_points) * 0.01
                stagnation_count = 0
        
        if best_found < 0.3807:
            break
    
    # Final evaluation of global best
    if global_best_value < best_found:
        best_found = global_best_value
        best_h = global_best.copy()
    
    return (best_h.tolist(), best_found, n_points)
