import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    t = 0
    T0 = 0.1
    tau = 1000
    
    while time.time() - start_time < budget_s - 1:
        # Update temperature
        T = T0 / (1 + t / tau)
        
        # Generate candidate solution
        h_prime = h + np.random.normal(0, T, n_points)
        h_prime = np.clip(h_prime, 0, 1)
        h_prime = h_prime * (n_points / 2) / np.sum(h_prime)
        
        # Compute overlap integrals
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlation) * dx
        
        correlation_prime = np.correlate(h_prime, 1 - h_prime, mode='full')
        c5_prime = np.max(correlation_prime) * dx
        
        # Accept or reject based on simulated annealing
        delta_c5 = c5_prime - c5_current
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_prime
            c5_current = c5_prime
            
        # Update best solution
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        t += 1
        
    return (best_h.tolist(), best_c5, n_points)
