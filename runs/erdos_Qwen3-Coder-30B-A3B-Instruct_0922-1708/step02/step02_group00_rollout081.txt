import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from the parent construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.random.rand(n_points) * 0.5
    
    # Scale to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Adam optimizer parameters
    alpha = 0.001
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    # Optimization loop
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute full correlation to find k that maximizes overlap
        correlation = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(correlation)
        # The maximum overlap for shift k_max
        c5 = correlation[k_max] * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient with respect to the max shift
        # correlation has length 2*n_points - 1, shift k is at index k
        # We compute gradient for the max shift k_max
        grad = np.zeros_like(h)
        
        # For each h[i], we need to compute its influence on the correlation at k_max
        # The derivative of correlation[k_max] w.r.t. h[i] depends on the positions:
        # From h[i] * (1 - h[i+k_max]) if this is valid
        # From h[i-k_max] * (-h[i]) if i-k_max >= 0
        
        # The shear operation to compute the partial derivative
        for i in range(n_points):
            # Contribution from h[i] to the correlation term when shifted k_max
            if i + k_max < 2 * n_points - 1:
                # This is the contribution to h[i] * (1 - h[i + k_max])
                # or equivalently from the i-th element of h to shift k_max correlation
                if i + k_max < n_points:
                    grad[i] += (1 - h[i + k_max]) * dx
                
                # The derivative from other terms: 
                # If i-k_max >= 0 then h[i] contributes to h[i-k_max] * (-h[i])
                # But we're taking derivative with respect to h[i]
                # So we consider i+k_max vs shift
                # In case of i + k_max in valid range, contribute 1
                # The derivative of (h[i] * (1-h[i+k_max])) w.r.t h[i] is (1 - h[i+k_max])
                # But we need to take shift into account correctly for scaling
                # The Pearson-style formula correction: Only the corresponding term contributes
                # The gradient of correlation[k_max] w.r.t. h[i] = 1 - h[i + k_max] if i + k_max < n
                # and also -h[i - k_max] if i - k_max >= 0
                # Let's recompute carefully for i-th parameter
                
                if i - k_max >= 0 and i - k_max < n_points:
                    grad[i] += -h[i - k_max] * dx
        
        # Clip gradient to prevent overflow
        grad = np.clip(grad, -10, 10)
        
        # Update Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        h_new = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Periodic perturbation to avoid local minima
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
