import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Use a different approach: simulated annealing
    # Initialize parameters for the algorithm
    current_h = h.copy()
    current_c5 = float('inf')
    
    # Track best solution
    best_h = current_h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated annealing parameters
    T = 1.0
    T_min = 1e-6
    alpha = 0.995
    max_iterations = 100000
    iteration = 0
    
    def compute_c5(h_val):
        correlation = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(correlation) * dx

    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Make a small random perturbation
        h_new = current_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Calculate new C5
        c5_new = compute_c5(h_new)
        
        # Accept or reject the new solution
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
        else:
            # Accept with probability based on temperature
            delta_e = c5_new - current_c5
            if np.random.rand() < np.exp(-delta_e / T):
                current_h = h_new.copy()
                current_c5 = c5_new
                
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Cool down the temperature
        T = max(T_min, T * alpha)
        
        if iteration % 1000 == 0:
            # Occasionally check if we should terminate early
            if time.time() - start_time > budget_s - 5:
                break
                
    # Final refinement using optimization
    if time.time() - start_time < budget_s - 5:
        try:
            # Use only the best solution as starting point
            h0 = best_h.copy()
            
            # Optimization with constraints
            def objective(h_vec):
                correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
                return np.max(correlation) * dx
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2
                
            constraints = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=constraints, 
                            options={'maxiter': 500, 'disp': False})
            
            if result.success:
                refined_h = result.x
                refined_c5 = objective(refined_h)
                
                # Ensure it meets the constraints
                refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
                refined_c5 = objective(refined_h)
                
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
