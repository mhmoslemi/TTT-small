from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points

        h = np.zeros(n_points)
        h[:n_points//2] = 1.0

    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # Try convex optimization approach
    try:
        # Set up the optimization problem
        # We want to minimize C5 = max_k ∫h(x)(1−h(x+k))dx
        # This is equivalent to minimizing the maximum correlation
        
        # Using CVXPY to solve a regularized version of the problem
        h_var = cp.Variable(n_points)
        
        # Limit on the maximum value
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Create a simpler continuous approximation approach
        def objective(h_vals):
            c5 = compute_c5(h_vals)
            return c5
        
        # Try different local search starting points
        all_starting_points = []
        
        # Starting point 1: initial_h_values if available
        all_starting_points.append(h.copy())
        
        # Starting point 2: block function
        block_h = np.zeros(n_points)
        block_h[:n_points//2] = 1.0
        all_starting_points.append(block_h)

        smooth_step_h = np.zeros(n_points)

            elif x > 1.0:
                smooth_step_h[i] = 2.0 - x
            else:
                smooth_step_h[i] = 1.0
        smooth_step_h = smooth_step_h * (n_points / 2) / np.sum(smooth_step_h)

        for starting_point in all_starting_points:

            correlation = np.correlate(h, 1 - h, mode='full')

                best_h = h.copy()

            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
        # Final check
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(correlation) * dx
        
        return (best_h.tolist(), final_c5, n_points)
