import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h"""
    correlation = correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def rescale_h(h, n_points):
    """Rescale h to ensure sum(h) = n_points / 2"""
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    return np.clip(h, 0, 1)

def mutate_individual(h, mutation_rate, n_points):
    """Mutate an individual with uniform mutation"""
    h_mut = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h_mut[i] += np.random.uniform(-0.1, 0.1)
    return rescale_h(h_mut, n_points)

def crossover_parents(p1, p2, crossover_rate):
    """Single-point crossover"""
    if np.random.random() > crossover_rate:
        return p1.copy(), p2.copy()
    
    crossover_point = np.random.randint(1, len(p1)-1)
    child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
    return child1, child2

def tournament_selection(population, fitness, tournament_size=2):
    """Tournament selection"""
    idx = np.random.choice(len(population), tournament_size)
    best_idx = idx[np.argmin([fitness[i] for i in idx])]
    return population[best_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    N_pop = 50
    C_rate = 0.8
    M_rate = 0.2
    max_generations = 1000
    disturbance_rate = 0.2

    # Population initialization
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = rescale_h(h_init, n_points)
        population.append(h_init)
    
    # Fill up population with random individuals
    while len(population) < N_pop:
        h = np.random.uniform(0, 1, n_points)
        h = rescale_h(h, n_points)
        population.append(h)

    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate fitness
        fitness = [evaluate_h(ind, dx) for ind in population]
        
        # Track best solution
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_c5 = fitness[min_idx]
            best_h = population[min_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination
        if best_c5 < 0.3807:
            break
            
        # Selection: keep top 50%
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:N_pop//2]]
        
        # Create new population
        new_population = selected.copy()
        
        # Elitism: preserve best individual
        new_population.append(best_h.copy())
        
        # Crossover and mutation
        while len(new_population) < N_pop:
            p1 = tournament_selection(population, fitness)
            p2 = tournament_selection(population, fitness)
            
            c1, c2 = crossover_parents(p1, p2, C_rate)
            
            c1 = mutate_individual(c1, M_rate, n_points)
            c2 = mutate_individual(c2, M_rate, n_points)
            
            new_population.extend([c1, c2])
        
        # Trim population to exact size
        new_population = new_population[:N_pop]
        
        # Apply disturbance to prevent stagnation
        if no_improvement_count > 50:
            for i in range(int(N_pop * disturbance_rate)):
                idx = np.random.randint(0, N_pop)
                new_population[idx] = np.random.uniform(0, 1, n_points)
                new_population[idx] = rescale_h(new_population[idx], n_points)
            no_improvement_count = 0
        
        population = new_population
        generation += 1
    
    # Final validation
    final_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
