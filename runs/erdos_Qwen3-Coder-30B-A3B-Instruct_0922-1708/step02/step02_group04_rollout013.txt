    prev_h = h.copy()

    for grid_size in grid_sizes:

            break

            # Downsample
            new_h = np.zeros(grid_size)
            for i in range(grid_size):
                start_idx = int(i * n_points / grid_size)
                end_idx = int((i + 1) * n_points / grid_size)
                new_h[i] = np.mean(prev_h[start_idx:end_idx])
            h = new_h
        elif grid_size > n_points:
            # Upsample using linear interpolation
            h_interp = np.interp(np.linspace(0, 1, grid_size), 
                                 np.linspace(0, 1, n_points), h)
            h = h_interp
            n_points = grid_size
        else:
            # Same size
            pass
        
        dx = 2.0 / n_points
        h = np.clip(h, 0, 1)
        
        if len(h) != n_points:
            h = h[:n_points]
        
        # Optimize on current grid
        curr_iter = 0
        improvement_count = 0

            correlation = np.correlate(h, 1 - h, mode='full')

                improvement_count = 0

                improvement_count += 1

                alpha *= 0.5

            grad = np.zeros_like(h)

    final_c5 = np.max(correlation_final) * (2.0 / len(best_h))
