import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Adaptive parameters
    max_iter = 1000
    delta = 0.01  # Initial decrement
    no_improvement_count = 0
    convergence_threshold = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps for all shifts
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
            delta = max(1e-4, 0.01 * best_c5)
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                delta *= 0.5
                no_improvement_count = 0
            delta = max(1e-4, delta)
            
        # If we've found a good enough solution, stop early
        if best_c5 < 0.3807:
            break
            
        # Find the worst shift
        k_max = np.argmax(correlation)
        
        # Formulate linear constraint around the worst shift
        # We want: sum h[i] * (1 - h[(i + k_max) % n_points]) * dx <= C5 - delta
        # This becomes: sum h[i] - sum h[i] * h[(i + k_max) % n_points]) * dx <= C5 - delta
        # This is equivalent to: sum h[i] * h[(i + k_max) % n_points] * dx >= C5 - delta - sum h[i] * dx
        
        # Use a more direct approach - sample gradient using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Apply gradient descent but stay within bounds
        learning_rate = 0.02
        
        h_new = h - learning_rate * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Instead of strict greedy descent, try some stochastic search to avoid local minima
        if iteration % 10 == 0:  # Occasionally perturb
            trial_h = h_new.copy() + np.random.normal(0, 0.005, n_points)
            trial_h = np.clip(trial_h, 0, 1)
            trial_h = trial_h * (n_points / 2) / np.sum(trial_h)
            trial_corr = np.correlate(trial_h, 1 - trial_h, mode='full')
            trial_c5 = np.max(trial_corr) * dx
            
            if trial_c5 < c5:
                h_new = trial_h
                
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
