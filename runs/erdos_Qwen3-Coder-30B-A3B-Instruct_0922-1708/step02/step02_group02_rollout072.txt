import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Create a list of all lags
    all_lags = list(range(1, n_points))
    
    # Precompute the matrix for correlation constraints
    # We'll build it iteratively using the most problematic lags
    while time.time() - start_time < budget_s - 1:
        # Compute current correlations
        correlation = np.correlate(h, 1 - h, mode='full')
        max_correlation = np.max(correlation)
        c5 = max_correlation * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Get the lag that produces maximum correlation
        max_idx = np.argmax(correlation)
        max_lag = max_idx - (n_points - 1)  # Convert full index to lag
        
        # Now we use a QP approach to optimize
        # Variables: h and an auxiliary variable t representing the maximum overlap
        h_vars = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = []
        # Normalize the integral constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        # Bound constraints
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        # Max overlap constraint
        # For each lag k, sum h[i] * (1 - h[i+k]) * dx <= t
        # Sum over i: 0 to n_points - k - 1
        
        # We'll use a subset of crucial lags for efficiency
        # For the immediate optimization, just use a few most problematic lags
        # Since we're looking for the max, we'll use the one with highest value
        k = max_lag
        if k > 0:
            # Sum from i = 0 to n_points - k - 1 of h_vars[i] * (1 - h_vars[i+k])
            expr = 0
            for i in range(n_points - k):
                expr += h_vars[i] * (1 - h_vars[i + k])
            constraints.append(expr * dx <= t)
        else:
            # Handle the negative lag - should not happen in this context 
            # But include to prevent IndexError in basic setup
            pass
        
        # Objective
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                if h_new is not None:
                    h = h_new.copy()
                    # Clip the values to be in [0,1]
                    h = np.clip(h, 0, 1)
                    # Rescale to sum to n_points / 2
                    h = h * (n_points / 2) / np.sum(h)
        except:
            # If QP fails, do gradient descent step instead
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
                
            # Update h using gradient descent
            lr = 0.005
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
