import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
    else:
        # Create a simple binary-like pattern
        h = np.zeros(n_points)
        # Use a pattern of 5 segments with gaps
        segment_width = 0.2
        gap_width = 0.1
        start_positions = np.linspace(0, 2 - segment_width, 5)
        for pos in start_positions:
            start_idx = int(pos / dx)
            end_idx = int((pos + segment_width) / dx)
            if end_idx <= n_points:
                h[start_idx:end_idx] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective(h_vec):
        # Ensure values are valid
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to satisfy constraint
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Optimize using differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time.time()
    
    # First perform a simpler search
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        # Perturb h slightly
        idx = np.random.randint(0, n_points)
        h_pert = h.copy()
        h_pert[idx] += np.random.normal(0, 0.01)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        c5 = objective(h_pert)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pert.copy()
        # Update h to better solution
        h = h_pert.copy()
    
    # Now use a more structured optimization
    res = differential_evolution(objective, bounds, maxiter=50, popsize=15, 
                                seed=seed, disp=False, workers=1)
    if res.success:
        h_opt = res.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        c5_opt = objective(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
