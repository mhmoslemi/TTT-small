import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    last_best_c5 = float('inf')
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Stop if no improvement in recent iterations
        if stagnation_count > 100:
            break
            
        # Adaptive perturbation strategy
        # 1. Small random perturbation with probability 0.7
        # 2. Large random jump with probability 0.2
        # 3. Neighborhood flip with probability 0.1
        
        if np.random.rand() < 0.7:
            # Small perturbation
            perturb = np.random.normal(0, 0.01, n_points)
            h_new = h + perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        elif np.random.rand() < 0.9:
            # Large jump - random single element
            i = np.random.randint(n_points)
            h_new = h.copy()
            h_new[i] = np.random.rand()
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
        else:
            # Neighborhood flip - pick a random region and flip
            region_size = max(1, n_points // 20)
            start_idx = np.random.randint(0, n_points - region_size)
            end_idx = start_idx + region_size
            h_new = h.copy()
            h_new[start_idx:end_idx] = 1 - h_new[start_idx:end_idx]
            # Rescale and normalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            
        h = h_new
    
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)

    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Ensure the constraints are met
    assert abs(np.sum(best_h) * dx - 1.0) < 1e-9, "Integral constraint violated"
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Value constraints violated"
    
    return (best_h.tolist(), final_c5, n_points)
