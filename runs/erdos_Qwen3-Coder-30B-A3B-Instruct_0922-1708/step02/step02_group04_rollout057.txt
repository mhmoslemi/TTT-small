import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):

    # Use the provided construction or start with symmetric step function

        n_points = len(h)

        h = np.ones(n_points)
        mid = n_points // 2
        h[:mid] = 0.3
        h[mid:] = 0.7
    
    dx = 2.0 / n_points
    target_integral = n_points / 2
    
    # Normalize initial h
    h = h * target_integral / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Critical shift sampling strategy
    def get_critical_shifts(h, n_shifts=15):
        shifts = []
        # Sample around likely critical regions
        for i in range(n_shifts):
            # Pick a point and look at nearby shifts
            idx = np.random.randint(0, n_points)
            base_shift = int(np.random.normal(0, n_points/8))
            shift = np.clip(base_shift, 1, n_points-1)
            shifts.append(shift)
        return list(set(shifts))  # Remove duplicates
    
    # Try different approaches to escape local minima
    step_sizes = [0.02, 0.01, 0.005]
    current_step_size = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        current_step_size = step_sizes[iteration % len(step_sizes)]
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Optimization via linear program with sampled constraints
        critical_shifts = get_critical_shifts(h)
        
        # Prepare constraints matrix
        n_constraints = len(critical_shifts)
        A_eq = np.zeros((1, n_points))  # Integral constraint
        b_eq = np.array([target_integral])
        A_ub = np.zeros((n_constraints, n_points))
        b_ub = np.zeros(n_constraints)
        
        # Build constraints
        for i, k in enumerate(critical_shifts):
            # For shift k: sum_{i} h[i] * (1 - h[i+k]) <= C * dx
            # Inequality: sum h[i] * (1 - h[i+k]) <= C * dx
            # We'll do this in stages, first compute the linear part:
            # sum h[i] - sum h[i] * h[i+k] <= C * dx
            # returns sum(h) - sum(h[i]*h[i+k]) <= C * dx
            # So we form: -sum(h[i]*h[i+k]) + sum(h) <= C*dx
            # Let weights[i] = 1 if i+k < n_points else 0
            # A_ub[i, ] = weights
            # But since it's a quadratic term, manually compute it
            # Let's build a simplified linearization
            
            # Approximate upper linearization of sum(h[i] * (1 - h[i+k]))
            # Linearizing around the current h
            # Let L = sum(h[i] * (1 - h[i+k]))  
            # For fixed h, this is (sum h[i]) - sum(h[i]*h[i+k])  
            # = 1 - sum(h[i]*h[i+k])
            linear_part = np.zeros(n_points)
            for j in range(n_points):
                if j + k < n_points:
                    linear_part[j] = h[j + k]
                else:
                    # Boundary conditions: assume 0 if out of range
                    linear_part[j] = 0
            A_ub[i, :] = -linear_part  # -sum(h[i]*h[i+k])
            # sum(h) = target_integral, so subtract from right side
            b_ub[i] = target_integral * dx - np.sum(h * (1 - linear_part))  # approximate
        
        # Set up the linear programming problem:
        # Minimize C subject to:
        # A_ub * h <= b_ub  
        # A_eq * h = b_eq
        # 0 <= h[i] <= 1
        # And add a constraint that h is somewhere bounded above by 1
        # One variable C to minimize
        # [A_ub] [h] <= [b_ub]
        # Let's solve with a hack: 
        # Create a test solution and tweak it
        
        # Instead of full LP, do a stochastic search
        # Generate neighborhood perturbations
        new_h = h.copy()
        
        # Random perturbation
        perturb = np.random.normal(0, current_step_size, n_points)
        new_h = h + perturb
        # Ensure feasibility
        new_h = np.clip(new_h, 0, 1)
        # Rescale to sum to n_points/2
        new_h = new_h * target_integral / np.sum(new_h)
        
        # Occasionally do larger jitter
        if np.random.rand() < 0.15:
            i = np.random.randint(n_points)
            new_h[i] = np.random.rand() * 0.5
            new_h = new_h * target_integral / np.sum(new_h)
            new_h = np.clip(new_h, 0, 1)
            
        h = new_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_integral / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
