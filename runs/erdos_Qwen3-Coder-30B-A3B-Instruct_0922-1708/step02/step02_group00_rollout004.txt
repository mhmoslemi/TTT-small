import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Multiple starts with noise
    starts = 5
    for start in range(starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize h
        h = best_h.copy() if start == 0 else np.random.uniform(0, 1, n_points)
        h = clip_and_scale(h, n_points)
        
        # Adam-like optimization with small steps
        for iter in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Evaluate C5
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Compute gradient using finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - c5) / eps
                
            # Update with adaptive step
            lr = 0.001
            h = h - lr * grad
            h = clip_and_scale(h, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
