import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use provided construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)

        # Use 45 points as in the parent construction

        h = np.ones(n_points) * 0.5

    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    # More sophisticated optimization approach
    def eval_overlap(h, dx):
        """Compute C5 value"""
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def optimize_segments(h, dx, n_seg=3):
        """Optimize h using piecewise segments with at most n_seg segments"""
        # 1. Compute clusters of similar values
        cluster_centers = np.linspace(0, 2, n_seg + 1)
        
        # 2. For simple case, we'll just optimize the values and boundaries of segments
        #    using simulated annealing approach within a structured space
        
        # Parameterize:
        # - values v1,...,vn_seg (in [0,1])
        # - segment boundaries 0 = b0 < b1 < ... < b_nseg = 2
        
        # We will be optimizing a compact parameterization
        
        def pack_params(values, boundaries):
            return np.concatenate([values, boundaries])
        
        def unpack_params(params, n_seg):
            values = params[:n_seg]
            boundaries = params[n_seg:]
            return values, boundaries

        # Optimizer function
        def obj(params):
            values, boundaries = unpack_params(params, n_seg)
            values = np.clip(values, 0, 1)
            boundaries = np.clip(boundaries, 0, 2)
            boundaries = np.sort(boundaries)
            boundaries = np.concatenate([[0], boundaries, [2]])  # Ensure order and bounds
            
            # Recompute h from segments (simplified: use step function)
            h_new = np.empty(n_points)
            for i in range(n_seg):
                start, end = int(boundaries[i]*n_points/2), int(boundaries[i+1]*n_points/2)
                if end <= start:
                    continue
                h_new[start:end] = values[i]
            h_new = np.clip(h_new, 0, 1)
            
            # adjust to correct integral
            integral = np.sum(h_new) * dx
            if integral > 0:
                h_new = h_new * (n_points / 2) / integral
            h_new = np.clip(h_new, 0, 1)
            
            return eval_overlap(h_new, dx)
        
        # Initial guess
        initial_values = np.ones(n_seg) * 0.5
        initial_boundaries = np.linspace(0.2, 1.8, n_seg - 1)
        initial_guess = pack_params(initial_values, initial_boundaries)
        
        # Optimize using scipy minimizer with appropriate constraints
        bounds = [(0,1)] * n_seg + [(0,2)] * (n_seg - 1)
        
        try:
            res = minimize(obj, initial_guess, method='Nelder-Mead',
                          options={'maxiter':50, 'disp':False})
            if not res.success:
                return h.copy()
            values, boundaries = unpack_params(res.x, n_seg)
            values = np.clip(values, 0, 1)
            boundaries = np.clip(boundaries, 0, 2)
            boundaries = np.sort(boundaries)
            boundaries = np.concatenate([[0], boundaries, [2]])

                h_new = h_new * (n_points / 2) / integral
                
            return np.clip(h_new, 0, 1)
        except:
            return h.copy()

        if iteration % 10 == 0:

        c5 = eval_overlap(h, dx)

            best_c5 = c5
            best_h = h.copy()

        if iteration - last_improvement > 200:
            break
            
        # Adaptive random perturbation with better control for resets
        perturb = np.random.normal(0, 0.02, n_points)
        h_new = h + perturb

            h_new = h_new * (n_points / 2) / integral

    # Final validation

    if integral > 0:

    best_h = np.clip(best_h, 0, 1)
    
    final_c5 = eval_overlap(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
