import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction or create random spikes
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        # Create random shape with spikes
        h = np.random.rand(n_points) * 0.5
        # Add some spikes to make it more interesting
        for i in range(5):
            idx = np.random.randint(0, n_points)
            h[idx] = 1.0
        h = np.clip(h, 0, 1)
    
    # Normalize to integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.001
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute all correlations and find max overlap shift
        corr = correlate(h, 1 - h, mode='full')
        max_shift_idx = np.argmax(corr)
        k_max = max_shift_idx - (n_points - 1)
        
        # Compute gradient for the critical shift k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Contribution from h[i] * (1 - h[i+k_max])
            if i + k_max >= 0 and i + k_max < n_points:
                grad[i] += 1 - h[i + k_max]
            # Contribution from h[i-k_max] * (-h[i])
            if i - k_max >= 0 and i - k_max < n_points:
                grad[i] += -h[i - k_max]
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        
        # Bias correction
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        
        # Update parameters
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Compute C5 for this h
        corr_full = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr_full) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
