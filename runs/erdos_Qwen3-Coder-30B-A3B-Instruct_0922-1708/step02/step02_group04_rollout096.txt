import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize population
    n_points = 45 if 'initial_h_values' in globals() else 100
    population_size = min(50, max(10, n_points // 2))
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_init = np.clip(h_init, 0, 1)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_init = np.clip(h_init, 0, 1)
    
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h.copy())
    
    # Helper to compute overlap
    def compute_c5(h):
        correlation = correlate(h, 1 - h, mode='full')
        return np.max(correlation) * (2.0 / n_points)
    
    # Track best solution
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        generation += 1
        
        # Evaluate fitness of population
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(c5)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Selection via roulette wheel
        inv_fitness = [1 / (f + 1e-12) for f in fitness]
        total_fitness = sum(inv_fitness)
        probabilities = [f / total_fitness for f in inv_fitness]
        
        # Generate new population
        new_population = [best_h.copy()]
        for _ in range(population_size - 1):
            # Select two parents
            parent1_idx = np.random.choice(population_size, p=probabilities)
            parent2_idx = np.random.choice(population_size, p=probabilities)
            
            # Crossover
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Segment-wise crossover with 50% chance
            crossover_mask = np.random.rand(n_points) < 0.5
            offspring = np.where(crossover_mask, parent1, parent2)
            
            # Mutation
            mutation_mask = np.random.rand(n_points) < 0.1
            if np.any(mutation_mask):
                noise = np.random.normal(0, 0.05, n_points)
                offspring = offspring + noise
                offspring = np.clip(offspring, 0, 1)
                
                # Rescale to meet integral constraint
                offspring = offspring * (n_points / 2) / np.sum(offspring)
                offspring = np.clip(offspring, 0, 1)
            
            new_population.append(offspring)
        
        population = new_population
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
