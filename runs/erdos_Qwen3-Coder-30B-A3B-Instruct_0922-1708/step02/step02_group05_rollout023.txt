import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # PSO parameters
    swarm_size = 50
    max_iter = 500
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize swarm
    particles = np.random.rand(swarm_size, n_points) * 0.5
    for i in range(swarm_size):
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    velocities = (np.random.rand(swarm_size, n_points) - 0.5) * 0.1
    
    # Track personal best
    personal_best = particles.copy()
    personal_best_scores = np.full(swarm_size, float('inf'))
    
    # Initialize global best
    global_best = h.copy()
    global_best_score = float('inf')
    
    start_time = time.time()
    current_iter = 0
    
    while time.time() - start_time < budget_s - 1 and current_iter < max_iter:
        current_iter += 1
        w = w_start - (w_start - w_end) * (current_iter / max_iter)
        
        # Evaluate all particles
        for i in range(swarm_size):
            # Compute C5
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            score = np.max(correlation) * dx
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = particles[i].copy()
                
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = particles[i].copy()
        
        # Update positions and velocities
        for i in range(swarm_size):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) +
                           c2 * r2 * (global_best - particles[i]))
            
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Final evaluation of the best solution
    correlation = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (global_best.tolist(), final_c5, n_points)
