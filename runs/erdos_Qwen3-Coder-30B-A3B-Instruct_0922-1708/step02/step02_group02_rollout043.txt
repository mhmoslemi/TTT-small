import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
    else:
        h = np.ones(n_points) * 0.5
        
    # Rescale to satisfy the constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Objective function for optimization
    def objective(h_vals):
        # Ensure h_vals are valid
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to satisfy integral constraint
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        # Compute correlation
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(correlation) * dx
        return c5
        
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using differential evolution
    result = differential_evolution(
        objective,
        bounds,
        seed=seed,
        maxiter=int(1000 * (budget_s/600)),
        disp=False,
        polish=True
    )
    
    # Final evaluation
    final_h = np.clip(result.x, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    # Return best solution found
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h
    
    return (best_h.tolist(), best_c5, n_points)
