import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral C5"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def ensure_feasibility(h, n_points):
    """Ensure h is in [0,1] and integrates to 1"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = ensure_feasibility(h, n_points)
    else:
        h = np.ones(n_points) * 0.5
        h = ensure_feasibility(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    iteration = 0
    
    # Use a hybrid approach: local optimization + occasional random restarts
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Perturb the current solution slightly to escape local minima
        if iteration % 10 == 0:
            # Random restart with a small perturbation
            h = best_h.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = ensure_feasibility(h, n_points)
        else:
            # Local optimization using scipy
            def objective(h_vec):
                h_modified = ensure_feasibility(h_vec, n_points)
                return evaluate_c5(h_modified, dx)
            
            def constraint_func(h_vec):
                return np.sum(h_vec) * dx - 1.0
            
            # Use SLSQP for constrained optimization
            try:
                result = minimize(
                    objective,
                    h,
                    method='SLSQP',
                    constraints={'type': 'eq', 'fun': constraint_func},
                    bounds=[(0, 1) for _ in range(n_points)],
                    options={'maxiter': 10, 'disp': False}
                )
                if result.success:
                    h = ensure_feasibility(result.x, n_points)
            except:
                # Fallback to simple gradient descent
                pass
            
            # Simple gradient descent with small steps
            eps = 1e-5
            grad = np.zeros_like(h)
            c5_original = evaluate_c5(h, dx)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = ensure_feasibility(h_plus, n_points)
                
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - c5_original) / eps
            
            lr = 0.05
            h_new = h - lr * grad
            h = ensure_feasibility(h_new, n_points)
        
        # Update best solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
