import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
        for _ in range(population_size - 1):
            h_var = h + np.random.normal(0, 0.05, n_points)
            h_var = np.clip(h_var, 0, 1)
            h_var = h_var * (n_points / 2) / np.sum(h_var)
            population.append(h_var)
    else:
        # Initialize with random spikes
        for _ in range(population_size):
            h = np.zeros(n_points)
            num_spikes = np.random.randint(5, 11)
            spike_positions = np.random.choice(n_points, num_spikes, replace=False)
            spike_heights = np.random.uniform(0.3, 0.7, num_spikes)
            for i, pos in enumerate(spike_positions):
                h[pos] = spike_heights[i]
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    def evaluate_individual(h):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def mutate(h, rate=0.15):
        h_new = h.copy()
        indices = np.random.choice(n_points, int(n_points * rate), replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-0.2, 0.2)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def crossover(h1, h2):
        point = np.random.randint(1, n_points - 1)
        h_new1 = np.concatenate([h1[:point], h2[point:]])
        h_new2 = np.concatenate([h2[:point], h1[point:]])
        h_new1 = h_new1 * (n_points / 2) / np.sum(h_new1)
        h_new2 = h_new2 * (n_points / 2) / np.sum(h_new2)
        return h_new1, h_new2
    
    # Main loop
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < 1000:
        # Evaluate population
        fitness = [evaluate_individual(h) for h in population]
        
        # Update best solution
        for i, fit in enumerate(fitness):
            if fit < best_c5:
                best_c5 = fit
                best_h = population[i].copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Create new population
        new_population = population[:population_size//2]  # Keep best
        
        # Generate offspring through crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = np.random.randint(0, population_size // 2)
            parent2_idx = np.random.randint(0, population_size // 2)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < 0.7:
                child1, child2 = crossover(parent1, parent2)
                new_population.extend([child1, child2])
            else:
                # Mutation
                child1 = mutate(parent1)
                child2 = mutate(parent2)
                new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
        generation += 1
    
    if best_h is None:
        # If no improvement found, use the best initial solution
        fitness = [evaluate_individual(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    return (best_h.tolist(), best_c5, n_points)
