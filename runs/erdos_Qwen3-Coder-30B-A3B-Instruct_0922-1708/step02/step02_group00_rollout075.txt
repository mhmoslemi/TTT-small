import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Multi-start optimization with adapted Adam-like strategy
    num_starts = 5
    for start_idx in range(num_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb initial solution
        if start_idx == 0:
            current_h = h.copy()
        else:
            # Add random noise to explore vicinity
            noise_scale = 0.05
            noise = np.random.normal(0, noise_scale, n_points)
            current_h = h + noise
            current_h = clip_and_scale(current_h, n_points)
            
        # Track history for adaptive learning
        vel = np.zeros_like(current_h)
        vel2 = np.zeros_like(current_h)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        
        # Optimization loop
        for iter in range(100):  # Limited iterations per start
            if time.time() - start_time > budget_s - 1:
                break
                
            c5 = evaluate_c5(current_h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
                
            # Gradient approximation
            grad = np.zeros_like(current_h)
            eps_step = 1e-5
            for i in range(n_points):
                h_plus = current_h.copy()
                h_plus[i] += eps_step
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - c5) / eps_step
                
            # Adam-like update with adaptive learning
            vel = beta1 * vel + (1 - beta1) * grad
            vel2 = beta2 * vel2 + (1 - beta2) * grad**2
            mhat = vel / (1 - beta1**(iter+1))
            vhat = vel2 / (1 - beta2**(iter+1))
            
            lr = 0.001
            current_h = current_h - lr * mhat / (np.sqrt(vhat) + eps)
            
            # Maintain feasibility
            current_h = clip_and_scale(current_h, n_points)
            
    return (best_h.tolist(), best_c5, n_points)
