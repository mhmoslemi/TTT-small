import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5

    # Scale to integral constraint
    h = h * (n_points / 2) / np.sum(h)

    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.001

    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)

    best_h = h.copy()
    best_c5 = float('inf')

    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s - 1:
        iteration += 1

        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        # Truncate to valid range for max
        max_idx = np.argmax(corr)
        max_shift = max_idx - (len(h) - 1)
        k0 = max_shift
        c5 = np.max(corr) * dx

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Compute gradient for the critical shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Contribution to gradient from term h[i](1 - h[i+k0])
            if 0 <= i + k0 < n_points:
                grad[i] += 1 - h[i + k0]
            # Contribution from term -h[i] * h[i-k0]
            if 0 <= i - k0 < n_points:
                grad[i] -= h[i - k0]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        h_new = h - alpha * m_hat / (np.sqrt(v_hat) + eps)

        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)

        h = h_new

        # Occasionally add noise to escape local minima
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    return (best_h.tolist(), best_c5, n_points)
