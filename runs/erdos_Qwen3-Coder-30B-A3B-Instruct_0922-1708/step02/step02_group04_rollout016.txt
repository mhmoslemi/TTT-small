import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with a simple initial guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Create a simple 3-segment construction
        h = np.ones(n_points) * 0.5
        # Split into 3 segments with different values
        seg1 = slice(0, n_points//3)
        seg2 = slice(n_points//3, 2*n_points//3)
        seg3 = slice(2*n_points//3, n_points)
        h[seg1] = 0.7
        h[seg2] = 0.5
        h[seg3] = 0.3
    
    # Normalize to integral = 1
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Optimization variables - parameters of segments
    # We will use 3 segments for simplicity and speed
    # Represent as [a1, a2, v1, v2, v3] where a1 < a2
    # Segment 1: [0, a1), value v1
    # Segment 2: [a1, a2), value v2  
    # Segment 3: [a2, 2), value v3
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Utility function to convert segment parameters to h
    def segments_to_h(segments):
        a1, a2, v1, v2, v3 = segments
        a1 = max(0, min(2, a1))
        a2 = max(a1, min(2, a2))
        v1 = max(0, min(1, v1))
        v2 = max(0, min(1, v2))
        v3 = max(0, min(1, v3))
        
        h_new = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < a1:
                h_new[i] = v1
            elif x < a2:
                h_new[i] = v2
            else:
                h_new[i] = v3
        return h_new
    
    # Objective function for optimization
    def compute_c5(segments):
        h_val = segments_to_h(segments)
        correlation = np.correlate(h_val, 1 - h_val, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Optimization with constraints
    def optimize_segments():
        # Parametrize segments: a1, a2, v1, v2, v3
        initial_guess = [0.5, 1.5, 0.7, 0.5, 0.3]
        
        # Optimization bounds
        bounds = [(0, 2), (0, 2), (0, 1), (0, 1), (0, 1)]
        
        # Objective
        def obj(segments):
            return compute_c5(segments)
        
        # Constraint to maintain integral = 1
        def constraint(segments):
            h_new = segments_to_h(segments)
            return np.sum(h_new) * dx - 1.0
        
        cons = {'type': 'eq', 'fun': constraint}
        
        # Optimize with Nelder-Mead for robustness
        try:
            res = minimize(obj, initial_guess, method='Nelder-Mead', 
                          options={'maxiter': 100, 'xatol': 1e-6, 'fatol': 1e-6})
            return res.x if res.success else initial_guess
        except:
            return initial_guess
    
    start_time = time.time()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < 1000:
        iteration += 1
        
        # Optimize segments
        best_segments = optimize_segments()
        h_opt = segments_to_h(best_segments)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        # Compute C5 for this
        correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Occasionally restart with a random variation
        if np.random.rand() < 0.05 or last_improvement > 20:
            h_rnd = h_opt.copy()
            # Add random variation
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    h_rnd[i] += (np.random.rand() - 0.5) * 0.1
            h_rnd = np.clip(h_rnd, 0, 1)
            h_rnd = h_rnd * (n_points / 2) / np.sum(h_rnd)
            correlation = np.correlate(h_rnd, 1 - h_rnd, mode='full')
            current_c5 = np.max(correlation) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_rnd.copy()
                last_improvement = 0
                
        # Random perturb of segments  
        if np.random.rand() < 0.1:
            perturbation = np.random.uniform(-0.2, 0.2, 5)
            perturbed = np.array(best_segments) + perturbation
            # Keep within bounds
            perturbed[0] = max(0, min(2, perturbed[0]))
            perturbed[1] = max(perturbed[0], min(2, perturbed[1]))
            perturbed[2] = max(0, min(1, perturbed[2]))
            perturbed[3] = max(0, min(1, perturbed[3]))
            perturbed[4] = max(0, min(1, perturbed[4]))
            
            h_pert = segments_to_h(perturbed)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            correlation = np.correlate(h_pert, 1 - h_pert, mode='full')
            current_c5 = np.max(correlation) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_pert.copy()
                last_improvement = 0
    
    return (best_h.tolist(), best_c5, n_points)
