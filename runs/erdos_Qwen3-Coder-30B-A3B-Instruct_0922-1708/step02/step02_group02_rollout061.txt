import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        # Create variants
        for _ in range(population_size):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Initialize random population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    best_c5_history = []
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
                
        best_c5_history.append(best_c5)
        if len(best_c5_history) > 10:
            best_c5_history.pop(0)
        
        # Selection: keep top 20%
        selected_indices = np.argsort(fitness)[:population_size // 5]
        selected_population = [population[i] for i in selected_indices]
        
        # Generate new population
        new_population = selected_population.copy()
        
        # Mutation and crossover
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < 0.5 and len(selected_population) >= 2:
                parent1 = selected_population[np.random.randint(0, len(selected_population))]
                parent2 = selected_population[np.random.randint(0, len(selected_population))]
                child = (parent1 + parent2) / 2.0
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            else:
                # Mutation
                parent = selected_population[np.random.randint(0, len(selected_population))]
                child = parent.copy()
                # Perturb 20% of elements
                indices = np.random.choice(n_points, int(0.2 * n_points), replace=False)
                for i in indices:
                    child[i] += np.random.normal(0, 0.02)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        population = new_population[:population_size]
        
        # Increase mutation rate if no improvement in recent iterations
        no_improvement_count += 1
        if no_improvement_count > 100:
            # Reset mutation rate and population
            no_improvement_count = 0
            
        # Occasionally use a random variant for diversity
        if np.random.rand() < 0.05 and len(best_c5_history) >= 2:
            # Add a small random variation of best solution
            if best_h is not None:
                new_h = best_h + np.random.normal(0, 0.01, n_points)
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                # Replace a random member of population
                idx = np.random.randint(0, len(population))
                population[idx] = new_h

    # Final evaluation of best solution
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to one of the population members if no best found
        h = population[0]
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return (h.tolist(), c5, n_points)
