import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap for the given h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def project_h(h, n_points):
    """Project h to feasible region: clip and rescale"""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    return h * scale

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    decrement = 0.01  # Starting decrement for constraint
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current maximum overlap
        c5_current = evaluate_c5(h, dx)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            no_improvement_count = 0
            decrement = 0.01  # Reset decrement on improvement
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                decrement *= 0.5  # Decay decrement
                decrement = max(decrement, 1e-4)
                no_improvement_count = 0
        
        # Find the worst shift (shift that gives maximum overlap)
        correlation = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(correlation)
        
        # Build constraint matrix and gradient for optimization
        # Constraint: sum(h[i] * (1 - h[(i + k_max) % n_points])) * dx <= c5_current - decrement
        # This is linearized as: sum(h[i] * (1 - h[(i + k_max) % n_points])) <= (c5_current - decrement) / dx
        
        # For optimization: minimize C5 subject to constraints
        # We'll solve the problem using a simple gradient-based approach with constraints
        
        # Create a simple constraint for the worst shift
        def constraint_function(h_vec):
            # For shift k_max, compute sum h[i] * (1 - h[(i + k_max) % n_points])
            # We want this sum <= (c5_current - decrement) / dx
            shift_sum = 0
            for i in range(n_points):
                j = (i + k_max) % n_points
                shift_sum += h_vec[i] * (1 - h_vec[j])
            return (c5_current - decrement) / dx - shift_sum
            
        # Use a simple heuristic: gradient descent with constraint projection
        # Compute approx gradient using finite differences
        eps = 1e-4
        
        # Compute gradient for all positions
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - c5_current) / eps
            
        # Gradient step with momentum and adaptive learning rate
        lr = 0.01
        momentum = 0.9
        if iteration == 1:
            velocity = -lr * grad
        else:
            velocity = momentum * velocity - lr * grad
            
        h_new = h + velocity
        
        # Project back to feasible region
        h_new = project_h(h_new, n_points)
        
        # Check improvement with global optimization
        if evaluate_c5(h_new, dx) <= evaluate_c5(h, dx) * 0.9999:
            h = h_new
        else:
            # If no improvement, scale down learning rate
            lr *= 0.5
            velocity = momentum * velocity - lr * grad
            h_new = h + velocity
            h_new = project_h(h_new, n_points)
            h = h_new
            
        # Occasionally restart with better points if we've stagnated
        if iteration % 200 == 0 and best_c5 < 0.3809:
            h = best_h.copy()
            h = project_h(h, n_points)
            decrement = 0.01
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
