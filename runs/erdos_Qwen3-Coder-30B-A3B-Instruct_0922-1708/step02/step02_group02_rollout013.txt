import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_ref = np.array(initial_h_values)
        h_ref = h_ref / np.sum(h_ref) * (n_points / 2)
        for _ in range(population_size):
            h = h_ref + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate all individuals in population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 20%
        elite_size = population_size // 5
        new_population = population[:elite_size]
        
        # Fill rest with crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            
            # Crossover
            if np.random.rand() < 0.5:
                crossover_point = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = (parent1 + parent2) / 2.0
            
            # Mutation
            if np.random.rand() < 0.8:
                mutate_indices = np.random.choice(n_points, int(0.2 * n_points), replace=False)
                for i in mutate_indices:
                    child[i] += np.random.normal(0, 0.02)
            
            # Clip and rescale
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Occasionally increase mutation rate
        if generation % 100 == 0 and generation > 0:
            pass  # no change needed for simplicity
    
    # Final evaluation
    if best_h is None:
        # If no improvement found, return the best from the last population
        corr = correlate(population[0], 1 - population[0], mode='full')
        best_c5 = np.max(corr) * dx
        best_h = population[0]
    
    return (best_h.tolist(), best_c5, n_points)
