import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Genetic algorithm approach
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    tournament_size = 3
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            new_h = h.copy()
            new_h += np.random.normal(0, 0.05, n_points)
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
        else:
            new_h = np.random.rand(n_points)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
        population.append(new_h)
    
    # If initial_h_values was provided, make sure it's in population
    if 'initial_h_values' in globals():
        population[0] = h.copy()
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate fitness for all individuals
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection using tournament
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            # Crossover
            mask = np.random.rand(n_points) < crossover_rate
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child1 += noise
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
                
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child2 += noise
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        # Keep only population_size individuals
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
