import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for given h."""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    start_time = time.time()
    
    # Use differential evolution for global optimization
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vals):
        # Normalize the sum to meet constraint
        h_vals = np.array(h_vals)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return evaluate_c5(h_vals, dx)
    
    # Define constraint to maintain sum constraint
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Keep track of best so far
    def callback(xk, convergence):
        nonlocal best_h, best_c5
        h_vals = np.array(xk)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        c5 = evaluate_c5(h_vals, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals.copy()
    
    # Force better initial solution
    callback(h, 0)
    
    # Use differential evolution algorithm
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=int(500 * (budget_s / 600.0)),
            popsize=30,
            mutation=(0.5, 1.0),
            recombination=0.7,
            seed=seed,
            callback=callback,
            disp=False,
            atol=1e-6,
            rtol=1e-6
        )
        
        # Update with final better value if needed
        if result.success:
            h_candidate = np.array(result.x)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            c5_candidate = evaluate_c5(h_candidate, dx)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
    except:
        # fallback to gradient descent if needed
        pass
    
    return (best_h.tolist(), best_c5, n_points)
