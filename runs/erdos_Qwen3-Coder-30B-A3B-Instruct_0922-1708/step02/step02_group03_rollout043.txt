import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = 0.9
    c2 = 0.9
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h.copy()
            # Add small random noise
            noise = np.random.normal(0, 0.01, n_points)
            particle = particle + noise
        else:
            particle = np.random.uniform(0, 1, n_points)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_values.append(float('inf'))
    
    gbest = h.copy()
    gbest_value = float('inf')
    
    # PSO iterations
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate particles
        for i in range(n_particles):
            particle = particles[i]
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_values[i]:
                pbest_values[i] = c5
                pbest[i] = particle.copy()
                
            if c5 < gbest_value:
                gbest_value = c5
                gbest = particle.copy()
                
        # Update global best
        if gbest_value < best_c5:
            best_c5 = gbest_value
            best_h = gbest.copy()
        
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            particle = particles[i]
            vel = velocities[i]
            
            # Update velocity
            vel = (w * vel + 
                   c1 * r1 * (pbest[i] - particle) + 
                   c2 * r2 * (gbest - particle))
            
            # Update position
            particle = particle + vel
            
            # Apply constraints
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            particles[i] = particle
            velocities[i] = vel
            
        # Occasionally perform local refinement
        if iteration % 20 == 0:
            # Refine global best with local optimization
            h_refined = gbest.copy()
            
            # Simple gradient-like refinement
            for _ in range(20):
                h_plus = h_refined.copy()
                eps = 1e-5
                h_plus = np.clip(h_plus + eps, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_refined = np.correlate(h_refined, 1 - h_refined, mode='full')
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_refined = np.max(corr_refined) * dx
                c5_plus = np.max(corr_plus) * dx
                
                grad = (c5_plus - c5_refined) / eps
                h_refined = h_refined - 0.01 * grad
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            gbest = h_refined.copy()
        
        if iteration % 100 == 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
