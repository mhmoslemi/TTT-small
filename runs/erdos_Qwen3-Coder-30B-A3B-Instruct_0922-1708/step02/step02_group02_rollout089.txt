import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = h_base / np.sum(h_base) * (n_points / 2)
        # Create variations
        for i in range(population_size):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Initialize with random vectors
        for i in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    generation = 0
    no_improvement = 0
    mutation_rate = 0.2
    crossover_rate = 0.5
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if no_improvement > 100:
            mutation_rate = 0.3
            
        # Selection: keep top 20%
        keep_count = population_size // 5
        selected = population[:keep_count]
        
        # Replacement: create new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(selected) >= 2:
                # Crossover
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                child = (parent1 + parent2) / 2
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            else:
                # Mutation
                parent = np.random.choice(selected)
                child = parent.copy()
                mutate_indices = np.random.choice(n_points, int(0.2 * n_points), replace=False)
                for i in mutate_indices:
                    child[i] += np.random.normal(0, 0.02)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
                
        population = new_population[:population_size]
        no_improvement += 1
        
    # Final evaluation of best solution
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to simple approach if something went wrong
        h = np.ones(n_points) * 0.5
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return (h.tolist(), c5, n_points)
