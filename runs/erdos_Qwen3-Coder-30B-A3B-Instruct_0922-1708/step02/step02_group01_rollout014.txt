from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points

        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search bounds
    C_low = 0.0
    C_high = 0.5
    epsilon = 1e-4
    
    while (C_high - C_low > epsilon) and (time.time() - start_time < budget_s - 1):
        mid = (C_low + C_high) / 2.0
        feasible = check_feasibility(h, mid, n_points, dx)
        
        if feasible:
            C_high = mid
            best_h = h.copy()
        else:
            C_low = mid
            
        # Solve LP to update h
        h = solve_lp(n_points, dx, C_high)
        
        # Adjust h to meet constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)

def check_feasibility(h, max_overlap, n_points, dx):
    # Build constraints matrix for LP
    A_eq = np.zeros((1, n_points))
    A_eq[0] = np.ones(n_points)
    b_eq = np.array([n_points / 2])
    
    # Build inequality constraints from overlap conditions
    # For each possible shift k (actually we'll discretize)
    A_ub_list = []
    b_ub_list = []
    
    # Process each shift
    for k_shift in range(n_points):
        # Create constraint: sum h[i] * (1 - h[(i-k_shift)%n_points]) * dx <= max_overlap
        row = np.zeros(n_points)
        for i in range(n_points):
            row[i] = h[(i - k_shift) % n_points] * dx
        A_ub_list.append(row)
        b_ub_list.append(max_overlap)
        
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    # Simple feasibility check with linear programming
    # f = zeros (to minimize a constant)
    f = np.zeros(n_points)
    
    try:
        result = linprog(f, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=[(0, 1)] * n_points, method='highs')
        return result.success 
    except:
        return False
