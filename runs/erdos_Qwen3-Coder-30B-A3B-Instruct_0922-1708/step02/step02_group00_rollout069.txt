import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_generations = 1000
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h_mut = h_init.copy()
            indices = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
            for i in indices:
                h_mut[i] += np.random.uniform(-0.2, 0.2)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            population.append(h_mut)
    else:
        # Generate random initial population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Keep top half
        next_pop = population[:population_size//2]
        
        # Generate new population through crossover and mutation
        while len(next_pop) < population_size:
            # Selection (tournament)
            idx1, idx2 = np.random.choice(len(population[:population_size//2]), size=2, replace=False)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover (single point)
            if np.random.rand() < 0.7:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1, parent2
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < 0.15:
                    indices = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
                    for i in indices:
                        child[i] += np.random.uniform(-0.2, 0.2)
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                
                next_pop.append(child)
        
        population = next_pop[:population_size]
        generation += 1
    
    return (best_h.tolist(), best_c5, n_points)
