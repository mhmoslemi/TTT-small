import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Step 1: Convex relaxation to get a good initialization
    if time.time() - start_time < budget_s / 2:
        try:
            # Formulate the convex relaxation
            h_var = cp.Variable(n_points)
            tau = cp.Variable(1)
            
            # Constraints for convex relaxation
            constraints = []
            constraints.append(cp.sum(h_var) == n_points / 2)
            constraints.append(h_var >= 0)
            constraints.append(h_var <= 1)
            
            # Compute all possible shifts for correlation
            for k in range(n_points):
                # Correlation term for shift k
                if k == 0:
                    corr = cp.sum(h_var * (1 - h_var))
                else:
                    # This is a simplified correlation constraint
                    corr = 0
                    for i in range(n_points):
                        j = i + k
                        if j < n_points:
                            corr += h_var[i] * (1 - h_var[j])
                        else:
                            break
                constraints.append(corr <= tau * n_points)
            
            # Objective
            objective = cp.Minimize(tau)
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_init = np.array(h_var.value).reshape(-1)
                h_init = np.clip(h_init, 0, 1)
                h_init = h_init * (n_points / 2) / np.sum(h_init)
                h = h_init.copy()
        except:
            pass
    
    # Step 2: Gradient-based refinement with adaptive learning rates
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral (C5)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Clip to feasible region and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
