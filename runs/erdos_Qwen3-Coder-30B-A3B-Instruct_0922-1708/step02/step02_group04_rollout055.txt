import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Adaptive strategy with linear programming
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Determine critical shifts based on h structure
        # Sample shifts near where h is likely to contribute to max overlap
        shift_indices = []
        for i in range(0, n_points, max(1, n_points // 10)):
            if h[i] > 0.1:  # Only consider significant contributions
                shift_indices.extend(range(0, n_points - i, max(1, n_points // 20)))
                shift_indices.extend(range(n_points - i, n_points, max(1, n_points // 20)))
                
        shift_indices = list(set(shift_indices))  # Remove duplicates
        shift_indices = sorted(shift_indices)
        
        # If no shifts, create default ones
        if not shift_indices:
            shift_indices = list(range(1, n_points // 4)) + list(range(n_points // 2, n_points - 1))
            
        shift_indices = [idx for idx in shift_indices if idx < n_points]
        
        # LP formulation to find better solution
        # Variables: h[0], h[1], ..., h[n_points-1], C
        num_vars = n_points + 1
        
        # Objective: minimize C
        c = np.zeros(num_vars)
        c[-1] = 1.0  # minimize C

        # Equality constraints: sum h_i = n_points/2
        A_eq = np.zeros((1, num_vars))
        A_eq[0, :-1] = 1.0
        b_eq = np.array([n_points / 2.0])
        
        # Inequality constraints: sum h_i * (1 - h_{i+k}) <= C*dx
        # We approximate overlap using current h structure
        A_ub = []
        b_ub = []
        
        # Build matrix A_ub for each shift k
        for k in shift_indices[:20]:  # Limit to 20 shifts for performance
            if k >= n_points:
                continue
            row = np.zeros(num_vars)
            # For shift k: sum h_i * (1 - h_{i+k}) <= C*dx
            for i in range(n_points - k):
                row[i] = 1.0
                row[i + k] -= 1.0
            row[-1] = -dx
            A_ub.append(row)
            b_ub.append(0.0)
        
        # Bounds for variables [0,1] for h values, C >= 0
        bounds = [(0, 1) for _ in range(n_points)] + [(0, None)]
        
        # Solve LP
        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                            bounds=bounds, method='highs')
            if result.success:
                new_h = np.clip(result.x[:-1], 0, 1)
                # Ensure constraint on integral
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                h = new_h
        except:
            # Fall back to random perturbation if LP fails
            perturb = np.random.normal(0, 0.01, n_points)
            h_new = h + perturb
            h_new = np.clip(h_new, 0, 1)
            # Rescale and normalize to make integral = 1
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
