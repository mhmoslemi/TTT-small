import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    delta = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
            delta = max(1e-4, 0.01 * best_c5)
        else:
            no_improvement_count += 1
            if no_improvement_count % 50 == 0:
                delta = max(1e-4, delta * 0.5)
        
        # Find worst shift
        max_shift = np.argmax(correlation)
        worst_k = max_shift - (n_points - 1)
        
        # Formulate constraint
        # S(k) = sum h[i] * (1 - h[(i+k)%n]) * dx <= C₅ - delta
        # This means: sum h[i] - sum h[i] * h[(i+k)%n] * dx <= C₅ - delta
        # => sum h[i] * h[(i+k)%n] * dx >= sum h[i] - (C₅ - delta)
        
        # Gradient approximation using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps

        # Calculate step using gradient descent
        lr = 0.01 * (1.0 / (1.0 + iteration / 500.0))  # Decaying learning rate
        h_new = h - lr * grad
        
        # Clip to [0,1] and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        if best_c5 < 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
