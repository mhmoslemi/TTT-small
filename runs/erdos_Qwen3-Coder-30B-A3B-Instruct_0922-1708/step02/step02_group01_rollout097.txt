import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    N_pop = 50
    C_rate = 0.8
    M_rate = 0.2
    disturbance_rate = 0.2
    tournament_size = 2
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        population.append(h_init)
    
    # Fill rest of population randomly
    while len(population) < N_pop:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    # Evaluate initial population
    def evaluate_individual(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Tournament selection
    def tournament_selection(pop, fitnesses):
        idxs = np.random.choice(len(pop), size=tournament_size)
        best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        return pop[best_idx]
    
    # Single-point crossover
    def crossover(parent1, parent2):
        if np.random.rand() > C_rate:
            return parent1.copy(), parent2.copy()
        crossover_point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    # Uniform mutation with repair
    def mutate(h):
        h_mut = h.copy()
        for i in range(n_points):
            if np.random.rand() < M_rate:
                h_mut[i] += np.random.uniform(-0.1, 0.1)
        # Clip and rescale
        h_mut = np.clip(h_mut, 0, 1)
        scale = (n_points / 2) / np.sum(h_mut)
        h_mut = h_mut * scale
        return h_mut
    
    # Helper to rescale
    def rescale(h):
        scale = (n_points / 2) / np.sum(h)
        return np.clip(h * scale, 0, 1)
    
    # Main loop
    best_h = None
    best_c5 = float('inf')
    last_improvement = 0
    generation = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitnesses = [evaluate_individual(h) for h in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            last_improvement = generation
        
        # Early termination
        if best_c5 < 0.3807:
            break
            
        # Selection
        selected = []
        for _ in range(N_pop):
            parent = tournament_selection(population, fitnesses)
            selected.append(parent)
        
        # Crossover
        new_population = []
        for i in range(0, N_pop, 2):
            p1, p2 = selected[i], selected[i+1] if i+1 < N_pop else selected[0]
            c1, c2 = crossover(p1, p2)
            new_population.extend([c1, c2])
        
        # Mutation
        for i in range(len(new_population)):
            new_population[i] = mutate(new_population[i])
        
        # Elitism: keep best from previous generation
        fitnesses_pop = [evaluate_individual(h) for h in population]
        elite_indices = np.argsort(fitnesses_pop)[:N_pop//2]
        for i, elite_idx in enumerate(elite_indices):
            new_population[i] = population[elite_idx].copy()
            
        # Disturbance
        if generation - last_improvement > 50:
            num_disturb = int(disturbance_rate * N_pop)
            for _ in range(num_disturb):
                idx = np.random.randint(N_pop)
                new_population[idx] = np.random.rand(n_points)
                new_population[idx] = rescale(new_population[idx])
                
        population = new_population
        generation += 1
        
    return (best_h.tolist(), best_c5, n_points)
