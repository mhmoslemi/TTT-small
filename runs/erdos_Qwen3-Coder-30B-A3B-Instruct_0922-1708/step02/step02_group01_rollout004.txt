import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Particle Swarm Optimization for C5 minimization
    n_points = 200
    dx = 2.0 / n_points
    n_particles = 50
    
    # Initialize particles
    particles = np.random.rand(n_particles, n_points) * 0.5
    velocities = np.random.rand(n_particles, n_points) * 0.1 - 0.05
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles[0] = h_init
    
    # Initialize personal bests
    personal_best = particles.copy()
    personal_best_values = np.full(n_particles, float('inf'))
    
    # Initialize global best
    global_best_idx = 0
    global_best_value = float('inf')
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    max_stagnation = 100
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            # Project onto feasible set
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            particles[i] = h
            
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_values[i]:
                personal_best_values[i] = c5
                personal_best[i] = h.copy()
                
                # Update global best
                if c5 < global_best_value:
                    global_best_value = c5
                    global_best_idx = i
                    stagnation_count = 0
                    
        # Update global best status
        if global_best_value < 0.3807:
            break
            
        stagnation_count += 1
        if stagnation_count > max_stagnation:
            # Restart particles near global best
            for i in range(n_particles):
                particles[i] = personal_best[global_best_idx] + np.random.rand(n_points) * 0.05 - 0.025
                velocities[i] = np.random.rand(n_points) * 0.1 - 0.05
            stagnation_count = 0
            
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            particles[i] += velocities[i]
    
    # Final evaluation of best solution
    best_h = personal_best[global_best_idx]
    best_h = np.clip(best_h, 0, 1)
    scale = (n_points / 2) / np.sum(best_h)
    best_h = best_h * scale
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
