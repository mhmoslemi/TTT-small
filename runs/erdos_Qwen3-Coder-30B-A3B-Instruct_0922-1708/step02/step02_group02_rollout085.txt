import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        # Create diverse variants
        for _ in range(population_size):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Initialize with random vectors
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness_scores.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Selection: Keep top 20%
        keep_count = max(1, population_size // 5)
        selected = population[:keep_count]
        
        # Create new population through crossover and mutation
        new_population = selected.copy()
        
        # Crossover: combine pairs
        while len(new_population) < population_size:
            parent1 = random.choice(selected)
            parent2 = random.choice(selected)
            # Simple average crossover
            child = (parent1 + parent2) / 2
            # Scale to satisfy constraint
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        
        # Mutation: modify some individuals
        mutation_rate = 0.2 if generation < 100 else 0.3
        for i in range(len(new_population)):
            if random.random() < mutation_rate:
                h = new_population[i]
                # Mutate 20% of elements
                indices_to_mutate = random.sample(range(n_points), max(1, n_points // 5))
                for idx in indices_to_mutate:
                    h[idx] += np.random.normal(0, 0.02)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                new_population[i] = h
        
        population = new_population
    
    # Final check with best found
    if best_h is not None:
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(correlation) * dx
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to simple approach if no good solution found
        h = np.ones(n_points) * 0.5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return (h.tolist(), c5, n_points)
