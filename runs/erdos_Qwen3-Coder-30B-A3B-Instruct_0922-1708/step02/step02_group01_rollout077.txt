import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution for better accuracy
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.random.rand(n_particles, n_points)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        # Normalize to sum to n_points/2
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles[0] = h_init
    
    # Initialize velocities
    velocities = np.zeros((n_particles, n_points))
    
    # Initialize personal best positions and values
    personal_best_positions = particles.copy()
    personal_best_values = np.full(n_particles, np.inf)
    
    # Initialize global best
    global_best_value = np.inf
    global_best_position = None
    
    # PSO parameters
    w = 0.7  # Inertia weight
    c1 = 1.5  # Cognitive parameter
    c2 = 1.5  # Social parameter
    
    start_time = time.time()
    iteration = 0
    stagnation_counter = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Project to feasible region
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = np.clip(h * scale, 0, 1)
            particles[i] = h  # Update particle position
            
            # Compute correlation
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < personal_best_values[i]:
                personal_best_values[i] = c5
                personal_best_positions[i] = h.copy()
                
                if c5 < global_best_value:
                    global_best_value = c5
                    global_best_position = h.copy()
                    stagnation_counter = 0  # Reset stagnation counter
                elif c5 < global_best_value + 1e-8:  # Tolerance
                    stagnation_counter += 1
            
            # If we've found a sufficiently good solution, stop early
            if global_best_value < 0.3807:
                break
        
        # Update global best if needed
        if global_best_position is not None:
            # Random restart if stuck
            if stagnation_counter > max_stagnation:
                for i in range(n_particles):
                    particles[i] = np.random.rand(n_points)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                velocities = np.zeros((n_particles, n_points))
                stagnation_counter = 0
            else:
                # Update velocities and positions
                for i in range(n_particles):
                    rand1 = np.random.rand(n_points)
                    rand2 = np.random.rand(n_points)
                    
                    # PSO velocity update
                    velocities[i] = (w * velocities[i] + 
                                     c1 * rand1 * (personal_best_positions[i] - particles[i]) +
                                     c2 * rand2 * (global_best_position - particles[i]))
                    
                    # Position update
                    particles[i] = particles[i] + velocities[i]
        
        if global_best_value < 0.3807:
            break
    
    # Final projection of global best to ensure feasibility
    if global_best_position is not None:
        global_best_position = np.clip(global_best_position, 0, 1)
        scale = (n_points / 2) / np.sum(global_best_position)
        global_best_position = np.clip(global_best_position * scale, 0, 1)
        
        # Final evaluation
        correlation = np.correlate(global_best_position, 1 - global_best_position, mode='full')
        final_c5 = np.max(correlation) * dx
    else:
        # Fallback to last evaluated particle if no global best was found
        final_c5 = np.inf
    
    # Return the best solution found
    if global_best_position is not None:
        return (global_best_position.tolist(), final_c5, n_points)
    else:
        # Return final particle if nothing was found
        return (particles[-1].tolist(), final_c5, n_points)
