import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Optimization loop with adaptive learning and momentum
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences and momentum
        grad = np.zeros_like(h)
        eps = 1e-4
        momentum = 0.9
        if iteration == 1:
            velocity = np.zeros_like(h)
        else:
            velocity = momentum * velocity - (1 - momentum) * grad
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps
            
        # Update with momentum
        lr = 0.005
        h_new = h - lr * velocity
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally try restarts or exploration steps
        if iteration % 200 == 0:
            # Try small random perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            h_try = h + perturbation
            h_try = np.clip(h_try, 0, 1)
            h_try = h_try * (n_points / 2) / np.sum(h_try)
            
            c5_try = compute_c5(h_try)
            if c5_try < c5:
                h = h_try

    return (best_h.tolist(), best_c5, n_points)
