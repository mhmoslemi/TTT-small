import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Multi-start optimization with particles
    n_particles = 10
    particles = []
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initialize particles
    particles.append(h.copy())
    for _ in range(n_particles - 1):
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        best_particle_c5 = float('inf')
        best_particle_idx = 0
        
        for i, particle in enumerate(particles):
            # Compute the overlap integral
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_particle_c5:
                best_particle_c5 = c5
                best_particle_idx = i
                
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
        
        # Update the best particle with gradient descent
        particle = particles[best_particle_idx]
        
        # Compute gradient with finite differences
        grad = np.zeros_like(particle)
        eps = 1e-4
        
        for j in range(n_points):
            p_plus = particle.copy()
            p_plus[j] += eps
            p_plus = np.clip(p_plus, 0, 1)
            p_plus = p_plus * (n_points / 2) / np.sum(p_plus)
            
            corr_plus = np.correlate(p_plus, 1 - p_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_particle_c5) / eps
        
        # Gradient descent step
        particle_new = particle - lr * grad
        
        # Clip and rescale
        particle_new = np.clip(particle_new, 0, 1)
        particle_new = particle_new * (n_points / 2) / np.sum(particle_new)
        
        # Apply perturbation occasionally
        if np.random.rand() < 0.05:
            noise = np.random.normal(0, 0.01, n_points)
            particle_new += noise
            particle_new = np.clip(particle_new, 0, 1)
            particle_new = particle_new * (n_points / 2) / np.sum(particle_new)
        
        particles[best_particle_idx] = particle_new
        
        # Adaptive learning rate
        if best_particle_c5 < 0.381:
            lr = max(0.001, lr * 0.995)
        
        # Occasionally restart particles
        if iteration % 1000 == 0:
            for i in range(n_particles):
                if i == 0 and 'initial_h_values' in globals():
                    # Keep the best performing particle
                    particles[i] = best_h.copy()
                else:
                    # Reinitialize with a new random particle
                    h_init = np.random.uniform(0, 1, n_points)
                    h_init = h_init * (n_points / 2) / np.sum(h_init)
                    particles[i] = h_init
    
    return (best_h.tolist(), best_c5, n_points)
