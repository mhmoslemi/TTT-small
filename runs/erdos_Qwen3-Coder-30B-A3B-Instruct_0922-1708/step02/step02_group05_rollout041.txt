import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def feasible(h):
        return np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) * dx - 1.0) < 1e-10
    
    # Start with a very simple heuristic solution if available
    if 'initial_h_values' in globals():
        current_h = h.copy()
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    # Local search with simulated annealing style approach
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-5
    
    # Generate initial population for genetic algorithm approach
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        # Use initial solution as elite member
        population.append(h.copy())
        while len(population) < population_size:
            # Random initializations with constraint satisfaction
            candidate = np.random.rand(n_points)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            population.append(candidate)
    else:
        for _ in range(population_size):
            candidate = np.random.rand(n_points)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            population.append(candidate)
            
    # GA loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute fitness for all population members
        fitness = []
        for i, ind in enumerate(population):
            if not feasible(ind):
                # Repair if needed
                ind = np.clip(ind, 0, 1)
                ind = ind * (n_points / 2) / np.sum(ind)
                population[i] = ind
            fitness.append(compute_c5(ind))
            
        # Update best if needed
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection (tournament)
        def tournament_selection(k=3):
            selected_indices = np.random.choice(len(population), k)
            selected_fitness = [fitness[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitness)]
            return population[winner_idx].copy()
        
        # Create new population via crossover and mutation
        new_population = [population[min_fitness_idx].copy()]  # Keep elite
        
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            # Crossover
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            
            # Mutation
            mutation_rate = 0.1
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, 0.05)
            
            # Enforce constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population[:population_size]
        
        # Update temperature
        temp = max(min_temp, temp * cooling_rate)
        
        # Occasionally local optimization with gradient descent
        if iteration % 10 == 0:
            if np.random.rand() < 0.3:
                # 30% chance to use gradient descent
                current = population[np.random.randint(len(population))]
                current_c5 = compute_c5(current)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current.copy()
    
    # Final check of best solution
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
