import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize population
    pop_size = min(50, max(10, n_points // 2))
    population = []
    
    # Add initial construction to population
    population.append(h.copy())
    
    # Add random individuals to population
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        population.append(h_rand)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Best solution tracking
    best_overall_h = h.copy()
    best_overall_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = []
        for ind in population:
            correlation = correlate(ind, 1 - ind, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
        # Update best overall
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_overall_c5:
            best_overall_c5 = fitness[min_fitness_idx]
            best_overall_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Early termination if no improvement
        if no_improvement_count > 100:
            break
            
        # Selection with roulette wheel
        min_fitness = min(fitness)
        if min_fitness == 0:
            # Prevent division by zero
            fitness = [f + 1e-10 for f in fitness]
        inverted_fitness = [1.0 / (f + 1e-12) for f in fitness]
        total_fitness = sum(inverted_fitness)
        probs = [f / total_fitness for f in inverted_fitness]
        
        selected = []
        for _ in range(pop_size):
            idx = np.random.choice(pop_size, p=probs)
            selected.append(population[idx].copy())
            
        # Crossover and mutation
        new_population = [best_overall_h.copy()]
        
        for i in range(0, pop_size - 1, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            # Crossover
            child1 = parent1.copy()
            child2 = parent2.copy()
            
            # Segment-wise crossover (simple average)
            crossover_point = np.random.randint(1, n_points - 1)
            child1[crossover_point:] = parent2[crossover_point:]
            child2[crossover_point:] = parent1[crossover_point:]
            
            # Mutation
            mutation_rate = 0.1
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                    child2[j] += np.random.normal(0, 0.05)
                    
            # Clamp and normalize
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
                
        # Replace population
        population = new_population[:pop_size]
        
        # Occasionally apply global perturbation
        if generation % 10 == 0:
            force_reset_idx = np.random.randint(pop_size)
            h_new = np.random.rand(n_points)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            population[force_reset_idx] = h_new

    # Final validation
    correlation = correlate(best_overall_h, 1 - best_overall_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_overall_h.tolist(), final_c5, n_points)
