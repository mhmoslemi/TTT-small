import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Best solution tracking
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Generate initialization strategies
    initial_strategies = []
    
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        if len(h_parent) != n_points:
            x_orig = np.linspace(0, 2, len(h_parent))
            x_new = np.linspace(0, 2, n_points)
            h_parent = np.interp(x_new, x_orig, h_parent)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        initial_strategies.append(h_parent.copy())
        # Add variations
        for _ in range(4):
            h_var = h_parent + np.random.normal(0, 0.02, n_points)
            h_var = clip_and_scale(h_var, n_points)
            initial_strategies.append(h_var.copy())
    else:
        # Random sparse initialization
        for _ in range(5):
            h = np.random.rand(n_points) * 0.5
            h = clip_and_scale(h, n_points)
            initial_strategies.append(h)
    
    # Run optimizations
    for i, h in enumerate(initial_strategies):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_current = h.copy()
        c5_current = compute_c5(h_current, dx)
        
        # Adam optimizer state
        m = np.zeros_like(h_current)
        v = np.zeros_like(h_current)
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8
        lr = 0.001
        t = 0
        
        # Optimization iteration
        for iter in range(500):
            if time.time() - start_time > budget_s - 1:
                break
            
            t += 1
            # Gradient approximation
            grad = np.zeros_like(h_current)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = compute_c5(h_plus, dx)
                grad[j] = (c5_plus - c5_current) / eps
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            h_new = h_current - lr * m_hat / (np.sqrt(v_hat) + epsilon)
            
            # Apply constraints
            h_new = clip_and_scale(h_new, n_points)
            
            c5_new = compute_c5(h_new, dx)
            
            # Accept improvement
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                
            # Early stopping based on improvement
            if iter > 10 and abs(c5_new - c5_current) < 1e-8:
                break

        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            
    # Ensure final validation
    if best_h is None:
        # Fallback uniform
        best_h = np.ones(n_points) * 0.5
        best_h = clip_and_scale(best_h, n_points)
        best_c5 = compute_c5(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
