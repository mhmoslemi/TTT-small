import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    correlation = correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def initialize_population(n_points, n_pop, initial_h_values=None):
    """Initialize population with optional initial solution"""
    population = []
    
    if initial_h_values is not None:
        # Use initial construction if available
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    # Fill the rest with random samples
    while len(population) < n_pop:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    return population

def tournament_selection(population, fitness, tournament_size=2):
    """Select an individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx]

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform single-point crossover"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_range=0.1):
    """Apply uniform mutation with repair"""
    h_mut = h.copy()
    
    for i in range(len(h_mut)):
        if np.random.random() < mutation_rate:
            h_mut[i] += np.random.uniform(-noise_range, noise_range)
    
    # Repair constraints
    h_mut = np.clip(h_mut, 0, 1)
    scale = (len(h_mut) / 2) / np.sum(h_mut)
    h_mut = np.clip(h_mut * scale, 0, 1)
    
    return h_mut

def genetic_algorithm(n_points=100, n_pop=50, budget_s=600, initial_h_values=None):
    """Main genetic algorithm implementation"""
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, n_pop, initial_h_values)
    
    # Evaluate initial population
    fitness = [evaluate_h(h, dx) for h in population]
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    generation = 0
    stagnant_count = 0
    max_stagnant = 50
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            stagnant_count = 0
        else:
            stagnant_count += 1
            
        # Early termination if target reached
        if best_c5 < 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep top 50%
        elite_count = n_pop // 2
        new_population.extend(population[:elite_count])
        
        # Generate offspring through selection, crossover, and mutation
        while len(new_population) < n_pop:
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        new_population = new_population[:n_pop]
        
        # Evaluate new population
        fitness = [evaluate_h(h, dx) for h in new_population]
        
        # Disturbance if stagnating
        if stagnant_count > max_stagnant:
            for i in range(n_pop // 5):  # disturb 20% of population
                idx = np.random.randint(0, n_pop)
                new_population[idx] = np.random.uniform(0, 1, n_points)
                new_population[idx] = new_population[idx] / np.sum(new_population[idx]) * (n_points / 2)
                fitness[idx] = evaluate_h(new_population[idx], dx)
            stagnant_count = 0
            
        population = new_population
        
    # Ensure final solution meets constraints
    best_h = np.clip(best_h, 0, 1)
    scale = (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h * scale, 0, 1)
    best_c5 = evaluate_h(best_h, dx)
    
    return best_h, best_c5, n_points

def run(seed=42, budget_s=600, **kwargs):
    """Main run function that returns the best solution found"""
    np.random.seed(seed)
    
    # Start with genetic algorithm
    h_values, c5_bound, n_points = genetic_algorithm(
        n_points=100,
        n_pop=50,
        budget_s=budget_s,
        initial_h_values=initial_h_values if 'initial_h_values' in globals() else None
    )
    
    return (h_values.tolist(), c5_bound, n_points)
