import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Initialize with piecewise constant function
        h = np.ones(n_points)
        
    # Normalize to have integral = 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define symbolic variables for CVXPY
    h_vars = Variable(n_points)
    
    # Initial guess for cvxpy
    h_vars.value = h.copy()
    
    # Optimization using CVXPY
    while time.time() - start_time < budget_s - 1:
        try:
            # Set up the dual optimization to minimize overlap
            # We minimize max_k int h(x)(1 - h(x+k)) dx = max_k int h(x) - h(x)h(x+k) dx
            # This is equivalent to minimizing sum_k int h(x) - h(x)h(x+k) dx
            # Since the integral is a convolution, we can use Fourier form
            
            # Set up the problem more directly: maximize the negative overlap 
            # This is equivalent to minimizing max_k int h(x)(1 - h(x+k)) dx
            h_vals = h_vars.value
            
            # Compute the overlap matrix using convolution
            correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_vals.copy()
                
            # Sample new optimization directions for exploration
            h_new = h_vals.copy() + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            if np.random.rand() < 0.3:
                h_vars.value = h_new
            else:
                # Use projected gradient
                loss = np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * dx
                # Approximate gradient via finite differences
                eps = 1e-4
                grad = np.zeros_like(h_vals)
                for i in range(n_points):
                    h_plus = h_vals.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - loss) / eps
                
                # Update step
                h_vars.value = h_vals - 0.005 * grad
                h_vars.value = np.clip(h_vars.value, 0, 1)
                h_vars.value = h_vars.value * (n_points / 2) / np.sum(h_vars.value)

        except Exception as e:
            # Fall back to simple gradient descent if CVXPY fails
            h_vals = best_h
            eps = 1e-5
            grad = np.zeros_like(h_vals)
            for i in range(n_points):
                h_plus = h_vals.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - best_c5) / eps
            
            h_vals = h_vals - 0.003 * grad
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            h_vars.value = h_vals
            
            # Evaluate and check improvement
            correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
            c5 = np.max(correlation) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_vals.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
