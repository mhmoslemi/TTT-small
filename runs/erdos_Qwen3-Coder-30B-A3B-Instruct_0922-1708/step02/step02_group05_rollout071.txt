import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    inertia = w_start
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            if len(particle) != n_points:
                particle = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(particle)), particle)
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
        else:
            particle = np.random.rand(n_points)
            particle = particle * (n_points / 2) / np.sum(particle)
            
        particles.append(particle.copy())
        
        velocity = np.random.uniform(-0.5, 0.5, n_points)
        velocities.append(velocity)
        
        # Evaluate particle
        correlation = np.correlate(particle, 1 - particle, mode='full')
        c5 = np.max(correlation) * dx
        
        pbest_positions.append(particle.copy())
        pbest_values.append(c5)
    
    # Initialize global best
    best_idx = np.argmin(pbest_values)
    gbest = pbest_positions[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    # PSO loop
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        # Update inertia weight
        inertia = w_start - (w_start - w_end) * (iteration / max_iter)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                             c1 * r1 * (pbest_positions[i] - particles[i]) +
                             c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(correlation) * dx
            
            # Update pbest
            if c5 < pbest_values[i]:
                pbest_values[i] = c5
                pbest_positions[i] = particles[i].copy()
                
                # Update gbest
                if c5 < gbest_value:
                    gbest_value = c5
                    gbest = particles[i].copy()
                    
    return (gbest.tolist(), gbest_value, n_points)
