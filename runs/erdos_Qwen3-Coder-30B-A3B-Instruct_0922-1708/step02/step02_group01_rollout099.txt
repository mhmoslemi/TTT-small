import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 50
    crossover_rate = 0.8
    mutation_rate = 0.2
    disturbance_rate = 0.2
    tournament_size = 2
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    else:
        h_init = np.ones(n_points) * 0.5
    
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_init.copy()
        else:
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        fitness.append(c5)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Tournament selection
        new_population = []
        for i in range(population_size):
            # Select parents via tournament
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[j] for j in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            new_population.append(population[winner_idx].copy())
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = new_population[i]
            parent2 = new_population[i+1] if i+1 < population_size else new_population[0]
            
            if np.random.rand() < crossover_rate:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            offspring.extend([child1, child2])
        
        # Mutation and rescaling
        mutated_population = []
        for h in offspring:
            if np.random.rand() < mutation_rate:
                # Uniform mutation
                noise = np.random.uniform(-0.1, 0.1, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
            
            # Rescale to maintain constraint
            h = h * (n_points / 2) / np.sum(h)
            mutated_population.append(h)
        
        # Elitism: keep best from previous generation
        elite_indices = np.argsort(fitness)[:population_size//2]
        for i, elite_idx in enumerate(elite_indices):
            mutated_population[i] = population[elite_idx].copy()
        
        # Evaluate new population
        population = mutated_population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Disturbance if no improvement for many generations
        if no_improvement_count > max_no_improvement:
            for i in range(int(population_size * disturbance_rate)):
                idx = np.random.randint(0, population_size)
                population[idx] = np.random.rand(n_points)
                population[idx] = population[idx] / np.sum(population[idx]) * (n_points / 2)
            no_improvement_count = 0
        
        # Early exit if target achieved
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
