import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    n_particles = 50
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            x_orig = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, n_points)
            h0 = np.interp(x_new, x_orig, h0)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Initialize particles and velocities
    particles = np.random.rand(n_particles, n_points) * 0.5 + 0.25
    particles[0] = h0
    velocities = np.random.randn(n_particles, n_points) * 0.01
    
    # Initialize best positions and values
    personal_best_pos = particles.copy()
    personal_best_val = np.full(n_particles, float('inf'))
    global_best_pos = particles[0].copy()
    global_best_val = float('inf')
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    max_stagnation = 100
    w, c1, c2 = 0.7, 1.5, 1.5
    restart_interval = 1000
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        if iteration % restart_interval == 0:
            # Occasionally restart particles with better points
            for i in range(n_particles):
                particles[i] = np.random.rand(n_points) * 0.5 + 0.25
                if i == 0:
                    particles[i] = global_best_pos.copy()
            velocities = np.random.randn(n_particles, n_points) * 0.01
            stagnation_count = 0
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            # Project to feasible region
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_val[i]:
                personal_best_val[i] = c5
                personal_best_pos[i] = h.copy()
                
            # Update global best
            if c5 < global_best_val:
                global_best_val = c5
                global_best_pos = h.copy()
                stagnation_count = 0
                
        stagnation_count += 1
        if stagnation_count > max_stagnation:
            # Restart stagnated particles
            for i in range(n_particles):
                if np.allclose(particles[i], global_best_pos, atol=1e-6):
                    particles[i] = np.random.rand(n_points) * 0.5 + 0.25
                else:
                    particles[i] = global_best_pos + np.random.randn(n_points) * 0.01
            velocities = np.random.randn(n_particles, n_points) * 0.01
            stagnation_count = 0
            
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best_pos[i] - particles[i]) +
                           c2 * r2 * (global_best_pos - particles[i]))
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
        # Early exit if target achieved
        if global_best_val < 0.3807:
            return (global_best_pos.tolist(), global_best_val, n_points)
            
    return (global_best_pos.tolist(), global_best_val, n_points)
