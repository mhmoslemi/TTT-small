import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Binary search bounds for C5
    C_low = 0.0
    C_high = 0.5
    epsilon = 1e-4
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iterations = 0
    
    # Efficient LP-based feasibility check
    def check_feasibility(C):
        # Construct constraints for the LP:
        # minimize 0 subject to:
        #   sum(h[i] * (1 - h[(i+k)%n_points]) * dx) <= C for all k
        #   0 <= h[i] <= 1
        #   sum(h) * dx = 1
        
        # Build constraint matrix efficiently
        A_eq = np.zeros((1, n_points))
        A_eq[0] = np.ones(n_points)
        b_eq = np.array([n_points / 2])  # sum(h) = n_points / 2 after rescaling
        
        # Inequality constraints
        A_ineq = []
        b_ineq = []
        
        # For each shift k, create a constraint
        for k in range(n_points):
            row = np.zeros(n_points)
            for i in range(n_points):
                row[i] = dx * (1 - h[(i+k) % n_points])
            A_ineq.append(row)
            b_ineq.append(C)
            
        A_ineq = np.array(A_ineq)
        b_ineq = np.array(b_ineq)
        
        # Bounds for each h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try to solve with linprog
        try:
            # We are only checking feasibility, not optimizing
            result = linprog(
                c=np.zeros(n_points),
                A_ub=A_ineq,
                b_ub=b_ineq,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method='highs',
                options={'presolve': True}
            )
            
            if result.success:
                return True, result.x
            else:
                return False, None
                
        except:
            return False, None
    
    # Refine solution using gradient steps with early termination
    def refine_solution(h, C):
        h = np.copy(h)
        for _ in range(100):  # Limited steps to keep it simple
            # Calculate overlap (approximate)
            correlation = np.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(correlation) * dx
            
            if max_overlap <= C + 1e-6:
                break
                
            # Simple gradient descent with fixed step
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                grad[i] = (np.max(corr_plus) * dx - max_overlap) / eps
                
            # Take step
            step_size = 0.01
            h_new = h - step_size * grad
            
            # Clamp and normalize
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
        return h
    
    # Binary search loop
    while time.time() - start_time < budget_s - 1 and (C_high - C_low) > epsilon:
        iterations += 1
        
        C_mid = (C_low + C_high) / 2
        
        feasible, new_h = check_feasibility(C_mid)
        
        if feasible:
            C_high = C_mid
            if new_h is not None:
                # Refine solution to get a better local minimum
                refined_h = refine_solution(new_h, C_mid)
                # Compute C5 for refined solution
                corr = np.correlate(refined_h, 1 - refined_h, mode='full')
                c5 = np.max(corr) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = refined_h.copy()
        else:
            C_low = C_mid
            
        # Occasionally refine the best solution found
        if iterations % 50 == 0:
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
    
    # Final refinement one more time
    best_h = refine_solution(best_h, C_high)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
