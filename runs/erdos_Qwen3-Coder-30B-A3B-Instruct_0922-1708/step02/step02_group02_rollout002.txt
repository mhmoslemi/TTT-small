import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            x_orig = np.linspace(0, 2, len(h0))
            x_target = np.linspace(0, 2, n_points)
            h0 = np.interp(x_target, x_orig, h0)
    else:
        # Blocky initialization
        h0 = np.ones(n_points)
        mid = n_points // 2
        h0[:mid] = 0.5
        h0[mid:] = 1.5
    
    h0 = h0 * (n_points / 2) / np.sum(h0)  # normalize to sum=50
    
    # Convex optimization approach using dual variables
    start_time = time.time()
    
    # Variables for optimization
    h = Variable(n_points, name="h")
    
    # Objective: minimize max_k ∫ h(x)(1 - h(x+k)) dx
    # This is equivalent to minimizing max_k (h * (1-h))[k]
    
    # Since we can't directly optimize max, we'll use another approach:
    # Set up the constraint that h is a valid density with sum = 1
    # And use cvxpy to enforce bound constraints
    
    # Reformulate using cvxpy
    prob = Problem(
        Minimize(0),  # dummy objective, we'll use constraint-based approach
        [
            h >= 0, 
            h <= 1,
            sum(h) == n_points / 2  # equivalent to ∫h=1
        ]
    )
    
    # Because this is hard to solve directly, we'll try a gradient-based 
    # approach with safe bounds, but with better initialization and updates
    
    # Best parameters so far: 
    h_best = h0.copy()
    best_c5 = float('inf')
    
    # Simple gradient descent approach that should be faster and robust
    learning_rate = 0.01
    max_iter = int(budget_s * 50)  # Conservative estimate
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution  
        correlation = np.correlate(h_best, 1 - h_best, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_best.copy()
            
        # Compute gradient numerically using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h_best.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # normalize
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Update
        h_new = h_best - learning_rate * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_best = h_new

    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx

    return (best_h.tolist(), final_c5, n_points)
