import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = 0.9
    c2 = 0.9
    v_max = 0.1
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h_part = h.copy()
            # Add small random noise
            noise = np.random.normal(0, 0.01, n_points)
            h_part += noise
            h_part = np.clip(h_part, 0, 1)
            h_part = h_part * (n_points / 2) / np.sum(h_part)
        else:
            h_part = np.random.rand(n_points)
            h_part = h_part * (n_points / 2) / np.sum(h_part)
        particles.append(h_part)
        velocities.append(np.zeros(n_points))
        pbest.append(h_part.copy())
        pbest_vals.append(float('inf'))
    
    gbest = h.copy()
    gbest_val = float('inf')
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate particles
        for i in range(n_particles):
            particle = particles[i]
            
            # Compute the overlap integral
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_vals[i]:
                pbest_vals[i] = c5
                pbest[i] = particle.copy()
                
            if c5 < gbest_val:
                gbest_val = c5
                gbest = particle.copy()
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
        
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            vel = (w * velocities[i] + 
                   c1 * r1 * (pbest[i] - particles[i]) + 
                   c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            vel = np.clip(vel, -v_max, v_max)
            
            particles[i] += vel
            
            # Clip and rescale
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            velocities[i] = vel
            
        # Occasionally do local refinement with scipy
        if iteration % 20 == 0:
            # Try a local search on the global best
            x0 = gbest.copy()
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                result = minimize(objective_func, x0, method='SLSQP', 
                                bounds=bounds, constraints=cons, 
                                options={'maxiter': 10})
                if result.success:
                    refined = result.x
                    refined = np.clip(refined, 0, 1)
                    refined = refined * (n_points / 2) / np.sum(refined)
                    correlation = np.correlate(refined, 1 - refined, mode='full')
                    c5_new = np.max(correlation) * dx
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = refined.copy()
            except:
                pass
                
        # Early termination check
        if iteration % 100 == 0 and iteration > 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)

def objective_func(h):
    n_points = len(h)
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx
