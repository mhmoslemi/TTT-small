import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated annealing parameters
    T0 = 0.1
    tau = 1000
    start_time = time.time()
    
    # Cooling schedule function
    def temperature(t):
        return T0 / (1 + t / tau)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        t = iter_count
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Generate candidate solution
        # Add noise
        noise_scale = temperature(t) * 0.1
        h_prime = h + np.random.normal(0, noise_scale, n_points)
        
        # Clip and rescale
        h_prime = np.clip(h_prime, 0, 1)
        h_prime = h_prime * (n_points / 2) / np.sum(h_prime)
        
        # Compute candidate C5
        correlation_prime = np.correlate(h_prime, 1 - h_prime, mode='full')
        c5_prime = np.max(correlation_prime) * dx
        
        # Accept or reject
        delta_c5 = c5_prime - c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temperature(t)):
            h = h_prime
        
        # Periodic validation
        if iter_count % 50 == 0:
            # Re-validate best solution
            validation_corr = np.correlate(best_h, 1 - best_h, mode='full')
            validated_c5 = np.max(validation_corr) * dx
            if validated_c5 < best_c5:
                best_c5 = validated_c5
    
    return (best_h.tolist(), best_c5, n_points)
