import numpy as np
from scipy.optimize import minimize
import time

def compute_overlap(h, dx):
    """Compute the maximum overlap for given h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def normalize_h(h, n_points):
    """Normalize h to have integral = 1"""
    return h * (n_points / 2) / np.sum(h)
