def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():

        if len(h) != n_points:

            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    tau = 1.0
    alpha = 0.001
    lambda_lag = 10.0
    mu = np.zeros(n_points)
    nu = np.zeros(n_points)
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Surrogate objective construction
        overlaps = []
        for k in range(-n_points+1, n_points):
            shift = np.roll(1-h, k)
            if k >= 0:
                overlap = np.sum(h[:n_points-k] * shift[k:])
            else:
                overlap = np.sum(h[-k:] * shift[:-k])
            overlaps.append(overlap * dx)
        
        # Smooth max approximation
        L_tau = 1/tau * np.log(np.sum(np.exp(tau * np.array(overlaps))))
        
        # Compute gradient via finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlaps_plus = []
            for k in range(-n_points+1, n_points):
                shift = np.roll(1-h_plus, k)
                if k >= 0:
                    overlap = np.sum(h_plus[:n_points-k] * shift[k:])
                else:
                    overlap = np.sum(h_plus[-k:] * shift[:-k])
                overlaps_plus.append(overlap * dx)
                
            L_plus = 1/tau * np.log(np.sum(np.exp(tau * np.array(overlaps_plus))))
            grad[i] = (L_plus - L_tau) / eps
            
        # Update h
        h_new = h - alpha * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adjust Lagrange multipliers
        constraint_violation = np.sum(h) - n_points / 2
        lambda_lag += 0.1 * constraint_violation
        
        # Decrease tau
        if iteration % 100 == 0:
            tau *= 0.99
            
    return (best_h.tolist(), best_c5, n_points)
