import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
        for _ in range(population_size - 1):
            h_mut = h.copy()
            indices = np.random.choice(n_points, int(0.15 * n_points), replace=False)
            for i in indices:
                h_mut[i] += np.random.uniform(-0.2, 0.2)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            population.append(h_mut)
    else:
        # Random spike initialization
        for _ in range(population_size):
            h = np.zeros(n_points)
            num_spikes = np.random.randint(5, 11)
            positions = np.random.choice(n_points, num_spikes, replace=False)
            for i in positions:
                h[i] = np.random.uniform(0.5, 1.0)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Create new population
        new_population = population[:population_size//2]
        
        # Elitism
        new_population.append(population[0])
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(population_size//2, 2, replace=False)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover with 70% probability
            if np.random.random() < 0.7:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                mutation_indices = np.random.choice(n_points, int(0.15 * n_points), replace=False)
                for i in mutation_indices:
                    child[i] += np.random.uniform(-0.2, 0.2)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        population = new_population[:population_size]
    
    return (best_h.tolist(), best_c5, n_points)
