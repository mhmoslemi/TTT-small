import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 45
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    def normalize_h(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate_h(h):
        # Normalize
        h = np.clip(h, 0, 1)
        h = normalize_h(h)
        
        # Evaluate overlap
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5, h
    
    # Initialize population
    pop_size = min(50, max(10, n_points // 2))
    population = []
    if 'initial_h_values' in globals():
        c5, h = evaluate_h(h_init)
        population.append((h, c5))
    
    # Generate random initial population
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h)
        c5, h = evaluate_h(h)
        population.append((h, c5))
    
    # Sort population
    population.sort(key=lambda x: x[1])
    best_c5 = population[0][1]
    best_h = population[0][0].copy()
    
    start_time = time.time()
    gen = 0
    no_improvement = 0
    max_no_improvement = 100
    
    # Mutation params
    mutation_rate = 0.1
    mutation_std = 0.05
    
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        gen += 1
        # Selection with roulette wheel
        fitnesses = np.array([x[1] for x in population])
        # Convert to probabilities
        inv_fitnesses = 1 / (fitnesses + 1e-10)
        probs = inv_fitnesses / np.sum(inv_fitnesses)
        
        new_population = [population[0]]  # Keep best
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Select parents
            parent_indices = np.random.choice(pop_size, size=2, p=probs)
            parent1, parent2 = population[parent_indices[0]][0], population[parent_indices[1]][0]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    mutate_indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
                    for idx in mutate_indices:
                        child[idx] += np.random.normal(0, mutation_std)
                
                # Clamp and normalize
                child = np.clip(child, 0, 1)
                child = normalize_h(child)
                
                # Evaluate
                c5, _ = evaluate_h(child)
                new_population.append((child, c5))
        
        # Trim population
        new_population.sort(key=lambda x: x[1])
        population = new_population[:pop_size]
        
        # Track best
        current_best_c5 = population[0][1]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0][0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
    
    return (best_h.tolist(), best_c5, n_points)
