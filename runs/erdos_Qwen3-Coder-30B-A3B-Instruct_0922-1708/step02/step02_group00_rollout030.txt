import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from provided parent or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Seed with a small spike
        h = np.zeros(n_points)
        h[n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    epsilon = 1e-8
    alpha = 0.001
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute correlations for all shifts
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the shift that maximizes the overlap
        max_idx = np.argmax(corr)
        k = max_idx - (len(h) - 1)  # This is the shift between 0 and len(h)-1
        
        # Gradient computation for the critical shift only
        grad = np.zeros_like(h)
        
        # For every index i, compute its contribution
        for i in range(n_points):
            # Contribution from h[i] * (1-h[i+k]) where i+k < n_points
            if i + k < n_points and i + k >= 0:
                grad[i] += 1 - h[i+k]
            # Contribution from -h[i] * h[i-k] where i-k >= 0
            if i - k >= 0:
                grad[i] -= h[i-k]
        
        grad *= dx  # scale by dx
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        
        h = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
