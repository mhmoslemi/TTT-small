import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with symmetric
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Create symmetric initial construction: h=1 in first half, h=0 in second half
        h = np.ones(n_points)
        mid = n_points // 2
        h[mid:] = 0
        # Perturb slightly to break symmetry
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Keep track of last few C5 values to detect stagnation
    recent_c5 = []
    max_recent = 10
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Update best solution if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Keep recent values
        recent_c5.append(c5)
        if len(recent_c5) > max_recent:
            recent_c5.pop(0)
        
        # Adapt step size based on recent performance
        avg_recent = np.mean(recent_c5) if len(recent_c5) > 1 else c5
        step_size = max(0.001, min(0.1, 0.01 * (1 + (avg_recent - c5) / (avg_recent + 1e-10))))
        
        # Make adaptive perturbation
        perturb = np.random.normal(0, step_size, n_points)
        
        # Create candidate for h
        h_new = h + perturb
        h_new = np.clip(h_new, 0, 1)
        # Normalize to constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Occasionally do systematic exploration
        if np.random.rand() < 0.1 and iteration > 10:
            # Flip some elements
            flip_idx = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
            h_new[flip_idx] = 1 - h_new[flip_idx]
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
