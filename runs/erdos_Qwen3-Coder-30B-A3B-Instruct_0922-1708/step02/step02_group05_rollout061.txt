import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Rescale to meet constraint ∫h = 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Convex relaxation for initialization
    start_time = time.time()
    
    # Formulate convex problem
    h_vars = cp.Variable(n_points)
    constraints = [0 <= h_vars, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
    
    # Sample a few shifts for the convex problem
    max_shifts = min(50, n_points)
    shift_indices = np.linspace(0, n_points - 1, max_shifts, dtype=int)
    
    # Create a convex approximation using relaxations
    objective = cp.Minimize(cp.sum(h_vars))
    
    # Solve the convex problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        h_init = np.array(h_vars.value).reshape(-1)
        h = np.clip(h_init, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient descent refinement
    while time.time() - start_time < budget_s - 1:
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-5

        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps

        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.05, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01

        # Update step
        h_new = h - lr * grad
        
        # Clip to feasible region and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
