import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation to get a good starting point
    if time.time() - start_time < budget_s * 0.6:
        # Formulate convex relaxation problem
        h_cvx = cp.Variable(n_points)
        tau = cp.Variable()
        
        # Constraints for convex relaxation
        constraints = [0 <= h_cvx, h_cvx <= 1]
        constraints.append(cp.sum(h_cvx) == n_points / 2)
        
        # Add constraints for each shift
        for k in range(n_points):
            # kth shift of the correlation
            corr_expr = 0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    corr_expr += h_cvx[i] * (1 - h_cvx[j])
                else:
                    break
            constraints.append(corr_expr <= tau * n_points)
        
        # Objective
        objective = cp.Minimize(tau)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=500)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_init = np.array(h_cvx.value).reshape(-1)
                if np.all(h_init >= 0) and np.all(h_init <= 1):
                    # Rescale to meet constraint
                    h_init = h_init * (n_points / 2) / np.sum(h_init)
                    h = np.clip(h_init, 0, 1)
        except:
            pass    # Continue with uniform initialization if convex relaxation fails
    
    # Gradient-based refinement using simulated annealing approach
    iteration = 0
    temp = 1.0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Adaptive simulated annealing with gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Temperature-based adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8)) * temp
        else:
            lr = 0.01 * temp
            
        # Update step with noise to escape local minima
        h_new = h - lr * grad + np.random.normal(0, 1e-6, n_points)
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Acceptance probability based on simulated annealing
        if c5 > c5 + 0.01:  # This condition is artificial; we're declining to accept worse solutions
            pass # Do not accept worse solution
        else:
            h = h_new
            
        # Decrease temperature
        temp *= 0.99999
        
    # Final computation using the best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
