import numpy as np
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum equals n_points/2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T0 = 0.1
    tau = 1000
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Current temperature
        t = iteration
        T = T0 / (1 + t / tau)
        
        # Generate candidate by adding Gaussian noise
        h_candidate = h + np.random.normal(0, T * 0.1, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to meet constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate candidate
        correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(correlation) * dx
        
        # Evaluate current
        correlation_current = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlation_current) * dx
        
        # Accept or reject based on Metropolis criterion
        delta_c5 = c5_candidate - c5_current
        if delta_c5 < 0 or np.random.rand() < math.exp(-delta_c5 / T):
            h = h_candidate.copy()
            c5_current = c5_candidate
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
