import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral C5 for given h"""
    correlation = correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def normalize_h(h, n_points):
    """Ensure h sums to n_points/2 and stays in [0,1]"""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    return h * scale

def tournament_selection(population, fitness, tournament_size=2):
    """Select an individual via tournament selection"""
    idx = np.random.choice(len(population), tournament_size)
    best_idx = idx[np.argmin([fitness[i] for i in idx])]
    return population[best_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Single-point crossover"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, max_mutation=0.1):
    """Uniform mutation with repair"""
    h = h.copy()
    mask = np.random.random(len(h)) < mutation_rate
    delta = np.random.uniform(-max_mutation, max_mutation, len(h))
    h[mask] += delta[mask]
    return np.clip(h, 0, 1)

def genetic_algorithm(seed, budget_s, n_points=100):
    """Genetic algorithm for minimizing C5"""
    np.random.seed(seed)
    dx = 2.0 / n_points
    N_pop = 50
    C_rate = 0.8
    M_rate = 0.2
    max_generations = 1000
    elite_size = N_pop // 2

    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = normalize_h(h, n_points)
        population.append(h)
    
    # Fill population
    while len(population) < N_pop:
        h = np.random.random(n_points)
        h = normalize_h(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    generation = 0
    stagnation_counter = 0
    last_best_c5 = best_c5
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < N_pop:
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            child1, child2 = crossover(parent1, parent2, C_rate)
            
            child1 = mutate(child1, M_rate)
            child1 = normalize_h(child1, n_points)
            
            child2 = mutate(child2, M_rate)
            child2 = normalize_h(child2, n_points)
            
            new_population.extend([child1, child2])
        
        # Trim to exact size
        new_population = new_population[:N_pop]
        
        # Evaluate new population
        new_fitness = [evaluate_h(h, dx) for h in new_population]
        
        # Update best solution
        current_best_idx = np.argmin(new_fitness)
        current_best_c5 = new_fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = new_population[current_best_idx].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        # Stagnation recovery
        if stagnation_counter > 50:
            for i in range(N_pop // 5):
                idx = np.random.randint(N_pop)
                new_population[idx] = np.random.random(n_points)
                new_population[idx] = normalize_h(new_population[idx], n_points)
            stagnation_counter = 0
        
        population = new_population
        fitness = new_fitness
        
        # Early termination
        if best_c5 < 0.3807:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main function to run the optimization"""
    best_h, best_c5 = genetic_algorithm(seed, budget_s)
    n_points = len(best_h)
    return (best_h.tolist(), best_c5, n_points)
