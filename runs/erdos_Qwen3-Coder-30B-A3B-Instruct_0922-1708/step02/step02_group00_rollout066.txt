import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def clip_and_rescale(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def mutate(h, mutation_rate=0.15):
    """Mutate h by adding noise to a fraction of points"""
    n_points = len(h)
    mask = np.random.rand(n_points) < mutation_rate
    noise = np.random.uniform(-0.2, 0.2, n_points)
    h_new = h + mask * noise
    return clip_and_rescale(h_new, n_points)

def crossover(parent1, parent2):
    """Single-point crossover between two parents"""
    n_points = len(parent1)
    point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        # Interpolate initial construction
        h_base = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_base))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_orig, h_base)
        h_interp = clip_and_rescale(h_interp, n_points)
        population.append(h_interp)
        
        # Add variations
        for _ in range(population_size - 1):
            h_mut = mutate(h_interp.copy())
            population.append(h_mut)
    else:
        # Initialize with random spikes
        for _ in range(population_size):
            h = np.zeros(n_points)
            n_spikes = np.random.randint(5, 11)
            positions = np.random.randint(0, n_points, n_spikes)
            heights = np.random.rand(n_spikes)
            for pos, height in zip(positions, heights):
                h[pos] = height
            h = clip_and_rescale(h, n_points)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50% (elitism)
        next_gen = population[:population_size//2]
        
        # Generate new candidates
        while len(next_gen) < population_size:
            # Selection: tournament
            i1, i2 = np.random.choice(population_size//2, 2, replace=False)
            parent1 = population[i1]
            parent2 = population[i2]
            
            # Crossover
            if np.random.rand() < 0.7:
                child1, child2 = crossover(parent1, parent2)
                child1 = clip_and_rescale(child1, n_points)
                child2 = clip_and_rescale(child2, n_points)
                next_gen.extend([child1, child2])
            else:
                # Just mutate
                child1 = mutate(parent1)
                child2 = mutate(parent2)
                next_gen.extend([child1, child2])
        
        population = next_gen[:population_size]
    
    if best_h is None:
        # Fallback
        fitness = [evaluate_h(h, dx) for h in population]
        i_best = np.argmin(fitness)
        best_h = population[i_best]
        best_c5 = fitness[i_best]
    
    return (best_h.tolist(), best_c5, n_points)
