import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    lr_initial = 0.001
    lr_decay = 0.999
    max_iter = 10000
    n_restarts = 5
    
    def compute_c5_and_gradient(h):
        # Compute correlation for all lags
        correlation = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(correlation)
        max_corr_val = correlation[max_corr_idx]
        
        # Analytic gradient for the active lag
        grad = np.zeros_like(h)
        k = max_corr_idx - (len(h) - 1)  # lag
        for i in range(len(h)):
            grad[i] = (1 - h[(i - k) % len(h)]) - h[(i + k) % len(h)]
            
        return max_corr_val * dx, grad, k
    
    lr = lr_initial
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_current = h.copy()
        last_improvement = 0
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            iteration += 1
            
            # Compute current value and gradient
            c5, grad, lag = compute_c5_and_gradient(h_current)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_current.copy()
                last_improvement = i
                
            # Update step
            h_new = h_current - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_current = h_new
            
            # Adaptive learning rate
            lr *= lr_decay
            
            # Early stopping
            if i - last_improvement > 500:
                break
                
    return (best_h.tolist(), best_c5, n_points)
