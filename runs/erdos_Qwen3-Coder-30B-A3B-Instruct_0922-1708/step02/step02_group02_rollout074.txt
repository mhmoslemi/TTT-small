import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Initialize QP variables
    h_vars = cp.Variable(n_points)
    t = cp.Variable()
    
    # Constraints list for QP
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2, t >= 0]
    
    # Keep track of worst lags
    worst_lags = []
    
    while time.time() - start_time < budget_s - 1:
        # Compute correlation once per iteration
        corr = np.correlate(h, 1-h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Build QP problem
        # Only include most active constraints
        active_lags = []
        if worst_lags:
            active_lags = worst_lags[-20:]  # Last 20 worst lags
        else:
            # Initially, use top lags with highest correlation
            active_lags = np.argsort(corr[-n_points+1:])[::-1][:20] + 1
        
        # Create linear constraints for the active lags
        new_constraints = constraints.copy()
        
        # Add the most significant correlation constraints
        for k in active_lags:
            # Compute indices for correlation
            # lag k means comparing h[i] with h[i+k] 
            # For a lag k, we have (n_points - k) valid comparisons
            expr = 0
            for i in range(n_points - k):
                expr += h_vars[i] * (1 - h_vars[i+k]) * dx
            
            new_constraints.append(expr <= t)
        
        # Solve QP
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, new_constraints)
        
        try:
            # Try to solve with different solvers if needed
            prob.solve(solver=cp.MOSEK, verbose=False, max_iter=1000)
        except:
            try:
                prob.solve(solver=cp.GLPK, verbose=False)
            except:
                # Fallback to a gradient-based approach if QP fails
                break
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = h_vars.value
            if h_new is not None:
                h = np.clip(h_new, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        # Update worst lags
        corr = np.correlate(h, 1-h, mode='full')
        worst_lags = np.argsort(corr[-n_points+1:])[::-1][:20] + 1
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
