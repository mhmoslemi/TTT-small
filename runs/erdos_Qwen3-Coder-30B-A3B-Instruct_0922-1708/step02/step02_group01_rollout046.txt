import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral C5 for given h"""
    correlation = correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def initialize_population(n_points, n_pop, initial_h_values=None):
    """Initialize population for genetic algorithm"""
    population = []
    
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h.copy())
    
    # Fill remaining population with random samples
    while len(population) < n_pop:
        h = np.random.random(n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h.copy())
    
    return population

def tournament_selection(population, fitness, tournament_size=2):
    """Select individual using tournament selection"""
    selected_idx = np.random.randint(0, len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitness[i] for i in selected_idx])]
    return population[best_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform single-point crossover"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(individual, mutation_rate=0.2, noise_scale=0.1):
    """Mutate individual with noise and rescale"""
    mutant = individual.copy()
    
    for i in range(len(mutant)):
        if np.random.random() < mutation_rate:
            mutant[i] += np.random.normal(0, noise_scale)
    
    # Clip to [0,1]
    mutant = np.clip(mutant, 0, 1)
    
    # Rescale to meet sum constraint
    scale = (len(mutant) / 2) / np.sum(mutant)
    mutant = np.clip(mutant * scale, 0, 1)
    
    return mutant

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    n_pop = 50
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points, n_pop, 
                                       initial_h_values=globals().get('initial_h_values'))
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Track generations without improvement
    no_improvement_count = 0
    max_no_improvement = 50
    
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
        
        if best_c5 < 0.3807:
            break
            
        no_improvement_count += 1
        
        # Selection: keep top 50%
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:n_pop//2]
        selected_pop = [population[i] for i in top_indices]
        selected_fitness = [fitness[i] for i in top_indices]
        
        # Generate new population via crossover and mutation
        new_population = selected_pop.copy()
        
        # Elitism: keep best individuals
        if len(selected_pop) < n_pop:
            # Add best from previous population
            for i in range(n_pop - len(selected_pop)):
                if i < len(top_indices):
                    idx = top_indices[i]
                    new_population.append(population[idx].copy())
        
        # Crossover and mutation
        while len(new_population) < n_pop:
            parent1 = tournament_selection(selected_pop, selected_fitness)
            parent2 = tournament_selection(selected_pop, selected_fitness)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Truncate to exactly n_pop
        new_population = new_population[:n_pop]
        population = new_population
        
        # Stagnation recovery
        if no_improvement_count > max_no_improvement:
            # Introduce some random individuals
            for i in range(n_pop // 5):
                new_individual = np.random.random(n_points)
                new_individual = new_individual / np.sum(new_individual) * (n_points / 2)
                idx_to_replace = np.random.randint(0, len(population))
                population[idx_to_replace] = new_individual
            
            no_improvement_count = 0
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to simple initial solution if something went wrong
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_orig, h)
            h = h / np.sum(h) * (n_points / 2)
        else:
            h = np.ones(n_points) * 0.5
            
        final_c5 = evaluate_h(h, dx)
        return (h.tolist(), final_c5, n_points)
