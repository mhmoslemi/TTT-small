import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def normalize_h(h, n_points):
    """Normalize h to have integral = 1"""
    return h * (n_points / 2) / np.sum(h)

def enforce_constraints(h, n_points):
    """Enforce all constraints on h"""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, n_points)
    return h

def crossover(parent1, parent2, n_points):
    """Create offspring from two parents via segment-wise averaging"""
    # Randomly decide crossover points
    n_crossovers = np.random.randint(1, n_points // 4)
    points = sorted(np.random.choice(n_points, n_crossovers, replace=False))
    points = [0] + points + [n_points]
    
    offspring = parent1.copy()
    for i in range(len(points)-1):
        if i % 2 == 1:  # Take from parent2 in odd segments
            offspring[points[i]:points[i+1]] = parent2[points[i]:points[i+1]]
    
    return offspring

def mutate(h, mutation_rate=0.1, std=0.05):
    """Apply Gaussian mutation to h"""
    mask = np.random.rand(len(h)) < mutation_rate
    h = h + np.random.normal(0, std, len(h)) * mask
    return h
