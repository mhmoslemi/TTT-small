    h = h * (n_points / 2) / np.sum(h)

    # Initialize tracking variables
    best_h = h.copy()
    best_c5 = float('inf')
    
    iteration = 0

        iteration += 1
        
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Create perturbed version with local refinement
        h_new = h.copy()
        
        # Randomly adjust some elements with differential adaptation
        n_adjust = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_adjust, replace=False)
        for i in indices:
            # Small random perturbation with adaptive magnitude
            delta = np.random.normal(0, 0.02)
            h_new[i] += delta
            h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Maintain integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)

        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx

            h = h_new
        else:
            # Occasionally accept worse steps to escape local minima
            if np.random.rand() < 0.05:
                h = h_new
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
