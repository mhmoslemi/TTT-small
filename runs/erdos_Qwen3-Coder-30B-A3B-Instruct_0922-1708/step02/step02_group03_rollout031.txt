import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral (C5)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute exact gradient
        max_shift = np.argmax(correlation)
        grad = np.zeros_like(h)
        
        # Derivative w.r.t. h[i] for the maximum correlation shift
        # Efficient computation of gradient for max correlation
        for i in range(n_points):
            if max_shift < n_points:
                # Forward shift
                j = (i + max_shift) % n_points
                grad[i] = (1 - h[j]) * dx
            else:
                # Backward shift
                j = (i - (max_shift - n_points + 1)) % n_points 
                grad[i] = (1 - h[j]) * dx
                
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Occasionally check for early termination
        if iteration % 50 == 0 and iteration > 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
