import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Initialize PSO parameters
    swarm_size = 50
    max_iter = 500
    w_start = 0.9
    w_end = 0.4
    c1 = c2 = 2.0
    
    # Initialize particles and velocities
    particles = np.random.rand(swarm_size, n_points) * 0.5
    for i in range(swarm_size):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    velocities = (np.random.rand(swarm_size, n_points) - 0.5) * 0.1
    
    # Track personal bests
    pbest = particles.copy()
    pbest_scores = np.full(swarm_size, np.inf)
    
    # Track global best
    gbest_idx = 0
    gbest_score = np.inf
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (particles[gbest_idx] - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            score = np.max(correlation) * dx
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                
            # Update global best
            if score < gbest_score:
                gbest_score = score
                gbest_idx = i
                
        # Periodic diversity maintenance
        if iteration % 50 == 0:
            # Perturb some particles to maintain diversity
            for i in range(swarm_size):
                if np.random.rand() < 0.3:
                    particles[i] = np.random.rand(n_points) * 0.5
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Return final best
    best_h = particles[gbest_idx].copy()
    best_c5 = gbest_score
    
    return (best_h.tolist(), best_c5, n_points)
