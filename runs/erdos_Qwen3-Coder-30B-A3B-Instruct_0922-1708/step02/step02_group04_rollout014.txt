import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    # Use the provided construction or start with uniform

    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Gradient-based optimization with structured search
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def compute_gradient(h):
        n = len(h)
        grad = np.zeros(n)
        # Compute contribution of each point to gradient
        for i in range(n):
            # For each shift k, compute gradient contribution
            for k in range(n):
                if i - k >= 0:
                    grad[i] += (1 - h[(i-k) % n]) * dx
                else:
                    grad[i] += (1 - h[(i-k+n) % n]) * dx
        return grad
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Try structured perturbation
        # Try several random searches with different methods
        if np.random.rand() < 0.7:
            # Local gradient descent step
            grad = compute_gradient(h)
            step_size = 0.01
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            # Rescale
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
        else:
            # Random perturbation with different amplitude
            perturb = np.random.normal(0, 0.02, n_points)
            h_new = h + perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)

    # Final validation
    best_h = np.clip(best_h, 0, 1)

    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
