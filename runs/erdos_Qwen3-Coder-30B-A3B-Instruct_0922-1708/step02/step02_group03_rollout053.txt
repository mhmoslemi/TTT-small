```python
import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.001
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # For tracking improvement
    last_improvement = 0
    patience = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute correlation for all lags
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        active_lag = np.argmax(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Compute exact gradient for the active lag
        grad = np.zeros_like(h)
        # The active lag in correlation corresponds to indices in h
        # correlation index k corresponds to shift k-n_points+1 in h
        lag = active_lag - n_points + 1
        
        # This is the gradient: d/dh[j] of max_k (int h(x) * (1-h(x+k)) dx) 
        # For fixed k, if h(x) appears in the integral at position x, 
        # then contribution to gradient w.r.t. h[j] is (1-h[j+k]) if j=x, 
        # and -h[j-k] if j=x+k, and 0 otherwise.
        # For the maximum k case, we'll compute the gradient symbolically
        # But we can estimate it from the fact that correlation value is sum(h(i)*h(i+k))
        
        # Calculate gradient: -h[j-k] + (1-h[j+k]) for active lag k
        # Since we integrate over [0,2], we actually have circular convolution 
        # But since we're looking at an explicit "max" over lags, 
        # we approximate gradient with the derivative for that active lag
        # Let's recompute gradient component by component directly from correlation
        
        # For i from 0 to n_points - 1:
        # locally, account for convolution being:
        #     h(k)*((1-h(state_shift_idx)) on the part that overlaps
        # Gradient of term h[i] * (1-h[i+k]) with respect to h[i] is (1-h[i+k])
        # Gradient of h[i] * h[i-k] with respect to h[i] is h[i-k]
        # But we're just taking partial of the maximum convolution value (i.e. argmax)
        # with respect to h[i].
        # For this, we look at how h[i] affects at the single active lag k
        # A better idea: since this is multivariable optimization with constraint,
        # use direct gradient: we expand the Lagrangian, compute the multiplier
        # But just assume that active lag gives us the steepest descent direction:
        # Exact gradient for each h[i]:
        # For each i, find where h[i] * (1-h[i+k]) or h[i] * h[i-k] appears in the maximized integral
        # Approximate:
        # d/dh[i] [Integral h(x) * (1-h(x+k)) dx] = (1 - h[i+k]) for positive contributions 
        # and similarly for the other terms
        # Use: for k-th lag: d/dh[i] term h[i]*h[i+k] is h[i+k] and d/dh[i] term h[i]*(1-h[i+k]) is (1-h[i+k])
        # But simplifying, for maximum overlap case, 
        # gradient[i] should represent the sensitivity of the max integral with respect to i
        
        # Correct exact gradient:
        # The j-th element of the gradient for (h*(1-h)) convolution with lag = active lag
        # Gradients: 
        # d/dh[j] \sum_i h[i] * (1-h[i+k]) = (1-h[j+k]) where valid
        # d/dh[j] \sum_i h[i] * h[i-k] = h[j-k] where valid
        # But maximizing these -> use subgradient method:
        # All terms contribute, but gradient of the maximum is relevant only for active lag 
        # The total gradient at any point is the gradient w.r.t. active lag,
        # from the convolution that maximizes the integral.
        # For h(i) to appear in the active contribution we compute:
        #   for lag k, h(i) multiplies (1-h(i+k)) and we want derivative w.r.t h(i)
        #   gradient[i] += (1-h[i+k]) if i+k < n_points , and -h[i-k] if i-k >= 0
        # So:
        # grad[i] = (1 - h[i + lag]) if i + lag < n_points, if same direction, and -h[i - lag] if i - lag >= 0
        # But we're doing correlation with 1-h, so it's cross-correlation
        
        # Redefine: correlation(x, 1-h) = sum_k h[k] (1-h[k+x]) = sum_k h[k]*1 - h[k]*h[k+x]
        # This is
        # sum_k h[k] * (1-h[k+x]) = sum_k h[k] - h[k]*h[k+x]
        # Thus d/dh[i] = - h[i+x]  
        # But we're also computing this with the opposite direction → 
        # Actually the simplest derivation:
        # Define x = h(k) * (1 - h(k + lag)) 
        # d x / dh(i) = 0 if i != k, or i != k + lag
        # if i == k, then residue is (1-h[k+lag]), 
        # if i == k + lag, then it's -h[k]
        # These are just the convolution kernel coefficients (???)
        # But we're dealing with the maximum of these expressions.
        # In order to compute gradient w.r.t. h[j], scan through correlation values.
        # For max over k, d/dh[j] max_k int h(i) * (1-h(i-k)) dx = the finite differences of that k for which it's maximal  
        # Alternately, we have max_k (sum_i h[i] * (1-h[i-k])) as a function of h.
        # For each h[j], the gradient is the sum: sum_k of d/dh[j] h[i] * (1-h[i-k]) when i matches j or i-k=j, 
        # i.e. h[j] contributes to term h[j] * (1 - h[j-k]) or h[j] * h[j+k] where i-k=j (i=j+k).
        # So gradient j += (1 - h[j-k]) when j-k = j (i.e. k=0), so we get a match. 
        # Or h[j] contributes to term h[j-k] * (1 - h[j-k-k]) 
        # What this means:
        # For lag k = active lag: 
        #   If i = j-k, then h[j] contributes to h[j-k] * (1 - h[j-k-k]) = h[j-k] * (1 - h[j-2k])
        #   If i = j+k, then h[j] contributes to h[j+k] * (1 - h[j+k-k]) = h[j+k] * (1 - h[j])  
        # Above equation is simplified wrong.
        # Best: correct gradient for h[j], considering that we're computing sequentially:
        # h(k)*h(k+l) part of convolution sum is i-th gradient contribution from d/dh[j] h[i]*h[i+l]  
        # Is: d/dh[j] h[i]*h[i+l] = 0 if j!=i and j!=i+l 
        # Is: h[i+l] if j=i, and h[i] if j=i+l
        # So we get: for lag l, if we're at i-th element, d/dh[j] contribution = h[i+l] when j=i and h[i] when j=i+l 
        # For the 1-h part: d/dh[j] h[i]*(1-h[i+l]) = 0 if j!=i and j!=i+l  
        # Is: -(1-h[i+l]) if j=i, and -h[i] if j=i+l
        # So combining for i-th position: sum_{l(all)} if lag is l, part contributes to d/dh[j]
        # Derivative of convolution sums wrt h[i]:
        # overall contributions are: 
        # h[i] * h[i+l] => d/dh[j] contribution is h[i+l] when j==i, and h[i] when j==i+l  
        # h[i] * (1 - h[i+l]) => d/dh[j] contribution is -(1 - h[i+l]) when j==i, and -h[i] when j==i+l
        # So both cases: we just check i-th element in the sum of correlation at shift l.
        # i-d element of gradient w.r.t. h: h[i+l] (if intersection in first case) 
        # plus -h[i] if second case, or +h[i] if it's the Second term 
        # But again, we only care about what contributes actively.
        # Better: Just find it.
        # correct: Let's redefine with a correct reindexing for cross correlation w.r.t. lag. 
        # corr = correlation(h, 1-h), so for lag = k: corr[k] = sum_i h[i] * (1-h[i-k]) 
        # d/dh[j] corr[k] is (1 - h[j-k]) if j-k >= 0, and -h[j+k] if j+k < n_points
        # But the key insight: 
        # We are maximizing the convolution over all lags
        # Therefore, when trying to compute the gradient of this max over h,
        # We need to determine which lag is active and compute the gradient for that. 
        # This is not standard; normally the gradient would behave nicely in the context of a functional,
        # but we're maximizing over lags, which makes it non-smooth and discontinuous
        # The "subgradient" approach is to take an arbitrary one among the active lags, e.g. [argmax_c corr(k) ]
        # Continue.
        # Actually use the fundamental theorem of calculus like before:
        # Or just retrace./
        # So for lag k with maximizing integral: 
        # d/dh[j] integral_{x} h(x) * h(x+k) dx is equal to h[j-k] when j-k >= 0, and h[j+k] when j+k < n_points  
        # d/dh[j] h(x) * (1-h(x+k)) dx = -(1-h[j-k]) when j-k >= 0 
        # Saying that for lag k: 
        # value is sum_i h[i] * (1-h[i-k]) = sum_i h[i] - h[i]*h[i-k]
        # So derivative is: [0 if i!=j] + [ + (1 - h[j-k]) if i=j and j-k >= 0 ] + [ - h[j+k] if i=j and j+k < n_points ]
        # Which gets simplified going back to gradient as:
        # gradient[j] += (1 - h[j-k]) (if j-k >= 0) and gradient[j] -= h[j+k] (if j+k < n_points)
        
        # We iterate through lags and average the gradient or keep one?
        # Let's just use the single active lag:
        # For element j, contributions in correlation sum:
        # If j - lag >= 0 then h[j] appears in term h[j-lag]*h[j-lag-k] with weight h[j-lag-k]
        # If j + lag < n_points then h[j] appears in term h[j+lag]*h[j+lag-k] with weight h[j+lag-k]
        # No. Let's actually rederive. 
        # corr_k = sum_{j} h_j * (1-h_{j-k}) -> Lagrangian method
        # 
        # Let's decide to take gradient for the lag that produces the highest cross correlation.
        # d/dh[i] ( \sum_j h_j * (1-h_{j-k}) ) = 
        # (1 - h[i-k]) if i-k >= 0
        # -h[i+k] if i+k < n_points  
        # Thus
        for i in range(n_points):
            if i - lag >= 0:
                grad[i] += 1 - h[i - lag]
            if i + lag < n_points:
                grad[i] -= h[i + lag]
                
        # Normalize the gradient direction for stability
        grad = grad / (np.linalg.norm(grad) + 1e-8)
        
        # Update with Adam  
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad

        # Bias correction
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        # Update parameters
        h = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Enforce constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping if no improvement in a while
        if iteration - last_improvement > patience and iteration > 10000:
            break
                
    return (best_h.tolist(), best_c5, n_points)
```
