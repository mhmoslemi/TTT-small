import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    not_improved_count = 0
    delta = 0.01
    
    # Add a more targeted optimization approach
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            not_improved_count = 0
            delta = max(1e-4, 0.01 * best_c5)
        else:
            not_improved_count += 1
            if not_improved_count > 50:
                delta *= 0.5
                not_improved_count = 0
                
        if best_c5 < 0.3807:
            break

        # Optimization step with regularization
        h_new = h.copy()
        
        # Gradient estimation using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Apply momentum-based update
        momentum = 0.9
        if iteration == 1:
            velocity = -0.01 * grad
        else:
            velocity = momentum * velocity - 0.01 * grad
            
        h_new = h + velocity
        
        # Apply penalty for constraint violations
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Occasionally try a more random perturbation
        if iteration % 20 == 0:
            h_new = np.random.uniform(0, 1, n_points)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
