import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 100

    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Coarse-to-fine optimization approach
    grid_sizes = [20, 40, 80, 160, 200]
    current_grid_size = n_points
    
    # Helper function to compute C5
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # Helper function to rescale h to exact integral
    def rescale_h(h_vals, target_sum=1.0):
        current_sum = np.sum(h_vals)
        if current_sum > 0:
            return h_vals * target_sum / current_sum
        else:
            return h_vals
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1:
        # If we've gone to smaller grids, start from a larger one
        for grid_size in grid_sizes:
            if grid_size <= current_grid_size:
                continue
                
            # Interpolate from current to new grid
            h_interp = np.interp(np.linspace(0, 2, grid_size), 
                               np.linspace(0, 2, current_grid_size), h)
            h = h_interp.copy()
            current_grid_size = grid_size
            break
        else:
            # We're at or above the highest grid, just do gradient descent
            pass
        
        # Gradient-based optimization
        iterations = 0
        last_improvement = 0
        
        while (time.time() - start_time < budget_s - 1 and 
               iterations - last_improvement < 50):
            iterations += 1
            # Current C5 value
            c5_current = compute_c5(h)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
            
            # Compute gradients via finite differences
            grad = np.zeros_like(h)
            eps = 1e-6
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = rescale_h(h_plus, n_points / 2)
                
                c5_plus = compute_c5(h_plus)
                
                grad[i] = (c5_plus - c5_current) / eps
            
            # Adaptive step size
            alpha = 0.01
            max_trials = 10
            for _ in range(max_trials):
                h_new = h - alpha * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = rescale_h(h_new, n_points / 2)
                
                c5_new = compute_c5(h_new)
                
                # Backtracking line search
                if c5_new < c5_current:
                    h = h_new.copy()
                    break
                else:
                    alpha *= 0.5
        
        # Perturb randomly to escape local minimum
        if np.random.rand() < 0.1:
            perturb = np.random.normal(0, 0.01, current_grid_size)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = rescale_h(h, n_points / 2)
    
    # Final refinement
    c5_final = compute_c5(best_h)
    if c5_final < best_c5:
        best_c5 = c5_final
    
    return (best_h.tolist(), best_c5, current_grid_size)
