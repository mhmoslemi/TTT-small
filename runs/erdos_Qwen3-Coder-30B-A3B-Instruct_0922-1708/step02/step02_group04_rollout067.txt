import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Use 3 segments to create a better structure
        # Start with basic configuration
        h = np.ones(n_points) * 0.5
        # Better split into segments with different values
        # Tend to have high values on the left and low values on the right
        segment1_end = n_points // 3
        segment2_end = 2 * n_points // 3
        h[:segment1_end] = 0.8
        h[segment1_end:segment2_end] = 0.2
        h[segment2_end:] = 0.1
    
    # Normalize to exact integral constraint
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Perform more targeted optimization
        # Perturb 5% of points with adaptive step size
        perturb_indices = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
        perturb_magnitude = 0.01 + 0.005 * np.random.rand()  # adaptive magnitude
        h_new = h.copy()
        for i in perturb_indices:
            h_new[i] += np.random.normal(0, perturb_magnitude)
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        # Rescale and normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Try random jump
        if np.random.rand() < 0.05:
            i = np.random.randint(n_points)
            h_new[i] = np.random.rand()
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
        
        h = h_new
    
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
