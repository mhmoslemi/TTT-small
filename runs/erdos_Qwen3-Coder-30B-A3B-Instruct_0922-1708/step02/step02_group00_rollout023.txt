import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Optimizer parameters
    lr = 0.001
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Adam optimizer variables
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    # Temperature for smooth max
    temp = 1.0
    temp_decay = 0.995
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute surrogate objective with smooth max
        # L(τ) = (1/τ) * log(∑ₖ exp(τ * ∑ᵢ hᵢ(1 - h_{i+k})))
        # Since we're maximizing, we minimize -L(τ)
        overlaps = np.array([np.sum(h[i:] * (1 - h[:len(h)-i])) for i in range(len(h))])
        surrogate = (1.0 / temp) * np.log(np.sum(np.exp(temp * overlaps)))
        
        # Compute gradient numerically
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlaps_plus = np.array([np.sum(h_plus[i:] * (1 - h_plus[:len(h_plus)-i])) for i in range(len(h_plus))])
            surrogate_plus = (1.0 / temp) * np.log(np.sum(np.exp(temp * overlaps_plus)))
            
            grad[i] = (surrogate_plus - surrogate) / eps

        # Update with Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adjust temperature
        if iteration % 100 == 0:
            temp *= temp_decay
            
    return (best_h.tolist(), best_c5, n_points)
