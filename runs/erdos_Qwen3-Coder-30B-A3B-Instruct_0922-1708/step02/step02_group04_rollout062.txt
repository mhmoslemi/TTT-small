import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():

        n_points = len(h)

        n_points = 100

    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Coarse-to-fine optimization approach
    grid_sizes = [20, 40, 80, 160, 200]
    
    # Start with current solution
    current_n = n_points
    current_h = h.copy()
    
    # Interpolate to requested grids
    def interpolate_h(h_in, n_out):
        if len(h_in) == n_out:
            return h_in.copy()
        if len(h_in) > n_out:
            # Downsample
            return h_in[::len(h_in)//n_out]
        else:
            # Upsample
            ratio = n_out / len(h_in)
            h_out = []
            for i in range(n_out):
                idx = i / ratio
                if idx >= len(h_in) - 1:
                    h_out.append(h_in[-1])
                else:
                    lower = int(idx)
                    upper = min(len(h_in) - 1, lower + 1)
                    weight = idx - lower
                    h_out.append(h_in[lower] * (1 - weight) + h_in[upper] * weight)
            return np.array(h_out)
    
    # Try optimization on different grid sizes
    for n in grid_sizes:
        if n > current_n:
            # Interpolate current solution to new grid
            current_h = interpolate_h(current_h, n)
        elif n < current_n:
            # Downsample with averaging for consistency
            current_h = current_h[::current_n//n]
        
        current_n = n
        dx = 2.0 / current_n
        
        # Optimize using projected gradient descent
        max_iter = 200
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute overlaps
            correlation = np.correlate(current_h, 1 - current_h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
            
            # Compute gradient (approximately, finite differences)
            gradient = np.zeros_like(current_h)
            eps = 1e-4
            
            # Compute numerical gradient for all positions
            for j in range(current_n):
                h_plus = current_h.copy()
                h_plus[j] = min(1.0, h_plus[j] + eps)
                # Rescale
                h_plus = h_plus * (current_n / 2) / np.sum(h_plus)
                h_plus = np.clip(h_plus, 0, 1)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad_val = (c5_plus - c5) / eps
                gradient[j] = grad_val
              
            # Gradient descent step with adaptive learning rate
            learning_rate = 0.1
            new_h = current_h - learning_rate * gradient
            # Clip and rescale
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (current_n / 2) / np.sum(new_h)
            
            # Check if improvement
            current_h = new_h
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final computation with best found values
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (current_n / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * (2.0 / current_n)
    
    return (best_h.tolist(), final_c5, current_n)
