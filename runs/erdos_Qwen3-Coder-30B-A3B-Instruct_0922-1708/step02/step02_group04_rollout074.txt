import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 50
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h_init = np.clip(h_init, 0, 1)
    
    # Parameters
    pop_size = min(50, max(10, n_points // 2))
    max_generations = 1000
    mutation_rate = 0.1
    elite_size = max(1, pop_size // 10)
    no_improve_limit = 100
    
    # Initialize population
    population = []
    population.append(h_init.copy())
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    best_fitness = float('inf')
    best_individual = h_init.copy()
    
    start_time = time.time()
    no_improve_count = 0
    
    # Callback to evaluate fitness
    def evaluate_fitness(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for population
        fitness_scores = [evaluate_fitness(h) for h in population]
        
        # Track best so far
        min_fitness = min(fitness_scores)
        if min_fitness < best_fitness:
            best_fitness = min_fitness
            best_individual = population[fitness_scores.index(min_fitness)].copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count >= no_improve_limit:
            # Reset with new diverse population
            population = [best_individual.copy()]
            for _ in range(pop_size - 1):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
                population.append(h)
            no_improve_count = 0
            
        # Selection using roulette wheel
        fitness_array = np.array(fitness_scores)
        # Invert for selection (lower is better)
        inverses = 1.0 / (fitness_array + 1e-12)
        probabilities = inverses / np.sum(inverses)
        
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness_array)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            # Select parents
            parent_indices = np.random.choice(pop_size, size=2, p=probabilities)
            parent1 = population[parent_indices[0]]
            parent2 = population[parent_indices[1]]
            
            # Crossover: single-point
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation for both children
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    mutate_idx = np.random.randint(n_points)
                    child[mutate_idx] = np.random.rand()
                    
                # Enforce constraints
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
                
        # Trim to exact population size
        population = new_population[:pop_size]
    
    # Final validate
    best_fitness = evaluate_fitness(best_individual)
    
    return (best_individual.tolist(), best_fitness, n_points)
