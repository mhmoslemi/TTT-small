import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    iteration = 0
    
    def compute_c5(h_values):
        correlation = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(correlation) * dx
    
    # Try a few different approaches to find a better solution
    def perturb_h(h_values, method='gradient'):
        h_new = h_values.copy()
        if method == 'gradient':
            eps = 1e-5
            grad = np.zeros_like(h_values)
            c5 = compute_c5(h_values)
            
            for i in range(n_points):
                h_plus = h_values.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_c5(h_plus)
                grad[i] = (c5_plus - c5) / eps
                
            lr = 0.01
            h_new = h_values - lr * grad
        elif method == 'random':
            # Add small random noise
            noise = np.random.normal(0, 0.01, n_points)
            h_new = h_values + noise
        else:
            h_new = h_values.copy()
            
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Try both gradient-based and random search strategies
    strategies = ['gradient', 'random', 'mixed', 'local']
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try different strategies
        strategy = strategies[iteration % len(strategies)]
        if strategy == 'gradient':
            h_new = perturb_h(h, 'gradient')
        elif strategy == 'random':
            h_new = perturb_h(h, 'random')
        elif strategy == 'mixed':
            # alternate between methods
            h_new = h if (iteration % 2) == 0 else perturb_h(h, 'gradient')
        else:  # local
            # Local search with small perturbations near current point
            h_new = h.copy()
            for i in range(n_points):
                if np.random.rand() < 0.1:  # 10% chance to mutate
                    h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.05), 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final check
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
