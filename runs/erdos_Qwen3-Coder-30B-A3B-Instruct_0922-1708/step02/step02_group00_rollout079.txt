import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    """Ensure h lies in [0,1] and sums to n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    """Compute maximum overlap integral"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Start multiple optimizations
    start_time = time.time()
    num_starts = 10
    for start in range(num_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize variation
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_orig, h)
            # Add noise
            noise_level = 0.05
            noise = np.random.normal(0, noise_level, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Create sparse random pattern
            h = np.zeros(n_points)
            nonzero_indices = np.random.choice(n_points, size=n_points//10, replace=False)
            h[nonzero_indices] = np.random.uniform(0.1, 0.9, len(nonzero_indices))
            h = clip_and_scale(h, n_points)
            
        # Adam optimizer settings
        alpha = 0.001
        beta1 = 0.9
        beta2 = 0.999
        epsilon = 1e-8
        
        # Adam state
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        
        # Optimization loop
        for iteration in range(500):  # per start
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current value
            c5 = compute_c5(h, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Compute gradient finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = compute_c5(h_plus, dx)
                grad[i] = (c5_plus - c5) / eps
            
            # Update Adam
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**(iteration + 1))
            v_hat = v / (1 - beta2**(iteration + 1))
            
            # Apply update
            h = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
            # Maintain constraints
            h = clip_and_scale(h, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
