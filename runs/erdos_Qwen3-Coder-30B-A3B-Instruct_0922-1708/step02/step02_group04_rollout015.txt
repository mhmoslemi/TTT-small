import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction or create a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Create a piecewise constant function
        h = np.ones(n_points) * 0.5

    # Normalize to ensure integral = 1
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization parameters
    max_iter = 10000
    iter_count = 0
    
    # Start with a reasonable number of segment points
    n_segments = 4
    segment_points = np.linspace(0, 2, n_segments + 1)
    
    def evaluate_c5(h_values):
        correlation = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(correlation) * dx
    
    def make_step_function_from_segments(segments):
        """Create step function from segment parameters."""
        # segments: [v1, v2, ..., vn] where n is number of segments
        # segment_edges: [0, a1, a2, ..., an, 2]
        segment_edges = np.linspace(0, 2, n_segments + 1)
        h_vals = np.zeros(n_points)
        for i in range(n_segments):
            start_idx = int(segment_edges[i] / dx)
            end_idx = int(segment_edges[i + 1] / dx)
            h_vals[start_idx:end_idx] = segments[i]
        return h_vals
    
    def optimize_segments():
        """Simple reconstructor of H and optimizes it."""
        # Initialize
        segments = np.ones(n_segments) * 0.5
        last_c5 = float('inf')
        iter_count = 0
        
        while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
            iter_count += 1
            
            # Create step function
            h_temp = make_step_function_from_segments(segments)
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            c5 = evaluate_c5(h_temp)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_temp.copy()
            
            # Simple gradient-like update
            # We do not have exact gradient, so we try random small changes
            idx = np.random.randint(n_segments)
            delta = np.random.normal(0, 0.05)
            segments[idx] = segments[idx] + delta
            segments[idx] = np.clip(segments[idx], 0, 1)
            
            # Normalize so that integral = 1
            h_temp = make_step_function_from_segments(segments)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h_temp = np.clip(h_temp, 0, 1)
            c5 = evaluate_c5(h_temp)
            
            # Accept or reject based on improvement in overlap
            if c5 < last_c5:
                last_c5 = c5
            else:
                # Revert if not better
                segments[idx] = segments[idx] - delta
            
            # Occasionally adjust all values to cope with overfitting
            if np.random.rand() < 0.05:
                segments = np.clip(segments + np.random.normal(0, 0.02, n_segments), 0, 1)
        
        return best_h, best_c5
    
    # Try optimization from multiple start points
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        # Start with a different initial choice
        initial_segments = np.random.rand(n_segments) * 0.5 + 0.25
        h_vals = make_step_function_from_segments(initial_segments)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        h_vals = np.clip(h_vals, 0, 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals.copy()
    
    # Final optimization phase
    final_h, final_c5 = optimize_segments()
    
    best_h = np.clip(final_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
