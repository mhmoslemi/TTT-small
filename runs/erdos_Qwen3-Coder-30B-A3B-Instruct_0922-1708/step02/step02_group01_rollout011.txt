import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Binary search bounds
    C_low, C_high = 0.0, 0.5
    epsilon = 1e-4
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iter_count = 0
    
    # Binary search
    while C_high - C_low > epsilon and time.time() - start_time < budget_s - 1:
        iter_count += 1
        C_mid = (C_low + C_high) / 2
        
        # Formulate LP constraints for feasibility check
        A_eq = np.zeros((1, n_points))
        A_eq[0] = np.ones(n_points)
        b_eq = np.array([n_points / 2])
        
        A_ub = []
        b_ub = []
        
        # Build overlap constraints
        for k in range(n_points):
            row = np.zeros(n_points)
            for i in range(n_points):
                j = (i + k) % n_points
                row[i] = h[i] - h[j]
            A_ub.append(row)
            b_ub.append(C_mid * dx)
            
        # Inequality constraints for bounds
        for i in range(n_points):
            A_ub.append([1 if j == i else 0 for j in range(n_points)])
            b_ub.append(1.0)  # h[i] <= 1
            A_ub.append([-1 if j == i else 0 for j in range(n_points)])
            b_ub.append(0.0)  # h[i] >= 0
            
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Linear programming solver for feasibility
        try:
            res = linprog(
                c=np.zeros(n_points),
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=[(0, 1) for _ in range(n_points)],
                method='highs'
            )
            
            if res.success:
                best_h = res.x
                best_c5 = C_mid
                C_high = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid
            
        # Periodically update best solution if we have it
        if time.time() - start_time > 0.1:
            # Evaluate current solution
            correlation = np.correlate(best_h, 1 - best_h, mode='full')
            c5_eval = np.max(correlation) * dx
            if c5_eval < best_c5:
                best_c5 = c5_eval
    
    # Final evaluation
    if time.time() - start_time < budget_s - 1:
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        c5_eval = np.max(correlation) * dx
        if c5_eval < best_c5:
            best_c5 = c5_eval
    
    return (best_h.tolist(), best_c5, n_points)
