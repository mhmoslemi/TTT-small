import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    tau = 1.0
    lr = 0.001
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Surrogate objective: smooth max
        def smooth_max(h_vals, tau):
            corr = np.correlate(h_vals, 1 - h_vals, mode='full')
            # Softmax approximation
            max_corr = np.max(corr) 
            exp_term = np.exp(tau * (corr - max_corr))
            return max_corr + np.log(np.sum(exp_term)) / tau
        
        # Compute gradient of the surrogate
        eps = 1e-5
        grad = np.zeros_like(h)
        base_val = smooth_max(h, tau)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            grad[i] = (smooth_max(h_plus, tau) - base_val) / eps

        # Update using gradient descent
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Anneal temperature
        if iteration % 100 == 0:
            tau *= 0.99
            lr *= 0.995

    return (best_h.tolist(), best_c5, n_points)
