import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_generations = 1000
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        for _ in range(population_size):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    last_improvement = 0
    generation = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            correlation = correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = generation
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 20%
        elite_size = population_size // 5
        new_population = population[:elite_size]
        
        # Generate new population through crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = np.random.choice(elite_size)
            parent2_idx = np.random.choice(elite_size)
            
            # Crossover
            if np.random.rand() < 0.5:
                alpha = np.random.rand()
                child = alpha * population[parent1_idx] + (1 - alpha) * population[parent2_idx]
            else:
                child = population[parent1_idx].copy()
                
            # Mutation
            mutation_rate = 0.2 if generation - last_improvement < 100 else 0.3
            mutation_mask = np.random.rand(n_points) < mutation_rate
            if np.any(mutation_mask):
                child[mutation_mask] += np.random.normal(0, 0.02, np.sum(mutation_mask))
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
    if best_h is None:
        # Fallback: try direct optimization as backup
        h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        correlation = correlate(h, 1 - h, mode='full')
        best_c5 = np.max(correlation) * dx
        best_h = h
    
    return (best_h.tolist(), best_c5, n_points)
