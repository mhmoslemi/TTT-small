import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters
    max_iter = 1000
    early_stopping = False
    adaptive_delta = 0.01
    delta_min = 1e-4
    delta_decay = 0.5
    no_improvement_count = 0
    max_no_improvement = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
            adaptive_delta = 0.01
        else:
            no_improvement_count += 1
            
        # If no improvement for a while, reduce delta
        if no_improvement_count >= max_no_improvement:
            adaptive_delta = max(delta_min, adaptive_delta * delta_decay)
            no_improvement_count = 0
            
        # Find the shift that produces maximum overlap
        k_max = np.argmax(correlation)
        max_corr = correlation[k_max]
        
        # Create a constraint for this shift
        # Constraint: sum_{i=0}^{n_points-1} h[i] * (1 - h[(i + k_max) % n_points]) * dx <= C₅_prev - adaptive_delta
        
        # Apply the method of Lagrange multipliers to solve the convex relaxation:
        # Minimize C₅ subject to:
        # 1. sum(h) * dx = 1
        # 2. sum(h[i] * (1 - h[(i + k_max) % n_points])) * dx <= C₅_prev - adaptive_delta
        # 3. 0 <= h[i] <= 1
        
        # Try to construct a better h by projecting the gradient of the constraint
        # This is simulating the dynamic optimization
        
        # Simplified gradient approach: penalize the dominant overlap
        h_projected = h.copy()
        
        # Generic heuristics to explore neighborhood:
        eps = 1e-4
        grad = np.zeros_like(h)
        
        # Approximate the gradient using finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Project gradient descent with momentum
        momentum = 0.9
        if iteration == 0:
            velocity = -0.01 * grad
        else:
            velocity = momentum * velocity - 0.01 * grad
            
        h_new = h + velocity
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally restart with better points if good solution found
        if iteration % 100 == 0 and best_c5 < 0.3808:
            h = best_h.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Early stopping if we are below target
        if best_c5 < 0.3807:
            early_stopping = True
            break
            
    return (best_h.tolist(), best_c5, n_points)
