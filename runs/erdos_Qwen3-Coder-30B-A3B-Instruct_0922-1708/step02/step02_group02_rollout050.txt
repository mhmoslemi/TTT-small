import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a block-like construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute the correlation matrix
    def compute_overlap(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Use feasibility-driven optimization with trust region
    def objective(h_vals):
        return compute_overlap(h_vals)
    
    # Constraints
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    def constraint_bounds(h_vals):
        return np.clip(h_vals, 0, 1) - h_vals
    
    # Try different starting strategies
    for strategy in range(3):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Initialize based on strategy
        if strategy == 0:
            # Use the parent construction if available
            h = best_h.copy()
        elif strategy == 1:
            # Random perturbation of the best solution
            h = best_h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Ad-hoc piecewise-linear function
            h = np.zeros(n_points)
            mid = n_points // 2
            h[:mid] = 1.0
            h[mid:] = 0.0
            h = h * (n_points / 2) / np.sum(h)
            
        # Use scipy minimize with SLSQP
        cons = [{'type': 'eq', 'fun': constraint_sum},
                {'type': 'ineq', 'fun': lambda x: x},
                {'type': 'ineq', 'fun': lambda x: 1 - x}]
        
        try:
            res = minimize(objective, h, method='SLSQP', constraints=cons, 
                          bounds=[(0, 1) for _ in range(n_points)], 
                          options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                current_h = res.x
                current_c5 = compute_overlap(current_h)
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
        except:
            pass
            
        # Try simple gradient ascent/projection for large steps
        lr = 0.01
        for _ in range(200):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Evaluate gradient by finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            base_c5 = compute_overlap(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_overlap(h_plus)
                grad[i] = (c5_plus - base_c5) / eps
            
            # Update step
            h_new = h - lr * grad
            # Project to feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            current_c5 = compute_overlap(h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final validation
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
