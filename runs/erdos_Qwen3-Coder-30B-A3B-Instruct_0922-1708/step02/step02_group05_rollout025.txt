import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize the population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        population.append(h_init)
    
    # Generate random initial population
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Mutation and crossover parameters
    F = 0.8
    CR = 0.7
    
    start_time = time.time()
    
    # Differential evolution loop
    max_generations = 500
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        new_population = []
        
        for i in range(population_size):
            # Select three distinct random individuals
            a, b, c = population[i], population[(i+1)%population_size], population[(i+2)%population_size]
            while a is population[i] or b is population[i] or c is population[i]:
                idx = np.random.randint(0, population_size)
                a = population[idx]
            while b is a or b is population[i]:
                idx = np.random.randint(0, population_size)
                b = population[idx]
            while c is a or c is b:
                idx = np.random.randint(0, population_size)
                c = population[idx]
            
            # Mutation
            v = a + F * (b - c)
            v = np.clip(v, 0, 1)
            v = v * (n_points / 2) / np.sum(v)
            
            # Crossover
            u = np.where(np.random.rand(n_points) < CR, v, population[i])
            
            # Ensure constraint satisfaction
            u = u * (n_points / 2) / np.sum(u)
            
            # Evaluate
            c5_u = evaluate_c5(u, dx)
            
            # Selection
            c5_i = evaluate_c5(population[i], dx)
            if c5_u < c5_i:
                new_population.append(u)
                if c5_u < best_c5:
                    best_c5 = c5_u
                    best_h = u.copy()
            else:
                new_population.append(population[i])
        
        population = new_population
    
    # Final check
    if best_h is None:
        # Pick best from final population
        c5_vals = [evaluate_c5(h, dx) for h in population]
        best_idx = np.argmin(c5_vals)
        best_h = population[best_idx]
        best_c5 = c5_vals[best_idx]
    
    return (best_h.tolist(), best_c5, n_points)
