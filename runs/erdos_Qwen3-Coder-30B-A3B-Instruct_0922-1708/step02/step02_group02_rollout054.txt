import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent solution or create a blocky function
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Create a blocky function as initial guess
        h_init = np.zeros(n_points)
        h_init[:n_points//4] = 1.0
        h_init[n_points//4:3*n_points//4] = 0.5
        h_init[3*n_points//4:] = 0.0
        
    # Normalize to sum to 50 (n_points/2)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define optimization variables
    h = Variable(n_points)
    
    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]
    
    # Duality approach: minimize max_k \int h(x)(1-h(x+k)) dx
    # Using convolution properties, we can reformulate this using duality
    
    # We will use an iterative approach to find a better solution:
    # 1. Start with initial h
    # 2. Use CVX to solve optimization problem
    # 3. Update h using gradient descent
    
    iteration = 0
    
    # For stability, we fix the number of iterations and time limit consistently
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
            
        # Try further optimization using CVXPY
        try:
            # Define a small perturbation problem
            # In hope to improve at this point in optimization
            # Reset h with current values and optimize
            h.value = best_h.copy()
            
            # Since this approach is computationally heavy, use a simple heuristic instead:
            # Adjust h based on gradient of correlation
            eps = 1e-5
            grad = np.zeros_like(best_h)
            for i in range(n_points):
                h_perturb = best_h.copy()
                h_perturb[i] += eps
                h_perturb = np.clip(h_perturb, 0, 1)
                # Rescale to sum to n_points/2
                h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
                
                corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
                c5_perturb = np.max(corr_perturb) * dx
                
                grad[i] = (c5_perturb - c5) / eps
            
            # Gradient descent step
            lr = 0.01
            h_new = best_h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            best_h = h_new
            
        except:
            # If optimization fails, fall back to simple gradient descent
            eps = 1e-5
            grad = np.zeros_like(best_h)
            for i in range(n_points):
                h_perturb = best_h.copy()
                h_perturb[i] += eps
                h_perturb = np.clip(h_perturb, 0, 1)
                # Rescale to sum to n_points/2
                h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
                
                corr_perturb = np.correlate(h_perturb, 1 - h_perturb, mode='full')
                c5_perturb = np.max(corr_perturb) * dx
                
                grad[i] = (c5_perturb - c5) / eps
                
            lr = 0.01
            h_new = best_h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            best_h = h_new
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
