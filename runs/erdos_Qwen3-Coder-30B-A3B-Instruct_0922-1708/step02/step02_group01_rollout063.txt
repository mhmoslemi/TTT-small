import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Set up binary search bounds
    C_low = 0.0
    C_high = 0.5
    tolerance = 1e-4
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    def check_feasibility(C):
        # Build constraints for linear programming
        A_eq = []
        b_eq = []
        A_ub = []
        b_ub = []
        
        # Equality constraint: sum(h) * dx = 1
        row = np.ones(n_points) * dx
        A_eq.append(row)
        b_eq.append(1.0)
        
        # Inequality constraints: each overlap should be <= C
        for k in range(n_points):
            # This constraint: sum h[i] * (1 - h[(i+k) % n_points]) * dx <= C
            row = np.zeros(n_points)
            for i in range(n_points):
                row[i] = dx * (1.0 - h[(i+k) % n_points])
            A_ub.append(row)
            b_ub.append(C)
            
        # Bound constraints
        for i in range(n_points):
            # Lower bound
            row = np.zeros(n_points)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(1.0)  # h[i] <= 1
            
            # Upper bound
            row = np.zeros(n_points)
            row[i] = -1
            A_ub.append(row)
            b_ub.append(0.0)  # h[i] >= 0
            
        # Solve
        try:
            c = np.zeros(n_points)  # minimizing 0*sum(h) is fine
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=[(0, 1) for _ in range(n_points)],
                          method='highs')
            if res.success:
                return True, res.x
            else:
                return False, None
        except:
            return False, None
    
    while time.time() - start_time < budget_s - 1 and C_high - C_low > tolerance:
        iteration += 1
        mid = (C_low + C_high) / 2
        
        feasible, new_h = check_feasibility(mid)
        if feasible:
            C_high = mid
            best_h = new_h
            best_c5 = mid
        else:
            C_low = mid
            
        # Every 20 iterations or time near limit
        if iteration % 20 == 0 or time.time() - start_time > budget_s - 5:
            # Recompute exact C5
            correlation = np.correlate(best_h, 1 - best_h, mode='full')
            c5_exact = np.max(correlation) * dx
            if c5_exact < best_c5:
                best_c5 = c5_exact
    
    if best_c5 == float('inf'):
        # Fall back to previous solution
        correlation = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(correlation) * dx
        best_h = h.copy()
    
    # Final check with evaluator
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), best_c5, n_points)
