import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 value for given h"""
    correlation = correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def rescale_h(h, n_points):
    """Rescale h to ensure sum equals n_points/2"""
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = np.clip(h * scale, 0, 1)
    return h

def mutate_individual(h, mutation_rate, n_points):
    """Apply mutation to an individual"""
    h_mut = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, 0.1)
    return rescale_h(h_mut, n_points)

def crossover_individuals(h1, h2, crossover_rate):
    """Perform single-point crossover between two individuals"""
    if np.random.random() > crossover_rate:
        return h1.copy(), h2.copy()
    
    crossover_point = np.random.randint(1, len(h1))
    h1_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h2_new = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    return h1_new, h2_new

def tournament_selection(population, fitness, tournament_size=2):
    """Select individuals using tournament selection"""
    selected = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitness = [fitness[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx].copy())
    return selected

def genetic_algorithm(seed, budget_s, n_points=100):
    np.random.seed(seed)
    
    # Parameters
    N_pop = 50
    C_rate = 0.8
    M_rate = 0.2
    elite_size = N_pop // 2
    dx = 2.0 / n_points
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = rescale_h(h_init, n_points)
        population.append(h_init)
    
    while len(population) < N_pop:
        h = np.random.random(n_points)
        h = rescale_h(h, n_points)
        population.append(h)
    
    # Main loop
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
        
        # Early termination
        if best_c5 < 0.3807:
            break
        
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Crossover and mutation
        new_population = []
        
        # Elitism
        elites = [population[i] for i in np.argsort(fitness)[:elite_size]]
        new_population.extend(elites)
        
        # Generate offspring
        for i in range(0, N_pop - elite_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
            offspring1, offspring2 = crossover_individuals(parent1, parent2, C_rate)
            offspring1 = mutate_individual(offspring1, M_rate, n_points)
            offspring2 = mutate_individual(offspring2, M_rate, n_points)
            new_population.extend([offspring1, offspring2])
            
        # Maintain population size
        population = new_population[:N_pop]
        
        # Stagnation detection and disturbance
        no_improvement_count += 1
        if no_improvement_count > 50:
            # Introduce random individuals
            for i in range(N_pop // 5):
                h = np.random.random(n_points)
                h = rescale_h(h, n_points)
                population[i] = h
            no_improvement_count = 0

    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
