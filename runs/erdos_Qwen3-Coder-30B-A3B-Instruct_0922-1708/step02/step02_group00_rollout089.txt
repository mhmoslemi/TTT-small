import numpy as np
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated annealing parameters
    T0 = 0.1
    tau = 1000
    temperature = T0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and temperature > 1e-6:
        iteration += 1
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Generate candidate solution
        # Add noise with strength proportional to temperature
        noise = np.random.normal(0, temperature, n_points)
        h_candidate = h + noise
        
        # Clip to [0,1] and rescale
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute candidate C5
        correlation_cand = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(correlation_cand) * dx
        
        # Accept with probability based on simulated annealing
        delta_c5 = c5_candidate - c5
        if delta_c5 < 0 or np.random.rand() < math.exp(-delta_c5 / temperature):
            h = h_candidate
            
        # Update temperature
        temperature = T0 / (1 + iteration / tau)
        
    return (best_h.tolist(), best_c5, n_points)
