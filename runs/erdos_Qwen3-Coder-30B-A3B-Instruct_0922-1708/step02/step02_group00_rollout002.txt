import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    num_starts = 5
    max_iter_per_start = 1000
    
    # Generate multiple initializations
    initializations = [h.copy()]
    for i in range(num_starts - 1):
        h_init = h + np.random.normal(0, 0.02, n_points)
        h_init = clip_and_scale(h_init, n_points)
        initializations.append(h_init)
    
    for start_idx, h in enumerate(initializations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        h = clip_and_scale(h, n_points)
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Simple gradient descent with adaptive steps
        for _ in range(max_iter_per_start):
            if time.time() - start_time >= budget_s - 1:
                break
                
            # Compute gradient via finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - c5) / eps
            
            # Update using simple gradient descent
            lr = 0.01
            h_new = h - lr * grad
            h_new = clip_and_scale(h_new, n_points)
            
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    
    return (best_h.tolist(), best_c5, n_points)
