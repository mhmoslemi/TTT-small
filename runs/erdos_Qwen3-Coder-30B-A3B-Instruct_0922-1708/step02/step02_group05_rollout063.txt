import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation to find a good starting point
    if time.time() - start_time < budget_s * 0.6:
        # Create variables for convex optimization
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize max correlation
        # We'll approximate by minimizing a quadratic relaxation
        # This is a simplified approach
        
        # Simple convex relaxation by relaxing h[i]*h[j] terms
        # Check if we can do better with direct convex program
        try:
            # Use a simple box-constrained problem to get a good initialization
            # Let's use a quadratic approximation to the overlap
            # Since this is challenging to express exactly as convex,
            # we'll just solve an easier feasible starting point
            constraints = [
                cp.sum(h_vars) == n_points / 2,  # Integral constraint
                h_vars >= 0,                     # Lower bound
                h_vars <= 1                      # Upper bound
            ]
            
            # Approximate objective: we'll try to make h smooth and balanced
            obj = cp.Minimize(cp.sum_squares(h_vars - 0.5))
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h = np.array(h_vars.value).clip(0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass  # Fall back to original initialization if needed
            
        current_time = time.time() - start_time
        if current_time > budget_s * 0.6:
            # Return early if we've already exceeded time
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            return (h.tolist(), c5, n_points)
    
    # Gradient descent refinement
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate based on gradient magnitude
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final refinement if needed
    if time.time() - start_time < budget_s - 0.5:
        # Perform one final optimization with scipy for better accuracy
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
            return -np.max(correlation) * dx
        
        # Use L-BFGS-B with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Compute initial guess
        h_guess = best_h.copy()
        
        try:
            result = minimize(objective, h_guess, method='L-BFGS-B', bounds=bounds, 
                            options={'maxiter': 100, 'gtol': 1e-6})
            if result.success:
                best_h = np.clip(result.x, 0, 1)
                best_h = best_h * (n_points / 2) / np.sum(best_h)
                correlation = np.correlate(best_h, 1 - best_h, mode='full')
                best_c5 = np.max(correlation) * dx
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
