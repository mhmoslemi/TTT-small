import numpy as np
from scipy.signal import convolve
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Create symmetric initial construction
        h = np.ones(n_points)
        mid = n_points // 2
        h[:mid] = 0.5
        h[mid:] = 1.5
    
    dx = 2.0 / n_points
    
    # Normalize to integral = 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1 and no_improvement < 100:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Make a perturbation
        # Use a combination of small random steps and larger jumps
        h_new = h.copy()
        
        # Small random step
        perturb = np.random.normal(0, 0.01, n_points)
        h_new += perturb
        
        # Occasionally do larger jumps
        if np.random.rand() < 0.15:
            i = np.random.randint(n_points)
            h_new[i] = np.random.rand() * 2
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Occasionally do a flip (more systematic)
        if np.random.rand() < 0.05:
            # Flip small neighborhood
            start_idx = np.random.randint(0, n_points - 5)
            for i in range(start_idx, start_idx + 5):
                h_new[i] = 1.0 - h_new[i]
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            
        h = h_new
    
    # Final check
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (final_h.tolist(), final_c5, n_points)
