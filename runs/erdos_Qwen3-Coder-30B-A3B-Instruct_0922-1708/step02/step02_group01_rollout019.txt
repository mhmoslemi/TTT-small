import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    delta = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            delta = max(1e-4, 0.01 * c5)
            
        if iteration - last_improvement > 50:
            delta *= 0.5
            
        # Find the worst shift
        k_max = np.argmax(correlation)
        k_max = k_max - (len(correlation) - 1) // 2  # Convert to actual shift
        
        # Optimization approach:
        # Try gradient descent with a different schedule and explore starting points
        lr = 0.005 if iteration < 100 else 0.001
        
        # Small perturbation to get gradient
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update with momentum
        momentum = 0.9
        if iteration == 1:
            velocity = -lr * grad
        else:
            velocity = momentum * velocity - lr * grad
            
        h_new = h + velocity
        
        # Clip and rescale to satisfy constraints
        h_new = np.clip(h_new, 0, 1)
        scale = (n_points / 2) / np.sum(h_new)
        h_new = h_new * scale
        
        h = h_new
        
        # Occasionally restart from best solution if no improvement
        if iteration % 200 == 0 and iteration > last_improvement + 100:
            h = best_h.copy()
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            
    return (best_h.tolist(), best_c5, n_points)
