import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent construction or a blocky function
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Create a blocky initial function
        h_init = np.zeros(n_points)
        h_init[20:40] = 1.0
        h_init[60:80] = 1.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)  # Rescale
    
    # Normalize initial vector
    h = h_init.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    # Set up convex optimization
    # Variables
    h_var = Variable(n_points)
    
    # Objective: minimize max_k ∫h(x)(1 - h(x+k))dx = max_k ||h * (1-h)||_1
    # We'll model this differently using a spectral approach
    
    # Since C5 = max_k ∫ h(x) (1 - h(x+k)) dx, 
    # and we know the convolution is involved, we can use a dual formulation.
    
    # The constraint ∫ h dx = 1 becomes sum(h) * dx = 1, so sum(h) = n_points / 2
    # And h in [0,1]
    
    # Instead of gradient descent, we directly solve a constrained problem
    # for a better local minimum
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define problem using CVXPY
    # Minimize over h
    # subject to: sum(h) = n_points/2, 0 <= h[i] <= 1
    
    # For better optimization, we use a semi-smooth approach:
    # Convert to a quadratic optimization problem by using a surrogate function
        
    # Use a different strategy: gradient-based optimization with
    # special handling for constraints.
    
    # Break early if nearly out of time
    while time.time() - start_time < budget_s - 1:
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Estimate gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        # Perturb each coordinate slightly
        for i in range(n_points):
            h_pert = h.copy()
            h_pert[i] += eps
            if h_pert[i] > 1:
                h_pert[i] = 1
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)  # Rescale
            
            corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
            c5_pert = np.max(corr_pert) * dx
            
            grad[i] = (c5_pert - c5) / eps
            
        # Perform step
        lr = 0.005
        h_new = h - lr * grad
        
        # Assert bounds and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept if improvement
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
