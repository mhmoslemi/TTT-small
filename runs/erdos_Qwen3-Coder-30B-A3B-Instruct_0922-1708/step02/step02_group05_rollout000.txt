import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define the objective function to minimize
    def objective(h):
        # Ensure it's within bounds
        h = np.clip(h, 0, 1)
        # Renormalize to meet the integral constraint
        h = h * (n_points / 2) / np.sum(h)
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints function to ensure integral constraint
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Use differential evolution for global optimization
    # Due to time constraints, limit generations to reasonable number
    maxiter = min(500, int(budget_s * 0.1))
    
    # Differential evolution setup
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=maxiter,
        popsize=50,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        atol=1e-6,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    # Final evaluation with best solution from DE
    final_h = np.clip(de_result.x, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = objective(final_h)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h
    
    # Fine-tune using gradient descent on best solution
    h = best_h.copy()
    
    # Perform gradient descent briefly
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate based on gradient magnitude
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update best solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
