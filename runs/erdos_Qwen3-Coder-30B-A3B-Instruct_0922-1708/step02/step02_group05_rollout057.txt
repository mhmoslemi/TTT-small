```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_overlap(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation for initial warm-up
    if time.time() - start_time < budget_s * 0.3:
        try:
            # Formulate and solve convex relaxation
            h_var = Variable(n_points)
            tau = Variable(1)
            
            # Constraints
            constraints = [0 <= h_var, h_var <= 1]
            constraints.append(sum(h_var) == n_points / 2)
            
            # For each shift k, add constraint
            for k in range(n_points):
                # Sum over valid indices for the shift
                if k < n_points:
                    expr = 0
                    for i in range(n_points):
                        if i + k < n_points:
                            expr += h_var[i] * (1 - h_var[i + k])
                        else:
                            break
                    constraints.append(expr <= tau * n_points)
            
            # Objective
            objective = Minimize(tau)
            prob = Problem(objective, constraints)
            
            # Solve with ECOS solver
            prob.solve(solver=ECOS, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_convex = np.array(h_var.value).flatten()
                h_convex = np.clip(h_convex, 0, 1)
                h_convex = h_convex * (n_points / 2) / np.sum(h_convex)
                h = h_convex.copy()
        except:
            pass  # Fallback to initial
    
    # Gradient descent refinement
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update if better solution found
        c5 = evaluate_overlap(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Estimate gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_overlap(h_plus, dx)
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Enforce constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
```
