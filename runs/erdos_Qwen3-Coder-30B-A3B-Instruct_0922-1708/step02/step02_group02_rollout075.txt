import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Use blocky initialization
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    max_iter = 1000
    iteration = 0
    
    # Precompute the correlation
    def compute_c5(h_values):
        correlation = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(correlation) * dx
    
    # Solve using convex optimization
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Use CVXPY to solve dual optimization problem
        h_var = Variable(n_points)
        
        # Objective: minimize max_k ∫ h(x)(1-h(x+k)) dx
        # Equivalent to minimizing C5 formulated as linear constraints
        # This part is critical and uses the structure
        
        # Dual approach: reformulate the constrained optimization problem
        # Let's formulate a simpler approach with gradient descent but 
        # improve it with constrained optimization
        
        # We'll construct a simpler ABSTRACT step:
        # Just improve with small tweaks and look for better local minima
        # This implementation is a second attempt with a sparse pattern
        
        # Perturb small components for local search
        steps = 10
        step_size = 0.005
        
        best_improvement = h.copy()
        min_c5 = current_c5
        
        for _ in range(steps):
            h_perturbed = h.copy()
            i = np.random.randint(0, n_points)
            # Try small perturbations
            perturbation = np.random.uniform(-step_size, step_size)
            h_perturbed[i] += perturbation
            # Keep in bounds
            h_perturbed = np.clip(h_perturbed, 0, 1)
            # Rescale to sum to 50
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            c5_test = compute_c5(h_perturbed)
            if c5_test < min_c5:
                min_c5 = c5_test
                best_improvement = h_perturbed.copy()
        
        # Use the best improvement
        if min_c5 < current_c5:
            h = best_improvement
        else:
            # Do a more global search
            # Create piecewise constant perturbation
            modes = 3
            h_new = np.ones(n_points)
            for k in range(modes):
                start = (k * n_points) // modes
                end = ((k + 1) * n_points) // modes
                h_new[start:end] = np.random.uniform(0.3, 0.7)
            
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            c5_test = compute_c5(h_new)
            if c5_test < current_c5:
                h = h_new

    # Final evaluation  
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
