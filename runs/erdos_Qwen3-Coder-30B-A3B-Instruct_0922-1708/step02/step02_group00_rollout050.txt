import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Multi-start optimization
    num_starts = 5
    for start_idx in range(num_starts):
        # Create variations of the initial solution
        if 'initial_h_values' in globals():
            # Add small random noise for variation
            variation = np.random.normal(0, 0.05, n_points)
            h = np.clip(initial_h_values + variation, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Random initialization with sparse pattern
            h = np.random.rand(n_points) * 0.3
            # Ensure some mass
            h[np.random.choice(n_points, size=max(1, n_points//10), replace=False)] = 0.8
            h = clip_and_scale(h, n_points)
        
        # Adam-like optimization
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.001
        
        for iter_idx in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current C5
            current_c5 = compute_c5(h, dx)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Compute gradient via finite differences
            grad = np.zeros_like(h)
            eps_val = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps_val
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = compute_c5(h_plus, dx)
                grad[i] = (c5_plus - current_c5) / eps_val
                
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**(iter_idx + 1))
            v_hat = v / (1 - beta2**(iter_idx + 1))
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Apply constraints
            h = clip_and_scale(h_new, n_points)
            
    return (best_h.tolist(), best_c5, n_points)
