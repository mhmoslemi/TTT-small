import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = c2 = 0.5
    velocities = np.zeros((n_particles, n_points))
    particles = np.random.random((n_particles, n_points)) * 0.5
    particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
    pbest = particles.copy()
    pbest_c5 = np.full(n_particles, float('inf'))
    
    # Initialize global best
    gbest = h.copy()
    gbest_c5 = float('inf')
    
    def evaluate(h_vec):
        """Evaluate the C5 bound for given h"""
        # Normalize
        h_norm = h_vec * (n_points / 2) / np.sum(h_vec)
        # Correlation
        correlation = np.correlate(h_norm, 1 - h_norm, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Initial evaluation
    for i in range(n_particles):
        c5 = evaluate(particles[i])
        if c5 < pbest_c5[i]:
            pbest_c5[i] = c5
            pbest[i] = particles[i].copy()
        if c5 < gbest_c5:
            gbest_c5 = c5
            gbest = particles[i].copy()
            
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # PSO update
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            particles[i] = particles[i] + velocities[i]
            
            # Clamp and rescale
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Update personal best
            c5 = evaluate(particles[i])
            if c5 < pbest_c5[i]:
                pbest_c5[i] = c5
                pbest[i] = particles[i].copy()
                if c5 < gbest_c5:
                    gbest_c5 = c5
                    gbest = particles[i].copy()
        
        # Periodic KKT refinement (every 100 iterations)
        if iteration % 100 == 0:
            # Start with global best
            h = gbest.copy()
            
            # Refine using gradient descent
            for _ in range(50):
                # Estimate gradient and update
                grad = np.zeros_like(h)
                eps = 1e-6
                for dim in range(len(h)):
                    h_plus = h.copy()
                    h_plus[dim] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c5_plus = evaluate(h_plus)
                    
                    grad[dim] = (c5_plus - evaluate(h)) / eps
                
                # Update with gradient and project back
                learning_rate = 0.01
                h_new = h - learning_rate * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                if abs(evaluate(h_new) - evaluate(h)) < 1e-8:
                    break
                h = h_new
            
            # Update gbest if better
            c5_new = evaluate(h)
            if c5_new < gbest_c5:
                gbest_c5 = c5_new
                gbest = h.copy()
        
        # Early termination
        if iteration % 500 == 0 and abs(best_c5 - gbest_c5) < 1e-6:
            break
            
    # Final evaluation of global best
    final_h = gbest.copy()
    final_c5 = evaluate(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
