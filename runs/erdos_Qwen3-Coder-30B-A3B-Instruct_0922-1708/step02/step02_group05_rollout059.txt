import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation step
    if time.time() - start_time < 0.5 * budget_s:
        # Define the convex relaxation
        h_var = cp.Variable(n_points)
        tau = cp.Variable()
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1]
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Add the moment constraints - we will compute all possible shifts
        # But this is computationally expensive for large n. So just do a few
        for shift in range(0, n_points//4):  # of course this has to be adjusted!
            # The correlation expression for a specific shift is:
            # sum_{i=0}^{n_points-1-shift} h[i] * (1 - h[i+shift])
            # Here we are forming a casual (non-circular) convolution-like constraint
            # But better: construct an appropriate matrix.
            if shift == 0:
                continue  # This is just the integral constraint

            corr_expr = 0
            contrib_expr = 0
            for i in range(n_points - shift):
                contrib_expr += h_var[i] * (1 - h_var[i+shift])
                
            constraints.append(contrib_expr <= tau * (n_points - shift))
            
        objective = cp.Minimize(tau)
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            if problem.status in ["optimal", "optimal_inaccurate"]:
                relaxed_h = h_var.value
                if relaxed_h is not None and np.all(relaxed_h >= 0) and np.all(relaxed_h <= 1):
                    h = np.clip(relaxed_h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
        except:
            pass  # If convex solve fails, fall back to default start
    
    # Gradient descent refinement
    while time.time() - start_time < budget_s - 1:
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Clip and rescale to maintain constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
