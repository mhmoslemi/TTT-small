import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try to perturb with gradient descent
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update h using gradient descent
        lr = 0.005
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Try to improve with SDP relaxation
    try:
        # If time allows, try SDP improvement 
        if time.time() - start_time < budget_s - 5:
            # Create a better starting point using optimization
            def objective(h):
                corr = np.correlate(h, 1 - h, mode='full')
                return -np.max(corr) * dx  # maximize the negative to minimize
            
            # Box constraints
            bounds = [(0, 1) for _ in range(n_points)]
            # Constraint for sum
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            # Run optimization with few iterations
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6})
            
            if result.success:
                h_sdp = np.clip(result.x, 0, 1)
                h_sdp = h_sdp * (n_points / 2) / np.sum(h_sdp)
                
                corr = np.correlate(h_sdp, 1 - h_sdp, mode='full')
                c5_sdp = np.max(corr) * dx
                
                if c5_sdp < best_c5:
                    best_c5 = c5_sdp
                    best_h = h_sdp.copy()
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
