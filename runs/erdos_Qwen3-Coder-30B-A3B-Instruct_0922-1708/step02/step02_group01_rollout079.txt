import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles and velocities
    particles = np.random.rand(n_particles, n_points) * 0.5
    velocities = np.random.rand(n_particles, n_points) * 0.1
    personal_best = particles.copy()
    personal_best_scores = np.full(n_particles, np.inf)
    
    # Set first particle to initial solution if available
    if 'initial_h_values' in globals():
        particles[0] = h.copy()
        personal_best[0] = h.copy()
        
    # Global best tracking
    global_best_idx = 0
    global_best_score = np.inf
    global_best = h.copy()
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            particle = particles[i]
            
            # Project to feasible region
            particle = np.clip(particle, 0, 1)
            scale = (n_points / 2) / np.sum(particle)
            particle = np.clip(particle * scale, 0, 1)
            
            # Compute C5
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_best[i] = particle.copy()
                
            # Update global best
            if c5 < global_best_score:
                global_best_score = c5
                global_best = particle.copy()
                stagnation_count = 0
                
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            
            # Velocity update
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) +
                           c2 * r2 * (global_best - particles[i]))
            
            # Position update
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            scale = (n_points / 2) / np.sum(particles[i])
            particles[i] = np.clip(particles[i] * scale, 0, 1)
            
            # Random restart heuristic
            if stagnation_count > 100 and iteration % 10 == 0:
                particles[i] = np.random.rand(n_points) * 0.5
                particles[i] = np.clip(particles[i], 0, 1)
                scale = (n_points / 2) / np.sum(particles[i])
                particles[i] = np.clip(particles[i] * scale, 0, 1)
                
        stagnation_count += 1
        if global_best_score < 0.3807:
            break
            
    # Final validation and return best solution
    best_h = global_best.copy()
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), best_c5, n_points)
