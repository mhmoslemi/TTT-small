import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Initialize multiple particles
    n_particles = 10
    particles = []
    for i in range(n_particles):
        if i == 0 and 'initial_h_values' in globals():
            h_init = h.copy()
        else:
            h_init = np.random.uniform(0, 1, n_points)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate each particle
        for i, h in enumerate(particles):
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            
            # Compute gradient with finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - c5) / eps
            
            # Update with gradient descent
            h_new = h - lr * grad
            
            # Add noise for exploration
            if np.random.random() < 0.05:
                noise = np.random.normal(0, 0.01, n_points)
                h_new += noise
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            
            # Occasionally adjust learning rate
            if iteration % 100 == 0 and lr > 0.001:
                lr *= 0.95
                
    return (best_h.tolist(), best_c5, n_points)
