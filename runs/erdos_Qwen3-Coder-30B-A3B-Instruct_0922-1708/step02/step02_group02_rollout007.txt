import numpy as np
from scipy.optimize import minimize_scalar
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
    else:
        # Create a binary comb-like structure: 5 pulses
        h = np.zeros(n_points)
        pulse_width = 0.2
        pulse_spacing = 0.25
        for i in range(5):
            start = i * (pulse_width + pulse_spacing)
            end = start + pulse_width
            start_idx = int(start / 2.0 * n_points)
            end_idx = int(end / 2.0 * n_points)
            if end_idx > start_idx:
                h[start_idx:end_idx] = 1.0
                
    # Adjust to make the integral equal to 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute indices
    indices = np.arange(n_points)
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Make a small perturbation
        h_new = h.copy()
        # Randomly choose a point to modify
        i = np.random.randint(0, n_points)
        # Apply a small change
        delta = np.random.uniform(-0.02, 0.02)
        h_new[i] += delta
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
