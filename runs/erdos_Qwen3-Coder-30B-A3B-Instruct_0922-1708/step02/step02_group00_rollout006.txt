import numpy as np
from scipy.signal import correlation
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.001
    
    # Initialize Adam moment estimates
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation_full = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation_full) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the critical shift (max overlap)
        max_idx = np.argmax(correlation_full)
        k_opt = max_idx - (n_points - 1)
        
        # Compute gradient for this critical shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Add to gradient for the term h[i]*(1-h[i+k_opt])
            if i + k_opt >= 0 and i + k_opt < n_points:
                grad[i] += 1 - h[i + k_opt]
            # Add to gradient for the term h[i+k_opt]*(1-h[i])
            if i - k_opt >= 0 and i - k_opt < n_points:
                grad[i] -= h[i - k_opt]
        
        # Apply Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
