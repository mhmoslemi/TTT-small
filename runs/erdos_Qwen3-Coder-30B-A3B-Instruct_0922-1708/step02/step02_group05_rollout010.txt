    # For convex relaxation, we make the constraint linear.

        # Compute the overlap integral
        correlation = np.correlate(h_current, 1 - h_current, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_current.copy()
            
        # Compute gradient: finite difference approximation
        grad = np.zeros_like(h_current)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.5, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.02
            
        h_new = h_current - lr * grad
        
        # Project new h vector back onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_current = h_new
        
        # Check for convergence or time limit
        if time.time() - start_time > time_left:
            break
            
    return (best_h.tolist(), best_c5, n_points)
