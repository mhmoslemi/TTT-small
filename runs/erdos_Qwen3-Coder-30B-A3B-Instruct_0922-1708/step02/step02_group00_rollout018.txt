import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
        for _ in range(population_size - 1):
            h_mut = h + np.random.normal(0, 0.05, n_points)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            population.append(h_mut)
    else:
        # Initialize with random spikes
        for _ in range(population_size):
            h = np.zeros(n_points)
            num_spikes = np.random.randint(5, 11)
            spike_positions = np.random.choice(n_points, num_spikes, replace=False)
            spike_heights = np.random.uniform(0.5, 1.0, num_spikes)
            for pos, height in zip(spike_positions, spike_heights):
                h[pos] = height
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < 1000:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Create new generation
        new_population = [population[0]]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = population[np.random.randint(0, population_size // 2)]
            parent2 = population[np.random.randint(0, population_size // 2)]
            
            # Crossover
            if np.random.rand() < 0.7:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
                
            # Mutation
            mut_rate = 0.15
            for i in range(n_points):
                if np.random.rand() < mut_rate:
                    child1[i] += np.random.normal(0, 0.1)
                    child1[i] = np.clip(child1[i], 0, 1)
                if np.random.rand() < mut_rate:
                    child2[i] += np.random.normal(0, 0.1)
                    child2[i] = np.clip(child2[i], 0, 1)
            
            # Rescale
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        population = new_population[:population_size]
        
    return (best_h.tolist(), best_c5, n_points)
