import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        # Create initial population
        for _ in range(population_size):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Random initialization
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Track best
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection (keep top 20%)
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size//5]
        selected = [population[i] for i in top_indices]
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Crossover
            if len(selected) >= 2 and random.random() < 0.5:
                parent1 = selected[random.randint(0, len(selected)-1)]
                parent2 = selected[random.randint(0, len(selected)-1)]
                # Blend crossover
                alpha = random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                # Mutation
                parent = selected[random.randint(0, len(selected)-1)]
                child = parent.copy()
                # Mutate 20% of elements
                num_mutate = int(0.2 * n_points)
                indices = random.sample(range(n_points), num_mutate)
                for i in indices:
                    child[i] += random.gauss(0, 0.02)
                child = np.clip(child, 0, 1)
            
            # Rescale to maintain sum constraint
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Check for improvement
        if generation % 10 == 0:
            if len(set([round(c, 6) for c in fitness])) == 1:
                no_improvement_count += 1
            else:
                no_improvement_count = 0
    
    # Final evaluation of best solution
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        return (best_h.tolist(), final_c5, n_points)
    else:
        # If no improvement found, use the first of the population
        h = population[0]
        corr = correlate(h, 1 - h, mode='full')
        final_c5 = np.max(corr) * dx
        return (h.tolist(), final_c5, n_points)
