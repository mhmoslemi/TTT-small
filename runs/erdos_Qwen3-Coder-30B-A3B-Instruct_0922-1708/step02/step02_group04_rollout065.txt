from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

    # Normalize to required integral
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = float('inf')
    
    iter_count = 0
    
    # Start optimization loop
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Evaluate current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Try to improve by local perturbations
        h_new = h.copy()
        
        # Pick a random subset of indices to modify
        n_modify = max(1, min(5, n_points // 10))
        indices = np.random.choice(n_points, size=n_modify, replace=False)
        
        for i in indices:
            # Small random change
            change = np.random.normal(0, 0.02)
            h_new[i] += change
            h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Normalize to satisfy constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept the new solution if it's better or with some probability
        new_correlation = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(new_correlation) * dx
        
        if new_c5 < c5 or np.random.rand() < 0.1:
            h = h_new
            
    # Final validation
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
