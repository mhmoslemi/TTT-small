import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    correlation = correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def normalize_h(h, n_points):
    """Normalize h to satisfy the constraint ∫h(x)dx = 1"""
    scale = (n_points / 2) / np.sum(h)
    return np.clip(h * scale, 0, 1)

def crossover(parent1, parent2, crossover_rate):
    """Single-point crossover with given rate"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate, n_points):
    """Uniform mutation with repair"""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.random() < mutation_rate:
            mutated[i] += np.random.normal(0, 0.05)
    return normalize_h(mutated, n_points)

def tournament_selection(population, fitnesses, tournament_size=2):
    """Tournament selection for parent selection"""
    selected = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_index].copy())
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    N_pop = 50
    C_rate = 0.8
    M_rate = 0.2
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    population = []
    for _ in range(N_pop):
        if np.random.random() < 0.3:  # Occasionally use init
            h = h_init.copy()
        else:
            h = np.random.random(n_points)
        h = normalize_h(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    start_time = time.time()
    generation = 0
    stagnation_count = 0
    last_best = best_c5
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and Mutation
        new_population = []
        for i in range(0, N_pop, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < N_pop else selected[0]
            
            child1, child2 = crossover(parent1, parent2, C_rate)
            child1 = mutate(child1, M_rate, n_points)
            child2 = mutate(child2, M_rate, n_points)
            
            new_population.extend([child1, child2])
        
        # Elitism: keep best from previous generation
        population = new_population[:N_pop]
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Replace worst with best if necessary
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Disturbance if stagnation
        if stagnation_count > 50:
            for i in range(int(N_pop * 0.2)):
                random_idx = np.random.randint(0, N_pop)
                population[random_idx] = normalize_h(np.random.random(n_points), n_points)
            stagnation_count = 0
            fitnesses = [evaluate_h(h, dx) for h in population]
            current_best_idx = np.argmin(fitnesses)
            if fitnesses[current_best_idx] < best_c5:
                best_h = population[current_best_idx].copy()
                best_c5 = fitnesses[current_best_idx]
                
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Final validation
    best_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), best_c5, n_points)
