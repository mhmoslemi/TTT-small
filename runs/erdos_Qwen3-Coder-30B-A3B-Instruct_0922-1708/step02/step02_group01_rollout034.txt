import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Multi-start optimization with multiple particles
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Initialize multiple particles
    particles = []
    n_particles = 10
    
    # First particle: initial value
    particles.append(h.copy())
    
    # Rest: random perturbations
    for _ in range(n_particles - 1):
        pert = np.random.uniform(-0.1, 0.1, n_points)
        h_pert = h + pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert.copy())
    
    # Optimization parameters
    learning_rate = 0.01
    momentum = 0.9
    noise_prob = 0.05
    noise_magnitude = 0.02
    
    iteration = 0
    particle_states = [0] * n_particles  # Track last improvement per particle
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update each particle
        for p_idx in range(n_particles):
            h = particles[p_idx]
            
            # Compute current C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Apply momentum and update
            velocity = momentum * (np.random.rand(n_points) - 0.5) - learning_rate * grad
            h_new = h + velocity
            
            # Add small random noise occasionally to escape local minima
            if np.random.random() < noise_prob:
                h_new += np.random.normal(0, noise_magnitude, n_points)
            
            # Clip and rescale to stay feasible
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[p_idx] = h_new
            
            # Occasionally restart particles that haven't improved in a while
            if particle_states[p_idx] > 100:
                particles[p_idx] = np.random.uniform(0, 1, n_points)
                particles[p_idx] = particles[p_idx] * (n_points / 2) / np.sum(particles[p_idx])
                particle_states[p_idx] = 0
            else:
                particle_states[p_idx] += 1
                
    return (best_h.tolist(), best_c5, n_points)
