import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Annealing parameters
    T0 = 0.1
    tau = 1000
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    t = 0
    
    while time.time() - start_time < budget_s - 1:
        t += 1
        
        # Current overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Temperature
        T = T0 / (1 + t / tau)
        
        # Perturb solution
        perturbation = np.random.normal(0, T, n_points)
        h_new = h + perturbation
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # New overlap integral
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept with probability
        delta = c5_new - c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            
    return (best_h.tolist(), best_c5, n_points)
