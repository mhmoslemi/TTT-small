import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    """Ensure h is in [0,1] and sums to n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    """Compute C5 bound from step function h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    n_starts = 5
    n_iterations = 500
    learning_rate = 0.005
    
    for start in range(n_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Different initializations
        if 'initial_h_values' in globals():
            # Add small noise
            noise_level = 0.05
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_orig, h)
            h = h + np.random.normal(0, noise_level, n_points) * h
            h = clip_and_scale(h, n_points)
        else:
            h = np.random.rand(n_points) * 0.5
            h = clip_and_scale(h, n_points)
        
        # Simple gradient descent with adaptive updates
        for iter_idx in range(n_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Approximate gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = compute_c5(h_plus, dx)
                grad[i] = (c5_plus - current_c5) / eps
            
            # Update with learning rate
            h = h - learning_rate * grad
            h = clip_and_scale(h, n_points)
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
