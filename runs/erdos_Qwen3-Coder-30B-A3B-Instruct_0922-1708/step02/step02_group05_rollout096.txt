from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points

    # Apply PSO with adaptive parameters

    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    gbest_position = None
    gbest_value = float('inf')
    
    # Initialize PSO swarm
    for _ in range(num_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            if len(particle) != n_points:
                particle = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(particle)), particle)
            particle = np.clip(particle, 0, 1)
        else:
            particle = np.random.rand(n_points) * 0.5
        
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.random.rand(n_points) * 0.5 - 0.25)
        # Evaluate particle
        correlation = np.correlate(particle, 1 - particle, mode='full')
        value = np.max(correlation) * dx
        pbest_positions.append(particle.copy())
        pbest_values.append(value)
        if value < gbest_value:
            gbest_value = value
            gbest_position = particle.copy()
    
    # PSO parameters
    w_start = 0.7
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    stagnation_counter = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update inertia weight
        w = w_start - (w_start - w_end) * (time.time() - start_time) / budget_s
        
        for i in range(num_particles):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            cognitive = c1 * r1 * (pbest_positions[i] - particles[i])
            social = c2 * r2 * (gbest_position - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            value = np.max(correlation) * dx
            
            # Update personal best
            if value < pbest_values[i]:
                pbest_values[i] = value
                pbest_positions[i] = particles[i].copy()
            
            # Update global best
            if value < gbest_value:
                gbest_value = value
                gbest_position = particles[i].copy()
                stagnation_counter = 0
            else:
                stagnation_counter += 1
                
        # Check for stagnation and perturb if needed
        if stagnation_counter > max_stagnation:
            # Perturb 20% of the swarm
            for i in range(int(0.2 * num_particles)):
                idx = np.random.randint(0, num_particles)
                particles[idx] = np.random.rand(n_points) * 0.5
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.rand(n_points) * 0.5 - 0.25
                stagnation_counter = 0
        
        # Early termination if improvement is sustained
        if gbest_value <= 0.3807:
            break
    
    # Final evaluation of the best solution
    correlation = np.correlate(gbest_position, 1 - gbest_position, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (gbest_position.tolist(), final_c5, n_points)
