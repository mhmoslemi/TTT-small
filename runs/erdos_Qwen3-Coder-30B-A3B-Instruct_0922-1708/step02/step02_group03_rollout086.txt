import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use a more sophisticated optimization method
    def objective(h):
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Use trust-constr for better convergence
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
    ]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with restarts
    max_restarts = 5
    restart_count = 0
    
    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        iteration += 1
        
        # Compute current C5
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Try a different initialization if we're stuck
        if iteration % 50 == 0 and current_c5 < 0.4:
            # Perturb the solution slightly
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Optimize using scipy
        try:
            # Use trust-constr for better convergence
            res = minimize(
                objective,
                h,
                method='trust-constr',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'verbose': 0}
            )
            
            if res.success:
                h = res.x
                # Renormalize
                h = h * (n_points / 2) / np.sum(h)
            else:
                # If optimization fails, perturb and continue
                h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
        except Exception:
            # Fallback to simple gradient-based update
            eps = 1e-6
            grad = np.zeros_like(h)
            current_c5 = objective(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = objective(h_plus)
                grad[i] = (c5_plus - current_c5) / eps
                
            # Simple gradient update
            h = h - 0.002 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Occasionally check for termination
        if iteration % 100 == 0:
            if time.time() - start_time > budget_s - 5:
                break
                
        # Occasionally try different initialization
        if iteration % 200 == 0:
            # Try a random restart
            h = np.random.rand(n_points) * 0.5
            h = h * (n_points / 2) / np.sum(h)
            restart_count += 1
            
    # Final check
    final_c5 = objective(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
