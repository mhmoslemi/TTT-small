import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Compute C5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def rescale_h(h, n_points):
    """Ensure sum(h) = n_points / 2"""
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return np.clip(h, 0, 1)

def mutate_h(h, rate=0.15, noise_range=0.2):
    """Mutate a step function"""
    h_new = h.copy()
    n_points = len(h)
    num_mutate = max(1, int(n_points * rate))
    indices = np.random.choice(n_points, num_mutate, replace=False)
    for i in indices:
        h_new[i] += np.random.uniform(-noise_range, noise_range)
    return rescale_h(h_new, n_points)

def crossover_h(h1, h2):
    """Single-point crossover"""
    n_points = len(h1)
    split = np.random.randint(1, n_points)
    h_new1 = np.concatenate([h1[:split], h2[split:]])
    h_new2 = np.concatenate([h2[:split], h1[split:]])
    return h_new1, h_new2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.15
    crossover_rate = 0.7
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = rescale_h(h_init, n_points)
        population.append(h_init)
        for _ in range(population_size - 1):
            population.append(mutate_h(h_init, mutation_rate))
    else:
        # Random spike initializations
        for _ in range(population_size):
            h = np.zeros(n_points)
            num_spikes = np.random.randint(3, 15)
            for _ in range(num_spikes):
                pos = np.random.randint(0, n_points)
                h[pos] = np.random.uniform(0.5, 1.0)
            h = rescale_h(h, n_points)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    gen = 0
    
    while time.time() - start_time < budget_s - 1 and gen < 1000:
        gen += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Create new population
        new_population = population[:population_size//2]  # Elitism
        
        while len(new_population) < population_size:
            # Selection (tournament)
            def tournament_selection(k=3):
                idxs = np.random.choice(len(population), k)
                best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
                return population[best_idx]
            
            if np.random.random() < crossover_rate and len(new_population) >= 2:
                p1 = tournament_selection()
                p2 = tournament_selection()
                o1, o2 = crossover_h(p1, p2)
                new_population.extend([o1, o2])
            else:
                parent = tournament_selection()
                child = mutate_h(parent, mutation_rate)
                new_population.append(child)
                
        population = new_population[:population_size]
    
    return (best_h.tolist(), best_c5, n_points)
