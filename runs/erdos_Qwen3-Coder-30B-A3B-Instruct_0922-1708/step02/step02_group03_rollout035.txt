import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use a more sophisticated optimization approach: L-BFGS with constraints
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute cross-correlation for all shifts
        correlation = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(correlation)
        return max_corr * dx
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Optimization parameters
    options = {'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current solution
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Create a constraint for the sum
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Use scipy.optimize.minimize with L-BFGS-B solver
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, options=options)
            if result.success:
                h = result.x
                # Rescale to match sum constraint
                h = h * (n_points / 2) / np.sum(h)
            else:
                # If optimization fails, perturb slightly and try again
                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback: just perturb and rescale
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally check for early termination
        if iteration % 50 == 0 and iteration > 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
