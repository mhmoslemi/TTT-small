import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use a more robust optimization approach
    # Try multiple random restarts to avoid local minima
    n_restarts = 10
    for restart in range(n_restarts):
        # Random initialization for this restart
        if restart == 0:
            current_h = h.copy()
        else:
            # Perturb the current best solution a bit
            current_h = best_h + np.random.normal(0, 0.05, n_points)
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            
        # Use L-BFGS-B for local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_vec):
            # Rescale to satisfy sum constraint
            h_rescaled = h_vec * (n_points / 2) / np.sum(h_vec)
            correlation = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
            return -np.max(correlation) * dx
        
        def grad(h_vec):
            h_rescaled = h_vec * (n_points / 2) / np.sum(h_vec)
            correlation = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
            max_k = np.argmax(correlation)
            
            # Compute gradient for the max correlation
            grad_vec = np.zeros_like(h_vec)
            eps = 1e-6
            
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                grad_vec[i] = (corr_plus[max_k] - correlation[max_k]) / eps
                
            return -grad_vec * dx
            
        try:
            # Minimization with bounds
            result = minimize(objective, current_h, method='L-BFGS-B', 
                             bounds=bounds, jac=grad, options={'maxiter': 500})
            if result.success:
                final_h = result.x * (n_points / 2) / np.sum(result.x)
                correlation = np.correlate(final_h, 1 - final_h, mode='full')
                c5 = -objective(result.x)  # Convert back to positive
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = final_h.copy()
        except:
            pass
            
        if time.time() - start_time > budget_s - 10:
            break
    
    # Final verification
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
