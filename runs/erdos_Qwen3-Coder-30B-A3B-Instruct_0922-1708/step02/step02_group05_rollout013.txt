import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Stage 1: Convex relaxation to get a good initialization
    if time.time() - start_time < budget_s * 0.6:
        # Use cvxpy for convex relaxation
        h_cvx = Variable(n_points)
        constraints = [h_cvx >= 0, h_cvx <= 1, sum(h_cvx) == n_points / 2]
        
        # We approximate the objective as a linear function
        # This is a heuristic approximation, not the actual objective
        objective = Minimize(sum(h_cvx))  # Dummy objective for convex solve
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                h_init = np.array(h_cvx.value).reshape(-1)
                h_init = np.clip(h_init, 0, 1)
                h_init = h_init * (n_points / 2) / np.sum(h_init)
                h = h_init.copy()
        except:
            pass
    
    # Stage 2: Refinement with projected gradient descent
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adaptive learning rate based on gradient magnitude
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step with adaptive learning rate
        h_new = h - lr * grad
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
