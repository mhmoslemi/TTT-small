def run(seed=42, budget_s=600, **kwargs):

        h[:n_points//2] = 1.0

    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')

    iteration = 0

        iteration += 1
        
        # Compute the current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive step size based on progress
        if iteration < 50:
            step_size = 0.02
        else:
            step_size = 0.01
        
        # Create perturbation
        perturb = np.random.normal(0, step_size, n_points)
        
        # Apply perturbation and keep within bounds
        h_new = h + perturb
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
