import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    current_time = start_time
    
    # First, try convex relaxation to get a good starting point
    if current_time - start_time < budget_s * 0.6:
        try:
            # Formulate the convex relaxation
            h_var = cp.Variable(n_points)
            tau = cp.Variable()
            
            # Constraints for convex relaxation
            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
            
            # Add correlation constraints for all shifts
            for k in range(n_points):
                # Compute correlation for shift k (convolution-like)
                corr = 0
                for i in range(n_points):
                    j = i + k
                    if j < n_points:
                        corr += h_var[i] * (1 - h_var[j])
                constraints.append(corr <= tau * n_points)
            
            # Minimize tau
            objective = cp.Minimize(tau)
            prob = cp.Problem(objective, constraints)
            
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_convex = np.array(h_var.value).reshape(-1)
                h_convex = np.clip(h_convex, 0, 1)
                h_convex = h_convex * (n_points / 2) / np.sum(h_convex)
                h = h_convex.copy()
        except:
            pass  # Fall back to uniform initialization if convex solver fails
        
        current_time = time.time()
    
    # Then do gradient descent refinement with adaptive learning rate
    iteration = 0
    while current_time - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute finite difference gradient with adaptive epsilon
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Ensure feasibility
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Clip and rescale to ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        current_time = time.time()
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
