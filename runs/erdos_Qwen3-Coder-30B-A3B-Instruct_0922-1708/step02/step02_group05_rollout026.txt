import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for given h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential Evolution with individual constraints
    def objective(h):
        # Ensure h is within bounds and satisfies the sum constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Run differential evolution
    # Sweep through different parameter settings to find best
    de_results = []
    
    # Perform multiple runs with different seeds and parameters
    for i in range(5):
        # Use a subset of population for efficiency
        result = differential_evolution(
            objective,
            bounds,
            seed=seed + i,
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1.0),
            recombination=0.7,
            constraints=cons
        )
        
        if not result.success:
            continue
        
        # Ensure constraints are met
        final_h = np.clip(result.x, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        c5 = evaluate_c5(final_h, dx)
        
        de_results.append((final_h, c5))
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = final_h.copy()
    
    # If DE did not improve on initial solution, use gradient-based method for refinement
    if best_c5 >= 0.381:
        # Refinement of the best solution found
        current_h = best_h.copy()
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Small random perturbations
            direction = np.random.normal(0, 0.05, n_points)
            new_h = current_h + direction
            
            # Project back to valid space
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Keep better solution
            c5_new = evaluate_c5(new_h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = new_h.copy()
                current_h = new_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
