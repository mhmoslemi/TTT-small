import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    swarm_size = 50
    w_start = 0.9
    w_end = 0.4
    c1 = c2 = 2.0
    
    # Initialize particles
    particles = [h.copy() for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.5, 0.5, n_points) for _ in range(swarm_size)]
    
    # Personal bests
    pbest_positions = [p.copy() for p in particles]
    pbest_c5 = [float('inf')] * swarm_size
    
    # Global best
    gbest_position = h.copy()
    gbest_c5 = float('inf')
    
    # Track stagnation
    stagnation_count = 0
    stagnation_threshold = 100
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Initial evaluation
    for i in range(swarm_size):
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        c5 = compute_c5(particles[i])
        if c5 < pbest_c5[i]:
            pbest_c5[i] = c5
            pbest_positions[i] = particles[i].copy()
        if c5 < gbest_c5:
            gbest_c5 = c5
            gbest_position = particles[i].copy()
    
    # PSO main loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / (budget_s / 10)
        
        # Update particles
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) +
                           c2 * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            c5 = compute_c5(particles[i])
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest_c5[i] = c5
                pbest_positions[i] = particles[i].copy()
                
            # Update global best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_position = particles[i].copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Stagnation handling
        if stagnation_count > stagnation_threshold:
            # Perturb some particles
            perturb_count = swarm_size // 10
            for i in range(perturb_count):
                idx = np.random.randint(0, swarm_size)
                particles[idx] = np.random.rand(n_points) * 0.5 + 0.25
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                c5 = compute_c5(particles[idx])
                pbest_c5[idx] = c5
                pbest_positions[idx] = particles[idx].copy()
                if c5 < gbest_c5:
                    gbest_c5 = c5
                    gbest_position = particles[idx].copy()
            stagnation_count = 0
            
        # Early termination if target achieved
        if gbest_c5 <= 0.3807:
            break    

    # Final evaluation of best solution
    final_c5 = compute_c5(gbest_position)
    
    return (gbest_position.tolist(), final_c5, n_points)
