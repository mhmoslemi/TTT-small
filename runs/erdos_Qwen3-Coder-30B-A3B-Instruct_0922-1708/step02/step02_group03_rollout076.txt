import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    learning_rate = 0.001
    
    # Initialize Adam moment estimates
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral (max over all lags)
        correlation = correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the active lag k corresponding to maximum correlation
        max_idx = np.argmax(correlation)
        # Determine k from the correlation index
        k = max_idx - (n_points - 1)
        
        # Compute exact gradient w.r.t. h for the maximum lag
        # The gradient for element h[i] when lag k is active is:
        # dC5/dh[i] = (1-h[i-k]) if i-k is valid and i-k>=0 else 0
        #           - h[i+k] if i+k is valid and i+k<n_points else 0
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i - k >= 0:
                grad[i] += -1.0
            if i + k < n_points:
                grad[i] += -1.0
                
        # Update with Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration+1))
        v_hat = v / (1 - beta2**(iteration+1))
        
        h_new = h - learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
