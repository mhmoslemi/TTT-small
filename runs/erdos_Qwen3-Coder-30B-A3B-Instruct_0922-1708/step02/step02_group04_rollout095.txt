import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    correlation = correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def normalize_h(h, n_points):
    """Ensure integral constraint and bounds"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def genetic_algorithm(seed, budget_s, n_points=45):
    np.random.seed(seed)
    
    # Initialize population
    pop_size = min(50, max(10, n_points // 2))
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = normalize_h(h_start, n_points)
        population.append(h_start)
    
    # Fill rest with random solutions
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = normalize_h(h, n_points)
        population.append(h)
    
    # Parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2
    max_no_improvement = 100
    no_improvement_count = 0
    
    best_h = None
    best_c5 = float('inf')
    dx = 2.0 / n_points
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        generation += 1
        
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Track best
        if best_h is not None:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Selection (roulette wheel)
        fitnesses = np.array(fitnesses)
        # Avoid negative or zero fitness
        fitnesses = np.maximum(fitnesses, 1e-10)
        # Convert to selection probability (lower is better)
        probs = 1.0 / (fitnesses + 1e-15)
        probs = probs / np.sum(probs)
        
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            # Select parents
            parent1_idx, parent2_idx = np.random.choice(pop_size, size=2, p=probs)
            parent1, parent2 = population[parent1_idx], population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Clone one parent
                child = parent1.copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                # Add Gaussian noise to some genes
                mutation_mask = np.random.rand(n_points) < mutation_rate
                noise = np.random.normal(0, 0.05, n_points)
                child += noise * mutation_mask
                child = normalize_h(child, n_points)
                
            new_population.append(child)
            
        population = new_population[:pop_size]
    
    # Final validation
    if best_h is not None:
        best_h = normalize_h(best_h, n_points)
        best_c5 = evaluate_h(best_h, dx)
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    # Use small number of points for speed
    n_points = min(200, max(50, 45))
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(seed, budget_s, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
