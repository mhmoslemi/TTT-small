import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a binary comb-like structure
        h = np.zeros(n_points)
        m = 5  # Number of pulses
        pulse_width = 0.2
        spacing = 0.25
        
        # Place m pulses at carefully chosen positions
        for i in range(m):
            start = i * (pulse_width + spacing)
            end = start + pulse_width
            start_idx = int(start / 2.0 * n_points)
            end_idx = int(end / 2.0 * n_points)
            if start_idx < n_points:
                h[start_idx:min(end_idx, n_points)] = 1.0
                
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Helper function to compute C5
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Slightly perturb h
        h_new = h.copy()
        # Select random indices to modify
        indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
        for i in indices:
            # Small random perturbation
            delta = np.random.normal(0, 0.01)
            h_new[i] = np.clip(h_new[i] + delta, 0, 1)
        
        # Rescale to meet constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if it's an improvement
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5:
            h = h_new
        else:
            # Occasionally accept worse solutions to escape local minima
            if np.random.random() < 0.05:
                h = h_new
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
