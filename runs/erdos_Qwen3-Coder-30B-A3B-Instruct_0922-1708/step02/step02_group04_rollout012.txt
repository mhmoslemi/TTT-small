import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    max_iter_no_improvement = 50
    iter_no_improvement = 0
    learning_rate = 0.01
    max_iter_per_grid = 100  # Reduced for faster convergence
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Coarse-to-fine optimization approach
    grid_sizes = [20, 40, 80, 160]
    current_grid_idx = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        current_grid = grid_sizes[current_grid_idx] if current_grid_idx < len(grid_sizes) else n_points
        
        # Interpolate to current grid size if needed
        if current_grid != n_points:
            new_h = np.zeros(current_grid)
            for i in range(current_grid):
                idx = int(i * n_points / current_grid)
                new_h[i] = h[min(idx, n_points-1)]
            h = new_h
            n_points = current_grid
        
        # Gradient-based optimization on current grid
        if current_grid <= 200:
            for _ in range(max_iter_per_grid):
                if time.time() - start_time > budget_s - 1:
                    break

                # Compute overlap values
                overlaps = np.zeros(2 * n_points - 1)
                for k in range(2 * n_points - 1):
                    overlap_sum = 0
                    for i in range(n_points):
                        j = (i - k + n_points) % n_points
                        overlap_sum += h[i] * (1 - h[j])
                    overlaps[k] = overlap_sum
                
                # Compute gradient (simplified gradient approximation)
                gradient = np.zeros(n_points)
                for i in range(n_points):
                    for k in range(2 * n_points - 1):
                        # Approximate gradient contribution per k
                        j = (i - k + n_points) % n_points
                        if k < n_points:
                            gradient[i] += (1 - h[j])
                        else:
                            gradient[i] += -h[j]
                
                # Update step
                h_new = h - learning_rate * gradient
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                h = h_new
                
                # Evaluate and track best
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    iter_no_improvement = 0
                else:
                    iter_no_improvement += 1
                    if iter_no_improvement > max_iter_no_improvement:
                        break

        # Move to next grid
        current_grid_idx = min(current_grid_idx + 1, len(grid_sizes) - 1)
        
        # Occasionally do random exploration
        if np.random.rand() < 0.1:
            # Random perturbation
            perturb = np.random.normal(0, 0.01, n_points)
            h_new = h + perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
