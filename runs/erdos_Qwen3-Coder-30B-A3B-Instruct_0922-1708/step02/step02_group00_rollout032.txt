import numpy as np
from scipy.signal import correlate
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.001
    
    # Initialize Adam variables
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute full correlation
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_corr = corr[k_max]
        
        # Objective value for current h
        c5 = max_corr * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient w.r.t. max overlap shift
        grad = np.zeros_like(h)
        
        # k_max is the index in full correlation; shift is k_max - (n-1)
        shift = k_max - (n_points - 1)
        
        # For shift k, we have h[i] contributes to h[i+k] in the correlation.
        # The derivative of h[i] with respect to the correlation at shift k
        # is (1 - h[i+k]) if i+k < n else 0.
        # Also, -h[i-k] contributes to ln(1-h[i-k]) in the full correlation.
        
        if shift >= 0:
            # h[i] contributes to the correlation at i+shift with factor (1-h[i+shift])
            # So ∂/∂h[i] of (h[i] * (1-h[i+shift])) = 1 - h[i+shift]
            for i in range(n_points - shift):
                grad[i] += 1 - h[i + shift]
        else:
            # h[i] contributes to the correlation at i-shift with factor -h[i-shift]
            # But since we're computing h * (1-h), it's more complex
            shifted_indices = np.arange(-shift, n_points)
            for i in range(n_points):
                if i - shift >= 0:
                    grad[i] += -h[i - shift]
                    
        # Further simplify: only valid contribs from diagonals
        # Analyze the correlogram to correctly identify gradient contributions
        grad = np.zeros_like(h)
        for i in range(n_points):
            # For the current shift, compute derivative carefully
            # At each i, we part contribute to the shift (i+k) term
            # Gradient is -1 if that term has unit difference in h_i
            # Exactly one entry in the correlation is influenced by h[i]
            # Formulate indirect relationship manually
            
            # Manually construct gradient for k_max shift
            # If k_max = i + shift, then for index i it's (1 - h[i+shift]) 
            # If k_max = shift - i, then for index i it's - h[shift-i]
            
            if shift >= 0:  # this is a positive lag
                if i + shift < n_points:
                    grad[i] = 1 - h[i + shift] 
            else:  # this is a negative lag
                if i - shift >= 0:
                    grad[i] = -h[i - shift]
                    
        # Update with Adam 
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        
        # Bias correction
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        
        # Update parameters
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodic perturbation
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
