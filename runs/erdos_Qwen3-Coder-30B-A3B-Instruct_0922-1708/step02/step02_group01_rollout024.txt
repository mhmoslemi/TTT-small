import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h"""
    correlation = correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def initialize_population(n_points, n_pop, initial_h=None):
    """Initialize population"""
    population = []
    
    if initial_h is not None:
        h = np.array(initial_h)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h.copy())
    
    # Fill with random individuals
    while len(population) < n_pop:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h.copy())
        
    return population

def mutate_individual(h, mutation_rate, dx):
    """Mutate individual with repair"""
    h_new = h.copy()
    
    for i in range(len(h_new)):
        if np.random.rand() < mutation_rate:
            h_new[i] += np.random.normal(0, 0.1)
    
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    
    # Rescale to maintain constraint
    scale = (len(h_new) / 2) / np.sum(h_new)
    h_new = np.clip(h_new * scale, 0, 1)
    
    return h_new

def crossover(parent1, parent2, crossover_rate):
    """Single point crossover"""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
        
    crossover_point = np.random.randint(1, len(parent1) - 1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def tournament_selection(population, fitness, tournament_size=2):
    """Select individual via tournament"""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    n_pop = 50
    c_rate = 0.8
    m_rate = 0.2
    elite_size = n_pop // 2
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = initialize_population(n_points, n_pop, initial_h_values)
    else:
        population = initialize_population(n_points, n_pop)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
        
        # Early exit if target achieved
        if best_c5 < 0.3807:
            break
            
        no_improvement_count += 1
        if no_improvement_count > max_no_improvement:
            # Perturb the population
            for i in range(int(0.2 * n_pop)):
                idx = np.random.randint(0, n_pop)
                population[idx] = np.random.rand(n_points)
                population[idx] = population[idx] / np.sum(population[idx]) * (n_points / 2)
            no_improvement_count = 0
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Create new population
        new_population = elite.copy()
        
        # Generate offspring
        while len(new_population) < n_pop:
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            child1, child2 = crossover(parent1, parent2, c_rate)
            
            child1 = mutate_individual(child1, m_rate, dx)
            child2 = mutate_individual(child2, m_rate, dx)
            
            new_population.extend([child1, child2])
            
        # Truncate to population size
        population = new_population[:n_pop]
        
    return (best_h.tolist(), best_c5, n_points)
