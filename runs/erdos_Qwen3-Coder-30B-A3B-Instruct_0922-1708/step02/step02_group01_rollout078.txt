import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    n_particles = 50
    dx = 2.0 / n_points
    max_iterations = int(500000)
    
    # Initialize particles
    particles = np.random.rand(n_particles, n_points) * 0.5
    velocities = np.zeros_like(particles)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles[0] = h_init
    
    # Initialize best solutions
    personal_best = particles.copy()
    personal_best_c5 = np.full(n_particles, float('inf'))
    global_best = particles[0].copy()
    global_best_c5 = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    inertia = w
    cognitive = c1
    social = c2
    
    start_time = time.time()
    iteration = 0
    stagnation_counter = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Project to feasible region
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            particles[i] = h
            
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best_c5[i] = c5
                personal_best[i] = h.copy()
            
            # Update global best
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best = h.copy()
                last_improvement = iteration
        
        # Check for stagnation
        if iteration - last_improvement > 100:
            # Reset velocities for all particles
            velocities = np.random.rand(n_particles, n_points) * 0.1 - 0.05
            
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (personal_best[i] - particles[i]) + 
                           social * r2 * (global_best - particles[i]))
            
            particles[i] = particles[i] + velocities[i]
        
        # Early stopping condition
        if global_best_c5 <= 0.3807:
            break
            
    # Final validation and return
    final_h = np.clip(global_best, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (final_h.tolist(), final_c5, n_points)
