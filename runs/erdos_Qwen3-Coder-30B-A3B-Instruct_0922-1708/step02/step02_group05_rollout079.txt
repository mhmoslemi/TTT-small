        h = h * (n_points / 2) / np.sum(h)

        h = h * (n_points / 2) / np.sum(h)

        return np.sum(h) - n_points / 2

    # Objective function to minimize C5
    def objective(h):
        # Rescale h to satisfy sum constraint
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Define constraints for scipy.optimize
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # We will instead implement a custom differential evolution approach
    # since scipy.optimize.differential_evolution might not be suitable for
    # this specific non-convex problem with exact sum constraint.
    
    # DE parameters
    population_size = 50
    max_generations = 1000
    mutation_factor = 0.8
    crossover_rate = 0.7
    
    # Initialize population
    def initialize_population(size, n_points):
        population = []
        for _ in range(size):
            individual = np.random.rand(n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
            population.append(individual)
        return population
    
    def evaluate_individual(individual):
        # Ensure it's valid
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        correlation = np.correlate(individual, 1 - individual, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Mutation
    def mutate(population, target_idx, F):
        a, b, c = population[np.random.choice(len(population), 3, replace=False)]
        mutant = a + F * (b - c)
        return mutant
    
    # Crossover
    def crossover(target, mutant, CR):
        mask = np.random.rand(len(target)) < CR
        trial = np.where(mask, mutant, target)
        return trial
    
    # Selection
    def select(population, trial, fitness_evals, idx):
        new_fitness = evaluate_individual(trial)
        if new_fitness < fitness_evals[idx]:
            return trial, new_fitness
        else:
            return population[idx], fitness_evals[idx]
    
    # DE evolution
    start_time = time.time()
    best_fitness = float('inf')
    best_individual = h.copy()
    
    population = initialize_population(population_size, n_points)
    fitness_evals = [evaluate_individual(ind) for ind in population]
    
    generation = 0
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        for i in range(population_size):
            # Mutation
            mutant = mutate(population, i, mutation_factor)
            # Clip to [0, 1]
            mutant = np.clip(mutant, 0, 1)
            # Renormalize
            mutant = mutant * (n_points / 2) / np.sum(mutant)
            # Crossover
            trial = crossover(population[i], mutant, crossover_rate)
            # Selection
            population[i], fitness_evals[i] = select(population, trial, fitness_evals, i)
            
            # Track best
            if fitness_evals[i] < best_fitness:
                best_fitness = fitness_evals[i]
                best_individual = population[i].copy()
        
        generation += 1
    
    # Return best solution found
    return (best_individual.tolist(), best_fitness, n_points)
