import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.random.rand(n_particles, n_points) * 0.5
    velocities = np.random.randn(n_particles, n_points) * 0.01
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        particles[0] = h_init / np.sum(h_init) * (n_points / 2)
    
    # Initialize personal best and global best
    personal_best = particles.copy()
    personal_best_fitness = np.full(n_particles, np.inf)
    global_best_idx = 0
    global_best_fitness = np.inf
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    stagnation_count = 0
    stagnation_max = 100
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness for all particles
        for i in range(n_particles):
            h = particles[i]
            # Clip and rescale to ensure constraint satisfaction
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            particles[i] = h
            
            # Compute correlation
            correlation = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(correlation) * dx
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best[i] = h.copy()
            
            # Update global best
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best_idx = i
                stagnation_count = 0
                
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            
            particles[i] += velocities[i]
            
        # Restart stagnated particles
        stagnation_count += 1
        if stagnation_count > stagnation_max:
            for i in range(n_particles):
                if np.allclose(personal_best_fitness[i], global_best_fitness, rtol=1e-6):
                    particles[i] = np.random.rand(n_points) * 0.5
                    velocities[i] = np.random.randn(n_points) * 0.01
            stagnation_count = 0
            
        # Early termination
        if global_best_fitness < 0.3807:
            break
            
    # Return best solution
    best_h = personal_best[global_best_idx]
    best_h = np.clip(best_h, 0, 1)
    scale = (n_points / 2) / np.sum(best_h)
    best_h = best_h * scale
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
