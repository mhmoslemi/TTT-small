import numpy as np
from scipy.optimize import differential_evolution
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with gradient-free method to avoid local minima
    def objective(h_vals):
        h = np.array(h_vals)
        # Rescale to correct integral
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # The key idea of this variation: use global optimization to escape local optima
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=1000,
            popsize=15,
            seed=seed,
            disp=False
        )
        h_opt = result.x
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
        c5_val = np.max(correlation) * dx
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
    except Exception as e:
        pass
    
    # Gradient descent refinement
    def compute_gradient(h_vals):
        h = np.array(h_vals)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        return grad
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient and update
        grad = compute_gradient(h)
        lr = 0.001 * (1.0 - (time.time() - start_time) / budget_s)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
