import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution for better accuracy
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.random.rand(n_particles, n_points)
    
    # If initial construction exists, use it
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        if len(h_initial) != n_points:
            x_orig = np.linspace(0, 2, len(h_initial))
            x_new = np.linspace(0, 2, n_points)
            h_initial = np.interp(x_new, x_orig, h_initial)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        particles[0] = h_initial
    
    # Normalize particles to sum to n_points/2
    for i in range(n_particles):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Initialize velocities
    velocities = np.zeros_like(particles)
    
    # Initialize personal bests
    personal_best = particles.copy()
    personal_best_values = np.full(n_particles, float('inf'))
    
    # Initialize global best
    global_best_idx = 0
    global_best_value = float('inf')
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Compute correlation (overlap integral)
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_values[i]:
                personal_best_values[i] = c5
                personal_best[i] = h.copy()
            
            # Update global best
            if c5 < global_best_value:
                global_best_value = c5
                global_best_idx = i
                stagnation_count = 0  # Reset stagnation
        
        # Early stopping if we found a good solution
        if global_best_value < 0.3807:
            break
            
        stagnation_count += 1
        if stagnation_count > 100:
            # Restart velocities
            velocities = np.random.rand(n_particles, n_points) * 0.1
            
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best[i] - particles[i]) +
                c2 * r2 * (personal_best[global_best_idx] - particles[i])
            )
            
            particles[i] += velocities[i]
            
            # Project onto feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            scale = (n_points / 2) / np.sum(particles[i])
            particles[i] = particles[i] * scale
        
        # Occasionally rescale particles to maintain diversity
        if iteration % 20 == 0:
            for i in range(n_particles):
                if np.random.rand() < 0.1:
                    particles[i] = np.random.rand(n_points)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Return best solution
    best_h = personal_best[global_best_idx]
    return (best_h.tolist(), global_best_value, n_points)
