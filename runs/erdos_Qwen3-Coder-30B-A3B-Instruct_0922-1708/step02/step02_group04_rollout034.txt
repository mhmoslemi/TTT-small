    best_c5 = float('inf')

        # Check current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx

            best_c5 = current_c5
            best_h = h.copy()

        perturb = np.random.normal(0, 0.01, n_points)
