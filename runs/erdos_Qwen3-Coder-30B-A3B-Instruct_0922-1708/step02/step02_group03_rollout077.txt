import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.001
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(correlation)
        k_active = max_corr_idx - (n_points - 1)
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute exact gradient with respect to active lag
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Compute gradient contribution for h[i] with respect to maximizer k
            # d/dh[i] (1 - h[i-k] * h[i]) = -h[i-k] if k is valid, same for shifted index
            if k_active >= 0:
                # h[i-k] term contributes to the correlation
                j = i - k_active
                if 0 <= j < n_points:
                    grad[i] -= h[j]
                # h[i+k] term comes from (1-h[x]) and is irrelevant in this gradient, 
                # but we can still include it if valid.
            else:
                j = i + abs(k_active)
                if 0 <= j < n_points:
                    grad[i] -= h[j]
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        
        # Bias correction
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        
        # Update h
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Occasionally check for early termination
        if iteration % 100 == 0 and iteration > 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
