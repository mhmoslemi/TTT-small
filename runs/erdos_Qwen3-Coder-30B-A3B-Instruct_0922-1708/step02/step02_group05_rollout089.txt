import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Convex relaxation approach
    start_time = time.time()
    
    # Create a convex relaxation by introducing auxiliary variables
    # This helps in better initialization
    try:
        # Formulate a simpler convex program to initialize
        h_init = h.copy()
        
        # Solve a relaxed version as initialization
        # This is a rough approximation to avoid local minima
        h_relaxed = np.zeros(n_points)
        mid_idx = n_points // 2
        h_relaxed[mid_idx-10:mid_idx+10] = 1.0
        h_relaxed = h_relaxed * (n_points / 2) / np.sum(h_relaxed)
        h = h_relaxed.copy()
        
    except:
        pass
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient descent with projection
    max_iters = 10000
    current_time = time.time()
    
    for i in range(max_iters):
        if time.time() - current_time > budget_s - 1:
            break
        
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient via finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Adaptive learning rate with momentum
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-12:
            lr = 0.05 / (grad_norm + 1e-8)
        else:
            lr = 0.05
            
        # Update with gradient
        h_new = h - lr * grad
        
        # Projection to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
