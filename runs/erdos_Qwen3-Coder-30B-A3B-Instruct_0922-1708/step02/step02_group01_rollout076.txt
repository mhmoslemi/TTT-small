import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    n_particles = 50
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Initialize particles
    particles = np.random.rand(n_particles, n_points) * 0.5 + 0.25
    particles[0] = h_init
    velocities = np.random.rand(n_particles, n_points) - 0.5
    personal_best = particles.copy()
    personal_best_scores = np.full(n_particles, float('inf'))
    
    # Global best tracking
    global_best_idx = 0
    global_best_score = float('inf')
    global_best_position = particles[0].copy()
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Project to feasible region
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            particles[i] = h
            
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_best[i] = h.copy()
            
            # Update global best
            if c5 < global_best_score:
                global_best_score = c5
                global_best_position = h.copy()
                global_best_idx = i
                last_improvement = iteration
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Check early termination
        if global_best_score < 0.3807:
            break
            
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (global_best_position - particles[i]))
            
            particles[i] += velocities[i]
        
        # Project particles to feasible region
        for i in range(n_particles):
            h = particles[i]
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            particles[i] = h * scale
            
        # Restart velocities if stagnation
        if stagnation_count > 100:
            velocities = np.random.rand(n_particles, n_points) - 0.5
            stagnation_count = 0
            
    # Final validation
    final_h = np.clip(global_best_position, 0, 1)
    scale = (n_points / 2) / np.sum(final_h)
    final_h = final_h * scale
    
    # Final computation of C5
    correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (final_h.tolist(), final_c5, n_points)
