import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        # Create population around the base
        for i in range(population_size):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Initialize with random solutions
        for i in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Evolutionary algorithm parameters
    mutation_rate = 0.2
    crossover_rate = 0.5
    elite_size = population_size // 5
    max_generations = 1000
    generation = 0
    
    # Track the best solution
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Create new population
        new_population = []
        
        # Keep elite
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection by tournament
            parent1 = population[np.random.randint(0, population_size // 2)]
            parent2 = population[np.random.randint(0, population_size // 2)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(n_points, int(n_points * 0.2), replace=False)
                for i in mutate_indices:
                    child[i] += np.random.normal(0, 0.02)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    if best_h is not None:
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(correlation) * dx
        return (best_h.tolist(), final_c5, n_points)
    
    # Return the best from the final population if no better was found
    final_fitness = []
    for h in population:
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        final_fitness.append(c5)
        
    best_idx = np.argmin(final_fitness)
    best_h = population[best_idx]
    final_c5 = final_fitness[best_idx]
    
    return (best_h.tolist(), final_c5, n_points)
