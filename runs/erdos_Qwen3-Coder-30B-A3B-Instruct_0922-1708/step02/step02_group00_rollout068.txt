import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral C5 for a given step function h."""
    correlation = correlate(h, 1 - h, mode='full')
    max_corr = np.max(correlation)
    return max_corr * dx

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population of step functions."""
    population = []
    
    if initial_h is not None:
        # Interpolate initial construction to target resolution
        x_orig = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_orig, initial_h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        
        # Generate variations
        for _ in range(pop_size - 1):
            h_mut = h.copy()
            # Apply small random perturbations
            perturb_idx = np.random.choice(n_points, size=n_points // 5, replace=False)
            for i in perturb_idx:
                h_mut[i] += np.random.normal(0, 0.05)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            population.append(h_mut)
    else:
        # Start with random spike-like constructors
        for _ in range(pop_size):
            h = np.zeros(n_points)
            num_spikes = np.random.randint(3, 15)
            for _ in range(num_spikes):
                pos = np.random.randint(0, n_points)
                height = np.random.uniform(0.3, 0.9)
                h[pos] = height
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    return population

def crossover(parent1, parent2):
    """Perform crossover between two parent step functions."""
    n = len(parent1)
    crossover_point = np.random.randint(1, n - 1)
    
    offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return offspring1, offspring2

def mutate(h, mutation_rate=0.15):
    """Apply mutation to a step function."""
    h_mut = h.copy()
    n = len(h)
    num_mutations = int(n * mutation_rate)
    mutation_indices = np.random.choice(n, num_mutations, replace=False)
    
    for i in mutation_indices:
        h_mut[i] += np.random.normal(0, 0.08)
    
    # Clip values to [0, 1]
    h_mut = np.clip(h_mut, 0, 1)
    
    # Rescale to meet constraint sum(h) = n/2
    h_mut = h_mut * (n / 2) / np.sum(h_mut)
    
    return h_mut

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    generations = 1000
    
    # Get initial population
    initial_h = globals().get('initial_h_values', None)
    population = initialize_population(n_points, pop_size, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Find current best
        current_min_c5 = min(fitness)
        if current_min_c5 < best_c5:
            best_c5 = current_min_c5
            best_h = population[fitness.index(current_min_c5)].copy()
        
        # Sort by fitness (C5)
        sorted_indices = np.argsort(fitness)
        top_half = sorted_indices[:pop_size // 2]
        
        # Create new population
        new_population = [population[i] for i in top_half]
        
        # Generate offspring via crossover and mutation
        while len(new_population) < pop_size:
            # Selection (tournament)
            idx1, idx2 = np.random.choice(top_half, 2, replace=False)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover
            if np.random.rand() < 0.7:
                offspring1, offspring2 = crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            offspring1 = mutate(offspring1)
            offspring2 = mutate(offspring2)
            
            new_population.extend([offspring1, offspring2])
        
        # Trim to population size
        population = new_population[:pop_size]
    
    if best_h is None:
        # Fallback to last population
        fitness = [evaluate_h(h, dx) for h in population]
        best_index = np.argmin(fitness)
        best_h = population[best_index]
        best_c5 = fitness[best_index]
    
    return (best_h.tolist(), best_c5, n_points)
