import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # GA parameters
    population_size = min(50, max(20, n_points // 2))
    mutation_rate = 0.1
    elite_size = population_size // 5
    max_no_improvement = 100
    no_improvement_count = 0
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals() and len(population) == 0:
            member = h.copy()
        else:
            member = np.random.rand(n_points)
            member = member * (n_points / 2) / np.sum(member)
        member = np.clip(member, 0, 1)
        population.append(member)
    
    if 'initial_h_values' in globals():
        population[0] = h.copy()
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    def evaluate_population():
        fitnesses = [compute_c5(individual) for individual in population]
        return np.array(fitnesses)
    
    def select_parents(fitnesses):
        # Roulette wheel selection
        inv_fitnesses = 1.0 / (fitnesses + 1e-15)
        probabilities = inv_fitnesses / np.sum(inv_fitnesses)
        indices = np.random.choice(len(population), size=2, p=probabilities)
        return population[indices[0]], population[indices[1]]
    
    def crossover(parent1, parent2):
        # Uniform crossover with segment averaging
        child = np.zeros_like(parent1)
        crossover_point = np.random.randint(1, n_points)
        child[:crossover_point] = parent1[:crossover_point]
        child[crossover_point:] = parent2[crossover_point:]
        # Normalize child
        child = child * (n_points / 2) / np.sum(child)
        child = np.clip(child, 0, 1)
        return child
    
    def mutate(individual):
        # Gaussian mutation
        mutated = individual.copy()
        num_mutations = max(1, int(mutation_rate * n_points))
        mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
        for i in mutation_indices:
            mutated[i] += np.random.normal(0, 0.05)
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    generation = 0
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        generation += 1
        fitnesses = evaluate_population()
        
        # Update best solution
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = fitnesses[sorted_indices]
        
        # Keep elite
        new_population = sorted_population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = select_parents(sorted_fitnesses[:population_size//2])
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        
        # Adaptive mutation rate (slowly increase)
        if generation % 20 == 0 and mutation_rate < 0.3:
            mutation_rate += 0.05
    
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
