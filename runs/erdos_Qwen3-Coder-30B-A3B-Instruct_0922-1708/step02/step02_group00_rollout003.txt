import numpy as np
from scipy.optimize import minimize
import time
import math

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Multi-start optimization with adaptive learning rate
    n_starts = 5
    for start in range(n_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize h for current start
        if start == 0:
            h_current = h.copy()
        else:
            # Add small noise
            noise_level = 0.05
            noise = np.random.normal(0, noise_level, n_points)
            h_current = h + noise
            h_current = clip_and_scale(h_current, n_points)
        
        # Adam optimizer parameters
        alpha = 0.001
        beta1 = 0.9
        beta2 = 0.999
        epsilon = 1e-8
        
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        
        # Optimization loop
        for iter in range(1000):  # Max iterations
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current C5
            c5_current = evaluate_c5(h_current, dx)
            
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
                
            # Compute gradient via finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - c5_current) / eps
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**(iter+1))
            v_hat = v / (1 - beta2**(iter+1))
            
            h_new = h_current - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
            h_current = clip_and_scale(h_new, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
