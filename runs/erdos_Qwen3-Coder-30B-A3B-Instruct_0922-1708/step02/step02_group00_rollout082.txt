import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        # Create a smooth initial function
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(3 * np.pi * x) * np.exp(-x**2)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    epsilon = 1e-8
    lr = 0.001
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current solution
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the critical shift
        max_idx = np.argmax(corr)
        k = max_idx - (len(h) - 1)
        
        # Compute gradient for the critical shift
        grad = np.zeros_like(h)
        
        # Forward shift gradient component
        if k >= 0:
            for i in range(len(h) - k):
                grad[i] += 1 - h[i + k]
                
        # Backward shift gradient component
        if k <= 0:
            for i in range(-k, len(h)):
                grad[i] += -h[i + k]
                
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Periodic global perturbation
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
