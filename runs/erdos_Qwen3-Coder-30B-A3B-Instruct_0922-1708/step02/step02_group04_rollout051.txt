import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0

    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx

    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current C5 value
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Hybrid optimization approach
        if iteration % 5 == 0:
            # Systematic refinement approach
            h_new = h.copy()
            # Identify problematic regions by computing overlap for all shifts
            # and flipping values where overlap is high
            overlap_values = np.correlate(h, 1 - h, mode='full') * dx
            # Flip few points in high overlap regions to escape local minima
            flip_indices = np.argsort(overlap_values)[-5:]
            for idx in flip_indices:
                i = idx % n_points
                # Flip a small value in h
                h_new[i] = 1 - h_new[i]
            
            # Rescale and normalize to integral = 1
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            h = h_new
        else:
            # Random perturbations with adaptive step size
            perturb = np.random.normal(0, 0.02, n_points)
            # Clip both ways to ensure feasibility after scaling
            h_new = h + perturb
            h_new = np.clip(h_new, 0, 1)
            # Rescale and normalize to make integral = 1
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Sometimes do a more significant step
            if np.random.rand() < 0.1:
                # Exploratory jump
                i = np.random.randint(n_points)
                h_new[i] = np.random.rand()
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_new = np.clip(h_new, 0, 1)
                
            # Accept the new solution if improvement or with probability
            new_c5 = compute_c5(h_new)
            if new_c5 < current_c5 or np.random.rand() < 0.3:
                h = h_new
    
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
