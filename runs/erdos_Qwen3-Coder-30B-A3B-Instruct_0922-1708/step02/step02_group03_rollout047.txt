import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Adaptive parameters
    learning_rate = 0.001
    lr_decay = 0.999
    eps = 1e-6
    no_improvement_count = 0
    max_no_improvement = 500
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Compute gradient more accurately
        grad = np.zeros_like(h)
        max_indices = np.argmax(correlation)
        max_k = max_indices - (len(h) - 1)
        
        # Exact gradient computation based on which k produces the max
        for i in range(n_points):
            # Compute gradient contribution from this point
            # The derivative at index i under lag k
            grad[i] = (1 - h[(i - max_k) % n_points]) - h[(i + max_k) % n_points]
            
        # Update h
        h_new = h - learning_rate * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Adaptive learning rate
        learning_rate *= lr_decay
        
        # Occasionally check for early termination
        if iteration % 100 == 0 and iteration > 0:
            if time.time() - start_time > budget_s - 5:
                break
                
        if no_improvement_count > max_no_improvement:
            # Reset with better initialization
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                if len(h) != n_points:
                    x_orig = np.linspace(0, 2, len(h))
                    x_new = np.linspace(0, 2, n_points)
                    h = np.interp(x_new, x_orig, h)
                h = h / np.sum(h) * (n_points / 2)
            else:
                # Random restart if we're stuck
                h = np.random.rand(n_points) * 0.5
                h = h * (n_points / 2) / np.sum(h)
            
            no_improvement_count = 0
            
    return (best_h.tolist(), best_c5, n_points)
