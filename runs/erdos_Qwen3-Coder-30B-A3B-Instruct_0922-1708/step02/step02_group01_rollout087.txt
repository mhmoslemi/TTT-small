import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Binary search bounds
    C_low = 0.0
    C_high = 0.5
    epsilon = 1e-4
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search loop
    while C_high - C_low > epsilon and (time.time() - start_time) < budget_s - 1:
        C_mid = (C_low + C_high) / 2.0
        
        # Formulate the LP for feasibility check
        # Objective: minimize 0 (just check feasibility)
        c = np.zeros(n_points)
        
        # Constraints: h[i] * (1 - h[(i+k)%n_points]) * dx <= C_mid for all shifts k
        # By summing over all overlaps, we have the right side of inequalities
        # We'll stack all h[i] * (1 - h[(i+k)%n_points]) terms for each k
        
        A_ub = []
        b_ub = []
        
        # For each shift k from 0 to n_points-1
        for k in range(n_points):
            # Create constraint for this shift
            row = np.zeros(n_points)
            for i in range(n_points):
                # h[i] * (1 - h[(i+k)%n_points]) term
                # For inequality: h[i] * (1 - h[(i+k)%n_points]) <= C_mid * dx
                # This is: -h[i] * h[(i+k)%n_points] + h[i] <= C_mid * dx
                # But the actual computation is a bit trickier
                row[i] = 1.0
            # Add offset due to 1-h influences
            offset_row = np.zeros(n_points)
            for i in range(n_points):
                offset_row[i] = -h[(i+k) % n_points]
                
            # Sum these two rows
            total_row = row + offset_row
            A_ub.append(total_row)
            b_ub.append(C_mid * dx)  # Since we need to compute h[i]*(1-h[(i+k)%n_points])

        # Add bounds on h: 0 <= h[i] <= 1
        A_ub.extend(np.eye(n_points))
        A_ub.extend(-np.eye(n_points))
        b_ub.extend([1.0] * n_points)  # h[i] <= 1
        b_ub.extend([0.0] * n_points)  # -h[i] <= 0  -> h[i] >= 0
        
        # Add sum constraint: sum(h[i]) * dx = 1
        A_eq = [np.ones(n_points) * dx]
        b_eq = [1.0]
        
        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=[(0, 1)] * n_points, method='highs')
            
            if result.success:
                # Check if we can improve h to reduce max overlap
                if C_mid < best_c5:
                    best_c5 = C_mid
                    best_h = result.x
                C_high = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid
            
        # Protect against over-optimizing in case of numerical issues
        if C_mid < best_c5:
            best_c5 = C_mid
            best_h = h.copy()
    
    # Refine the result with final computation
    # Compute C5 value for best_h
    def compute_C5(h_vec):
        # Compute overlaps with 1-h
        correlations = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Max overlap times dx
        return np.max(correlations) * dx
    
    # Re-ensure we have valid solution
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    final_c5 = compute_C5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
