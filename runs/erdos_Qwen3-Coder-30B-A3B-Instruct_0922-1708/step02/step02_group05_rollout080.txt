import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation to get a good initialization
    if time.time() - start_time < budget_s / 3:
        # Formulate and solve convex relaxation
        h_var = cp.Variable(n_points)
        tau = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
        ]
        
        # Add shifted correlation constraints (approximation)
        for k in range(n_points):
            # Approximate correlation by summing relevant terms
            corr_sum = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_sum += h_var[i] * (1 - h_var[j])
            constraints.append(corr_sum <= tau * n_points)  # Scale by n_points to match dx
        
        prob = cp.Problem(cp.Minimize(tau), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_init = h_var.value
                h_init = np.clip(h_init, 0, 1)
                h_init = h_init * (n_points / 2) / np.sum(h_init)
                h = h_init
        except:
            pass  # Fall back to previous initialization
    
    # Gradient descent refinement
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate based on gradient magnitude
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
