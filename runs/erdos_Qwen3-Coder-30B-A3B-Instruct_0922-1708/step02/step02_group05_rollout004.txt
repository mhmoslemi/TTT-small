import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h, n_points):
    """Evaluate the C5 bound for given h"""
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = evaluate_h(h, n_points)
    
    start_time = time.time()
    
    # Define constraints for differential evolution
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.min(h)  # Should be >= 0
    
    def constraint_bounds_max(h):
        return 1 - np.max(h)  # Should be >= 0
    
    # Use scipy's differential evolution for global optimization
    # Objective function
    def obj_func(h):
        return evaluate_h(h, n_points)
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': constraint_bounds},
        {'type': 'ineq', 'fun': constraint_bounds_max}
    ]
    
    # differential_evolution parameters
    de_kwargs = {
        'bounds': bounds,
        'constraints': constraints,
        'maxiter': 1000,
        'popsize': 50,
        'tol': 1e-6,
        'recombination': 0.7,
        'mutation': (0.5, 1.0),
        'seed': seed,
        'workers': 1,
        'updating': 'immediate',
        'disp': False
    }
    
    # Run differential evolution
    try:
        result = differential_evolution(obj_func, bounds, **de_kwargs)
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5_bound = evaluate_h(h, n_points)
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h.copy()
        else:
            # Fallback to using the best h we had
            pass
    except Exception as e:
        # Fallback to gradient descent if DE fails
        pass
    
    # Continue with a few iterations of gradient descent from the DE solution
    h = best_h.copy()
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 5000:
        iteration += 1
        
        # Evaluate current C5
        c5 = evaluate_h(h, n_points)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_h(h_plus, n_points)
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
