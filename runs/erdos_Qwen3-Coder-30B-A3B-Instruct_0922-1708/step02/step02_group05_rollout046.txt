import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize swarm
    particles = np.array([h.copy() for _ in range(swarm_size)])
    velocities = np.random.uniform(-0.5, 0.5, (swarm_size, n_points))
    
    # Evaluate initial particles
    pbest = np.array([particle.copy() for particle in particles])
    pbest_vals = np.full(swarm_size, np.inf)
    
    gbest = h.copy()
    gbest_val = np.inf
    
    start_time = time.time()
    
    # PSO iteration loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        # Evaluate particles
        for i in range(swarm_size):
            particle = particles[i]
            
            # Compute overlap integral (C5)
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_vals[i]:
                pbest_vals[i] = c5
                pbest[i] = particle.copy()
                
            if c5 < gbest_val:
                gbest_val = c5
                gbest = particle.copy()
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (pbest[i] - particles[i])
                           + c2 * r2 * (gbest - particles[i]))
            
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    return (gbest.tolist(), gbest_val, n_points)
