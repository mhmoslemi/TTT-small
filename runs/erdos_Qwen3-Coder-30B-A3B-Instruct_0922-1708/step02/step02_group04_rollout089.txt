import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use a gradient-based optimization approach
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    def compute_gradient(h):
        n = len(h)
        grad = np.zeros(n)
        
        # Compute overlap for all shifts
        correlation = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(correlation)
        max_overlap = correlation[max_k]
        
        # Compute gradient analytically for the max overlap
        # Overlap at shift k is sum_i h[i] * (1 - h[(i-k)%n])
        # Gradient is -sum_{i such that i-k is valid} h[(i-k)%n] if i contributes
        # But this is complex, so use finite differences
        eps = 1e-6
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            c5_plus = compute_c5(h_plus)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        return grad
    
    # Optimizer
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current C5 value
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Gradient step
        try:
            grad = compute_gradient(h)
            step_size = 0.01
            h_new = h - step_size * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Accept step if it improves
            new_c5 = compute_c5(h_new)
            if new_c5 < current_c5:
                h = h_new
            else:
                # Try smaller step
                h_new = h - step_size * 0.1 * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                new_c5 = compute_c5(h_new)
                if new_c5 < current_c5:
                    h = h_new
        except:
            # Fallback: random perturbation
            perturb = np.random.normal(0, 0.01, n_points)
            h_new = h + perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
        
        # Occasionally escape local minima
        if np.random.rand() < 0.05:
            noise = np.random.normal(0, 0.02, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
