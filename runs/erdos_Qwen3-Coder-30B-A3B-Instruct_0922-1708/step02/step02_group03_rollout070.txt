import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    lr_initial = 0.005
    lr_decay = 0.999
    eps = 1e-6
    max_iter = 10000
    n_restarts = 5
    lr = lr_initial
    
    # Try multiple random restarts with different initializations
    restarts_done = 0
    while time.time() - start_time < budget_s - 1 and restarts_done < n_restarts:
        # Simple restart: perturb current best
        h = best_h.copy() + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Optimization loop per restart
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break
                
            iteration += 1
            
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Compute gradient
            grad = np.zeros_like(h)
            max_k_idx = np.argmax(correlation)
            # max_k is k in [-n+1, n-1]
            max_k = max_k_idx - (n_points - 1)
            
            # More efficient gradient computation
            for i in range(n_points):
                # Essential: use only the maximum k correlation
                # For lag k, gradient w.r.t. h[i] 
                # is 1 - h[(i-k)%n] if k is positive, 
                # and -h[(i+k)%n] if k is negative
                if max_k >= 0:
                    grad[i] = 1 - h[(i - max_k) % n_points]
                else:
                    # max_k < 0 implies we shift the index forward
                    grad[i] = -h[(i + max_k) % n_points]
            
            # Update h
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
            lr *= lr_decay
            
            # Occasionally check for early termination
            if iter_count % 500 == 0 and iter_count > 0:
                if time.time() - start_time > budget_s - 5:
                    break
                    
        restarts_done += 1
        
    return (best_h.tolist(), best_c5, n_points)
