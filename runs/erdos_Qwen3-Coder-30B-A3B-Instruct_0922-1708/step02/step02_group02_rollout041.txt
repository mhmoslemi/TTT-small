import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0

    # Try different strategies: gradient descent, and local search with SDP relaxation
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try multiple different update strategies for escaping local optima
        strategies = []
        
        # Gradient descent
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update h using gradient descent
        lr = 0.005
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        strategies.append(('grad', h_new.copy()))
        
        # Random perturbations
        for _ in range(3):
            h_perturb = h + np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            strategies.append(('perturb', h_perturb.copy()))
        
        # Try a local search using SDP relaxation idea, but simplified version
        # Use a convex relaxation approach by formulating as a QP (quadratic program)  
        try:
            # Start with heuristics: look for piecewise constant solutions
            h_piecewise = h.copy()
            # Apply some sort of thresholding
            h_piecewise = np.clip(h_piecewise, 0, 1)
            # Rescale
            h_piecewise = h_piecewise * (n_points / 2) / np.sum(h_piecewise)
            strategies.append(('piecewise', h_piecewise))
        except:
            pass
            
        # Small perturbation around current best
        h_small_perturb = h + np.random.normal(0, 0.001, n_points)
        h_small_perturb = np.clip(h_small_perturb, 0, 1)
        h_small_perturb = h_small_perturb * (n_points / 2) / np.sum(h_small_perturb)
        strategies.append(('small_perturb', h_small_perturb))
        
        # Try each strategy
        for name, h_try in strategies:
            correlation_try = np.correlate(h_try, 1 - h_try, mode='full')
            c5_try = np.max(correlation_try) * dx
            
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()  # Update for next round
                
        # Update h to be best found so far
        h = best_h.copy()
        
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
