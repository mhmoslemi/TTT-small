import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Binary comb approach: 5 pulses
        h = np.zeros(n_points)
        pulse_width = 0.2
        pulse_spacing = 0.25
        start_positions = [0.0, 0.25, 0.5, 0.75, 1.0]
        for i in range(5):
            start = int(start_positions[i] / 2.0 * n_points)
            end = int((start_positions[i] + pulse_width) / 2.0 * n_points)
            if end > start:
                h[start:end] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Use differential evolution with appropriate bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial evaluation
    c5_initial = objective(h)
    if c5_initial < best_c5:
        best_c5 = c5_initial
        best_h = h.copy()
    
    # Run optimization
    try:
        result = differential_evolution(
            objective, 
            bounds, 
            maxiter=1000,
            popsize=15,
            mutation=(0.5, 1.0),
            recombination=0.7,
            seed=seed,
            disp=False
        )
        if result.success:
            h_opt = result.x.copy()
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Local refinement with gradient-based method
    max_iter = 1000
    lr = 0.001
    decay_rate = 0.999
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        if i % 10 == 0:
            c5_current = objective(best_h)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = best_h.copy()
        
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = objective(h_plus)
            
            grad[j] = (c5_plus - best_c5) / eps
        
        # Update with gradient
        best_h = best_h - lr * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        lr *= decay_rate
        
    # Final evaluation of best solution
    final_c5 = objective(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
