import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T0 = 0.1
    tau = 1000
    start_time = time.time()
    
    iteration = 0
    T = T0
    
    while time.time() - start_time < budget_s - 1 and T > 1e-6:
        iteration += 1
        # Compute current overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate candidate solution
        noise_strength = T
        h_candidate = h + np.random.normal(0, noise_strength, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute candidate overlap
        correlation_cand = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(correlation_cand) * dx
        
        # Accept or reject based on simulated annealing criteria
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate
        
        # Update temperature
        T = T0 / (1 + iteration / tau)
    
    return (best_h.tolist(), best_c5, n_points)
