import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_new = np.linspace(0, 2, n_points)
            h_base = np.interp(x_new, x_orig, h_base)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        population.append(h_base.copy())
        # Generate variations
        for _ in range(population_size - 1):
            h_var = h_base + np.random.normal(0, 0.05, n_points)
            h_var = np.clip(h_var, 0, 1)
            h_var = h_var * (n_points / 2) / np.sum(h_var)
            population.append(h_var.copy())
    else:
        # Create initial population with random spikes
        for _ in range(population_size):
            # Random number of spikes (5-10)
            n_spikes = np.random.randint(5, 11)
            h = np.zeros(n_points)
            spike_positions = np.random.choice(n_points, size=n_spikes, replace=False)
            for pos in spike_positions:
                h[pos] = np.random.uniform(0.5, 1.0)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate all candidates
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50%
        survivors = population[:population_size//2]
        new_population = survivors[:]
        
        # Create new population via crossover and mutation
        while len(new_population) < population_size:
            # Selection (tournament)
            idx1 = np.random.randint(0, len(survivors))
            idx2 = np.random.randint(0, len(survivors))
            if fitness[idx1] < fitness[idx2]:
                parent1 = survivors[idx1]
            else:
                parent1 = survivors[idx2]
            parent1 = parent1.copy()
            
            # Crossover
            if np.random.random() < 0.7:
                if len(survivors) > 1:
                    idx2 = np.random.randint(0, len(survivors))
                    while idx2 == idx1:
                        idx2 = np.random.randint(0, len(survivors))
                    parent2 = survivors[idx2].copy()
                    
                    # Single point crossover
                    crossover_point = np.random.randint(1, n_points - 1)
                    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                else:
                    child1 = parent1.copy()
                    child2 = parent1.copy()
            else:
                child1 = parent1.copy()
                child2 = parent1.copy()
            
            # Mutation
            for child in [child1, child2]:
                # Mutate 15% of points
                n_mutate = int(0.15 * n_points)
                indices = np.random.choice(n_points, size=n_mutate, replace=False)
                for i in indices:
                    delta = np.random.uniform(-0.2, 0.2)
                    child[i] += delta
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        population = new_population[:population_size]
    
    return (best_h.tolist(), best_c5, n_points)
