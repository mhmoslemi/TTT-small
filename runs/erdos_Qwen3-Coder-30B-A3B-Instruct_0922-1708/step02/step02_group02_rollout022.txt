import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency (we'll use only relevant ones)
    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try a different approach: QP-based refinement
        # Define variables
        h_var = Variable(n_points)
        t = Variable()
        
        # Define constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            sum(h_var) == n_points / 2,
            t >= 0
        ]
        
        # Add correlation constraints
        for k in range(1, n_points):
            # Compute the correlation sum for lag k
            # The constraint is: sum(h[i] * (1 - h[i+k])) * dx <= t
            # We iterate over valid positions for this lag
            corr_sum = 0
            for i in range(n_points - k):
                corr_sum += h_var[i] * (1 - h_var[i+k])
            # Add constraint
            constraints.append(corr_sum * dx <= t)
        
        # Create and solve the problem
        objective = Minimize(t)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array(h_var.value).reshape(-1)
                # Clip to [0,1]
                h_new = np.clip(h_new, 0, 1)
                # Rescale to meet sum constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # Fallback to gradient descent if solver fails
                grad = np.zeros_like(h)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5) / eps
                lr = 0.005
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except:
            # Fallback to gradient descent on failure
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            lr = 0.005
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
