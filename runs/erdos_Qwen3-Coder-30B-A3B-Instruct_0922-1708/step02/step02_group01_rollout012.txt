import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Binary search bounds
    C_low = 0.0
    C_high = 0.5
    epsilon = 1e-4
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    while (C_high - C_low > epsilon) and (time.time() - start_time < budget_s - 1):
        mid = (C_low + C_high) / 2.0
        
        # Formulate linear constraints for feasibility check
        # Each inequality: sum h[i] * (1 - h[(i+k) % n_points]) * dx <= mid
        # This becomes: sum h[i] * (1 - h[(i+k) % n_points]) <= mid / dx
        
        # Linearize: use variable bounds and approximate:
        # For each shift k, we have constraint:
        # sum_{i=0}^{n_points-1} h[i] * (1 - h[(i+k) % n_points]) * dx <= mid
        # This is quadratic in h, but we approximate as linear with current h
        
        # Build the constraint matrix for this mid value
        # We'll use a hack: fixed linearization based on a test h
        
        # Constraints of the form: A @ h <= b
        num_constraints = n_points  # one constraint per shift
        A = np.zeros((num_constraints, n_points))
        b = np.full(num_constraints, mid / dx)
        
        # For each shift, compute the constraint coefficients
        for k in range(n_points):
            for i in range(n_points):
                A[k, i] += 1.0 * dx
                A[k, (i + k) % n_points] -= dx  # subtract the term related to h_i * h_{i+k}
        
        # Bound constraints for h: 0 <= h[i] <= 1
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Constraint that sum h[i] = 1 (keeping in terms of dx, = n_points/2)
        A_eq = np.ones((1, n_points))
        b_eq = [n_points / 2]
        
        # Solve the linear program
        res = linprog(
            c=np.zeros(n_points),
            A_ub=A,
            b_ub=b,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method='highs',
            options={'time_limit': 5}
        )
        
        if res.success:
            h_candidate = res.x
            # Verify constraint that sum is exact
            assert abs(np.sum(h_candidate) - n_points / 2) < 1e-6
            
            # Compute the actual C5 using convolution
            correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            computed_c5 = np.max(correlation) * dx
            
            if computed_c5 < best_c5:
                best_c5 = computed_c5
                best_h = h_candidate.copy()
                
            if computed_c5 <= mid:
                C_high = mid
            else:
                C_low = mid
        else:
            # Check if infeasible; raise lower bound
            C_low = mid
            
    return (best_h.tolist(), best_c5, n_points)
