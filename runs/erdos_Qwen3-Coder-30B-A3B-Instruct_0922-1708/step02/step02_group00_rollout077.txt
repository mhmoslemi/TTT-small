import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using parent or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    n_starts = 5
    iteration = 0
    
    # Generate diverse initializations
    starts = []
    starts.append(h.copy())
    
    # Add noisy variations
    for _ in range(4):
        noise = np.random.normal(0, 0.05, n_points)
        h_var = h + noise
        h_var = clip_and_scale(h_var, n_points)
        starts.append(h_var.copy())
    
    for i_start, h_start in enumerate(starts):
        h = h_start.copy()
        
        # Adam optimizer parameters
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.001
        
        # Optimization loop
        while time.time() - start_time < budget_s - 1 and iteration < 500:
            iteration += 1
            
            # Compute current C5
            c5 = compute_c5(h, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Compute gradient using finite differences
            grad = np.zeros(n_points)
            eps_val = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps_val
                h_plus = clip_and_scale(h_plus, n_points)
                
                c5_plus = compute_c5(h_plus, dx)
                grad[i] = (c5_plus - c5) / eps_val
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**(iteration))
            v_hat = v / (1 - beta2**(iteration))
            
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = clip_and_scale(h, n_points)
    
    # Final validation
    best_h = clip_and_scale(best_h, n_points)
    best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
