import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    learning_rate = 0.001
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    t = 0  # Adam time step
    
    while time.time() - start_time < budget_s - 1:
        t += 1
        
        # Compute all correlations and find maximum
        corr_full = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr_full)
        max_corr = corr_full[max_lag_idx]
        c5 = max_corr * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute exact gradient for the active lag
        # The active lag is max_lag_idx - (n_points - 1)
        active_lag = max_lag_idx - (n_points - 1)
        
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Compute gradient for h[i] based on the contribution to C5
            # For lag k, ∂C5/∂h[i] = 1 - h[i-k] if valid, -h[i+k] if valid
            if active_lag >= 0:
                if i >= active_lag:
                    grad[i] += -h[i - active_lag]
                if i + active_lag < n_points:
                    grad[i] += -h[i + active_lag]
            else:
                if i >= -active_lag:
                    grad[i] += -(1 - h[i + active_lag])
                if i - active_lag < n_points:
                    grad[i] += -(1 - h[i - active_lag])
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        # Rescale to ensure integral is correct
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Early termination check
        if t % 500 == 0 and time.time() - start_time > budget_s - 5:
            break
    
    return (best_h.tolist(), best_c5, n_points)
