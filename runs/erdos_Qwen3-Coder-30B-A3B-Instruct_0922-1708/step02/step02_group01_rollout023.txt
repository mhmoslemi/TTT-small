import numpy as np
import time

def evaluate_h(h, n_points, dx):
    """Evaluate C5 for given h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def initialize_population(n_points, n_pop, initial_h=None):
    """Initialize population with optional initial solution"""
    population = []
    
    if initial_h is not None:
        h = np.array(initial_h)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h.copy())
    
    # Fill rest of population with random samples
    while len(population) < n_pop:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h.copy())
    
    return population

def mutate_individual(h, mutation_rate, n_points):
    """Mutate individual with noise and repair constraints"""
    h_mut = h.copy()
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, 0.1)
    
    # Repair constraints
    h_mut = np.clip(h_mut, 0, 1)
    scale_factor = (n_points / 2) / np.sum(h_mut)
    h_mut = h_mut * scale_factor
    
    return h_mut

def crossover(parent1, parent2):
    """Single-point crossover"""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def tournament_selection(population, fitness, tournament_size=2):
    """Tournament selection for parent selection"""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitness[i] for i in selected_idx])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    n_pop = 50
    c_rate = 0.8
    m_rate = 0.2
    
    # Initialize
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    
    population = initialize_population(n_points, n_pop, initial_h)
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        fitness = []
        
        # Evaluate fitness for all individuals
        for h in population:
            c5 = evaluate_h(h, n_points, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
                
        # Early termination if good enough
        if best_c5 < 0.3807:
            break
            
        stagnation_count += 1
        if stagnation_count > 50:
            # Disturbance - reset 20% of population
            num_to_reset = max(1, int(0.2 * n_pop))
            for _ in range(num_to_reset):
                idx = np.random.randint(0, n_pop)
                population[idx] = np.random.rand(n_points)
                population[idx] = population[idx] / np.sum(population[idx]) * (n_points / 2)
            stagnation_count = 0
        
        # Selection
        selected_parents = []
        for _ in range(n_pop):
            parent = tournament_selection(population, fitness)
            selected_parents.append(parent)
        
        # Crossover and mutation
        new_population = []
        
        # Elitism - keep top 20%
        elite_indices = np.argsort(fitness)[:int(0.2 * n_pop)]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < n_pop:
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            if np.random.random() < c_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate_individual(child1, m_rate, n_points)
            child2 = mutate_individual(child2, m_rate, n_points)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:n_pop]
        
        # Periodic check for early termination
        if generation % 10 == 0:
            if best_c5 < 0.3809:
                # More aggressive restart if getting close
                break
    
    # Final validation
    if best_h is not None:
        # Ensure constraint satisfaction
        best_h = np.clip(best_h, 0, 1)
        scale = (n_points / 2) / np.sum(best_h)
        best_h = best_h * scale
        final_c5 = evaluate_h(best_h, n_points, dx)
        return (best_h.tolist(), final_c5, n_points)
    
    # Return from last generation if no improvement
    final_fitness = [evaluate_h(h, n_points, dx) for h in population]
    best_idx = np.argmin(final_fitness)
    return (population[best_idx].tolist(), final_fitness[best_idx], n_points)
