import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute lags we care about
    max_lag = n_points - 1
    
    # For quadratic programming approach
    # We want to minimize max_k int h(x)(1-h(x+k)) dx
    # equivalently: minimize t such that for all k: int h(x)(1-h(x+k)) dx <= t
    # This translates to linear constraints
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlation) * dx
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # QP formulation
        # Variables: h[0], ..., h[n_points-1], t
        h_var = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, sum(h_var) == n_points / 2, t >= 0]
        
        # Add correlation constraints for active lags
        # Only consider lags that are relevant for current solutions
        # Correlations computed on full data
        full_corr = np.correlate(h, 1 - h, mode='full')
        
        # Create a subset of most active lags to be most efficient
        active_lags = np.argsort(-full_corr)[:min(10, len(full_corr))]
        
        # Only generate constraints for these lags
        for k in active_lags:
            # The correlation term at lag k is sum_{i=0}^{n_points-k-1} h[i] * (1 - h[i+k])
            # We express this in matrix form (this is the constraint that this term <= t)
            
            # For index i in original h, in correlation, we look at h[i+k] 
            # Over range from 0 to n_points-k-1
            p = (np.arange(0, n_points - k) * 2 - 1) * dx  # h * (1 - h_shifted) contributions for this lag
            p = p * (1 - np.concatenate([h[k:], np.zeros(k)]))  # Actual products
            # Actually, better just to construct the coefficients properly:
            
            if k == 0:
                continue  # This is a special case, we do not use it as it's just a sum
            
            # Build the linear constraint manually
            # We'll create a mixture of guaranteed helpfulness and surviving traces
            active_indices = np.arange(n_points - k)
            # Might be needed to manually sum to match correlate result
            # This constraint: sum_{i=0}^{n_points-k-1} h[i] * (1 - h[i+k]) <= t
            # Let's try another theoretical kernel-based robust approach
            
            # Generate constraint for this specific lag only
            # But we already have exact value from full corr - just use the lag index
            c = np.zeros(n_points)
            c[active_indices] = 1.0  # coefficient for h[i]  
            d = np.zeros(n_points)
            d[active_indices + k] = 1.0  # coefficient for h[i+k]  
            constraints.append(0.5 * np.sum(c * h) - 0.5 * np.sum(d * h) - t <= 0)  # Actually a simpler approach
            
        # Simpler approach: just separate constraints for critical ones
        # Instead of full QP solve (which is costly), use the current structure and
        # just project in gradient descent manner
        
        # Let's do a gradient step with constraints using a feasible direction search
        # Gradient estimate:
        eps = 1e-4
        grad = np.zeros(n_points)
        
        # Use mini-batch of important lags to estimate gradient
        # Sample lags with big contribution
        idxs = np.argsort(full_corr)[::-1][:min(5, len(full_corr))]
        for i in range(n_points):
            dummy_h = h.copy()
            dummy_h[i] += eps
            dummy_h = np.clip(dummy_h, 0, 1)
            dummy_h = dummy_h * (n_points / 2) / np.sum(dummy_h)  # Keep sum = n_points/2
            
            corr_dummy = np.correlate(dummy_h, 1 - dummy_h, mode='full')
            c5_dummy = np.max(corr_dummy) * dx
            
            grad[i] = (c5_dummy - c5_current) / eps
            
        # Gradient step
        lr = 0.02
        h_new = h - lr * grad
        
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
