```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from provided parent construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        # Random spikes
        h = np.random.rand(n_points) * 0.5
        
    # Scale to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    epsilon = 1e-8
    alpha = 0.001
    t = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        t += 1
        
        # Compute correlations for all shifts
        corr = correlate(h, 1 - h, mode='full')
        # The full correlation is of length 2*n_points - 1.
        # For the shift k in [0, 2*n_points - 2], the corresponding
        # correlation value is at index k.
        max_k = np.argmax(corr)
        max_corr_val = corr[max_k]
        c5 = max_corr_val * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient for the maximum correlation shift
        grad = np.zeros(n_points)
        # Now we compute the gradient for only the dominant shift
        # The shift k is from 0 to 2*n_points-2 (inclusive)
        k = max_k  # shift value (0 to 2*n_points-2)
        # For the shift k, contribution to gradient:
        # if i+k < n_points then d/dh_i = 1 - h[i+k] 
        # if i-k >= 0 then d/dh_i -= h[i-k]
        # We will do this more clearly
        # We treat the convolution as a sum over indices
        # i ranges from 0 to n_points-1
        # For shift k, the correlation term is sum_{j} h[i] * (1 - h[i+k-j])
        # which is equivalent when j runs from 0 to n_points-1 to:
        #    h[i] * (1 - h[i+k]) if i+k < n_points
        #    -h[i] * h[i-k] if i-k >= 0

        # s = h[i] * (1 - h[i+k-j]) for each j
        # The gradient is the partial derivative of this with respect to h[i]
        # If i+k < n_points, then d/dh_i = 1 - h[i+k]  
        # If i-k >= 0, then also includes a term -h[i-k]
        # We process this as:
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1 - h[i + k]) * dx
                
            # Incompatibility: need to update when i+k >= n_points. 
            # But in this, we realize we should be  using convolution operation correctly.
            # More precise:
            # Consider that correlation computes sum_{j} h[j] * (1 - h[j-k]) for shift k
            # So derivative of the overlap with respect to h[i] includes terms:
            # components where j = i-k, j = i+k, etc.
            # Using auto-diff or analog:
            # If j = i-k, then h[j] contributes via d/dh[i] = h[j] (corresponding to h[i-k] term)
            # If j = i+k, then h[j] contributes via d/dh[i] = 1-h[j] (corresponding to (1-h[i+k]) term)
            # This corresponds to correct indexing for the full correlation. Let's recompute gradient properly
            
        # We'll compute the full gradient following standard way:
        # Full gradient correspond to d C5 / d h_i 
        # Using the full "valid" correlation approach (which is what we defined)
        # Now the update
        # For easier computation: use convolution similarly  
        # Derivative wrt h[i] of sum over j of same terms as in correlation:
        # orig corr is corr_{k} = sum h[j] * (1 - h[j-k])
        # So derivative w.r.t h[i]: d/dh[i] = sum for j such that j-k=i or j=i+k of (1-h[j]) - h[j]* (which is not correct)
        # This is more tricky, let's avoid this for now.
        # Let's use gradient shift-based approximation more directly:
        # Change only where it affects that maximum shift
        # Gradient computation: look at position i contributed to the maximum shift k. 
        # For i such that i+k < n_points, there is term h[i] (1 - h[i+k])
        # If i-k >= 0, also there is h[i] h[i-k]
        # So d/d h[i] = (1 - h[i+k]) * dx if i+k < n_points
        # + (nonzero if exists) -h[i-k] * dx if i-k >= 0  
        # We'll compute exactly with kernel positions
        # gradient stored: d/dh[i] = d/dh[i] sum_j h[j] * (1-h[j-k]) for minimum k

        # Correct construction of gradients!
        grad = np.zeros(n_points)
        # We are updating gradient wrt h[i] of correlation k 
        # where k causes maximum correlation 
        # i.e. correlation = sum_{j env} h[j] * (1 - h[j - k])
        # The pole of h[j-k] is at j = i+k, where we assume sum over h[j] (left side no issues)
        # But j-k = i => j = i+k hence ker h_j = 1-h_{i+k} (also contrasts)
        # d/dh[i]  = (1 - h[i+k]) if i+k < n_points
        # d/dh[i] -= h[i-k] if i-k >= 0
        # So:
        # For i in 0 to n_points-1: 
        #     grad[i] += (1 - h[i+k]) if i+k < n_points
        #     grad[i] -= h[i-k] if i-k >= 0
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1 - h[i + k]) * dx
            if i - k >= 0:
                grad[i] -= h[i - k] * dx
                
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Perturbation to escape local minima
        if t % 500 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # For large arrays or complicated evaluations, we could add early-stopping
        # But mean optimization is ongoing small for now.
    
    return (best_h.tolist(), best_c5, n_points)
```
