import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or create a random starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        # Create a piecewise constant function with spikes
        h = np.ones(n_points) * 0.5
        # Add some spikes in the middle
        h[40:60] = 1.0
        h[30:40] = 0.0
        h[60:70] = 0.0
        
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Adam optimizer parameters
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    alpha = 0.001
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current solution
        correlation = correlate(h, 1 - h, mode='full')
        # The maximum correlation is at index n_points-1
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute the gradient with respect to the worst-case overlap shift
        max_k = np.argmax(correlation)
        k = max_k - (n_points - 1)
        
        # Gradient computation (only for the max overlap shift)
        grad = np.zeros(n_points)
        for i in range(n_points):
            # derivative of h_i (1 - h_{i+k})
            if i + k >= 0 and i + k < n_points:
                grad[i] += (1 - h[i+k])
            # derivative of (1-h_i) h_{i-k}
            if i - k >= 0 and i - k < n_points:
                grad[i] += -h[i-k]
                
        # Adam update step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        
        h_new = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Periodic perturbation to escape local minima
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
