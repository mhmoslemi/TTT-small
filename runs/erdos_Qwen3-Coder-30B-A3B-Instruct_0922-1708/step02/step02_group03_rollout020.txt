import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    lr_initial = 0.005
    lr_decay = 0.999
    max_iter = 10000
    n_restarts = 5
    
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        h = best_h.copy()
        lr = lr_initial
        
        last_improvement = 0
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            iteration += 1
            
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            max_corr_idx = np.argmax(correlation)
            max_corr = correlation[max_corr_idx]
            c5 = max_corr * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = i
                
            # Compute gradient with respect to the active index
            grad = np.zeros_like(h)
            active_k = max_corr_idx - (len(h) - 1)
            
            for j in range(n_points):
                # Compute gradient for j-th element
                h_plus = h.copy()
                eps = 1e-6
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                max_corr_plus = corr_plus[max_corr_idx]
                c5_plus = max_corr_plus * dx
                
                grad[j] = (c5_plus - c5) / eps
            
            # Update h with adaptive learning rate
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
            # Update learning rate
            lr *= lr_decay
            
            # Early stopping
            if i - last_improvement > 500:
                break
                
    return (best_h.tolist(), best_c5, n_points)
