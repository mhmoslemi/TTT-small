import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    tau = 1.0
    alpha = 0.001
    
    # Adam optimizer params
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute surrogate objective
        shifts = correlation * dx
        # Smooth max approximation
        log_sum_exp = np.log(np.sum(np.exp(tau * shifts)))
        smooth_max = log_sum_exp / tau
        
        # Gradient of smooth max
        exp_shifts = np.exp(tau * shifts)
        softmax_weights = exp_shifts / np.sum(exp_shifts)
        
        # Gradient w.r.t. h
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Estimate gradient via finite difference of overlaps
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / 1e-5
        
        # Update using Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        h_new = h - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptively decrease tau
        if iteration % 100 == 0:
            tau *= 0.99
            
        # Occasionally try a heuristic restart
        if iteration % 1000 == 0:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
