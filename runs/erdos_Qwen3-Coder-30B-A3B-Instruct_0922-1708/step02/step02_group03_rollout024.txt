import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    lr_initial = 0.001
    lr_decay = 0.999
    max_iter = 10000
    n_restarts = 5
    
    # For each restart, perform gradient-based optimization
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_current = h.copy()
        lr = lr_initial
        no_improvement_count = 0
        best_in_restart = float('inf')
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            iteration += 1
            
            # Compute the overlap integral
            correlation = np.correlate(h_current, 1 - h_current, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_current.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            
            if no_improvement_count > 500:
                break
                
            # Find the active lag that produces the maximum
            max_idx = np.argmax(correlation)
            active_lag = max_idx - (len(h_current) - 1)
            
            # Compute exact gradient for this lag
            grad = np.zeros_like(h_current)
            for j in range(len(h_current)):
                if active_lag >= 0:
                    # h[j] influences (1-h[j+lag]) term
                    grad[j] = -1.0 * (1.0 - h_current[j+active_lag])
                    # Also needs to consider the cross-term for lag-positive
                    if j + active_lag < len(h_current):
                        grad[j] += h_current[j+active_lag]
                else:
                    # Consider negative lag
                    grad[j] = -1.0 * (1.0 - h_current[j+active_lag])
                    if j + active_lag >= 0 and j + active_lag < len(h_current):
                        grad[j] += h_current[j+active_lag]
                        
            # Update h
            h_new = h_current - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_current = h_new
            
            # Adaptive learning rate adjustment
            lr *= lr_decay
            
    return (best_h.tolist(), best_c5, n_points)
