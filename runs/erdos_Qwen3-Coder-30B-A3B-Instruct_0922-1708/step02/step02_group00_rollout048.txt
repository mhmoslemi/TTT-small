import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Optimized gradient-based approach with adaptive learning rate
    def compute_c5_and_grad(h):
        # Compute all correlations efficiently
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5, correlation

    # Gradient computation for a single variable
    def finite_diff_grad(h, i, eps=1e-5):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
        c5_plus, _ = compute_c5_and_grad(h_plus)
        
        c5_minus, _ = compute_c5_and_grad(h)
        return (c5_plus - c5_minus) / eps

    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5, _ = compute_c5_and_grad(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adaptive gradient descent
        grad = np.zeros_like(h)
        for i in range(n_points):
            grad[i] = finite_diff_grad(h, i)
            
        # Update with adaptive step size
        lr = 0.01 / (1 + iteration / 1000)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Periodic stochastic restarts
        if iteration % 100 == 0:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            # Add small random noise to escape local optima
            h += np.random.normal(0, 0.001, n_points)
        
    return (best_h.tolist(), best_c5, n_points)
