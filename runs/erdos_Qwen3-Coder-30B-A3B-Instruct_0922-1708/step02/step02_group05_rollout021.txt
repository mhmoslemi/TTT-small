import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO Parameters
    swarm_size = 50
    max_iter = 500
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            pos = np.array(initial_h_values)
            if len(pos) != n_points:
                pos = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(pos)), pos)
            pos = np.clip(pos, 0, 1)
            pos = pos * (n_points / 2) / np.sum(pos)
        else:
            pos = np.random.rand(n_points)
            pos = pos * (n_points / 2) / np.sum(pos)
        
        particles.append(pos.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        pbest_positions.append(pos.copy())
        pbest_scores.append(float('inf'))
    
    gbest_position = particles[0].copy()
    gbest_score = float('inf')
    
    # PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        # Evaluate each particle
        for i in range(swarm_size):
            pos = particles[i]
            
            # Compute overlap integral
            correlation = np.correlate(pos, 1 - pos, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_scores[i]:
                pbest_scores[i] = c5
                pbest_positions[i] = pos.copy()
                
            if c5 < gbest_score:
                gbest_score = c5
                gbest_position = pos.copy()
                
        # Update particles
        for i in range(swarm_size):
            pos = particles[i]
            vel = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (w * vel + 
                   c1 * r1 * (pbest_positions[i] - pos) + 
                   c2 * r2 * (gbest_position - pos))
            
            # Update position
            pos = pos + vel
            
            # Apply constraints
            pos = np.clip(pos, 0, 1)
            pos = pos * (n_points / 2) / np.sum(pos)
            
            particles[i] = pos
            velocities[i] = vel
            
        if gbest_score < best_c5:
            best_c5 = gbest_score
            best_h = gbest_position.copy()
            
    return (best_h.tolist(), best_c5, n_points)
