import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    
    # Set up differential evolution parameters
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Custom constraint for sum(h) = n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Suppress warnings for faster execution
    import warnings
    warnings.filterwarnings("ignore")
    
    # Run DE optimization
    result = differential_evolution(
        lambda x: evaluate_c5(x, dx),
        bounds,
        constraints=constraints,
        maxiter=int(budget_s / 2),
        popsize=30,
        seed=seed,
        disp=False
    )
    
    # Compute final C5 for the best solution
    final_h = result.x
    final_c5 = evaluate_c5(final_h, dx)
    
    # Keep best result during optimization
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    # Re-evaluate to make sure it's still valid
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
