import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    learning_rate = 0.005
    eps = 1e-6
    lr_decay = 0.9995
    
    # Random restarts
    n_restarts = 5
    
    for restart in range(n_restarts):
        # Start with the best solution so far or the initialization
        if restart == 0:
            current_h = best_h.copy()
        else:
            # Random perturbation
            current_h = h + np.random.normal(0, 0.02, n_points)
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        last_improvement = 0
        for iter_count in range(10000):  # Max iterations per restart
            if time.time() - start_time > budget_s - 2:
                break
                
            iteration += 1
            # Compute the overlap integral
            correlation = np.correlate(current_h, 1 - current_h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
                last_improvement = iter_count
                
            # Compute gradient (exact computation)
            grad = np.zeros_like(current_h)
            max_k = np.argmax(correlation)
            # Find the actual lag that produces maximum correlation
            lag = max_k - (n_points - 1)
            
            # Compute gradient for this specific lag
            for i in range(n_points):
                # Gradient computation for this specific lag
                if i + lag >= 0 and i + lag < n_points:
                    grad[i] += (1 - current_h[i + lag]) * dx
                if i - lag >= 0 and i - lag < n_points:
                    grad[i] -= current_h[i - lag] * dx
                    
            # Update h
            current_h_new = current_h - learning_rate * grad
            
            # Clip and rescale
            current_h_new = np.clip(current_h_new, 0, 1)
            current_h_new = current_h_new * (n_points / 2) / np.sum(current_h_new)
            current_h = current_h_new
            
            # Adaptive learning rate
            if iter_count % 100 == 0:
                learning_rate *= lr_decay
                
            # Early stopping if no improvement
            if iter_count - last_improvement > 500:
                break
                
    return (best_h.tolist(), best_c5, n_points)
