import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy the constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = project_to_feasible(h, n_points)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    start_time = time.time()
    generation = 0
    
    # Use a simple gradient-based local search with restarts
    max_generations = int(budget_s * 20)  # Roughly 20 ops/sec
    
    while generation < max_generations and time.time() - start_time < budget_s - 1:
        # Do a few gradient steps
        h_current = h.copy()
        c5_current = best_c5
        
        for step in range(5):  # Take a few steps of gradient descent
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros_like(h_current)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                
                c5_plus = evaluate_h(h_plus, dx)
                grad[i] = (c5_plus - c5_current) / eps
            
            # Adaptive learning rate
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-10:
                lr = min(0.1, 1.0 / (grad_norm + 1e-8))
            else:
                lr = 0.01
                
            h_new = h_current - lr * grad
            h_new = project_to_feasible(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
            else:
                break
                
        # Check if we improved
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            
        # Make a random perturbation to avoid local minima
        h = h_current.copy()
        mutation_strength = 0.05
        for i in range(n_points):
            if np.random.random() < 0.1:  # 10% chance of mutation
                h[i] += np.random.normal(0, mutation_strength)
                
        h = project_to_feasible(h, n_points)
        
        # Perturb also the best solution to avoid getting stuck
        if np.random.random() < 0.05:
            h = best_h.copy()
            for i in range(n_points):
                if np.random.random() < 0.05:
                    h[i] += np.random.normal(0, 0.05)
            h = project_to_feasible(h, n_points)
            
        generation += 1
    
    # Final evaluation of best solution
    final_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
