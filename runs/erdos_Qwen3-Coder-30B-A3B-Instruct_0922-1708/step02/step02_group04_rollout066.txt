from scipy.optimize import minimize

        h = np.ones(n_points) * 0.5
        # Set up 3 segments: high, low, high
        split1 = n_points // 3
        split2 = 2 * n_points // 3
        h[:split1] = 0.75
        h[split1:split2] = 0.25
        h[split2:] = 0.75
    
    dx = 2.0 / n_points
    
    # Normalize to integral = 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = float('inf')
    
    iteration = 0
    last_improvement = 0
    
    # Helper to evaluate C5 based on overlap correlation
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
        
    # Function to evaluate gradient of overlap with respect to h values
    def evaluate_gradient(h_vals, k):
        # This is a simple and approximate gradient
        # We'll just modify the values around where the maximum correlation occurs
        n = len(h_vals)
        grad = np.zeros(n)
        for i in range(n):
            # Small perturbation
            h_plus = h_vals.copy()
            h_plus[i] += 1e-6
            h_plus = np.clip(h_plus, 0, 1)
            
            h_minus = h_vals.copy()
            h_minus[i] -= 1e-6
            h_minus = np.clip(h_minus, 0, 1)
            
            # Ensure integral constraint is satisfied
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            c5_plus = compute_c5(h_plus)
            c5_minus = compute_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2e-6)
        return grad
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1 and iteration - last_improvement < 100:

        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Fine-tune h using a local optimization approach
        # We'll use the gradient-based approach by sampling the correlation for a fixed k
        # and optimizing around it

        noise = np.random.normal(0, 0.01, n_points)
        h_new = h + noise
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)

        if iteration % 5 == 0:
            # Apply small smoothing
            h_smooth = np.convolve(h, np.ones(5)/5, mode='same')

            h = h_smooth
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
