import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    n_particles = 50
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            x_orig = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, n_points)
            h0 = np.interp(x_new, x_orig, h0)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    else:
        h0 = np.ones(n_points) * 0.5
    
    particles = np.random.random((n_particles, n_points)) * 0.5
    particles[0] = h0
    
    # Initialize velocities
    velocities = np.random.random((n_particles, n_points)) * 0.01 - 0.005
    
    # Personal best and global best tracking
    pbest = particles.copy()
    pbest_c5 = np.full(n_particles, float('inf'))
    
    gbest_idx = 0
    gbest_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Project to feasible region
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            
            # Compute correlation
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest_c5[i] = c5
                pbest[i] = h.copy()
            
            # Update global best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_idx = i
                stagnation_count = 0
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (pbest[gbest_idx] - particles[i]))
            
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            scale = (n_points / 2) / np.sum(particles[i])
            particles[i] = particles[i] * scale
            
        # Stagnation check and reset
        if iteration % max_stagnation == 0:
            stagnation_count += 1
            if stagnation_count > 1:
                # Reset velocities
                velocities = np.random.random((n_particles, n_points)) * 0.01 - 0.005
                stagnation_count = 0
                
        # Check for early termination
        if gbest_c5 < 0.3807:
            break
    
    # Final evaluation of global best
    final_h = pbest[gbest_idx]
    final_correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (final_h.tolist(), final_c5, n_points)
