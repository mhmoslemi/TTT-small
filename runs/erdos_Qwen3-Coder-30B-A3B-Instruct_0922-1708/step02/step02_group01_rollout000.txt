import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = int(budget_s * 1000)  # Rough estimate
    
    # Initialize particles
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    # Scale to desired sum
    for i in range(n_particles):
        scale = (n_points / 2) / np.sum(particles[i])
        particles[i] = np.clip(particles[i] * scale, 0, 1)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles[0] = h_init
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Track best solution
    personal_best = particles.copy()
    personal_best_scores = np.full(n_particles, float('inf'))
    global_best_idx = 0
    global_best_score = float('inf')
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            correlation = np.correlate(h, 1 - h, mode='full')
            score = np.max(correlation) * dx
            
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = h.copy()
                
            if score < global_best_score:
                global_best_score = score
                global_best_idx = i
                stagnation_count = 0
                
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            scale = (n_points / 2) / np.sum(particles[i])
            particles[i] = particles[i] * scale
            
        # Check for stagnation
        stagnation_count += 1
        if stagnation_count > 100:
            # Restart velocities
            velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
            stagnation_count = 0
            
        # Early termination if we found a good enough solution
        if global_best_score < 0.3807:
            break
    
    # Return best solution
    best_h = personal_best[global_best_idx]
    return (best_h.tolist(), global_best_score, n_points)
