import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T0 = 0.1
    tau = 1000
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Cooling schedule
        T = T0 / (1 + iteration / tau)
        
        # Generate candidate solution
        h_candidate = h + np.random.normal(0, T, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute C5 for current and candidate
        corr_current = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(corr_current) * dx
        
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr_candidate) * dx
        
        # Accept or reject the candidate
        delta_c5 = c5_candidate - c5_current
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate.copy()
            c5 = c5_candidate
        else:
            c5 = c5_current
            
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
