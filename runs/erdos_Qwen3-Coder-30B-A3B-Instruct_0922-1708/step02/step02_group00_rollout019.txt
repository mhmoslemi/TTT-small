import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_new = np.linspace(0, 2, n_points)
            h_base = np.interp(x_new, x_orig, h_base)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        population.append(h_base.copy())
        for _ in range(population_size - 1):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Generate initial random solutions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Evolution loop
    gen = 0
    max_generations = 1000
    mutation_rate = 0.15
    crossover_rate = 0.7
    elite_size = population_size // 10
    
    while time.time() - start_time < budget_s - 1 and gen < max_generations:
        gen += 1
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        # Sort by fitness
        indices = np.argsort(fitness)
        sorted_population = [population[i] for i in indices]
        sorted_fitness = [fitness[i] for i in indices]
        
        # Elitism
        new_population = sorted_population[:elite_size]
        
        # Generate new population
        while len(new_population) < population_size:
            # Selection (tournament)
            def tournament_select():
                k = 3
                idx = np.random.choice(population_size, k)
                best_idx_in_tournament = idx[np.argmin([fitness[i] for i in idx])]
                return sorted_population[best_idx_in_tournament]
            
            # Crossover
            if np.random.random() < crossover_rate:
                p1 = tournament_select()
                p2 = tournament_select()
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
                child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
                children = [child1, child2]
            else:
                # Mutation only
                parent = tournament_select()
                child = parent.copy()
                num_mutations = int(mutation_rate * n_points)
                mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
                for i in mutation_indices:
                    child[i] += np.random.normal(0, 0.1)
                children = [child]
            
            # Apply bounds and normalization
            for child in children:
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    return (best_h.tolist(), best_c5, n_points)
