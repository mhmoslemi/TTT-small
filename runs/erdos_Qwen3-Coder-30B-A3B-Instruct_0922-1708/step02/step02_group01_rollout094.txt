import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    delta = 0.01
    
    # Helper to compute overlap integrals
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
            delta = 0.01 * best_c5
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                delta *= 0.5
                delta = max(delta, 1e-4)
                no_improvement_count = 0
                
        # Find the largest overlap
        k_max = 0
        max_overlap = 0
        for k in range(n_points):
            overlap = 0.0
            for i in range(n_points):
                overlap += h[i] * (1 - h[(i + k) % n_points])
            overlap *= dx
            if overlap > max_overlap:
                max_overlap = overlap
                k_max = k
                
        # Define the constraint for worst shift
        # This constraint effectively says that the overlap for the worst shift
        # should be less than or equal to current_C5 - delta
        # We approximate this with projected gradient steps
        
        # Perform gradient-based update
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = 0.0
            for j in range(n_points):
                overlap_plus += h_plus[j] * (1 - h_plus[(j + k_max) % n_points])
            overlap_plus *= dx
            
            overlap = 0.0
            for j in range(n_points):
                overlap += h[j] * (1 - h[(j + k_max) % n_points])
            overlap *= dx
            
            grad[i] = (overlap_plus - overlap) / eps
            
        # Update h with momentum
        lr = 0.01
        momentum = 0.9
        if iteration == 1:
            velocity = -lr * grad
        else:
            velocity = momentum * velocity - lr * grad
            
        h_new = h + velocity
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        scale = (n_points / 2) / np.sum(h_new)
        h_new = h_new * scale
        
        h = h_new
        
        # Occasionally restart if we're stuck (only for small datasets)
        if iteration % 1000 == 0:
            h = best_h.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            delta = 0.01 * best_c5
            
        # Early termination if target reached
        if best_c5 < 0.3807:
            break
            
    # Final validation and recomputation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
