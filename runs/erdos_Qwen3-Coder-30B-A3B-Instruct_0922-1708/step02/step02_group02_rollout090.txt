from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    if 'initial_h_values' in globals():

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()

        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Use a more sophisticated optimization approach
        # Try different step sizes and directions
        grad = np.zeros_like(h)
        eps = 1e-4
        
        # Compute gradient with finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Use momentum-based update
        if iteration > 1:
            momentum = 0.9
            velocity = momentum * velocity + (1 - momentum) * grad
        else:
            velocity = grad
            
        # Update h using momentum-based gradient descent
        lr = 0.01
        h_new = h - lr * velocity
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally perform local search using Nelder-Mead
        if iteration % 10 == 0:
            # Sample different perturbations
            h_current = h.copy()
            for _ in range(10):
                perturbation = np.random.normal(0, 0.05, n_points)

    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
