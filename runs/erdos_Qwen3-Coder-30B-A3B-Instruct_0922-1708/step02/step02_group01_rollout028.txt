import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use 200 points for higher resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.random.rand(n_particles, n_points) * 0.5
    velocities = np.random.rand(n_particles, n_points) * 0.01
    
    # If initial construction exists, use it as one particle
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        particles[0] = h_init / np.sum(h_init) * (n_points / 2)
    
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iterations = int(budget_s * 1000)  # Roughly 1000 iterations per second
    
    # Personal bests
    personal_bests = particles.copy()
    personal_best_c5 = np.full(n_particles, float('inf'))
    
    # Global best
    global_best_idx = 0
    global_best_c5 = float('inf')
    
    start_time = time.time()
    stagnation_count = 0
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Ensure feasibility with projection
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < personal_best_c5[i]:
                personal_best_c5[i] = c5
                personal_bests[i] = h.copy()
                
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_idx = i
                best_h = h.copy()
                stagnation_count = 0
                
        # Update global best tracking
        stagnation_count += 1
        if stagnation_count > 100:
            # Restart velocities
            velocities = np.random.rand(n_particles, n_points) * 0.01
            stagnation_count = 0
            
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # PSO velocity update
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            v = (w * v + 
                 c1 * r1 * (personal_bests[i] - h) + 
                 c2 * r2 * (personal_bests[global_best_idx] - h))
            
            # Update position
            h_new = h + v
            
            # Projection to feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            velocities[i] = v
            
    # Ensure final solution is valid
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(correlation) * dx
    else:
        # Fallback to initial solution if no better found
        h_init = np.clip(particles[0], 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        correlation = np.correlate(h_init, 1 - h_init, mode='full')
        best_c5 = np.max(correlation) * dx
        best_h = h_init
        
    return (best_h.tolist(), best_c5, n_points)
