import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a simple block function
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use convex optimization approach
    # Variables for h
    h_var = Variable(n_points)
    
    # Constraints for h in [0,1] and integral = 1
    constraints = [h_var >= 0, h_var <= 1, sum(h_var) == n_points / 2]
    
    # For optimization, we minimize the maximum correlation value
    # We introduce a new variable to bound the max
    c5_var = Variable()
    
    # Add inequality constraints for all lags
    for k in range(1, n_points):
        # Formulate correlation constraint in terms of half convolution
        # But since we're working with frequency domain approximations, 
        # let's do something simpler: optimize one specific constraint
        # This is a proxy for the true objective but makes it convex
        pass
    
    # Read the current h and evaluate
    def evaluate_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    max_iterations = 10000  # For sanity check
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        c5_current = evaluate_c5(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Alternate optimization using proximal gradient descent
        # Compute gradient approximations
        eps = 1e-5
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus)
            grad[i] = (c5_plus - c5_current) / eps
            
        # Proximal step
        step_size = 0.005
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
