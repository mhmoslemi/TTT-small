import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    delta = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            delta = max(1e-4, delta * 0.95)
            
        # Find the worst shift
        max_k = np.argmax(correlation)
        # For full correlation, we adjust the indexing
        # The maximum correlation index is in [0, 2*n_points-2]
        # But it maps differently to actual shifts for the convolution.
        # we want to find k such that overlap = h[i] * (1-h[i+k]) for some i 
        # For mode="full", the index k_start = n_points - 1 maps to shift 0
        # In general, shift k maps to index k_start + k, so k = index - (n_points - 1)
        # for overlap at index, it's mapping to shift (index - (n_points-1)) from 0 to 2*n_points-2
        # shifts in range [0, 2*n_points-2] corresponds to shifts in [0, 2*n_points-2] 
        # actually, since h has length n_points and we correlate with 1-h, we shall directly check.
        # The actual shift makes sense for the cycle from 0 to n_points - 1 if it's properly interpreted.
        # if we compute with mode='full', for this particular use it seems k validly spans from 0 to n_points - 1
        # for our full convolution at index `idx`, actual shift is idx - (n_points-1). but to be consistent with our reasoning let's recompute
        
        # Actually, recompute directly
        # To get overlap at shift k:
        # sum_{i=0}^{n_points-1} h[i] * (1 - h[(i+k) % n_points]) 
        # From correlation:
        # correlation[shift_index] = sum_{i=0}^{n_points-1} h[i] * (1 - h[(i + shift_index - (n_points-1)) % n_points])
        # if shift_index = n_points-1+k, then actual shift = k. So:
        # shift = shift_index - (n_points-1)  
        # normalize shift to be in 0, ..., n_points-1 
        normalized_shift = max_k - (n_points - 1)
        if normalized_shift < 0:
            normalized_shift += n_points
        worst_shift = normalized_shift
        
        # Create a small change which is in the direction of reducing overlap at the worst shift
        # Just construct one change direction 
        h_plus = h.copy()
        epsilon = 1e-3
        perturbation = np.zeros_like(h)
        
        # try small changes in one variable
        pert_idx = worst_shift  # agressive but simple
        if worst_shift >= 0 and worst_shift < n_points:
            # The overlap at shift k is sum_{i} h[i] * (1 - h[i+k])
            # so we reduce overlap by decreasing h[pert_idx] or increasing h[pert_idx + k]
            # but to reduce it biggest, we do it with the one that makes the most impact
            # When we increase h[i], it decreases overlap in the contribution to shift k  
            # So we try to increase h[pert_idx] to reduce overlap contribution in shift k
            
            h_plus[pert_idx] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            if c5_plus < c5:
                h = h_plus.copy()
            else:
                h_plus[pert_idx] -= 2*epsilon
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                if c5_plus < c5:
                    h = h_plus.copy()
        
        # Manual update if above does nothing effective
        if iteration % 20 == 0:
            # Simple random perturbation with rescaling
            h = h + 0.005 * (np.random.random(n_points) - 0.5)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Scaling back to fixed sum
        if iteration % 80 == 0:
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
        
        # Early stopping
        if c5 < 0.3807:
            break
            
        # If no progress, reduce step size
        if iteration - last_improvement > 50:
            delta /= 2.0
            if delta < 1e-4:
                delta = 1e-4
                
    return (best_h.tolist(), best_c5, n_points)
