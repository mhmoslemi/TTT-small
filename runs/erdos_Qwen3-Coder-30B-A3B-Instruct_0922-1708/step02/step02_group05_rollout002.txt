import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_h(h_values, n_points):
    """Evaluate the C5 bound for given h values."""
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use parent construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function to minimize
    def objective(h):
        # Normalize h
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Compute overlap integral
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint function
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Create constraint dictionary
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Use differential evolution for global optimization
    start_time = time.time()
    
    # Run optimization using scipy's differential evolution which is well-suited for this kind of problem
    # We limit the number of iterations to stay within budget
    maxiter = min(1000, int(budget_s / 0.5))  # Adjust number of iterations based on time
    
    result = differential_evolution(
        objective,
        bounds,
        maxiter=maxiter,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        constraints=constraints,
        disp=False,
        atol=1e-6,
        rtol=1e-6
    )
    
    # Check time remaining
    elapsed_time = time.time() - start_time
    remaining_time = budget_s - elapsed_time
    
    # If we still have time, continue refining with local optimization
    if remaining_time > 1.0:
        # refine using scipy.optimize.minimize with L-BFGS-B
        from scipy.optimize import minimize
        
        # Make sure the result is valid
        h_final = np.clip(result.x, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        
        # Local refinement using L-BFGS-B
        constraints_lbfgs = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        # Run local optimization
        try:
            res_lbfgs = minimize(
                objective,
                h_final,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints_lbfgs,
                options={'maxiter': 500}
            )
            # Check refinement
            refined_h = np.clip(res_lbfgs.x, 0, 1)
            refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
            refined_c5 = evaluate_h(refined_h, n_points)
        except:
            refined_h = h_final
            refined_c5 = evaluate_h(refined_h, n_points)
            
        final_h = refined_h
        final_c5 = refined_c5
    else:
        final_h = np.clip(result.x, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_c5 = evaluate_h(final_h, n_points)
    
    # Final validation of constraints
    sum_h = np.sum(final_h)
    assert abs(sum_h - n_points / 2) < 1e-6, f"Sum constraint violated: {sum_h}"
    assert all(0 <= h <= 1 for h in final_h), "Bound constraint violated"
    
    return (final_h.tolist(), final_c5, n_points)
