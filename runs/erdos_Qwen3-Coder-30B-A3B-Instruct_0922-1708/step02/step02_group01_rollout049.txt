import numpy as np
import time
from scipy.signal import convolve

def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h"""
    # Compute correlation using convolution
    correlation = convolve(h, 1 - h, mode='full')
    # The maximum correlation corresponds to the C5 value
    c5 = np.max(correlation) * dx
    return c5

def create_initial_population(n_points, n_pop, initial_h_values=None):
    """Create initial population for GA"""
    population = []
    
    # If initial construction available, use it
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        # Normalize to required sum
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill rest of population with random samples
    while len(population) < n_pop:
        h = np.random.random(n_points)
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        population.append(h.copy())
        
    return population

def tournament_selection(population, fitness, tournament_size=2):
    """Select parent using tournament selection"""
    selected_indices = []
    for _ in range(len(population)):
        # Randomly select contestants
        contestants = np.random.choice(len(population), tournament_size)
        # Choose the one with best fitness (lowest C5)
        winner = contestants[np.argmin([fitness[i] for i in contestants])]
        selected_indices.append(winner)
    return selected_indices

def crossover(parent1, parent2, rate=0.8):
    """Perform single-point crossover"""
    if np.random.random() > rate:
        return parent1.copy(), parent2.copy()
        
    # Single-point crossover
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, rate=0.2, mutation_strength=0.1):
    """Mutate individual with uniform noise and repair"""
    h_mut = h.copy()
    # Apply mutation to random genes
    for i in range(len(h_mut)):
        if np.random.random() < rate:
            # Add uniform noise
            h_mut[i] += np.random.uniform(-mutation_strength, mutation_strength)
            # Repair constraints
            h_mut[i] = np.clip(h_mut[i], 0, 1)
            
    # Rescale to maintain sum constraint
    scale = (len(h_mut) / 2) / np.sum(h_mut)
    h_mut = np.clip(h_mut * scale, 0, 1)
    return h_mut

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    n_pop = 50
    c_rate = 0.8
    m_rate = 0.2
    mutation_strength = 0.1
    
    dx = 2.0 / n_points
    
    # Initialize population
    start_time = time.time()
    
    population = create_initial_population(n_points, n_pop, initial_h_values if 'initial_h_values' in globals() else None)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0  # Reset stagnation counter
        
        # Early termination if we've reached target
        if best_c5 < 0.3807:
            break
            
        stagnation_count += 1
        
        # Selection: tournament selection
        selected_indices = tournament_selection(population, fitness)
        selected_population = [population[i] for i in selected_indices]
        
        # Create new population through crossover and mutation
        new_population = []
        
        # Keep top 20% (elitism)
        elite_count = n_pop // 5
        sorted_indices = np.argsort(fitness)[:elite_count]
        for i in sorted_indices:
            new_population.append(population[i].copy())
            
        # Generate offspring through crossover and mutation
        while len(new_population) < n_pop:
            # Select two parents
            idx1 = np.random.randint(len(selected_population))
            idx2 = np.random.randint(len(selected_population))
            
            parent1 = selected_population[idx1]
            parent2 = selected_population[idx2]
            
            # Crossover
            child1, child2 = crossover(parent1, parent2, c_rate)
            
            # Mutation
            child1 = mutate(child1, m_rate, mutation_strength)
            child2 = mutate(child2, m_rate, mutation_strength)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:n_pop]
        
        # Stagnation handling: disturb 20% of population if no improvement
        if stagnation_count > 50:
            num_to_disturb = max(1, n_pop // 5)
            indices_to_disturb = np.random.choice(n_pop, num_to_disturb, replace=False)
            for i in indices_to_disturb:
                population[i] = np.random.random(n_points)
                # Rescale to maintain constraint
                population[i] = population[i] * (n_points / 2) / np.sum(population[i])
                population[i] = np.clip(population[i], 0, 1)
            stagnation_count = 0
        
    return (best_h.tolist(), best_c5, n_points)
