import numpy as np
import time

def project_h(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 using correlation"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.random.rand(n_particles, n_points) * 0.5
    velocities = np.random.randn(n_particles, n_points) * 0.01
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        particles[0] = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize best
    best_h = None
    best_c5 = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Track personal bests
    personal_best_h = particles.copy()
    personal_best_c5 = np.full(n_particles, float('inf'))
    
    global_best_h = None
    global_best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    stagnation = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = project_h(particles[i], n_points)
            c5 = evaluate_c5(h, dx)
            
            if c5 < personal_best_c5[i]:
                personal_best_c5[i] = c5
                personal_best_h[i] = h.copy()
            
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_h = h.copy()
                stagnation = 0
                
        if global_best_c5 < best_c5:
            best_c5 = global_best_c5
            best_h = global_best_h.copy()
            
        # Update stagnation
        if stagnation > 100:
            # Restart velocities
            velocities = np.random.randn(n_particles, n_points) * 0.01
            stagnation = 0
            
        stagnation += 1
        
        # Update particles
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best_h[i] - particles[i]) + 
                           c2 * r2 * (global_best_h - particles[i]))
            
            particles[i] = particles[i] + velocities[i]
            
            # Project after update
            particles[i] = project_h(particles[i], n_points)
        
        # Early exit condition
        if best_c5 < 0.3807:
            break
            
    if best_h is None:
        best_h = global_best_h.copy() if global_best_h is not None else particles[0].copy()
        best_c5 = evaluate_c5(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
