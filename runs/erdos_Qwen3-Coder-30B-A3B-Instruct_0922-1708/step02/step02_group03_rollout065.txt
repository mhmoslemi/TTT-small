import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 0.9
    inertia = w
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            pos = np.array(initial_h_values)
            if len(pos) != n_points:
                x_orig = np.linspace(0, 2, len(pos))
                x_new = np.linspace(0, 2, n_points)
                pos = np.interp(x_new, x_orig, pos)
            pos = pos / np.sum(pos) * (n_points / 2)
        else:
            pos = np.random.rand(n_points) * 0.5
            pos = pos * (n_points / 2) / np.sum(pos)
        particles.append(pos.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(pos.copy())
        pbest_vals.append(float('inf'))
    
    # Global best
    gbest = h.copy()
    gbest_val = float('inf')
    
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            pos = particles[i]
            
            # Ensure constraints
            pos = np.clip(pos, 0, 1)
            pos = pos * (n_points / 2) / np.sum(pos)
            particles[i] = pos
            
            # Compute C5
            correlation = np.correlate(pos, 1 - pos, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_vals[i]:
                pbest[i] = pos.copy()
                pbest_vals[i] = c5
                
                if c5 < gbest_val:
                    gbest = pos.copy()
                    gbest_val = c5
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = pos.copy()
                        no_improvement_count = 0
                        
        # Update global best
        if gbest_val < best_c5:
            best_c5 = gbest_val
            best_h = gbest.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update velocities and positions
        for i in range(n_particles):
            pos = particles[i]
            vel = velocities[i]
            
            # Random numbers for PSO
            r1 = np.random.rand()
            r2 = np.random.rand()
            
            # Update velocity
            vel = inertia * vel + c1 * r1 * (pbest[i] - pos) + c2 * r2 * (gbest - pos)
            
            # Update position
            pos = pos + vel
            
            # Clamp position and ensure constraint
            pos = np.clip(pos, 0, 1)
            pos = pos * (n_points / 2) / np.sum(pos)
            
            particles[i] = pos.copy()
            velocities[i] = vel.copy()
            
        # Reduce inertia after some iterations
        if iteration % 100 == 0 and inertia > 0.3:
            inertia *= 0.95
            
        # Occasionally KKT-style optimization for best particle
        if iteration % 20 == 0:
            # Try refining globally best with projection of gradient
            h_refine = gbest.copy()
            prev_c5 = gbest_val
            
            # Simple gradient descent
            for step in range(10):
                # Compute gradients numerically
                grad = np.zeros(n_points)
                eps = 1e-5
                for j in range(n_points):
                    h_plus = h_refine.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - prev_c5) / eps
                
                # Gradient descent step
                h_refine -= 0.001 * grad
                h_refine = np.clip(h_refine, 0, 1)
                h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
                
                # Compute new correlation
                corr = np.correlate(h_refine, 1 - h_refine, mode='full')
                new_c5 = np.max(corr) * dx
                
                if new_c5 < prev_c5:
                    prev_c5 = new_c5
                    gbest = h_refine.copy()
                    gbest_val = prev_c5
        
        if no_improvement_count > 500:
            break
            
    return (best_h.tolist(), best_c5, n_points)
