```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Convex relaxation to get a better initial point
    start_time = time.time()
    
    # Set up relaxated optimization problem
    h_cvx = Variable(n_points)
    tau = Variable(1)
    
    # Constraints for convex relaxation
    constraints = [h_cvx >= 0, h_cvx <= 1, cp.sum(h_cvx) == n_points / 2]
    # Approximate the max constraint by adding all shift constraints
    # (This is a looser relaxation but computationally tractable)
    
    # Add all shift constraints
    for k in range(n_points):
        # Compute the correlation for shift k with 1-h
        corr_expr = 0
        for i in range(n_points):
            j = i + k
            if j < n_points:
                corr_expr += h_cvx[i] * (1 - h_cvx[j])
            else:
                break
        constraints.append(corr_expr <= tau * n_points)
    
    # Objective
    objective = Minimize(tau)
    prob = Problem(objective, constraints)
    
    # Try solving the convex relaxation
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=100)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_init = np.array(h_cvx.value).flatten()
            # Rescale and clip
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            h = h_init.copy()
    except:
        pass
    
    # Gradient-based refinement
    iteration = 0
    
    # Optimizer loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Feasibility maintenance
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
```
