import numpy as np
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)

    # Current best
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated annealing parameters
    T0 = 0.1
    tau = 1000
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Generate candidate
        T = T0 / (1 + iteration / tau)
        noise_strength = T * 0.1
        
        h_new = h + np.random.normal(0, noise_strength, n_points)
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to ensure constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute candidate C5
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept with probability
        if c5_new < c5 or np.random.rand() < math.exp(-(c5_new - c5) / T):
            h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
