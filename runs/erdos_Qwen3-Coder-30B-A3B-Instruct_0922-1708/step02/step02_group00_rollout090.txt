import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
        # Generate variations
        for _ in range(population_size - 1):
            h_mut = h.copy()
            indices = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
            for i in indices:
                h_mut[i] += np.random.uniform(-0.2, 0.2)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            population.append(h_mut)
    else:
        # Generate random starting population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    gen = 0
    max_generations = 1000
    
    while time.time() - start_time < budget_s - 1 and gen < max_generations:
        gen += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50% for breeding
        keep = population_size // 2
        population = population[:keep]
        
        # Generate offspring
        new_population = population.copy()
        
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = np.random.randint(0, keep)
            parent2_idx = np.random.randint(0, keep)
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < 0.7:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < 0.15:
                    indices = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
                    for i in indices:
                        child[i] += np.random.uniform(-0.2, 0.2)
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                
                new_population.append(child)
                if len(new_population) >= population_size:
                    break
        
        population = new_population
    
    return (best_h.tolist(), best_c5, n_points)
