import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Create variations around parent
    starts = [h.copy()]
    for i in range(5):
        pert = h + np.random.normal(0, 0.05, n_points)
        starts.append(clip_and_scale(pert, n_points))
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    for start_h in starts:
        if time.time() - start_time > budget_s - 1:
            break
            
        h = start_h.copy()
        
        # Optimization loop
        for i in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            iteration += 1
            c5 = compute_c5(h, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Gradient approximation using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = clip_and_scale(h_plus, n_points)
                
                c5_plus = compute_c5(h_plus, dx)
                grad[j] = (c5_plus - c5) / eps
            
            # Adam optimizer parameters
            beta1, beta2 = 0.9, 0.999
            lr = 0.001
            
            # Simple Adam-like update with clipping
            h_new = h - lr * grad
            h_new = clip_and_scale(h_new, n_points)
            
            h = h_new
    
    return (best_h.tolist(), best_c5, n_points)
