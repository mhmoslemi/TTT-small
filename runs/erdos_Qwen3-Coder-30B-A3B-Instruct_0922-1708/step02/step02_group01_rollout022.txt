import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Population-based genetic algorithm for optimization
    N_pop = 50
    C_rate = 0.8
    M_rate = 0.2
    disturbance_rate = 0.2
    tournament_size = 2
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    # Fill remaining population with random individuals
    while len(population) < N_pop:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    population = np.array(population)
    
    # Track best found solution
    best_h = None
    best_c5 = float('inf')
    
    def evaluate_individual(h):
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute correlation (overlap integral)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def tournament_selection(population, fitnesses):
        idxs = np.random.choice(len(population), size=tournament_size)
        best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        return population[best_idx]
    
    def crossover(parent1, parent2):
        if np.random.rand() > C_rate:
            return parent1.copy(), parent2.copy()
            
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        # Rescale to satisfy constraint
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        return child1, child2
    
    def mutate(individual):
        if np.random.rand() > M_rate:
            return individual.copy()
            
        mutated = individual.copy()
        idxs = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for i in idxs:
            noise = np.random.uniform(-0.1, 0.1)
            mutated[i] += noise
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    start_time = time.time()
    generation = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness for entire population
        fitnesses = [evaluate_individual(h) for h in population]
        
        # Update best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early termination if target reached
        if best_c5 < 0.3807:
            break
            
        # Selection
        new_population = []
        
        # Elitism - keep top 50%
        sorted_indices = np.argsort(fitnesses)
        elites = [population[i] for i in sorted_indices[:N_pop // 2]]
        new_population.extend(elites)
        
        # Generate offspring through selection, crossover, and mutation
        while len(new_population) < N_pop:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
            
        new_population = new_population[:N_pop]
        population = np.array(new_population)
        
        # Disturbance step to avoid stagnation
        if stagnation_count > 50:
            for i in range(int(N_pop * disturbance_rate)):
                idx = np.random.randint(0, N_pop)
                population[idx] = np.random.rand(n_points)
                population[idx] = population[idx] * (n_points / 2) / np.sum(population[idx])
            stagnation_count = 0
        
        generation += 1

    # Final validation and evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_individual(best_h)
        # Ensure we return the best found solution with final evaluation
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fall back to a random solution if nothing found
        h = np.ones(n_points) * 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_individual(h)
        return (h.tolist(), c5, n_points)
