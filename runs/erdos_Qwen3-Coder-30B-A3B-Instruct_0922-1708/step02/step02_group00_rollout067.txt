import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given step function h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def ensure_constraints(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def crossover(parent1, parent2, crossover_rate=0.7):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(individual, mutation_rate=0.15, noise_range=0.2):
    """Mutate an individual."""
    mutated = individual.copy()
    n = len(mutated)
    n_mutate = max(1, int(n * mutation_rate))
    
    indices = np.random.choice(n, n_mutate, replace=False)
    for i in indices:
        mutated[i] += np.random.uniform(-noise_range, noise_range)
    
    return ensure_constraints(mutated, n)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial values if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = ensure_constraints(h, n_points)
        population.append(h)
        for _ in range(population_size - 1):
            h_mut = mutate(h, mutation_rate=0.15, noise_range=0.1)
            population.append(h_mut)
    else:
        # Generate random start population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = ensure_constraints(h, n_points)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < 1000:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50% and generate new population
        new_population = population[:population_size // 2]
        
        # Add mutated copies of top individuals
        for i in range(population_size // 2):
            parent = population[i]
            mutated_h = mutate(parent, mutation_rate=0.15, noise_range=0.1)
            new_population.append(mutated_h)
        
        # Add crossover offspring
        while len(new_population) < population_size:
            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            child1, child2 = crossover(population[idx1], population[idx2])
            new_population.extend([child1, child2])
        
        # Keep population size
        population = new_population[:population_size]
        
        generation += 1
    
    return (best_h.tolist(), best_c5, n_points)
