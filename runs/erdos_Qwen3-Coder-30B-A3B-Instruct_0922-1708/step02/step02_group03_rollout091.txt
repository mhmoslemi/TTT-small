from scipy.optimize import minimize
import time
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points

        if len(h) != n_points:

            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 0.9
    max_iter = 1000
    kkt_every = 50
    
    # Initialize particles
    particles = []
    velocities = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            part = np.array(initial_h_values)
            if len(part) != n_points:
                x_orig = np.linspace(0, 2, len(part))
                x_new = np.linspace(0, 2, n_points)
                part = np.interp(x_new, x_orig, part)
            part = part / np.sum(part) * (n_points / 2)
        else:
            part = np.random.rand(n_points) * 0.5
            part = part * (n_points / 2) / np.sum(part)
        particles.append(part.copy())
        velocities.append(np.zeros(n_points))
    
    # Global best
    gbest_idx = 0
    gbest = particles[gbest_idx].copy()
    gbest_c5 = float('inf')
    
    iteration = 0
    no_improvement = 0
    max_no_improvement = 500
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        no_improvement += 1
        
        # Update particles
        for i, (particle, vel) in enumerate(zip(particles, velocities)):
            # Evaluate current particle
            corr = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
                no_improvement = 0
                
            # Update global best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest = particle.copy()
                
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            new_vel = (w * vel + 
                      c1 * r1 * (particles[i] - particle) + 
                      c2 * r2 * (gbest - particle))
            
            new_pos = particle + new_vel
            
            # Clip and rescale
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * (n_points / 2) / np.sum(new_pos)
            
            particles[i] = new_pos
            velocities[i] = new_vel
            
        # KKT refinement periodically
        if iteration % kkt_every == 0 and iteration > 0:
            # Apply KKT refinement to current gbest
            refinement_h = gbest.copy()
            refinement_c5 = gbest_c5
            best_refinement_h = refinement_h.copy()
            best_refinement_c5 = refinement_c5
            
            # Simple line search with gradients
            for _ in range(20):
                # Compute gradient using finite differences
                grad = np.zeros(n_points)
                eps = 1e-5
                
                corr_ref = np.correlate(refinement_h, 1 - refinement_h, mode='full')
                ref_c5 = np.max(corr_ref) * dx
                
                for i in range(n_points):
                    h_plus = refinement_h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[i] = (c5_plus - ref_c5) / eps
                
                # Update with gradient descent
                step_size = 0.001
                new_h = refinement_h - step_size * grad
                
                # Clip and rescale
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                # Evaluate new point
                corr_new = np.correlate(new_h, 1 - new_h, mode='full')
                new_c5 = np.max(corr_new) * dx
                
                if new_c5 < refinement_c5:
                    refinement_h = new_h
                    refinement_c5 = new_c5
                else:
                    break
                    
            if refinement_c5 < gbest_c5:
                gbest = refinement_h.copy()
                gbest_c5 = refinement_c5
            
            # Update the global best if refined version is better
            if gbest_c5 < best_c5:
                best_c5 = gbest_c5
                best_h = gbest.copy()
                no_improvement = 0
                
        # Early termination check
        if no_improvement > max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
