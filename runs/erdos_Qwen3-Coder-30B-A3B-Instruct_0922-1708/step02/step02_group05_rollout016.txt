import numpy as np
import time

def evaluate_h(h, dx):
    """Compute C5 for given h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def ensure_feasible(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = ensure_feasible(h, n_points)
    else:
        h = np.ones(n_points) * 0.5

    # Genetic algorithm parameters
    population_size = 50
    generations = 0
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    start_time = time.time()

    # Initialize population
    population = [ensure_feasible(np.random.rand(n_points), n_points) for _ in range(population_size)]
    if 'initial_h_values' in globals():
        population[0] = h.copy()

    def fitness(individual):
        return evaluate_h(individual, dx)

    while time.time() - start_time < budget_s - 1:
        generations += 1
        # Evaluate population
        fitnesses = [fitness(ind) for ind in population]
        
        # Find best individual in current population
        min_fit_idx = np.argmin(fitnesses)
        if fitnesses[min_fit_idx] < best_c5:
            best_c5 = fitnesses[min_fit_idx]
            best_h = population[min_fit_idx].copy()

        # Tournament selection
        def tournament():
            k = 3
            indices = np.random.choice(population_size, k)
            fits = [fitnesses[i] for i in indices]
            winner_idx = indices[np.argmin(fits)]
            return population[winner_idx].copy()

        # Crossover
        def crossover(parent1, parent2):
            cr = 0.5
            mask = np.random.rand(n_points) < cr
            child = np.where(mask, parent1, parent2)
            return child

        # Mutation
        def mutate(individual):
            mut_rate = 0.2
            mask = np.random.rand(n_points) < mut_rate
            noise = np.random.normal(0, 0.1, n_points)
            individual = individual + np.where(mask, noise, 0)
            individual = ensure_feasible(individual, n_points)
            return individual

        # Create offspring
        offspring = []
        for i in range(population_size // 2):
            parent1 = tournament()
            parent2 = tournament()
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            child1 = mutate(child1)
            child2 = mutate(child2)
            offspring.extend([child1, child2])

        # Replace worst in population
        sorted_indices = np.argsort(fitnesses)
        for i in range(population_size // 2):
            population[sorted_indices[-(i+1)]] = offspring[i]

    return (best_h.tolist(), best_c5, n_points)
