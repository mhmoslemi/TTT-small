import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Genetic Algorithm parameters
    population_size = min(50, max(20, n_points))
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 0
    no_improvement_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals() and np.random.rand() < 0.3:
            ind = h.copy()
        else:
            ind = np.random.rand(n_points)
        ind = ind * (n_points / 2) / np.sum(ind)
        ind = np.clip(ind, 0, 1)
        population.append(ind)
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        generations += 1
        
        # Evaluate fitness (C5 values) for each individual
        fitness = []
        for ind in population:
            correlation = np.correlate(ind, 1 - ind, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Update the best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Selection with roulette wheel
        fitness = np.array(fitness)
        # Normalize fitness (lower is better)
        fitness = 1.0 / (fitness + 1e-10)
        fitness = fitness / np.sum(fitness)
        selected_indices = np.random.choice(population_size, size=population_size, p=fitness)
        selected_population = [population[i] for i in selected_indices]
        
        # Crossover and mutation
        new_population = []
        
        for i in range(0, population_size, 2):
            parent1 = selected_population[i]
            parent2 = selected_population[(i+1) % population_size] if i+1 < population_size else selected_population[0]
            
            # Crossover
            if np.random.rand() < crossover_rate and population_size > 1:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] = max(0, min(1, child1[j] + np.random.normal(0, 0.05)))
                if np.random.rand() < mutation_rate:
                    child2[j] = max(0, min(1, child2[j] + np.random.normal(0, 0.05)))
            
            # Normalize and enforce constraints
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child1 = np.clip(child1, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            child2 = np.clip(child2, 0, 1)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Replace old population
        population = new_population
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
