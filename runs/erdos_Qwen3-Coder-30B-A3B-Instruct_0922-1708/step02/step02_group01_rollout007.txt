import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Multi-start approach with multiple particles
    num_particles = 10
    particles = []
    for i in range(num_particles):
        if i == 0 and 'initial_h_values' in globals():
            # Use initial construction as first particle
            particles.append(h.copy())
        else:
            # Random starting points
            h_init = np.random.rand(n_points) * 0.5 + 0.25
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            particles.append(h_init)
    
    # Velocity tracking for each particle
    velocities = [np.zeros(n_points) for _ in range(num_particles)]
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate each particle
        for p_idx in range(num_particles):
            h = particles[p_idx]
            
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                lr = max(lr * 1.1, 0.001)  # Adaptive learning rate
            
            # Compute gradient with finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update with momentum
            momentum = 0.9
            velocities[p_idx] = momentum * velocities[p_idx] - lr * grad
            h_new = h + velocities[p_idx]
            
            # Apply constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Occasionally perturb to escape local minima
            if np.random.rand() < 0.05:
                noise = np.random.normal(0, 0.02, n_points)
                h_new = h_new + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
            particles[p_idx] = h_new
        
        # Occasionally restart with better points
        if iteration % 500 == 0:
            # Restart with best solution so far
            for p_idx in range(num_particles):
                if p_idx == 0:
                    particles[p_idx] = best_h.copy()
                else:
                    h_init = np.random.rand(n_points) * 0.5 + 0.25
                    h_init = h_init * (n_points / 2) / np.sum(h_init)
                    particles[p_idx] = h_init
                    
            lr = 0.01  # Reset learning rate
    
    return (best_h.tolist(), best_c5, n_points)
