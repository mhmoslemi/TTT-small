import numpy as np
from scipy.optimize import differential_evolution
import time

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def evaluate_constraint(h, n_points):
    return np.sum(h) - n_points / 2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, dx)
    
    start_time = time.time()
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    def constraint_func(h):
        return evaluate_constraint(h, n_points)
    
    constraint = {'type': 'eq', 'fun': constraint_func}
    
    # Differential Evolution optimization
    result = differential_evolution(
        lambda h: compute_c5(h, dx),
        bounds,
        constraints=constraint,
        maxiter=int(budget_s / 2),
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    # Check the result
    if result.success:
        final_h = result.x
        final_c5 = result.fun
    else:
        final_h = best_h
        final_c5 = best_c5
    
    # Ensure final solution meets constraints
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
