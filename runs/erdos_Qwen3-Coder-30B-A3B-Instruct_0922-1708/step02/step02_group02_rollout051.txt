import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a block-like construction
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 0.5
        h[mid:] = 1.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use a simplified trust-region approach
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 500:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try perturbing in a few directions
        h_trial = h.copy()
        perturb_idx = np.random.randint(0, n_points)
        step_size = 0.01 * (0.5 + np.random.random() * 0.5)
        
        # Try small perturbations to local areas
        for i in range(max(0, perturb_idx-2), min(n_points, perturb_idx+3)):
            if np.random.random() > 0.7:
                h_trial[i] += step_size * (np.random.random() - 0.5)
        
        # Ensure bounds and normalization
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Check improvement
        corr_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
        c5_trial = np.max(corr_trial) * dx
        
        if c5_trial < c5:
            h = h_trial.copy()
    
    # Final evaluation
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
