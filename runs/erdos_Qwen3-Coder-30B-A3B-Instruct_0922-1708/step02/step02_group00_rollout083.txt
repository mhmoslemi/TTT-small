import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from provided parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        # Initial construction using a spike pattern
        h = np.zeros(n_points)
        h[20:30] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.001
    
    # Momentum and velocity for Adam
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    t = 0  # time step for Adam
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        t += 1
        
        # Compute correlation and identify the critical shift
        correlation = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(correlation)
        max_val = correlation[k_max]
        
        # Compute gradient for the critical shift only
        grad = np.zeros(n_points)
        if k_max < n_points:
            # Contribution from h[i] * (1 - h[i+k_max])
            for i in range(n_points):
                if i + k_max < n_points:
                    grad[i] += 1 - h[i + k_max]
                    # Gradient w.r.t. h[i+k_max] also contributes
                    grad[i + k_max] -= h[i]
                    # Correct for the case where i+k_max >= n_points
                    if i + k_max >= n_points:
                        grad[i] += 0  # No effect on elements beyond the range
        else:
            # For longer shifts: i+k_max >= n_points, so no valid 'h[i+k_max]'
            # Compute corr over full range including extra
            for i in range(n_points):
                if i + k_max - n_points + 1 < n_points and i + k_max - n_points + 1 >= 0:
                    grad[i] += -h[i + k_max - n_points + 1]
                    if i >= k_max - n_points + 1:
                        # This part has contribution from one term
                        grad[i - (k_max - n_points + 1)] += 1 - h[i]
        
        # Update Adam parameters
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - alpha * m_hat / (np.sqrt(v_hat) + eps)

        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Compute new C5 and store best
        correlation_new = correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Occasionally add noise to escape local minima
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
