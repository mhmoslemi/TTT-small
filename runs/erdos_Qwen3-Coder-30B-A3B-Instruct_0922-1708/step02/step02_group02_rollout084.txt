import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a binary comb pattern
        h = np.zeros(n_points)
        m = 5  # number of pulses
        w = 1.0 / m  # width of each pulse
        s = 1.0 / (m - 1)  # spacing between pulses
        for i in range(m):
            start = i * (w + s)
            end = start + w
            start_idx = int(start / 2.0 * n_points)
            end_idx = int(end / 2.0 * n_points)
            if end_idx > start_idx:
                h[start_idx:end_idx] = 1.0
        h = h / np.sum(h) * (n_points / 2)
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_c5(h):
        h = np.array(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Precompute indices for efficient updates
    def perturb_h(h, idx, delta):
        h_new = h.copy()
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Use differential evolution with bounds to explore
    bounds = [(0, 1) for _ in range(n_points)]
    
    optimizer = differential_evolution(
        lambda x: compute_c5(x), 
        bounds,
        maxiter=1000,
        popsize=15,
        seed=seed,
        disp=False
    )
    
    # Initial evaluation
    initial_c5 = compute_c5(h)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
        best_h = h.copy()
    
    # Run optimization
    start_time = time.time()
    eval_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Do a few differential evolution steps
        optimizer = differential_evolution(
            lambda x: compute_c5(x), 
            bounds,
            maxiter=50,
            popsize=15,
            seed=int(time.time()) % 1000000,
            disp=False
        )
        
        if optimizer.success and optimizer.fun < best_c5:
            best_c5 = optimizer.fun
            best_h = optimizer.x.copy()
            
        eval_count += 1
        
        # Occasionally try random adjustments
        if eval_count % 10 == 0:
            h_trial = best_h.copy()
            # Mutate a few randomly selected points
            for _ in range(5):
                idx = np.random.randint(n_points)
                delta = np.random.uniform(-0.05, 0.05)
                h_trial = perturb_h(h_trial, idx, delta)
            
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                
        if best_c5 < 0.3807:  # Early stopping if target met
            break
            
    # Final refinement with gradient descent
    def compute_gradient(h, eps=1e-4):
        grad = np.zeros_like(h)
        base_c5 = compute_c5(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Run gradient ascent steps if improvement is possible
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        grad = compute_gradient(best_h)
        lr = 0.01
        h_new = best_h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        else:
            break
            
    # Final computation
    final_c5 = compute_c5(best_h)
            
    return (best_h.tolist(), final_c5, n_points)
