import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def correlation(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    num_starts = 5
    max_iter_per_start = 500
    
    for start in range(num_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize with slight noise
        if start == 0:
            h_start = h.copy()
        else:
            h_start = h + np.random.normal(0, 0.02, n_points)
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
            
        h_current = h_start.copy()
        
        for iteration in range(max_iter_per_start):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute C5
            c5 = correlation(h_current, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_current.copy()
                
            # Gradient approximation using finite differences
            grad = np.zeros_like(h_current)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = clip_and_scale(h_plus, n_points)
                
                c5_plus = correlation(h_plus, dx)
                grad[i] = (c5_plus - c5) / eps
            
            # Update using adaptive learning rate
            lr = 0.005
            h_new = h_current - lr * grad
            
            # Clip and rescale
            h_new = clip_and_scale(h_new, n_points)
            h_current = h_new
            
    return (best_h.tolist(), best_c5, n_points)
