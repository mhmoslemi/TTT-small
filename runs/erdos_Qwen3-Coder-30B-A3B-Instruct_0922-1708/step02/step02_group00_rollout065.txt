import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_new = np.linspace(0, 2, n_points)
            h_base = np.interp(x_new, x_orig, h_base)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        population.append(h_base.copy())
        for _ in range(population_size - 1):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Start with random spikes and small variations
        for _ in range(population_size):
            h = np.zeros(n_points)
            # Place 5 spikes at random locations
            positions = np.random.choice(n_points, 5, replace=False)
            for i in positions:
                h[i] = np.random.uniform(0.5, 1.0)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < 1000:
        generation += 1
        
        # Evaluate fitness of current population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Create new population
        new_population = []
        # Keep top 10% (elitism)
        elite_count = max(1, population_size // 10)
        new_population.extend(population[:elite_count])
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness, 3)
            parent2 = tournament_selection(population, fitness, 3)
            
            # Crossover with 70% probability
            if np.random.rand() < 0.7:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1, 0.15)
            child2 = mutate(child2, 0.15)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    return (best_h.tolist(), best_c5, n_points)

def tournament_selection(population, fitness, k):
    """Select a parent using tournament selection"""
    idx = np.random.choice(len(population), k)
    best_idx = idx[np.argmin([fitness[i] for i in idx])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    """Single-point crossover"""
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Normalize sum to match constraint
    child1 = child1 * (len(child1) / 2) / np.sum(child1)
    child2 = child2 * (len(child2) / 2) / np.sum(child2)
    
    return child1, child2

def mutate(h, rate):
    """Gaussian mutation with normalization"""
    h = h.copy()
    n = len(h)
    mutations = np.random.choice(n, int(n * rate), replace=False)
    for i in mutations:
        h[i] += np.random.normal(0, 0.1)
        h[i] = np.clip(h[i], 0, 1)
    
    # Normalize sum
    h = h * (len(h) / 2) / np.sum(h)
    return h
