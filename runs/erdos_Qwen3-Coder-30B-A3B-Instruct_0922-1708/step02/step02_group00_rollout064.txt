import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Annealing parameters
    T0 = 0.1
    tau = 1000
    min_T = 1e-6
    
    iteration = 0
    T = T0
    
    while time.time() - start_time < budget_s - 1 and T > min_T:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate candidate
        noise_strength = T * 0.1
        h_prime = h + np.random.normal(0, noise_strength, n_points)
        h_prime = np.clip(h_prime, 0, 1)
        h_prime = h_prime * (n_points / 2) / np.sum(h_prime)
        
        # Compute candidate C5
        correlation_prime = np.correlate(h_prime, 1 - h_prime, mode='full')
        candidate_c5 = np.max(correlation_prime) * dx
        
        # Accept or reject
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_prime
        
        # Update temperature
        T = T0 / (1 + iteration / tau)
    
    return (best_h.tolist(), best_c5, n_points)
