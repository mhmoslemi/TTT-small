import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    decrement = 0.01
    
    # Adversarial refinement approach: iteratively minimize worst overlap
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlaps for all k values
        # Use the full convolution to get all correlations
        correlation = np.correlate(h, 1-h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
            decrement = 0.01  # Reset decrement on improvement
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                decrement *= 0.5
                no_improvement_count = 0
                decrement = max(decrement, 1e-4)
        
        # Find the shift that causes the maximum overlap
        # The length of correlation is 2*n_points - 1
        # The valid shifts are from 0 to 2*n_points - 2
        # Real shifts k are from 0 to n_points - 1 (considering wraparound)
        # However, we care about the maximum value which can come from any valid k
        max_idx = np.argmax(correlation)
        k_max = max_idx - (n_points - 1)  # In range [-n_points+1, n_points-1]
        if k_max < 0:
            k_max += n_points  # Map to [0, n_points-1]
        
        # Now we want to add a constraint that this particular overlap is reduced
        # Let's do a small perturbation or gradient step to reduce max overlap
        
        # Compute finite differences for gradient estimation
        grad = np.zeros_like(h)
        eps = 1e-4
        
        # We compute gradients via finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1-h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adjust this: instead of gradient descent, solve constraint-based problem
        # Learn about shift k_max and constrain S(k_max) < c5 - decrement
        
        lr = 0.01
        h_new = h - lr * grad
        
        # Clip values
        h_new = np.clip(h_new, 0, 1)
        # Normalize to correct sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally perturb with an auxiliary method to explore
        if iteration % 50 == 0 and best_c5 < 0.381:
            # Add a small random perturbation to escape local minima
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final check
    correlation = np.correlate(best_h, 1-best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
