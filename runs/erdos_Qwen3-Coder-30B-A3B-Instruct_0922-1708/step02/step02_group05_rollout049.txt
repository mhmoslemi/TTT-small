import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    swarm_size = 50
    c1, c2 = 2.0, 2.0
    w_start, w_end = 0.9, 0.4
    
    # Initialize swarm
    positions = np.random.rand(swarm_size, n_points) * 0.5
    velocities = (np.random.rand(swarm_size, n_points) - 0.5) * 0.1
    
    # Normalize and scale initial positions
    for i in range(swarm_size):
        positions[i] = np.clip(positions[i], 0, 1)
        positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
    
    personal_best_positions = positions.copy()
    personal_best_values = np.full(swarm_size, float('inf'))
    global_best_position = None
    global_best_value = float('inf')
    
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Evaluate initial swarm
    for i in range(swarm_size):
        c5 = compute_c5(positions[i])
        personal_best_values[i] = c5
        if c5 < global_best_value:
            global_best_value = c5
            global_best_position = positions[i].copy()
    
    # PSO loop
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / 1000
        
        # Update each particle
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best_positions[i] - positions[i]) + 
                            c2 * r2 * (global_best_position - positions[i]))
            
            # Update position
            positions[i] += velocities[i]
            
            # Apply constraints
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
            
            # Evaluate
            c5 = compute_c5(positions[i])
            
            # Update personal best
            if c5 < personal_best_values[i]:
                personal_best_values[i] = c5
                personal_best_positions[i] = positions[i].copy()
                
                # Update global best
                if c5 < global_best_value:
                    global_best_value = c5
                    global_best_position = positions[i].copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            
        # Maintain diversity
        if stagnation_count > max_stagnation:
            for i in range(swarm_size // 10):
                idx = np.random.randint(swarm_size)
                positions[idx] = np.random.rand(n_points) * 0.5
                positions[idx] = positions[idx] * (n_points / 2) / np.sum(positions[idx])
                velocities[idx] = (np.random.rand(n_points) - 0.5) * 0.1
            stagnation_count = 0
            
        # Early exit if target reached
        if global_best_value <= 0.3807:
            break
            
    # Final evaluation of the best solution to ensure it's correct
    final_c5 = compute_c5(global_best_position)
    
    return (global_best_position.tolist(), final_c5, n_points)
