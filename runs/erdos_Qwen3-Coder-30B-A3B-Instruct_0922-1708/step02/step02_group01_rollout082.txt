import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Multi-start with 10 particles
    particles = []
    for i in range(10):
        if i == 0 and 'initial_h_values' in globals():
            h_start = h.copy()
        else:
            h_start = np.random.uniform(0, 1, n_points)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
        particles.append(h_start)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Adaptive learning rate and momentum
    lr = 0.01
    momentum = 0.9
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update each particle
        for p_id, h in enumerate(particles):
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Compute gradient with finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
                
            # Adaptive step size and momentum
            velocity = momentum * velocity - lr * grad if iteration > 1 else -lr * grad
            
            h_new = h + velocity
            
            # Add noise with 5% probability to escape local minima
            if np.random.random() < 0.05:
                h_new += np.random.normal(0, 0.01, n_points)
                
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[p_id] = h_new
            
        # Occasionally restart particles with best solution
        if iteration % 50 == 0 and best_c5 < 0.381:
            for p_id in range(10):
                particles[p_id] = best_h.copy()
                particles[p_id] = np.clip(particles[p_id], 0, 1)
                particles[p_id] = particles[p_id] * (n_points / 2) / np.sum(particles[p_id])
                
    return (best_h.tolist(), best_c5, n_points)
