import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    learning_rate = 0.01
    eps = 1e-6
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 0.9
    inertia = w
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h.copy()
            particle += np.random.normal(0, 0.01, n_points)
        else:
            particle = np.random.uniform(0, 1, n_points)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_vals.append(float('inf'))
    
    gbest = h.copy()
    gbest_val = float('inf')
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        for i in range(n_particles):
            particle = particles[i]
            
            # Compute the overlap integral
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_vals[i]:
                pbest_vals[i] = c5
                pbest[i] = particle.copy()
                
            if c5 < gbest_val:
                gbest_val = c5
                gbest = particle.copy()
                
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
        
        # Update particles using PSO
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            r1, r2 = np.random.random(2)
            vel = inertia * vel + c1 * r1 * (pbest[i] - particle) + c2 * r2 * (gbest - particle)
            particle = particle + vel
            
            # Clip and rescale
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            particles[i] = particle
            velocities[i] = vel
            
        # Occasionally apply gradient-based refinement
        if iteration % 100 == 0:
            # Refine global best
            h_ref = gbest.copy()
            for _ in range(20):
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_ref.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - gbest_val) / eps
                
                h_new = h_ref - learning_rate * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_ref = h_new
                
                # Recompute and check improvement
                corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                new_c5 = np.max(corr_new) * dx
                if new_c5 < gbest_val:
                    gbest_val = new_c5
                    gbest = h_new.copy()
                    
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_new.copy()
                    break
            
        # Occasionally check for early termination
        if iteration % 500 == 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
