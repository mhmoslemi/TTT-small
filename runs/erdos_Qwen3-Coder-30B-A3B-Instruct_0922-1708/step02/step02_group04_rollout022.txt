import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use provided initial construction or create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Population size
    population_size = min(50, max(10, n_points // 2))
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            ind = np.array(initial_h_values)
        else:
            ind = np.random.rand(n_points)
            ind = ind / np.sum(ind) * (n_points / 2)
        population.append(ind)
    
    # Keep track of improvement
    no_improvement_count = 0
    max_no_improvement = 100
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for ind in population:
            corr = correlate(ind, 1 - ind, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Find and track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            # Reset population if no improvement for many generations
            population = []
            for _ in range(population_size):
                if 'initial_h_values' in globals():
                    ind = np.array(initial_h_values)
                else:
                    ind = np.random.rand(n_points)
                    ind = ind / np.sum(ind) * (n_points / 2)
                population.append(ind)
            no_improvement_count = 0
            
        # Selection based on inverse fitness (higher fitness = better)
        fitness = np.array(fitness)
        fitness = 1 / (fitness + 1e-10)  # Avoid division by zero
        fitness = np.clip(fitness, 0, 1e10)  # Clamp to reasonable values
        probs = fitness / np.sum(fitness)
        
        # Generate new population
        new_population = [best_h]  # Elitism: keep best individual
        
        while len(new_population) < population_size:
            # Select two parents
            parents = np.random.choice(population_size, size=2, p=probs)
            parent1 = population[parents[0]]
            parent2 = population[parents[1]]
            
            # Crossover: blend two parents with some probability
            if np.random.rand() < 0.5:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Average crossover
                child = (parent1 + parent2) / 2.0
                
            # Mutate
            if np.random.rand() < 0.1:  # 10% chance of mutation
                mutate_idx = np.random.randint(0, n_points)
                child[mutate_idx] += np.random.normal(0, 0.05)
                
            # Clamp to [0,1]
            child = np.clip(child, 0, 1)
            
            # Normalize to correct integral
            child = child / np.sum(child) * (n_points / 2)
            
            new_population.append(child)
            
        population = new_population
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
