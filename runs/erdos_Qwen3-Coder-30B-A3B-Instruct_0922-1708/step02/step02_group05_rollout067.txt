import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use a more sophisticated approach: adaptive gradient descent with momentum
    # and alternating between gradient steps and local searches
    velocity = np.zeros_like(h)
    momentum = 0.9
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Apply momentum to gradient
        velocity = momentum * velocity + (1 - momentum) * grad
        
        # Adaptive step size based on gradient norm
        grad_norm = np.linalg.norm(velocity)
        if grad_norm > 1e-10:
            adaptive_lr = min(learning_rate, 1.0 / (grad_norm + 1e-8))
        else:
            adaptive_lr = learning_rate
            
        # Update step with adaptive learning rate
        h_new = h - adaptive_lr * velocity
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Occasionally perform a local search using scipy
        if iteration % 10 == 0:
            # Local search around current solution using scipy minimize
            def obj(h_local):
                h_local = np.clip(h_local, 0, 1)
                h_local = h_local * (n_points / 2) / np.sum(h_local)
                correlation = np.correlate(h_local, 1 - h_local, mode='full')
                return np.max(correlation) * dx
                
            def constraint_sum(h_local):
                return np.sum(h_local) - n_points / 2
                
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                result = minimize(obj, h_new, method='SLSQP', bounds=bounds, constraints=cons, 
                                options={'maxiter': 5, 'disp': False})
                if result.success:
                    h_new = np.clip(result.x, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
            except:
                pass
                
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
