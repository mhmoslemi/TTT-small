import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    swarm_size = 50
    inertia_weight = 0.7
    cognitive_param = 2.0
    social_param = 2.0
    max_iterations = 500
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            pos = np.array(initial_h_values)
            if len(pos) != n_points:
                pos = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(pos)), pos)
            pos = np.clip(pos, 0, 1)
            pos = pos * (n_points / 2) / np.sum(pos)
        else:
            pos = np.random.rand(n_points)
            pos = pos * (n_points / 2) / np.sum(pos)
        particles.append(pos)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        pbest_positions.append(pos.copy())
        
        # Evaluate initial particle
        correlation = np.correlate(pos, 1 - pos, mode='full')
        value = np.max(correlation) * dx
        pbest_values.append(value)
        
        if value < best_c5:
            best_c5 = value
            best_h = pos.copy()
    
    # PSO main loop
    gbest_index = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_index]
    gbest_value = pbest_values[gbest_index]
    
    iteration = 0
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Update inertia weight
        w = 0.7 - (0.3 * iteration / max_iterations)
        
        for i in range(swarm_size):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = cognitive_param * r1 * (pbest_positions[i] - particles[i])
            social = social_param * r2 * (gbest_position - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            current_value = np.max(correlation) * dx
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest_positions[i] = particles[i].copy()
                
                # Update global best if needed
                if current_value < gbest_value:
                    gbest_value = current_value
                    gbest_position = particles[i].copy()
                    stagnation_count = 0
                    
                    if gbest_value < best_c5:
                        best_c5 = gbest_value
                        best_h = gbest_position.copy()
            else:
                stagnation_count += 1
        
        # Check for stagnation
        if stagnation_count > max_stagnation:
            # Perturb particles
            for i in range(swarm_size):
                if np.random.rand() < 0.1:  # Perturb 10% of particles
                    perturbation = np.random.normal(0, 0.1, n_points)
                    particles[i] += perturbation
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            stagnation_count = 0
            
        # Occasionally sample the best solution found so far
        if iteration % 50 == 0:
            correlation = np.correlate(gbest_position, 1 - gbest_position, mode='full')
            gbest_value = np.max(correlation) * dx
            if gbest_value < best_c5:
                best_c5 = gbest_value
                best_h = gbest_position.copy()
    
    # Final check of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Ensure final solution is correct
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
