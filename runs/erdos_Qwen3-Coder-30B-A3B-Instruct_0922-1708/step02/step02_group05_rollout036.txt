import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation phase (first 30% of time)
    convex_time = budget_s * 0.3
    if time.time() - start_time < convex_time - 1:
        # Simplified convex relaxation approach
        # We create an objective that approximates the behavior and solve
        try:
            # Use a simpler heuristic to initialize better
            # Try to make h as piecewise constant with one large peak
            h = np.zeros(n_points)
            mid = n_points // 2
            h[mid-10:mid+10] = 1.0
            h = h * (n_points / 2) / np.sum(h)
        except:
            pass
    
    # Main optimization loop (remaining time)
    while time.time() - start_time < budget_s - 1:
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adaptive learning rate with momentum
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update with momentum
        h_new = h - lr * grad
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new  # Overwrite h with updated version

    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    c5 = np.max(correlation) * dx
    return (best_h.tolist(), c5, n_points)
