import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    N_pop = 50
    C_rate = 0.8
    M_rate = 0.2
    disturbance_rate = 0.2
    tournament_size = 2
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        population.append(h_init)
    
    # Fill remaining population with random individuals
    while len(population) < N_pop:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Function to compute C5
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Evaluate initial population
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        # Sort population by fitness
        sort_idx = np.argsort(fitness)
        sorted_pop = [population[i] for i in sort_idx]
        sorted_fitness = [fitness[i] for i in sort_idx]
        
        # Keep top 50% (elitism)
        elite_size = N_pop // 2
        elite = sorted_pop[:elite_size]
        elite_fitness = sorted_fitness[:elite_size]
        
        # Generate new population
        new_population = elite[:]
        new_fitness = elite_fitness[:]
        
        # Tournament selection and crossover
        while len(new_population) < N_pop:
            # Tournament selection
            idx1 = np.random.randint(0, N_pop // 2)
            idx2 = np.random.randint(0, N_pop // 2)
            parent1 = sorted_pop[idx1]
            parent2 = sorted_pop[idx2]
            
            # Crossover
            if np.random.rand() < C_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < M_rate:
                    mutation_strength = 0.1
                    noise = np.random.uniform(-mutation_strength, mutation_strength, n_points)
                    child += noise
                    # Repair constraint
                    child = np.clip(child, 0, 1)
                    scale = (n_points / 2) / np.sum(child)
                    child = child * scale
                    
                new_population.append(child)
                new_fitness.append(compute_c5(child))
        
        # Trim to exact population size
        population = new_population[:N_pop]
        fitness = new_fitness[:N_pop]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Apply disturbance if no improvement for a while
        if no_improvement_count > 50:
            num_disturb = int(N_pop * disturbance_rate)
            for _ in range(num_disturb):
                idx = np.random.randint(0, N_pop)
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                population[idx] = h
                fitness[idx] = compute_c5(h)
            no_improvement_count = 0
            
        # Early exit if target achieved
        if best_c5 < 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
