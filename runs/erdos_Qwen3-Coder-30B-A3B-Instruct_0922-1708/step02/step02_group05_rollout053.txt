import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for given h array."""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Set up bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Equality constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the objective function for optimization
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Solver settings
    start_time = time.time()
    
    # Use a subset of the total budget for solving
    remaining_budget = budget_s - 1
    
    # Use differential evolution with constraints
    # This algorithm may be expensive but explores better
    result = differential_evolution(
        objective,
        bounds,
        constraints={'type': 'eq', 'fun': constraint},
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: None  # Just to avoid printing
    )
    
    if result.success and evaluate_c5(result.x, dx) < best_c5:
        best_h = result.x.copy()
        best_c5 = evaluate_c5(best_h, dx)
    
    # For remaining time, do some local refinement if needed
    # Try simpler local search starting from best
    # Keep only the best solution during the full time budget

    # Reevaluate best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final check on C5
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
