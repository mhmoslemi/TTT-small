import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Particle swarm with 10 particles
    n_particles = 10
    particles = []
    velocities = []
    best_particle = 0
    best_particle_c5 = float('inf')
    
    # Initialize particles
    for i in range(n_particles):
        if i == 0 and 'initial_h_values' in globals():
            # Use the initial construction for first particle
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_new = np.linspace(0, 2, n_points)
                h_init = np.interp(x_new, x_orig, h_init)
            h_init = h_init / np.sum(h_init) * (n_points / 2)
        else:
            # Random initialization
            h_init = np.random.uniform(0, 1, n_points)
            h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    
    # Optimization loop
    iteration = 0
    lr = 0.01
    momentum = 0.9
    perturb_prob = 0.05
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for p in range(n_particles):
            h = particles[p]
            
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            if c5 < best_particle_c5:
                best_particle = p
                best_particle_c5 = c5
            
            # Compute gradient using finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update particle position
            velocities[p] = momentum * velocities[p] - lr * grad
            particles[p] = particles[p] + velocities[p]
            
            # Apply constraints
            particles[p] = np.clip(particles[p], 0, 1)
            particles[p] = particles[p] * (n_points / 2) / np.sum(particles[p])
            
            # Occasionally perturb particles if no improvement
            if np.random.random() < perturb_prob:
                noise = np.random.normal(0, 0.05, n_points)
                particles[p] = particles[p] + noise
                particles[p] = np.clip(particles[p], 0, 1)
                particles[p] = particles[p] * (n_points / 2) / np.sum(particles[p])
            
            # Occasionally restart with best solution
            if iteration % 500 == 0 and best_c5 < 0.381:
                particles[p] = best_h.copy()
                particles[p] = np.clip(particles[p], 0, 1)
                particles[p] = particles[p] * (n_points / 2) / np.sum(particles[p])
                lr = min(lr * 1.1, 0.1)
        
        # Adjust learning rate based on progress
        if iteration % 100 == 0:
            lr = max(lr * 0.95, 0.001)
    
    return (best_h.tolist(), best_c5, n_points)
