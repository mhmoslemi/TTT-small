import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    else:
        h_init = np.ones(n_points) * 0.5
        
    # Multi-start optimization
    num_particles = 10
    particles = []
    best_c5 = float('inf')
    best_h = None
    
    # Initialize particles
    for i in range(num_particles):
        if i == 0:
            h = h_init.copy()
        else:
            # Perturb the initial solution for diversification
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        
    start_time = time.time()
    iteration = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Process each particle
        for i, h in enumerate(particles):
            # Compute the overlap integral for current particle
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - c5) / eps
            
            # Update with gradient descent
            h_new = h - lr * grad
            
            # Apply constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Occasionally perturb to escape local minima
            if np.random.random() < 0.05:
                h_new = h_new + np.random.normal(0, 0.02, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            
        # Adjust learning rate adaptively
        if iteration % 100 == 0:
            lr = max(lr * 0.95, 1e-5)
    
    return (best_h.tolist(), best_c5, n_points)
