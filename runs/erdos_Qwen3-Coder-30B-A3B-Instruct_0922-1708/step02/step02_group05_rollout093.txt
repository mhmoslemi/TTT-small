import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Try different start points to avoid local minima
    starts = [h]
    for _ in range(10):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        starts.append(h_rand)
    
    # Genetic algorithm approach
    pop_size = 50
    generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.5
    tournament_size = 3
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i < len(starts):
            individual = starts[i]
        else:
            individual = np.random.rand(n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    best_overall = best_h.copy()
    best_overall_c5 = best_c5
    
    # GA steps
    for gen in range(generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Track best in generation
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_overall_c5:
            best_overall_c5 = fitness[min_fitness_idx]
            best_overall = population[min_fitness_idx].copy()
            
        # Selection, crossover, mutation
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(population[min_fitness_idx])
        
        while len(new_population) < pop_size:
            # Tournament selection
            selected = []
            for _ in range(2):  # For two parents
                tournament = np.random.choice(pop_size, tournament_size)
                best_in_tourney = tournament[np.argmin([fitness[i] for i in tournament])]
                selected.append(population[best_in_tourney])
            
            # Crossover
            parent1, parent2 = selected
            child1, child2 = parent1.copy(), parent2.copy()
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points)
                child1[:crossover_point] = parent2[:crossover_point]
                child2[:crossover_point] = parent1[:crossover_point]
                
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    # Add Gaussian noise
                    child1[i] += np.random.normal(0, 0.05)
                    # Clamp to [0,1] and rescale
                    child1 = np.clip(child1, 0, 1)
                    child1 = child1 * (n_points / 2) / np.sum(child1)
                
                if np.random.rand() < mutation_rate:
                    # Add Gaussian noise
                    child2[i] += np.random.normal(0, 0.05)
                    # Clamp to [0,1] and rescale
                    child2 = np.clip(child2, 0, 1)
                    child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        population = new_population[:pop_size]
    
    # Final evaluation
    final_correlation = np.correlate(best_overall, 1 - best_overall, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (best_overall.tolist(), final_c5, n_points)
