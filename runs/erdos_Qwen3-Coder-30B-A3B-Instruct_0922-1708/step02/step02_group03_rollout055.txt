import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    # Learning rate
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        t += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute exact gradient
        max_k = np.argmax(correlation)
        grad = np.zeros_like(h)
        
        # Exact gradient computation for each point
        for i in range(n_points):
            # Compute partial derivative w.r.t. h[i]
            # For shift k that maximizes correlation
            # We only consider the terms that depend on h[i]
            k = max_k - n_points + 1
            
            # Compute the derivative contribution
            temp_h = h.copy()
            temp_h[i] += 1e-6  # Small perturbation
            temp_h = np.clip(temp_h, 0, 1)
            temp_h = temp_h * (n_points / 2) / np.sum(temp_h)  # Rescale
            
            # Manual correlation computation for just one point change
            # We don't actually compute the full correlation for efficiency
            temp_corr = np.zeros(2 * n_points - 1)
            for j in range(n_points):
                temp_corr[(j - i) % (2 * n_points - 1)] += temp_h[j] * (1 - temp_h[i]) if i != j else temp_h[j] * (1 - temp_h[j])
            
            temp_c5 = np.max(temp_corr) * dx
            grad[i] = (temp_c5 - c5) / 1e-6
            
        # Update using Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Occasionally check for early termination
        if iteration % 100 == 0 and iteration > 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
