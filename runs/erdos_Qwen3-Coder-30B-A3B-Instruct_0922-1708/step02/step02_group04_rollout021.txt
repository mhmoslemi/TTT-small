import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 50
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize GA parameters
    pop_size = min(50, max(20, n_points))
    generations = 1000
    mutation_rate = 0.1
    elite_size = pop_size // 5
    if elite_size < 1:
        elite_size = 1
    
    # Create initial population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        h_new = np.random.rand(n_points)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        population.append(h_new)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    no_improvement_count = 0
    max_no_improvement = 100
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all population members
        fitness = []
        for ind in population:
            corr = correlate(ind, 1 - ind, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind.copy()
                no_improvement_count = 0
        
        no_improvement_count += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        while len(new_population) < pop_size:
            # Tournament selection (2 participants)
            idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover
            if np.random.rand() < 0.5:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                offspring = np.where(mask, parent1, parent2)
            else:
                # Arithmetic crossover
                alpha = np.random.rand()
                offspring = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mut_indices = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
                for i in mut_indices:
                    offspring[i] += np.random.normal(0, 0.05)
            
            # Ensure bounds
            offspring = np.clip(offspring, 0, 1)
            
            # Normalize
            offspring = offspring * (n_points / 2) / np.sum(offspring)
            
            new_population.append(offspring)
        
        population = new_population
        
        # Adaptive mutation rate
        if no_improvement_count > 20:
            mutation_rate = min(0.5, mutation_rate + 0.05)
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
