import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 45
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize initial vector to sum to n_points/2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h_init = np.clip(h_init, 0, 1)
    
    # GA parameters
    pop_size = min(50, max(20, n_points // 2))
    generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_count = pop_size // 10
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            population.append(h_init.copy())
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    
    # Track best solution
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    no_improvement_count = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Update best if needed
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_c5 = fitness[min_idx]
            best_h = population[min_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Early stopping
        if no_improvement_count > 100:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_pop = sorted_pop[:elite_count]
        
        # Generate offspring
        while len(new_pop) < pop_size:
            # Tournament selection
            parent1 = sorted_pop[np.random.randint(elite_count)]
            parent2 = sorted_pop[np.random.randint(elite_count)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, 0.03)
                
                # Clip and rescale
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
                # Add to population
                new_pop.append(child)
                
        population = new_pop[:pop_size]
    
    # Final validation of best solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
