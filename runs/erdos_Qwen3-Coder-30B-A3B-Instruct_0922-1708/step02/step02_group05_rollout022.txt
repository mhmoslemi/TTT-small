import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize swarm
    particles = [h.copy() for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.5, 0.5, n_points) for _ in range(swarm_size)]
    
    pbest_positions = [p.copy() for p in particles]
    pbest_values = [float('inf')] * swarm_size
    
    gbest_position = h.copy()
    gbest_value = float('inf')
    
    # Initialize PSO loop
    for iter_idx in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iter_idx / max_iter)
        
        # Evaluate all particles
        for i in range(swarm_size):
            particle = particles[i]
            # Ensure particle is feasible
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Compute correlation
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_values[i]:
                pbest_values[i] = c5
                pbest_positions[i] = particle.copy()
                
            if c5 < gbest_value:
                gbest_value = c5
                gbest_position = particle.copy()
            
            # Update particles
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (pbest_positions[i] - particles[i]) +
                            c2 * r2 * (gbest_position - particles[i]))
            
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
        if gbest_value < best_c5:
            best_c5 = gbest_value
            best_h = gbest_position.copy()
            
    return (best_h.tolist(), best_c5, n_points)
