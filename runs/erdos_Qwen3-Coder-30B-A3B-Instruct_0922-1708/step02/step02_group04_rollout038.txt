import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Optimization approach inspired by simulated annealing with restarts
    # and structured search using gradient information
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current C5 value using correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Try structured perturbations
        h_new = h.copy()
        
        # 1. Small random perturbation
        perturb = np.random.normal(0, 0.01, n_points)
        h_new += perturb
        
        # 2. Clip to feasible range
        h_new = np.clip(h_new, 0, 1)
        
        # 3. Rescale to maintain integral = 1
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # 4. Occasionally make larger changes
        if np.random.rand() < 0.1:
            # Modify a single point
            i = np.random.randint(n_points)
            h_new[i] = np.random.rand()
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            
        # Test with this new point
        h = h_new
    
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
