import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
        for _ in range(population_size - 1):
            mutation_rate = 0.15
            h_mut = h.copy()
            indices = np.random.choice(n_points, size=int(mutation_rate * n_points), replace=False)
            for i in indices:
                h_mut[i] += np.random.uniform(-0.2, 0.2)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            population.append(h_mut)
    else:
        for _ in range(population_size):
            # Random spikes pattern
            h = np.zeros(n_points)
            num_spikes = np.random.randint(5, 11)
            indices = np.random.choice(n_points, size=num_spikes, replace=False)
            for i in indices:
                h[i] = np.random.uniform(0.5, 1.0)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    def evaluate_h(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def mutate(h, rate=0.15):
        h_new = h.copy()
        indices = np.random.choice(n_points, size=int(rate * n_points), replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-0.2, 0.2)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def crossover(h1, h2):
        crossover_point = np.random.randint(1, n_points-1)
        h_new1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
        h_new2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
        h_new1 = h_new1 * (n_points / 2) / np.sum(h_new1)
        h_new2 = h_new2 * (n_points / 2) / np.sum(h_new2)
        return h_new1, h_new2
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        if generation > 1000:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        # Selection (top 50%)
        selected = population[:population_size//2]
        
        # Generate new population
        new_population = selected.copy()
        
        # Elitism: keep best
        new_population.append(population[0])
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < 0.7:
                idx1, idx2 = np.random.choice(len(selected), size=2, replace=False)
                h1, h2 = crossover(selected[idx1], selected[idx2])
                new_population.append(h1)
                if len(new_population) < population_size:
                    new_population.append(h2)
            else:
                # Mutation
                idx = np.random.choice(len(selected))
                h_mut = mutate(selected[idx])
                new_population.append(h_mut)
        
        population = new_population
    
    return (best_h.tolist(), best_c5, n_points)
