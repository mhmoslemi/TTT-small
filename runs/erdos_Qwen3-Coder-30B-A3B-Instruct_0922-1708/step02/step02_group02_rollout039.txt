import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
    else:
        h_base = np.ones(n_points) * 0.5
    
    # Create initial population
    for _ in range(population_size):
        h = h_base + np.random.uniform(-0.05, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate current population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 20%
        elite_size = population_size // 5
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = population[np.random.randint(0, population_size // 2)]
            parent2 = population[np.random.randint(0, population_size // 2)]
            
            # Crossover
            if np.random.random() < 0.5:
                # Uniform crossover
                mask = np.random.random(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Arithmetic crossover
                alpha = np.random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.random() < 0.3:  # mutation rate
                mutation_indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
                for i in mutation_indices:
                    child[i] += np.random.normal(0, 0.02)
            
            # Clip and rescale
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        return (best_h.tolist(), final_c5, n_points)
    
    # If no improvement found, return original
    h_orig = h_base if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
    corr = correlate(h_orig, 1 - h_orig, mode='full')
    final_c5 = np.max(corr) * dx
    return (h_orig.tolist(), final_c5, n_points)
