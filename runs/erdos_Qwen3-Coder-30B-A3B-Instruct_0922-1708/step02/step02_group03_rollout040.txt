import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    learning_rate = 0.005
    eps = 1e-6
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 0.9
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    gbest = None
    gbest_c5 = float('inf')
    
    # Initialize PSO swarm
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h.copy()
            particle += np.random.normal(0, 0.05, n_points)
        else:
            particle = np.random.uniform(0, 1, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_c5.append(float('inf'))
    
    # PSO iterations
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update particles
        for i in range(n_particles):
            # Compute fitness
            particle = particles[i]
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particle.copy()
                pbest_c5[i] = c5
                
            # Update global best
            if c5 < gbest_c5:
                gbest = particle.copy()
                gbest_c5 = c5
                
            # Update best solution so far
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
            
            # Update velocity and position
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particle) + 
                            c2 * r2 * (gbest - particle))
            particles[i] = particle + velocities[i]
            
            # Clip and rescale
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
        # Occasionally do local refinement
        if iteration % 50 == 0:
            # Do gradient-based refinement on gbest
            for _ in range(10):
                # Compute gradient
                grad = np.zeros_like(gbest)
                for j in range(n_points):
                    h_plus = gbest.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - gbest_c5) / eps
                
                # Update gbest
                gbest_new = gbest - learning_rate * grad
                gbest_new = np.clip(gbest_new, 0, 1)
                gbest_new = gbest_new * (n_points / 2) / np.sum(gbest_new)
                
                # Evaluate
                corr = np.correlate(gbest_new, 1 - gbest_new, mode='full')
                new_c5 = np.max(corr) * dx
                
                if new_c5 < gbest_c5:
                    gbest = gbest_new
                    gbest_c5 = new_c5
                    if gbest_c5 < best_c5:
                        best_c5 = gbest_c5
                        best_h = gbest.copy()
                        
                # Sometimes make larger steps for faster exploration
                if np.random.random() < 0.3:
                    gbest = gbest_new
                    
        # Early termination
        if iteration % 100 == 0 and iteration > 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
