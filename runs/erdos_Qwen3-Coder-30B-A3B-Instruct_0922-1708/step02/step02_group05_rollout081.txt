import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to target sum
    h = h * target_sum / np.sum(h)
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation step
    if time.time() - start_time < budget_s * 0.6:
        # Formulating the convex relaxation
        h_cvx = Variable(n_points)
        tau = Variable()
        
        # Constraints
        constraints = [0 <= h_cvx, h_cvx <= 1, sum(h_cvx) == target_sum]
        
        # Adding constraints for all valid shifts
        for k in range(n_points):
            # Compute the correlation for shift k manually
            corr_sum = 0
            valid = 0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    corr_sum += h_cvx[i] * (1 - h_cvx[j])
                    valid += 1
            if valid > 0:
                constraints.append(corr_sum <= tau * valid)
        
        # Objective
        objective = Minimize(tau)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
            if prob.status == 'optimal':
                h_relaxed = h_cvx.value
                h_relaxed = np.clip(h_relaxed, 0, 1)
                h_relaxed = h_relaxed * target_sum / np.sum(h_relaxed)
                h = h_relaxed.copy()
        except:
            pass
    
    # Gradient descent refinement
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute correlation with 1-h
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        h = h_new
    
    return (best_h.tolist(), best_c5, n_points)
