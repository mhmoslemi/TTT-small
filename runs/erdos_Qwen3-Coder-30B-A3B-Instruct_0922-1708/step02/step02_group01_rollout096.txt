import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def genetic_algorithm(n_points=100, budget_s=600):
    # Parameters
    N_pop = 50
    C_rate = 0.8
    M_rate = 0.2
    elitism_rate = 0.5
    disturbance_rate = 0.2
    tournament_size = 2
    
    dx = 2.0 / n_points
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h.copy())
    
    while len(population) < N_pop:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    population = np.array(population)
    
    best_h = None
    best_c5 = float('inf')
    generation = 0
    stagnation_count = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
        
        if best_c5 < 0.3807:
            break
            
        # Selection: tournament
        selected = []
        for _ in range(N_pop):
            tournament = np.random.choice(N_pop, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = []
        
        # Elitism: keep top individuals
        elites = [population[i] for i in np.argsort(fitness)[:int(N_pop * elitism_rate)]]
        new_population.extend(elites)
        
        # Generate offspring
        while len(new_population) < N_pop:
            # Crossover
            if np.random.rand() < C_rate and len(selected) >= 2:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                children = [child1, child2]
            else:
                # No crossover, just copy one parent
                parent = np.random.choice(selected)
                children = [parent.copy()]
                
            # Mutation
            for child in children:
                # Apply mutation
                for i in range(n_points):
                    if np.random.rand() < M_rate:
                        # Add noise
                        child[i] += np.random.uniform(-0.1, 0.1)
                        # Ensure within bounds
                        child[i] = max(0, min(1, child[i]))
                        
                # Rescale to maintain constraint
                scale_factor = (n_points / 2) / np.sum(child)
                child = child * scale_factor
                new_population.append(child)
                
                if len(new_population) >= N_pop:
                    break
        
        # Ensure population size
        while len(new_population) > N_pop:
            new_population.pop()
            
        population = np.array(new_population)
        
        # Check for stagnation and disturbance
        stagnation_count += 1
        if stagnation_count > 50:
            # Disturbation: add random individuals
            for i in range(int(N_pop * disturbance_rate)):
                h = np.random.uniform(0, 1, n_points)
                h = h / np.sum(h) * (n_points / 2)
                idx = np.random.randint(0, N_pop)
                population[idx] = h
            stagnation_count = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h, c5 = genetic_algorithm(budget_s=budget_s)
    return (h.tolist(), c5, 100)
