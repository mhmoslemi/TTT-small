import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a binary comb pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Binary comb pattern with 5 pulses
        h = np.zeros(n_points)
        pulse_width = 0.2
        pulse_spacing = 0.25
        pulses = 5
        for i in range(pulses):
            start = i * (pulse_width + pulse_spacing)
            end = start + pulse_width
            start_idx = int(start / 2 * n_points)
            end_idx = int(end / 2 * n_points)
            if start_idx < n_points:
                h[start_idx:end_idx] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use scipy.optimize for better convergence
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = -np.max(correlation) * dx  # Negative for minimization
        return c5
    
    def constraint_func(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization step
    try:
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, 
                         options={'maxiter': 100})
        if result.success:
            h = result.x.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    # Refinement with local iterations
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Try local optimizations
        h_new = h.copy()
        
        # Random small perturbations
        for i in range(n_points):
            if np.random.random() < 0.1:
                perturb = np.random.normal(0, 0.01)
                h_new[i] += perturb
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final computation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
