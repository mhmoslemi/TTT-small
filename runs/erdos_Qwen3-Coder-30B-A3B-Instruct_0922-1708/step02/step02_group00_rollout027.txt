import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    """Ensure h is in [0,1] and sums to n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    """Compute the maximum overlap integral"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    num_starts = 5
    start_points = [h.copy()]
    
    # Generate variations of initial solution
    for _ in range(num_starts - 1):
        h_var = h + np.random.normal(0, 0.05, n_points)
        h_var = clip_and_scale(h_var, n_points)
        start_points.append(h_var)
    
    # Run optimization from different initial points
    for start_idx, h_start in enumerate(start_points):
        h = h_start.copy()
        # Adam-like update parameters
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        alpha = 0.001
        t = 0
        
        while time.time() - start_time < budget_s - 1 and iteration < 500:
            iteration += 1
            t += 1
            
            # Compute the overlap integral
            c5 = compute_c5(h, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Estimate gradient via finite differences
            grad = np.zeros_like(h)
            eps_val = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps_val
                h_plus = clip_and_scale(h_plus, n_points)
                
                c5_plus = compute_c5(h_plus, dx)
                grad[i] = (c5_plus - c5) / eps_val
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            h_new = h - alpha * m_hat / (np.sqrt(v_hat) + eps)
            
            # Ensure feasibility
            h_new = clip_and_scale(h_new, n_points)
            h = h_new
    
    return (best_h.tolist(), best_c5, n_points)
