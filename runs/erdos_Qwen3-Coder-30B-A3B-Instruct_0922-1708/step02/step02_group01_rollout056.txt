import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Multiple particles for better exploration
    n_particles = 10
    particles = []
    for i in range(n_particles):
        if i == 0 and 'initial_h_values' in globals():
            # Use initial construction for first particle
            h0 = h.copy()
        else:
            # Random initialization for others
            h0 = np.random.rand(n_points)
            h0 = h0 / np.sum(h0) * (n_points / 2)
        particles.append(h0)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Adaptive learning rates
    lr_schedule = [0.01, 0.005, 0.002, 0.001]
    lr_idx = 0
    
    # Particle tracking
    particle_c5s = [float('inf')] * n_particles
    particle_best_h = [p.copy() for p in particles]
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update all particles
        for p_idx, particle_h in enumerate(particles):
            # Compute the overlap integral
            correlation = np.correlate(particle_h, 1 - particle_h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < particle_c5s[p_idx]:
                particle_c5s[p_idx] = c5
                particle_best_h[p_idx] = particle_h.copy()
                lr_idx = min(lr_idx + 1, len(lr_schedule) - 1)
                
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle_h.copy()
                lr_idx = min(lr_idx + 1, len(lr_schedule) - 1)
            
            # Compute gradient with finite differences
            grad = np.zeros_like(particle_h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = particle_h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Adaptive step size and momentum
            lr = lr_schedule[lr_idx]
            momentum = 0.9
            
            # Occasionally perturb with noise
            if np.random.rand() < 0.05:
                noise = np.random.normal(0, lr * 0.1, n_points)
                particle_h = particle_h + noise
            
            # Gradient update
            velocity = momentum * (particle_h - particle_h) - lr * grad
            h_new = particle_h + velocity
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[p_idx] = h_new
            
            # Occasionally restart particles with best points if we've made good progress
            if iteration % 200 == 0 and best_c5 < 0.381:
                for i in range(n_particles):
                    particles[i] = particle_best_h[i].copy()
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                lr_idx = 0
        
        # Periodically improve global best
        if iteration % 50 == 0:
            for p_idx in range(n_particles):
                if particle_c5s[p_idx] < best_c5:
                    best_c5 = particle_c5s[p_idx]
                    best_h = particle_best_h[p_idx].copy()
    
    # Final computation of C5 for the best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
