import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    max_points = 100
    min_points = 45
    n_points = min_points
    
    # Use provided initial construction or create symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Ensure initial integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Adaptive optimization approach
    while time.time() - start_time < budget_s - 1:
        # Sample critical shifts for constraints
        shifts = []
        # Sample shifts around critical regions
        num_shifts = min(20, n_points // 2)
        for i in range(num_shifts):
            k = np.random.randint(1, n_points // 2)
            shifts.append(k)
        
        # Create variables for optimization
        h_vars = Variable(n_points)
        C = Variable()
        
        # Define constraints
        constraints = [h_vars >= 0, h_vars <= 1]
        
        # Integral constraint
        constraints.append(sum(h_vars) == n_points / 2)
        
        # Overlap constraints
        for k in shifts:
            # Compute convolution
            overlap_expr = 0
            for i in range(n_points):
                if i + k < n_points:
                    overlap_expr += h_vars[i] * (1 - h_vars[i+k])
            constraints.append(overlap_expr <= C * dx)
        
        # Objective
        objective = Minimize(C)
        prob = Problem(objective, constraints)
        
        try:
            # Warm-start with previous solution
            for i in range(n_points):
                h_vars[i].value = best_h[i]
            prob.solve(solver=CVXOPT, verbose=False, max_iters=1000)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array([h_vars[i].value for i in range(n_points)])
                h_new = np.clip(h_new, 0, 1)
                
                # Rescale to meet integral constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Update best solution
                correlation = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(correlation) * dx
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    
                h = h_new.copy()
                
        except:
            # Fallback to simple perturbation
            perturb = np.random.normal(0, 0.02, n_points)
            h_new = h + perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
        
        # Occasionally increase resolution for better exploration
        if np.random.rand() < 0.1 and n_points < max_points:
            n_points += 5
            dx = 2.0 / n_points
            h = np.interp(np.linspace(0, 2, n_points), 
                         np.linspace(0, 2, len(h)), h)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
