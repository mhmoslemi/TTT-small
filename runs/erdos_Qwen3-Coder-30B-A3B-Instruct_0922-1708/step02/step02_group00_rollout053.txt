import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    """Ensure h is in [0,1] and sums to n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def evaluate_c5(h, dx):
    """Compute C5 = max correlation of h with (1-h)"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Multi-start strategy
    n_starts = 5
    for start in range(n_starts):
        # Perturb the starting point
        if start == 0:
            h_current = h.copy()
        else:
            # Add small noise
            h_current = h + np.random.normal(0, 0.02, n_points)
            h_current = clip_and_scale(h_current, n_points)
        
        # Adam optimizer parameters
        m_t = np.zeros_like(h_current)
        v_t = np.zeros_like(h_current)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        alpha = 0.001
        
        # Optimization loop
        for i in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute C5 for current point
            c5_current = evaluate_c5(h_current, dx)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
            
            # Compute finite difference gradient
            grad = np.zeros_like(h_current)
            eps_fd = 1e-5
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += eps_fd
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[j] = (c5_plus - c5_current) / eps_fd
            
            # Adam update
            m_t = beta1 * m_t + (1 - beta1) * grad
            v_t = beta2 * v_t + (1 - beta2) * grad**2
            m_hat = m_t / (1 - beta1**(i+1))
            v_hat = v_t / (1 - beta2**(i+1))
            
            # Update parameters
            h_current = h_current - alpha * m_hat / (np.sqrt(v_hat) + eps)
            
            # Ensure feasibility
            h_current = clip_and_scale(h_current, n_points)
        
        iteration += 500
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h.tolist(), best_c5, n_points)
