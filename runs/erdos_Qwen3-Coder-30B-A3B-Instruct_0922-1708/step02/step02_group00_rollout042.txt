import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_generations = 1000
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        # Generate variations
        for _ in range(population_size - 1):
            h_var = h_init + np.random.normal(0, 0.05, n_points)
            h_var = np.clip(h_var, 0, 1)
            h_var = h_var * (n_points / 2) / np.sum(h_var)
            population.append(h_var)
    else:
        # Initialize with random spikes
        for _ in range(population_size):
            h = np.zeros(n_points)
            num_spikes = np.random.randint(5, 11)
            spike_positions = np.random.randint(0, n_points, num_spikes)
            spike_heights = np.random.rand(num_spikes)
            for i, pos in enumerate(spike_positions):
                h[pos] = spike_heights[i]
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Create new population
        new_population = []
        # Elitism: Keep top 10%
        elite_size = population_size // 10
        new_population.extend(population[:elite_size])
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = population[np.random.randint(0, population_size // 2)]
            parent2 = population[np.random.randint(0, population_size // 2)]
            
            # Crossover
            if np.random.rand() < 0.7:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                # Mutate 15% of points
                mutate_indices = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
                for idx in mutate_indices:
                    child[idx] += np.random.normal(0, 0.1)
                # Clip and rescale
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
                if len(new_population) < population_size:
                    new_population.append(child)
                else:
                    break
        
        population = new_population
    
    # Final check of best solution
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
