import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):

        n_points = len(h)

        h = np.ones(n_points)
        mid = n_points // 2
        h[mid-2:mid+3] = 0.7  # peaked
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    
    # Ensure integral = 1.0 exactly
    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    max_iterations = 500  # Add safeguard
    
    # Define critical shift distances to test
    max_shift = n_points - 1
    shift_indices = np.arange(1, max_shift + 1)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Calculate current overlap maximum
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Linear programming approach:
        # Minimize C subject to constraints for each shift k
        # sum_i h_i * (1 - h_{i+k}) <= C * dx  for all valid k
        # sum(h) = n/2
        # 0 <= h_i <= 1 for all i
        
        # Sample more critical shifts
        critical_shifts = []
        for k in shift_indices:
            # For each valid shift k, compute correlation coefficient
            # We only care about those that are most constraining
            valid_indices = range(n_points - k)
            if len(valid_indices) > 0:
                correlation_k = np.sum(h[valid_indices] * (1 - h[valid_indices + k]))
                if correlation_k > 0.95 * best_c5 * dx or k == 1 or k == n_points//2:
                    critical_shifts.append(k)
        
        # Add a few more shifts around extremes
        critical_shifts.extend([1, n_points//2, n_points-1])
        critical_shifts = list(set(critical_shifts))  # unique shifts
        
        # Constraint matrix: for each shift, we create a constraint row
        n_constraints = len(critical_shifts) + 1  # plus normalization
        
        # Set up LP
        A_eq = np.zeros((1, n_points))
        A_eq[0,:] = 1.0
        b_eq = np.array([n_points/2.0])
        
        A_ub = np.zeros((n_constraints, n_points))
        b_ub = np.zeros(n_constraints)
        
        # Add shift constraints
        for i, k in enumerate(critical_shifts):
            # h_i * (1 - h_{i+k}) <= C * dx  =>  h_i - h_i * h_{i+k} <= C * dx
            # For LP, we reframe as: - h_i * h_{i+k} <= C * dx - h_i
            # But the product h_i * h_{i+k} makes this non-linear.
            # Work instead with the actual cross terms.
            rows = []
            cols = []
            values = []
            for j in range(n_points - k):
                rows.append(i)
                cols.append(j)
                values.append(1.0)
                rows.append(i)
                cols.append(j + k)
                values.append(-h[j])  # coefficient of h_{i+k}

            # For LINPROG, the constraint is A_ub * h <= b_ub
            # Each constraint:
            # sum_i a_i * h_i <= b
            # For this specific case:
            # sum_{i=0}^{n-k-1} h_i - h_i * h_{i+k} <= C * dx
            
            # Approach: use 0-1 upper bounds on h_i h_{i+k} to linearize
            # Since h_i, h_{i+k} in [0,1], their product is bounded by:
            # h_i * h_{i+k} <= min(h_i, h_{i+k})
            
            # Alternative approach (not using complements):
            # Let's just compute once the actual correlations
            A_ub[i, :] = h[:]
            A_ub[i, :n_points-k] -= h[:n_points-k] * h[k:]
            b_ub[i] = 0  # update later
            # b_ub[i] = C * dx with C = best_c5 + epsilon (guess)
        
        # Replace the linearization with numerical approach 
        # Try updating h using coordinate ascent or direct method
        # Conservative hill-climbing
        new_h = h.copy()
        step_size = 0.03
        if np.random.rand() < 0.6:  # Small update
            perturbation = np.random.normal(0, step_size, n_points)
        else:  # Larger update
            perturbation = np.random.choice([-step_size, step_size], n_points)
        
        new_h += perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate the new solution
        corr_new = np.correlate(new_h, 1 - new_h, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < current_c5:
            h = new_h
        else:
            # Occasionally accept worse moves to escape local optima
            if np.random.rand() < 0.05:
                h = new_h
    
    # Final adjustment and validate
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    correlation = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
