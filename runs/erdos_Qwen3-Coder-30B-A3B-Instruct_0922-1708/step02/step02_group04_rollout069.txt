import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Segment-based optimization approach
    # Represent h as piecewise constant with segments
    def segments_to_h(segments, n_points, dx):
        """Convert segment representation to h array"""
        h = np.zeros(n_points)
        for i, (a, b, v) in enumerate(segments):
            # Map continuous interval [a,b] to discrete indices
            start_idx = int(a / dx)
            end_idx = int(b / dx)
            # Ensure valid indexing
            start_idx = max(0, min(start_idx, n_points-1))
            end_idx = max(0, min(end_idx, n_points))
            if start_idx < end_idx:
                h[start_idx:end_idx] = v
        return h
    
    def h_to_segments(h, n_points, dx):
        """Convert h array to piecewise segments"""
        # Look for groups of similar values and merge them
        segments = []
        start_idx = 0
        prev_val = h[0]
        
        for i in range(1, n_points):
            if abs(h[i] - prev_val) > 1e-6:
                # End previous segment
                if start_idx < i:
                    segments.append((start_idx * dx, i * dx, prev_val))
                prev_val = h[i]
                start_idx = i
        
        # Add final segment
        if start_idx < n_points:
            segments.append((start_idx * dx, n_points * dx, prev_val))
        return segments
    
    def get_c5(segments, n_points, dx):
        """Compute maximum overlap for given segments"""
        h = segments_to_h(segments, n_points, dx)
        correlation = np.correlate(h, 1 - h, mode='full')
        # The full correlation array is length 2*n_points - 1
        # The corresponding shift values are from -(n_points-1) to (n_points-1)
        # Maximum overlap corresponds to shift = -1 to 1 (center part)
        # But we need to extract all relevant overlap values
        # Method: integrate h(x)(1-h(x+k)) for all k
        # This matches maximum value in correlation
    
        c5 = np.max(correlation) * dx
        return c5
    
    def adjust_segments(segments, n_points, dx, target_k):
        """Adjust segments to lower overlap at target_k"""
        # Calculate overlap contributions at given shift
        new_segments = []
        total_integral = 0.0
        
        # For simplicity, we will do a greedy adjustment
        # This is rough implementation, more detailed analysis would compute
        # mask for specific overlap contribution at k
        # But for optimization we'll sample some key intervals
        
        for i, (a, b, v) in enumerate(segments):
            new_segments.append((a, b, v))
            total_integral += v * (b - a)
        
        # Simple approach: find where contribution to overlap occurs at k
        # Downsample to prevent drift
        step = max(1, len(segments) // 4)
        for i in range(0, len(segments), step):
            if segments[i][2] > 0.01:  # Only adjust if not nearly zero
                a, b, v = segments[i]
                # Reduce value slightly
                new_v = max(0.0, v - 0.02)
                new_segments[i] = (a, b, new_v)
        
        return new_segments
    
    def optimize_segments(segments, n_points, dx, max_iter=50):
        """Optimize segments via local search"""
        best_segments = [s[:] for s in segments]
        best_c5 = float('inf')
        
        for iter_idx in range(max_iter):
            # Compute overlap
            c5 = get_c5(best_segments, n_points, dx)
            if c5 < best_c5:
                best_c5 = c5
                # Make shallow copy
                best_segments = [s[:] for s in best_segments]
            
            # Make small random adjustments
            new_segments = [s[:] for s in best_segments]
            i = np.random.randint(len(new_segments))
            a, b, v = new_segments[i]
            # Perturb value and edges
            noise = np.random.randn() * 0.02
            v = max(0.0, min(1.0, v + noise))
            # Small perturbation to edges to maintain spacing
            edge_noise = np.random.randn() * 0.01
            a = max(0.0, min(a + edge_noise, b - 0.01))
            b = max(a + 0.01, min(b + edge_noise, 2.0))
            new_segments[i] = (a, b, v)
            
            # Ensure integral constraint
            h = segments_to_h(new_segments, n_points, dx)
            integral = np.sum(h) * dx
            if abs(integral - 1.0) > 1e-6:
                h = h * (1.0 / integral)
                new_segments = h_to_segments(h, n_points, dx)
            
            # Replace with modified segments
            best_segments = new_segments
        
        return best_segments, best_c5
    
    # Start with segments derived from h
    segments = h_to_segments(h, n_points, dx)
    iteration = 0
    
    # Initialize with simple segments if no parent
    if not ('initial_h_values' in globals()):
        # Manually create good starting structure
        segments = [
            (0.0, 0.8, 0.7),
            (0.8, 1.2, 0.7),
            (1.2, 2.0, 0.0)
        ]
        # Make sure integral is 1
        integral = sum(v * (b - a) for a, b, v in segments)
        scaled = [ (a, b, v * (1.0/integral)) for a, b, v in segments]
        segments = scaled
    
    # Main loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Optimize segments
        segments, c5 = optimize_segments(segments, n_points, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            # Convert to full h representation
            h = segments_to_h(segments, n_points, dx)
            best_h = h.copy()
        
        # Occasionally reinitialize a new set of segments
        if iteration % 20 == 0:
            # Do an exploratory re-clustering
            h = segments_to_h(segments, n_points, dx)
            h = np.clip(h, 0, 1)
            segment_list = h_to_segments(h, n_points, dx)
            segments = [s[:] for s in segment_list]
    
    # Final evaluation
    h_final = segments_to_h(segments, n_points, dx)
    h_final = np.clip(h_final, 0, 1)
    integral_check = np.sum(h_final) * dx
    if abs(integral_check - 1.0) > 1e-6:
        h_final = h_final * (1.0 / integral_check)
    
    correlation = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(correlation) * (2.0 / n_points)
    
    return (h_final.tolist(), final_c5, n_points)
