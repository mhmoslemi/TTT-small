import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation approach
    # Construct a convex relaxation by linearizing constraints
    # This is a simplified version - in practice, you'd solve a proper SDP
    # For now, we'll use the initial solution and refine with gradient descent
    
    # First, let's optimize with a more robust approach
    def compute_c5(h_vals):
        # Compute correlation
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def project_h(h_vals):
        # Ensure h is in [0,1] and sum to n_points/2
        h_new = np.clip(h_vals, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Optimizer with adaptive learning rate
    def optimize_step(h_vals, iterations=50):
        h_current = h_vals.copy()
        current_c5 = compute_c5(h_current)
        
        for i in range(iterations):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros_like(h_current)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = project_h(h_plus)
                
                c5_plus = compute_c5(h_plus)
                grad[j] = (c5_plus - current_c5) / eps
            
            # Compute adaptive learning rate
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-10:
                lr = min(0.05, 1.0 / (grad_norm + 1e-8))
            else:
                lr = 0.01
                
            # Update
            h_new = h_current - lr * grad
            h_new = project_h(h_new)
            
            new_c5 = compute_c5(h_new)
            if new_c5 < current_c5:
                h_current = h_new
                current_c5 = new_c5
            else:
                # Try smaller step if no improvement
                lr *= 0.5
                h_new = h_current - lr * grad
                h_new = project_h(h_new)
                new_c5 = compute_c5(h_new)
                if new_c5 < current_c5:
                    h_current = h_new
                    current_c5 = new_c5
        
        return h_current, current_c5
    
    # Try different initialization strategies
    h_best = h.copy()
    c5_best = compute_c5(h)
    max_iterations = 500
    
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Random perturbation
        h_pert = h_best + np.random.normal(0, 0.01, n_points)
        h_pert = project_h(h_pert)
        
        # Optimize from this point
        h_opt, c5_opt = optimize_step(h_pert, 100)
        
        # Also run from original starting point
        h_opt2, c5_opt2 = optimize_step(h_best, 50)
        
        # Update best if needed
        if c5_opt < c5_best:
            h_best = h_opt
            c5_best = c5_opt
        if c5_opt2 < c5_best:
            h_best = h_opt2
            c5_best = c5_opt2
    
    # Final verification
    final_h = project_h(h_best)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
