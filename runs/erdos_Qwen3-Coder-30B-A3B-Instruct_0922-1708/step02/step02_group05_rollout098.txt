import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Initialize PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia_weight = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles and velocities
    particles = np.random.uniform(0, 1, (swarm_size, n_points))
    velocities = np.random.uniform(-0.5, 0.5, (swarm_size, n_points))
    
    # Normalize particles to meet constraint
    for i in range(swarm_size):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Initialize personal best positions and values
    pbest_positions = particles.copy()
    pbest_values = np.full(swarm_size, float('inf'))
    
    # Initialize global best
    gbest_position = None
    gbest_value = float('inf')
    
    start_time = time.time()
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(swarm_size):
            particle = particles[i]
            
            # Compute overlap integral
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_values[i]:
                pbest_values[i] = c5
                pbest_positions[i] = particle.copy()
                
                # Update global best
                if c5 < gbest_value:
                    gbest_value = c5
                    gbest_position = particle.copy()
        
        # Update velocities and positions
        for i in range(swarm_size):
            # Update inertia weight (linearly decreasing)
            new_inertia = inertia_weight - (inertia_weight - 0.4) * iteration / max_iter
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (new_inertia * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Final evaluation of best solution
    if gbest_position is not None:
        correlation = np.correlate(gbest_position, 1 - gbest_position, mode='full')
        final_c5 = np.max(correlation) * dx
        best_h = gbest_position
    else:
        # Fallback to original h if no improvement
        correlation = np.correlate(h, 1 - h, mode='full')
        final_c5 = np.max(correlation) * dx
        best_h = h
    
    return (best_h.tolist(), final_c5, n_points)
