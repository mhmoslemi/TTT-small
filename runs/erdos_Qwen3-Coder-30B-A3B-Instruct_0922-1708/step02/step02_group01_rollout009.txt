import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Multi-start approach with 10 particles
    particles = []
    for i in range(10):
        if i == 0 and 'initial_h_values' in globals():
            # Use initial construction
            h_start = h.copy()
        else:
            # Random construction
            h_start = np.random.rand(n_points)
            h_start = h_start / np.sum(h_start) * (n_points / 2)
        particles.append(h_start)
    
    # Optimization parameters
    lr_schedule = [0.01, 0.005, 0.002, 0.001]
    lr_idx = 0
    momentum = 0.9
    velocity_buffer = [np.zeros(n_points) for _ in range(10)]
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for p_idx, h in enumerate(particles):
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                lr_idx = min(lr_idx + 1, len(lr_schedule) - 1)
            
            # Gradient estimation using finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Adaptive learning rate and momentum
            lr = lr_schedule[lr_idx]
            velocity = momentum * velocity_buffer[p_idx] - lr * grad
            velocity_buffer[p_idx] = velocity
            
            h_new = h + velocity
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Occasionally perturb
            if np.random.random() < 0.05:
                noise = np.random.normal(0, 0.02, n_points)
                h_new = h_new + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
            particles[p_idx] = h_new
            
        # Occasionally restart with best points
        if iteration % 100 == 0 and best_c5 < 0.381:
            for i in range(10):
                if i == 0 and 'initial_h_values' in globals():
                    particles[i] = h.copy()
                else:
                    particles[i] = np.random.rand(n_points)
                    particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
            lr_idx = 0
            
    return (best_h.tolist(), best_c5, n_points)
