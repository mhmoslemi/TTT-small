import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use scipy.optimize for better convergence properties
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run optimization for most of the time budget
    remaining_time = budget_s - 1
    
    # Try multiple random restarts to find better local minima
    for restart in range(5):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Initialize with random perturbation for every restart
        h = np.random.uniform(0.3, 0.7, n_points)
        h = h * (n_points / 2) / np.sum(h)
        
        # Optimize with different methods
        try:
            res = minimize(objective, h, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 500})
            h_opt = res.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            # Evaluate result
            correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                
        except:
            pass
            
        # Occasionally check if we've run out of time
        if time.time() - start_time > remaining_time - 20:
            break
    
    # Final refinement with gradient-based method
    try:
        # Use a less aggressive optimization for final tuning
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Gradient-free method to fine-tune further
        for i in range(1000):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Small random perturbations
            perturb = np.random.normal(0, 0.005, n_points)
            h_trial = h_refined + perturb
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            correlation = np.correlate(h_trial, 1 - h_trial, mode='full')
            c5_trial = np.max(correlation) * dx
            
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h_refined = h_trial.copy()
                
    except:
        pass
    
    # Return the best result found
    return (best_h.tolist(), best_c5, n_points)
