        h = np.ones(n_points) * 0.5

    best_c5 = float('inf')

        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Perturb in a structured way
        # Try a few different perturbation strategies
        for _ in range(2):
            # Random perturbation with adaptive step size
            perturbation = np.random.normal(0, 0.01, n_points)
            h_new = h + perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)

            c5_new = np.max(correlation_new) * dx
            
            if c5_new < c5:
                h = h_new.copy()
                
        # Try to refine using a quadratic approximation
        if iteration % 10 == 0:
            # Use a simple gradient estimate with finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)

                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            
            # Gradient descent step
            lr = 0.01
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)

    # Make sure it meets constraints
    if abs(np.sum(best_h) * dx - 1.0) > 1e-10:
        # Rescale to meet constraint
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    return (best_h.tolist(), final_c5, n_points)
