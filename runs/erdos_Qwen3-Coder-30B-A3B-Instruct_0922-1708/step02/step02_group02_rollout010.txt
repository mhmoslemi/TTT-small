import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        # Create variants with small noise
        for _ in range(population_size):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Initialize with random distributions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)

    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    no_improvement_count = 0
    mutation_rate = 0.2
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Early stopping if no improvement in several generations
        if generation > 1 and fitness[0] >= best_c5:
            no_improvement_count += 1
            if no_improvement_count > 100:
                mutation_rate = 0.3
        else:
            no_improvement_count = 0
        
        # Selection: keep top 20%
        keep_count = max(1, population_size // 5)
        selected = population[:keep_count]
        
        # Replacement: create new population
        new_population = selected.copy()
        
        # Crossover: combine pairs
        while len(new_population) < population_size:
            parent1 = random.choice(selected)
            parent2 = random.choice(selected)
            # Single point crossover (or uniform)
            crossover_point = random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            # Maintain constraint
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        
        # Mutation
        for i in range(len(new_population)):
            if random.random() < mutation_rate:
                child = new_population[i].copy()
                # Mutate 20% of elements
                mutate_indices = random.sample(range(n_points), max(1, int(0.2 * n_points)))
                for idx in mutate_indices:
                    child[idx] += np.random.normal(0, 0.02)
                # Clip and rescale
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population[i] = child
        
        population = new_population
        
    # Final computation of best solution
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
    else:
        # Fallback
        best_h = population[0] if population else np.ones(n_points) * 0.5
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
