import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a binary comb pattern
        h = np.zeros(n_points)
        # Create 5 pulses with specific spacing and width
        pulse_width = 0.2
        pulse_spacing = 0.25
        pulse_positions = [0.1, 0.45, 0.8, 1.15, 1.5]
        for i in range(5):
            start = int(max(0, pulse_positions[i] / 2 * n_points))
            end = int(min(n_points, (pulse_positions[i] + pulse_width) / 2 * n_points))
            if start < end:
                h[start:end] = 1.0
        
        # Ensure sum is correct
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    max_iter = 5000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Gradient approximation using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update with gradient descent - adaptive learning rate
        lr = 0.01 / (1 + iteration / 100.0)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
