import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # PSO parameters
    swarm_size = 50
    w_start = 0.7
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_iter = 500
    stagnation_threshold = 100
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            pos = np.array(initial_h_values)
            pos = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(pos)), pos)
            pos = np.clip(pos, 0, 1)
            pos = pos * (n_points / 2) / np.sum(pos)
        else:
            pos = np.random.uniform(0, 1, n_points)
            pos = pos * (n_points / 2) / np.sum(pos)
        
        particles.append(pos.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        pbest_positions.append(pos.copy())
        pbest_values.append(float('inf'))
    
    gbest_position = h.copy()
    gbest_value = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update inertia weight linearly
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Compute current position's objective value
            pos = particles[i]
            correlation = np.correlate(pos, 1 - pos, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_values[i]:
                pbest_values[i] = c5
                pbest_positions[i] = pos.copy()
            
            # Update global best if needed
            if c5 < gbest_value:
                gbest_value = c5
                gbest_position = pos.copy()
                last_improvement = iteration
        
        # Check for stagnation
        if iteration - last_improvement > stagnation_threshold:
            # Perturb random particles
            for _ in range(swarm_size // 10):
                idx = np.random.randint(swarm_size)
                particles[idx] = np.random.uniform(0, 1, n_points)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.uniform(-0.5, 0.5, n_points)
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.random(2)
            vel = w * velocities[i] \
                  + c1 * r1 * (pbest_positions[i] - particles[i]) \
                  + c2 * r2 * (gbest_position - particles[i])
            
            velocities[i] = vel
            new_pos = particles[i] + vel
            
            # Apply constraints
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * (n_points / 2) / np.sum(new_pos)
            
            particles[i] = new_pos
    
    # Final evaluation of the best solution
    correlation = np.correlate(gbest_position, 1 - gbest_position, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (gbest_position.tolist(), final_c5, n_points)
