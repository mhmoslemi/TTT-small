import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        for i in range(population_size):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for i in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        if no_improvement_count >= max_no_improvement:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
        
        # Sort population
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 20% (elitism)
        elite_size = population_size // 5
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = np.random.choice(elite_size)
            parent2_idx = np.random.choice(elite_size)
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < 0.5:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = (parent1 + parent2) / 2.0
            
            # Mutation
            mutate_mask = np.random.rand(n_points) < 0.2
            if np.any(mutate_mask):
                child[mutate_mask] += np.random.normal(0, 0.02, np.sum(mutate_mask))
                child = np.clip(child, 0, 1)
            
            # Rescale to meet constraint
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:population_size]
        no_improvement_count += 1
    
    # Final evaluation
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        return (best_h.tolist(), final_c5, n_points)
    else:
        # If no improvement yet, return the best from initial population
        h = population[0]
        corr = correlate(h, 1 - h, mode='full')
        final_c5 = np.max(corr) * dx
        return (h.tolist(), final_c5, n_points)
