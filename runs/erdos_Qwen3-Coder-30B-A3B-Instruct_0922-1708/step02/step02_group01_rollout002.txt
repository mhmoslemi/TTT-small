import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    n_particles = 50
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            x_orig = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, n_points)
            h0 = np.interp(x_new, x_orig, h0)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Initialize particle positions and velocities
    positions = np.random.rand(n_particles, n_points) * 0.5
    positions[0] = h0
    velocities = np.random.rand(n_particles, n_points) * 0.1 - 0.05
    
    # Projection function to ensure constraints
    def project(h):
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        return h
    
    # Compute C5 for a given h
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize personal and global bests
    pbest_positions = positions.copy()
    pbest_values = np.array([compute_c5(p) for p in positions])
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    start_time = time.time()
    stagnation_count = 0
    max_stagnation = 100
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        stagnation_count += 1
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - positions[i]) + 
                            c2 * r2 * (gbest_position - positions[i]))
            
            # Update position
            positions[i] += velocities[i]
            
            # Project to feasible region
            positions[i] = project(positions[i])
            
            # Evaluate
            current_c5 = compute_c5(positions[i])
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest_positions[i] = positions[i].copy()
                
                # Update global best
                if current_c5 < gbest_value:
                    gbest_value = current_c5
                    gbest_position = positions[i].copy()
                    stagnation_count = 0
        
        # Restart velocities if stagnation
        if stagnation_count > max_stagnation:
            velocities = np.random.rand(n_particles, n_points) * 0.1 - 0.05
            stagnation_count = 0
            
        # Early termination
        if gbest_value < 0.3807:
            break
    
    return (gbest_position.tolist(), gbest_value, n_points)
