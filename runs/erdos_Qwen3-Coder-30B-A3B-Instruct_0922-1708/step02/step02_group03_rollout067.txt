import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Initialize swarm parameters
    n_particles = 50
    w = 0.7
    c1 = c2 = 0.9
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    gbest = None
    gbest_value = float('inf')
    
    # Initialize particles
    for i in range(n_particles):
        particle = h.copy()
        # Add small random noise
        particle += np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_values.append(float('inf'))
    
    # Helper function to compute C5
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # PSO main loop
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update each particle
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            
            # Compute current fitness
            current_c5 = compute_c5(particle)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest[i] = particle.copy()
            
            # Update global best
            if current_c5 < gbest_value:
                gbest_value = current_c5
                gbest = particle.copy()
                
            # Update velocity and position
            r1, r2 = np.random.random(2)
            velocity = (w * velocity + 
                       c1 * r1 * (pbest[i] - particle) + 
                       c2 * r2 * (gbest - particle))
            
            particle = particle + velocity
            
            # Apply constraints
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            particles[i] = particle
            velocities[i] = velocity
        
        # Occasionally refine the best solution using gradient-based method
        if iteration % 50 == 0 and gbest is not None:
            # Do local refinement of gbest using projected gradient descent
            refine_h = gbest.copy()
            
            # Gradient approximation
            eps = 1e-6
            grad = np.zeros(n_points)
            
            for j in range(n_points):
                h_plus = refine_h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr_ref = np.correlate(refine_h, 1 - refine_h, mode='full')
                c5_ref = np.max(corr_ref) * dx
                
                grad[j] = (c5_plus - c5_ref) / eps
                
            # Gradient descent step
            learning_rate = 0.05
            refine_h_new = refine_h - learning_rate * grad
            
            # Apply constraints
            refine_h_new = np.clip(refine_h_new, 0, 1)
            refine_h_new = refine_h_new * (n_points / 2) / np.sum(refine_h_new)
            
            # Update gbest
            new_c5 = compute_c5(refine_h_new)
            if new_c5 < gbest_value:
                gbest_value = new_c5
                gbest = refine_h_new.copy()
                
            # Update particles to be closer to gbest
            for i in range(n_particles):
                particles[i] = 0.9 * particles[i] + 0.1 * gbest
                
        # Early termination condition
        if iteration % 1000 == 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (gbest.tolist(), gbest_value, n_points)
