import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a blocky initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    # Rescale to meet constraint
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Function to compute overlap integral
    def compute_c5(h_vals):
        h = np.array(h_vals)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Gradient computation using finite differences
    def compute_gradient(h_vals, eps=1e-5):
        grad = np.zeros_like(h_vals)
        c5_base = compute_c5(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = compute_gradient(h)
        
        # Update using gradient descent with adaptive learning rate
        lr = 0.01 / (1.0 + iteration * 0.001)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Local search around best solution
        if np.random.random() < 0.1:
            h_try = best_h.copy() + np.random.normal(0, 0.01, n_points)
            h_try = np.clip(h_try, 0, 1)
            h_try = h_try * (n_points / 2) / np.sum(h_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                
        h = h_new
        
    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
