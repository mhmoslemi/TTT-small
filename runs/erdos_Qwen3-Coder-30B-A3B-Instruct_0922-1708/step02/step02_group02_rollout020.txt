import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_lags = n_points // 2  # Consider only first half of lags for efficiency
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        
    h = h_init / np.sum(h_init) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Compute max overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Set up QP
        h_vars = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            sum(h_vars) == n_points / 2,
            t >= 0
        ]
        
        # Add lag constraints
        # Create matrix for efficient computation
        if len(correlation) > 0:
            lag_indices = np.arange(len(correlation)) - (len(correlation) - 1) // 2
            # Only use relevant lags (not too long)
            relevant_lags = lag_indices[np.abs(lag_indices) <= max_lags]
            
            for k in relevant_lags:
                if k >= 0:
                    # h[i] * (1 - h[i+k]) for i = 0..n-k-1
                    # This translates to a sum of products
                    # Use numpy for efficient computation
                    if k == 0:
                        # special case: h[i] * (1 - h[i]) = h[i] - h[i]^2
                        slack = 0
                        for i in range(n_points - k):
                            slack += (h_vars[i] - h_vars[i]**2)
                    else:
                        # sum of h[i]*(1 - h[i+k]) = sum(h[i]) - sum(h[i]*h[i+k])
                        slack = 0
                        for i in range(n_points - k):
                            slack += h_vars[i] * (1 - h_vars[i + k])
                    constraints.append(slack <= t * dx)
                else:
                    # Use k as negative indices
                    k_abs = -k
                    # This means h[i+k] for i = k..n-1, but need h[i] and h[i+k]
                    # i+k >= 0 means i >= -k and i < n
                    # So i ranges from max(0,-k) to n-1
                    if k_abs > n_points:
                        continue
                    slack = 0
                    for i in range(max(0, k_abs), n_points):
                        slack += h_vars[i] * (1 - h_vars[i - k_abs])
                    constraints.append(slack <= t * dx)
        
        # Objective
        objective = Minimize(t)
        
        # Solve QP
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                new_h = np.array(h_vars.value).flatten()
                h = np.clip(new_h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # If QP fails, fallback to gradient-based approach
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                corr = np.correlate(h, 1 - h, mode='full')
                c5_cur = np.max(corr) * dx
                grad[i] = (c5_plus - c5_cur) / eps
            lr = 0.001
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
