import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    Δ = 0.01
    no_improvement_count = 0
    
    # Heuristic: use different learning rates and search strategies
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
            Δ = max(1e-4, 0.01 * c5)
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                Δ = max(1e-4, Δ * 0.5)
                
        if best_c5 < 0.3807:
            break
            
        # Find the index of maximum correlation
        max_idx = np.argmax(correlation)
        k_max = max_idx - (len(h) - 1)
        
        # Create a modified h that penalizes the worst overlap
        h_new = h.copy()
        if k_max >= 0:
            # Offset automatically wraps
            for i in range(len(h)):
                h_new[i] -= 0.01 * h[i] * (1 - h[(i + k_max) % len(h)])
        else:
            # k_max < 0, so h[i+k_max] wraps from the end
            for i in range(len(h)):
                h_new[i] -= 0.01 * h[i] * (1 - h[(i + k_max) % len(h)])
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
