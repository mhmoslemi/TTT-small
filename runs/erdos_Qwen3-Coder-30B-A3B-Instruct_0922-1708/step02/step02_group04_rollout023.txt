import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 60
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters
    population_size = min(50, max(20, n_points // 3))
    generations = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals() and np.random.rand() < 0.3:
            individual = h.copy()
        else:
            individual = np.random.rand(n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
            individual = np.clip(individual, 0, 1)
        population.append(individual)
    
    # Function to evaluate fitness (C5)
    def evaluate_c5(h):
        correlation = correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Main evolutionary loop
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        generations += 1
        
        # Evaluate population
        fitness_scores = []
        for individual in population:
            c5 = evaluate_c5(individual)
            fitness_scores.append(c5)
        
        # Track best in population
        current_best_idx = np.argmin(fitness_scores)
        current_best_c5 = fitness_scores[current_best_idx]
        current_best_h = population[current_best_idx].copy()
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Selection (roulette wheel)
        fitness_scores = np.array(fitness_scores)
        # Avoid division by zero and negative fitness
        fitness_scores = np.maximum(1e-10, 1.0 / (fitness_scores + 1e-10))
        total_fitness = np.sum(fitness_scores)
        
        if total_fitness == 0:
            # fallback to uniform selection
            probabilities = np.ones(population_size) / population_size
        else:
            probabilities = fitness_scores / total_fitness
        
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(current_best_h.copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Select two parents
            parent1_idx = np.random.choice(population_size, p=probabilities)
            parent2_idx = np.random.choice(population_size, p=probabilities)
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover (uniform crossover)
            if np.random.rand() < 0.7:  # 70% chance of crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            mutation_rate = 0.1
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                    child2[i] += np.random.normal(0, 0.05)
            
            # Clamping and re-normalization
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population[:population_size]
    
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
