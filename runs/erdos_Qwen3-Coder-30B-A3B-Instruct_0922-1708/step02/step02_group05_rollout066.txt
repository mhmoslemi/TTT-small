import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Setup for genetic algorithm
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    tournament_size = 3
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h_test = np.random.rand(n_points)
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        population.append(h_test)
    
    # Add the best known solution to population
    if 'initial_h_values' in globals():
        population[0] = h.copy()
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate fitness for all individuals
        fitness = []
        for i in range(population_size):
            corr = np.correlate(population[i], 1 - population[i], mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = population[i].copy()
        
        # Selection using tournament
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and Mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % population_size]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.1)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.1)
            
            # Repair feasibility
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        # Keep only the population_size individuals
        population = new_population[:population_size]
        
        # Occasionally add new random individuals to maintain diversity
        if iteration % 10 == 0:
            for i in range(0, min(5, population_size//10)):
                h_rand = np.random.rand(n_points)
                h_rand = np.clip(h_rand, 0, 1)
                h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
                idx = np.random.randint(0, population_size)
                population[idx] = h_rand

    return (best_h.tolist(), best_c5, n_points)
