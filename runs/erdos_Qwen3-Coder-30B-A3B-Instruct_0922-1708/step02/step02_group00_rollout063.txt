import numpy as np
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated annealing parameters
    T0 = 0.1
    tau = 1000.0
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Current temperature
        T = T0 / (1.0 + iteration / tau)
        
        # Generate candidate solution
        h_new = h + np.random.normal(0, T, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate candidates
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept with probability
        delta = c5_new - c5
        if delta < 0 or np.random.random() < math.exp(-delta / T):
            h = h_new
            c5 = c5_new
            
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
