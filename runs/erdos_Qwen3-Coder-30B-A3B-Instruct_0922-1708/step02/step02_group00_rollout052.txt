import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent solution or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Multi-start with variations
    starts = []
    if 'initial_h_values' in globals():
        starts.append(h)
        # Add variations
        for _ in range(4):
            noise = np.random.normal(0, 0.05, n_points)
            h_var = h + noise
            h_var = clip_and_scale(h_var, n_points)
            starts.append(h_var)
    else:
        # Random sparse solutions
        for _ in range(5):
            h_rand = np.random.rand(n_points)
            h_rand = clip_and_scale(h_rand, n_points)
            starts.append(h_rand)
    
    for i, h_start in enumerate(starts):
        h = h_start.copy()
        lr = 0.001
        beta1, beta2 = 0.9, 0.999
        m, v = np.zeros_like(h), np.zeros_like(h)
        epsilon = 1e-8
        
        for iteration in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute objective
            c5 = compute_c5(h, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Estimate gradient using finite differences
            eps = 1e-5
            grad = np.zeros_like(h)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = compute_c5(h_plus, dx)
                grad[j] = (c5_plus - c5) / eps
                
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**(iteration + 1))
            v_hat = v / (1 - beta2**(iteration + 1))
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
            h = clip_and_scale(h_new, n_points)
            
    return (best_h.tolist(), best_c5, n_points)
