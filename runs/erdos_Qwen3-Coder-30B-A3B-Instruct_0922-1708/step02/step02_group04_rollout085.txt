    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use a more robust optimization approach with adaptive steps and better convergence
    # and attempt to move away from local optima periodically
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # More structured optimization approach using gradient approximation
        # Compute gradient approximation (simplified)
        gradient = np.zeros_like(h)
        for i in range(n_points):
            # Forward difference approximation
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + 1e-4)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            gradient[i] = (c5_plus - c5) / 1e-4
        
        # Take gradient step
        step_size = 0.01
        h_new = h - step_size * gradient
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)

        # Occasionally escape local minima with random perturbation

    best_h = np.clip(best_h, 0, 1)

    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
