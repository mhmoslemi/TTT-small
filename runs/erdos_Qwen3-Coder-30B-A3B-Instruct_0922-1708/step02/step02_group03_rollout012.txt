from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use a more sophisticated optimization approach
    # Try multiple restarts with different initializations
    restarts = 5
    restart_interval = budget_s // (restarts + 1)

        if time.time() - start_time > budget_s - 10:
            break
            
        # Random perturbation of the current best
        if restart == 0:
            current_h = h.copy()
        else:
            # Initialize with small random noise
            current_h = best_h + np.random.normal(0, 0.05, n_points)
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            
        # Optimization loop for this restart
        for iter in range(500):  # limit iterations per restart
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute the overlap integral
            correlation = np.correlate(current_h, 1 - current_h, mode='full')

                best_c5 = c5
                best_h = current_h.copy()
                
            # More sophisticated gradient estimation using finite differences
            eps = 1e-4
            grad = np.zeros_like(current_h)

            for i in range(n_points):

                grad[i] = (c5_plus - c5) / eps

            learning_rate = max(0.001, 0.01 * np.exp(-iter/100))

            current_h_new = current_h - learning_rate * grad

            # Occasionally check early termination during restart
            if iter % 100 == 0 and iter > 0:
                if time.time() - start_time > budget_s - 10:
                    break
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
