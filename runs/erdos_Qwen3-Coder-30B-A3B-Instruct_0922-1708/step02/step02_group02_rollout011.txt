import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population with variants of initial_h_values or random
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        h_base = h_base / np.sum(h_base) * (n_points / 2)
        
        # Create variants
        for _ in range(population_size):
            h = h_base + np.random.uniform(-0.05, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Initialize random population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            correlation = correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if generation > 1 and fitness[0] >= fitness[1]:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        # Adjust strategy if no improvement
        if no_improvement_count > max_no_improvement:
            no_improvement_count = 0
            # Increase mutation rate
            mutation_rate = 0.3
        else:
            mutation_rate = 0.2
            
        # Create new population
        new_population = []
        
        # Keep top 20% (elitism)
        elite_size = population_size // 5
        new_population.extend(population[:elite_size])
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = population[random.randint(0, population_size // 2)]
            parent2 = population[random.randint(0, population_size // 2)]
            
            # Crossover
            if random.random() < 0.5:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Arithmetic crossover
                alpha = random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if random.random() < mutation_rate:
                num_mutations = max(1, int(0.2 * n_points))
                mutation_indices = random.sample(range(n_points), num_mutations)
                for i in mutation_indices:
                    child[i] += random.gauss(0, 0.02)
            
            # Clip and rescale
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population[:population_size]
    
    # Final check
    if best_h is not None:
        correlation = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(correlation) * dx
    else:
        # If nothing was found, use the best in final population
        h = population[0]
        correlation = correlate(h, 1 - h, mode='full')
        final_c5 = np.max(correlation) * dx
        best_h = h
    
    return (best_h.tolist(), final_c5, n_points)
