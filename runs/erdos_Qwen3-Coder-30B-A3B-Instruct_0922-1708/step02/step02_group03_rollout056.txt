import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute exact gradient
        max_k = np.argmax(correlation)
        # Relative shift for the maximum value
        if max_k >= n_points - 1:
            shift = max_k - (n_points - 1)
        else:
            shift = max_k
        
        grad = np.zeros_like(h)
        # Compute gradient component by component
        for i in range(n_points):
            # At index i, the contribution to the kernel is h[i]*(1-h[(i+shift)%n_points])
            # So d/dh[i] = (1-h[(i+shift)%n_points]) - h[(i+shift)%n_points] = 1-2*h[(i+shift)%n_points]
            j = (i + shift) % n_points
            grad[i] = 1 - 2 * h[j]
        
        # Update with Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            learning_rate = max(0.0001, learning_rate * 0.99)
            
        h_new = h - learning_rate * m_hat / (np.sqrt(v_hat) + eps_adam)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        if iteration % 100 == 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
