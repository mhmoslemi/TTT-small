import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use scipy's minimize with L-BFGS-B for better convergence
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Define the constraint for sum to be exactly n_points/2
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    iteration = 0
    
    # Initial value for optimization
    h_init = h.copy()
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Optimize with L-BFGS-B
        res = minimize(objective, h_init, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 10})
        
        if res.success:
            h_opt = res.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            # Check for improvement
            correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                
            h_init = h_opt
        else:
            # Fallback: gradient descent with momentum
            h = h_init.copy()
            learning_rate = 0.01
            momentum = 0.9
            v = np.zeros_like(h)
            
            # Compute gradient numerically
            eps = 1e-6
            grad = np.zeros_like(h)
            
            # Create temporary vectors for gradient computation
            h_plus = h.copy()
            h_plus[0] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Compute gradient using finite difference approximation
            corr = np.correlate(h, 1 - h, mode='full')
            c5_current = np.max(corr) * dx
            
            grad[0] = (c5_plus - c5_current) / eps
            
            # Apply momentum update
            v = momentum * v - learning_rate * grad
            h_new = h + v
            
            # Ensure feasible solution
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_init = h_new
            
            # Check for improvement
            correlation = np.correlate(h_new, 1 - h_new, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        
        if iteration % 50 == 0 and time.time() - start_time > budget_s - 5:
            break
            
    return (best_h.tolist(), best_c5, n_points)
