import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():

        n_points = len(h)

    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Coarse to fine optimization
    grid_sizes = [20, 40, 80]
    
    for grid_size in grid_sizes:
        if grid_size > n_points:
            # Interpolate h to this grid size
            h_new = np.interp(np.linspace(0, 2, grid_size), np.linspace(0, 2, n_points), h)
            h = h_new
            n_points = grid_size
        else:
            # Reduce resolution by averaging pairs of points
            if grid_size < n_points:
                n = grid_size
                h_new = np.zeros(n)
                for i in range(n):
                    start_idx = i * (n_points // n)
                    end_idx = (i + 1) * (n_points // n)
                    h_new[i] = np.mean(h[start_idx:end_idx])
                h = h_new
                n_points = grid_size
        
        dx = 2.0 / n_points
        
        # Gradient-like descent on this grid
        for iteration in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Compute gradient approximation
            grad = np.zeros(n_points)
            for i in range(n_points):
                # Approximate gradient with finite difference
                h_test = h.copy()
                h_test[i] += 1e-4
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                
                corr_test = np.correlate(h_test, 1 - h_test, mode='full')
                c5_test = np.max(corr_test) * dx
                
                grad[i] = (c5_test - c5) / 1e-4
                
            # Line search with step size
            alpha = 0.01
            improved = False
            
            for _ in range(10):
                h_new = h - alpha * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr_new) * dx
                
                if c5_new < c5:
                    h = h_new
                    improved = True
                    break
                alpha *= 0.5
                
            if not improved:
                # Random perturbations for exploration
                perturb = np.random.normal(0, 0.01, n_points)
                h_new = h + perturb
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr_new) * dx
                
                if c5_new < c5:
                    h = h_new
    
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx

    return (best_h.tolist(), final_c5, n_points)
