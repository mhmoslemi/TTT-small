import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Define constraint for sum(h) = 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Wrapper for the objective function to satisfy bounds and constraints
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Add the sum constraint to the bounds
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Use differential evolution for global optimization
    start_time = time.time()
    
    # Run differential evolution with time constraint
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=1000, 
        popsize=15, 
        seed=seed,
        constraints=constraints,
        strategy='best1bin',
        atol=1e-6,
        tol=1e-6
    )
    
    # Extract best solution
    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_opt = np.max(correlation) * dx
    
    # Final check
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt
    
    # Try local refinement with gradient-based optimization
    # Only if we have time left
    if time.time() - start_time < budget_s * 0.9:
        from scipy.optimize import minimize
        
        # Define a new objective for local optimization
        def local_objective(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            return c5
            
        # Local optimization with bounds
        bounds_list = [(0, 1) for _ in range(n_points)]
        try:
            result_local = minimize(
                local_objective,
                h_opt,
                method='L-BFGS-B',
                bounds=bounds_list,
                options={'maxiter': 500}
            )
            
            if result_local.success:
                h_local = result_local.x
                h_local = np.clip(h_local, 0, 1)
                h_local = h_local * (n_points / 2) / np.sum(h_local)
                correlation = np.correlate(h_local, 1 - h_local, mode='full')
                c5_local = np.max(correlation) * dx
                
                if c5_local < best_c5:
                    best_h = h_local
                    best_c5 = c5_local
        except:
            pass  # If local optimization fails, we keep current best
    
    return (best_h.tolist(), best_c5, n_points)
