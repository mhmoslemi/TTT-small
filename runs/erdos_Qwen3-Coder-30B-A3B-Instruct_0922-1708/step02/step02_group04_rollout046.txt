import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():

        n_points = len(h)

        n_points = 100

    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    last_best_c5 = float('inf')
    
    # Genetic algorithm parameters
    population_size = min(50, max(20, n_points // 4))
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_count = population_size // 10
    
    # Create initial population
    population = []
    fitness = []
    
    for _ in range(population_size):
        if 'initial_h_values' in globals() and len(population) == 0:
            h_individual = h.copy()
        else:
            h_individual = np.random.rand(n_points)
            h_individual = h_individual * (n_points / 2) / np.sum(h_individual)
            h_individual = np.clip(h_individual, 0, 1)
        population.append(h_individual)
    
    # Evaluation function
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate population
        fitness = []
        for h_individual in population:
            c5 = compute_c5(h_individual)
            fitness.append(c5)
        
        # Track best solution
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # If no improvement for many generations, reset or terminate
        if no_improvement_count > 100:
            break
        
        # Selection (tournament + roulette wheel)
        selected = []
        for _ in range(population_size):
            # Tournament selection
            tournament_size = 3
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Create new population through crossover and mutation
        new_population = []
        
        # Elitism: keep the best
        elite_indices = np.argsort(fitness)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover on segments
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:  # 10% chance to mutate each gene
                        change = np.random.normal(0, 0.03)
                        child[i] += change
                        
            # Ensure feasibility
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        # Trim to exactly population_size
        population = new_population[:population_size]
        
        # Occasionally increase mutation rate if no improvements
        if no_improvement_count > 20:
            mutation_rate = min(0.3, mutation_rate + 0.05)
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
