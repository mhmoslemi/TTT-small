import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    n_particles = 10
    
    # Multiple starting points
    particles = []
    for i in range(n_particles):
        if i == 0 and 'initial_h_values' in globals():
            particles.append(h.copy())
        else:
            h_init = np.random.uniform(0, 1, n_points)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            particles.append(h_init)
    
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    particle_best = [float('inf')] * n_particles
    particle_best_h = [None] * n_particles
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update each particle
        for p in range(n_particles):
            h = particles[p]
            
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < particle_best[p]:
                particle_best[p] = c5
                particle_best_h[p] = h.copy()
                
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Compute gradient with finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
                
            # Update with momentum
            momentum = 0.9
            velocities[p] = momentum * velocities[p] - lr * grad
            h_new = h + velocities[p]
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Perturb occasionally to escape local minima
            if np.random.rand() < 0.05:
                noise = np.random.normal(0, 0.01, n_points)
                h_new = np.clip(h_new + noise, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
            particles[p] = h_new
            
            # Adaptive learning rate
            if c5 < particle_best[p]:
                lr = min(lr * 1.1, 0.1)
            else:
                lr = max(lr * 0.9, 1e-4)
                
    return (best_h.tolist(), best_c5, n_points)
