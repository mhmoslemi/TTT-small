import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided construction or start with a good initial guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Start with a piecewise constant function with 3 segments
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_c5(h_values):
        # Compute correlation to find maximum overlap
        correlation = np.correlate(h_values, 1 - h_values, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Record performance
    last_improvement = 0
    iteration = 0
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1 and iteration < 10000:
        iteration += 1
        
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
        
        # Stop if no improvement for a while
        if iteration - last_improvement > 500:
            break
            
        # Create small perturbation
        perturb = np.random.normal(0, 0.01, n_points)
        h_new = h + perturb
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        # Occasionally do larger steps
        if np.random.rand() < 0.1:
            i = np.random.randint(n_points)
            h_new[i] = np.random.rand()
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            
        h = h_new
        
        # Every 100 iterations, see if we're stuck
        if iteration % 100 == 0 and iteration - last_improvement > 300:
            # Jump to a new random configuration
            h = np.random.rand(n_points) * 0.5 + 0.25
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
