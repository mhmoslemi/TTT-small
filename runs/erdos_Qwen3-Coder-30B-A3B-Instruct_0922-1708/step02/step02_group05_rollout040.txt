import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Mutation-based optimization with local search
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Create new candidate via mutation
        h_new = h.copy()
        
        # Perturb a few random indices
        num_perturb = max(1, n_points // 20)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            delta = np.random.normal(0, 0.05)
            h_new[i] += delta
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        # Rescale to satisfy integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept if better or with some probability (simulated annealing)
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        if c5_new < c5 or np.random.rand() < 0.05:
            h = h_new
            
    return (best_h.tolist(), best_c5, n_points)
