import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
    else:
        # Create a simple binary pattern
        h = np.zeros(n_points)
        # Pulse pattern: 5 pulses of width 0.2, spaced by 0.25
        for i in range(5):
            start = i * 0.25
            end = start + 0.2
            start_idx = int(start/dx)
            end_idx = int(end/dx)
            h[start_idx:end_idx] = 1.0
        
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use global optimization to find better solution
    result = differential_evolution(objective, bounds, maxiter=100, seed=seed, 
                                   popsize=15, mutation=(0.5, 1), recombination=0.7)
    
    # Extract best solution from optimization
    final_h = np.clip(result.x, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    # Evaluate final C5
    correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h.copy()
    
    # Try local refinement with gradient information
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 200:
        iteration += 1
        
        # Local search around current best
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = final_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - final_c5) / eps
            
        # Update with gradient descent
        lr = 0.0005
        new_h = final_h - lr * grad
        
        # Clip and rescale
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate new solution
        corr_new = np.correlate(new_h, 1 - new_h, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < final_c5:
            final_h = new_h.copy()
            final_c5 = c5_new
        else:
            # Occasionally try small random perturbations    
            if np.random.rand() < 0.1:
                pert = np.random.normal(0, 0.005, n_points)
                new_h = final_h + pert
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                corr_new = np.correlate(new_h, 1 - new_h, mode='full')
                c5_new = np.max(corr_new) * dx
                
                if c5_new < final_c5:
                    final_h = new_h.copy()
                    final_c5 = c5_new
    
    # Final evaluation
    correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (final_h.tolist(), final_c5, n_points)
