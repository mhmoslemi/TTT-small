import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Set up constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
    
    # Cache for correlation computations
    correlation_cache = {}
    
    def compute_c5(h_vec):
        """Computes the maximum overlap integral C5 for given h."""
        # Normalize h to satisfy constraint
        h_norm = h_vec * (n_points / 2) / np.sum(h_vec)
        # Compute correlation with (1-h)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        # Max over all lags
        c5 = np.max(corr) * dx
        return c5
    
    def compute_gradient(h_vec, eps=1e-6):
        """Numerical gradient computation of C5."""
        h_norm = h_vec * (n_points / 2) / np.sum(h_vec)
        c5_base = compute_c5(h_vec)
        grad = np.zeros_like(h_vec)
        
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            grad[i] = (compute_c5(h_plus) - c5_base) / eps

        return grad
    
    # Initialize optimization
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    iteration = 0
    
    # PSO Parameters
    n_particles = 20
    w = 0.7
    c1, c2 = 0.9, 0.9
    
    # Initialize particles
    particles = [h.copy() for _ in range(n_particles)]
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    pbest = [p.copy() for p in particles]
    pbest_scores = [compute_c5(p) for p in particles]
    gbest = best_h.copy()
    gbest_score = best_c5
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # PSO Update Logic
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) +
                            c2 * r2 * (gbest - particles[i]))
            
            # Update particle position
            particles[i] += velocities[i]
            
            # Apply bounds and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            score = compute_c5(particles[i])
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                
                # Update global best if needed
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    
        # Update best solution found so far
        if gbest_score < best_c5:
            best_c5 = gbest_score
            best_h = gbest.copy()
            
        # Apply local refinement every 20 iterations
        if iteration % 20 == 0:
            # Gradient-based local search at gbest
            for _ in range(5):
                grad = compute_gradient(gbest)
                gbest -= 0.001 * grad
                gbest = np.clip(gbest, 0, 1)
                gbest = gbest * (n_points / 2) / np.sum(gbest)
                current_c5 = compute_c5(gbest)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = gbest.copy()
        
        # Early termination check
        if time.time() - start_time > budget_s - 5:
            break
            
    return (best_h.tolist(), best_c5, n_points)
