import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured approach
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Use a binary comb-like structure
        # Create 5 pulses, each of width 0.2, spaced to make integral = 1
        h = np.zeros(n_points)
        pulse_width = 0.2
        pulse_start = np.linspace(0, 1.0, 5, endpoint=False)  # 0, 0.25, 0.5, 0.75, 1.0
        pulse_positions = [i * 0.25 for i in range(5)]
        for pos in pulse_positions:
            start_idx = int(pos / 2.0 * n_points)
            end_idx = int((pos + pulse_width) / 2.0 * n_points)
            if end_idx > start_idx:
                h[start_idx:end_idx] = 1.0
                
        # Scale to make integral = 1
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Simple local search with simulated annealing approach
    temp = 1.0
    cooling_rate = 0.9995
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Create a neighbor solution by small random perturbations
        h_new = h.copy()
        
        # Randomly select some indices to modify
        num_changes = np.random.randint(1, max(1, n_points // 20))
        indices = np.random.choice(n_points, num_changes, replace=False)
        
        for idx in indices:
            # Small random update
            delta = np.random.normal(0, 0.02)
            h_new[idx] += delta
            
        # Project back to feasible bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy the constraint ∫h(x)dx = 1
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept or reject based on Metropolis criterion
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        if c5_new < c5 or np.random.random() < np.exp(-(c5_new - c5) / temp):
            h = h_new.copy()
            
        temp *= cooling_rate
        
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
