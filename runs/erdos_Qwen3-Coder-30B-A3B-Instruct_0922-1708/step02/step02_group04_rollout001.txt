import time

def run(seed=42, budget_s=600, **kwargs):

        h[mid:] = 0.0
        # Add small noise
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Try adaptive perturbations
        h_new = h.copy()
        
        # Select random indices to modify
        num_modify = max(1, n_points // 20)
        indices = np.random.choice(n_points, num_modify, replace=False)
        
        # Apply perturbations
        for i in indices:
            delta = np.random.normal(0, 0.01)
            h_new[i] += delta
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)

        if np.random.rand() < 0.05:
            # Do a larger random change
            i = np.random.randint(n_points)

            h_new = h_new * (n_points / 2) / np.sum(h_new)

        # Occasionally reinitialize using a better seed or structure
        if iteration % 1000 == 0:

    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
