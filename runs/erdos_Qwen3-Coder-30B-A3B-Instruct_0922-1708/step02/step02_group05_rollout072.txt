import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    swarm_size = 50
    inertia_weight = 0.7
    c1 = c2 = 2.0
    max_iterations = 500
    
    # Initialize particles and velocities
    particles = np.random.rand(swarm_size, n_points) * 0.5
    for i in range(swarm_size):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        
    velocities = (np.random.rand(swarm_size, n_points) - 0.5) * 0.1
    pbest = particles.copy()
    pbest_vals = np.full(swarm_size, float('inf'))
    gbest_idx = 0
    gbest_val = float('inf')
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Evaluate initial particles
    for i in range(swarm_size):
        c5 = compute_c5(particles[i])
        if c5 < pbest_vals[i]:
            pbest_vals[i] = c5
            pbest[i] = particles[i].copy()
        if c5 < gbest_val:
            gbest_val = c5
            gbest_idx = i
    
    # PSO loop
    iteration = 0
    stagnation_count = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        stagnation_count += 1
        
        # Update velocity and position for each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            inertia = inertia_weight
            if iteration > max_iterations * 0.8:
                inertia = 0.4
                
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (pbest[gbest_idx] - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate and update bests
            c5 = compute_c5(particles[i])
            
            if c5 < pbest_vals[i]:
                pbest_vals[i] = c5
                pbest[i] = particles[i].copy()
                
            if c5 < gbest_val:
                gbest_val = c5
                gbest_idx = i
                stagnation_count = 0
                
        # Periodic diversity maintenance
        if stagnation_count > 50 and iteration % 20 == 0:
            for i in range(swarm_size // 5):
                idx = np.random.randint(0, swarm_size)
                particles[idx] = np.random.rand(n_points) * 0.5
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                # Update bests for this particle
                c5 = compute_c5(particles[idx])
                if c5 < pbest_vals[idx]:
                    pbest_vals[idx] = c5
                    pbest[idx] = particles[idx].copy()
                if c5 < gbest_val:
                    gbest_val = c5
                    gbest_idx = idx
                
        # Early termination if we reached target
        if gbest_val <= 0.3807:
            break
    
    return (pbest[gbest_idx].tolist(), gbest_val, n_points)
