def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:

            h_orig = np.array(initial_h_values)

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Try different optimization strategies
    strategies = [
        lambda h, lags: local_search_optimization(h, lags, dx, best_c5),
        lambda h, lags: gradient_based_optimization(h, lags, dx, best_c5),
        lambda h, lags: simple_heuristic_optimization(h, lags, dx, best_c5)
    ]
    
    strategy_idx = 0
    while time.time() - start_time < budget_s - 1:
        # Try the current strategy
        strategy = strategies[strategy_idx % len(strategies)]
        h_new, c5_new = strategy(h.copy(), lags)
        
        # Update if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new
        strategy_idx += 1
        
        # Occasionally reinitialize with better solution
        if strategy_idx % 100 == 0:
            h = best_h.copy()
            
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    return (best_h.tolist(), final_c5, n_points)

def local_search_optimization(h, lags, dx, best_c5):
    """Local search strategy."""
    n_points = len(h)
    
    # Try to improve by small perturbations
    for _ in range(5):
        idx = np.random.randint(0, n_points)
        step = np.random.uniform(-0.05, 0.05)
        h_new = h.copy()
        h_new[idx] += step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            return h_new, c5
            
    return h, best_c5

def gradient_based_optimization(h, lags, dx, best_c5):
    """Gradient-based optimization."""
    n_points = len(h)
    
    # Compute current value
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    
    # Simple gradient approximation
    grad = np.zeros_like(h)
    eps = 1e-4
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
        
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        
        grad[i] = (c5_plus - c5) / eps
    
    # Update using gradient descent
    lr = 0.02
    h_new = h - lr * grad
    
    # Clip and rescale
    h_new = np.clip(h_new, 0, 1)
    h_new = h_new * (n_points / 2) / np.sum(h_new)
    
    correlation = np.correlate(h_new, 1 - h_new, mode='full')
    c5_new = np.max(correlation) * dx
    
    return h_new, c5_new

def simple_heuristic_optimization(h, lags, dx, best_c5):
    """Simple heuristic optimization."""
    n_points = len(h)
    
    # Try constructing an "optimal" sawtooth pattern
    h_new = np.zeros(n_points)
    period = n_points // 4  # Periods of oscillation
    
    for i in range(n_points):
        if i % period < period // 2:
            h_new[i] = 0.0
        else:
            h_new[i] = 1.0
    
    # Scale to correct sum
    h_new = h_new * (n_points / 2) / np.sum(h_new)
    
    # Try to perturb gradually
    h_new += np.random.normal(0, 0.01, n_points)
    h_new = np.clip(h_new, 0, 1)
    h_new = h_new * (n_points / 2) / np.sum(h_new)
    
    correlation = np.correlate(h_new, 1 - h_new, mode='full')
    c5_new = np.max(correlation) * dx
    
    return h_new, c5_new
