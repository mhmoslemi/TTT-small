import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
        
        # Generate variations
        for _ in range(population_size - 1):
            h_var = h.copy()
            # Mutation
            mask = np.random.random(n_points) < 0.15
            noise = np.random.uniform(-0.2, 0.2, n_points)
            h_var += noise * mask
            h_var = np.clip(h_var, 0, 1)
            h_var = h_var * (n_points / 2) / np.sum(h_var)
            population.append(h_var.copy())
    else:
        # Initialize with random spikes
        for _ in range(population_size):
            h = np.zeros(n_points)
            num_spikes = np.random.randint(5, 11)
            positions = np.random.choice(n_points, num_spikes, replace=False)
            values = np.random.uniform(0.3, 0.7, num_spikes)
            h[positions] = values
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < 1000:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            correlation = correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection (top 50%)
        selected_indices = np.argsort(fitness)[:population_size//2]
        selected_population = [population[i] for i in selected_indices]
        
        # Generate new population
        new_population = selected_population.copy()
        
        # Elitism: keep top performer
        if len(selected_population) > 0:
            top_h = selected_population[0]
            new_population.append(top_h.copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if len(selected_population) >= 2:
                parents = np.random.choice(len(selected_population), 2, replace=False)
                parent1, parent2 = selected_population[parents[0]], selected_population[parents[1]]
                
                # Crossover
                if np.random.random() < 0.7:
                    # Single-point crossover
                    crossover_point = np.random.randint(1, n_points - 1)
                    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                else:
                    child1, child2 = parent1.copy(), parent2.copy()
                
                # Mutation
                for child in [child1, child2]:
                    mask = np.random.random(n_points) < 0.15
                    noise = np.random.uniform(-0.2, 0.2, n_points)
                    child += noise * mask
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Create new random individual
                h = np.zeros(n_points)
                num_spikes = np.random.randint(5, 11)
                positions = np.random.choice(n_points, num_spikes, replace=False)
                values = np.random.uniform(0.3, 0.7, num_spikes)
                h[positions] = values
                h = h * (n_points / 2) / np.sum(h)
                new_population.append(h)
        
        population = new_population[:population_size]
    
    return (best_h.tolist(), best_c5, n_points)
