from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Evolutionary strategy parameters
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    generations = 0
    no_improvement_count = 0
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h_new = h + np.random.normal(0, 0.05, n_points)
        else:
            h_new = np.random.rand(n_points) * 0.5
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        population.append(h_new)
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate population
        fitness_scores = []
        for h_ind in population:
            correlation = np.correlate(h_ind, 1 - h_ind, mode='full')
            c5 = np.max(correlation) * dx
            fitness_scores.append(c5)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness_scores)
        if fitness_scores[min_fitness_idx] < best_c5:
            best_c5 = fitness_scores[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Selection: keep top 20%
        sorted_indices = np.argsort(fitness_scores)
        elite_size = population_size // 5
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Create new population through crossover and mutation
        new_population = elite.copy()
        
        # Crossover
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(elite) >= 2:
                i1, i2 = np.random.choice(len(elite), 2, replace=False)
                parent1, parent2 = elite[i1], elite[i2]
                # Average crossover
                child = (parent1 + parent2) / 2.0
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            else:
                # Mutation
                if len(elite) > 0:
                    parent = elite[np.random.randint(len(elite))]
                    child = parent.copy()
                    # Mutate 20% of elements
                    mutate_indices = np.random.choice(n_points, int(0.2 * n_points), replace=False)
                    for i in mutate_indices:
                        child[i] += np.random.normal(0, 0.02)
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                    new_population.append(child)
                else:
                    # Random initialization if no elite
                    child = np.random.rand(n_points) * 0.5
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                    new_population.append(child)
        
        population = new_population[:population_size]
        
        # Increase mutation if no improvement for 100 generations
        if no_improvement_count > 100:
            mutation_rate = 0.3
        else:
            mutation_rate = 0.2
            
        generations += 1
    
    # Final check with best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
