import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Genetic algorithm approach
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    tournament_size = 3
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals() and iteration == 0:
            # Use the initial solution as first member
            individual = h.copy()
        else:
            individual = np.random.random(n_points) * 0.5
        # Normalize to sum to n_points / 2
        individual = individual * (n_points / 2) / np.sum(individual)
        individual = np.clip(individual, 0, 1)
        population.append(individual)
    
    iteration = 0
    gen = 0
    while time.time() - start_time < budget_s - 1:
        gen += 1
        iteration += 1
        
        # Evaluate fitness (inverse of C5)
        fitness_scores = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness_scores.append(-c5)  # Negative because we minimize
            
        # Find best solution in current population
        best_in_pop_idx = np.argmax(fitness_scores)
        if fitness_scores[best_in_pop_idx] < -best_c5:
            best_c5 = -fitness_scores[best_in_pop_idx]
            best_h = population[best_in_pop_idx].copy()
        
        # Selection: tournament
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            parent = population[winner_idx].copy()
            new_population.append(parent)
        
        # Crossover and mutation
        next_generation = []
        for i in range(0, population_size, 2):
            parent1 = new_population[i]
            parent2 = new_population[i+1] if i+1 < population_size else new_population[0]
            
            # Crossover
            if np.random.random() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.random() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.1)
                if np.random.random() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.1)
            
            # Ensure feasible solutions
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            next_generation.extend([child1, child2])
            
        population = next_generation[:population_size]
    
    # Final evaluation with original method
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Verification
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) * dx - 1) < 1e-10
    
    return (best_h.tolist(), final_c5, n_points)
