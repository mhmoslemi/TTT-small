import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Compute the maximum overlap for given h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def initialize_from_parent():
    """Initialize from the parent construction if available"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        dx = 2.0 / n_points
        # Normalize to exact integral of 1
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        return h, n_points
    else:
        # Default initialization
        n_points = 45
        h = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        return h, n_points

def update_h_from_segments(segments, n_points):
    """Convert segments to step function h"""
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for v, a, b in segments:
        start_idx = int(a / dx)
        end_idx = int(b / dx)
        if start_idx < n_points:
            h[start_idx:end_idx] = v
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from parent or default
    h, n_points = initialize_from_parent()
    dx = 2.0 / n_points
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Segment-based approach
    # Try different numbers of segments to find a good configuration
    for n_segments in [3, 4, 5]:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize segments (value, start, end)
        segments = []
        segment_width = 2.0 / n_segments
        
        # Create segments with non-uniform spacing based on known good solutions
        for i in range(n_segments):
            v = np.random.rand() * 0.7 + 0.3  # Random valid value
            a = i * segment_width
            b = (i + 1) * segment_width
            segments.append((v, a, b))
        
        # Optimization with Nelder-Mead
        def objective(params):
            # Map parameters to segments
            segments_new = []
            for i in range(n_segments):
                # Values
                v = np.clip(params[i], 0, 1)
                # Start positions (but must be ordered)
                a = params[n_segments + i] if i == 0 else segments_new[i-1][2]
                b = a + 0.1 * (1 + np.random.rand())  # Perturb width
                # Clamp position to [0,2]
                a = np.clip(a, 0, 2)
                b = np.clip(b, a, 2)
                segments_new.append((v, a, b))
            
            # Normalize segment widths to maintain integral = 1
            total_width = sum(b - a for _, a, b in segments_new)
            if total_width > 0:
                scale = 2.0 / total_width
                for i in range(n_segments):
                    a = segments_new[i][1]
                    b = segments_new[i][2]
                    segments_new[i] = (segments_new[i][0], a, b * scale)
            
            h_new = update_h_from_segments(segments_new, n_points)
            # Ensure integral constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute overlap
            return evaluate_overlap(h_new, dx)
        
        # Use optimization loop
        max_iter = 100
        for iter in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Sample random starting parameters for this optimizer
            x0 = []
            for i in range(n_segments):
                x0.append(np.random.rand() * 0.7 + 0.3)  # value
                x0.append(i * 0.2)  # start pos
            x0 = np.array(x0)
            
            # Optimize with Nelder-Mead
            try:
                result = minimize(objective, x0, method='Nelder-Mead', 
                                tol=1e-6, options={'maxiter': 50})
                if not result.success:
                    continue
            except:
                continue
                    
            # Convert back to segments and h
            segments_new = []
            for i in range(n_segments):
                v = np.clip(result.x[i], 0, 1)
                a = result.x[n_segments + i] if i == 0 else segments_new[i-1][2]
                b = a + 0.5  # Approximate width
                segments_new.append((v, a, b))
            
            # Normalize widths again
            total_width = sum(b - a for _, a, b in segments_new)
            if total_width > 0:
                scale = 2.0 / total_width
                for i in range(n_segments):
                    a = segments_new[i][1]
                    b = segments_new[i][2]
                    segments_new[i] = (segments_new[i][0], a, b * scale)
                    
            h_new = update_h_from_segments(segments_new, n_points)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            
            c5 = evaluate_overlap(h_new, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        
    # Final accurate evaluation
    final_c5 = evaluate_overlap(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
