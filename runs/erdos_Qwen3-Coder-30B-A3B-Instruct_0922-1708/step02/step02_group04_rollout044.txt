import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided construction or start with a smart initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Start with a simple piecewise approximation of a good construction
        n_points = 45
        h = np.zeros(n_points)
        # Hardcoded good pattern: high value short segments
        # First segment high value
        h[0:10] = 0.7
        # Second segment low value
        h[10:25] = 0.1
        # Third segment high value
        h[25:35] = 0.8
        # Fourth segment low value
        h[35:] = 0.1
        
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Optimization with small step sizes for refined search
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Perturb all elements with increasing perturbation size
        step_size = 0.01 * (1 + 0.01 * iteration)
        perturb = np.random.normal(0, step_size, n_points)
        h_new = h + perturb
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize integral to 1
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # Occasionally reset to fine-tune more
            if iteration % 20 == 0:
                h = best_h.copy()
            else:
                h = h_new.copy()
        else:
            # Sometimes jump to local area
            if np.random.rand() < 0.05:
                h = h_new.copy()
            else:
                h = best_h.copy()
    
    # Final refinement to best found solution
    c5_final = compute_c5(best_h)
    
    return (best_h.tolist(), c5_final, n_points)
