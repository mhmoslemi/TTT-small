import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a simple construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Construct a binary-like pattern with 5 pulses
        h = np.zeros(n_points)
        pulse_width = 0.2
        spacing = 0.25
        for i in range(5):
            start = i * (pulse_width + spacing)
            end = start + pulse_width
            start_idx = int(start / 2.0 * n_points)
            end_idx = int(end / 2.0 * n_points)
            if end_idx > start_idx:
                h[start_idx:end_idx] = 1.0
                
        # Normalize so that sum(h) * dx = 1
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Try to improve using a local search approach
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Slightly perturb h to explore neighborhood
        h_new = h.copy()
        
        # Select a few random indices to perturb
        indices = np.random.choice(n_points, size=min(10, n_points//10), replace=False)
        
        for i in indices:
            # Small random perturbation
            perturb = np.random.uniform(-0.05, 0.05)
            h_new[i] += perturb
            
        # Ensure h_new stays in [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to ensure integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final computation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
