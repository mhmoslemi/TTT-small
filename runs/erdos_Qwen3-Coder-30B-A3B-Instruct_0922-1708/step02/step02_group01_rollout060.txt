import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Binary search bounds
    C_low = 0.0
    C_high = 0.5
    eps = 1e-4
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # For binary search
    while C_high - C_low > eps and time.time() - start_time < budget_s - 1:
        iteration += 1
        mid = (C_low + C_high) / 2
        
        # Formulate constraints for feasibility check
        # minimize 0 subject to A_ub * h <= b_ub, A_eq * h = b_eq
        # We want to check if there exists h such that all overlap constraints hold
        
        # Build constraint matrix for all shifts
        A_ub = []
        b_ub = []
        A_eq = []
        b_eq = []
        
        # Equality constraint: sum(h) = n_points / 2
        row_eq = np.ones(n_points)
        A_eq.append(row_eq)
        b_eq.append(n_points / 2)
        
        # Inequality constraints: for each shift k, 
        # sum(h[i] * (1 - h[(i+k) % n_points]) * dx) <= mid
        for k in range(n_points):
            row = np.zeros(n_points)
            for i in range(n_points):
                row[i] += dx * (1 - h[(i + k) % n_points])
                row[(i + k) % n_points] += -dx * h[i]
            A_ub.append(row)
            b_ub.append(mid)
            
        # Add bounds constraints: 0 <= h[i] <= 1
        for i in range(n_points):
            row = np.zeros(n_points)
            row[i] = 1
            A_ub.append(row)
            b_ub.append(1)
            A_ub.append(-row)
            b_ub.append(0)
            
        # Solve LP feasibility problem
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        A_eq = np.array(A_eq)
        b_eq = np.array(b_eq)
        
        # Try to solve feasibility problem
        try:
            # Use linear programming solver (defaults to 'highs')
            result = linprog(
                c=np.zeros(n_points),  # dummy objective
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=[(0, 1) for _ in range(n_points)],
                method='highs',
                options={'presolve': True}
            )
            
            if result.success:
                # feasible solution
                h_candidate = result.x
                # Compute actual C5 value
                conv = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(conv) * dx
                
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
                    
                C_high = mid
            else:
                C_low = mid
                
        except Exception as e:
            C_low = mid
            
        # Occasionally update h to best found so far
        if iteration % 5 == 0:
            h = best_h.copy()
            
    # Final evaluation of best solution
    conv = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(conv) * dx
    
    # Ensure tight final constraints
    h_final = best_h.copy()
    h_final = h_final / np.sum(h_final) * (n_points / 2)
    conv_final = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(conv_final) * dx
    
    return (h_final.tolist(), final_c5, n_points)
