import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a step function
        h = np.zeros(n_points)
        for i in range(5):
            start_idx = int(i * n_points / 5)
            end_idx = int((i + 1) * n_points / 5)
            h[start_idx:end_idx] = 1.0 / 5.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Define objective function for optimization
    def objective(h_vals):
        h_local = np.array(h_vals)
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * (n_points / 2) / np.sum(h_local)
        correlation = np.correlate(h_local, 1 - h_local, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Optimize using differential evolution with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}]
    
    start_time = time.time()
    iteration = 0
    
    # Perform optimization with early stopping
    try:
        result = differential_evolution(
            objective, 
            bounds, 
            seed=seed,
            maxiter=1000,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.7,
            adapt=(0.1, 0.9)
        )
        
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            correlation = np.correlate(final_h, 1 - final_h, mode='full')
            final_c5 = np.max(correlation) * dx
            
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
                
    except Exception:
        pass

    # Final refinement with gradient-based method
    max_iter = 1000
    lr = 0.01
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        eps = 1e-5
        grad = np.zeros(n_points)
        c5_base = objective(best_h)
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = objective(h_plus)
            grad[j] = (c5_plus - c5_base) / eps
            
        # Update
        new_h = best_h - lr * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Accept if better
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            
        if np.allclose(best_h, new_h, atol=1e-6):
            break

    # Final verification
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
