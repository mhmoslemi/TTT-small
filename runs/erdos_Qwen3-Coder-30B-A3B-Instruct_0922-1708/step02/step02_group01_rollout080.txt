import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Initialize multiple particles
    n_particles = 10
    particles = []
    for i in range(n_particles):
        if i == 0 and 'initial_h_values' in globals():
            # Use the initial construction for first particle
            h_init = h.copy()
        else:
            # Random initialization for other particles
            h_init = np.random.uniform(0, 1, n_points)
            h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Learning rate schedule
    lr_schedule = [0.01, 0.005, 0.002, 0.001]
    lr_idx = 0
    
    # Momentum
    momentum = 0.9
    
    # Velocity tracking for each particle
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Track the best particle index
    best_particle_idx = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        c5_vals = []
        for i, particle in enumerate(particles):
            # Compute the overlap integral for each particle
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            c5_vals.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
                best_particle_idx = i
                # Adjust learning rate if we improved significantly
                lr_idx = min(lr_idx + 1, len(lr_schedule) - 1)
        
        # Update each particle with gradient descent
        for i in range(n_particles):
            particle = particles[i]
            
            # Compute gradient with finite differences
            grad = np.zeros_like(particle)
            eps = 1e-4
            
            # Perturb each dimension
            for j in range(n_points):
                particle_plus = particle.copy()
                particle_plus[j] += eps
                particle_plus = np.clip(particle_plus, 0, 1)
                particle_plus = particle_plus * (n_points / 2) / np.sum(particle_plus)
                
                corr_plus = np.correlate(particle_plus, 1 - particle_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr_orig = np.correlate(particle, 1 - particle, mode='full')
                c5_orig = np.max(corr_orig) * dx
                
                grad[j] = (c5_plus - c5_orig) / eps
            
            # Adaptive learning rate
            lr = lr_schedule[lr_idx]
            
            # Velocity update
            velocities[i] = momentum * velocities[i] - lr * grad
            
            # Update particle
            particle_new = particle + velocities[i]
            
            # Clip and rescale
            particle_new = np.clip(particle_new, 0, 1)
            particle_new = particle_new * (n_points / 2) / np.sum(particle_new)
            
            particles[i] = particle_new
            
            # Occasionally perturb for exploration
            if np.random.random() < 0.05:
                noise = np.random.uniform(-0.02, 0.02, n_points)
                particles[i] = np.clip(particles[i] + noise, 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    return (best_h.tolist(), best_c5, n_points)
