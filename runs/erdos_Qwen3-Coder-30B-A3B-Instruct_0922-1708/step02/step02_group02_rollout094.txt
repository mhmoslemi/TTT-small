import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def objective(h_vec):
        # Rescale to sum to n_points/2
        h_rescaled = h_vec * (n_points / 2) / np.sum(h_vec)
        # Clip to valid range
        h_rescaled = np.clip(h_rescaled, 0, 1)
        # Compute correlation with 1-h
        correlation = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Handle constraint that sum(h) = n_points/2
    # We'll use a penalty method for the sum constraint
    
    def constrained_objective(h_vec):
        # Rescale to correct sum
        h_normalized = h_vec * (n_points / 2) / np.sum(h_vec)
        # Clip
        h_clipped = np.clip(h_normalized, 0, 1)
        # Compute objective
        correlation = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Optimization with DE
    try:
        result = differential_evolution(
            constrained_objective, 
            bounds, 
            maxiter=1000,
            popsize=15,
            recombination=0.7,
            seed=seed,
            atol=1e-6,
            disp=False
        )
        
        # Update best based on DE result
        if result.success:
            h_opt = result.x * (n_points / 2) / np.sum(result.x)
            h_opt = np.clip(h_opt, 0, 1)
            correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5_val = np.max(correlation) * dx
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
    except Exception as e:
        pass  # Fall back to gradient-based approach
    
    # Local improvement with gradient descent
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
            
        # Try to perturb with gradient descent
        grad = np.zeros_like(best_h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            # Rescale to sum constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_plus = np.clip(h_plus, 0, 1)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update with gradient descent
        lr = 0.005
        h_new = best_h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        best_h = h_new
        
        # Occasionally restart with random perturbation
        if iteration % 50 == 0:
            # Add some randomness
            noise = np.random.normal(0, 0.02, n_points)
            best_h = best_h + noise
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
