import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    particles = 10
    particle_h = [h.copy() for _ in range(particles)]
    particle_best_c5 = [float('inf')] * particles
    particle_best_h = [h.copy() for _ in range(particles)]
    particle_lr = [0.01] * particles
    particle_no_improve = [0] * particles
    
    # Multi-start optimization
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        for p in range(particles):
            h = particle_h[p].copy()
            
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < particle_best_c5[p]:
                particle_best_c5[p] = c5
                particle_best_h[p] = h.copy()
                particle_no_improve[p] = 0
            elif particle_no_improve[p] > 100:  # Perturb if no improvement for 100 steps
                h = particle_best_h[p].copy()
                noise = np.random.uniform(-0.05, 0.05, n_points)
                h += noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                particle_no_improve[p] = 0
                
            # Compute gradient with finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update with adaptive learning rate
            lr = particle_lr[p]
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particle_h[p] = h_new
            particle_no_improve[p] += 1
            
            # Adjust learning rate
            if particle_no_improve[p] < 50:
                particle_lr[p] = min(0.05, particle_lr[p] * 1.01)
            else:
                particle_lr[p] = max(0.0001, particle_lr[p] * 0.99)
        
        # Update global best
        for p in range(particles):
            if particle_best_c5[p] < best_c5:
                best_c5 = particle_best_c5[p]
                best_h = particle_best_h[p].copy()
    
    return (best_h.tolist(), best_c5, n_points)
