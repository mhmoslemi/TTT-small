import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use a gradient-based optimization approach with adaptive learning rate
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    def compute_gradient(h_vals):
        n = len(h_vals)
        grad = np.zeros(n)
        # Compute gradient w.r.t. each h[i]
        for i in range(n):
            # Gradients via finite differences: compute how small perturbation 
            # in h[i] affects the maximum overlap
            eps = 1e-6
            h_plus = h_vals.copy()
            h_minus = h_vals.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            # Normalize 
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            # Ensure feasibility
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            c5_plus = compute_c5(h_plus)
            c5_minus = compute_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    # Adaptive learning rate
    alpha = 0.01
    patience = 0
    max_patience = 20
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        grad = compute_gradient(h)
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        # Rescale to get correct integral
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Adaptive learning rate updates
        if patience > max_patience:
            alpha *= 0.5
            patience = 0
            
        if alpha < 1e-6:
            break
            
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
