import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Ensure sum constraint is met for initial guess
    h = h * (n_points / 2) / np.sum(h)
    
    # Use differential evolution for better global search
    start_time = time.time()
    
    # Run DE for a portion of the budget
    de_budget = budget_s - 1  # Leave 1 second for final validation
    
    # Create a wrapper that satisfies constraints
    def constrained_objective(h):
        # Normalize the vector to meet sum constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return objective(h)
    
    # Use differential evolution with constraints
    result = differential_evolution(
        constrained_objective,
        bounds,
        maxiter=1000,
        popsize=50,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        tol=1e-6,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    if not result.success:
        final_h = h
    else:
        final_h = np.clip(result.x, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    # Refine the solution with local search if time permits
    if time.time() - start_time < de_budget - 1:
        # Try gradient-based refinement on the DE result
        h_refined = final_h.copy()
        for _ in range(100):
            if time.time() - start_time > de_budget - 1:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c5_plus = objective(h_plus)
                c5 = objective(h_refined)
                grad[i] = (c5_plus - c5) / eps
            
            # Update step with adaptive learning rate
           _grad_norm = np.linalg.norm(grad)
            if _grad_norm > 1e-10:
                lr = min(0.1, 1.0 / (_grad_norm + 1e-8))
            else:
                lr = 0.01
                
            h_new = h_refined - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_refined = h_new
    
    # Final evaluation
    final_h = np.clip(h_refined, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = objective(final_h)
    
    # Return best solution found
    return (final_h.tolist(), float(final_c5), n_points)
