import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated annealing parameters
    T0 = 0.1
    tau = 1000
    start_time = time.time()
    
    # Initial evaluation
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Cooling schedule
        T = T0 / (1 + iteration / tau)
        
        # Generate candidate solution by adding noise
        h_new = h + np.random.normal(0, T * 0.1, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate candidate
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        candidate_c5 = np.max(corr_new) * dx
        
        # Accept or reject
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_c5 = candidate_c5
            
        # Track best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
