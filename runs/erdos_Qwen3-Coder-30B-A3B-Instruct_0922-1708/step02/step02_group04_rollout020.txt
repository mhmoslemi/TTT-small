import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use provided initial construction or create a basic one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize population
    population_size = 50
    population = []
    fitness = []
    
    # Add initial h as first member
    population.append(h.copy())
    correlation = correlate(h, 1 - h, mode='full')
    f = np.max(correlation) * dx
    fitness.append(f)
    
    # Fill population with random variations
    for i in range(1, population_size):
        h_new = np.random.rand(n_points)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        population.append(h_new)
        correlation = correlate(h_new, 1 - h_new, mode='full')
        f = np.max(correlation) * dx
        fitness.append(f)
    
    best_h = population[np.argmin(fitness)].copy()
    best_c5 = min(fitness)
    
    start_time = time.time()
    no_improvement_count = 0
    max_no_improvement = 100
    mutation_rate = 0.1
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Selection using roulette wheel
        fitness_array = np.array(fitness)
        inv_fitness = 1 / (fitness_array + 1e-10)  # Avoid division by zero
        probs = inv_fitness / np.sum(inv_fitness)
        selected_indices = np.random.choice(population_size, size=population_size, p=probs)
        
        # Crossover and mutation
        new_population = [best_h.copy()]  # Keep best
        new_fitness = [best_c5]
        
        for i in range(1, population_size, 2):
            idx1, idx2 = selected_indices[i], selected_indices[i+1]
            parent1, parent2 = population[idx1], population[idx2]
            
            # Crossover: one-point
            crossover_point = np.random.randint(1, n_points)
            offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutate
            for off in [offspring1, offspring2]:
                if np.random.rand() < mutation_rate:
                    mut_indices = np.random.choice(n_points, size=int(mutation_rate * n_points), replace=False)
                    for idx in mut_indices:
                        off[idx] += np.random.normal(0, 0.05)
                    off = np.clip(off, 0, 1)
                    # Rescale to sum = n_points/2
                    off = off * (n_points / 2) / np.sum(off)
                new_population.append(off)
                correlation = correlate(off, 1 - off, mode='full')
                f = np.max(correlation) * dx
                new_fitness.append(f)
        
        # Replace population
        population = new_population[:population_size]
        fitness = new_fitness[:population_size]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive mutation rate
        if no_improvement_count > 20:
            mutation_rate = min(0.5, mutation_rate + 0.05)
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
