import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    num_starts = 5
    max_iter = 500
    
    # Multi-start optimization
    for start in range(num_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate variation of the solution
        h_start = h.copy()
        if 'initial_h_values' in globals():
            noise_factor = 0.05
            noise = np.random.normal(0, noise_factor, n_points)
            h_start = h_start + noise
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
        else:
            # If no parent, generate sparse random pattern
            h_start = np.random.rand(n_points) * 0.5
            h_start = h_start * (n_points / 2) / np.sum(h_start)
            
        h_current = h_start.copy()
        
        # Adam-like optimization with small steps
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            c5_current = evaluate_c5(h_current, dx)
            
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
                
            # Compute gradient with finite differences
            eps = 1e-4
            grad = np.zeros_like(h_current)
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[j] = (c5_plus - c5_current) / eps
                
            # Adaptive update with momentum
            lr = 0.001
            h_current = h_current - lr * grad
            h_current = clip_and_scale(h_current, n_points)
    
    # Final check and cleanup
    final_c5 = evaluate_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
