import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with symmetric construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Start with a symmetric construction
        h = np.ones(n_points)
        mid = n_points // 2
        h[:mid] = 0.5
        h[mid:] = 1.5
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1 and iteration - last_improvement < 100:
        iteration += 1
        
        # Compute the current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
        
        # Compute local gradient estimate
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + 0.01)
            h_minus = h.copy()
            h_minus[i] = max(0.0, h[i] - 0.01)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) * dx / 0.02
        
        # Gradient-based update with adaptive step size
        step_size = 0.005 / (1.0 + iteration / 1000.0)
        h_new = h - step_size * grad
        
        # Apply neighborhood flips for escape from local minima
        if np.random.rand() < 0.1:
            # Flip neighbors
            for _ in range(n_points // 20):  # Flip about 5% of the points
                i = np.random.randint(n_points)
                h_new[i] = 1.0 - h_new[i]
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final validation and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
