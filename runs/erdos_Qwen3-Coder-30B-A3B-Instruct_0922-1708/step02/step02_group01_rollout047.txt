import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # GA parameters
    N_pop = 50
    C_rate = 0.8
    M_rate = 0.2
    elite_size = N_pop // 2
    max_generations = 1000
    disturbance_rate = 0.2
    stagnant_generations = 0
    max_stagnant = 50
    
    # Initialize population
    population = []
    population_fitness = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        population.append(h.copy())
        correlation = np.correlate(h, 1 - h, mode='full')
        fitness = np.max(correlation) * dx
        population_fitness.append(fitness)
    
    # Fill the rest of population randomly
    for _ in range(N_pop - len(population)):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand / np.sum(h_rand) * (n_points / 2)
        population.append(h_rand)
        correlation = np.correlate(h_rand, 1 - h_rand, mode='full')
        fitness = np.max(correlation) * dx
        population_fitness.append(fitness)
    
    # Track best solution
    best_idx = np.argmin(population_fitness)
    best_h = population[best_idx].copy()
    best_c5 = population_fitness[best_idx]
    
    # For tracking stagnation
    best_fitness_history = [best_c5]
    
    start_time = time.time()
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        for i in range(len(population)):
            if i < len(population_fitness):
                continue
            correlation = np.correlate(population[i], 1 - population[i], mode='full')
            fitness = np.max(correlation) * dx
            population_fitness.append(fitness)
            
        # Selection: Tournament selection
        new_population = []
        new_fitness = []
        
        # Elitism: keep best individuals
        sorted_indices = np.argsort(population_fitness)
        for i in range(elite_size):
            new_population.append(population[sorted_indices[i]].copy())
            new_fitness.append(population_fitness[sorted_indices[i]])
            
        # Tournament selection for remaining
        while len(new_population) < N_pop:
            # Tournament selection
            tournament_size = 2
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [population_fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Clone the winner
            new_individual = population[winner_idx].copy()
            
            # Crossover with another random individual
            if np.random.rand() < C_rate:
                other_idx = np.random.choice(len(population))
                crossover_point = np.random.randint(1, n_points - 1)
                new_individual[:crossover_point] = population[other_idx][:crossover_point]
                
            # Mutation
            if np.random.rand() < M_rate:
                for j in range(n_points):
                    if np.random.rand() < 0.1:  # mutation rate
                        new_individual[j] += np.random.normal(0, 0.05)
                        
                # Ensure boundary constraints
                new_individual = np.clip(new_individual, 0, 1)
                # Rescale to maintain sum constraint
                scale = (n_points / 2) / np.sum(new_individual)
                new_individual = new_individual * scale
                
            new_population.append(new_individual)
            correlation = np.correlate(new_individual, 1 - new_individual, mode='full')
            new_fitness.append(np.max(correlation) * dx)
            
        # Apply elitism for next generation
        population = new_population
        population_fitness = new_fitness
        
        # Check for improvement
        current_best = np.min(population_fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_idx = np.argmin(population_fitness)
            best_h = population[best_idx].copy()
            stagnant_generations = 0
            best_fitness_history.append(best_c5)
        else:
            stagnant_generations += 1
            
        # Disturbance if stagnation
        if stagnant_generations >= max_stagnant:
            for _ in range(int(N_pop * disturbance_rate)):
                idx = np.random.randint(0, N_pop)
                new_h = np.random.rand(n_points)
                new_h = new_h / np.sum(new_h) * (n_points / 2)
                population[idx] = new_h
                correlation = np.correlate(new_h, 1 - new_h, mode='full')
                population_fitness[idx] = np.max(correlation) * dx
            stagnant_generations = 0
            
        # Early termination if target reached
        if best_c5 < 0.3807:
            break
            
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
