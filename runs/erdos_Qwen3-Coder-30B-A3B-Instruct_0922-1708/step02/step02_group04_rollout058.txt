import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided construction or start with symmetric initial guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45

    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # LP solver-based approach to find improved solution
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5 for comparison
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try several local search variants
        if iteration % 2 == 0:
            # Gradient-like step with local adjustment
            # Approximate gradient using finite difference
            step_size = 0.01
            h_plus = h + step_size
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            if c5_plus < c5:
                h = h_plus
        else:
            # Local stochastic search with rescaling
            perturb = np.random.normal(0, 0.015, n_points)
            h_new = h + perturb
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Occasionally perform significant changes
            if np.random.rand() < 0.15:
                i = np.random.randint(n_points)
                h_new[i] = np.random.rand() * 0.5
                h_new = h_new * (n_points / 2) / np.sum(h_new)

            h = h_new
            
        # Periodic CLA (combined local and global adjustment)
        if iteration % 5 == 0:
            # Use a slightly different local search
            local_shift = np.random.normal(0, 0.01, n_points)
            h_local = h + local_shift
            h_local = np.clip(h_local, 0, 1)
            h_local = h_local * (n_points / 2) / np.sum(h_local)
            
            # Rescale to maintain constraints
            h = h_local
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Validate and compute final C5
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
