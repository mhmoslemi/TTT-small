import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Multi-start approach with 10 different initializations
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Initialize particles
    particles = []
    particles.append(h.copy())
    for _ in range(9):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        particles.append(h_rand)
    
    lr = 0.01
    momentum = 0.9
    velocities = [np.zeros(n_points) for _ in range(10)]
    particle_best_c5 = [float('inf')] * 10
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for p_idx, h_p in enumerate(particles):
            # Compute correlation for current particle
            correlation = np.correlate(h_p, 1 - h_p, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < particle_best_c5[p_idx]:
                particle_best_c5[p_idx] = c5
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_p.copy()
            
            # Compute gradient with finite differences
            grad = np.zeros_like(h_p)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_p.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update using momentum-based gradient descent
            velocities[p_idx] = momentum * velocities[p_idx] - lr * grad
            h_new = h_p + velocities[p_idx]
            
            # Apply constraint handling
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Occasionally perturb particles at low probability to escape local minima
            if np.random.random() < 0.05:
                noise = np.random.normal(0, 0.01, n_points)
                h_new = h_new + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
            particles[p_idx] = h_new
            
            # Reinitialize poor performing particles
            if c5 > 0.39 and iteration % 50 == 0:
                h_new = np.random.uniform(0, 1, n_points)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                particles[p_idx] = h_new
                velocities[p_idx] = np.zeros(n_points)
        
        # Adjust learning rate based on improvement
        if iteration % 100 == 0:
            lr = max(lr * 0.95, 1e-5)
    
    return (best_h.tolist(), best_c5, n_points)
