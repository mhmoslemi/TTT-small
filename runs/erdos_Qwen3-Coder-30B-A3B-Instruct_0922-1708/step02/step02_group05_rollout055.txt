import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation initialization (if time allows)
    if time.time() - start_time < budget_s * 0.6:
        # Create optimization variables
        h_var = cp.Variable(n_points)
        tau = cp.Variable()
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Get all shift indices for correlation
        shifts = range(n_points)
        
        # Add constraints for each shift
        for k in shifts:
            # Compute correlation constraint
            corr_expr = 0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    corr_expr += h_var[i] * (1 - h_var[j])
                else:
                    break
            constraints.append(corr_expr <= tau * n_points)
        
        # Problem
        prob = cp.Problem(cp.Minimize(tau), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_candidate = h_var.value
            if h_candidate is not None:
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                h = h_candidate.copy()
        except:
            pass
            
    # Gradient descent refinement
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate based on gradient magnitude
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
