    h = h * (n_points / 2) / np.sum(h)

    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    last_best = float('inf')
    
    # Optimization parameters
    base_step_size = 0.01
    max_step_size = 0.1

        iteration += 1
        
        # Compute the current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
            last_best = c5
        else:
            stagnation_count += 1
        
        # Adaptive step size based on progress
        current_step_size = min(base_step_size * (1 + stagnation_count * 0.01), max_step_size)
        
        # Random perturbations
        perturb = np.random.normal(0, current_step_size, n_points)

        h_new = np.clip(h_new, 0, 1)

    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
