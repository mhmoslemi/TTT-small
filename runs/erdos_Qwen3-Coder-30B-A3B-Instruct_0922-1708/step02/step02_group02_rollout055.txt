import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a binary comb-like pattern
        h = np.zeros(n_points)
        # Use 5 pulses with careful spacing
        m = 5
        w = 0.2  # pulse width
        s = 0.25  # spacing
        # Positions of the pulses
        start_positions = [0.0, 0.25, 0.5, 0.75, 1.0]
        for i in range(m):
            start = start_positions[i]
            end = start + w
            # Convert to indices
            start_idx = int(start / 2.0 * n_points)
            end_idx = int(end / 2.0 * n_points)
            if end_idx > start_idx:
                h[start_idx:end_idx] = 1.0
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Helper to compute overlap integral
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Gradient-free optimization using simulated annealing inspired approach
    # Principles: sample random perturbations and accept improvements or with some probability
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < 50000:
        iter_count += 1
        
        # Make a random change
        idx = np.random.randint(n_points)
        old_val = current_h[idx]
        
        # Mutate
        if np.random.rand() < 0.5:
            # Change by small amount
            delta = np.random.normal(0, 0.01) * (1.0 if np.random.rand() < 0.5 else -1.0)
            new_val = old_val + delta
        else:
            # Binary toggle
            new_val = 1.0 - old_val if old_val == 0.0 else 0.0
        
        # Ensure feasibility
        new_val = np.clip(new_val, 0, 1)
        current_h[idx] = new_val
        
        # Rescale to meet constraint
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # Evaluate
        new_c5 = compute_c5(current_h)
        
        # Accept or reject
        if new_c5 <= current_c5:
            current_c5 = new_c5
        else:
            # Accept with probability based on temperature (simplified)
            if np.random.rand() < 0.01:
                current_c5 = new_c5
            else:
                # Revert
                current_h[idx] = old_val
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()

    # Final computation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
