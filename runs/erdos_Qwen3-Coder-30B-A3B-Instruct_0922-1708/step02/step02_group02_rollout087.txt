import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        # Create variants
        for _ in range(population_size):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Random initialization
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    generation = 0
    no_improvement_count = 0
    mutation_rate = 0.2
    crossover_rate = 0.5
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness_scores = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness_scores.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Check for improvement
        if generation > 1:
            if fitness_scores[0] < best_c5:
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        # Adjust mutation rate if no improvement
        if no_improvement_count > 100:
            mutation_rate = 0.3
            no_improvement_count = 0
            
        # Selection: Keep top 20%
        selected_count = population_size // 5
        selected_population = population[:selected_count]
        selected_fitness = fitness_scores[:selected_count]
        
        # Replacement: Generate new population
        new_population = selected_population.copy()
        
        # Crossover and Mutation
        while len(new_population) < population_size:
            if np.random.random() < crossover_rate and len(selected_population) >= 2:
                # Crossover
                i1, i2 = np.random.choice(len(selected_population), 2, replace=False)
                parent1, parent2 = selected_population[i1], selected_population[i2]
                alpha = np.random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
                # Ensure sum constraint
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            else:
                # Mutation
                parent = selected_population[np.random.randint(len(selected_population))]
                child = parent.copy()
                # Mutate 20% of elements
                mutate_indices = np.random.choice(n_points, int(0.2 * n_points), replace=False)
                for i in mutate_indices:
                    child[i] += np.random.normal(0, 0.02)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
