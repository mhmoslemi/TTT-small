import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Multi-start approach with 10 particles
    particles = []
    for i in range(10):
        if i == 0 and 'initial_h_values' in globals():
            # Use initial construction as first particle
            h_init = h.copy()
        else:
            # Random initialization
            h_init = np.random.rand(n_points)
            h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init.copy())
    
    # Keep track of best solution for each particle
    particle_best_c5 = [float('inf')] * 10
    particle_best_h = [h.copy() for h in particles]
    
    # Parameters
    lr = 0.01
    momentum = 0.9
    noise_prob = 0.05
    
    # Store velocities
    velocities = [np.zeros(n_points) for _ in range(10)]
    
    while time.time() - start_time < budget_s - 1:  # Leave some buffer
        iteration += 1
        
        # Update all particles in parallel
        for p_idx in range(10):
            h = particles[p_idx]
            
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < particle_best_c5[p_idx]:
                particle_best_c5[p_idx] = c5
                particle_best_h[p_idx] = h.copy()
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update with momentum
            velocities[p_idx] = momentum * velocities[p_idx] - lr * grad
            h_new = h + velocities[p_idx]
            
            # Perturb with noise occasionally
            if np.random.random() < noise_prob:
                noise = np.random.normal(0, 0.02, n_points)
                h_new += noise
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[p_idx] = h_new
        
        # Occasionally restart with better points
        if iteration % 50 == 0:
            lr = max(0.001, lr * 0.995)
            
            # Restart particles from good solutions
            for p_idx in range(10):
                if particle_best_c5[p_idx] < 0.381:
                    noise = np.random.normal(0, 0.01, n_points)
                    h_new = particle_best_h[p_idx] + noise
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    particles[p_idx] = h_new
    
    return (best_h.tolist(), best_c5, n_points)
