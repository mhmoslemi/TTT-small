import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation to get a better starting point (first 60% of time)
    time_left = budget_s * 0.6
    conv_start = time.time()
    
    # Use CVXPY for a convex relaxation approach
    try:
        h_vars = Variable(n_points)
        # Objective: approximate minimization of max correlation
        # We linearize the problem for convexity
        constraints = [h_vars >= 0, h_vars <= 1, sum(h_vars) == n_points / 2]
        prob = Problem(Minimize(0), constraints)
        prob.solve(solver=ECOS, verbose=False)
        if prob.status == 'optimal':
            h = np.array(h_vars.value).reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    # Gradient descent refinement (remaining 40% of time)
    time_left = budget_s * 0.4
    gd_start = time.time()
    
    # Function to evaluate C5 given current h
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Evaluate initial point
    current_c5 = compute_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Gradient-based optimization
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute finite difference gradient 
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
