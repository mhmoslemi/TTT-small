import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction or use a block function
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Create a piecewise constant function
        h_init = np.zeros(n_points)
        h_init[20:40] = 1.0  # Two intervals
        h_init[60:80] = 1.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)  # Rescale
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define optimization variables
    h = Variable(n_points)
    
    # Objective: minimize the maximum correlation
    # We'll solve the dual formulation using cvxpy
    
    # Compute the correlation matrix for all lags
    # Instead of full convolution, use a simpler yet effective method
    # Try optimization focusing on the structure
    
    # Optimize directly using a gradient-based method
    # But start with improved initialization
    
    x = h_init.copy()
    
    def compute_c5(h_vec):
        # Compute correlation efficiently
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Gradient descent with projection
    lr = 0.01
    max_iter = 10000
    tol = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = compute_c5(x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = x.copy()
        
        # Compute gradient numerically
        grad = np.zeros_like(x)
        eps = 1e-5
        
        for i in range(n_points):
            x_plus = x.copy()
            x_plus[i] += eps
            x_plus = np.clip(x_plus, 0, 1)
            x_plus = x_plus * (n_points / 2) / np.sum(x_plus)
            
            c5_plus = compute_c5(x_plus)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update step
        x_new = x - lr * grad
        
        # Project onto constraints
        x_new = np.clip(x_new, 0, 1)
        x_new = x_new * (n_points / 2) / np.sum(x_new)
        
        # Check convergence
        if np.linalg.norm(x_new - x) < tol:
            break
            
        x = x_new
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
