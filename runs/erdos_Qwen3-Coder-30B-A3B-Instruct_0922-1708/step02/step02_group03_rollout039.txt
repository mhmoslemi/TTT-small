from scipy.optimize import minimize
import time

    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

        return np.sum(h_vec) - n_points / 2

    patience_counter = 0

        # Use minimize with SLSQP for better convergence
        try:

                            constraints=constraints, options={'maxiter': 5, 'ftol': 1e-6})
            if result.success:
                h = result.x
                # Ensure bounds and sum constraint
                h = np.clip(h, 0, 1)

            eps = 1e-6
