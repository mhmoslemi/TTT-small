import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Adam optimizer parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.001
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the active lag k that maximizes the correlation
        max_idx = np.argmax(correlation)
        k = max_idx - (len(h) - 1)
        
        # Compute exact gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Gradient contribution from h[i] to lag k
            # The derivative of correlation[i] w.r.t. h[i] is
            # (1-h[i-k]) if k >= 0 and i-k is valid, and (1-h[i+k]) if k < 0 and i+k is valid
            # i-k = 0 means the current h[i] appears in the (i-k)th term, 
            # though we adjust the indexing.
            if k >= 0:
                # For h[i] to appear in the lag k term, we look at h[i-k]
                if i - k >= 0:
                    grad[i] += -1.0  # -1 from the 1-h[i-k] term
                if i + k < n_points:
                    grad[i] += 1.0  # +1 from the h[i+k] term
            else:
                # k < 0, so we look at h[j-k] with j = i+k  
                # So h[i] appears in h[i+k], but i+k might be out of bounds
                if i + k >= 0:
                    grad[i] += 1.0
                if i - k < n_points:
                    grad[i] += -1.0
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        # Bias correction
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        
        # Update h
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
