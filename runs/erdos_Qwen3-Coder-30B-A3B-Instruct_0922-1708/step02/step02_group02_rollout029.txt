import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or use a blocky function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a blocky starting function
        h = np.zeros(n_points)
        h[:20] = 0.5
        h[20:40] = 1.0
        h[40:60] = 0.5
        h[60:80] = 1.0
        h[80:] = 0.5
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Try to improve with convex optimization
    iteration = 0
    try:
        while time.time() - start_time < budget_s - 1:
            iteration += 1
            
            # Compute current C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Convex optimization approach
            # Optimize h to minimize max_k integral of h(x)(1-h(x+k))
            h_var = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
            
            # Objective: minimize max correlation
            # Approximate the max by using a large penalty
            # But since we can't directly use max in CVXPY, we numerically sample
            
            # Instead, try a direct gradient descent approach
            # But first let's try a different and more promising approach
            
            break  # Skip CVX optimization to prefer later technique
    except:
        pass
    
    # Gradient-based refinement using scipy.optimize
    try:
        # Use scipy minimize to refine the solution
        n_points = 100
        
        # Objective function
        def obj(h_vec):
            h = np.array(h_vec)
            h = h / np.sum(h) * (n_points / 2)
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = -np.max(correlation) * dx  # Negative because we minimize
            return c5
            
        # Constraints
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        def constraint_lower(h_vec):
            return h_vec - 0
            
        def constraint_upper(h_vec):
            return 1 - h_vec
            
        # Initial point
        h_init = best_h.copy()
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Constraints for scipy.optimize
        constraints = [
            {'type': 'eq', 'fun': constraint_sum},
        ]
        
        # Try a few optimization steps
        for i in range(10):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Use a quasi-Newton method with bounds
            result = minimize(
                obj,
                h_init,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 20}
            )
            
            if result.success:
                h_init = result.x
            else:
                # fallback to simple gradient descent if needed
                pass
                
            # Check if we found a better solution
            correlation = np.correlate(h_init, 1 - h_init, mode='full')
            c5 = np.max(correlation) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_init.copy()
            
    except Exception as e:
        # If optimization fails, return previous best
        pass
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
