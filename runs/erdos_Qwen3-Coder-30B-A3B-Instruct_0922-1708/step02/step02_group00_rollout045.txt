import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Optimized gradient-based approach with deeper exploration
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Use a more sophisticated optimization approach with adaptive steps
        # Implementing a version of LBFGS with constraint handling
        # But also mixing in local gradient approximations to avoid local minima
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # Gradient approximation using finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update using adaptive learning rate and momentum
        if iteration == 1:
            velocity = np.zeros_like(h)
        lr = 0.01 / (1 + iteration / 100)
        momentum = 0.9
        velocity = momentum * velocity - lr * grad
        h_new = h + velocity
        
        # Better projection and normalization
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Occasionally add a small random perturbation to escape local optima
        if iteration % 100 == 0:
            h_new += np.random.normal(0, 0.005, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
        h = h_new
        
        # Occasionally try a heuristic restart with a better initial
        if iteration % 500 == 0:
            # Use a strategy of mixing the best solution so far with random samples
            # but keep some structure from previous best
            h = 0.3 * best_h + 0.7 * np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
