import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given step function h."""
    correlation = correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def rescale_h(h, n_points):
    """Rescale h to have sum = n_points / 2."""
    return h * (n_points / 2) / np.sum(h)

def mutate_individual(h, mutation_rate=0.15, noise_range=0.2):
    """Apply mutation to a single individual."""
    h_mut = h.copy()
    num_mutate = max(1, int(len(h) * mutation_rate))
    indices = np.random.choice(len(h), num_mutate, replace=False)
    for i in indices:
        noise = np.random.uniform(-noise_range, noise_range)
        h_mut[i] += noise
    h_mut = np.clip(h_mut, 0, 1)
    return rescale_h(h_mut, len(h))

def crossover_individuals(h1, h2):
    """Perform crossover between two individuals."""
    n = len(h1)
    crossover_point = np.random.randint(1, n)
    h_child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h_child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    return rescale_h(h_child1, n), rescale_h(h_child2, n)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        if len(h_start) != n_points:
            x_orig = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, n_points)
            h_start = np.interp(x_new, x_orig, h_start)
        h_start = np.clip(h_start, 0, 1)
        h_start = rescale_h(h_start, n_points)
        population.append(h_start)
        # Add random variations
        for _ in range(population_size - 1):
            population.append(mutate_individual(h_start))
    else:
        # Generate diverse initial population
        for _ in range(population_size):
            # Create random spike-like pattern
            h_random = np.zeros(n_points)
            num_spikes = np.random.randint(5, 11)
            for _ in range(num_spikes):
                pos = np.random.randint(0, n_points)
                height = np.random.uniform(0.3, 0.9)
                h_random[pos] = height
            h_random = rescale_h(h_random, n_points)
            population.append(h_random)
    
    # Evolution loop
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    generation = 0
    max_generations = 1000
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection (tournament)
        selected_indices = []
        tournament_size = 3
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            winner_idx = min(tournament_indices, key=lambda i: fitness[i])
            selected_indices.append(winner_idx)
        
        # Create new population
        new_population = []
        elite_count = population_size // 10
        # Keep best individuals (elitism)
        sorted_indices = np.argsort(fitness)[:elite_count]
        for i in sorted_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Crossover
            parent1_idx, parent2_idx = np.random.choice(selected_indices, 2, replace=False)
            if np.random.rand() < 0.7:  # 70% chance of crossover
                child1, child2 = crossover_individuals(population[parent1_idx], population[parent2_idx])
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Just mutate
                parent = population[np.random.choice(selected_indices)]
                child = mutate_individual(parent)
                new_population.append(child)
        
        # Ensure population size
        population = new_population[:population_size]
        generation += 1
    
    if best_h is None:
        # Fallback to last evaluated best
        fitness = [evaluate_h(h, dx) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = rescale_h(best_h, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
