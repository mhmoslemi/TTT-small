import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():

        if len(h) != n_points:

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Convex relaxation to get initial feasible point
    start_time = time.time()
    
    # Solve convex relaxation
    h_relaxed = h.copy()
    
    # Use a simple heuristic to initialize, then refine with gradient descent
    # First let's do a small convex optimization to get a better starting point
    
    # Even simpler: use a smooth approximation
    # Try a simple initial guess that might be better for smoothness
    
    # Alternative: start with a better structured solution
    # We might have a better solution by using an iterative approach

    # Quick structured initial hypothesis
    h = np.zeros(n_points)
    # Create a piecewise linear function that's feasible
    mid = n_points // 2
    for i in range(mid):
        h[i] = 0.5 + 0.5 * (i / mid)
    for i in range(mid, n_points):
        h[i] = 0.5 + 0.5 * ((n_points - i) / (n_points - mid))
    
    # Normalize
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Now refine with projected gradient descent
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        # The full convolution has 2*n_points-1 elements, center is at index n_points-1
        # max index of overlap is n_points-1 -> check elements around center
        return np.max(correlation) * dx
    
    def compute_grad(h_vec, eps=1e-5):
        grad = np.zeros_like(h_vec)
        c5_current = compute_c5(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_current) / eps
        return grad
    
    iter_time = time.time() - start_time
    remaining_time = budget_s - iter_time
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    max_iter = 2000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        c5_current = compute_c5(h)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        grad = compute_grad(h)
        
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            lr = min(0.1, 1.0 / (grad_norm + 1e-8))
        else:
            lr = 0.01
            
        # Update
        h_new = h - lr * grad
        
        # Projection to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
