    start_time = time.time()

    iter_count = 0

            best_c5 = current_c5

        grad = np.zeros(n_points)

            grad[i] = (c5_plus - current_c5) / eps
        
        # Adaptive learning rate

        else:

        # Update

        h_new = np.clip(h_new, 0, 1)

    # Final verification
    final_c5 = compute_c5(best_h)
