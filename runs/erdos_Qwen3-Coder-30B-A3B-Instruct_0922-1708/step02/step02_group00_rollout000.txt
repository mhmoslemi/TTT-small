import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    """Ensure h is in [0,1] and sums to n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_c5(h, dx):
    """Compute the maximum overlap integral"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Create initial guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.random.rand(n_points) * 0.5
        
    h = clip_and_scale(h, n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    # Multi-start procedure
    for run_id in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random initialization around current best or uniform
        if np.random.rand() < 0.7:
            # Start near best solution
            h_start = best_h.copy() + np.random.normal(0, 0.02, n_points)
        else:
            # Random initialization
            h_start = np.random.rand(n_points) * 0.5
            
        h = clip_and_scale(h_start, n_points)
        
        # Simple gradient ascent with adaptive learning rate for 200 iterations
        for iter_count in range(200):
            if time.time() - start_time > budget_s - 1:
                break
                
            c5_current = compute_c5(h, dx)
            
            # Evaluate gradient via finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = compute_c5(h_plus, dx)
                grad[i] = (c5_plus - c5_current) / eps
            
            # Update step size adaptively
            lr = 0.001 / (1 + iter_count / 50)
            h_new = h - lr * grad
            
            h = clip_and_scale(h_new, n_points)
            
            # Update best found
            c5_new = compute_c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
