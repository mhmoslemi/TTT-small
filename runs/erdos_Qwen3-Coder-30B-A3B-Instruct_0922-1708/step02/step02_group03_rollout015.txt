import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = c2 = 0.9
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h.copy() + np.random.normal(0, 0.01, n_points)
        else:
            particle = np.random.rand(n_points) * 0.5 + 0.25
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_vals.append(float('inf'))
    
    gbest = h.copy()
    gbest_val = float('inf')
    
    # PSO main loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            particle = particles[i]
            
            # Compute overlap integral
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_vals[i]:
                pbest_vals[i] = c5
                pbest[i] = particle.copy()
                
            if c5 < gbest_val:
                gbest_val = c5
                gbest = particle.copy()
                
        # Update global best
        if gbest_val < best_c5:
            best_c5 = gbest_val
            best_h = gbest.copy()
            
        # Update particles
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocity = (w * velocity + 
                       c1 * r1 * (pbest[i] - particle) + 
                       c2 * r2 * (gbest - particle))
            
            # Clip velocity
            velocity = np.clip(velocity, -v_max, v_max)
            
            # Update position
            particle = particle + velocity
            
            # Boundary conditions
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            particles[i] = particle
            velocities[i] = velocity
        
        # Occasionally do KKT refinement
        if iteration % 20 == 0:
            # KKT refinement on gbest
            refined = gbest.copy()
            for _ in range(20):
                # Compute the overlap integral for this particle
                correlation = np.correlate(refined, 1 - refined, mode='full')
                c5_val = np.max(correlation) * dx
                
                # Approximate the gradient
                eps = 1e-5
                grad = np.zeros_like(refined)
                for j in range(n_points):
                    h_plus = refined.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5_val) / eps
                
                # Update using gradient descent
                step_size = 0.05
                refined = refined - step_size * grad
                refined = np.clip(refined, 0, 1)
                refined = refined * (n_points / 2) / np.sum(refined)
                
                # Early termination if improvement is minimal
                if np.linalg.norm(grad) < 1e-6:
                    break
            
            # Check if refined solution is better
            correlation = np.correlate(refined, 1 - refined, mode='full')
            refined_c5 = np.max(correlation) * dx
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined.copy()
        
        # Early termination check
        if iteration % 100 == 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
