import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided construction or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = clip_and_scale(h, n_points)
    else:
        h = np.ones(n_points) * 0.5
        h = clip_and_scale(h, n_points)
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Multi-start with different variations
    starts = []
    if 'initial_h_values' in globals():
        starts.append(h.copy())
        # Add variations
        for _ in range(4):
            variation = h + np.random.normal(0, 0.02, n_points)
            starts.append(clip_and_scale(variation, n_points))
    else:
        # Random sparse starts
        for _ in range(5):
            h_start = np.random.rand(n_points) * 0.5
            h_start = clip_and_scale(h_start, n_points)
            starts.append(h_start)
            
    # Optimization with adaptive steps
    for start_idx, h in enumerate(starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adam-like optimization
        m = np.zeros(n_points)
        v = np.zeros(n_points)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        alpha = 0.001
        
        for iter_idx in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Gradient approximation via finite differences
            eps_val = 1e-5
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps_val
                h_plus = clip_and_scale(h_plus, n_points)
                c5_plus = compute_c5(h_plus, dx)
                grad[i] = (c5_plus - c5) / eps_val
                
            # Adam updates
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**(iter_idx + 1))
            v_hat = v / (1 - beta2**(iter_idx + 1))
            h = h - alpha * m_hat / (np.sqrt(v_hat) + eps)
            
            h = clip_and_scale(h, n_points)
            
    return (best_h.tolist(), best_c5, n_points)
