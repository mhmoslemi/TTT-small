import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h_vals, dx):
    """Evaluate the overlap integral for given h values"""
    h = np.array(h_vals)
    correlation = correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def initialize_population(n_points, n_pop, initial_h):
    """Initialize population with diverse individuals"""
    population = []
    
    # Add the initial solution if available
    if initial_h is not None:
        h = np.array(initial_h)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill remaining population with random samples
    while len(population) < n_pop:
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    return population

def selection(population, fitness_scores, tournament_size=2):
    """Tournament selection"""
    selected = []
    n_pop = len(population)
    
    for _ in range(n_pop):
        # Tournament
        tournament_indices = np.random.choice(n_pop, tournament_size)
        tournament_fitness = [fitness_scores[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx].copy())
    
    return selected

def crossover(parent1, parent2, crossover_rate=0.8):
    """Single-point crossover"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
        
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(individual, mutation_rate=0.2, max_mutation=0.1):
    """Mutation with constraint repair"""
    mutated = individual.copy()
    
    for i in range(len(mutated)):
        if np.random.random() < mutation_rate:
            # Add noise
            mutated[i] += np.random.normal(0, max_mutation)
    
    # Repair constraints
    mutated = np.clip(mutated, 0, 1)
    scale = (len(mutated) / 2) / np.sum(mutated)
    mutated = mutated * scale
    
    return mutated

    n_points = 100
    dx = 2.0 / n_points
    n_pop = 50
    max_generations = 1000
    mutation_rate = 0.2

    # Track best

    best_c5 = float('inf')
    stagnation_count = 0
    last_improvement = 0
    
    start_time = time.time()
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness_scores.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = generation
                
        # Early termination
        if best_c5 < 0.3807:
            break
            
        # Selection
        selected = selection(population, fitness_scores)
        
        # Create new population through crossover and mutation
        new_population = []
        
        # Elitism: carry forward top individuals
        sorted_indices = np.argsort(fitness_scores)
        for i in range(n_pop // 2):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate offspring
        while len(new_population) < n_pop:
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
            
        # Trim to exactly n_pop
        population = new_population[:n_pop]
        
        # Stagnation handling
        if generation - last_improvement > 50:
            # Disturb population
            for i in range(n_pop // 5):  # Reset 20% of population
                idx = np.random.randint(0, n_pop)
                population[idx] = np.random.random(n_points)
                population[idx] = population[idx] * (n_points / 2) / np.sum(population[idx])
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to some random solution
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        final_c5 = evaluate_h(h, dx)
        return (h.tolist(), final_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s, **kwargs)
