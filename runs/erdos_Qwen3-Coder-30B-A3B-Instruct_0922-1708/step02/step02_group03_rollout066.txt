import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    learning_rate = 0.01
    eps = 1e-6
    max_iter = 10000
    
    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = c2 = 0.5
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Initialize particles
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            if len(particle) != n_points:
                x_orig = np.linspace(0, 2, len(particle))
                x_new = np.linspace(0, 2, n_points)
                particle = np.interp(x_new, x_orig, particle)
            particle = particle / np.sum(particle) * (n_points / 2)
        else:
            particle = np.random.rand(n_points) * 0.5
            particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        # Compute initial value
        correlation = np.correlate(particle, 1 - particle, mode='full')
        pbest_vals.append(np.max(correlation) * dx)
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (pbest[i] - particles[i])
                           + c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Clip and rescale
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate particle
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            particle_val = np.max(correlation) * dx
            
            # Update personal best
            if particle_val < pbest_vals[i]:
                pbest_vals[i] = particle_val
                pbest[i] = particles[i].copy()
                
                # Update global best
                if particle_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = particle_val
        
        # Take a few gradient steps using L-BFGS on best solution
        if iteration % 10 == 0:
            # Convert best to a form suitable for scipy
            h = gbest.copy()
            
            def obj(h_vec):
                h = h_vec.reshape(-1)
                correlation = np.correlate(h, 1 - h, mode='full')
                return np.max(correlation) * dx
            
            def grad(h_vec):
                h = h_vec.reshape(-1)
                n = len(h)
                # Compute gradient numerically
                grad_vals = np.zeros(n)
                eps = 1e-6
                for i in range(n):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    h_minus = h.copy()
                    h_minus[i] -= eps
                    h_minus = np.clip(h_minus, 0, 1)
                    h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                    corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                    c5_minus = np.max(corr_minus) * dx
                    
                    grad_vals[i] = (c5_plus - c5_minus) / (2 * eps)
                return grad_vals.ravel()
            
            # Use L-BFGS to refine the solution
            try:
                result = minimize(obj, h, method='L-BFGS-B', bounds=[(0,1)]*n_points, 
                                jac=grad, options={'maxiter': 50, 'disp': False})
                if result.success:
                    h = result.x.reshape(-1)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    # Update gbest with refined solution
                    correlation = np.correlate(h, 1 - h, mode='full')
                    new_c5 = np.max(correlation) * dx
                    if new_c5 < gbest_val:
                        gbest_val = new_c5
                        gbest = h.copy()
            except:
                pass
        
        # Start optimizing from the best solution
        if gbest_val < best_c5:
            best_c5 = gbest_val
            best_h = gbest.copy()
            
        # Occasionally check for early termination
        if iteration % 100 == 0:
            if time.time() - start_time > budget_s - 5:
                break
    
    return (best_h.tolist(), best_c5, n_points)
