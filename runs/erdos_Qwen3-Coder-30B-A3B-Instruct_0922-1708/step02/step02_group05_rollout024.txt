import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    swarm_size = 50
    inertia_weight = 0.7
    cognitive_param = 2.0
    social_param = 2.0
    
    # Initialize swarm
    particles = []
    velocities = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            if len(particle) != n_points:
                particle = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(particle)), particle)
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
        else:
            particle = np.random.rand(n_points)
            particle = particle * (n_points / 2) / np.sum(particle)
            
        particles.append(particle)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    personal_best = [p.copy() for p in particles]
    personal_best_c5 = [float('inf')] * swarm_size
    gbest = particles[0].copy()
    gbest_c5 = float('inf')
    
    stagnation_counter = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update each particle
        for i in range(swarm_size):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = cognitive_param * r1 * (personal_best[i] - particle)
            social = social_param * r2 * (gbest - particle)
            velocity = inertia_weight * velocity + cognitive + social
            
            # Update position
            particle = particle + velocity
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Store updated particle
            particles[i] = particle
            velocities[i] = velocity
            
            # Evaluate particle
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best[i] = particle.copy()
                personal_best_c5[i] = c5
                
            # Update global best
            if c5 < gbest_c5:
                gbest = particle.copy()
                gbest_c5 = c5
                stagnation_counter = 0
            else:
                stagnation_counter += 1
                
        # Stagnation handling
        if stagnation_counter >= max_stagnation:
            for i in range(swarm_size):
                if np.random.rand() < 0.1:  # Perturb 10% of swarm
                    particles[i] = np.clip(particles[i] + np.random.normal(0, 0.01, n_points), 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            stagnation_counter = 0
            
        # Adapt inertia weight
        inertia_weight = 0.7 - (0.3 * iteration / 500)
        if inertia_weight < 0.4:
            inertia_weight = 0.4
    
    # Final evaluation with best solution
    correlation = np.correlate(gbest, 1 - gbest, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (gbest.tolist(), final_c5, n_points)
