from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Define the objective function to minimize
    def objective(h_vals):
        # Ensure h_vals is within bounds
        h_vals = np.clip(h_vals, 0, 1)
        # Normalize to integral = 1
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # Define constraints for the optimization
    def constraint_integral(h_vals):
        return np.sum(h_vals) * dx - 1.0

    # Create bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using scipy minimize
    while time.time() - start_time < budget_s - 1:
        iteration += 1

        c5 = objective(best_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()

        # Local optimization using minimize
        try:
