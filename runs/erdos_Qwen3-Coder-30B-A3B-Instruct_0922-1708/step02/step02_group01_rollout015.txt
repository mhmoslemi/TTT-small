import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000

        h = np.ones(n_points) * 0.5

    best_c5 = float('inf')
    
    # Helper to compute overlap integral
    def compute_overlap(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Helper to compute constraint for a shift k
    def compute_constraint(h_vec, k):
        constraint_coeffs = np.zeros(n_points)
        for i in range(n_points):
            constraint_coeffs[i] = h_vec[i] * (1 - h_vec[(i + k) % n_points])
        return constraint_coeffs
    
    # Project h to feasible region
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        scale = (n_points / 2) / np.sum(h_vec)
        h_vec = h_vec * scale
        return h_vec
    
    # Initialize iterations
    iteration = 0
    Δ = 0.01  # decrement for constraint
    no_improvement_count = 0
    min Δ = 1e-4  # minimum value for Δ
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Compute current overlap
        c5 = compute_overlap(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
            Δ = 0.01  # Reset Δ on improvement
        else:
            no_improvement_count += 1
            if no_improvement_count % 50 == 0:
                Δ = max(Δ * 0.5, minΔ)
        
        # Find the shift with maximum overlap
        max_k = -1
        max_overlap = -float('inf')

            for i in range(n_points):
                sum_overlap += h[i] * (1 - h[(i + k) % n_points])
            if sum_overlap > max_overlap:
                max_overlap = sum_overlap
                max_k = k
        
        if max_k == -1:
            break
    
        # Define a linear constraint based on max_k shift
        # We want: sum(h[i] * (1 - h[i + max_k])) * dx <= C₅_prev - Δ
        # Rearranging: sum(h[i] * (1 - h[i + max_k])) <= (C₅_prev - Δ) / dx

        constraints_rhs = np.zeros(n_points)
        
        # Build linear constraint matrix: coefficients of h
        # Each row is constraint relating h[i] with shifted h[i + max_k]
        for i in range(n_points):
            constraints_matrix[i, i] = 1.0
            constraints_matrix[i, (i + max_k) % n_points] = -h[i]
        constraints_rhs = np.ones(n_points) * (c5 - Δ) / dx
        
        # Try projection-based heuristic to reduce overlap for max_k
        # Create objective to minimize the maximum overlap constraint for this shift
        # Use a simple variation direction
        
        # Instead of full optimization (which is overkill), use gradient ascent
        # Compute gradient for change in overlap in direction of h with shift k
        
        # Simple gradient descent in the space of h
        # We might want to decrease the overlap for shift k by simply adjusting h
        
        # Gradient-based update
        h_new = h.copy()
        
        # Adjust h to reduce the specific shift's contribution to overlap
        # For each i, adjust the h[i] to reduce term h[i] * (1 - h[i + k])

        for i in range(n_points):
            # Approximate gradient w.r.t h[i] for this shift
            # A_i = h[i] * (1 - h[i + k])

            grad_i = (1 - h[i + max_k % n_points])
            h_new[i] = h_new[i] - η * grad_i
            
        h_new = project_h(h_new)
        
        h = h_new
         
        # For convergence check
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
