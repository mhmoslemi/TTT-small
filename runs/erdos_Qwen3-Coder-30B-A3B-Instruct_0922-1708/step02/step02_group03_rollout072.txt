import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    lr_initial = 0.005
    lr_decay = 0.9995
    lr = lr_initial
    eps = 1e-6
    
    # Number of restarts
    n_restarts = 5
    
    for restart in range(n_restarts):
        h_current = h.copy()
        last_improvement = 0
        patience = 500
        lr_current = lr
        
        while time.time() - start_time < budget_s - 1 and iteration < 10000:
            iteration += 1
            
            # Compute the overlap integral
            correlation = np.correlate(h_current, 1 - h_current, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_current.copy()
                last_improvement = iteration
                
            # Compute gradient for the maximizer lag
            max_lag_idx = np.argmax(correlation)
            if max_lag_idx >= len(correlation) // 2:
                lag = max_lag_idx - len(correlation) // 2
            else:
                lag = max_lag_idx - len(correlation) // 2
                
            # Compute gradient
            grad = np.zeros_like(h_current)
            for i in range(n_points):
                # Forward difference
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update h
            h_new = h_current - lr_current * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_current = h_new
            
            # Adaptive learning rate
            lr_current *= lr_decay
            
            # Early stopping if no improvement
            if iteration - last_improvement > patience:
                break
                
        # Random restart
        if restart < n_restarts - 1:
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
