import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a simple starting point with spikes
        h = np.zeros(n_points)
        h[20:30] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.001
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < 1000000:
        iteration += 1
        
        # Compute overlap integral
        correlation = correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the shift maximizing overlap
        k_max = np.argmax(correlation)
        corr_max = correlation[k_max]
        k_offset = k_max - (n_points - 1)
        
        # Compute gradient w.r.t. that shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Add contribution from h[i] to overlap at shift k_offset
            j_plus = i + k_offset
            if 0 <= j_plus < n_points:
                grad[i] += (1 - h[j_plus])
                
            # Add contribution from h[j] to overlap at shift k_offset
            j_minus = i - k_offset
            if 0 <= j_minus < n_points:
                grad[i] -= h[j_minus]
                
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Perturb occasionally to escape local optima
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
