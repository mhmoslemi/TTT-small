import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    learning_rate = 0.005
    eps = 1e-6
    
    # PSO settings
    n_particles = 50
    w = 0.7
    c1 = c2 = 0.9
    velocities = np.zeros((n_particles, n_points))
    particles = np.random.rand(n_particles, n_points) * 0.5
    particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
    personal_best = particles.copy()
    personal_best_fitness = np.full(n_particles, float('inf'))
    global_best = None
    global_best_fitness = float('inf')
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update global best from PSO
        for i in range(n_particles):
            # Evaluate current particle
            particle_h = particles[i]
            correlation = np.correlate(particle_h, 1 - particle_h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < personal_best_fitness[i]:
                personal_best_fitness[i] = c5
                personal_best[i] = particle_h.copy()
            
            if c5 < global_best_fitness:
                global_best_fitness = c5
                global_best = particle_h.copy()
        
        # Update best solution found so far
        if global_best_fitness < best_c5:
            best_c5 = global_best_fitness
            best_h = global_best.copy()
            
        # PSO update
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            particles[i] += velocities[i]
            
            # Clip and rescale
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        
        # Occasionally apply local refinement with gradient
        if iteration % 20 == 0:
            # Local gradient-based refinement on global best
            h = global_best.copy()
            for _ in range(10):
                # Compute gradient numerically
                grad = np.zeros_like(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[i] = (c5_plus - global_best_fitness) / eps
                
                h_new = h - learning_rate * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
                # Recompute fitness
                correlation = np.correlate(h, 1 - h, mode='full')
                new_c5 = np.max(correlation) * dx
                if new_c5 < global_best_fitness:
                    global_best = h.copy()
                    global_best_fitness = new_c5

    return (best_h.tolist(), best_c5, n_points)
