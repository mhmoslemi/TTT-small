import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 value for a given h"""
    correlation = correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def initialize_population(n_points, n_pop, initial_h=None):
    """Initialize population with or without initial solution"""
    population = []
    
    if initial_h is not None:
        # Normalize initial solution to sum to n_points/2
        scale = (n_points / 2) / np.sum(initial_h)
        h_normalized = np.clip(initial_h * scale, 0, 1)
        population.append(h_normalized)
    
    # Fill remaining population with random samples
    while len(population) < n_pop:
        h = np.random.random(n_points)
        scale = (n_points / 2) / np.sum(h)
        h = np.clip(h * scale, 0, 1)
        population.append(h)
    
    return population

def tournament_selection(population, fitness, tournament_size=2):
    """Select an individual using tournament selection"""
    selected_idx = np.random.randint(0, len(population))
    for _ in range(tournament_size - 1):
        idx = np.random.randint(0, len(population))
        if fitness[idx] < fitness[selected_idx]:
            selected_idx = idx
    return population[selected_idx]

def crossover(parent1, parent2, rate=0.8):
    """Perform single-point crossover"""
    if np.random.random() > rate:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, rate=0.2, noise_range=0.1):
    """Apply uniform mutation with repair"""
    h_new = h.copy()
    
    for i in range(len(h_new)):
        if np.random.random() < rate:
            h_new[i] += np.random.uniform(-noise_range, noise_range)
    
    # Ensure constraints
    h_new = np.clip(h_new, 0, 1)
    scale = (len(h_new) / 2) / np.sum(h_new)
    h_new = np.clip(h_new * scale, 0, 1)
    
    return h_new

def genetic_algorithm(n_points=100, n_pop=50, budget_s=600, initial_h=None):
    """Main genetic algorithm implementation"""
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, n_pop, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
        
        # Early termination
        if best_c5 < 0.3807:
            break
            
        # Selection, crossover, mutation
        new_population = []
        
        # Keep top 50% as elite
        elite_indices = np.argsort(fitness)[:n_pop//2]
        for idx in elite_indices:
            new_population.append(population[idx])
        
        # Generate offspring
        while len(new_population) < n_pop:
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:n_pop]
        
        # Stagnation handling
        stagnation_count += 1
        if stagnation_count > 50:
            # Disturb 20% of population
            n_disturb = max(1, n_pop // 5)
            for _ in range(n_disturb):
                idx = np.random.randint(0, n_pop)
                h = np.random.random(n_points)
                scale = (n_points / 2) / np.sum(h)
                h = np.clip(h * scale, 0, 1)
                population[idx] = h
            stagnation_count = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    initial_h = None

    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(n_points=100, n_pop=50, 
                                       budget_s=budget_s, initial_h=initial_h)
    
    return (best_h.tolist(), best_c5, 100)
