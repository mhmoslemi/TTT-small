                # Interpolate to next grid
                h = np.interp(np.linspace(0, 1, next_grid), 
                             np.linspace(0, 1, current_grid), h)
                current_grid = next_grid

        for _ in range(30):
            if time.time() - start_time > budget_s - 1:

            correlation = np.correlate(h, 1 - h, mode='full')

                    # Indices in h: i+k, indices in (1-h): i

                        grad[i + k_shift] += 1 - h[i]

            step_size = 0.001
