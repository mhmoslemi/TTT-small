import numpy as np

import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Use gradient-free optimization approach
        # Try perturbing in random directions
        h_new = h.copy()
        directions = np.random.normal(0, 0.01, n_points)
        h_new += directions
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Test the new point
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < c5:
            h = h_new
            
        # Occasionally use a more systematic approach
        if iteration % 10 == 0:
            # Find the largest correlation constraint
            correlations = np.correlate(h, 1 - h, mode='full')
            max_corr_idx = np.argmax(correlations)
            max_corr_val = correlations[max_corr_idx]
            
            # Try to improve by adjusting h in a neighborhood
            # Simple local search around where correlation was highest
            idx = max_corr_idx
            if idx > 0 and idx < len(h):
                # Adjust h[idx] to decrease the correlation
                h_test = h.copy()
                h_test[idx] = min(1.0, h_test[idx] + 0.01)
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                corr_test = np.correlate(h_test, 1 - h_test, mode='full')
                c5_test = np.max(corr_test) * dx
                
                if c5_test < c5:
                    h = h_test
                    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
