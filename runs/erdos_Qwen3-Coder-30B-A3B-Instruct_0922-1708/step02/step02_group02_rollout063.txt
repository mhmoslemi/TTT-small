import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        # Create variants
        for _ in range(population_size):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Initialize random population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Evolution parameters
    mutation_rate = 0.2
    crossover_rate = 0.5
    generations = 0
    stagnant = 0
    
    while time.time() - start_time < budget_s - 1:
        generations += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection: Keep top 20%
        idxs = np.argsort(fitness)
        new_population = [population[i] for i in idxs[:population_size//5]]
        
        # Replacement
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < crossover_rate and len(new_population) >= 2:
                i1, i2 = np.random.choice(len(new_population), 2, replace=False)
                parent1, parent2 = new_population[i1], new_population[i2]
                crossover_point = np.random.randint(1, n_points-1)
                child = (parent1[:crossover_point] + parent2[crossover_point:]) / 2.0
                child = np.concatenate([child, parent2[crossover_point:]])
                child = child * (n_points / 2) / np.sum(child)
            else:
                # Mutation
                parent = new_population[np.random.randint(len(new_population))]
                child = parent.copy()
                mutate_idx = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
                for i in mutate_idx:
                    child[i] += np.random.normal(0, 0.02)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Adjust mutation rate if no improvement for a while
        if generations % 100 == 0 and best_c5 < 0.381:
            mutation_rate = min(0.3, mutation_rate + 0.01)
            stagnant += 1
            if stagnant > 5:
                break  # Stop if no improvement for 500 generations
    
    # Final evaluation of best solution
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        return (best_h.tolist(), final_c5, n_points)
    else:
        # If no good solution yet found, use simple approach
        h = np.ones(n_points) * 0.5
        corr = correlate(h, 1 - h, mode='full')
        final_c5 = np.max(corr) * dx
        return (h.tolist(), final_c5, n_points)
