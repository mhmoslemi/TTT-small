import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    learning_rate = 0.005
    eps = 1e-6
    max_iterations = int(budget_s * 10)
    
    while iteration < max_iterations and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update h using gradient descent with momentum
        h_new = h - learning_rate * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Occasionally do a more thorough refinement step
        if iteration % 50 == 0:
            # Try a local optimization with scipy
            def objective(h_vec):
                h_clipped = np.clip(h_vec, 0, 1)
                h_scaled = h_clipped * (n_points / 2) / np.sum(h_clipped)
                correlation = np.correlate(h_scaled, 1 - h_scaled, mode='full')
                return -np.max(correlation) * dx  # Negative because we minimize

            # Use scipy for more accurate step
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 10})
            
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                h = new_h
                
        # Occasionally check for early termination
        if iteration % 100 == 0 and iteration > 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
