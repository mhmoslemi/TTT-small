import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute the lags for correlation
    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1:
        # Compute correlation and find max
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(correlation)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try to improve using a QP approach
        # Variables
        h_vars = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1]
        constraints.append(sum(h_vars) == n_points / 2)
        constraints.append(t >= 0)
        
        # For each lag k, add constraint: sum h[i] * (1 - h[i+k]) * dx <= t
        for k in lags:
            # We need to compute this for the appropriate indices where both h[i], h[i+k] exist
            if k < n_points:
                expr = 0
                for i in range(n_points - k):
                    expr += h_vars[i] * (1 - h_vars[i + k]) * dx
                constraints.append(expr <= t)
            else:
                # Beyond max lag, these correlations would only be zero in this simplification
                pass
                
        # Objective
        objective = Minimize(t)
        
        # Create and solve problem
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
            new_h = h_vars.value
            if new_h is not None:
                h = np.clip(new_h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)  # Normalize again
        except:
            # If optimization fails, fall back to gradient approach for small changes
            # Gradient step using finite differences
            eps = 1e-5
            grad = np.zeros_like(h)
            corri = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corri) * dx
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
                
            lr = 0.001
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5 = np.max(correlation)
    
    return (best_h.tolist(), c5, n_points)
