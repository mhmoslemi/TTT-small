import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use provided initial solution or create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 50
        h = np.random.rand(n_points)
    
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Population-based GA approach
    pop_size = min(50, max(10, n_points // 2))
    population = []
    fitnesses = []
    
    # Initialize population
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            individual = h.copy()
        else:
            individual = np.random.rand(n_points)
        individual = individual * (n_points / 2) / np.sum(individual)
        individual = np.clip(individual, 0, 1)
        population.append(individual)
    
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    gen = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        gen += 1
        # Evaluate fitness
        fitnesses = [compute_c5(ind) for ind in population]
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[np.argmin(fitnesses)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Selection (tournament selection)
        selected = []
        tournament_size = 3
        
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            winner = tournament[np.argmin([fitnesses[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover and Mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[min(i+1, pop_size-1)]
            
            # Crossover: uniform
            mask = np.random.rand(n_points) > 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
            
            # Mutation: Gaussian perturbation
            mutation_rate = 0.1
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.05)
            
            # Ensure feasibility
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
    
    # Final cleanup and validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
