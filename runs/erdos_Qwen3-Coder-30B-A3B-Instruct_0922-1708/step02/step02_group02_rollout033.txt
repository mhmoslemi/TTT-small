import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create initial binary comb-like structure
        h = np.zeros(n_points)
        m = 5  # Number of pulses
        w = 0.2  # Width of each pulse
        s = 0.25  # Spacing between pulses
        total_width = m * w + (m - 1) * s
        scale = 1.0 / total_width
        start_positions = np.linspace(0, 2 - total_width, m)
        
        for i in range(m):
            start_idx = int(start_positions[i] / dx)
            end_idx = int((start_positions[i] + w) / dx)
            if end_idx > start_idx:
                h[start_idx:end_idx] = 1.0
        
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h_vals):
        # Ensure h_vals is properly scaled
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    def objective(h_vals):
        return compute_c5(h_vals)
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run optimization
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use differential evolution to explore the space
    de_result = differential_evolution(
        objective, 
        bounds, 
        maxiter=100,
        popsize=15,
        seed=seed,
        disp=False,
        callback=lambda x, convergence: None
    )
    
    if not de_result.success:
        # Fall back to local optimization if DE fails
        try:
            from scipy.optimize import minimize
            res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                          options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                updated_h = res.x
                updated_c5 = compute_c5(updated_h)
                if updated_c5 < best_c5:
                    best_c5 = updated_c5
                    best_h = updated_h
        except:
            pass
    else:
        updated_h = de_result.x
        updated_c5 = compute_c5(updated_h)
        if updated_c5 < best_c5:
            best_c5 = updated_c5
            best_h = updated_h
    
    # Fine-tune with gradient-based method for final refinement
    if time.time() - start_time < budget_s - 2:
        try:
            from scipy.optimize import minimize
            def refine_objective(h_vals):
                return compute_c5(h_vals)
                
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(refine_objective, best_h, method='L-BFGS-B', 
                          bounds=bounds, options={'maxiter': 300})
            if res.success:
                final_h = res.x
                final_c5 = compute_c5(final_h)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = final_h
        except:
            pass
    
    # Final check
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), float(final_c5), n_points)
