import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use a more sophisticated optimizer
    def objective(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B for better convergence properties
    method = 'L-BFGS-B'
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Use scipy.optimize.minimize for better convergence
        result = minimize(
            objective, 
            h, 
            method=method,
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-6}
        )
        
        if result.success:
            h = result.x
            # Ensure the sum constraint is met
            h = h * (n_points / 2) / np.sum(h)
            # Keep values in bounds
            h = np.clip(h, 0, 1)
        else:
            # Fallback to gradient descent if optimization fails
            eps = 1e-6
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update with gradient
            learning_rate = 0.001
            h_new = h - learning_rate * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
        # Occasionally check for early termination
        if iteration % 100 == 0 and iteration > 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
