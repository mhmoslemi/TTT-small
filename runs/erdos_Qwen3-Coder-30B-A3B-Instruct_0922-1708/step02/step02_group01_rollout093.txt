import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    delta = 0.01
    
    # Heuristic: use a more sophisticated optimization approach
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5 value
        correlations = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlations) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
            delta = max(1e-4, delta * 0.95)
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                delta *= 0.5
                no_improvement_count = 0
            
        # Find worst shift and add its constraint
        k_max = np.argmax(correlations)
        
        # Try adaptive optimization with multiple starting points
        best_step = None
        best_c5_local = c5
        
        # Generate multiple random starts
        for _ in range(5):
            # Start from current solution with small perturbation
            h_perturbed = h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            # Local optimization using L-BFGS
            try:
                result = minimize(
                    lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
                    h_perturbed,
                    method='L-BFGS-B',
                    bounds=[(0, 1)] * n_points,
                    options={'maxiter': 20}
                )
                if result.success:
                    optimized = result.x
                    optimized = np.clip(optimized, 0, 1)
                    optimized = optimized * (n_points / 2) / np.sum(optimized)
                    local_c5 = np.max(np.correlate(optimized, 1 - optimized, mode='full')) * dx
                    
                    if local_c5 < best_c5_local:
                        best_c5_local = local_c5
                        best_step = optimized
            except:
                continue
                
        if best_step is not None:
            h = best_step
        else:
            # If no improvement, try gradient descent with adaptive learning rate
            eps = 1e-4
            grad = np.zeros_like(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Adaptive step size
            lr = 0.01 / (1 + iteration / 500)
            
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
        # Occasionally restart with best points
        if iteration % 100 == 0:
            # Restart with best solution so far
            h = best_h.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            delta = 0.01  
            
        if best_c5 < 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
