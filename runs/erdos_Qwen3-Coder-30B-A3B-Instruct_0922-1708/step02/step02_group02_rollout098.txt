import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try to perturb with gradient descent, but also use QP approach 
        # Build a reduced problem for better performance
        
        # Use a simpler gradient based approach for exploratory steps
        # But also do QP for local refinement
        
        # Use QP approach on top k lags
        k_lags = min(50, len(lags))
        recent_lags = lags[:k_lags]
        
        # Use CVXPY to solve a QP with active constraints
        h_vars = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints: correlation <= t for top lags
        constraints = [0 <= h_vars, h_vars <= 1, cp.sum(h_vars) == n_points / 2, t >= 0]
        
        # Add some key correlation constraints
        for k in recent_lags:
            # Compute the sum over valid indices
            corr_sum = 0
            for i in range(n_points - k):
                corr_sum += h_vars[i] * (1 - h_vars[i + k]) * dx
            constraints.append(corr_sum <= t)
            
        # Objective 
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_qp = h_vars.value
                h_new = np.clip(h_qp, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new.copy()
            else:
                # fallback to gradient approach if QP fails
                grad = np.zeros_like(h)
                eps = 1e-5

                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[i] = (c5_plus - c5) / eps
                    
                # Update h using gradient descent
                lr = 0.002
                h_new = h - lr * grad
                
                # Clip and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
        except:
            # Fallback to gradient descent
            grad = np.zeros_like(h)
            eps = 1e-5

            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
                
            # Update h using gradient descent
            lr = 0.002
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
