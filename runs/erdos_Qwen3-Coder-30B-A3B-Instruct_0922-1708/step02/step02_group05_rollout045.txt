import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = c2 = 2.0
    
    # Initialize particles and velocities
    particles = np.random.rand(swarm_size, n_points) * 0.5
    velocities = np.random.rand(swarm_size, n_points) * 0.5 - 0.25
    pbest = particles.copy()
    pbest_scores = np.full(swarm_size, float('inf'))
    gbest_idx = 0
    gbest_score = float('inf')
    
    # Ensure particles satisfy constraint
    for i in range(swarm_size):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        # Compute initial scores
        correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
        score = np.max(correlation) * dx
        pbest_scores[i] = score
        if score < gbest_score:
            gbest_score = score
            gbest_idx = i
            
    # PSO main loop
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (particles[gbest_idx] - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to satisfy integral constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            score = np.max(correlation) * dx
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                
            # Update global best
            if score < gbest_score:
                gbest_score = score
                gbest_idx = i
                
        # Check for stagnation and add diversity
        if iteration % 100 == 0:
            # Randomly perturb some particles to avoid local minima
            perturb_indices = np.random.choice(swarm_size, size=swarm_size // 10, replace=False)
            for idx in perturb_indices:
                particles[idx] = np.random.rand(n_points) * 0.5
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
    
    # Final evaluation of the best solution
    final_h = particles[gbest_idx].copy()
    final_correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (final_h.tolist(), final_c5, n_points)
