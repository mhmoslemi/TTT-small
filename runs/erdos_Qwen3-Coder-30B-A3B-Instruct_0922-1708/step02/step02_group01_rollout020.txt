import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    correlation = signal.convolve(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def initialize_population(n_points, N_pop, initial_h_values=None):
    """Initialize population for GA"""
    population = []
    
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    # Fill remaining population with random samples
    remaining = N_pop - len(population)
    for _ in range(remaining):
        h = np.random.random(n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, tournament_size=2):
    """Select parent using tournament selection"""
    selected_idx = np.random.choice(len(population), tournament_size)
    winner_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[winner_idx]

def crossover(parent1, parent2, crossover_rate):
    """Perform single-point crossover"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    # Single point crossover
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(individual, mutation_rate, dx):
    """Mutate individual with uniform noise and rescale"""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.random() < mutation_rate:
            mutated[i] += np.random.uniform(-0.1, 0.1)
    
    # Clip and rescale to maintain constraints
    mutated = np.clip(mutated, 0, 1)
    scale = (len(mutated) / 2) / np.sum(mutated)
    mutated = mutated * scale
    
    return mutated

def genetic_algorithm(n_points, N_pop=50, C_rate=0.8, M_rate=0.2, budget_s=600):
    """Main genetic algorithm implementation"""
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, N_pop)
    
    # Evaluate initial fitness
    fitnesses = [evaluate_h(ind, dx) for ind in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    generation = 0
    stagnation_counter = 0
    best_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
            best_improvement = 0
        else:
            best_improvement += 1
            
        # Early termination if target reached
        if best_c5 < 0.3807:
            break
            
        # Create new population through selection, crossover, and mutation
        new_population = [population[0]]  # Elitism: keep best
            
        # Selection and reproduction
        while len(new_population) < N_pop:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2, C_rate)
            
            # Mutation
            child1 = mutate(child1, M_rate, dx)
            child2 = mutate(child2, M_rate, dx)
            
            new_population.extend([child1, child2])
            
        # Keep only population size
        population = new_population[:N_pop]
        
        # Recalculate fitness
        fitnesses = [evaluate_h(ind, dx) for ind in population]
        
        # Disturbance if no improvement for 50 generations
        if best_improvement > 50:
            for i in range(int(0.2 * N_pop)):
                if len(population) > i:
                    population[i] = np.random.random(n_points)
                    population[i] = population[i] / np.sum(population[i]) * (n_points / 2)
            best_improvement = 0
            fitnesses = [evaluate_h(ind, dx) for ind in population]
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(n_points, budget_s=budget_s)
    
    # Final validation
    dx = 2.0 / n_points
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
