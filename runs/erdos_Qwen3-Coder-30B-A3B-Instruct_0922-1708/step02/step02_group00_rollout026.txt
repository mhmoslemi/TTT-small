import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def clip_and_scale(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Generate initial candidates
    candidates = []
    
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        if len(h_parent) != n_points:
            x_orig = np.linspace(0, 2, len(h_parent))
            x_new = np.linspace(0, 2, n_points)
            h_parent = np.interp(x_new, x_orig, h_parent)
        candidates.append(h_parent.copy())
        # Add noisy variations
        for _ in range(4):
            noise = np.random.normal(0, 0.05, n_points)
            h_noisy = h_parent + noise
            h_noisy = clip_and_scale(h_noisy, n_points)
            candidates.append(h_noisy)
    else:
        # Random initialization
        for _ in range(5):
            h = np.random.rand(n_points) * 0.5
            h = clip_and_scale(h, n_points)
            candidates.append(h)
    
    # Helper to compute gradient using finite differences
    def compute_grad(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_c5 = evaluate_c5(h, dx)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = clip_and_scale(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.001
    
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Start optimization from each candidate
    for i, h in enumerate(candidates):
        if time.time() - start_time > budget_s - 1:
            break
        h = h.copy()
        t = 0
        for it in range(500):
            if time.time() - start_time > budget_s - 1:
                break
            t += 1
            grad = compute_grad(h)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h -= lr * m_hat / (np.sqrt(v_hat) + eps)
            h = clip_and_scale(h, n_points)
            
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    if best_h is None:
        # fallback to initial approach
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_orig, h)
        else:
            h = np.ones(n_points) * 0.5
        best_h = clip_and_scale(h, n_points)
        best_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
