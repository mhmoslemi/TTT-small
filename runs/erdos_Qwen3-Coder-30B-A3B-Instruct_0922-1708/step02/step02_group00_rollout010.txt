import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T0 = 0.1
    tau = 1000
    current_temp = T0
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Generate candidate solution
        # Add noise scaled by temperature
        noise = np.random.normal(0, current_temp * 0.1, n_points)
        h_candidate = h + noise
        
        # Clip and rescale
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute candidate overlap
        correlation_c = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_c = np.max(correlation_c) * dx
        
        # Acceptance criterion
        delta = c5_c - c5
        if delta < 0 or np.random.rand() < np.exp(-delta / current_temp):
            h = h_candidate.copy()
        
        # Update temperature
        current_temp = T0 / (1 + iteration / tau)
        
        # Occasionally reset temperature if stuck
        if iteration % 1000 == 0 and current_temp < 1e-6:
            current_temp = T0
    
    return (best_h.tolist(), best_c5, n_points)
