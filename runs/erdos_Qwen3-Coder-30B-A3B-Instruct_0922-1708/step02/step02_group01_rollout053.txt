import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    n_particles = 50
    max_iter = int(budget_s * 1000)  # Rough estimate for iterations
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        particles.append(np.ones(n_points) * 0.5)
        velocities.append(np.zeros(n_points))
        
    # Initialize remaining particles
    for i in range(n_particles - len(particles)):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
        velocities.append(np.random.randn(n_points) * 0.01)
        
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Track best solutions
    personal_best = particles.copy()
    personal_best_values = np.full(n_particles, float('inf'))
    global_best_idx = 0
    global_best_value = float('inf')
    
    start_time = time.time()
    iteration = 0
    stagnation_counter = 0
    stagnation_threshold = 100
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Evaluate each particle
        for i in range(n_particles):
            h = particles[i]
            
            # Project if needed (clip and rescale)
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            particles[i] = h
            
            # Evaluate objective
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_values[i]:
                personal_best_values[i] = c5
                personal_best[i] = h.copy()
                
            # Update global best
            if c5 < global_best_value:
                global_best_value = c5
                global_best_idx = i
                stagnation_counter = 0
        
        # PSO update
        r1, r2 = np.random.rand(2, n_particles, n_points)
        velocities = (w * velocities + 
                      c1 * r1 * (personal_best - particles) + 
                      c2 * r2 * (personal_best[global_best_idx] - particles))
        
        particles += velocities
        
        # Projection after velocity update
        for i in range(n_particles):
            h = particles[i]
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            particles[i] = h
            
        # Restart mechanism if stagnation
        stagnation_counter += 1
        if stagnation_counter > stagnation_threshold:
            # Reset velocities for some particles
            for i in range(n_particles):
                if np.random.rand() < 0.3:  # 30% chance to reset
                    velocities[i] = np.random.randn(n_points) * 0.01
            stagnation_counter = 0
            
        # Early termination if good enough
        if global_best_value < 0.3807:
            break
            
    # Return best solution
    best_h = personal_best[global_best_idx]
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    scale = (n_points / 2) / np.sum(best_h)
    best_h = best_h * scale
    
    # Recompute final value
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
