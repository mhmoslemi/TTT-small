import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with provided parent or uniform distribution
    if 'initial_h_values' in globals():

        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points) * 0.5
        
    # Objective function to minimize
    def obj(h):
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Bounds for each variable

    # Use differential evolution with custom budget
    start_time = time.time()

    best_h = h0.copy()

    # Differential evolution parameters
    population_size = 50
    max_generations = 1000
    mutation_factor = 0.8
    crossover_rate = 0.7
    
    # Track best solution
    best_individual = h0.copy()
    best_fitness = obj(h0)
    
    # Run differential evolution manually with budget control
    for gen in range(max_generations):
