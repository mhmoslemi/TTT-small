import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Track improvement
    last_improvement_iter = 0
    delta = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute all overlaps for current h
        overlaps = []
        for k in range(n_points):
            # Compute overlap for shift k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlaps.append(overlap * dx)
        
        c5 = max(overlaps)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement_iter = iteration
            delta = max(0.001, 0.01 * c5)
        
        # Find worst shift
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Gradient-based update
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = 0
            for k in range(n_points):
                j = (k + i) % n_points
                overlap_plus += h_plus[k] * (1 - h_plus[j])
            overlap_plus *= dx
            
            grad[i] = (overlap_plus - max_overlap) / eps
        
        # Update step
        lr = 5e-3
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Reset delta if improvement was recent
        if iteration - last_improvement_iter > 50:
            delta *= 0.5
            delta = max(1e-4, delta)
        
        # Early exit if target reached
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
