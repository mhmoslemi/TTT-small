import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        h_base = h_base / np.sum(h_base) * (n_points / 2)
        
        # Create initial population with small perturbations
        for _ in range(population_size):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Initialize with random valid h vectors
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    gen = 0
    no_improvement = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        gen += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Track improvement
        if gen == 1 or fitness[0] < best_c5:
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 20% (elitism)
        elite_size = population_size // 5
        new_population = population[:elite_size]
        
        # Create offspring via crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_select(population, fitness, 3)
            parent2 = tournament_select(population, fitness, 3)
            
            # Crossover (uniform)
            if random.random() < 0.5:
                child = (parent1 + parent2) / 2.0
            else:
                child = parent1.copy()
                mask = np.random.rand(n_points) < 0.5
                child[mask] = parent2[mask]
            
            # Mutation
            mutate(child, 0.2 if no_improvement < 50 else 0.3)
            
            # Ensure constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Final evaluation
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to simple gradient descent if evolutionary failed
        return simple_gradient_descent(seed, budget_s, n_points)

def tournament_select(population, fitness, k):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), k, replace=False)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx]

def mutate(h, mutation_rate):
    """Mutate h with given rate"""
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h[i] += random.gauss(0, 0.02)
    
def simple_gradient_descent(seed, budget_s, n_points):
    """Fallback gradient descent approach"""
    np.random.seed(seed)
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update
        learning_rate = 0.005
        h_new = h - learning_rate * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    return (best_h.tolist(), best_c5, n_points)
