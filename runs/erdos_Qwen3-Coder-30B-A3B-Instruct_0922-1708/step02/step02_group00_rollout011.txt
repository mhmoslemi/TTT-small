import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated annealing parameters
    T0 = 0.1
    tau = 1000
    start_time = time.time()
    
    iteration = 0
    T = T0
    
    while time.time() - start_time < budget_s - 1 and T > 1e-6:
        iteration += 1
        
        # Compute current overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Generate candidate solution
        h_prime = h + np.random.normal(0, T, n_points)
        h_prime = np.clip(h_prime, 0, 1)
        h_prime = h_prime * (n_points / 2) / np.sum(h_prime)
        
        # Compute candidate overlap
        correlation_prime = np.correlate(h_prime, 1 - h_prime, mode='full')
        c5_prime = np.max(correlation_prime) * dx
        
        # Accept with probability based on simulated annealing
        delta = c5_prime - c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h = h_prime
            
        # Update temperature
        T = T0 / (1 + iteration / tau)
        
    return (best_h.tolist(), best_c5, n_points)
