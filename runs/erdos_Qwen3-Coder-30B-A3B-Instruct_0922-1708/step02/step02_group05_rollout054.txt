import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define objective function to minimize
    def objective(h):
        # Ensure h is within bounds and satisfies integral constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
        
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the constraint for the integral
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Use differential evolution for global optimization
    # We limit iterations to fit within budget
    maxiter = max(50, int(budget_s / 5))  # Roughly 5 seconds per iteration
    popsize = min(50, max(10, n_points // 2))
    
    try:
        result = differential_evolution(
            objective,
            bounds,
            constraints=constraints,
            maxiter=maxiter,
            popsize=popsize,
            seed=seed,
            disp=False,
            atol=1e-6,
            rtol=1e-6
        )
        
        if not result.success:
            h_final = h.copy()
        else:
            h_final = result.x
            
        # Ensure final solution satisfies constraints
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        
        # Compute final C5
        correlation_final = np.correlate(h_final, 1 - h_final, mode='full')
        final_c5 = np.max(correlation_final) * dx
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final.copy()
            
    except Exception:
        # Fallback to original approach if DE fails
        # This is a simpler, robust fallback
        current_h = h.copy()
        for iteration in range(200):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Simple gradient-like approach
            grad = np.zeros_like(current_h)
            eps = 1e-5
            c5_base = objective(current_h)
            
            for i in range(n_points):
                h_plus = current_h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c5_plus = objective(h_plus)
                grad[i] = (c5_plus - c5_base) / eps
            
            # Update with learning rate
            lr = 0.01
            current_h_new = current_h - lr * grad
            current_h_new = np.clip(current_h_new, 0, 1)
            current_h_new = current_h_new * (n_points / 2) / np.sum(current_h_new)
            
            current_h = current_h_new
            
            # Evaluate and store best
            c5_current = objective(current_h)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = current_h.copy()
    
    # Final evaluation
    correlation_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
