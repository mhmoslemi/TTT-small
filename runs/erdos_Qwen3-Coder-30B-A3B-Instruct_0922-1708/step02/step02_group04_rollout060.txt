import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    max_iter = 1000
    no_improve_count = 0
    max_no_improve = 100
    
    # Initialize grid sizes to explore for progressive refinement
    grid_sizes = [20, 40, 80, 160] 
    grid_idx = 0
    current_n = grid_sizes[0] if grid_sizes else n_points
    
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def project_to_feasible(h_vals):
        # Ensure sum(h) = n/2
        h_vals = h_vals * (current_n / 2) / np.sum(h_vals)
        # Ensure bounds [0,1]
        return np.clip(h_vals, 0, 1)
    
    def update_h(h_vals, step_size=0.01):
        # Compute overlaps for all k
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k] * dx
        
        # Compute gradient estimate (simplified)
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            # Compute contribution to overlap at k = max_k
            k = max_k - (len(h_vals) - 1) # Convert to offset 
            # Approximate gradient contribution
            if i + k >= 0 and i + k < len(h_vals):
                grad[i] = 1 - h_vals[i + k]
            else:
                grad[i] = 0
                
        # Simple gradient step
        h_new = h_vals - step_size * grad
        h_new = project_to_feasible(h_new)
        return h_new
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute C5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > max_no_improve:
            # Try more aggressive exploration
            if grid_idx < len(grid_sizes) - 1:
                grid_idx += 1
                current_n = grid_sizes[grid_idx]
                h = project_to_feasible(np.interp(np.linspace(0, 1, current_n), 
                                                 np.linspace(0, 1, len(h)), h))
                no_improve_count = 0
                continue
                
        # Apply local update
        h_try = update_h(h, 0.01)
        h_try = project_to_feasible(h_try)
        
        new_c5 = compute_c5(h_try)
        if new_c5 < c5:
            h = h_try
        else:
            # Random perturbation + rescaling
            h_perturbed = h + np.random.normal(0, 0.005, len(h))
            h_perturbed = project_to_feasible(h_perturbed)
            if compute_c5(h_perturbed) < c5:
                h = h_perturbed
                
    # Final check and cleaning
    final_c5 = compute_c5(best_h)
    final_h = project_to_feasible(best_h)
    
    return (final_h.tolist(), final_c5, current_n)
