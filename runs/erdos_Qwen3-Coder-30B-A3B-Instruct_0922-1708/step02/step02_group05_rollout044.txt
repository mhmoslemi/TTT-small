import numpy as np
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def repair_feasibility(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init)
        h_init = repair_feasibility(h_init, n_points)
        population.append(h_init)
    
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = repair_feasibility(h, n_points)
        population.append(h)
    
    # Genetic algorithm
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Tournament selection
    def tournament_selection(pop, k=3):
        idxs = np.random.choice(len(pop), k)
        fitnesses = [evaluate_c5(pop[i], dx) for i in idxs]
        return pop[idxs[np.argmin(fitnesses)]]
    
    # Crossover
    def crossover(p1, p2, cr=0.5):
        mask = np.random.rand(len(p1)) < cr
        child = np.where(mask, p1, p2)
        return child
    
    # Mutation
    def mutate(h, mu=0.2, sigma=0.1):
        mask = np.random.rand(len(h)) < mu
        h = h + np.random.normal(0, sigma, len(h)) * mask
        return repair_feasibility(h, n_points)
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate current population
        fitnesses = [evaluate_c5(h, dx) for h in population]
        
        # Track best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
        
        # Selection
        new_population = []
        # Keep elite
        elite_idx = min_idx
        new_population.append(population[elite_idx])
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    if best_h is None:
        # Fallback to initial if no better found
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init)
            best_h = repair_feasibility(h_init, n_points)
            best_c5 = evaluate_c5(best_h, dx)
        else:
            best_h = np.ones(n_points) * 0.5
            best_h = repair_feasibility(best_h, n_points)
            best_c5 = evaluate_c5(best_h, dx)
    
    # Final validation
    best_h = repair_feasibility(best_h, n_points)
    best_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
