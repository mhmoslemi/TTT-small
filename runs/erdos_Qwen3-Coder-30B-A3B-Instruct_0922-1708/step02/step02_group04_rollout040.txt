    start_time = time.time()

        mid = n_points // 2
        h_vals[:mid//2] = 0.7
        h_vals[mid//2:3*mid//4] = 0.3
        h_vals[3*mid//4:] = 0.7
        # Normalize to exact constraint
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        h_vals = np.clip(h_vals, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_vals.copy()
    best_c5 = float('inf')
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_vals.copy()
        
        # Generate new candidate h_vals
        h_new = h_vals.copy()
        
        # Random perturbations
        if np.random.random() < 0.7:
            # Small random changes
            perturb = np.random.normal(0, 0.02, n_points)
            h_new = h_new + perturb
            h_new = np.clip(h_new, 0, 1)

            # Larger jump
            i = np.random.randint(n_points)

    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
