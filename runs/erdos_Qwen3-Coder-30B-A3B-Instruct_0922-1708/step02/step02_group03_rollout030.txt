import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Optimizer state for Adam
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute exact gradient
        max_shift = np.argmax(correlation)
        # The cross-correlation at shift k corresponds to:
        # sum(h[i] * (1 - h[(i + k) % n])) 
        # For each i, we have partial derivative of this term wrt h[i]
        # Let's find gradient explicitly: if shift k gives max,
        # the gradient should look at the differences for each element
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Compute gradient element-wise
            # For each term h[i] * (1 - h[(i + k) % n]), we want to compute
            # d/dh[i] of the maximum cross-correlation
            # assuming the maximum is achieved at max_shift
            
            # When computing correlation at shift max_shift,
            # we consider h[i] * (1 - h[(i + max_shift) % n])
            # The derivative w.r.t h[i] is (1 - h[(i + max_shift) % n])
            # This is only when i corresponds to the max contributing term
            
            # Alternative: computed by finite differences for robustness
            # But here we attempt exact computation:
            # If term in correlation is h[i] * (1 - h[i + max_shift]), 
            # gradient wrt h[i] = (1 - h[i + max_shift])
            # But i is not a fixed position with fixed max_shift contribution
            # Actual computation via finite differences or by more precise logic
            
            # Approximate with finite differences
            eps = 1e-6
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        
        h_new = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Occasionally check for early termination
        if iteration % 100 == 0 and iteration > 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
