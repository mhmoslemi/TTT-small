import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    n_particles = 50
    dx = 2.0 / n_points
    max_iter = int(500000)  # Roughly 600s budget
    
    # Initialize particles
    particles = np.random.rand(n_particles, n_points) * 0.5
    velocities = np.random.randn(n_particles, n_points) * 0.01
    personal_best = particles.copy()
    personal_best_scores = np.full(n_particles, float('inf'))
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles[0] = h_init
        personal_best[0] = h_init
    
    # Global best tracking
    global_best_idx = 0
    global_best_score = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Stagnation tracking
    stagnation_count = 0
    last_improvement = 0
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Ensure feasibility
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            
            # Update particle position
            particles[i] = h
            
            # Compute overlap
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_best[i] = h.copy()
                
                # Update global best
                if c5 < global_best_score:
                    global_best_score = c5
                    global_best_idx = i
                    stagnation_count = 0
                    last_improvement = iter_count
                elif iter_count - last_improvement > 100:
                    stagnation_count += 1
                    if stagnation_count > 100:
                        # Restart velocities
                        velocities[i] = np.random.randn(n_points) * 0.01
                        stagnation_count = 0
            else:
                # Update velocities
                r1, r2 = np.random.rand(2)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (personal_best[i] - particles[i]) +
                                c2 * r2 * (particles[global_best_idx] - particles[i]))
                
                # Update positions
                particles[i] = particles[i] + velocities[i]
                
                # Ensure feasibility after update
                particles[i] = np.clip(particles[i], 0, 1)
                scale = (n_points / 2) / np.sum(particles[i])
                particles[i] = particles[i] * scale
                
        # Periodic velocity reset for stagnation
        if stagnation_count > 100:
            velocities = np.random.randn(n_particles, n_points) * 0.01
            stagnation_count = 0
            
        # Early stopping
        if global_best_score < 0.3807:
            break
    
    # Final validation
    best_h = particles[global_best_idx]
    best_h = np.clip(best_h, 0, 1)
    scale = (n_points / 2) / np.sum(best_h)
    best_h = best_h * scale
    
    # Recompute final C5
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
