import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    n_particles = 50
    dx = 2.0 / n_points
    max_iter = int(1e6)
    
    # Initialize particles
    particles = np.random.rand(n_particles, n_points) * 0.5
    velocities = np.random.randn(n_particles, n_points) * 0.01
    
    # Add initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        particles[0] = h_init / np.sum(h_init) * (n_points / 2)
    
    # Initialize best
    best_c5 = float('inf')
    best_h = None
    global_best_idx = 0
    global_best_c5 = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Synchronize particles with constraints
    def project(h):
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        return h
    
    # Initialize personal bests
    personal_bests = particles.copy()
    personal_best_c5 = np.full(n_particles, float('inf'))
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            # Project to feasible set
            h = project(h)
            particles[i] = h
            
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best_c5[i] = c5
                personal_bests[i] = h.copy()
            
            # Update global best
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_idx = i
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            vel = (w * velocities[i] + 
                   c1 * r1 * (personal_bests[i] - particles[i]) + 
                   c2 * r2 * (particles[global_best_idx] - particles[i]))
            velocities[i] = vel
            particles[i] += vel
            
        # Occasionally restart velocities if stagnation detected
        if iter_count % 100 == 0:
            velocities = np.random.randn(n_particles, n_points) * 0.01
            
    if best_h is None:
        best_h = particles[global_best_idx]
        
    return (best_h.tolist(), best_c5, n_points)
