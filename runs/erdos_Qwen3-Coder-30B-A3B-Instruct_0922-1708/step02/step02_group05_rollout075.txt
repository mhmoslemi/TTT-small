import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, n_points):
    """Evaluate the C5 bound for given h"""
    dx = 2.0 / n_points
    # Compute the correlation using numpy's correlate
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initialize using provided parent or uniform
    if 'initial_h_values' in globals():

        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Normalize the initial guess to meet the integral constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Objective function to minimize C5
    def objective(h):
        # Ensure the integral constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, n_points)
    
    # Add bounds to the constraints
    # We'll create a custom optimization loop since we have constraints
    # differential_evolution approach
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # DE parameters
    popsize = 50
    maxiter = 500  # We'll limit iterations based on time
    
    # Use a simple custom DE with time tracking
    population = []
    fitnesses = []
    
    # Initialize population
    for _ in range(popsize):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        fitness = objective(h)
        fitnesses.append(fitness)
    
    # Find best in initial population
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # DE iteration with time tracking
    iteration = 0
    F = 0.8
    CR = 0.7
    
    while time.time() - start_time < budget_s - 1 and iteration < maxiter:
        iteration += 1
        
        new_population = []
        new_fitnesses = []

            # Select three different random individuals
            idxs = np.random.choice(popsize, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + F * (b - c)
            # Ensure bounds
            mutant = np.clip(mutant, 0, 1)
            # Renormalize
            mutant = mutant * (n_points / 2) / np.sum(mutant)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, mutant, population[i])
            
            # Evaluate new individual
            fitness = objective(trial)
            
            # Selection
            if fitness < fitnesses[i]:
                new_population.append(trial)
                new_fitnesses.append(fitness)
                
                # Update best solution
                if fitness < best_c5:
                    best_c5 = fitness
                    best_h = trial.copy()

                new_population.append(population[i])
                new_fitnesses.append(fitnesses[i])

        fitnesses = new_fitnesses

                    best_c5 = fitnesses[i]
                    best_h = population[i].copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)
