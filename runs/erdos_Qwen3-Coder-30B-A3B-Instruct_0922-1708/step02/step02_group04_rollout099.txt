import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    # Normalize to exact integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # GA parameters
    pop_size = min(50, max(20, n_points // 2))
    max_generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = pop_size // 4
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            individual = h.copy()
        else:
            individual = np.random.rand(n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
            individual = np.clip(individual, 0, 1)
        population.append(individual)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        generation += 1
        
        # Evaluate fitness of population
        fitness_scores = []
        for individual in population:
            corr = correlate(individual, 1 - individual, mode='full')
            c5 = np.max(corr) * dx
            fitness_scores.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = individual.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Generate rest through selection, crossover, mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(sorted_population, sorted_fitness, 3)
            parent2 = tournament_selection(sorted_population, sorted_fitness, 3)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            mutate(child1, mutation_rate)
            mutate(child2, mutation_rate)
            
            # Normalize and clip
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child1 = np.clip(child1, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            child2 = np.clip(child2, 0, 1)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        new_population = new_population[:pop_size]
        population = new_population
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)

def tournament_selection(population, fitness_scores, tournament_size):
    """Select an individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), 
                                        size=tournament_size, 
                                        replace=False)
    tournament_fitness = [fitness_scores[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx].copy()

def crossover(parent1, parent2):
    """Single-point crossover"""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(individual, mutation_rate):
    """Gaussian mutation"""
    for i in range(len(individual)):
        if np.random.rand() < mutation_rate:
            individual[i] += np.random.normal(0, 0.05)
