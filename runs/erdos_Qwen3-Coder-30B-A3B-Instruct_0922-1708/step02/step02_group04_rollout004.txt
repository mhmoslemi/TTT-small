    best_h = h.copy()
    best_c5 = float('inf')

    # Try a few different strategies to avoid local minima

            best_h = h.copy()
        
        # Adapting step size based on progress
        elapsed = time.time() - start_time
        if elapsed < budget_s * 0.3:
            step_size = 0.02
        elif elapsed < budget_s * 0.6:
            step_size = 0.01
        else:
            step_size = 0.005
            
        # Random perturbations with adaptive step size
        perturb = np.random.normal(0, step_size, n_points)
        # Clip both ways to ensure feasibility after scaling
        h_new = h + perturb
