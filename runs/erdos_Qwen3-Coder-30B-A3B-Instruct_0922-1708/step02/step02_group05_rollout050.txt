import numpy as np
from scipy.optimize import differential_evolution
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given h."""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def constraint_sum(h, target_sum):
    """Constraint function for sum of h."""
    return np.sum(h) - target_sum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * target_sum / np.sum(h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    constraints = {'type': 'eq', 'fun': lambda x: constraint_sum(x, target_sum)}
    
    # Objective function
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Use differential evolution with custom bounds and constraints
    start_time = time.time()
    
    # Initial best solution
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Differential evolution parameters
    population_size = 50
    max_generations = 500
    mutation_factor = 0.8
    crossover_rate = 0.7
    
    # Keep track of current best solution
    current_best_h = best_h.copy()
    current_best_c5 = best_c5
    
    # For debugging, we'll iterate manually to ensure we respect the time limit
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate new population
        new_population = []
        
        # Start with current best solution
        new_population.append(current_best_h.copy())
        
        # Generate rest of the population
        for _ in range(population_size - 1):
            # Select three distinct individuals
            idxs = np.random.choice(n_points, 3, replace=False)
            a, b, c = current_best_h, current_best_h, current_best_h
            
            # Mutation step
            mutant = a + mutation_factor * (b - c)
            
            # Clip to [0,1]
            mutant = np.clip(mutant, 0, 1)
            
            # Renormalize
            mutant = mutant * target_sum / np.sum(mutant)
            
            # Crossover step
            trial = np.where(np.random.rand(n_points) < crossover_rate, mutant, current_best_h)
            
            new_population.append(trial)
        
        # Evaluate all individuals in population
        population_fitness = [objective(pop) for pop in new_population]
        
        # Selection: keep best individual
        min_idx = np.argmin(population_fitness)
        new_best_h = new_population[min_idx]
        new_best_c5 = population_fitness[min_idx]
        
        # Update current best if improved
        if new_best_c5 < current_best_c5:
            current_best_h = new_best_h.copy()
            current_best_c5 = new_best_c5
            
        # Update global best
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
