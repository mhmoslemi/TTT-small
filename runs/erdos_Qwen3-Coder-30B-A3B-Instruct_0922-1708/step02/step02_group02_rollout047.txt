import numpy as np
from cvxpy import Variable, Minimize, Problem, sum, multiply, diag, sqrt
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute lag indices
    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1:
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # QP approach for iterative improvement
        # Variables: h[0:n_points], t (max overlap)
        h_var = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1]
        constraints.append(sum(h_var) == n_points / 2)
        constraints.append(t >= 0)
        
        # Add overlap constraints for all lags
        for k in lags:
            # For lag k, compute sum of h[i]*(1-h[i+k]) for valid i
            # Use a compact form for constraint
            corr_expr = 0
            for i in range(n_points - k):
                corr_expr += h_var[i] * (1 - h_var[i + k])
            constraints.append(corr_expr * dx <= t)
            
        # Objective
        objective = Minimize(t)
        prob = Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array(h_var.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                # Re-scale to match constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # If QP fails, fallback to gradient step
                eps = 1e-5
                grad = np.zeros_like(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    corr = np.correlate(h, 1 - h, mode='full')
                    c5 = np.max(corr) * dx
                    
                    grad[i] = (c5_plus - c5) / eps
                
                # Update h using gradient descent
                lr = 0.001
                h_new = h - lr * grad
                
                # Clip and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
        except Exception:
            # Fallback to gradient step if any error
            eps = 1e-5
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update h using gradient descent
            lr = 0.001
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
