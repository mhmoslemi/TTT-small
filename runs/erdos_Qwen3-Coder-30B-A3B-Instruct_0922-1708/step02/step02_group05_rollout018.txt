import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Random search with periodic gradient refinement
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Occasionally refine with gradient step
        if iteration % 10 == 0:
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
                
            # Adaptive learning rate
            grad_norm = np.linalg.norm(grad)
            lr = min(0.1, 1.0 / (grad_norm + 1e-8)) if grad_norm > 1e-10 else 0.01
            
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
        else:
            # Random mutation
            mutation_magnitude = 0.05
            mask = np.random.random(n_points) < 0.3
            h += np.random.normal(0, mutation_magnitude, n_points) * mask
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
