import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        # Create diverse initial population
        for _ in range(population_size):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Random initialization
        for _ in range(population_size):
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Selection: keep top 20%
        keep_size = max(1, population_size // 5)
        selected = population[:keep_size]
        
        # Create new population
        new_population = selected.copy()
        
        # Add offspring
        while len(new_population) < population_size:
            # Mutation
            if np.random.random() < 0.5 or len(new_population) < population_size // 2:
                parent = selected[np.random.randint(0, len(selected))]
                h_mut = parent.copy()
                # Mutate 20% of elements
                mutate_indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
                for i in mutate_indices:
                    h_mut[i] += np.random.normal(0, 0.02)
                h_mut = np.clip(h_mut, 0, 1)
                h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
                new_population.append(h_mut)
            else:
                # Crossover
                parent1 = selected[np.random.randint(0, len(selected))]
                parent2 = selected[np.random.randint(0, len(selected))]
                h_cross = (parent1 + parent2) / 2.0
                h_cross = h_cross * (n_points / 2) / np.sum(h_cross)
                new_population.append(h_cross)
        
        population = new_population[:population_size]
        
        # Check for stagnation
        if generation > 100:
            if abs(fitness[0] - best_c5) < 1e-6:
                stagnation_count += 1
                if stagnation_count > 20:
                    # Increase mutation rate if stagnation occurs
                    for i in range(len(population)):
                        h_mut = population[i].copy()
                        mutate_indices = np.random.choice(n_points, size=int(0.3 * n_points), replace=False)
                        for j in mutate_indices:
                            h_mut[j] += np.random.normal(0, 0.02)
                        h_mut = np.clip(h_mut, 0, 1)
                        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
                        population[i] = h_mut
            else:
                stagnation_count = 0
    
    # Final check
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to simple approach if no better solution found
        h_simple = np.ones(n_points) * 0.5
        h_simple = h_simple * (n_points / 2) / np.sum(h_simple)
        corr = correlate(h_simple, 1 - h_simple, mode='full')
        final_c5 = np.max(corr) * dx
        return (h_simple.tolist(), final_c5, n_points)
