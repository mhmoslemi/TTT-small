import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use provided initial construction or create a simple one
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 45
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize initial vector to correct integral
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h_init = np.clip(h_init, 0, 1)
    
    # GA parameters
    pop_size = min(50, max(10, n_points // 2))
    max_generations = 1000
    mutation_rate = 0.1
    elite_count = pop_size // 5
    stagnation_limit = 100
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.7:
            h = h_init.copy()
            # Perturb slightly
            h += np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
        population.append(h)
    
    # Best solution tracking
    best_h = h_init.copy()
    best_c5 = float('inf')
    last_improvement = 0
    
    start_time = time.time()
    gen = 0
    
    while time.time() - start_time < budget_s - 1 and gen < max_generations:
        gen += 1
        
        # Evaluate all individuals
        fitness = []
        for h in population:
            correlation = correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = gen
        
        # Early stopping if no improvement
        if gen - last_improvement > stagnation_limit:
            break
            
        # Selection: roulette wheel
        fitness = np.array(fitness)
        # invert fitness for selection (lower is better)
        inverted_fitness = 1.0 / (fitness + 1e-12)
        scores = inverted_fitness / np.sum(inverted_fitness)
        
        # Elitism - keep best individuals
        elite_indices = np.argsort(fitness)[:elite_count]
        new_population = [population[i].copy() for i in elite_indices]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Select two parents
            idxs = np.random.choice(pop_size, size=2, p=scores)
            parent1, parent2 = population[idxs[0]], population[idxs[1]]
            
            # Crossover: blend
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutate
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.03, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            
            # Ensure integral constraint
            child = child * (n_points / 2) / np.sum(child)
            child = np.clip(child, 0, 1)
            
            new_population.append(child)
            
        population = new_population[:pop_size]
        
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
