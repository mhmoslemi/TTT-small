import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    start_time = time.time()
    
    # Prepare integer indices for correlation computation
    # We'll use CVXPY to solve the optimization problem with different constraints
    while time.time() - start_time < budget_s - 1:
        # Compute current correlation and C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Use quadratic programming approach
        # Variables
        h_var = Variable(n_points)
        t = Variable()
        
        # Objective: minimize t
        objective = Minimize(t)
        
        # Constraints
        constraints = [0 <= h_var, h_var <= 1, 
                       sum(h_var) == n_points / 2,
                       t >= 0]
        
        # Correlation constraints (only a subset at a time for efficiency)
        # The idea is to identify the most problematic lags and add them
        # Use a greedy approach based on current correlation
        corr_indices = np.argsort(correlation)[::-1]  # descending order
        
        # Select top k lags that contribute most to current max correlation
        # This is a heuristic to focus on most relevant constraints
        k = min(20, len(lags))  # Limit number of constraints to avoid overfitting
        selected_lags = corr_indices[1:k+1]  # Skip index 0, it's the unshifted correlation
        
        # Convert full correlation indices to actual lags
        actual_lags = selected_lags - (n_points - 1)  # see how correlate works
        
        # Generate constraints for these
        for i, lag in enumerate(selected_lags):
            if lag >= n_points - 1:
                # lag = 2n - 2 - i -> corresponds to lag i - n + 2 = 2 - (1 + i - n) = 2 - (n + 1 - i)
                corr_val = correlation[lag]
                # Sum of h[i] (1 - h[i + lag]) over valid indices
                left = i
                right = min(n_points - lag - 1, n_points - 1)
                valid_idx = np.arange(left, right + 1)
                const = sum(h_var[j] * (1 - h_var[j + lag]) for j in valid_idx) * dx
                constraints += [const <= t]
        
        # Solve the problem
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status == 'optimal':
                h_new = np.array(h_var.value).flatten()
                h = h_new.copy()
            else:
                # Fallback to gradient descent if QP fails
                grad = np.zeros_like(h)
                eps = 1e-5
                c5 = np.max(correlation) * dx
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)

                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx

                    grad[i] = (c5_plus - c5) / eps

                lr = 0.005
                h_new = h - lr * grad

                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except Exception:
            # Fallback on error
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)

                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx

                grad[i] = (c5_plus - c5) / eps

            lr = 0.005
            h_new = h - lr * grad

            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
