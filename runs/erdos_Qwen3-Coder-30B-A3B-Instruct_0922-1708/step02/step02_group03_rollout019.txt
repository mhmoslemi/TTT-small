import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    learning_rate = 0.002
    eps = 1e-6
    
    # For PSO-like behavior with adaptation
    velocity = np.zeros(n_points)
    inertia_weight = 0.7
    cognitive_param = 0.9
    social_param = 0.9
    
    # Track best solution found
    pbest_h = h.copy()
    pbest_c5 = float('inf')
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Update personal best
        if c5 < pbest_c5:
            pbest_c5 = c5
            pbest_h = h.copy()
            
        # PSO update step
        r1, r2 = np.random.random(2)
        velocity = (inertia_weight * velocity + 
                   cognitive_param * r1 * (pbest_h - h) + 
                   social_param * r2 * (best_h - h))
        
        h_new = h + velocity
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Occasionally apply local optimization with randomized restarts
        if iteration % 50 == 0:
            # Try simpler gradient descent with small perturbations
            trials = 5
            for _ in range(trials):
                h_trial = h + np.random.normal(0, 0.01, n_points)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
                corr_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
                c5_trial = np.max(corr_trial) * dx
                
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
        
        # Termination condition
        if iteration % 500 == 0:
            if time.time() - start_time > budget_s - 5:
                break
                
    return (best_h.tolist(), best_c5, n_points)
