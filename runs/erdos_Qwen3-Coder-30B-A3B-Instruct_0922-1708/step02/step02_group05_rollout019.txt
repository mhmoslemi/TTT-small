import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Initialize population for genetic algorithm
    population_size = 50
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            ind = np.array(initial_h_values)
            if len(ind) != n_points:
                ind = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(ind)), ind)
            ind = np.clip(ind, 0, 1)
        else:
            ind = np.random.rand(n_points)
        ind = ind * (n_points / 2) / np.sum(ind)
        population.append(ind)
    
    # Add the initial solution to the population if it exists
    if 'initial_h_values' in globals():
        population[0] = h.copy()
    
    start_time = time.time()
    generation = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and generation - last_improvement < 50:
        generation += 1
        
        # Evaluate fitness for all individuals
        fitness = []
        for ind in population:
            correlation = np.correlate(ind, 1 - ind, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
        # Keep track of best solution in population
        best_in_pop = min(fitness)
        if best_in_pop < best_c5:
            best_c5 = best_in_pop
            best_h = population[fitness.index(best_in_pop)].copy()
            last_improvement = generation
            
        # Selection: tournament
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
            
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % population_size]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            mutation_rate = 0.2
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.1)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.1)
                    
            # Repair
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        
    return (best_h.tolist(), best_c5, n_points)
