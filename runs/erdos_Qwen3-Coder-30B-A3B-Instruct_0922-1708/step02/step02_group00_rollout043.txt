import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute the overlap integral C5
    correlation = correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def initialize_population(n_points, population_size, initial_h_values=None):
    if initial_h_values is not None:
        # Interpolate to target resolution
        x_orig = np.linspace(0, 2, len(initial_h_values))
        x_new = np.linspace(0, 2, n_points)
        h0 = np.interp(x_new, x_orig, initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Initialize with uniform distribution
        h0 = np.ones(n_points) * 0.5
        
    population = [h0.copy()]
    for _ in range(population_size - 1):
        # Create variation with small mutations
        h = h0.copy()
        # Mutate 20% of points
        indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
        for i in indices:
            h[i] += np.random.uniform(-0.1, 0.1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    return population

def mutate(h, rate=0.2):
    h_new = h.copy()
    indices = np.random.choice(len(h), size=int(rate * len(h)), replace=False)
    for i in indices:
        h_new[i] += np.random.uniform(-0.1, 0.1)
    h_new = np.clip(h_new, 0, 1)
    h_new = h_new * (len(h_new) / 2) / np.sum(h_new)
    return h_new

def crossover(h1, h2):
    # Simple single-point crossover
    n = len(h1)
    point = np.random.randint(1, n)
    h_new1 = np.concatenate([h1[:point], h2[point:]])
    h_new2 = np.concatenate([h2[:point], h1[point:]])
    h_new1 = h_new1 * (n / 2) / np.sum(h_new1)
    h_new2 = h_new2 * (n / 2) / np.sum(h_new2)
    return h_new1, h_new2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 1000
    
    # Initialize population
    population = initialize_population(n_points, population_size, initial_h_values)
    
    # Evaluate initial population
    fitness = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    generation = 0
    
    while (time.time() - start_time < budget_s - 1) and generation < max_generations:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50% as parents
        parents = population[:population_size//2]
        parent_fitness = fitness[:population_size//2]
        
        # Generate new population
        new_population = [population[0].copy()]  # Elitism
        
        # Generate offspring through crossover and mutation
        while len(new_population) < population_size:
            # Select two parents
            idx1 = np.random.randint(0, len(parents))
            idx2 = np.random.randint(0, len(parents))
            
            # Crossover with 70% probability
            if np.random.rand() < 0.7:
                h1, h2 = crossover(parents[idx1], parents[idx2])
                new_population.append(h1)
                if len(new_population) < population_size:
                    new_population.append(h2)
            else:
                # Mutate one parent
                h = mutate(parents[idx1])
                new_population.append(h)
        
        # Evaluate new population
        population = new_population
        fitness = [evaluate_h(h, dx) for h in population]
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
    
    return (best_h.tolist(), best_c5, n_points)
