import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1:
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Formulate and solve QP
        h_vars = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            sum(h_vars) == n_points / 2,
            t >= 0
        ]
        
        # Add correlation constraints for a subset of lags based on current violations
        # Get lags with highest correlation
        corr_indices = np.argsort(correlation)[-min(20, len(correlation)):][::-1]
        corr_lags = []
        for i in corr_indices:
            if i >= n_points:
                corr_lags.append(i - n_points)
            else:
                corr_lags.append(i)
        
        # Add just the most problematic lags
        active_lags = corr_lags[:min(10, len(corr_lags))] if corr_lags else [0]
        
        for k in active_lags:
            if k == 0:
                continue
            # Sum over pairs (i, i+k) where both indices exist
            lag_sum = 0
            for i in range(n_points - k):
                lag_sum += h_vars[i] * (1 - h_vars[i + k])
            constraints.append(lag_sum * dx <= t)
        
        # Objective
        objective = Minimize(t)
        
        # Solve QP
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                new_h = h_vars.value
                if new_h is not None:
                    h = np.clip(new_h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)  # Ensure constraint
                else:
                    # Fall back to gradient descent if QP fails
                    h = h + 0.001 * (np.random.rand(n_points) - 0.5)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            else:
                # QP failed, fall back to gradient descent
                h = h + 0.001 * (np.random.rand(n_points) - 0.5)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Any error in optimization, use gradient descent
            h = h + 0.001 * (np.random.rand(n_points) - 0.5)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
