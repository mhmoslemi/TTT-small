import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 constant for given step function h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population with diverse step functions."""
    population = []
    
    if initial_h is not None:
        h0 = np.array(initial_h)
        if len(h0) != n_points:
            x_orig = np.linspace(0, 2, len(h0))
            x_new = np.linspace(0, 2, n_points)
            h0 = np.interp(x_new, x_orig, h0)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        population.append(h0)
        
        # Generate variations around the initial solution
        for _ in range(pop_size - 1):
            h = h0.copy()
            # Random small perturbations
            perturb = np.random.uniform(-0.2, 0.2, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Start with diverse random functions
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    return population

def crossover(parent1, parent2):
    """Perform crossover between two parents."""
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.15):
    """Mutate a step function."""
    h = h.copy()
    n = len(h)
    n_mutate = max(1, int(n * mutation_rate))
    indices = np.random.choice(n, n_mutate, replace=False)
    for i in indices:
        h[i] += np.random.uniform(-0.2, 0.2)
    h = np.clip(h, 0, 1)
    h = h * (n / 2) / np.sum(h)
    return h

def select_parents(population, fitnesses, num_parents):
    """Select top parents based on fitness (low C5 is better)."""
    sorted_indices = np.argsort(fitnesses)
    return [population[i] for i in sorted_indices[:num_parents]]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    generations = 1000
    
    # Initialize
    initial_h = globals().get('initial_h_values', None)
    population = initialize_population(n_points, pop_size, initial_h)
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < generations:
        generation += 1
        
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Select parents (top 50%)
        num_parents = pop_size // 2
        parents = select_parents(population, fitnesses, num_parents)
        
        # Generate new population
        new_population = parents.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Select two random parents
            p1, p2 = np.random.choice(parents, 2, replace=False)
            
            # Crossover with 70% probability
            if np.random.rand() < 0.7:
                child1, child2 = crossover(p1, p2)
            else:
                child1, child2 = p1.copy(), p2.copy()
                
            # Mutation
            child1 = mutate(child1, 0.15)
            child2 = mutate(child2, 0.15)
            
            new_population.extend([child1, child2])
        
        # Keep only the pop_size individuals
        population = new_population[:pop_size]
    
    # Final evaluation of best solution
    if best_h is not None:
        best_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist() if best_h is not None else [0.5]*n_points, best_c5, n_points)
