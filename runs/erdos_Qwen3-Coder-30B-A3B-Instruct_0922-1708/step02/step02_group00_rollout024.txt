import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Define smooth max approximation for better optimization
    def smooth_max(x, tau=1.0):
        """Softmax approximation of maximum"""
        return (1.0 / tau) * np.log(np.sum(np.exp(tau * x)))
        
    # Define surrogate objective
    def surrogate_objective(h, tau=1.0):
        # Compute all correlations
        correlation = np.correlate(h, 1 - h, mode='full')
        # Smooth the max
        return smooth_max(correlation, tau) * dx
    
    # Gradient estimation
    def estimate_gradient(h, eps=1e-5, tau=1.0):
        grad = np.zeros_like(h)
        base_obj = surrogate_objective(h, tau)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            obj_plus = surrogate_objective(h_plus, tau)
            grad[i] = (obj_plus - base_obj) / eps
            
        return grad
    
    # Optimization loop
    tau = 1.0  # Temperature for smooth max
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient
        grad = estimate_gradient(h, tau=tau)
        
        # Update using gradient descent
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update temperature
        if iteration % 100 == 0:
            tau *= 0.95
            
        # Occasionally try a heuristic restart
        if iteration % 1000 == 0:
            # Restart with better random initialization
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
