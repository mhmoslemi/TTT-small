import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize with provided construction or create simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Initialize with 3 segments: high, low, high
        h = np.ones(n_points) * 0.25
        # First segment: 0.5 width, value 0.75
        mid1 = n_points // 3
        mid2 = 2 * n_points // 3
        h[:mid1] = 0.75
        h[mid1:mid2] = 0.25
        h[mid2:] = 0.75
        
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Objective function - we want to minimize max overlap
    def objective(params):
        # Reshape params into h
        h_temp = np.array(params)
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        
        correlation = np.correlate(h_temp, 1 - h_temp, mode='full')
        return np.max(correlation) * dx

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Optimizer: minimize the max overlap
        # We construct initial simplex based on current solution, and optimize
        initial_params = h.copy()
        
        # Random neighbor perturbation to avoid local minima
        perturbed = initial_params + np.random.normal(0, 0.02, n_points)
        perturbed[perturbed < 0] = 0
        perturbed[perturbed > 1] = 1
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        
        # Use Nelder-Mead with a subset of points to explore better solutions
        try:
            result = minimize(objective, perturbed, method='Nelder-Mead',
                                options={'maxiter': 50, 'xtol': 1e-6, 'ftol': 1e-6})
            if result.success:
                h_new = np.clip(result.x, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                new_c5 = objective(h_new)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                h = h_new.copy()
            else:
                # fallback to simple perturbation
                h = perturbed
        except:
            # Fallback: simple neighborhood search
            h = perturbed
            
        # Occasionally do a bigger jump to escape local minima
        if np.random.rand() < 0.05:
            idx = np.random.randint(0, n_points)
            h[idx] = np.random.rand()
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)

    # Final validation and cleanup
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
