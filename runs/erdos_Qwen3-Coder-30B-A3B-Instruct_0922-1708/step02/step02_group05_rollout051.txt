import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define constraints for optimization
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function to minimize
    def objective(h):
        # Ensure h is within bounds
        h = np.clip(h, 0, 1)
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Use differential evolution for global optimization
    # Try different population sizes and numbers of generations
    popsize = 50
    maxiter = 500
    mutation = (0.5, 1.0)
    crossover = 0.7
    
    # Set up DE options
    de_options = {
        'popsize': popsize,
        'maxiter': maxiter,
        'mutation': mutation,
        'crossover': crossover,
        'seed': seed,
        'strategy': 'best1bin',
        'tol': 1e-6,
        'recombination': 0.7,
        'disp': False
    }
    
    # Run DE
    try:
        result = differential_evolution(
            objective,
            bounds,
            constraints=constraints,
            **de_options
        )
        
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        # Compute final C5
        correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
        c5_opt = np.max(correlation) * dx
        best_h = h_opt
        best_c5 = c5_opt
        
    except Exception as e:
        # Fallback to initial if DE fails
        pass
        
    # Refinement with local search
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < 1000:
        iter_count += 1
        
        # Random perturbation to escape local optima
        h = best_h.copy()
        perturbation = np.random.normal(0, 0.01, n_points)
        h = h + perturbation
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Occasionally reinitialize from best if stagnation
        if iter_count % 100 == 0:
            h = best_h.copy()
            noise = np.random.normal(0, 0.02, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
