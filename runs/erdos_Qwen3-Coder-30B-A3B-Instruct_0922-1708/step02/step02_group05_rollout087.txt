import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Convex relaxation to get a good starting point
    start_time = time.time()
    
    # Create optimization variables
    h_cvx = cp.Variable(n_points)
    
    # Objective: minimize max correlation
    # For convex relaxation, we'll minimize the sum of all correlations
    # rather than the maximum, to make it convex
    
    # This is a rough approximation approach
    # We'll solve the convex problem first
    constraints = [
        h_cvx >= 0,
        h_cvx <= 1,
        cp.sum(h_cvx) == n_points / 2
    ]
    
    # Use a simple proxy for convex relaxation
    obj = cp.Minimize(cp.sum_squares(h_cvx))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = h_cvx.value.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    # Now refinement with gradient descent
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Calculate initial C5
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Gradient descent loop
    max_iterations = 20000
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-12:
            lr = min(0.1, 1.0 / (grad_norm + 1e-12))
        else:
            lr = 0.01
            
        # Update step
        h_new = h - lr * grad
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally recompute C5 to prevent oscillation
        if iteration % 100 == 0:
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final re-computation and validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
