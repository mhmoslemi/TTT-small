import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_generations = 1000
    population_size = 50
    mutation_rate = 0.15
    crossover_rate = 0.7
    elite_size = population_size // 10

    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_new = np.linspace(0, 2, n_points)
            h_base = np.interp(x_new, x_orig, h_base)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        population.append(h_base)
        for _ in range(population_size - 1):
            h_mut = h_base.copy()
            indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
            h_mut[indices] += np.random.uniform(-0.2, 0.2, len(indices))
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            population.append(h_mut)
    else:
        # Initialize with random spikes
        for _ in range(population_size):
            h = np.zeros(n_points)
            num_spikes = np.random.randint(5, 11)
            spike_positions = np.random.randint(0, n_points, num_spikes)
            spike_heights = np.random.uniform(0.5, 1.0, num_spikes)
            for i, pos in enumerate(spike_positions):
                h[pos] = spike_heights[i]
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            if np.random.random() < crossover_rate and len(new_population) + 1 < population_size:
                # Crossover
                parent1, parent2 = np.random.choice(population[:population_size//2], 2, replace=False)
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                for child in [child1, child2]:
                    indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
                    child[indices] += np.random.uniform(-0.2, 0.2, len(indices))
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                    new_population.append(child)
                    if len(new_population) >= population_size:
                        break
            else:
                # Mutation only
                parent = np.random.choice(population[:population_size//2])
                child = parent.copy()
                indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
                child[indices] += np.random.uniform(-0.2, 0.2, len(indices))
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
                
        population = new_population[:population_size]
    
    return (best_h.tolist(), best_c5, n_points)
