            # Interpolate to finer grid
            h_new = np.zeros(grid_n)
            for i in range(grid_n):
                idx = i * current_n / grid_n
                idx_floor = int(np.floor(idx))
                idx_ceil = min(idx_floor + 1, current_n - 1)
                frac = idx - idx_floor

            h = h_new
            current_n = grid_n

            h = h_new
            current_n = grid_n
