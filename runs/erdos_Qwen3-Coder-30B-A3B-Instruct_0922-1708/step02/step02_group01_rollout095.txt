import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    N_pop = 50
    C_rate = 0.8
    M_rate = 0.2
    disturbance_rate = 0.2
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    # Fill the rest of the population
    for _ in range(N_pop - len(population)):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    population = np.array(population)
    
    # Keep track of the best solution
    best_h = population[0].copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    gen = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        gen += 1
        
        # Evaluate all individuals in the population
        fitness = []
        for h in population:
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
        
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
            
        no_improvement_count += 1
        
        # Selection: tournament selection
        selected = []
        tournament_size = 2
        for _ in range(N_pop):
            tournament = np.random.choice(N_pop, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover
        new_population = []
        for i in range(0, N_pop, 2):
            if np.random.rand() < C_rate and i+1 < N_pop:
                parent1 = selected[i]
                parent2 = selected[i+1]
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = selected[i].copy()
                child2 = selected[min(i+1, N_pop-1)].copy()
                
            new_population.extend([child1, child2])
        
        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < M_rate:
                # Add noise to each gene
                noise = np.random.uniform(-0.1, 0.1, n_points)
                new_population[i] = new_population[i] + noise
                
                # Clip values
                new_population[i] = np.clip(new_population[i], 0, 1)
                
                # Rescale to maintain constraint
                scale = (n_points / 2) / np.sum(new_population[i])
                new_population[i] = new_population[i] * scale
        
        # Apply elitism
        # Keep top 50% of old population
        elite_indices = np.argsort(fitness)[:N_pop//2]
        elite_population = population[elite_indices]
        
        # Replace bottom 50% of new population
        new_population = new_population[:N_pop//2] + [elite_population[i].tolist() for i in range(N_pop//2)]
        
        # Disturbance if no improvement for a while
        if no_improvement_count > 50:
            for i in range(int(N_pop * disturbance_rate)):
                idx = np.random.randint(0, N_pop)
                new_population[idx] = np.random.rand(n_points)
                new_population[idx] = new_population[idx] / np.sum(new_population[idx]) * (n_points / 2)
        
        population = np.array(new_population)
        
        # Occasionally restart with better points
        if gen % 50 == 0 and no_improvement_count > 30:
            for i in range(N_pop):
                if np.random.rand() < 0.3:
                    population[i] = np.random.rand(n_points)
                    population[i] = population[i] / np.sum(population[i]) * (n_points / 2)
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
