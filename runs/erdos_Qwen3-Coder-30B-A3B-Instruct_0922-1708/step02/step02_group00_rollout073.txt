        correlation = np.correlate(h, 1 - h, mode='full')

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # L(τ) = (1/τ) * log(∑ₖ exp(τ * ∑ᵢ hᵢ(1 - h_{i+k})))

        if len(overlaps) > 0:

            h = np.random.rand(n_points) * 0.5
