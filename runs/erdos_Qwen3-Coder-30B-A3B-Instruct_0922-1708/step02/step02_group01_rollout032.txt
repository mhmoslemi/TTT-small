import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Multi-start optimization with perturbations
    n_particles = 10
    particles = []
    
    # Initialize particles
    particles.append(h.copy())
    for i in range(n_particles - 1):
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
    
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    lr = 0.01
    momentum = 0.9
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for p_idx, particle in enumerate(particles):
            # Compute the overlap integral
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
            
            # Compute gradient with finite differences
            grad = np.zeros_like(particle)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = particle.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update particle with momentum
            velocities[p_idx] = momentum * velocities[p_idx] - lr * grad
            particles[p_idx] = particle + velocities[p_idx]
            
            # Clip and rescale
            particles[p_idx] = np.clip(particles[p_idx], 0, 1)
            particles[p_idx] = particles[p_idx] * (n_points / 2) / np.sum(particles[p_idx])
            
            # Occasionally perturb particle
            if np.random.random() < 0.05:
                noise = np.random.uniform(-0.05, 0.05, n_points)
                particles[p_idx] = particles[p_idx] + noise
                particles[p_idx] = np.clip(particles[p_idx], 0, 1)
                particles[p_idx] = particles[p_idx] * (n_points / 2) / np.sum(particles[p_idx])
        
        # Occasionally restart particles
        if iteration % 100 == 0 and best_c5 < 0.381:
            for p_idx in range(n_particles):
                # Add some noise to restart
                perturbation = np.random.uniform(-0.05, 0.05, n_points)
                particles[p_idx] = best_h + perturbation
                particles[p_idx] = np.clip(particles[p_idx], 0, 1)
                particles[p_idx] = particles[p_idx] * (n_points / 2) / np.sum(particles[p_idx])
    
    return (best_h.tolist(), best_c5, n_points)
