import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    n_particles = 50
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iter = int(budget_s * 1000)  # Rough estimate for iterations
    
    # Initialize particles
    particles = np.random.random((n_particles, n_points)) * 0.5
    velocities = np.random.random((n_particles, n_points)) - 0.5
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles[0] = h_init
    
    # Initialize personal best and global best
    personal_best = particles.copy()
    personal_best_fitness = np.full(n_particles, float('inf'))
    
    global_best_idx = 0
    global_best_fitness = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Evaluate fitness for all particles
        for i in range(n_particles):
            h = particles[i]
            # Project to feasible region
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = np.clip(h * scale, 0, 1)
            particles[i] = h
            
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(correlation) * dx
            
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best[i] = h.copy()
                
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best_idx = i
                
        # Update velocities and positions
        for i in range(n_particles):
            if i == global_best_idx:
                continue
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (personal_best[i] - particles[i]) 
                           + c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            
            particles[i] += velocities[i]
            
            # Project to feasible region
            h = particles[i]
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = np.clip(h * scale, 0, 1)
            particles[i] = h
            
        # Occasionally restart particles if stagnation
        if iteration % 100 == 0:
            for i in range(1, n_particles):  # Skip first
                if np.random.random() < 0.1:
                    particles[i] = np.random.random(n_points) * 0.5
                    velocities[i] = np.random.random(n_points) - 0.5
                    
        # Early termination check
        if global_best_fitness < 0.3807:
            break
            
    # Return the best solution found
    best_h = personal_best[global_best_idx]
    return (best_h.tolist(), global_best_fitness, n_points)
