import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Hyperparameters
    lr_initial = 0.001
    lr_decay = 0.999
    lr = lr_initial
    
    # Number of restarts
    n_restarts = 5
    
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        h = best_h.copy() if restart > 0 else h.copy()
        
        last_improvement = 0
        for i in range(10000):
            if time.time() - start_time > budget_s - 1:
                break
                
            iteration += 1
            
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = i
            
            # Compute gradient for the maximal lag
            max_lag_idx = np.argmax(correlation)
            max_lag = max_lag_idx - (len(h) - 1)  # Convert to actual lag
            
            grad = np.zeros_like(h)
            for j in range(n_points):
                # Gradient of h[j] for lag max_lag
                left_idx = (j - max_lag) % n_points
                right_idx = (j + max_lag) % n_points
                grad[j] = (1 - h[left_idx]) - h[right_idx]
            
            # Update h
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
            # Adaptive learning rate
            lr *= lr_decay
            
            # Early stopping
            if i - last_improvement > 500:
                break
                
    return (best_h.tolist(), best_c5, n_points)
