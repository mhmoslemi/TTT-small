import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a binary-like comb function
        h = np.zeros(n_points)
        # Use 5 pulses, each of width 0.2
        pulse_width = 0.2
        spacing = 0.25
        # Place 5 non-overlapping pulses
        for i in range(5):
            start = i * (pulse_width + spacing)
            end = start + pulse_width
            start_idx = int(start / 2.0 * n_points)
            end_idx = int(end / 2.0 * n_points)
            if end_idx > n_points:
                end_idx = n_points
            if start_idx < n_points:
                h[start_idx:end_idx] = 1.0
        # Rescale to unit integral
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')

    start_time = time.time()
    
    # Objective function to minimize
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize using differential evolution
    result = differential_evolution(
        objective,
        bounds,
        maxiter=1000,
        popsize=15,
        tol=1e-6,
        recombination=0.7,
        mutation=(0.5, 1),
        seed=seed,
        callback=lambda x, convergence: None
    )
    
    # Extract the best solution
    final_h = result.x.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(correlation) * dx

    return (final_h.tolist(), final_c5, n_points)
