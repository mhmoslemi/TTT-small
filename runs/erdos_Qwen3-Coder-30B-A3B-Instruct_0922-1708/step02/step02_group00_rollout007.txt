import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.random.uniform(0.1, 0.9, n_points)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Adam optimizer variables
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.001
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    t = 0
    
    while time.time() - start_time < budget_s - 1 and t < 1000000:
        t += 1
        
        # Compute all correlations and find maximum
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find critical shift k that maximizes overlap
        max_k = np.argmax(corr)
        # The full correlation is size 2*n_points - 1
        # k from 0 to 2*n_points - 2
        # middle is at n_points - 1 (k = n_points - 1 corresponds to k=0 in shifted indexing)
        k = max_k - (n_points - 1)
        
        # Compute gradient w.r.t. h for this shift
        grad = np.zeros(n_points)
        if k >= 0:
            # for i such that i+k < n_points, partial derivative w.r.t. h[i] is 1 - h[i+k]
            for i in range(n_points - k):
                grad[i] += 1 - h[i + k]
        else:
            # for i such that i+k >= 0, partial derivative w.r.t. h[i] is -h[i-k]
            k_abs = -k
            for i in range(k_abs, n_points):
                grad[i] -= h[i - k_abs]
                
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Clip and re-scale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally perturb
        if t % 500 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
