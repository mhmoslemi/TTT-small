import numpy as np
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute lag indices for efficiency
    max_lag = n_points - 1
    all_lags = list(range(1, max_lag + 1))
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Use QP to address the optimization by minimizing the maximum overlap
        # Variables
        h_vars = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add most problematic correlation constraints
        # Compute worst-case correlations
        corr_vals = np.correlate(h, 1 - h, mode='full') * dx
        # Select top lags to focus on
        sorted_lags = sorted(range(len(corr_vals)), key=lambda k: corr_vals[k], reverse=True)
        # Use top 50 lags for constraints, or all if less
        top_lags = sorted_lags[:min(50, len(sorted_lags))]
        
        # Add linear constraints to the QP problem
        for lag_idx in top_lags:
            # Only process lags that actually contribute
            if lag_idx < len(corr_vals) and corr_vals[lag_idx] > 0:
                # Constraint: sum_{i} h[i] * (1 - h[i + lag_idx]) * dx <= t
                expr = 0.0
                # Compute correlation sum for specific lag
                for i in range(n_points - lag_idx):
                    expr += h_vars[i] * (1 - h_vars[i + lag_idx])
                expr *= dx
                constraints.append(expr <= t)
        
        # Objective: minimize t
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_new = np.array(h_vars.value).flatten()
                # Fix numerical issues
                h_new = np.clip(h_new, 0, 1)
                scale_factor = (n_points / 2) / np.sum(h_new)
                h_new *= scale_factor
                h = h_new
            else:
                # Fallback: gradient descent as backup
                grad = np.zeros_like(h)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5) / eps
                lr = 0.001  # Reduced learning rate
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except:
            # Fallback on gradient descent if QP fails
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            lr = 0.001
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
