import numpy as np
from scipy.optimize import differential_evolution
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Objective function for differential evolution
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize using differential evolution with custom bounds
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=1000,
        popsize=15,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        tol=1e-6,
        disp=False
    )
    
    h_final = de_result.x
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Final evaluation
    correlation = np.correlate(h_final, 1 - h_final, mode='full')
    c5_final = np.max(correlation) * dx
    
    return (h_final.tolist(), c5_final, n_points)
