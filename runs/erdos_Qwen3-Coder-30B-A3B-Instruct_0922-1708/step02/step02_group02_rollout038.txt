import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
    else:
        h_base = np.ones(n_points) * 0.5
    
    for i in range(population_size):
        if i == 0:
            h = h_base.copy()
        else:
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            corr = correlate(h, 1-h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Selection: keep top 20%
        keep_count = max(1, population_size // 5)
        selected = population[:keep_count]
        
        # Create new population through mutation and crossover
        new_population = selected.copy()
        
        while len(new_population) < population_size:
            # Crossover
            if len(selected) >= 2 and np.random.random() < 0.5:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            else:
                # Mutation
                parent = np.random.choice(selected)
                child = parent.copy()
                # Mutate 20% of elements
                indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
                for i in indices:
                    child[i] += np.random.normal(0, 0.02)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        population = new_population
        
        # Track improvement
        if generation % 10 == 0:
            if generation > 100:
                no_improvement_count += 1
                if no_improvement_count > 10:
                    # Increase mutation rate
                    for i in range(len(population)):
                        if np.random.random() < 0.3:
                            h = population[i]
                            indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
                            for j in indices:
                                h[j] += np.random.normal(0, 0.05)
                            h = np.clip(h, 0, 1)
                            h = h * (n_points / 2) / np.sum(h)
                            population[i] = h
            else:
                no_improvement_count = 0
    
    # Final evaluation
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
    else:
        # Re-evaluate the best from final population
        fitness_final = []
        for h in population:
            corr = correlate(h, 1-h, mode='full')
            c5 = np.max(corr) * dx
            fitness_final.append(c5)
        best_idx = np.argmin(fitness_final)
        best_h = population[best_idx]
        final_c5 = fitness_final[best_idx]
    
    return (best_h.tolist(), final_c5, n_points)
