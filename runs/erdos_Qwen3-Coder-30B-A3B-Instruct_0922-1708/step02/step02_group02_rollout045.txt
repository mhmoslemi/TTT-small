import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute lags for overlap computation
    max_lag = n_points - 1
    lags = list(range(1, max_lag))
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute overlaps for all lags
        correlations = []
        for k in lags:
            corr = 0.0
            for i in range(n_points - k):
                corr += h[i] * (1 - h[i + k])
            correlations.append(corr * dx)
        
        # Find maximum overlap
        c5 = np.max(correlations)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # QP optimization
        # Decision variables
        h_var = Variable(n_points)
        t = Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            sum(h_var) == n_points / 2,
            t >= 0
        ]
        
        # Correlation constraints
        for i, k in enumerate(lags):
            # This is a bit tricky to express in CVXPY; using a specific
            # approach based on linearization of the integral
            corr_expr = 0
            for j in range(n_points - k):
                corr_expr += h_var[j] * (1 - h_var[j + k])
            constraints.append(corr_expr * dx <= t)
        
        # Objective
        objective = Minimize(t)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_new = h_var.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    # Normalize to correct integral
                    h = h * (n_points / 2) / np.sum(h)
                else:
                    # Fall back to gradient step if solver fails
                    h = h + 0.01 * (best_h - h)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            else:
                # Fall back to gradient step
                h = h + 0.01 * (best_h - h)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fall back to gradient step if solver fails
            h = h + 0.01 * (best_h - h)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
