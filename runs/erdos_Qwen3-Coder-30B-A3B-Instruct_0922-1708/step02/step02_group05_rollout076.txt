import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Define objective function to minimize
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Ensure sum constraint by adjusting the last variable
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    constraint = {'type': 'eq', 'fun': constraint_func}
    
    start_time = time.time()
    
    # Use differential evolution with population size and generations
    # Adjust parameters for reasonable time usage
    # Max generations to roughly fit within budget
    maxiter = min(500, int(budget_s / 2) if budget_s > 2 else 100)
    popsize = min(50, max(10, n_points // 2))
    
    # Run differential evolution
    result = differential_evolution(
        objective,
        bounds,
        args=(),
        maxiter=maxiter,
        popsize=popsize,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False,
        polish=True
    )
    
    if not result.success:
        # Fallback to gradient descent
        h = h.copy()
        for _ in range(2000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Simple gradient-based update
            eps = 1e-5
            grad = np.zeros_like(h)
            c5_base = objective(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = objective(h_plus)
                grad[i] = (c5_plus - c5_base) / eps
            
            # Adaptive learning rate
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-10:
                lr = min(0.05, 0.1 / (grad_norm + 1e-8))
            else:
                lr = 0.01
                
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
            # Check if this improves the solution
            c5_current = objective(h)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
    
    # Final check with the result from differential evolution
    if result.success:
        h_final = np.clip(result.x, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        final_c5 = objective(h_final)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final.copy()
    
    # Final safety check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
