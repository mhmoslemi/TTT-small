import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 1500
    mutation_rate = 0.2
    crossover_rate = 0.5
    tournament_size = 3
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h_temp = np.random.rand(n_points)
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        population.append(h_temp.copy())
    
    # Add the initial solution if it exists and is valid
    if 'initial_h_values' in globals():
        population[0] = h.copy()
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all population members
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        # Selection (tournament)
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            new_population.append(population[winner_idx].copy())
            
        # Crossover and Mutation
        for i in range(0, population_size, 2):
            parent1 = new_population[i]
            parent2 = new_population[(i+1) % population_size]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
                
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.05)
                    child2[j] += np.random.normal(0, 0.05)
                    
            # Repair constraints (clip and rescale)
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            # Replace worst individuals
            worst_indices = np.argsort(fitness)[-2:]
            population[worst_indices[0]] = child1
            population[worst_indices[1]] = child2
    
    # Final optimization of the best solution found
    def obj(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Use scipy minimize to polish the best solution
    result = minimize(obj, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points)
    if result.success:
        h_final = result.x.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        correlation_final = np.correlate(h_final, 1 - h_final, mode='full')
        final_c5 = np.max(correlation_final) * dx
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final
    
    return (best_h.tolist(), best_c5, n_points)
