import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 500
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize initial solution
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Particle positions and velocities
    positions = np.zeros((swarm_size, n_points))
    velocities = np.zeros((swarm_size, n_points))
    pbest_positions = np.zeros((swarm_size, n_points))
    pbest_values = np.full(swarm_size, np.inf)
    gbest_position = None
    gbest_value = np.inf
    
    # Initialize particles
    for i in range(swarm_size):
        # Start with a modified version of the initial solution
        if i == 0:
            positions[i] = h_init.copy()
        else:
            # Add small random noise to diversity
            positions[i] = h_init + np.random.normal(0, 0.1, n_points)
        
        # Ensure feasible values
        positions[i] = np.clip(positions[i], 0, 1)
        # Rescale to satisfy constraint
        positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
        
        # Update personal best
        corr = np.correlate(positions[i], 1 - positions[i], mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < pbest_values[i]:
            pbest_values[i] = c5
            pbest_positions[i] = positions[i].copy()
            
        # Update global best
        if c5 < gbest_value:
            gbest_value = c5
            gbest_position = positions[i].copy()
    
    # PSO parameters
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - positions[i]) +
                           c2 * r2 * (gbest_position - positions[i]))
            
            # Update position
            positions[i] += velocities[i]
            
            # Apply constraints
            positions[i] = np.clip(positions[i], 0, 1)
            # Rescale to satisfy constraint sum(h) = n_points/2
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
            
            # Evaluate
            corr = np.correlate(positions[i], 1 - positions[i], mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < pbest_values[i]:
                pbest_values[i] = c5
                pbest_positions[i] = positions[i].copy()
                
                # Update global best
                if c5 < gbest_value:
                    gbest_value = c5
                    gbest_position = positions[i].copy()
    
    # Final validation with exact method
    if gbest_position is not None:
        corr = np.correlate(gbest_position, 1 - gbest_position, mode='full')
        final_c5 = np.max(corr) * dx
    
    return (gbest_position.tolist(), final_c5, n_points)
