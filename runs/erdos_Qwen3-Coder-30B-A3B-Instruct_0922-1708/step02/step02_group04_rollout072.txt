import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    correlation = correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def enforce_constraints(h, n_points):
    """Enforce sum(h) = n_points / 2 and 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def crossover(parent1, parent2, n_points):
    """Perform segment-wise crossover."""
    # Use a simple uniform crossover
    mask = np.random.rand(n_points) > 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(h, mutation_rate=0.1, std=0.05):
    """Apply Gaussian mutation."""
    mask = np.random.rand(len(h)) < mutation_rate
    h = h + np.random.normal(0, std, len(h)) * mask
    return h

def select_parents(population, fitnesses):
    """Select parents using roulette wheel selection."""
    # Convert fitnesses to probabilities (lower is better)
    min_fitness = np.min(fitnesses)
    if min_fitness == np.max(fitnesses):
        # All same fitness
        probabilities = np.ones(len(fitnesses)) / len(fitnesses)
    else:
        # Avoid numerical issues
        fitnesses_shifted = fitnesses - min_fitness + 1e-10
        probabilities = 1.0 / fitnesses_shifted
        probabilities = probabilities / np.sum(probabilities)
    
    indices = np.random.choice(len(population), size=2, p=probabilities)
    return population[indices[0]], population[indices[1]]

def genetic_algorithm(n_points, population_size=50, budget_s=600):
    """Genetic algorithm for optimizing h."""
    start_time = time.time()
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = enforce_constraints(h, n_points)
        population.append(h)
    else:
        h = np.ones(n_points) * 0.5
        h = enforce_constraints(h, n_points)
        population.append(h)
        
    # Fill the rest of the population
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = enforce_constraints(h, n_points)
        population.append(h)
        
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    no_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 100:
        generation += 1
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        
        # Keep top half
        next_population = sorted_population[:population_size // 2]
        
        # Generate new offspring
        while len(next_population) < population_size:
            parent1, parent2 = select_parents(sorted_population, fitnesses)
            child = crossover(parent1, parent2, n_points)
            child = mutate(child)
            child = enforce_constraints(child, n_points)
            next_population.append(child)
        
        population = next_population
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use a reasonable discretization
    n_points = 100
    
    h_best, c5_best = genetic_algorithm(n_points, budget_s=budget_s)
    
    if h_best is None:
        # Fallback to simple approach if GA didn't find anything
        h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
        h = enforce_constraints(h, n_points)
        c5_best = evaluate_h(h, 2.0 / n_points)
        h_best = h
    
    return (h_best.tolist(), c5_best, n_points)
