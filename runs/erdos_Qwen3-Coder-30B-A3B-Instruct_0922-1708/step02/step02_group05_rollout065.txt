import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Genetic Algorithm Implementation
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.5
    tournament_size = 3
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            individual = h.copy()
            # Add small random noise
            noise = np.random.normal(0, 0.05, n_points)
            individual += noise
            individual = np.clip(individual, 0, 1)
            individual = individual * (n_points / 2) / np.sum(individual)
        else:
            individual = np.random.random(n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # Track current elite
    elite_h = h.copy()
    elite_c5 = evaluate_h(elite_h, dx)
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate fitness for all individuals
        fitness = []
        for individual in population:
            c5 = evaluate_h(individual, dx)
            fitness.append(c5)
            
        # Find current best in population
        current_best_idx = np.argmin(fitness)
        current_best_h = population[current_best_idx].copy()
        current_best_c5 = fitness[current_best_idx]
        
        if current_best_c5 < elite_c5:
            elite_c5 = current_best_c5
            elite_h = current_best_h.copy()
            
        # Selection, crossover, mutation
        new_population = []
        
        # Keep elite
        new_population.append(elite_h.copy())
        
        # Tournament selection + crossover + mutation
        while len(new_population) < population_size:
            # Tournament selection
            selected_indices = np.random.choice(len(population), tournament_size)
            selected_fitness = [fitness[i] for i in selected_indices]
            winner_index = selected_indices[np.argmin(selected_fitness)]
            
            parent1 = population[winner_index]
            
            # Crossover
            if np.random.random() < crossover_rate:
                parent2_idx = np.random.choice(len(population))
                parent2 = population[parent2_idx]
                crossover_point = np.random.randint(1, n_points)
                offspring = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                offspring = parent1.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    # Gaussian mutation
                    offspring[i] += np.random.normal(0, 0.05)
            
            # Clip and rescale to maintain constraints
            offspring = np.clip(offspring, 0, 1)
            offspring = offspring * (n_points / 2) / np.sum(offspring)
            
            new_population.append(offspring)
            
        population = new_population
        
    return (elite_h.tolist(), elite_c5, n_points)

def evaluate_h(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5
