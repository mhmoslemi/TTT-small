import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    
    # Initialize swarm
    particles = [h.copy() for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.5, 0.5, n_points) for _ in range(swarm_size)]
    pbest = [p.copy() for p in particles]
    pbest_vals = [float('inf')] * swarm_size
    gbest = h.copy()
    gbest_val = float('inf')
    
    # Function to evaluate C5
    def compute_c5(h_arr):
        correlation = np.correlate(h_arr, 1 - h_arr, mode='full')
        return np.max(correlation) * dx
    
    # Initialize evaluation
    for i in range(swarm_size):
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        val = compute_c5(particles[i])
        if val < pbest_vals[i]:
            pbest_vals[i] = val
            pbest[i] = particles[i].copy()
        if val < gbest_val:
            gbest_val = val
            gbest = particles[i].copy()
    
    best_c5 = gbest_val
    best_h = gbest.copy()
    
    # PSO Loop
    iteration = 0
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            val = compute_c5(particles[i])
            
            # Update personal best
            if val < pbest_vals[i]:
                pbest_vals[i] = val
                pbest[i] = particles[i].copy()
                
            # Update global best
            if val < gbest_val:
                gbest_val = val
                gbest = particles[i].copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Check for stagnation
        if stagnation_count > max_stagnation:
            # Perturb particles
            for i in range(int(swarm_size * 0.1)):
                idx = np.random.randint(0, swarm_size)
                particles[idx] += np.random.normal(0, 0.05, n_points)
                particles[idx] = np.clip(particles[idx], 0, 1)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                val = compute_c5(particles[idx])
                if val < pbest_vals[idx]:
                    pbest_vals[idx] = val
                    pbest[idx] = particles[idx].copy()
                if val < gbest_val:
                    gbest_val = val
                    gbest = particles[idx].copy()
            stagnation_count = 0
            
        if gbest_val < best_c5:
            best_c5 = gbest_val
            best_h = gbest.copy()
            
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
