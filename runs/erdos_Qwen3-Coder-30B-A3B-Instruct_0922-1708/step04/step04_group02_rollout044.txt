import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency  
    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Use a gradient-like approach with small perturbations
        h_new = h.copy()
        
        # Sample some indices to modify
        sample_indices = np.random.choice(n_points, size=min(10, n_points//2), replace=False)
        step = 0.01
        
        for i in sample_indices:
            if np.random.random() < 0.7:  # 70% chance to try perturbation
                direction = np.random.choice([-1, 1])
                if 0 <= h[i] + direction * step <= 1:
                    h_new[i] = h[i] + direction * step
                    
                    # Ensure sum constraint holds
                    sum_diff = np.sum(h_new) - n_points / 2
                    if abs(sum_diff) > 1e-8:
                        # Adjust other elements
                        indices = [j for j in range(n_points) if j != i]
                        diff = sum_diff / len(indices)
                        h_new[indices] = np.clip(h_new[indices] - diff, 0, 1)
                        
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    
                    # Evaluate new function
                    corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                    c5_new = np.max(corr_new) * dx
                    
                    if c5_new < c5:
                        h = h_new.copy()
                        c5 = c5_new
                        
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h.copy()

    # Final refinement of best solution
    h_final = best_h.copy()
    max_iter = 30
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient-free local search
        h_test = h_final.copy()
        
        # Try perturbations
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.005)
        if 0 <= h_test[idx] + delta <= 1:
            h_test[idx] += delta
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            c5_test = np.max(corr_test) * dx
            
            if c5_test < np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx:
                h_final = h_test
    
    return (h_final.tolist(), best_c5, n_points)
