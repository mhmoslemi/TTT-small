import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Sparse representation optimization approach
    def sparse_reconstruct(pos, h_vals):
        """Reconstruct full h from sparse representation"""
        reconstructed = np.zeros(n_points)
        pos_scaled = np.array(pos) * (n_points / 2)
        for i in range(len(h_vals)):
            start = int(pos_scaled[i])
            end = int(pos_scaled[i+1]) if i+1 < len(pos_scaled) else n_points
            reconstructed[start:end] = h_vals[i]
        return reconstructed
    
    def compute_sparse_c5(pos, h_vals):
        """Compute C5 for sparse representation"""
        h_full = sparse_reconstruct(pos, h_vals)
        correlation = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(correlation) * dx
    
    def optimize_sparse():
        # Use 10 segments for sparse representation
        m = 10
        n_vars = 2 * m - 1  # m-1 positions, m heights
        
        # Initial guess
        pos = np.linspace(0, 2, m+1)
        h_vals = np.ones(m) * (1.0 / m)
        
        def obj(vars_vec):
            # Parse variables
            pos_vec = np.array([0] + [vars_vec[i] for i in range(m-1)] + [2])
            h_vec = np.array([vars_vec[i+m-1] for i in range(m)])
            
            # Ensure bounds
            pos_vec = np.clip(pos_vec, 0, 2)
            pos_vec = np.sort(pos_vec)
            h_vec = np.clip(h_vec, 0, 1)
            
            # Normalize to unit integral
            integral = np.sum([(pos_vec[i+1] - pos_vec[i]) * h_vec[i] for i in range(m)])
            h_vec = h_vec * (n_points / 2) / integral
            
            return compute_sparse_c5(pos_vec, h_vec)
        
        # Constraints
        def constraint_pos(vars_vec):
            pos_vec = np.array([0] + [vars_vec[i] for i in range(m-1)] + [2])
            return np.diff(pos_vec)  # Should be positive
            
        def constraint_integral(vars_vec):
            pos_vec = np.array([0] + [vars_vec[i] for i in range(m-1)] + [2])
            h_vec = np.array([vars_vec[i+m-1] for i in range(m)])
            h_vec = np.clip(h_vec, 0, 1)
            integral = np.sum([(pos_vec[i+1] - pos_vec[i]) * h_vec[i] for i in range(m)])
            return integral - (n_points / 2)
        
        # Optimize
        try:
            cons = [
                {'type': 'ineq', 'fun': lambda x: constraint_pos(x)},
                {'type': 'eq', 'fun': lambda x: constraint_integral(x)}
            ]
            
            bounds = [(0.001, 1.999) if i < m-1 else (0.001, 0.999) for i in range(n_vars)]
            
            res = minimize(
                obj, 
                np.concatenate([0.2 * np.ones(m-1), 0.1 * np.ones(m)]),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-6}
            )
            
            if res.success:
                pos_final = np.array([0] + [res.x[i] for i in range(m-1)] + [2])
                h_final = np.array([res.x[i+m-1] for i in range(m)])
                h_final = np.clip(h_final, 0, 1)
                integral = np.sum([(pos_final[i+1] - pos_final[i]) * h_final[i] for i in range(m)])
                h_final = h_final * (n_points / 2) / integral
                return pos_final, h_final
        except:
            pass
        return None, None
    
    # Try several optimization approaches within budget
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlation) * dx
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Apply sparse optimization every few iterations
        if iteration % 5 == 0:
            try:
                pos, h_sparse = optimize_sparse()
                if pos is not None:
                    h = sparse_reconstruct(pos, h_sparse)
            except:
                pass
        # Perturb current solution
        else:
            h_test = h.copy()
            # Make small changes
            change_idx = np.random.choice(n_points)
            delta = np.random.normal(0, 0.01)
            h_test[change_idx] = np.clip(h_test[change_idx] + delta, 0, 1)
            
            # Rescale to unit integral
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            c5_test = np.max(corr_test) * dx
            
            if c5_test < c5_current:
                h = h_test.copy()

    # Final refinement using gradient-based optimization within remaining time
    try:
        # Refine with a direct coarse grid search for final optimization
        h_final = best_h.copy()
        current_c5 = best_c5
        
        # Test larger number of perturbations on sample points
        sample_indices = np.random.choice(n_points, size=min(20, n_points//2), replace=False)
        
        for i in sample_indices:
            for delta in [-0.02, -0.01, 0.01, 0.02]:
                h_test = h_final.copy()
                h_test[i] += delta
                
                # Ensure valid values and normalization
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                
                corr_test = np.correlate(h_test, 1 - h_test, mode='full')
                c5_test = np.max(corr_test) * dx
                
                if c5_test < current_c5:
                    h_final = h_test
                    current_c5 = c5_test
                    
        best_c5 = current_c5
        best_h = h_final
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
