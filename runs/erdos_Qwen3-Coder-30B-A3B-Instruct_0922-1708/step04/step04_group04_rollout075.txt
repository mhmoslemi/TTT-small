import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum is correct
    h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    
    # Optimization with CVXPY
    h_var = cp.Variable(n_points)
    C = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1
    ]
    
    # Add overlap constraints for all shifts
    for k in range(1, n_points):
        if time.time() - start_time > budget_s - 1:
            break
        # Compute overlap for shift k
        if k <= n_points // 2:
            # Overlap for positive shift k
            # Sum h[i] * (1 - h[i+k]) for i = 0..n_points-k-1
            overlap = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
            constraints.append(overlap <= C)
        else:
            # For negative shift -k, we use the same logic but swap sides
            # Sum h[i+k] * (1 - h[i]) for i = 0..n_points-k-1 (shifted)
            overlap = cp.sum(cp.multiply(h_var[k:], 1 - h_var[:n_points-k]))
            constraints.append(overlap <= C)
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Solve using ECOS
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h_var.value
            c5_val = C.value
            if c5_val is not None:
                h = np.array(h_sol)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                c5_val = max(np.correlate(h, 1 - h, mode='full')) * dx
    except:
        pass # fall back to current solution
    
    # Final refinement with gradient descent if needed
    h = np.array(h)
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break

        # Mutate h slightly
        h_old = h.copy()
        
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Estimate gradient with finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_test = h.copy()
            h_test[i] += eps
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            c5_test = np.max(corr_test) * dx
            grad[i] = (c5_test - c5) / eps

        # Update
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-4)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Check progress
        correlation_new = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation_new) * dx
        if c5_new < c5:
            c5 = c5_new
        else:
            h = h_old
            break

    # Final checks
    h_final = np.clip(h, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    correlation_final = np.correlate(h_final, 1 - h_final, mode='full')
    c5_final = np.max(correlation_final) * dx

    return (h_final.tolist(), float(c5_final), n_points)
