import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # LP relaxation approach
    # Variables
    h_var = Variable(n_points)
    w_var = Variable((n_points, n_points))
    C5_var = Variable()

    # Constraints
    constraints = [h_var >= 0, h_var <= 1]
    constraints.append(sum(h_var) == n_points / 2)
    
    # Add constraints for w
    for i in range(n_points):
        for k in range(n_points):
            j = i + k
            if j < n_points:
                constraints.append(w_var[i, k] <= h_var[i])
                constraints.append(w_var[i, k] <= h_var[j])
                constraints.append(w_var[i, k] >= 0)
    
    # Sum constraints for each shift k
    for k in range(n_points):
        if k == 0:
            continue
        shift_sum = 0
        for i in range(n_points - k):
            shift_sum += (h_var[i] - w_var[i, k]) * dx
        constraints.append(shift_sum <= C5_var)
    
    # Objective
    objective = Minimize(C5_var)
    
    # Solve the LP
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=1000)
        h_lp = h_var.value
        c5_lp = C5_var.value
        if h_lp is not None and c5_lp is not None:
            # Normalize
            h_lp = np.clip(h_lp, 0, 1)
            h_lp = h_lp * (n_points / 2) / np.sum(h_lp)
            # Update best solution
            if c5_lp < best_c5:
                best_h = h_lp.copy()
                best_c5 = c5_lp
    except:
        pass
    
    # If that didn't work, try gradient descent on best solution found so far
    h = best_h.copy()
    
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    def objective(h_vals):
        return compute_c5(h_vals)
    
    # Gradient descent refinement
    max_iter = 3000
    lr = 0.01
    epsilon = 1e-6
    
    grad_h = np.zeros_like(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h)
        
        # Finite difference for gradient approximation
        grad = np.zeros_like(h)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] = min(1.0, h_plus[j] + epsilon)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[j] = (c5_plus - current_c5) / epsilon
            
        # Update
        h_new = h - lr * grad
        # Re-normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        lr = max(1e-6, lr * 0.999)
        
        # Update best if improved
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
