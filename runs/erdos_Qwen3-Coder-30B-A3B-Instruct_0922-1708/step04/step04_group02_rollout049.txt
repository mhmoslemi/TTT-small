import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure bounds and normalization
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated annealing parameters
    temperature = 1.0
    cooling_factor = 0.99
    iterations_per_temp = 100
    perturb_size = 0.1
    neighborhood_size = max(1, n_points // 10)
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 2:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Perturb h
        h_new = h.copy()
        
        # Randomly select indices to perturb
        indices = np.random.choice(n_points, size=neighborhood_size, replace=False)
        
        # Perturb selected indices
        for i in indices:
            # Random change with bounded magnitude
            delta = np.random.uniform(-perturb_size, perturb_size)
            h_new[i] += delta
            
        # Clip to [0,1] bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject based on simulated annealing
        delta = new_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / temperature):
            h = h_new
            current_c5 = new_c5
            
        # Update temperature
        if iteration % iterations_per_temp == 0:
            temperature *= cooling_factor
    
    # Final refinement with local search
    refinement_iter = 500
    for _ in range(refinement_iter):
        if time.time() - start_time > budget_s - 1:
            break
        h_new = h.copy()
        i = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.05, 0.05)
        h_new[i] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            
    # Final check of the solution
    correlation_final = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(correlation_final) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
