import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO Parameters
    n_particles = 30
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest = None
    gbest_val = float('inf')
    
    for _ in range(n_particles):
        # Initialize particle with some randomness
        part = h + np.random.normal(0, 0.1, n_points)
        part = np.clip(part, 0, 1)
        part = part * (n_points / 2) / np.sum(part)
        particles.append(part.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(part.copy())
        pbest_vals.append(float('inf'))
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            particle = particles[i]
            
            # Compute fitness (C5)
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_vals[i]:
                pbest[i] = particle.copy()
                pbest_vals[i] = c5
                
            if c5 < gbest_val:
                gbest = particle.copy()
                gbest_val = c5
                
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            particle = particles[i]
            vel = velocities[i]
            
            # Update velocity
            vel = (w * vel + 
                   c1 * r1 * (pbest[i] - particle) + 
                   c2 * r2 * (gbest - particle))
            
            # Clip velocity
            vel_max = 0.2
            vel = np.clip(vel, -vel_max, vel_max)
            
            # Update position
            particle += vel
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            particles[i] = particle.copy()
            
            # Early stopping if no improvement
            if iteration % 10 == 0 and gbest_val < best_c5:
                if gbest_val < best_c5:
                    best_c5 = gbest_val
                    best_h = gbest.copy()
    
    # Final check and refinement
    if gbest is not None:
        correlation = np.correlate(gbest, 1 - gbest, mode='full')
        final_c5 = np.max(correlation) * dx
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = gbest.copy()
    
    # Try CVX refinement on best solution found
    try:
        # Define the optimization variables and constraints
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Objective: minimize the maximum overlap
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass

    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
