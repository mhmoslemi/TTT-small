import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use the provided parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a uniform distribution
        h = np.ones(n_points) * (n_points / 2) / n_points
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    def compute_c5(h_values):
        correlation = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(correlation) * dx
    
    # Sparse representation optimization
    def sparse_represent(h_sparse, n_points):
        # Convert sparse representation to full
        h_full = np.zeros(n_points)
        segment_count = len(h_sparse) // 2
        # Assumes positions are stored in first half, heights in second half
        positions = h_sparse[:segment_count]
        heights = h_sparse[segment_count:]
        
        # Linear interpolation between segments
        x = np.linspace(0, 2, n_points)
        for i in range(segment_count):
            start = int(positions[i] * n_points / 2)
            end = int(positions[i+1] * n_points / 2) if i+1 < segment_count else n_points
            h_full[start:end] = heights[i]
        
        return h_full
    
    # Objective function for sparse representation
    def sparse_obj(h_sparse):
        h_full = sparse_represent(h_sparse, n_points)
        return compute_c5(h_full)
    
    # Constraint function for sparse representation
    def sparse_constraint(h_sparse):
        h_full = sparse_represent(h_sparse, n_points)
        return np.sum(h_full) * dx - 1.0
    
    # Initialize sparse representation
    n_segments = 5
    initial_positions = np.linspace(0, 2, n_segments+1)
    initial_heights = np.ones(n_segments) * 0.5
    initial_sparse = np.concatenate([initial_positions[1:-1], initial_heights])
    
    constraints = {'type': 'eq', 'fun': sparse_constraint}
    
    # Carve out time budget for optimization
    optimization_budget = budget_s * 0.8
    optimization_start = time.time()
    
    # Run with different starting points for increasing chance of finding better solution
    for attempt in range(10):
        if time.time() - optimization_start > optimization_budget:
            break
            
        # Initialize sparse representation with random starts
        positions = np.sort(np.random.rand(n_segments-1)) * 2
        positions = np.concatenate([[0], positions, [2]])
        heights = np.random.rand(n_segments)
        heights = heights / np.sum(heights) * (n_points / 2) / (2 / n_points)
        h_sparse = np.concatenate([positions[1:-1], heights])
        
        # Optimize the sparse representation
        options = {'disp': False, 'maxiter': 100}
        result = minimize(sparse_obj, h_sparse, method='SLSQP', 
                         constraints=constraints, options=options, 
                         bounds=[(0,2)]*(n_segments-1) + [(0,1)]*n_segments)
        
        if result.success:
            h_full = sparse_represent(result.x, n_points)
            c5 = compute_c5(h_full)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_full.copy()
    
    # Fine-tune with direct optimization if we have time
    if time.time() - start_time < budget_s - 2:
        # Perform local refinement on the best solution
        h = best_h.copy()
        # Try small perturbations to improve the final result
        for _ in range(100):
            h_test = h.copy()
            # Randomly perturb a few points
            idx = np.random.randint(0, n_points)
            step = np.random.normal(0, 0.01)
            h_test[idx] = np.clip(h_test[idx] + step, 0, 1)
            # Rescale
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            c5_test = compute_c5(h_test)
            
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
    
    return (best_h.tolist(), best_c5, n_points)
