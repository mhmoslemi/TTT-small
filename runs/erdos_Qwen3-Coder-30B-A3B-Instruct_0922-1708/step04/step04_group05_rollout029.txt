import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure sum(h) * dx = 1, i.e., sum(h) = n_points / 2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Get the best initial value
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = float('inf')
    
    # PSO Parameters
    n_particles = 20
    max_iter = 200
    inertia_w = 0.9
    c1 = 1.5
    c2 = 1.5
    velocity_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for i in range(n_particles):
        # Start near initial solution with noise
        pos = h_init + np.random.normal(0, 0.05, n_points)
        pos = np.clip(pos, 0, 1)
        pos = pos * (n_points / 2) / np.sum(pos)
        particles.append(pos)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
        p_best.append(pos.copy())
        p_best_vals.append(float('inf'))
    
    # Global best
    g_best_idx = 0
    g_best = particles[0].copy()
    g_best_val = float('inf')
    
    start_time = time.time()
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        for i in range(n_particles):
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_w * velocities[i] + 
                             c1 * r1 * (p_best[i] - particles[i]) + 
                             c2 * r2 * (g_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to ensure constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate this particle
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = c5
                
                # Update global best
                if c5 < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = c5
                    
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = particles[i].copy()
        
        # Update inertia weight
        inertia_w = 0.9 - ((0.9 - 0.4) * iter_num) / max_iter
        
        # Adaptive particle moves
        if iter_num % 50 == 0 and iter_num > 0:
            for i in range(n_particles):
                # Add noise to maintain exploration
                if np.random.rand() < 0.3:
                    particles[i] += np.random.normal(0, 0.02, n_points)
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Line search refinement for top 5 particles
    top_particles = sorted([(p_best_vals[i], i) for i in range(n_particles)])[:5]
    
    for _, i in top_particles:
        if time.time() - start_time > budget_s - 5:
            break
        
        h_ref = p_best[i].copy()
        
        for _ in range(100):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Gradient estimate
            grad = np.zeros(n_points)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h_ref.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            if np.linalg.norm(grad) < 1e-6:
                break
                
            # Line search with backtracking
            lr = 0.1
            h_new = h_ref - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            # Backtracking line search
            while c5_new >= np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx and lr > 1e-6:
                lr *= 0.5
                h_new = h_ref - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr_new) * dx
            
            if c5_new < np.max(np.correlate(h_ref, 1 - h_ref, mode='full')) * dx:
                h_ref = h_new.copy()
                
                # Update if better
                corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
                c5_ref = np.max(corr_new) * dx
                if c5_ref < best_c5:
                    best_h = h_ref.copy()
                    best_c5 = c5_ref
            else:
                break
    
    # Final refinement check
    corr_best = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_best) * dx
    
    return (best_h.tolist(), final_c5, n_points)
