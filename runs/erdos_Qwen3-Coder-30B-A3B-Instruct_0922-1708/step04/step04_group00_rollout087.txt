import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Solve LP relaxation
    try:
        # Variables
        h_var = cp.Variable(n_points)
        w_vars = []
        for k in range(2 * n_points - 1):
            w_vars.append(cp.Variable(n_points))
        
        # LP constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add constraint for each shift
        for shift in range(2 * n_points - 1):
            # Shifted indices
            if shift < n_points:
                indices1 = range(shift, n_points)
                indices2 = range(0, n_points - shift)
            else:
                indices1 = range(0, n_points - (shift - n_points + 1))
                indices2 = range(shift - n_points + 1, n_points)
                
            # Ensure bounds
            for i in range(len(indices1)):
                constraints.append(w_vars[shift][i] <= h_var[indices1[i]])
                constraints.append(w_vars[shift][i] <= h_var[indices2[i]])
                constraints.append(w_vars[shift][i] >= 0)
                # Introduce c5 constraint
                if shift == 0:  # Main diagonal
                    constraints.append(w_vars[shift][i] >= h_var[indices1[i]] - 1)
        
        # Objective (minimal C5)
        C5 = cp.Variable()
        for shift in range(2 * n_points - 1):
            # Difference from h(x) and h(x+k) for each offset
            for i in range(len(indices1)):
                constraints.append(w_vars[shift][i] <= h_var[indices1[i]] + h_var[indices2[i]] - 1)
        
        # Objective is to minimize max correlation
        objective = cp.Minimize(C5)
        
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.GLPK, verbose=False)
        
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h = h_var.value
            # Normalize and clip
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            # Update best solution
            if np.max(np.correlate(h, 1 - h, mode="full")) * dx < best_c5:
                best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
                best_h = h.copy()
    except:
        pass
    
    # Gradient descent refinement
    h_refine = best_h.copy()
    max_refine_iter = 10000
    lr = 0.001
    
    for refine_iter in range(max_refine_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h_refine)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_refine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update step
        h_new = h_refine - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update learning rate
        lr = max(1e-6, lr * 0.9995)
        
        h_refine = h_new
        
        # Update best if improved
        correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_refine.copy()
    
    return (best_h.tolist(), best_c5, n_points)
