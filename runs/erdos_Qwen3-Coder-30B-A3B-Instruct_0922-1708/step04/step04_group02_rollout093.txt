import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_segments = 10
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using parent construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Interpolate to our resolution
        x_orig = np.linspace(0, 2, len(h_init))
        x_target = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_target, x_orig, h_init)
        # Convert to sparse representation
        h_sparse = np.zeros(n_segments)
        pos_sparse = np.linspace(0, 2, n_segments + 1)
        for i in range(n_segments):
            segment_size = n_points // n_segments
            start = i * segment_size
            end = (i+1) * segment_size if i < n_segments - 1 else n_points
            h_sparse[i] = np.mean(h_interp[start:end])
    else:
        # Initialize with uniform distribution
        h_sparse = np.ones(n_segments) * (1.0 / n_segments)
        pos_sparse = np.linspace(0, 2, n_segments + 1)
    
    # Adjust to satisfy constraint sum(h_sp * width) = 1
    widths = np.diff(pos_sparse)
    current_sum = np.sum(h_sparse * widths)
    h_sparse = h_sparse * (1.0 / current_sum)
    
    # Helper function to compute h(x) for any x given sparse representation
    def construct_h(pos, h_vals):
        def h(x):
            # Find the segment containing x
            idx = np.searchsorted(pos, x, side="right") - 1
            idx = max(0, min(idx, len(h_vals) - 1))
            return h_vals[idx]
        return h
    
    # Critical lags to evaluate
    def get_critical_lags(pos, h_vals):
        # All positive and negative lags that could be critical
        lags = set()
        for i in range(len(pos)-1):
            for j in range(len(pos)-1):
                # Lag between segment boundaries
                lags.add(pos[j+1] - pos[i])
                lags.add(pos[i] - pos[j+1])
        lags = sorted(list(lags))
        # Be sure to include 0 and some boundary points
        if 0 not in lags:
            lags.insert(0, 0)
        return np.array(lags)
    
    # Objective function to minimize
    def compute_c5(pos_vals, h_vals):
        # Recreate the step function h
        h_func = construct_h(pos_vals, h_vals)
        
        # Compute all critical lags
        lags = get_critical_lags(pos_vals, h_vals)
        max_overlap = 0
        
        for k in lags:
            # Integral = sum over segments of (h(x) * (1 - h(x+k)) * width)
            integral = 0
            for i in range(len(pos_vals)-1):
                # Region [pos[i], pos[i+1]]
                x_start, x_end = pos_vals[i], pos_vals[i+1]
                h_val = h_vals[i]
                
                # Shifted interval [x_start+k, x_end+k] 
                if x_start + k >= 2 or x_end + k < 0:
                    # No overlap
                    integral += h_val * 1.0 * (x_end - x_start)
                    continue
                
                # Determine overlap interval
                a = max(0, x_start + k)
                b = min(2, x_end + k)
                
                if a >= b:
                    integral += h_val * 1.0 * (x_end - x_start)
                    continue
                
                # In overlap region
                overlap_len = b - a
                
                # Compute h(x+k) for this overlap
                k_start = max(0, a - k)
                k_end = min(2, b - k)
                
                # Find the segment in k-domain
                idx_k = np.searchsorted(pos_vals, k_start, side="right") - 1
                idx_k = max(0, min(idx_k, len(h_vals) - 1))
                
                h_k_val = h_vals[idx_k]
                integral += h_val * (1 - h_k_val) * overlap_len
                
            max_overlap = max(max_overlap, integral)
        
        return max_overlap
    
    # Build optimization variables
    def unpack_vars(vars):
        pos = np.array(vars[:n_segments-1])
        h = np.array(vars[n_segments-1:])
        # Ensure pos strictly increasing
        pos = np.concatenate([[0], np.sort(pos), [2]])
        # Ensure h in [0,1]
        h = np.clip(h, 0, 1)
        # Normalize so sum(width * h) = 1
        widths = np.diff(pos)
        norm_factor = np.sum(widths * h)
        if norm_factor > 0:
            h = h / norm_factor
        return pos, h
    
    def objective(vars):
        pos, h = unpack_vars(vars)
        return compute_c5(pos, h)
    
    # Initial guess processing
    x0 = np.concatenate([pos_sparse[1:-1], h_sparse])  # Remove first and last positions
    bounds = [(0, 2)] * (n_segments - 1) + [(0, 1)] * n_segments
    
    best_c5 = float('inf')
    best_solution = x0.copy()
    
    # Run optimization with multiple starts
    start_time = time.time()
    
    for trial in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Slightly randomize initial values
        x_trial = x0.copy()
        x_trial[:n_segments-1] += np.random.normal(0, 0.02, n_segments-1)
        x_trial[n_segments-1:] += np.random.normal(0, 0.02, n_segments)
        x_trial[n_segments-1:] = np.clip(x_trial[n_segments-1:], 0, 1)
        
        # Optimize with SLSQP
        try:
            res = minimize(objective, x_trial, method='SLSQP', 
                          bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-8})
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_solution = res.x.copy()
        except:
            continue
    
    # Extract best solution
    final_pos, final_h = unpack_vars(best_solution)
    
    # Reconstruct h at full resolution for reporting
    h_full = np.zeros(n_points)
    for i in range(len(final_pos)-1):
        start_idx = int((final_pos[i]/2) * n_points)
        end_idx = int((final_pos[i+1]/2) * n_points)
        value = final_h[i]
        h_full[start_idx:end_idx] = value
    
    # Make sure sum is correct
    h_full = h_full / np.sum(h_full) * (n_points/2)
    
    # Final C5 calculation
    corr = np.correlate(h_full, 1-h_full, mode='full')
    c5 = np.max(corr) * dx
    
    return (h_full.tolist(), c5, n_points)
