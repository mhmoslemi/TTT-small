import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation approach
    while time.time() - start_time < budget_s * 0.4:
        try:
            # Formulate the convex relaxation
            h_relaxed = cp.Variable(n_points)
            C = cp.Variable()
            
            # Constraints
            constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points / 2]
            
            # Create overlap constraint for each lag
            for k in range(1, n_points):
                # This part is tricky to express exactly in CVXPY due to circular convolution
                # For now, we'll use a simplified version that allows the solver to work with constraints
                # and refine with local search
                pass
            
            # Objective
            objective = cp.Minimize(C)
            
            # Solve convex relaxation
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed_sol = h_relaxed.value
                
                # Integer rounding heuristic
                threshold = np.median(h_relaxed_sol)
                h_candidate = (h_relaxed_sol > threshold).astype(float)
                # Ensure exact sum constraint
                current_sum = np.sum(h_candidate)
                diff = n_points / 2 - current_sum
                if abs(diff) > 1e-5:
                    # Find indices to flip
                    indices = np.argsort(np.abs(h_relaxed_sol - threshold))
                    flip_indices = indices[:int(abs(diff))]
                    if diff > 0:
                        # Add ones
                        h_candidate[flip_indices] = 1.0
                    else:
                        # Remove ones
                        h_candidate[flip_indices] = 0.0
                        
                # Normalize
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                
                # Evaluate
                correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(correlation) * dx
                
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
                    
        except:
            pass
            
    # Local search neighborhood improvement
    iteration = 0
    while time.time() - start_time < budget_s * 0.7 and iteration < 200:
        iteration += 1
        
        # Try small perturbations around best solution
        h_new = best_h.copy()
        mutated = False
        
        # Try a few random modifications
        for _ in range(5):
            idx = np.random.randint(n_points)
            # Try to make small change
            change = np.random.uniform(-0.02, 0.02)
            if 0 <= h_new[idx] + change <= 1:
                h_new[idx] += change
                mutated = True
                
        if mutated:
            # Ensure sum constraint
            old_sum = np.sum(h_new)
            if abs(old_sum - n_points / 2) > 1e-6:
                # Distribute across other indices
                diff = (n_points / 2 - old_sum) / (n_points - 1)
                for i in range(n_points):
                    if i != idx:
                        h_new[i] += diff
                        
            h_new = np.clip(h_new, 0, 1)
            
            # Compute new C5
            correlation = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(correlation) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
    # Final refinement - try some additional search techniques
    if time.time() - start_time < budget_s * 0.95:
        # Binary search type refinement
        h_final = best_h.copy()
        for i in range(50):
            best_local_improvement = False
            for j in range(n_points):
                for delta in [-0.01, 0.01]:
                    h_test = h_final.copy()
                    if 0 <= h_test[j] + delta <= 1:
                        h_test[j] += delta
                        # Normalize sum
                        h_test = h_test * (n_points / 2) / np.sum(h_test)
                        correlation = np.correlate(h_test, 1 - h_test, mode='full')
                        c5_test = np.max(correlation) * dx
                        if c5_test < best_c5:
                            best_c5 = c5_test
                            h_final = h_test.copy()
                            best_local_improvement = True
                            
            if not best_local_improvement:
                break
                
        best_h = h_final
        
    # Final validation
    try:
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        c5_final = np.max(correlation) * dx
        if c5_final < best_c5:
            best_c5 = c5_final
    except:
        pass
        
    return (best_h.tolist(), best_c5, n_points)
