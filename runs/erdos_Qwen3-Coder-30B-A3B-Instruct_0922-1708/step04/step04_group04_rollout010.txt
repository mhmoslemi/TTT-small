import numpy as np
import time
from heapq import heappush, heappop
from scipy.signal import convolve

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Helper to compute overlaps efficiently
    def compute_overlaps(h_vec):
        # Compute all overlaps using convolution
        corr = convolve(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Main optimization loop
    window_width = 5  
    perturbation_scale = 0.02
    improvement_threshold = 1e-6
    stall_count = 0
    max_stall = 50
    
    # Initial computation
    current_c5 = compute_overlaps(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 10000:
        iteration += 1
        # Compute all overlaps for prioritization
        overlaps = []
        h_reversed = 1 - h
        
        # Compute convolution for all shifts
        corr_full = convolve(h, h_reversed, mode='full')
        
        # Compute for each valid shift
        shift_range = range(-(n_points-1), n_points)
        for k in shift_range:
            if k == 0:
                continue  # Skip zero shift
            # Find the contribution from shift k
            corr_k = corr_full[n_points-1+k] * dx
            if corr_k > 0:  # Only consider non-trivial overlaps
                overlaps.append((-corr_k, k))  # Negative for max-heap
        
        if not overlaps:
            break
        
        # Convert to heap
        import heapq
        heapq.heapify(overlaps)
        
        # Get shift with maximum overlap
        neg_overlap_val, best_k = heapq.heappop(overlaps)
        overlap_val = -neg_overlap_val
        
        if overlap_val < improvement_threshold:
            if stall_count > max_stall:
                # Apply random perturbation
                perturb_idx = np.random.randint(0, n_points)
                h[perturb_idx] = np.clip(h[perturb_idx] + np.random.normal(0, perturbation_scale), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count = 0
            
            # Select region around k for adjustment
            # Identify start and end indices
            if best_k < 0:
                # Shift left
                start_idx = 0
                end_idx = n_points + best_k
            else:
                # Shift right
                start_idx = best_k
                end_idx = n_points
                
            # Adjust h in the effective overlap region
            adjustment_magnitude = 0.01
            if best_k < 0:
                for i in range(start_idx, min(end_idx, start_idx + window_width)):
                    h[i] = np.clip(h[i] - adjustment_magnitude, 0, 1)
            else:
                for i in range(max(0, end_idx - window_width), end_idx):
                    h[i] = np.clip(h[i] - adjustment_magnitude, 0, 1)
            
            # Rescale
            h = h * (n_points / 2) / np.sum(h)
        
        # Check if new solution is better
        new_c5 = compute_overlaps(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            stall_count = 0
        
        # Occasionally refresh overlaps to prevent getting stuck on local minima
        if iteration % 50 == 0:
            # Apply small random perturbation
            perturb_mask = np.random.random(n_points) < 0.2
            perturbation = np.random.normal(0, perturbation_scale, n_points)
            h[perturb_mask] = np.clip(h[perturb_mask] + perturbation[perturb_mask], 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
