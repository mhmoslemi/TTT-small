import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Solve relaxation using CVXPY
    # Decision variables
    h_vars = Variable(n_points)
    w_vars = Variable((n_points, n_points))
    c5_var = Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        sum(h_vars) == n_points / 2
    ]
    
    # Add auxiliary constraints
    for i in range(n_points):
        for k in range(n_points):
            if i + k < n_points:
                # w[i, k] <= h[i]
                constraints.append(w_vars[i, k] <= h_vars[i])
                # w[i, k] <= h[i+k] 
                constraints.append(w_vars[i, k] <= h_vars[i+k])
                # w[i, k] >= 0
                constraints.append(w_vars[i, k] >= 0)
                
            # Link w and h for correlation computation
            # h[i] - w[i,k] <= C5
            if i + k < n_points:
                constraints.append(h_vars[i] - w_vars[i, k] <= c5_var)
    
    # Objective
    objective = Minimize(c5_var)
    
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False, max_iters=1000)
        h_solution = h_vars.value
        if h_solution is not None:
            h = np.clip(h_solution, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    # Refinement with gradient descent
    max_refine_iter = 1000
    lr = 0.01
    
    for refine_iter in range(max_refine_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update learning rate
        lr = max(1e-6, lr * 0.999)
        
        h = h_new
        
    # Final objective computation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
