from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Use a combination of global search and refinement
    # Start with constraint-aware gradient descent from initial
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Gradient descent with constraint projection
    max_iter = 1000
    lr = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute approximate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update with projection
        h_new = h - lr * grad
        h_new = project_h(h_new)
        
        # Reduce learning rate
        lr = max(1e-6, lr * 0.999)
        h = h_new
        
    # Try random restarts to escape local minima
    n_restarts = 20
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        h_restart = h + np.random.normal(0, 0.05, n_points)
        h_restart = project_h(h_restart)
        
        # Local refinement
        for i in range(500):
            if time.time() - start_time > budget_s - 1:
                break
            c5 = compute_c5(h_restart)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_restart.copy()
                
            grad = np.zeros_like(h_restart)

                c5_plus = compute_c5(h_plus)
                grad[j] = (c5_plus - c5) / eps
                
            h_restart_new = h_restart - 0.01 * grad
            h_restart_new = project_h(h_restart_new)
            h_restart = h_restart_new
    
    return (best_h.tolist(), best_c5, n_points)
