import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to correct integral
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    start_time = time.time()
    
    # Iterative convex optimization approach
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Since CVXPY is exact, we'll call it manually once with constraints
    # But we'll use a modified approach: run multiple optimizations
    # with different starting points and hybridize
    
    for iter_num in range(20):  # 20 iterations should be enough
        if time.time() - start_time > budget_s - 5:
            break
            
        # Define h and C as CVXPY variables
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            C >= 0
        ]
        
        # Add the overlap constraints
        for k in range(1, n_points):
            # For each shift, compute the overlap
            # We'll do it manually with sum for now
            # Actually simplify the cross-correlation constraint as done in the original:
            # Each term in the correlation is h[i] * (1 - h[i+k])
            # "full" mode: shift k = 0 to n_points-1 (which python actually considers as -n_points+1 to n_points-1)
            # But in our case, since we're being careful, let's compute the full correlation terms
            # Let's make sure we handle all valid indices: i in [0,n_points-1] and i+k in [0,n_points-1]
            corr_expr = 0
            for i in range(n_points):
                if i + k < n_points:
                    corr_expr += h_var[i] * (1 - h_var[i + k])
            constraints.append(corr_expr <= C)
        
        # Objective
        objective = cp.Minimize(C)
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, feastol=1e-5)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                # Clip and rescale to make sure it's exactly feasible
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Compute actual C5
                correlation = np.correlate(h_new, 1 - h_new, mode='full')
                c5 = np.max(correlation) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    
        except Exception:
            pass  # Skip in case of solver issues
        
        # Perturb the current solution for next optimization:
        # create a new start point by small mutation
        h = best_h.copy()
        # Add a small random perturbation
        pert = np.random.normal(0, 0.01, n_points)
        h_perturbed = h + pert
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        h = h_perturbed.copy()

    # Final refinement with gradient descent if still time
    h = best_h.copy()
    for iter in range(100):  # Limit to 100 iter
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Gradient estimate using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-4)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update if better
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Check for convergence
        if np.abs(c5 - c5_new) < 1e-10:
            break
    
    return (best_h.tolist(), best_c5, n_points)
