import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation approach
    # Phase 1: Solve relaxation using CVXPY
    # We'll minimize sum of all correlations (convex) to get initial guess
    print("Starting convex relaxation phase...")
    try:
        # Create variables
        h_var = cp.Variable(n_points)
        
        # Objective: sum of correlations (approximation)
        # This is an over-simplified proxy for the actual objective
        # but provides a reasonable initial point
        obj = cp.Minimize(cp.sum_squares(h_var))
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Solve the problem
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_init = h_var.value
            h = np.clip(h_init, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    except Exception as e:
        print(f"CVX relaxation failed: {e}")
        pass
    
    # Phase 2: Projected gradient descent refinement
    print("Starting refinement phase...")
    
    # Function to compute correlation and C5
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Project function to maintain constraint
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Gradient estimation via finite differences
    def estimate_gradient(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        base_c5 = compute_c5(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Main refinement loop
    max_no_improvement = 50
    no_improvement_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current c5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Estimate gradient with finite differences
        grad = estimate_gradient(h)
        
        # Adaptive learning rate
        lr = 0.01 / (1 + 0.001 * np.linalg.norm(grad))
        
        # Update step
        h_new = h - lr * grad
        h_new = project_h(h_new)
        
        # Check for improvement
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5:
            h = h_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    # Ensure the best solution is returned
    return (best_h.tolist(), final_c5, n_points)
