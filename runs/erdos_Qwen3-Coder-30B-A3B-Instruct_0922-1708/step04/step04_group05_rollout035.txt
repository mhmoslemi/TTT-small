import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary by thresholding, set count to exactly n_points/2
    h = (h_init >= 0.5).astype(float)
    ones_count = np.sum(h)
    if ones_count > n_points / 2:
        # Find indices of 1s and flip some to 0
        one_indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(one_indices, int(ones_count - n_points / 2), replace=False)
        h[flip_indices] = 0
    elif ones_count < n_points / 2:
        # Find indices of 0s and flip some to 1
        zero_indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(zero_indices, int(n_points / 2 - ones_count), replace=False)
        h[flip_indices] = 1
    
    # Evaluate initial C5
    corr_full = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr_full) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    T = 1.0  # Initial temperature
    cooling_rate = 0.999
    perturbation_freq = 1000
    stagnation_limit = 1000
    stagnation_count = 0
    
    # Simulated annealing
    for i in range(10000):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Make a move: flip a random bit
        idx = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[idx] = 1 - h_new[idx]
        
        # Project to ensure exactly n_points/2 ones
        ones_count_new = np.sum(h_new)
        if ones_count_new > n_points / 2:
            # Flip some 1s to 0s
            one_indices = np.where(h_new == 1)[0]
            flip_indices = np.random.choice(one_indices, int(ones_count_new - n_points / 2), replace=False)
            h_new[flip_indices] = 0
        elif ones_count_new < n_points / 2:
            # Flip some 0s to 1s
            zero_indices = np.where(h_new == 0)[0]
            flip_indices = np.random.choice(zero_indices, int(n_points / 2 - ones_count_new), replace=False)
            h_new[flip_indices] = 1
        
        # Compute new C5
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            stagnation_count = 0
        else:
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                h = h_new
                current_c5 = new_c5
            stagnation_count += 1
        
        T *= cooling_rate
        
        # Random perturbation occasionally
        if i % perturbation_freq == 0 and i > 0:
            # Reset 5% of values randomly
            reset_count = int(n_points * 0.05)
            reset_indices = np.random.choice(n_points, reset_count, replace=False)
            for idx in reset_indices:
                h[idx] = np.random.rand() >= 0.5
            # Project to feasibility
            ones_count_h = np.sum(h)
            if ones_count_h > n_points / 2:
                one_indices = np.where(h == 1)[0]
                flip_indices = np.random.choice(one_indices, int(ones_count_h - n_points / 2), replace=False)
                h[flip_indices] = 0
            elif ones_count_h < n_points / 2:
                zero_indices = np.where(h == 0)[0]
                flip_indices = np.random.choice(zero_indices, int(n_points / 2 - ones_count_h), replace=False)
                h[flip_indices] = 1
        
        if stagnation_count > stagnation_limit:
            break
    
    # Final eval for best solution found
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
