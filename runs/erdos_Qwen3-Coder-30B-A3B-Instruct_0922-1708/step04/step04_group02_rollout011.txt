import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    swarm_size = 100
    max_iter = 900
    ω = 0.7
    φ_p = 2.0
    φ_g = 2.0
    local_search_freq = 50
    rejuvenation_freq = 200
    resolution_steps = [50, 100, 200]
    
    dx = 2.0 / n_points
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Generate first particle as initial h (or scaled uniform)
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Generate remaining particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    pbests = [p.copy() for p in particles]
    pbest_vals = [float('inf')] * swarm_size
    
    # Initialize global best
    gbest_idx = 0
    gbest = particles[0].copy()
    gbest_val = float('inf')
    
    for i in range(swarm_size):
        h = particles[i]
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < pbest_vals[i]:
            pbest_vals[i] = c5
            pbests[i] = h.copy()
        if c5 < gbest_val:
            gbest_val = c5
            gbest = h.copy()
            gbest_idx = i
    
    # PSO Main Loop
    iteration = 0
    last_improvement = 0
    current_res = 0
    best_improvement = gbest_val
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Evaluate and update personal bests
        for i in range(swarm_size):
            h = particles[i]
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < pbest_vals[i]:
                pbest_vals[i] = c5
                pbests[i] = h.copy()
            
            if c5 < gbest_val:
                gbest_val = c5
                gbest = h.copy()
                last_improvement = iteration
                if gbest_val < best_improvement:
                    best_improvement = gbest_val
        
        # Update velocity and position for all particles
        r1, r2 = np.random.rand(2)
        for i in range(swarm_size):
            v = velocities[i]
            h = particles[i]
            
            # Update velocity
            v = ω * v + φ_p * r1 * (pbests[i] - h) + φ_g * r2 * (gbest - h)
            
            # Update position
            h_new = h + v
            
            # Repair
            h_new = np.clip(h_new, 0, 1)
            h_sum = np.sum(h_new)
            target_sum = n_points / 2
            
            if abs(h_sum - target_sum) > 1e-8:
                if h_sum > target_sum:
                    # Reduce largest elements
                    sorted_indices = np.argsort(h_new)[::-1]
                    excess = h_sum - target_sum
                    for idx in sorted_indices:
                        if excess <= 0:
                            break
                        reduce_amount = min(excess, h_new[idx])
                        h_new[idx] -= reduce_amount
                        excess -= reduce_amount
                else:
                    # Increase smallest elements
                    sorted_indices = np.argsort(h_new)
                    deficit = target_sum - h_sum
                    for idx in sorted_indices:
                        if deficit <= 0:
                            break
                        increase_amount = min(deficit, 1 - h_new[idx])
                        h_new[idx] += increase_amount
                        deficit -= increase_amount
            
            # Rescale to exact sum
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            velocities[i] = v
        
        # Local search every few iterations
        if iteration % local_search_freq == 0:
            # Select top 10% for local search
            top_indices = np.argsort(pbest_vals)[:swarm_size // 10]
            for i in top_indices:
                p = pbests[i].copy()
                # Simple gradient descent
                for _ in range(10):
                    corr = correlate(p, 1 - p, mode='full')
                    c5 = np.max(corr) * dx
                    
                    # Simple gradient approximation
                    grad = np.zeros_like(p)
                    eps = 1e-5
                    for j in range(n_points):
                        p_plus = p.copy()
                        p_plus[j] += eps
                        if p_plus[j] > 1:
                            p_plus[j] = 1
                        if p_plus[j] < 0:
                            p_plus[j] = 0
                        p_plus = p_plus * (n_points / 2) / np.sum(p_plus)
                        corr_plus = correlate(p_plus, 1 - p_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - c5) / eps
                    
                    # Update
                    p = p - 0.001 * grad
                    p = np.clip(p, 0, 1)
                    p = p * (n_points / 2) / np.sum(p)
                    if np.allclose(p, pbests[i], atol=1e-5):
                        break
                pbests[i] = p
                corr = correlate(p, 1 - p, mode='full')
                c5 = np.max(corr) * dx
                if c5 < pbest_vals[i]:
                    pbest_vals[i] = c5
        
        # Rejuvenation
        if iteration % rejuvenation_freq == 0:
            # Replace worst 10%
            worst_indices = np.argsort(pbest_vals)[-swarm_size // 10:]
            for i in worst_indices:
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[i] = h
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
    
    # Final pass
    final_candidates = []
    for i in range(swarm_size):
        corr = correlate(pbests[i], 1 - pbests[i], mode='full')
        c5 = np.max(corr) * dx
        final_candidates.append((pbests[i].copy(), c5))
    
    final_candidates.sort(key=lambda x: x[1])
    best_h = final_candidates[0][0]
    best_c5 = final_candidates[0][1]
    
    return (best_h.tolist(), best_c5, n_points)
