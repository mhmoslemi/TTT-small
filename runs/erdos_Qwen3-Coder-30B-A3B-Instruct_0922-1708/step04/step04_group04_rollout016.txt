import numpy as np
from scipy.stats import qmc
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Latin Hypercube Sampling approach with gradient refinement
    lhs_samples = 50
    max_grad_steps = 500
    
    # Generate Latin Hypercube samples
    sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs = sampler.random(n=lhs_samples)
    
    # Rescale each LHS sample to meet sum constraint
    h_candidates = []
    for i in range(lhs_samples):
        sample = lhs[i]
        # Transform to [0,1] and rescale to sum to n_points/2
        h_sample = sample * (n_points / 2) / np.sum(sample)
        h_sample = np.clip(h_sample, 0, 1)
        h_candidates.append(h_sample)
    
    # Also include the initial solution for diversity
    h_candidates.append(h.copy())
    
    # helper function for computing C5
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # gradient computation using finite differences
    def compute_gradient(h_vec, eps=1e-5):
        grad = np.zeros_like(h_vec)
        c5_base = compute_c5(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            c5_plus = compute_c5(h_plus)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    for i, h_current in enumerate(h_candidates):
        if time.time() - start_time > budget_s - 1:
            break
        
        h_refined = h_current.copy()
        c5_refined = compute_c5(h_refined)
        
        # Gradient descent refinement
        lr = 0.01
        prev_c5 = float('inf')
        no_improvement_count = 0
        
        for step in range(max_grad_steps):
            if time.time() - start_time > budget_s - 1:
                break
                
            grad = compute_gradient(h_refined)
            h_new = h_refined - lr * grad
            
            # Enforce bounds and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = compute_c5(h_new)
            
            if c5_new < c5_refined:
                h_refined = h_new
                c5_refined = c5_new
                no_improvement_count = 0
                if lr < 0.0001:  # Reduce learning rate if too small
                    lr *= 0.99
            else:
                no_improvement_count += 1
                if no_improvement_count > 20:
                    lr *= 0.5  # Reduce learning rate when not improving
                    no_improvement_count = 0
                if lr < 1e-6:
                    break
            
            if abs(c5_new - prev_c5) < 1e-8:
                break
            prev_c5 = c5_new
            
        # Update global best
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Final validation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
