import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate diverse initial candidates
    candidates = [h_init]
    for _ in range(19):
        pert = np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_init + pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        candidates.append(h_pert)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    # Gradient-based local optimization
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_gradient(h):
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_eps = h.copy()
            h_eps[i] += eps
            h_eps = project_feasible(h_eps)
            c5_plus = compute_c5(h_eps)
            
            h_neg = h.copy()
            h_neg[i] -= eps
            h_neg = project_feasible(h_neg)
            c5_neg = compute_c5(h_neg)
            
            grad[i] = (c5_plus - c5_neg) / (2 * eps)
        return grad
    
    candidate_count = 0
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all current candidates
        c5_values = [compute_c5(h) for h in candidates]
        
        # Track best solution
        min_idx = np.argmin(c5_values)
        if c5_values[min_idx] < best_c5:
            best_c5 = c5_values[min_idx]
            best_h = candidates[min_idx].copy()
        
        # Generate new candidates using gradient descent on worst
        worst_idx = np.argmax(c5_values)
        h_current = candidates[worst_idx]
        
        # Gradient descent step
        lr = 0.01 / (1 + iteration * 0.001)
        grad = compute_gradient(h_current)
        h_new = h_current - lr * grad
        
        h_new = project_feasible(h_new)
        
        # Add new candidate
        if candidate_count < 20:
            candidates.append(h_new)
            candidate_count += 1
        else:
            # Replace worst candidate
            worst_new = np.argmax([compute_c5(h) for h in candidates])
            candidates[worst_new] = h_new
        
        # Occasionally add random candidates
        if iteration % 100 == 0:
            for _ in range(3):
                new_h = np.random.rand(n_points)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                candidates.append(new_h)
                candidate_count += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
