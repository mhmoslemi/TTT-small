import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # First phase: Convex relaxation using CVXPY
    print("Starting convex relaxation phase...")
    try:
        # Formulate the convex relaxation
        h_var = cp.Variable(n_points)
        # We want to minimize sum of overlaps, not the max
        # But for relaxation, we use a simpler approach
        # Minimize trace of matrix multiplication
        
        # Create a simple relaxation using the sum of overlaps
        # Not exactly what we want, but we'll refine it later
        
        objective = cp.Minimize(cp.sum(h_var))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h = h_var.value
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            print("Initial solution from convex relaxation obtained.")
        else:
            print("Convex relaxation failed. Using uniform distribution.")
            
    except Exception as e:
        print(f"Convex relaxation failed: {e}")
        pass
    
    # Second phase: Projected gradient descent
    print("Starting gradient descent refinement phase...")
    
    def evaluate(h_vals):
        h = np.array(h_vals)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def gradient(h_vals, eps=1e-4):
        h = np.array(h_vals)
        base_val = evaluate(h)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            grad[i] = (evaluate(h_plus) - base_val) / eps
        return grad
    
    iteration = 0
    max_no_improvement = 100
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        # Gradient descent step
        if iteration % 20 == 0:
            # Occasionally recompute gradient with a smaller epsilon
            eps = 1e-5
        else:
            eps = 1e-4
        
        try:
            grad = gradient(h, eps)
        except Exception as e:
            # Fallback to finite difference approximation
            h_plus = h.copy()
            h_plus[0] += 1e-4
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            h_minus = h.copy()
            h_minus[0] -= 1e-4
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad = np.zeros_like(h)
            grad[0] = (c5_plus - c5_minus) / (2e-4)
            grad[0] = np.clip(grad[0], -1e5, 1e5)  # Clip for stability
            
        # Update with adaptive learning rate
        lr = 0.01 / (1 + iteration * 1e-3)
        h_new = h - lr * grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            improved = True
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        h = h_new
        
        if no_improvement_count >= max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
