import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Configuration
    n_segments = 8
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        # Interpolate to target resolution
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
    else:
        h = np.ones(n_points) * 0.5
        
    # Sparse representation: segments [start_pos, height] with fixed ends
    pos_init = np.linspace(0, 2, n_segments + 1)
    h_init = np.ones(n_segments) * (1.0 / n_segments)
    
    # Rescale to satisfy integral constraint
    integral = np.sum(h_init * np.diff(pos_init))
    h_init = h_init * (1.0 / integral) * 0.5  # Target integral = 1
    
    # Optimization variables: positions (excluding first and last) and heights
    variables = np.concatenate([pos_init[1:-1], h_init])
    
    best_c5 = float('inf')
    best_vars = variables.copy()
    
    start_time = time.time()
    
    # Critical lag identification - all lag values for correlation
    lags = range(1, n_points)
    
    def compute_overlap(pos, h_vals):
        # Create full h function from sparse representation
        h_full = np.zeros(n_points)
        for i in range(len(pos)-1):
            start_idx = int(pos[i] / dx)
            end_idx = int(pos[i+1] / dx)
            h_full[start_idx:end_idx] = h_vals[i]
        
        # Compute correlation
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def objective(vars):
        pos = np.concatenate([[0], vars[:n_segments-1], [2]])
        h_vals = vars[n_segments-1:]
        
        # Ensure bounds
        h_vals = np.clip(h_vals, 0, 1)
        
        # Ensure integral constraint via rescaling
        integral = np.sum(h_vals * np.diff(pos))
        if integral > 0:
            h_vals = h_vals * (1.0 / integral)
        
        return compute_overlap(pos, h_vals)
    
    def constraint(vars):
        pos = np.concatenate([[0], vars[:n_segments-1], [2]])
        h_vals = vars[n_segments-1:]
        # Pos must be increasing
        return np.diff(pos)
    
    # Optimization
    while time.time() - start_time < budget_s - 1:
        try:
            # Try multiple starting points
            for _ in range(5):
                # Random initialization
                pos_rand = np.random.rand(n_segments-1) * 2
                pos_rand = np.sort(pos_rand)
                pos_rand = np.concatenate([[0], pos_rand, [2]])
                h_rand = np.random.rand(n_segments)
                h_rand = np.clip(h_rand, 0, 1)
                integral = np.sum(h_rand * np.diff(pos_rand))
                if integral > 0:
                    h_rand = h_rand * (1.0 / integral) * 0.5
                
                vars_initial = np.concatenate([pos_rand[1:-1], h_rand])
                
                # Optimization
                result = minimize(objective, vars_initial, method='SLSQP', 
                                bounds=[(0, 2)] * (n_segments-1) + [(0, 1)] * n_segments,
                                constraints={'type': 'ineq', 'fun': lambda x: np.diff(np.concatenate([[0], x[:n_segments-1], [2]]))},
                                options={'maxiter': 50})
                
                if result.success:
                    c5 = objective(result.x)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_vars = result.x.copy()
        except:
            pass
        
        # Local refinement step
        try:
            pos = np.concatenate([[0], best_vars[:n_segments-1], [2]])
            h_vals = np.clip(best_vars[n_segments-1:], 0, 1)
            integral = np.sum(h_vals * np.diff(pos))
            if integral > 0:
                h_vals = h_vals * (1.0 / integral) * 0.5
            best_vars = np.concatenate([pos[1:-1], h_vals])
        except:
            pass
    
    # Final refinement with full correlation
    pos_final = np.concatenate([[0], best_vars[:n_segments-1], [2]])
    h_vals_final = np.clip(best_vars[n_segments-1:], 0, 1)
    integral = np.sum(h_vals_final * np.diff(pos_final))
    if integral > 0:
        h_vals_final = h_vals_final * (1.0 / integral)
        
    # Create full h vector
    h_full = np.zeros(n_points)
    for i in range(len(pos_final)-1):
        start_idx = int(pos_final[i] / dx)
        end_idx = int(pos_final[i+1] / dx)
        h_full[start_idx:end_idx] = h_vals_final[i]
    
    # Compute final C5
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_full.tolist(), final_c5, n_points)
