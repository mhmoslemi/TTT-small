import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Guided random search
    perturbation_scale = 0.1
    iterations = 1000
    
    for i in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate perturbation
        perturbation = np.random.normal(0, perturbation_scale, n_points)
        h_perturbed = h + perturbation
        
        # Clip and rescale
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Compute the overlap integral
        correlation = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()
            
        # Update perturbation scale
        perturbation_scale *= 0.99
        
        if i % 50 == 0 and perturbation_scale < 1e-6:
            perturbation_scale = 1e-6
    
    # Gradient descent refinement
    iterations = 500
    lr = 0.01
    eps = 1e-4
    
    for i in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute approximate gradient using finite differences
        grad = np.zeros_like(h)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
            
        # Update h using gradient descent
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute the overlap integral
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
