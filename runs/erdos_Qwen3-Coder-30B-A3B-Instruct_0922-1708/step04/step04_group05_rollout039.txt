import time

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation with exactly n_points/2 ones
    h = (h_init >= 0.5).astype(int)
    count_ones = np.sum(h)
    if count_ones > n_points // 2:
        # Need to flip some ones to zeros
        zero_indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(zero_indices, count_ones - n_points // 2, replace=False)
        h[flip_indices] = 0
    elif count_ones < n_points // 2:
        # Need to flip some zeros to ones
        one_indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(one_indices, n_points // 2 - count_ones, replace=False)
        h[flip_indices] = 1
    
    # Ensure exactly n_points/2 ones
    h = h.astype(float)
    h = np.round(h).astype(int)
    current_count = np.sum(h)
    if current_count > n_points // 2:
        diff = current_count - n_points // 2
        one_indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(one_indices, diff, replace=False)
        h[flip_indices] = 0
    elif current_count < n_points // 2:
        diff = n_points // 2 - current_count
        zero_indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(zero_indices, diff, replace=False)
        h[flip_indices] = 1
    
    # Ensure feasibility: exactly n_points/2 ones
    h = h.astype(float)
    h = np.round(h).astype(int)
    
    # Helper functions
    def compute_correlation(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h):
        """Ensure h has exactly n_points/2 ones."""
        count = np.sum(h)
        if count == n_points // 2:
            return h
        elif count > n_points // 2:
            diff = count - n_points // 2
            positions = np.where(h == 1)[0]
            flip_indices = np.random.choice(positions, diff, replace=False)
            h[flip_indices] = 0
        else:
            diff = n_points // 2 - count
            positions = np.where(h == 0)[0]
            flip_indices = np.random.choice(positions, diff, replace=False)
            h[flip_indices] = 1
        return h
    
    # Compute initial value
    current_c5 = compute_correlation(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T = 1.0
    decay_factor = 0.999
    perturbation_rate = 0.05
    update_freq = 1000
    
    # Main loop
    iter_count = 0
    while time.time() - start_time < budget_s - 2:
        iter_count += 1
        
        # Keep track of last improvement
        if iter_count % update_freq == 0:
            # Periodic perturbation
            num_perturb = int(n_points * perturbation_rate)
            perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
            h[perturb_indices] = 1 - h[perturb_indices]
            h = project_feasible(h)
        
        # Pick a random index to flip
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]  # Flip the bit
        
        # Project to feasible set
        h_new = project_feasible(h_new)
        
        # Compute new C5
        new_c5 = compute_correlation(h_new)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
        else:
            # Accept with some probability
            delta = new_c5 - current_c5
            prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                h = h_new
                current_c5 = new_c5
        
        # Update temperature
        T = max(1e-10, T * decay_factor)
        
        # Early exit if no improvement found for a while (simplified approach)
        if iter_count % 10000 == 0 and current_c5 > best_c5 + 1e-6:
            # Very basic stagnation check
            pass
    
    return (best_h.tolist(), best_c5, n_points)
