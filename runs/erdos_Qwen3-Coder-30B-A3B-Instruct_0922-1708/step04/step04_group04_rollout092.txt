import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    max_iterations = 1000
    iteration = 0
    
    # LHS sampling approach
    def generate_lhs_samples(count, n):
        # Generate Latin Hypercube Samples
        samples = []
        for _ in range(count):
            # Generate random permutation
            perm = np.random.permutation(n)
            sample = np.zeros(n)
            for i in range(n):
                sample[i] = (perm[i] + np.random.random()) / n
            # Scale to [0, 1] and enforce sum constraint
            sample = sample * (n / 2) / np.sum(sample)
            samples.append(sample)
        return samples
    
    # Refinement function
    def refine_candidate(h_start, max_iter=200):
        h = h_start.copy()
        best_h_refined = h.copy()
        best_c5_refined = float('inf')
        
        for iter in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5_current = np.max(correlation) * dx
            
            if c5_current < best_c5_refined:
                best_c5_refined = c5_current
                best_h_refined = h.copy()
            
            # Compute gradient with finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Gradient descent step
            lr = 0.01 / (1 + 0.001 * iter)
            h_new = h - lr * grad
            
            # Clip and rescale to meet constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Early stopping
            if iter > 50 and abs(c5_current - best_c5_refined) < 1e-8:
                break
                
        return best_h_refined, best_c5_refined
    
    # Generate LHS samples
    lhs_samples = generate_lhs_samples(50, n_points)
    
    # Try each LHS sample with refinement
    for sample in lhs_samples:
        if time.time() - start_time > budget_s - 2:
            break
            
        refined_h, refined_c5 = refine_candidate(sample, max_iter=150)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h.copy()
    
    # Final refinement of best solution
    final_h, final_c5 = refine_candidate(best_h, max_iter=100)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h.copy()

    # Ensure final constraints are met
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
