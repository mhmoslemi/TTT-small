import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    n_particles = 20
    max_iter = 200
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        
    # Normalize to integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    inertia_decay = (w_start - w_end) / max_iter
    
    # Initialize swarm
    swarm_pos = []
    swarm_vel = []
    swarm_best_pos = []
    swarm_best_val = []
    
    for _ in range(n_particles):
        pos = h_init + np.random.normal(0, 0.01, n_points)
        pos = np.clip(pos, 0, 1)
        pos = pos * (n_points / 2) / np.sum(pos)
        swarm_pos.append(pos.copy())
        swarm_vel.append(np.random.normal(0, 0.001, n_points))
        swarm_best_pos.append(pos.copy())
        corr = np.correlate(pos, 1 - pos, mode='full')
        swarm_best_val.append(np.max(corr) * dx)
    
    # PSO main loop
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - inertia_decay * it
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            swarm_vel[i] = (w * swarm_vel[i] + 
                          c1 * r1 * (swarm_best_pos[i] - swarm_pos[i]) + 
                          c2 * r2 * (swarm_best_pos[np.argmin(swarm_best_val)] - swarm_pos[i]))
            
            # Update position
            swarm_pos[i] += swarm_vel[i]
            swarm_pos[i] = np.clip(swarm_pos[i], 0, 1)
            swarm_pos[i] = swarm_pos[i] * (n_points / 2) / np.sum(swarm_pos[i])
            
            # Evaluate
            corr = np.correlate(swarm_pos[i], 1 - swarm_pos[i], mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < swarm_best_val[i]:
                swarm_best_val[i] = c5
                swarm_best_pos[i] = swarm_pos[i].copy()
                
        # Add noise occasionally to prevent stagnation
        if it % 20 == 0:
            for i in range(n_particles):
                noise = np.random.normal(0, 0.0005, n_points)
                swarm_pos[i] = np.clip(swarm_pos[i] + noise, 0, 1)
                swarm_pos[i] = swarm_pos[i] * (n_points / 2) / np.sum(swarm_pos[i])
    
    # Refinement via local search using scipy
    top_candidates = np.argsort(swarm_best_val)[:5]
    best_c5 = min(swarm_best_val)
    best_h = swarm_best_pos[top_candidates[0]]
    
    # Refine each of the top candidates
    for i in top_candidates:
        h0 = swarm_best_pos[i].copy()
        corr = np.correlate(h0, 1 - h0, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Local refinement using scipy minimize
        def objective(h):
            # Clip h to [0,1]
            h = np.clip(h, 0, 1)
            # Renormalize to integral constraint
            h = h * (n_points / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode='full')
            return -np.max(corr) * dx  # negative because we minimize
        
        def constraint(h):
            return np.sum(h) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-6})
            ref_c5 = -res.fun
            if ref_c5 < current_c5:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                corr_refined = np.correlate(h_refined, 1 - h_refined, mode='full')
                refined_c5 = np.max(corr_refined) * dx
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = h_refined
        except:
            pass
            
        if time.time() - start_time > budget_s - 2:
            break
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
