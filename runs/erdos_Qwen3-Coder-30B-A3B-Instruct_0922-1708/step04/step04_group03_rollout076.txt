import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation using CVXPY
    # variables
    h_cvx = cp.Variable(n_points)
    # constraints
    constraints = [h_cvx >= 0, h_cvx <= 1, cp.sum(h_cvx) == n_points / 2]
    # objective: minimize sum of all correlations
    # For a convex relaxation, we'll minimize the sum of all overlap terms
    # The exact convex form for this is difficult, so instead:
    # we minimize a proxy based on linearized cross terms
    
    # This is a rough approximation - it's probably not very tight but
    # helps us get a reasonable starting solution
    # Let's try solving a simpler optimization with convex relaxation:
    # minimize: sum_i h[i] (1 - mean(h)), 
    # which is a simplification but still captures the idea of reducing
    # the magnitude of overlaps
    
    # Better approach for convex relaxation: 
    # Since this is a highly structured overlap, we can try to create a 
    # reasonable starting point by using a low-pass filtered approach
    # We will implement a two-phase approach
    
    # Phase 1: Convex relaxation approximation with some advanced dynamics
    # We create a smooth initial vector using some prior knowledge
    
    # Phase 1: Use a density modulation approach to get smoother solution
    h_smooth = h.copy()
    # Apply Gaussian smoothing to smooth the function
    kernel = np.exp(-np.linspace(-2, 2, 21)**2/2)
    kernel = kernel / np.sum(kernel)
    h_smooth = np.convolve(h_smooth, kernel, mode='same')
    
    # Clip and rescale
    h_smooth = np.clip(h_smooth, 0, 1)
    h_smooth = h_smooth * (n_points / 2) / np.sum(h_smooth)
    
    # Phase 2: Improve using projected gradient descent with careful updates
    current_h = h_smooth.copy()
    iteration = 0
    max_no_improvement = 50
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        # Evaluate current solution
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
            improved = True
            
        # Projected gradient descent step
        grad = np.zeros_like(current_h)
        eps = 1e-5
        
        # Compute gradient using finite differences
        for i in range(n_points):
            h_plus = current_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update step with adaptive learning rate
        # Small adaptive learning rate based on gradient magnitude
        lr = 0.05 / (1 + np.linalg.norm(grad) * 1e-3)
        new_h = current_h - lr * grad
        
        # Project back to feasible set: [0,1] and sum constraint
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        current_h = new_h
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Occasionally restart from good solution for escape
        if iteration % 200 == 0 and iteration > 0:
            current_h = best_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
