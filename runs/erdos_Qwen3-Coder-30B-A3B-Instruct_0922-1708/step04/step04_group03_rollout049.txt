import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    learning_rate = 0.1
    eps = 1e-4
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current correlation and max overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(correlation) * dx
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Compute directional derivatives for each h[i]
        derivatives = np.zeros(n_points)
        for i in range(n_points):
            # Increase h[i]
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Decrease h[i]
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Compute derivative
            derivatives[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Update h using gradient descent
        h_new = h - learning_rate * derivatives
        
        # Ensure constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adjust learning rate
        if iteration % 50 == 0:
            learning_rate *= 0.99
            
    return (best_h.tolist(), best_c5, n_points)
