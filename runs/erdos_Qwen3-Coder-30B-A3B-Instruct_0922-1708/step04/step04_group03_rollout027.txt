import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation with CVXPY
    # We want to minimize sum of overlaps, which is a good proxy for the max overlap
    # Define CVXPY variables
    h_cvx = cp.Variable(n_points)
    
    # Objective: minimize sum of overlaps
    # For a fixed shift k, the overlap is sum of h[i] * (1 - h[i+k])
    # We approximate by minimizing the sum over all k
    objective = cp.Minimize(cp.sum(cp.multiply(h_cvx, 1 - h_cvx)))
    
    # Constraints
    constraints = [
        h_cvx >= 0,
        h_cvx <= 1,
        cp.sum(h_cvx) == n_points / 2
    ]
    
    # Solve the convex problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = h_cvx.value
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    # Phase 2: Local refinement using projected gradient descent
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def compute_gradient(h, eps=1e-4):
        grad = np.zeros_like(h)
        c5_base = compute_c5(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    # Refinement loop
    max_iter = 1000
    lr = 0.01
    no_improvement_count = 0
    patience = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_current = compute_c5(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        grad = compute_gradient(h)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_current:
            h = h_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= patience:
                # Reduce learning rate
                lr *= 0.9
                no_improvement_count = 0
                
    # Final check for best solution
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (best_h.tolist(), best_c5, n_points)
