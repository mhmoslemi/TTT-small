import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation approach
    # Create a relaxed version using cvxpy to initialize better
    h_relaxed = h.copy()
    
    # Constraints for the relaxed problem
    h_vars = cp.Variable(n_points)
    
    # Objective: minimize sum of overlaps (simplified convex relaxation)
    # We'll use a heuristic to prevent overlap too well
    objective = cp.Minimize(cp.sum_squares(h_vars))
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2
    ]
    
    # Create problem and solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        h_candidate = h_vars.value
        if h_candidate is not None:
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            h = h_candidate.copy()
    except:
        pass  # Fall back to original if no luck
    
    # Phase 2: Projected gradient descent using finite differences
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        c5_base = compute_c5(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    max_iter = 1000
    lr = 0.01
    no_improvement_limit = 50
    no_improvement_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        grad = compute_gradient(h)
        h_new = h - lr * grad
        
        # Pull back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate
        if i % 100 == 0 and i > 0:
            lr *= 0.9
            
        if no_improvement_count >= no_improvement_limit:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
