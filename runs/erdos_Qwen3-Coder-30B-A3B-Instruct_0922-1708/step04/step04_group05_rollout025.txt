import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise use uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        
    # Normalize to satisfy integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h = h_init.copy()
    
    # PSO parameters
    n_particles = 20
    max_iter = 200
    w_max, w_min = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    inertia_decay = (w_max - w_min) / max_iter
    
    # Initialize swarm
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    for i in range(n_particles):
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    personal_best = particles.copy()
    personal_best_fitness = np.full(n_particles, np.inf)
    global_best_idx = 0
    global_best_fitness = np.inf
    global_best = h.copy()
    
    start_time = time.time()
    
    # PSO loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        w = w_max - inertia_decay * iter_count
        
        # Evaluate all particles
        for i in range(n_particles):
            particle = particles[i]
            
            # Evaluate C5 for current particle
            correlation_full = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation_full) * dx
            
            # Update personal best
            if c5 < personal_best_fitness[i]:
                personal_best_fitness[i] = c5
                personal_best[i] = particle.copy()
                
            # Update global best
            if c5 < global_best_fitness:
                global_best_fitness = c5
                global_best = particle.copy()
                global_best_idx = i
                
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) +
                           c2 * r2 * (global_best - particles[i]))
            
            particles[i] += velocities[i]
            
            # Clamp to valid range
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Maintain integral constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Refinement with local search
    best_h = global_best.copy()
    best_c5 = global_best_fitness
    
    # Perform line search around top solutions
    top_candidates = []
    for i in range(n_particles):
        top_candidates.append((personal_best_fitness[i], personal_best[i].copy()))
    top_candidates.sort(key=lambda x: x[0])
    
    # Try up to 5 best solutions with refinement
    num_refine = min(5, len(top_candidates))
    for _, candidate in top_candidates[:num_refine]:
        # Simple gradient-based refinement
        ref_h = candidate.copy()
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Compute gradient approximation
            grad = np.zeros(n_points)
            eps = 1e-4
            base_c5 = np.max(np.correlate(ref_h, 1 - ref_h, mode='full')) * dx
            
            for j in range(n_points):
                h_plus = ref_h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - base_c5) / eps
            
            # Update with gradient descent
            lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-2)
            ref_h = ref_h - lr * grad
            ref_h = np.clip(ref_h, 0, 1)
            ref_h = ref_h * (n_points / 2) / np.sum(ref_h)
            
            # Check for improvement
            new_c5 = np.max(np.correlate(ref_h, 1 - ref_h, mode='full')) * dx
            if new_c5 < base_c5:
                base_c5 = new_c5
                ref_h = ref_h.copy()
        
        # Update if better than previous best
        final_c5 = np.max(np.correlate(ref_h, 1 - ref_h, mode='full')) * dx
        if final_c5 < best_c5:
            best_h = ref_h.copy()
            best_c5 = final_c5

    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    # Ensure integral constraint is met
    assert np.abs(np.sum(best_h) * dx - 1.0) < 1e-10, "Integral constraint violated"
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Bounds constraint violated"
    
    return (best_h.tolist(), final_c5, n_points)
