import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # PSO parameters
    n_particles = 30
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    vmax = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals() and np.random.rand() < 0.5:
            # Use parent for some particles
            particle_h = h.copy()
        else:
            particle_h = np.random.uniform(0, 1, n_points)
            particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
            particle_h = np.clip(particle_h, 0, 1)
        particles.append(particle_h)
        velocities.append(np.random.uniform(-vmax, vmax, n_points))
        pbest.append(particle_h.copy())
        pbest_c5.append(float('inf'))
    
    gbest = h.copy()
    gbest_c5 = float('inf')
    
    # PSO iterations
    max_iter = 300
    no_improvement = 0
    max_no_improvement = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Compute fitness
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particle.copy()
                pbest_c5[i] = c5
                
            # Update global best
            if c5 < gbest_c5:
                gbest = particle.copy()
                gbest_c5 = c5
                
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
                
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            vel = (inertia * vel + 
                   c1 * r1 * (pbest[i] - particle) + 
                   c2 * r2 * (gbest - particle))
            
            # Clip velocity
            vel = np.clip(vel, -vmax, vmax)
            
            # Update position
            particle += vel
            
            # Enforce constraints
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            velocities[i] = vel
            particles[i] = particle
            
        # Early stopping
        if gbest_c5 < best_c5:
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > max_no_improvement:
                break
    
    # Local refinement with Nelder-Mead on best solution
    h = best_h.copy()
    x0 = h.copy()
    
    def objective(x):
        # Normalize x
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        correlation = np.correlate(x, 1-x, mode='full')
        return -np.max(correlation) * dx  # Negative because we minimize
    
    def constraint(x):
        return np.sum(x) - n_points / 2
        
    # Nelder-Mead local search
    for i in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation for the current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Try to improve with Nelder-Mead
        try:
            # Use smaller dimension for Nelder-Mead simplex
            # Keep 90% of the values fixed, only optimize 10%
            indices = np.random.choice(n_points, size=int(0.1*n_points), replace=False)
            
            # Make a copy of current solution for simplex
            h_current = h.copy()
            h_fixed = h_current.copy()
    
            # simplex constraints
            simplex_size = 10
            simplex = [h_current]
            for _ in range(simplex_size):
                perturb = np.zeros_like(h_current)
                for idx in indices:
                    perturb[idx] = np.random.normal(0, 0.05)
                h_perturbed = h_current + perturb
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
                simplex.append(h_perturbed)
    
            # Use Nelder-Mead on subset
            def centroid(x_simplex):
                return np.mean(x_simplex, axis=0)
            
            def new_simplex(h_ref):
                new_s = [h_ref]
                for i in range(1, len(simplex)):
                    # distorted version
                    x2 = centroid([h_ref] + [simplex[j] for j in range(1, len(simplex)) if j != i])
                    x_new = 2 * x2 - simplex[i]
                    # constrain
                    x_new = np.clip(x_new, 0, 1)
                    x_new = x_new * (n_points / 2) / np.sum(x_new) 
                    new_s.append(x_new)
                return new_s
            
            iter_count = 0
            while iter_count < 10:
                if time.time() - start_time > budget_s - 1:
                    break
                    
                for i in range(len(simplex)): 
                    opt_h = simplex[i]
                    corr = np.correlate(opt_h, 1-opt_h, mode='full')
                    val = np.max(corr) * dx
                    if val < best_c5:
                        best_c5 = val
                        best_h = opt_h.copy()
                        
                # Simplex optimization step
                simplex = new_simplex(h_current)
                iter_count += 1
                
        except Exception:
            pass
        
        # Regular pso-based update to mix with global search
        # Update with simple gradient
        x_fixed = h.copy()
        grad = np.zeros_like(x_fixed)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = x_fixed.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # update with gradient descent step
        lr = 0.01
        h_new = x_fixed - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update learning rate
        lr = max(1e-6, lr * 0.999)
        
        h = h_new if np.max(np.correlate(h_new, 1-h_new, mode='full')) * dx < 
                 np.max(np.correlate(h, 1-h, mode='full')) * dx else h
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
