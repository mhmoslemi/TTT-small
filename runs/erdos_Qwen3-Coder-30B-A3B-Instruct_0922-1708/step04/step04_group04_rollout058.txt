import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Main iterative convex optimization
    max_iter = 100
    tol = 1e-5
    step_size = 0.5
    backtracking_ratio = 0.5
    solver_tol = 1e-4
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current correlation to determine bounds
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Formulate convex approximation of the problem
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(h_var) == n_points / 2]
        constraints += [0 <= h_var[i] for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        
        # Linearized cross-correlation constraints for all k
        # For each shift k, we have sum over i h_i * (1 - h_{i+k}) <= C
        # Use majorization: h_{i+k} ~ h^{old}_{i+k} 
        for k in range(1, n_points):
            # Compute shifted indices correctly for circular shifting in correlation
            shifted_indices = np.arange(n_points) - k
            shifted_indices = shifted_indices % n_points
            
            # Linearized constraint: sum_i h_i * (1 - h_old_{i+k}) <= C
            linear_constraint = cp.sum(cp.multiply(h_var, 1 - h[shifted_indices])) <= C
            constraints.append(linear_constraint)
        
        # Objective
        objective = cp.Minimize(C)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, feastol=solver_tol, abstol=solver_tol)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
            else:
                h_new = h  # fallback
        except:
            h_new = h  # fallback
            
        # Line search
        step = step_size
        for _ in range(20):  # max backtrack steps
            h_candidate = h_new * step + h * (1 - step)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            candidate_corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            candidate_c5 = np.max(candidate_corr) * dx
            
            if candidate_c5 < current_c5:
                h = h_candidate
                break
            step *= backtracking_ratio
        else:
            # If line search fails, continue with h_new
            h = h_new
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Convergence check
        if iteration > 0 and abs(current_c5 - np.max(np.correlate(h, 1 - h, mode='full') * dx)) < tol:
            break
            
    # Final check for best solution
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
