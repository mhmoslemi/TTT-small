import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy the constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Use CVXPY for convex optimization
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1,
        C >= 0
    ]
    
    # Compute overlap constraints for all possible shifts
    for k in range(1, n_points):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Build the overlap expression for shift k
        # When k is positive, we correlate h with 1 - h shifted by k
        # This is equivalent to sum(h[i] * (1 - h[i+k])) for valid i
        
        # Indices for valid correlations
        valid_i = range(n_points - k)
        overlap_terms = []
        for i in valid_i:
            overlap_terms.append(h_var[i] * (1 - h_var[i + k]))
        
        # The sum of the overlap terms must be <= C * dx
        overlap_sum = cp.sum(overlap_terms)
        constraints.append(overlap_sum <= C * dx)
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers = ['ECOS', 'SCS']
    for solver in solvers:
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            prob.solve(solver=solver, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                sol_h = h_var.value
                if sol_h is not None:
                    sol_h = np.clip(sol_h, 0, 1)
                    sol_h = sol_h * (n_points / 2) / np.sum(sol_h)
                    
                    # Compute final objective value C5
                    correlation = np.correlate(sol_h, 1 - sol_h, mode='full')
                    c5 = np.max(correlation) * dx
                    
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = sol_h.copy()
        except:
            continue
            
    # Final refinement using gradient descent
    h = best_h.copy()
    for i in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5 via correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Estimate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5) / eps
            
        # Gradient descent step
        lr = 0.1 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update best solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Check convergence
        if np.abs(c5 - c5_new) < 1e-12:
            break
    
    return (best_h.tolist(), best_c5, n_points)
