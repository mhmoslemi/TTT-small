import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with provided parent if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Try to construct a reasonable starting point
        h_init = np.ones(n_points) * 0.5
        
    # Objective function to minimize C5
    def objective(h):
        # Ensure h is within bounds
        h = np.clip(h, 0, 1)
        # Normalize to correct sum
        h = h * (n_points / 2) / np.sum(h)
        # Compute the overlap integral
        correlation = correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Constraint to ensure integral equals 1
    def constraint_func(h):
        return np.sum(h) * dx - 1
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try different optimization strategies
    best_h = h_init.copy()
    best_c5 = objective(best_h)
    
    # Try differential evolution first
    try:
        result = differential_evolution(objective, bounds, seed=seed, maxiter=50, popsize=15)
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
    except:
        pass
        
    # Local optimization
    try:
        from scipy.optimize import minimize
        # L-BFGS-B handles bounds better
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                          options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9})
        if result.success:
            h_candidate = result.x
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
    except:
        pass
    
    # Fine tune with manual gradient descent on selected points
    current_h = best_h.copy()
    current_c5 = best_c5
    learning_rate = 0.01
    
    # Try a few iterations of gradient descent on a subset of points
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
            
        sample_indices = np.random.choice(n_points, size=max(5, n_points // 10), replace=False)
        grad = np.zeros_like(current_h)
        eps = 1e-4
        
        for i in sample_indices:
            h_plus = current_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = objective(h_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update
        new_h = current_h - learning_rate * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate
        new_c5 = objective(new_h)
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Final validation
    final_c5 = objective(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
