import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):

    population_size = 100
    generations = 100

        if 'initial_h_values' in globals():

            if len(h) != n_points:
                h_orig = np.array(initial_h_values)
                x_orig = np.linspace(0, 2, len(h_orig))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h_orig)
        else:
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
        population.append(h.copy())
    
    # Evolutionary algorithm
    for g in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Selection: keep top 20%
        top_indices = sorted_indices[:population_size // 5]
        top_candidates = [population[i] for i in top_indices]
        
        # Create new generation
        new_population = top_candidates[:]
        
        # Crossover: average top individuals
        while len(new_population) < population_size:
            idx1, idx2 = np.random.choice(len(top_candidates), 2, replace=False)
            h1 = top_candidates[idx1]
            h2 = top_candidates[idx2]
            # Create offspring
            alpha = np.random.rand()
            offspring = alpha * h1 + (1 - alpha) * h2
            # Ensure constraint sum = 50
            offspring = offspring / np.sum(offspring) * (n_points / 2)
            offspring = np.clip(offspring, 0, 1)
            new_population.append(offspring)

        for i in range(len(new_population)):

                new_population[i] += noise

                new_population[i] = new_population[i] / np.sum(new_population[i]) * (n_points / 2)

        # Periodic CVX refinement of best solution
        if g % 20 == 0 and g > 0:
            # Refine top candidates with convex optimize
            for h in population[:10]:

                    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]

                        h_refined = h_vars.value

                            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
                            c5_refined = np.max(corr) * dx
                            if c5_refined < best_c5:
                                best_c5 = c5_refined
                                best_h = h_refined
                except Exception:
                    pass
    
    # Final adjustment with a global optimizer
    try:
        # Use a more robust continuous optimization with quadratically constrained quadratic program
        h_vars = cp.Variable(n_points)
        # Reminder: we want to minimize max_k integral h(x)(1-h(x+k)) dx
        # This is complex and will resort to heuristic
        # For now, use the result from fast evolutionary algorithm and fine-tune
        # Refine one more with CVXPY on top candidates
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        if prob.status == cp.OPTIMAL:

                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)

                c5_cvx = np.max(corr) * dx
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass

    # Final check
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
