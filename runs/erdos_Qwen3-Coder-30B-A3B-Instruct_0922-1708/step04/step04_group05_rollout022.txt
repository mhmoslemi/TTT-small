import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Generate diverse initial candidates
    candidates = [h.copy()]
    for _ in range(19):
        noise = np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h + noise, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        candidates.append(h_pert)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Evaluate all candidates
        candidate_c5s = []
        for i, h_candidate in enumerate(candidates):
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5 = np.max(corr) * dx
            candidate_c5s.append(c5)
        
        # Select best current candidate
        best_idx = np.argmin(candidate_c5s)
        h = candidates[best_idx]
        current_c5 = candidate_c5s[best_idx]
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Gradient-based refinement for current best
        for _ in range(10):
            if time.time() - start_time > budget_s - 2:
                break
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                grad[i] = (c5_plus - c5) / eps
            
            # Gradient descent step
            lr = 0.005 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            
            # Ensure bounds and integral constraint
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
            # Early exit if stagnating
            new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if abs(new_c5 - current_c5) < 1e-8:
                break
            current_c5 = new_c5
        
        # Add new candidate periodically
        if iter_count % 10 == 0 and time.time() - start_time < budget_s - 2:
            noise = np.random.normal(0, 0.02, n_points)
            h_new = np.clip(h + noise, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            candidates.append(h_new)
            if len(candidates) > 20:
                candidates.pop(0)
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
