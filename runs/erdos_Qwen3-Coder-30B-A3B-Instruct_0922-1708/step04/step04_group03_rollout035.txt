import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    max_no_improvement = 100
    no_improvement_count = 0
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        # First, solve a relaxed convex problem for initial solution
        if iteration == 1:
            # Convex relaxation using CVXPY
            h_var = cp.Variable(n_points)
            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
            
            # For convex relaxation, minimize sum of all correlations (approximation)
            # This is not exactly what we want but gives us a good starting point
            # For now, use a simpler heuristic
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            improved = True
            
        # Gradient descent refinement
        h_refined = h.copy()
        lr = 0.01
        
        for refine_iter in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute correlation using numpy
            correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5 = np.max(correlation) * dx
            
            # Compute gradient with finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - c5) / eps
                
            # Update using gradient
            h_new = h_refined - lr * grad
            
            # Enforce constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Check improvement
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr) * dx
            
            if c5_new < c5:
                h_refined = h_new
                improved = True
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_refined.copy()
            else:
                break  # No descent, stop refinement
                
        h = h_refined
        
        # Early stopping
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (best_h.tolist(), best_c5, n_points)
