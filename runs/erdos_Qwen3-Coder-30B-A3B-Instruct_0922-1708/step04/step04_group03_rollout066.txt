import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # DE parameters
    population_size = 50
    F = 0.8
    CR = 0.9
    max_no_improvement = 100
    local_search_freq = 20
    
    # Initialize population
    population = []
    fitness = []
    for _ in range(population_size):
        if 'initial_h_values' in globals() and len(population) == 0:
            individual = h.copy()
        else:
            individual = np.random.uniform(0, 1, n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        corr = np.correlate(individual, 1 - individual, mode='full')
        fitness.append(np.max(corr) * dx)
    
    # Find best individual
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        new_population = []
        new_fitness = []
        improved = False
        
        for i in range(population_size):
            # Select three different individuals
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = a + F * (b - c)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, mutant, population[i])
            
            # Enforce constraints
            trial = np.clip(trial, 0, 1)
            trial = trial * (n_points / 2) / np.sum(trial)
            
            # Evaluate
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_fitness = np.max(corr) * dx
            
            # Selection
            if trial_fitness < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_fitness)
                if trial_fitness < best_fitness:
                    best_fitness = trial_fitness
                    best_individual = trial.copy()
                    improved = True
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
            
        population = new_population
        fitness = new_fitness
        
        # Local search on best individual every few iterations
        if iteration % local_search_freq == 0:
            h = best_individual.copy()
            for refine_iter in range(20):
                correlation = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(correlation) * dx
                
                # Compute gradient with finite differences
                grad = np.zeros_like(h)
                eps = 1e-4
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5) / eps
                
                # Adaptive learning rate
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h - lr * grad
                
                # Clip and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                h = h_new
                
                # Check for improvement
                corr = np.correlate(h, 1 - h, mode='full')
                c5_refine = np.max(corr) * dx
                
                if c5_refine < best_fitness:
                    best_fitness = c5_refine
                    best_individual = h.copy()
                    
            # Update best fitness from the refined individual
            corr = np.correlate(best_individual, 1 - best_individual, mode='full')
            best_fitness = np.max(corr) * dx
        
        # Early stopping
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
    
    return (best_individual.tolist(), best_fitness, n_points)
