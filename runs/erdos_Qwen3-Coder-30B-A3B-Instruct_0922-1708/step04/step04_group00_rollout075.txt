import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.spatial.distance import cdist
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Use differential evolution for global optimization
    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential Evolution with constraints
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            callback=lambda x, convergence: None,
            disp=False
        )
        if not result.success:
            result = differential_evolution(
                objective,
                bounds,
                maxiter=100,
                popsize=10,
                mutation=(0.8, 1),
                recombination=0.7,
                seed=seed+100,
                disp=False
            )
        h_opt = np.clip(result.x, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
        c5_opt = np.max(correlation) * dx
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    except:
        pass
    
    # Local refinement with Nelder-Mead on best solution
    if time.time() - start_time < budget_s - 2:
        try:
            # Create initial simplex around the best solution
            h_refine = best_h.copy()
            simplex_size = 0.05
            simplex = [h_refine]
            for i in range(len(h_refine)):
                pert = np.zeros_like(h_refine)
                pert[i] = simplex_size
                simplex.append(np.clip(h_refine + pert, 0, 1))
            
            # Nelder-Mead optimization
            def ref_obj(h_vec):
                h = np.clip(h_vec, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                correlation = np.correlate(h, 1 - h, mode='full')
                return np.max(correlation) * dx
            
            # Find the center of mass to transform problem
            center = np.mean(simplex, axis=0)
            simplex = [x - center for x in simplex]
            
            # Nelder-Mead iteration
            max_iter = 100
            for i in range(max_iter):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Sort vertices by function value
                values = [ref_obj(x + center) for x in simplex]
                indices = np.argsort(values)
                simplex = [simplex[i] for i in indices]
                values = [values[i] for i in indices]
                
                # Compute centroid excluding worst
                centroid = np.mean(simplex[:-1], axis=0)
                
                # Reflection point
                reflected = (1 + 1) * centroid - simplex[-1]
                f_reflected = ref_obj(reflected + center)
                
                if values[0] <= f_reflected < values[-2]:
                    simplex[-1] = reflected
                elif f_reflected < values[0]:
                    # Expansion
                    expanded = (1 + 2) * centroid - simplex[-1]
                    f_expanded = ref_obj(expanded + center)
                    if f_expanded < f_reflected:
                        simplex[-1] = expanded
                    else:
                        simplex[-1] = reflected
                else:
                    # Contraction
                    if f_reflected < values[-1]:
                        contracted = (1 + 0.5) * centroid - 0.5 * simplex[-1]
                    else:
                        contracted = (1 - 0.5) * simplex[-1] + 0.5 * centroid
                    simplex[-1] = contracted
                
                # Update best
                current_best = np.array(simplex[0]) + center
                current_best = np.clip(current_best, 0, 1)
                current_best = current_best * (n_points / 2) / np.sum(current_best)
                correlation = np.correlate(current_best, 1 - current_best, mode='full')
                current_c5 = np.max(correlation) * dx
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_best.copy()
        except:
            pass
    
    # Final check with exact computation
    try:
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        c5_final = np.max(correlation) * dx
        if c5_final < best_c5:
            best_c5 = c5_final
    except:
        pass
    
    # Ensure proper normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
