import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective_and_gradient(h):
        # Ensure h is within bounds
        h = np.clip(h, 0, 1)
        # Enforce sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        return c5
    
    def objective(h):
        return objective_and_gradient(h)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for sum
    def sum_constraint(h):
        return np.sum(h) - n_points / 2
    
    # Use trust region reflective method
    max_iter = 1000
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Perform optimization step
        try:
            result = minimize(
                objective,
                h,
                method='TRF',
                bounds=bounds,
                options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-8},
                constraints={'type': 'eq', 'fun': sum_constraint}
            )
            h = result.x
        except:
            # Fallback to simple gradient descent
            eps = 1e-4
            grad = np.zeros_like(h)
            c5_base = objective(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = objective_and_gradient(h_plus)
                grad[i] = (c5_plus - c5_base) / eps
            lr = 0.01
            h = h - lr * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Check current solution
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        iteration += 1
        
        # Early stopping
        if iteration % 50 == 0 and iteration > 0:
            if best_c5 < 0.38085:
                break
    
    # Final refinement
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        try:
            result = minimize(
                objective,
                best_h,
                method='TRF',
                bounds=bounds,
                options={'maxiter': 5, 'ftol': 1e-10, 'gtol': 1e-10},
                constraints={'type': 'eq', 'fun': sum_constraint}
            )
            best_h = result.x
        except:
            break
        c5 = objective(best_h)
        if c5 < best_c5:
            best_c5 = c5
        else:
            break
    
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
