import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary: 1 where h >= 0.5, 0 otherwise
    h = (h_init >= 0.5).astype(int)
    
    # Project to exact count of 50 ones
    count_ones = np.sum(h)
    if count_ones > n_points // 2:
        # Flip some 1s to 0s
        zero_indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(zero_indices, count_ones - n_points // 2, replace=False)
        h[flip_indices] = 0
    elif count_ones < n_points // 2:
        # Flip some 0s to 1s
        one_indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(one_indices, n_points // 2 - count_ones, replace=False)
        h[flip_indices] = 1
    
    # Evaluate initial C5
    def compute_c5(h):
        corr = np.correlate(h, 1-h, mode="full")
        return np.max(corr) * dx
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing with binary variables
    T = 1.0  # Initial temperature
    decay_factor = 0.999
    perturbation_frequency = 1000
    stagnation_limit = 1000
    stagnation_count = 0
    
    start_time = time.time()
    
    # Main loop
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Select a random index to flip
        idx = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[idx] = 1 - h_new[idx]
        
        # Ensure exactly n_points//2 ones
        count_ones_new = np.sum(h_new)
        if count_ones_new > n_points // 2:
            # Find 1s and randomly flip one to 0
            ones_idx = np.where(h_new == 1)[0]
            flip_idx = np.random.choice(ones_idx)
            h_new[flip_idx] = 0
        elif count_ones_new < n_points // 2:
            # Find 0s and randomly flip one to 1
            zeros_idx = np.where(h_new == 0)[0]
            flip_idx = np.random.choice(zeros_idx)
            h_new[flip_idx] = 1
            
        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        # Accept or reject the move
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_c5 = new_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Perturbation to escape local minima
        if iteration % perturbation_frequency == 0 and iteration > 0:
            # Reset 5% of values randomly
            num_reset = n_points // 20
            reset_indices = np.random.choice(n_points, num_reset, replace=False)
            for idx in reset_indices:
                h[idx] = np.random.randint(0, 2)
            
            # Project to exact count
            count_ones = np.sum(h)
            if count_ones > n_points // 2:
                ones_idx = np.where(h == 1)[0]
                flip_indices = np.random.choice(ones_idx, count_ones - n_points // 2, replace=False)
                h[flip_indices] = 0
            elif count_ones < n_points // 2:
                zeros_idx = np.where(h == 0)[0]
                flip_indices = np.random.choice(zeros_idx, n_points // 2 - count_ones, replace=False)
                h[flip_indices] = 1
        
        # Decrease temperature
        T *= decay_factor
        
        # Early termination if stagnating
        if stagnation_count > stagnation_limit:
            break
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
