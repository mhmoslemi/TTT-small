import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Function to compute C5
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Function to project h into feasible region
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Function to compute objective and gradient
    def obj_and_grad(h_vec):
        h_vec = project_h(h_vec)
        c5 = compute_c5(h_vec)
        
        # Compute gradient using finite differences
        eps = 1e-4
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps
            
        return c5, grad
    
    # Use trust region reflective method for optimization
    max_iter = 500
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use scipy minimize with trust region reflective method
        result = minimize(
            lambda x: obj_and_grad(x)[0],
            h.copy(),
            method='TRF',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=lambda x: obj_and_grad(x)[1],
            options={'maxiter': 10, 'verbose': 0}
        )
        
        h = project_h(result.x)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early stopping if no improvement for a while
        if iter_count > 20 and abs(prev_c5 - c5) < 1e-6:
            break
        prev_c5 = c5
    
    # Final validation and refinement
    h = project_h(best_h)
    final_c5 = compute_c5(h)
    
    return (h.tolist(), final_c5, n_points)
