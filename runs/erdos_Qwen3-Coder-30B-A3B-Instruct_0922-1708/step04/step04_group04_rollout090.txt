import numpy as np
from scipy.stats import qmc
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Latin Hypercube Sampling
    sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs_samples = sampler.random(n=50)
    
    # Convert LHS to step function candidates
    candidates = []
    for sample in lhs_samples:
        h_candidate = np.interp(sample, np.linspace(0, 1, n_points), np.linspace(0, 1, n_points))
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        candidates.append(h_candidate)
    
    # Add original if exists
    if 'initial_h_values' in globals():
        candidates.append(h.copy())
    
    # Gradient descent for each candidate
    max_iter = 500
    lr = 0.01
    min_lr = 1e-4
    patience = 50
    patience_counter = 0
    best_iter = 0
    
    for i, h_candidate in enumerate(candidates):
        if time.time() - start_time > budget_s - 1:
            break
            
        h = h_candidate.copy()
        prev_c5 = float('inf')
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Gradient via finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - c5) / eps
            
            # Update parameters
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Check improvement
            correlation = np.correlate(h, 1 - h, mode='full')
            c5_new = np.max(correlation) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                best_iter = iteration
                patience_counter = 0
            else:
                patience_counter += 1
                if patience_counter >= patience:
                    lr = max(lr / 2, min_lr)
                    patience_counter = 0
            
            if abs(c5 - c5_new) < 1e-6:
                break
                
    return (best_h.tolist(), best_c5, n_points)
