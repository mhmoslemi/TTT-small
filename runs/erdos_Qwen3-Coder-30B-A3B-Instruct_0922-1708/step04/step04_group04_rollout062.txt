import numpy as np
import time
from heapq import heappush, heappop
from scipy.fft import fft, ifft

def overlap_computation(h, n_points):
    """Compute all overlaps using FFT convolution"""
    # Create the conjugate sequence (1 - h)
    conj_h = 1 - h
    
    # Use FFT convolution to compute all overlaps
    # This is equivalent to np.correlate(h, conj_h, mode='full')
    # But more efficient for large sequences
    n = 2 * n_points - 1
    h_extended = np.concatenate([h, np.zeros(n - n_points)])
    conj_h_extended = np.concatenate([conj_h, np.zeros(n - n_points)])
    
    # Doing convolution via FFT
    H = fft(h_extended)
    Conj_H = fft(conj_h_extended)
    conv_result = ifft(H * np.conj(Conj_H))
    
    # Convert back to real values and extract the full result
    conv_real = np.real(conv_result)
    # Extract the valid parts corresponding to shifts [-n+1, n-1]
    # but we only care about the
    overlap_values = conv_real[n_points-1:n_points+n_points-1]
    return overlap_values

def prioritize_shifts(overlaps):
    """Build priority queue of shifts with maximum overlaps"""
    # Create a max heap using negative values
    heap = []
    for k, val in enumerate(overlaps):
        heappush(heap, (-val, k - (len(overlaps) - 1)))
    return heap

def apply_adjustments(h, k, w, n_points):
    """Apply targeted adjustment for shift k within window w"""
    h_new = h.copy()
    n = len(h)
    # Compute indicies for shift k
    if k >= 0:
        # Positive shift (h[i] looks at h[i+k])
        # Valid indices where we can compare h[i] with h[i+k]
        start_idx = 0
        end_idx = n - k
    else:
        # Negative shift (h[i+k] looks at h[i])
        # Valid indices where we can compare h[i+k] with h[i]
        start_idx = -k
        end_idx = n
        
    # Target region for adjustment within k width
    adjust_start = max(0, start_idx - w)
    adjust_end = min(end_idx + w, n)
    
    # Direct adjustment in the affecting window
    # For overlap reduction: modify h values near the relevant regions
    for i in range(adjust_start, adjust_end):
        # For small changes with constraints
        # If k = 0, we don't have this problem
        
        # Instead of being greedy, we look to slightly alter values
        # toward reducing overlap for this shift k
        
        # This is a simplified strategy:
        # Only modify the h values that contribute most to this overlap
        if k >= 0 and i < n - k:
            # Adjust both h[i] and h[i+k] to reduce their product
            # tanh-guided adjustment (combination of linear and sigmoid behavior)
            # based on values
            val = h[i]
            val_k = h[i + k]
            # Move values to reduce the product h[i] * (1 - h[i+k])
            # perturb locates where changes should occur
            perturb_factor = 0.01  # control the magnitude of change
            
            # Decrease contribution to overlap
            h_new[i] = max(0, val - perturb_factor * (val * (1 - val_k)))
            h_new[i + k] = max(0, val_k - perturb_factor * (val_k * (1 - val)))
            
    # Ensure sum is preserved exactly
    h_new = h_new * (n_points / 2) / np.sum(h_new)
    return np.clip(h_new, 0, 1)

def random_perturb(h, w, pert_scale):
    """Introduce controlled randomness"""
    h_new = h.copy()
    n = len(h)
    # Randomly choose a central point to perturb within some range
    center = np.random.randint(0, n)
    start_idx = max(0, center - w)
    end_idx = min(n, center + w)
    
    for i in range(start_idx, end_idx):
        # Add small random noise
        noise = np.random.normal(0, pert_scale)
        h_new[i] += noise
    
    # Clip and normalize
    h_new = np.clip(h_new, 0, 1)
    return h_new * (len(h_new) / 2) / np.sum(h_new)

def validate_h(h):
    """Ensure h is within bounds and has correct sum"""
    h = np.clip(h, 0, 1)
    h_scale_sum = len(h) / 2 
    actual_sum = np.sum(h)
    h = h * (h_scale_sum / actual_sum)
    return h

    # Initialize priority queue of overlaps for all shifts
    overlaps = overlap_computation(h, n_points)

    pq = prioritize_shifts(overlaps)

    no_improvement_count = 0

    perturb_scale_base = 0.02

            overlaps = overlap_computation(h, n_points)
            pq = prioritize_shifts(overlaps)
            
        current_time = time.time()
        if current_time - start_time > budget_s - 1:
            break
            
        # Get top element
        if not pq:

        h_new = random_perturb(h_new, window_width, perturb_scale_base)

        new_overlaps = overlap_computation(h_new, n_points)

    # Final evaluation
    final_overlaps = overlap_computation(best_h, n_points)
