import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Scale to meet integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h = h_init.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Frequency-based optimization with gradient descent
    max_iter = 1000
    lr = 0.001
    patience = 100
    patience_counter = 0
    prev_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute overlap using FFT
        h_fft = np.fft.fft(h)
        one_minus_h_fft = np.fft.fft(1 - h)
        overlap_fft = h_fft * np.conj(one_minus_h_fft)
        overlap = np.fft.ifft(overlap_fft).real
        
        # Find maximum overlap index
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx]
        
        # Compute C5
        c5 = max_overlap * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            # Early termination
            break
        
        # Gradient computation in frequency domain
        # We want to minimize overlap at index max_idx
        k_shift = max_idx - n_points + 1  # convert to shift value
        if k_shift <= 0:
            k_shift = 0
            
        # Compute frequency domain filter
        omega = 2 * np.pi * np.fft.fftfreq(n_points, dx)
        filter_freq = np.exp(-1j * omega * k_shift)
        
        # Apply filter to frequency domain
        h_filtered_fft = h_fft * filter_freq
        h_filtered = np.fft.ifft(h_filtered_fft).real
        
        # Project back to feasible region
        h_new = np.clip(h_filtered, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update with momentum
        h = (1 - lr) * h + lr * h_new
        
        # Re-project
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if i % 50 == 0:
            # Add some noise to escape local minima
            noise = np.random.normal(0, 0.001, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation
    final_overlap = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_overlap) * dx
    
    return (best_h.tolist(), final_c5, n_points)
