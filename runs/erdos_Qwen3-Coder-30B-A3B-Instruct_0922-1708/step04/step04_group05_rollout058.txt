import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, n_points):
    """Compute C5 = max_k int h(x)(1 - h(x+k)) dx"""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_step(h, n_points, step_size=0.001):
    """Perform gradient step to reduce C5, with finite difference approximation"""
    dx = 2.0 / n_points
    h_original = h.copy()
    grad = np.zeros(n_points)
    eps = 1e-4
    
    # Evaluate original C5
    c5_original = compute_c5(h, n_points)
    
    for i in range(n_points):
        h_plus = h_original.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        c5_plus = compute_c5(h_plus, n_points)
        grad[i] = (c5_plus - c5_original) / eps
    
    h_new = h_original - step_size * grad
    return project_feasible(h_new, n_points)

    h = project_feasible(h, n_points)
    
    best_h = h.copy()

    # Optimization loop
    iter_count = 0

        h_new = gradient_step(h, n_points, step_size=0.001)
