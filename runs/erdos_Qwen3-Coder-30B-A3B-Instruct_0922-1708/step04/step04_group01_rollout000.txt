import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function: first half 1.0, second half 0.0
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h.copy() + np.random.normal(0, 0.01, n_points)
        else:
            particle = np.random.uniform(0, 1, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(particle.copy())
        correlation = np.correlate(particle, 1 - particle, mode='full')
        pbest_vals.append(np.max(correlation) * dx)
    
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    if gbest_val < best_c5:
        best_c5 = gbest_val
        best_h = gbest.copy()
    
    # PSO main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            velocity = w * velocity + c1 * r1 * (pbest[i] - particle) + c2 * r2 * (gbest - particle)
            
            # Update particle
            particle = particle + velocity
            
            # Enforce bounds
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Compute fitness
            correlation = np.correlate(particle, 1 - particle, mode='full')
            fitness = np.max(correlation) * dx
            
            # Update personal best
            if fitness < pbest_vals[i]:
                pbest[i] = particle.copy()
                pbest_vals[i] = fitness
                
                # Update global best
                if fitness < gbest_val:
                    gbest = particle.copy()
                    gbest_val = fitness
                    
                    if gbest_val < best_c5:
                        best_c5 = gbest_val
                        best_h = gbest.copy()
            
            particles[i] = particle
            velocities[i] = velocity
    
    # Fine-tune with CVX if time permits
    if time.time() - start_time < budget_s - 1:
        try:
            # Define the optimization variables and constraints
            h_vars = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Objective: minimize the maximum overlap of a flexible arrangement
            # We need to compute the maximum correlation as a semidefinite program for this step
            
            # This is the advanced solver step
            # For now perform a focused optimization
            # Use best solution so far as initialization
            
            # Refine with local optimization to potentially improve result
            def eval_c5(h_vec):
                h_vec = np.array(h_vec)
                corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
                return np.max(corr)
            
            # Local optimization from best result
            result = minimize(eval_c5, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points,
                             options={'maxiter': 100})
            
            if result.success:
                refined_h = result.x.copy()
                # Normalize to sum = n_points/2
                refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
                refined_c5 = eval_c5(refined_h)
                
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
                    
        except Exception:
            pass
            
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
