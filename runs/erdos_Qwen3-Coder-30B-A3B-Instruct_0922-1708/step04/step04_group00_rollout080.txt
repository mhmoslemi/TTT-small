import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    def objective(h_vec):
        return compute_c5(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Define bounds (0 to 1 for each component)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for sum
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Trust-region reflective method setup
    max_iter = 500
    iter_count = 0
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize using trust region reflective method
        res = minimize(objective, h, bounds=bounds, constraints=constraint, 
                        method='TRF', options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-8})
        
        if res.success:
            h = res.x
            # Project to feasible region
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Update best solution
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        iter_count += 1
        
        # Early stopping conditions
        if iter_count > 100 and iter_count % 10 == 0:
            # Check if we're close to local minimum
            if best_c5 < 0.381:  # Early exit if we're close to target
                break
    
    # Final refinement
    h_final = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        c5_old = compute_c5(h_final)
        
        # Estimate gradient via finite differences
        eps = 1e-5
        grad = np.zeros_like(h_final)
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_old) / eps
        
        # Update step
        lr = 1e-3
        h_new = h_final - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Only accept if improvement
        c5_new = compute_c5(h_new)
        if c5_new < c5_old:
            h_final = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
