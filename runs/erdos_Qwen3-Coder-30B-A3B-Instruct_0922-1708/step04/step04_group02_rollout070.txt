import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    max_iterations = 10000
    
    # Initialize h from initial_h_values if available, otherwise use uniform
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize tracking variables
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    temperature = 1.0
    cooling_rate = 0.99
    min_temperature = 1e-6
    iteration = 0
    start_time = time.time()
    
    # Main SA loop
    while time.time() - start_time < budget_s - 5 and iteration < max_iterations:
        iteration += 1
        
        # Compute current C₅
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate candidate solution
        h_candidate = h.copy()
        
        # Select 10% of indices for perturbation
        num_perturb = max(1, n_points // 10)
        perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
        
        # Perturb selected values
        for i in perturb_indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_candidate[i] += delta
        
        # Ensure feasibility
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Maintain sum constraint
        old_sum = np.sum(h)
        new_sum = np.sum(h_candidate)
        diff = (n_points / 2) - new_sum
        
        # Distribute the difference among all points
        if abs(diff) > 1e-10:
            h_candidate += diff / n_points
        
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute candidate C₅
        correlation_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(correlation_candidate) * dx
        
        # Accept or reject candidate
        delta = candidate_c5 - current_c5
        
        if delta < 0 or np.random.random() < np.exp(-delta / temperature):
            h = h_candidate.copy()
        
        # Update temperature
        if iteration % 100 == 0:
            temperature = max(min_temperature, temperature * cooling_rate)
    
    # Final refinement with local search
    refinement_steps = 500
    for _ in range(refinement_steps):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Small random perturbation and acceptance
        h_candidate = h.copy()
        index = np.random.randint(0, n_points)
        h_candidate[index] += np.random.uniform(-0.05, 0.05)
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Maintain integral constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate
        correlation_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(correlation_candidate) * dx
        
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = h_candidate.copy()
            h = h_candidate.copy()
    
    # Final verification
    correlation_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
