import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Trust region method parameters
    trust_radius = 0.1
    trust_radius_min = 1e-6
    decay_rate = 0.99
    max_iter = 500
    max_stagnation = 50
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_gradient(h, eps=1e-4):
        grad = np.zeros_like(h)
        c5_base = compute_c5(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    def solve_subproblem(grad, trust_radius):
        # Simple gradient descent within trust region
        step = -grad
        step_norm = np.linalg.norm(step)
        if step_norm > trust_radius:
            step = step * trust_radius / step_norm
        return step
    
    stagnation_count = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                break

        grad = compute_gradient(h)
        step = solve_subproblem(grad, trust_radius)
        h_new = h + step
        h_new = project_to_feasible(h_new)

        # Update trust region
        h = h_new
        trust_radius = max(trust_radius_min, trust_radius * decay_rate)
    
    # Final refinement with local search on best solution
    h = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        grad = compute_gradient(h)
        step = -0.01 * grad
        h_new = h + step
        h_new = project_to_feasible(h_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return (best_h.tolist(), best_c5, n_points)
