import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization using scipy minimize with bounds and constraints
    def objective(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Define bounds (0 <= h[i] <= 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Optimization parameters
    max_optimize_iter = 500
    initial_step = 0.01
    max_iter = 1000
    
    # Optimization loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimization with L-BFGS-B
        result = minimize(objective, h.copy(), method='L-BFGS-B', bounds=bounds, 
                          constraints=constraints, options={'maxiter': max_optimize_iter,
                                                          'ftol': 1e-12, 'gtol': 1e-12})
        
        if result.success:
            h_new = result.x
        else:
            # Fallback to random perturbation if optimization fails
            h_new = h + np.random.normal(0, initial_step, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
        # Evaluate new solution
        c5_new = objective(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Update h for next round
        h = h_new.copy()
        
        # Early stopping if the target is reached
        if best_c5 <= 0.3807:
            break
            
    # Final refinement if needed
    if best_c5 > 0.3807 and time.time() - start_time < budget_s - 1:
        # Try small local perturbations to refine further
        h_refined = best_h.copy()
        steps = [0.001, 0.0005, 0.0001, 0.00005]
        for step in steps:
            if time.time() - start_time > budget_s - 1:
                break
            h_candidate = h_refined + np.random.normal(0, step, n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
        
    return (best_h.tolist(), best_c5, n_points)
