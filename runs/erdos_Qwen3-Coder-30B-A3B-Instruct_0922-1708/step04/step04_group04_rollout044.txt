    def latin_hypercube(n, samples):
        result = np.zeros((samples, n))
        for i in range(n):
            # Generate uniform random numbers for this dimension
            u = np.random.uniform(i / n, (i + 1) / n, samples)
            # Shuffle to create LHS
            np.random.shuffle(u)
            result[:, i] = u
        return result

        h_vec = np.clip(h_vec, 0, 1)

        return h_vec

        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
