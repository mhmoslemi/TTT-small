import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    h = (h_init >= 0.5).astype(int)
    # Adjust to ensure exactly n_points/2 ones
    count_ones = np.sum(h)
    if count_ones > n_points // 2:
        # Need to flip some 1s to 0s
        zero_indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(zero_indices, count_ones - n_points // 2, replace=False)
        h[flip_indices] = 0
    elif count_ones < n_points // 2:
        # Need to flip some 0s to 1s
        one_indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(one_indices, n_points // 2 - count_ones, replace=False)
        h[flip_indices] = 1
    
    # Compute initial correlation and C5
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    perturbation_rate = 0.05
    max_iter = 10000
    no_improve_limit = 1000
    no_improve_count = 0
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make a change: randomly flip a bit
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]  # Flip the bit
        
        # Project to ensure exactly n_points/2 ones
        count_ones = np.sum(h_new)
        if count_ones > n_points // 2:
            # Flip back some 1s to 0s
            zero_indices = np.where(h_new == 1)[0]
            flip_indices = np.random.choice(zero_indices, count_ones - n_points // 2, replace=False)
            h_new[flip_indices] = 0
        elif count_ones < n_points // 2:
            # Flip back some 0s to 1s
            one_indices = np.where(h_new == 0)[0]
            flip_indices = np.random.choice(one_indices, n_points // 2 - count_ones, replace=False)
            h_new[flip_indices] = 1

        # Compute new correlation and C5
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx

        # Decide whether to accept the change
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            no_improve_count = 0
        else:
            # Accept with probability based on temperature
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
                current_c5 = new_c5
            no_improve_count += 1

        # Decrease temperature
        T *= cooling_rate
        
        # Periodic perturbation to escape local minima
        if iteration % 1000 == 0 and iteration > 0:
            num_perturb = int(n_points * perturbation_rate)
            random_indices = np.random.choice(n_points, num_perturb, replace=False)
            for i in random_indices:
                h[i] = 1 - h[i]
            # Project to correct count
            count_ones = np.sum(h)
            if count_ones > n_points // 2:
                zero_indices = np.where(h == 1)[0]
                flip_indices = np.random.choice(zero_indices, count_ones - n_points // 2, replace=False)
                h[flip_indices] = 0
            elif count_ones < n_points // 2:
                one_indices = np.where(h == 0)[0]
                flip_indices = np.random.choice(one_indices, n_points // 2 - count_ones, replace=False)
                h[flip_indices] = 1

        # Early termination if no improvement for a while
        if no_improve_count > no_improve_limit:
            break

    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
