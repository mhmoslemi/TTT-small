import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution approach
    resolutions = [50, 100, 200]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize population
        population_size = 100
        population = []
        
        # Start with the provided initial solution or random
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_init)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.ones(n_points) * 0.5
            h = h * (n_points / 2) / np.sum(h)
            
        population.append(h)
        
        # Generate random feasible solutions
        for _ in range(population_size - 1):
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
            
        # Genetic algorithm for current resolution
        for gen in range(200):  # Generations
            if time.time() - start_time > budget_s - 1:
                break
                
            # Evaluate fitness
            fitness = []
            for h in population:
                corr = correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * (2.0 / n_points)
                fitness.append(c5)
                
            # Select top 50%
            sorted_indices = np.argsort(fitness)
            population = [population[i] for i in sorted_indices[:population_size//2]]
            fitness = [fitness[i] for i in sorted_indices[:population_size//2]]
            
            # Create new population through crossover and mutation
            new_population = population.copy()
            while len(new_population) < population_size:
                # Tournament selection
                i1 = np.random.randint(0, len(population))
                i2 = np.random.randint(0, len(population))
                parent1 = population[i1] if fitness[i1] < fitness[i2] else population[i2]
                
                i3 = np.random.randint(0, len(population))
                i4 = np.random.randint(0, len(population))
                parent2 = population[i3] if fitness[i3] < fitness[i4] else population[i4]
                
                # Crossover
                if np.random.random() < 0.6:
                    child1, child2 = crossover(parent1, parent2, n_points)
                else:
                    child1 = parent1.copy()
                    child2 = parent2.copy()
                    
                # Mutation
                child1 = mutate(child1, n_points, gen, 200)
                child2 = mutate(child2, n_points, gen, 200)
                
                # Repair
                child1 = repair(child1, n_points)
                child2 = repair(child2, n_points)
                
                new_population.extend([child1, child2])
                
            population = new_population[:population_size]
            
            # Keep track of best
            for i, h in enumerate(population):
                if fitness[i] < best_c5:
                    best_c5 = fitness[i]
                    best_h = h.copy()
                    
        # Gradient refinement on best solution
        if best_h is not None:
            h_refined = gradient_refinement(best_h.copy(), n_points, start_time, budget_s)
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            c5_refined = np.max(corr) * (2.0 / n_points)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined

    return (best_h.tolist(), best_c5, len(best_h))

def crossover(parent1, parent2, n_points):
    # Simulated Binary Crossover
    eta = 20
    beta = np.random.random(n_points)
    gamma = np.where(beta <= 0.5, (2 * beta) ** (1.0 / (eta + 1)), 
                     (1.0 / (2 * (1 - beta))) ** (1.0 / (eta + 1)))
    
    child1 = 0.5 * ((1 + gamma) * parent1 + (1 - gamma) * parent2)
    child2 = 0.5 * ((1 - gamma) * parent1 + (1 + gamma) * parent2)
    
    return child1, child2

def mutate(child, n_points, generation, max_generations):
    # Polynomial mutation
    eta_m = 20
    mutation_rate = 1.0 / n_points
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            if np.random.random() < 0.5:
                delta = np.random.random()
                delta = (delta ** (1.0 / (eta_m + 1)) - 1) * 0.5
                child[i] = np.clip(child[i] + delta, 0, 1)
            else:
                delta = np.random.random()
                delta = 1 - (1 - delta) ** (1.0 / (eta_m + 1))
                child[i] = np.clip(child[i] + delta, 0, 1)
    return child

def repair(child, n_points):
    # Clip values to [0,1]
    child = np.clip(child, 0, 1)
    # Rescale to sum to n_points/2
    child = child * (n_points / 2) / np.sum(child)
    return child

def gradient_refinement(h, n_points, start_time, budget_s):
    dx = 2.0 / n_points
    max_iter = 1000
    lr = 0.001
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient with finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair(h_plus, n_points)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            g = (c5_plus - compute_c5(h, n_points, dx)) / eps
            grad[j] = g
            
        # Update
        h = h - lr * grad
        h = repair(h, n_points)
        
    return h

def compute_c5(h, n_points, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
