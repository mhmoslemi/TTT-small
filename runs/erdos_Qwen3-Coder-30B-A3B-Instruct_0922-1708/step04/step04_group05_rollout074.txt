import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        # Construct a reasonable initial guess
        h_init = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Collect initial candidates
    candidates = [h_init.copy()]
    for i in range(19):
        h_perturbed = h_init + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        candidates.append(h_perturbed)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Evaluate all candidates
        candidate_c5s = []
        for i, h in enumerate(candidates):
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            candidate_c5s.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Find the best candidate
        best_idx = np.argmin(candidate_c5s)
        h = candidates[best_idx].copy()
        
        # Gradient descent step
        num_steps = 50
        for step in range(num_steps):
            if time.time() - start_time > budget_s - 2:
                break
            
            # Compute the gradient approximated via finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                # Restore sum constraint
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update h
            lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            
            # Project onto constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
            # Early stopping if improvement is minimal
            new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if abs(new_c5 - np.max(np.correlate(candidates[best_idx], 1 - candidates[best_idx], mode='full')) * dx) < 1e-10:
                break
        
        # Store the new candidate
        candidates[best_idx] = h.copy()
        
        # Add a new candidate periodically
        if iter_count % 10 == 0 and len(candidates) < 30:
            noise = np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h + noise, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            candidates.append(h_new)
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
