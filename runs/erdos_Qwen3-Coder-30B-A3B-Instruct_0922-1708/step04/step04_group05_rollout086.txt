import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary with threshold 0.5
    h = (h_init >= 0.5).astype(int)
    
    # Ensure exactly n_points/2 ones
    ones_count = np.sum(h)
    needed = n_points // 2
    if ones_count > needed:
        # Set some ones to zeros
        zeros_indices = np.where(h == 1)[0]
        np.random.shuffle(zeros_indices)
        for i in range(ones_count - needed):
            h[zeros_indices[i]] = 0
    elif ones_count < needed:
        # Set some zeros to ones
        ones_indices = np.where(h == 0)[0]
        np.random.shuffle(ones_indices)
        for i in range(needed - ones_count):
            h[ones_indices[i]] = 1
    
    # Compute initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T = 1.0
    decay = 0.999
    perturb_freq = 1000
    perturb_rate = 0.05
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iter_count += 1
        
        # Make a candidate move
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]  # Flip the bit
        
        # Ensure exactly n_points/2 ones
        ones_count_new = np.sum(h_new)
        needed = n_points // 2
        if ones_count_new != needed:
            # Adjust to get exactly needed ones
            if ones_count_new > needed:
                # Need to flip some 1s to 0s
                ones_indices = np.where(h_new == 1)[0]
                np.random.shuffle(ones_indices)
                flip_count = ones_count_new - needed
                for j in range(flip_count):
                    h_new[ones_indices[j]] = 0
            else:
                # Need to flip some 0s to 1s
                zeros_indices = np.where(h_new == 0)[0]
                np.random.shuffle(zeros_indices)
                flip_count = needed - ones_count_new
                for j in range(flip_count):
                    h_new[zeros_indices[j]] = 1
        
        # Compute new C5
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject the move
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            current_c5 = new_c5
            
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Periodic random perturbation
        if iter_count % perturb_freq == 0:
            num_perturb = int(n_points * perturb_rate)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for i in indices:
                h[i] = 1 - h[i]
            
            # Rebalance
            ones_count = np.sum(h)
            needed = n_points // 2
            if ones_count > needed:
                ones_indices = np.where(h == 1)[0]
                np.random.shuffle(ones_indices)
                for i in range(ones_count - needed):
                    h[ones_indices[i]] = 0
            elif ones_count < needed:
                zeros_indices = np.where(h == 0)[0]
                np.random.shuffle(zeros_indices)
                for i in range(needed - ones_count):
                    h[zeros_indices[i]] = 1
        
        # Update temperature
        T *= decay
    
    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
