import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Generate correlation matrix for gradient computation
    def compute_correlation(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr
    
    def evaluate_c5(h):
        corr = compute_correlation(h)
        return np.max(corr) * dx
    
    def compute_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_c5 = evaluate_c5(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # First, solve convex relaxation to get a good starting point
    # Minimize sum of all overlaps (including k=0 case which is not needed)
    # But for relaxation, we consider all possible overlaps
    
    # Relaxation using CVXPY with sum of overlaps minimized (not maximum)
    x = cp.Variable(n_points)
    constraints = [0 <= x, x <= 1, cp.sum(x) == n_points / 2]
    
    # We want to minimize sum of all overlap terms:
    # For each k, sum_i h[i] * (1-h[i+k])
    # Not directly expressible via CVXPY linear constraints (nonlinear)
    # We'll use a simpler convex relaxation
    
    # Relaxation: Minimize total sum of all correlations
    # Minimize sum(h*(1-h)) over all positions  
    # But that's a continuous relaxation.
    # So start with our current guess and then refine it
    
    # Start with refinement from local solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Set up a gradient descent-like refinement with momentum
    momentum = 0.9
    velocity = np.zeros_like(h)
    lr = 0.005
    
    no_improvement = 0
    max_no_improvement = 50
    
    start_time = time.time()
    
    iterations = 0
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        iterations += 1
        # Compute gradient for current h
        grad = compute_gradient(best_h)
        
        # Update with momentum
        velocity = momentum * velocity - lr * grad
        new_h = best_h + velocity
        
        # Ensure new_h is in [0,1]
        new_h = np.clip(new_h, 0, 1)
        # Rescale to sum to n_points/2
        new_h = new_h * (n_points / 2) / np.sum(new_h)

        new_c5 = evaluate_c5(new_h)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Occasionally reevaluate current best with gradient computation
        if iterations % 20 == 0:
            c5_current = evaluate_c5(best_h)
            if c5_current < best_c5:
                best_c5 = c5_current
                
    return (best_h.tolist(), best_c5, n_points)
