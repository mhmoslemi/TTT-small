import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Differential Evolution parameters
    population_size = 50
    F = 0.8
    CR = 0.9
    max_no_improvement = 100
    
    # Initialize population
    population = []
    fitness = []
    
    # Add the initial solution to population if available
    if 'initial_h_values' in globals():
        population.append(h.copy())
        corr = np.correlate(h, 1-h, mode='full')
        fitness.append(np.max(corr) * dx)
    else:
        # Start with random solutions
        for _ in range(population_size):
            h_init = np.random.uniform(0, 1, n_points)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            population.append(h_init)
            corr = np.correlate(h_init, 1-h_init, mode='full')
            fitness.append(np.max(corr) * dx)
    
    # Add missing solutions to population to reach size
    while len(population) < population_size:
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        corr = np.correlate(h_init, 1-h_init, mode='full')
        fitness.append(np.max(corr) * dx)
    
    # Find the best individual
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    no_improvement = 0
    iteration = 0
    
    # DE optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        # For each individual in the population
        for i in range(population_size):
            # Select three distinct individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            crossover_mask = np.random.rand(n_points) < CR
            trial = np.where(crossover_mask, mutant, population[i])
            
            # Ensure constraint (sum = n_points/2)
            trial = trial * (n_points / 2) / np.sum(trial)
            
            # Evaluate trial vector
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_c5 = np.max(corr) * dx
            
            # Selection
            if trial_c5 < fitness[i]:
                population[i] = trial.copy()
                fitness[i] = trial_c5
                improved = True
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
        
        # Local refinement every 20 iterations
        if iteration % 20 == 0:
            # Refine the best solution
            h_refine = best_h.copy()
            for j in range(100):
                corr = np.correlate(h_refine, 1 - h_refine, mode='full')
                c5 = np.max(corr) * dx
                
                # Gradient estimation via finite differences
                grad = np.zeros_like(h_refine)
                eps = 1e-5
                for k in range(n_points):
                    h_plus = h_refine.copy()
                    h_plus[k] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[k] = (c5_plus - c5) / eps
                
                # Adaptive learning rate
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h_refine - lr * grad
                
                # Clip and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                h_refine = h_new
                
                # Update best if improved
                corr = np.correlate(h_refine, 1 - h_refine, mode='full')
                c5_refine = np.max(corr) * dx
                
                if c5_refine < best_c5:
                    best_c5 = c5_refine
                    best_h = h_refine.copy()
            
            # Update population with refined best solution if it's better
            best_idx = np.argmin(fitness)
            if best_c5 < fitness[best_idx]:
                population[best_idx] = best_h.copy()
                fitness[best_idx] = best_c5
        
        # Early stopping based on no improvement
        if improved:
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement >= max_no_improvement:
                break
    
    return (best_h.tolist(), best_c5, n_points)
