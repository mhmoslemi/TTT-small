import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Iterative convex optimization with line search
    max_iter = 100
    alpha = 0.5  # Line search step size
    tol = 1e-5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute current correlation maximum for line search
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlation) * dx
        
        # Formulate and solve convex subproblem
        h_var = cp.Variable(n_points)
        c_var = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(h_var) == n_points / 2]
        constraints += [0 <= h_var[i] for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        
        # Linearize the cross-correlation constraints
        # h_i * (1 - h_{i+k}) is approximated linearly at current h
        for k in range(1, n_points):
            # For full correlation, consider all valid shifts
            # k corresponds to a right shift of k points
            # We are adding the constraint that:
            # sum_{i=0}^{n-k-1} h_i * (1 - h_{i+k}) <= c
            # Which is: sum h_i - sum h_i * h_{i+k} <= c
            # Linearized using current solution h:
            # sum h_i - sum h_i * h_{i+k}^{old} <= c
            # So: sum h_i - sum h_i * h_{i+k}^{old} <= c
            # implies sum h_i * h_{i+k}^{old} - sum h_i >= -c
            # or: sum h_i * h_{i+k}^{old} <= sum h_i - c
            
            # Alternative: directly approximate correlation
            # sum_{i=0}^{n-1} h_i * (1 - h_{i+k}) <= c
            # sum h_i - sum h_i * h_{i+k} <= c
            # sum h_i * h_{i+k} >= sum h_i - c 
            
            # In terms of variables:
            # sum_{i=0}^{n-k-1} h_i * (1 - h_{i+k}) <= c
            # sum h_i - sum_{i=0}^{n-k-1} h_i * h_{i+k} <= c
            
            # This becomes a constraint about the
            # sum h_i * h_{i+k} vars we need to define.
            # For solver this, use quadratic constraint wp one convex relaxation.
            # Better: Use linear approximation
            # Approximate sum h_i * (1 - h_{i+k}) <= c
            # sum h_i - sum h_i * h_{i+k} <= c
            # sum h_i * h_{i+k} >= sum h_i - c
            
            # So approximate h_i * h_{i+k} as h_i * h_old_{i+k}  
            # similarly for other k values...
            
            # Actually, let's instead do a direct constraint with shifted sums:
            corr_term = 0
            # sum_{i=0}^{n-k-1} h_i * (1 - h_{i+k})
            # = sum_{i=0}^{n-k-1} h_i - sum_{i=0}^{n-k-1} h_i * h_{i+k}
            for i in range(n_points - k):
                corr_term += h_var[i] - h_var[i] * h[i+k]  # note: h is old value
            
            constraints.append(corr_term <= c_var)
            
        # Objective
        objective = cp.Minimize(c_var)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False, eps=1e-4)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                c5_new = c_var.value
            else:
                # Fallback for infeasible or unbounded
                h_new = h.copy()
                c5_new = c5_current
                
        except:
            h_new = h.copy()
            c5_new = c5_current
            
        # Line search
        step_size = alpha
        for _ in range(10):  # At most 10 backtracks
            h_candidate = h.copy()
            h_candidate += step_size * (h_new - h)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_test = np.max(corr) * dx
            
            if c5_test < c5_current:
                break
            step_size *= 0.5
            
        h = h_candidate.copy()
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h.copy()

        # Early stopping if no improvement and close to convergence
        if abs(c5_current - c5_test) < tol:
            break
            
    return (best_h.tolist(), best_c5, n_points)
