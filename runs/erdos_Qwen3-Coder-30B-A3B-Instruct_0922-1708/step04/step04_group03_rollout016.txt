import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Rescale to meet constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # DE parameters
    pop_size = 50
    F = 0.8
    CR = 0.9
    max_iter = 500
    
    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        h_random = np.random.uniform(0, 1, n_points)
        h_random = h_random * (n_points / 2) / np.sum(h_random)
        h_random = np.clip(h_random, 0, 1)
        population.append(h_random)
    
    # Evaluate initial population
    fitness = []
    for h_pop in population:
        corr = correlate(h_pop, 1 - h_pop, mode='full')
        fit = np.max(corr) * dx
        fitness.append(fit)
    
    # Find best individual
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    start_time = time.time()
    no_improvement = 0
    max_no_improvement = 100
    
    # Main DE loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        new_population = []
        new_fitness = []
        
        for i in range(pop_size):
            # Select three different individuals
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            
            # Mutation
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (n_points / 2) / np.sum(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            
            # Evaluate trial
            corr = correlate(trial, 1 - trial, mode='full')
            trial_fitness = np.max(corr) * dx
            
            # Selection
            if trial_fitness < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_fitness)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
                
            # Update best
            if trial_fitness < best_fitness:
                best_h = trial.copy()
                best_fitness = trial_fitness
                no_improvement = 0
            else:
                no_improvement += 1
        
        # Update population
        population = new_population
        fitness = new_fitness
        
        # Local refinement every 20 iterations
        if iteration % 20 == 0 and iteration > 0:
            h_local = best_h.copy()
            for _ in range(20):
                corr = correlate(h_local, 1 - h_local, mode='full')
                current_max = np.max(corr) * dx
                
                # Gradient estimation
                grad = np.zeros(n_points)
                eps = 1e-4
                for j in range(n_points):
                    h_plus = h_local.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - current_max) / eps
                
                # Update with small step
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h_local - lr * grad
                
                # Ensure constraint
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Accept improvement
                corr_new = correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr_new) * dx
                
                if c5_new < best_fitness:
                    h_local = h_new.copy()
                    best_fitness = c5_new
                    no_improvement = 0
                    best_h = h_local.copy()
                else:
                    break
                    
            # Update best in population
            for idx, pop_h in enumerate(population):
                if np.allclose(pop_h, best_h, atol=1e-10):
                    break
            else:
                # Replace worst individual
                worst_idx = np.argmax(fitness)
                population[worst_idx] = best_h.copy()
                corr = correlate(best_h, 1 - best_h, mode='full')
                fitness[worst_idx] = np.max(corr) * dx
        
        if no_improvement >= max_no_improvement:
            break
    
    return (best_h.tolist(), best_fitness, n_points)
