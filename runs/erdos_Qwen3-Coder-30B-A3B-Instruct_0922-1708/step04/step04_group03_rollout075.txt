import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation using CVXPY
    # Variables
    h_vars = cp.Variable(n_points)
    
    # Constraints
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
    
    # Objective: minimize sum of all overlaps (relaxed)
    # This makes the problem convex
    total_overlap = 0
    for k in range(n_points):
        # For each shift k, compute sum of h[i] * (1 - h[(i+k)%n])
        shift_sum = 0
        for i in range(n_points):
            shift_sum += h_vars[i] * (1 - h_vars[(i + k) % n_points])
        total_overlap += shift_sum
    
    # Solve relaxed problem
    prob = cp.Problem(cp.Minimize(total_overlap), constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = np.array(h_vars.value).reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass  # Fall back to previous initialization if optimization fails

    # Phase 2: Projected gradient descent
    def compute_overlap(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    max_no_improvement = 50
    no_improvement_count = 0
    iteration = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current value
        current_c5 = compute_overlap(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient via finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update with gradient descent
        h_new = h - lr * grad
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        new_c5 = compute_overlap(h_new)
        if new_c5 < current_c5:
            h = h_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Reduce learning rate if stuck
            lr *= 0.99
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Occasionally restart with better point
        if iteration % 100 == 0 and iteration > 0:
            # Try a random perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            h_try = h + perturbation
            h_try = np.clip(h_try, 0, 1) 
            h_try = h_try * (n_points / 2) / np.sum(h_try)
            try_c5 = compute_overlap(h_try)
            if try_c5 < best_c5:
                h = h_try
                best_c5 = try_c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
