import time

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points

    else:
        h_init = np.ones(n_points) * 0.5
    
    # Scale to meet integral constraint
    h = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Warm start with a few gradient steps
    for i in range(50):
        if time.time() - start_time > budget_s - 5:
            break
        # Gradient descent with finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            grad[j] = (c5_plus - c5) / eps
        
        # Update
        lr = 0.001
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Track best
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Main improvement loop

    final_c5 = np.max(corr) * dx
