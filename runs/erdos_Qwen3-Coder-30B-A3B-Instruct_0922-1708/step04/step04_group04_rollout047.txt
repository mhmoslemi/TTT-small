import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h = np.clip(h_init, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # More targeted approach using convex optimization
    # Formulate as a convex problem that's close to the original
    # Variable: h[i], C
    # minimize C
    # subject to sum(h) = n/2
    #         h[i] in [0,1]
    #         for each shift k, sum(h[i] * (1 - h[i+k])) <= C
    # This is a non-convex constraint due to h[i]*h[i+k] terms
    # We will use the fact that we can overestimate with linear constraints
    
    # But let's simplify approach: keep our core strategy but focus
    # on improvements:
    # 1. Better initialization
    # 2. Direct convex optimization on relaxed version 
    # 3. Local refinement
    
    # Reinitialize for a cleaner start
    h = h.copy()

    # Try optimizing with CVXPY
    # DCP approach: treat max as variable and linearize constraints
    try:
        # Try our improved optimization approach
        max_iter = 100
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Define variables
            h_vars = cp.Variable(n_points)
            C = cp.Variable()
            
            # Constraints
            constraints = [
                cp.sum(h_vars) == n_points / 2.0,
                h_vars >= 0,
                h_vars <= 1,
                C >= 0
            ]
            
            # Overestimate constraints for all shifts
            for k in range(1, n_points):
                # For shift k, we need to compute sum h[i] * (1 - h[i+k])
                if k < n_points:
                    expr = 0
                    for i in range(n_points-k):
                        expr += h_vars[i] * (1 - h_vars[i+k])
                    # We'll linearize smoothe with quadratic overestimate approach
                    # Constraint: expr <= C
                    constraints.append(expr <= C)
            
            # Objective
            objective = cp.Minimize(C)
            
            # Formulate and solve
            prob = cp.Problem(objective, constraints)
            
            # Try different solvers
            try:
                prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            except:
                try:
                    prob.solve(solver=cp.SCS, verbose=False, max_iters=100)
                except:
                    break  # If we can't solve, fallback to continued local search
                    
            if prob.status == cp.OPTIMAL:
                # Extract solution
                h_opt = h_vars.value
                if h_opt is not None:
                    h_opt = np.maximum(0, np.minimum(1, h_opt))
                    # Rescale to match constraint
                    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                    
                    # Evaluate
                    correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
                    c5 = np.max(correlation) * dx
                    
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
                        
                    # Local refinement step
                    step_size = 0.001
                    for _ in range(20):
                        if time.time() - start_time > budget_s - 1:
                            break
                        # Simple gradient descent on correlations 
                        grad = np.zeros_like(h_opt)
                        eps = 1e-5
                        for i in range(n_points):
                            h_plus = h_opt.copy()
                            h_plus[i] += eps
                            h_plus = np.clip(h_plus, 0, 1)
                            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                            c5_plus = np.max(corr_plus) * dx
                            grad[i] = (c5_plus - c5) / eps
                        
                        h_opt_new = h_opt - step_size * grad
                        h_opt_new = np.clip(h_opt_new, 0, 1)
                        h_opt_new = h_opt_new * (n_points / 2) / np.sum(h_opt_new)
                        
                        corr = np.correlate(h_opt_new, 1 - h_opt_new, mode='full')
                        c5_new = np.max(corr) * dx
                        
                        if c5_new < c5:
                            h_opt = h_opt_new
                            c5 = c5_new
                        else:
                            break

                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()

    except Exception as e:
        # Fall back to old method
        pass
    
    # Final refinement
    h = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
    
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-4)
        h_new = h - lr * grad
        
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        
        if np.abs(c5 - c5_new) < 1e-10:
            break
            
    return (best_h.tolist(), best_c5, n_points)
