import numpy as np
from scipy.stats import qmc
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Latin Hypercube Sampling
    sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs_samples = sampler.random(n=50)
    
    # Convert to array in [0,1]^n_points, then ensure sum constraint
    candidates = []
    for i in range(len(lhs_samples)):
        candidate = lhs_samples[i]
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        candidates.append(candidate)
    
    # Add the parent solution if it exists
    if 'initial_h_values' in globals():
        candidates.append(h)
    
    # Function to compute C5
    def compute_c5(h_val):
        correlation = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(correlation) * dx
    
    # Gradient descent refinement for each candidate
    for i, candidate in enumerate(candidates):
        if time.time() - start_time > budget_s - 1:
            break
            
        h = candidate.copy()
        c5 = compute_c5(h)
        
        # Gradient descent
        max_iter = 500
        lr = 0.01
        eps = 1e-5
        min_lr = 1e-4
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient via finite differences
            grad = np.zeros(n_points)
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                h_minus = h.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                c5_plus = compute_c5(h_plus)
                c5_minus = compute_c5(h_minus)
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = compute_c5(h_new)
            
            # Adaptive learning rate
            if c5_new < c5:
                c5 = c5_new
                h = h_new
            else:
                lr = max(lr / 2, min_lr)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            if iteration > 10 and abs(c5 - c5_new) < 1e-6:
                break
    
    # Final refinement with standard gradient descent
    h = best_h.copy()
    max_iter = 200
    lr = 0.01
    eps = 1e-5
    min_lr = 1e-5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        c5 = compute_c5(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[j] = (c5_plus - c5) / eps
        
        # Update
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < c5:
            c5 = c5_new
            h = h_new
        else:
            lr = max(lr / 2, min_lr)
            
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if iteration > 10 and abs(c5 - c5_new) < 1e-6:
            break
    
    return (best_h.tolist(), best_c5, n_points)
