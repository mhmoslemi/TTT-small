import numpy as np
from scipy.stats import qmc
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from the parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    lhs_samples = 50
    max_grad_steps = 500
    eps = 1e-5
    
    # Create Latin Hypercube Samples
    sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs = sampler.random(n=lhs_samples)
    
    # Transform LHS to [0,1]^n_points and ensure sum constraint
    candidates = []
    for i in range(lhs_samples):
        candidate = lhs[i]
        # Normalize to [0,1] and then rescale to sum to n_points/2
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * (n_points / 2) / np.sum(candidate) 
        candidates.append(candidate.copy())
    
    # Add initial h if it satisfies constraints
    if np.abs(np.sum(h) - n_points/2) < 1e-5:
        candidates.append(h.copy())
    
    # Evaluate each candidate with gradient descent
    for i, candidate in enumerate(candidates):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_current = candidate.copy()
        c5_current = float('inf')
        
        # Gradient descent steps
        lr = 0.01
        for iter_step in range(max_grad_steps):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current C5
            correlation = np.correlate(h_current, 1 - h_current, mode='full')
            c5_current_iter = np.max(correlation) * dx
            
            if c5_current_iter < c5_current:
                c5_current = c5_current_iter
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h_current.copy()
                    
            # Compute gradient via finite differences
            grad = np.zeros(n_points)
            for j in range(n_points):
                # Perturb in both directions
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h_current.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[j] = (c5_plus - c5_minus) / (2.0 * eps)
            
            # Update with adaptive learning rate
            h_new = h_current - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_current = h_new
            
            # Decrease learning rate if no improvement for a while
            if iter_step % 50 == 0 and iter_step > 0:
                lr = max(1e-4, lr * 0.5)
                
    return (best_h.tolist(), best_c5, n_points)
