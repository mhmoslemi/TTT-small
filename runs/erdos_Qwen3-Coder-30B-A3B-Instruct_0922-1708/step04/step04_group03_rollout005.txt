import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation using CVXPY
    # Create a relaxed version to get good initial point
    h_relaxed = h.copy()
    
    # Use CVXPY to solve a relaxed version
    try:
        # Set up a simplified convex optimization to find a good starting point
        h_cvx = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum(cp.multiply(h_cvx, 1 - h_cvx)))
        constraints = [
            cp.sum(h_cvx) == n_points / 2,
            h_cvx >= 0,
            h_cvx <= 1
        ]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_relaxed = np.array(h_cvx.value).flatten()
            h_relaxed = np.clip(h_relaxed, 0, 1)
            h_relaxed = h_relaxed * (n_points / 2) / np.sum(h_relaxed)
    except:
        pass # If CVXPY fails, continue with original h
    
    # Phase 2: Local refinement with projected gradient descent
    h = h_relaxed.copy()
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Project onto constraints
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Gradient descent with adaptive learning rate
    max_iter = 1000
    lr = 0.01
    patience = 50
    no_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_old = compute_c5(h)
        
        # Compute finite difference gradient
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_old) / eps

        # Update with projected gradient descent
        h_new = h - lr * grad
        h_new = project_h(h_new)
        h = h_new
        
        # Update best solution
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            lr *= 0.9
            
        # Early stopping
        if no_improvement >= patience:
            break
    
    # Final refinement if better solution found
    if best_c5 < compute_c5(h):
        h = best_h.copy()
        
    correlation = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(correlation) * dx
        
    return (best_h.tolist(), best_c5, n_points)
