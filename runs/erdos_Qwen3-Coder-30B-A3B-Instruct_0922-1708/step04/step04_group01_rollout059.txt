import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Grid search setup
    coarse_points = 20
    fine_points = 100
    dx_coarse = 2.0 / coarse_points
    dx_fine = 2.0 / fine_points
    
    # Initial solution from parent or uniform distribution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        if len(h_start) != coarse_points:
            x_orig = np.linspace(0, 2, len(h_start))
            x_target = np.linspace(0, 2, coarse_points)
            h_start = np.interp(x_target, x_orig, h_start)
    else:
        h_start = np.ones(coarse_points) * 0.5
    
    h_start = h_start * (coarse_points / 2) / np.sum(h_start)
    
    # Grid search
    best_c5 = float('inf')
    best_h = h_start.copy()
    start_time = time.time()
    
    # Sample grid points in [0,1]^(coarse_points) space
    # We want sum(h) = coarse_points / 2 = 10
    n_grid_samples = 1000
    grid_samples = []
    
    while len(grid_samples) < n_grid_samples and time.time() - start_time < budget_s - 200:
        h_sample = np.random.rand(coarse_points)
        h_sample = h_sample * (coarse_points / 2) / np.sum(h_sample)
        grid_samples.append(h_sample)
        
    # Evaluate each grid sample
    best_grid_h = None
    best_grid_c5 = float('inf')
    
    for h in grid_samples:
        h_clipped = np.clip(h, 0, 1)
        correlation = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        c5 = np.max(correlation) * dx_coarse
        if c5 < best_grid_c5:
            best_grid_c5 = c5
            best_grid_h = h_clipped.copy()
            
    # Refinement phase with high resolution
    # Upsample the best coarse solution
    h_refine = np.interp(np.linspace(0, 2, fine_points), 
                         np.linspace(0, 2, coarse_points), 
                         best_grid_h)
    
    best_h_fine = h_refine.copy()
    best_c5_fine = best_grid_c5
    
    # Gradient-based refinement
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx_fine
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec * (fine_points / 2) / np.sum(h_vec)
        
    # Apply many gradient steps
    lr = 0.01
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute approximate gradient
        grad = np.zeros_like(h_refine)
        eps = 1e-5
        c5_base = compute_c5(h_refine)
        
        for i in range(fine_points):
            h_plus = h_refine.copy()
            h_plus[i] += eps
            h_plus_proj = project_h(h_plus)
            c5_plus = compute_c5(h_plus_proj)
            grad[i] = (c5_plus - c5_base) / eps
            
        # Update with gradient
        h_new = h_refine - lr * grad
        h_refine = project_h(h_new)
        
        # Check if improved
        current_c5 = compute_c5(h_refine)
        if current_c5 < best_c5_fine:
            best_c5_fine = current_c5
            best_h_fine = h_refine.copy()
            
    # Final check with CVX
    try:
        h_vars = cp.Variable(fine_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == fine_points / 2]
        # Objective is max of convolution - we'll compute it manually rather than using cvxpy's conv
        correlation = np.correlate(best_h_fine, 1 - best_h_fine, mode='full')
        c5_final = np.max(correlation) * dx_fine
    except:
        c5_final = compute_c5(best_h_fine)

    return (best_h_fine.tolist(), c5_final, fine_points)
