import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation approach
    # Minimize sum of all overlaps (not just max)
    # Use CVXPY for this
    h_var = cp.Variable(n_points)
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # Use the fact that we want to minimize max correlation
    # So approximate: min sum of all terms
    # This is a convex relaxation - best we can do with convex programming
    
    # For simplicity, we use a gradient-based refinement from the relaxed solution
    # But since CVXPY can solve the convex relaxation, let's see what we get
    
    # Relaxation: minimize trace
    # sum_i h[i] * (sum_k h[(i+k) mod n] * 1)  
    # This is still hard, so better approach: minimize trace of one k
    # Actually an even simpler relaxation: min sum(h[i] * (1 - h[i]))
    # But that is wrong. Let's just use a proper relaxation:
    
    # For now, do a gradient descent type approach from the initial
   
    # Gradient descent refined approach
    max_iter = 5000
    lr = 0.01
    patience = 50
    
    current_c5 = float('inf')
    no_improvement = 0
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
        
        # Compute gradient using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            # Perturb h[i] forward
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update step
        h_new = h - lr * grad
        
        # Enforce constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
