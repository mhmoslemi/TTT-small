import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points

    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Relaxation approach to get a starting point
    try:
        # Define Variable
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # We need to bound the maximum overlap
        # Let's make it an explicit constraint using a slack variable
        C = cp.Variable()
        
        # Loop over lags to enforce them
        lags = list(range(1, n_points))
        for k in lags:
            # Convolution constraint (approximated in discrete form)
            # Here we compute a few key lags to reduce complexity
            if k >= n_points // 2:
                continue
            corr_k = cp.sum(cp.multiply(h_vars[:-k], 1 - h_vars[k:]))
            constraints.append(corr_k <= C * dx)
            
        # Objective
        obj = cp.Minimize(C)
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_relaxed = h_vars.value
            # Round to binary with thresholding
            threshold = np.median(h_relaxed)
            h_rounded = (h_relaxed > threshold).astype(float)
            
            # Ensure sum constraint
            current_sum = np.sum(h_rounded)
            diff = current_sum - n_points / 2
            if diff > 0:
                # Need to reduce ones
                indices = np.where(h_rounded == 1)[0]
                np.random.shuffle(indices)
                for i in range(int(diff)):
                    h_rounded[indices[i]] = 0
            elif diff < 0:
                # Need to increase zeros
                indices = np.where(h_rounded == 0)[0]
                np.random.shuffle(indices)
                for i in range(int(-diff)):
                    h_rounded[indices[i]] = 1
                    
            h = h_rounded
    except:
        pass
    
    iteration = 0
    # Local search
    while time.time() - start_time < budget_s - 1:
        iteration += 1  
        
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Random small perturbation
        h_new = h.copy()
        
        # Try perturb 10 random indices
        indices = np.random.choice(n_points, 10, replace=False)
        for i in indices:
            delta = np.random.uniform(-0.02, 0.02)
            h_new[i] += delta
            
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < c5:
            h = h_new
            
    # Final check with refined optimization / multi-starts
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        correlation = np.correlate(h_init, 1 - h_init, mode='full')
        c5_current = np.max(correlation) * dx
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_init.copy()

    return (best_h.tolist(), best_c5, n_points)
