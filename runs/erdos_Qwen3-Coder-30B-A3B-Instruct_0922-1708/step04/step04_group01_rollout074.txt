import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Coarse grid search
    coarse_size = 20
    coarse_candidates = []
    
    # Generate a small set of coarse candidates for exploration
    for i in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        h_c = np.random.rand(coarse_size)
        h_c = h_c * (coarse_size / 2) / np.sum(h_c)
        
        # Interpolate to fine grid
        h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, coarse_size), h_c)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fine.copy()
    
    # Refinement using CVXPY
    try:
        for i in range(50):  # Limited iterations
            if time.time() - start_time > budget_s - 5:
                break
                
            h_cvx = np.random.rand(n_points)
            h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
            
            # We will solve a simpler version of the problem using direct optimization
            # by making a few iterations with simple SGD-like steps
            for it in range(100):
                if time.time() - start_time > budget_s - 5:
                    break
                    
                # Gradient approximation
                eps = 1e-4
                grad = np.zeros_like(h_cvx)
                
                for j in range(n_points):
                    h_test = h_cvx.copy()
                    h_test[j] += eps
                    h_test = np.clip(h_test, 0, 1)
                    h_test = h_test * (n_points / 2) / np.sum(h_test)
                    
                    corr_test = np.correlate(h_test, 1 - h_test, mode='full')
                    c5_test = np.max(corr_test) * dx
                    
                    h_test = h_cvx.copy()
                    h_test[j] -= eps
                    h_test = np.clip(h_test, 0, 1)
                    h_test = h_test * (n_points / 2) / np.sum(h_test)
                    
                    corr_test2 = np.correlate(h_test, 1 - h_test, mode='full')
                    c5_test2 = np.max(corr_test2) * dx
                    
                    grad[j] = (c5_test - c5_test2) / (2 * eps)
                
                # Update
                lr = 0.005
                h_cvx = h_cvx - lr * grad
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
            # Evaluate
            correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cvx.copy()
                
    except Exception as e:
        pass
    
    # Final evaluation and return
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx

    return (best_h.tolist(), final_c5, n_points)
