import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO Parameters
    n_particles = 30
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Initialize PSO
    positions = np.random.rand(n_particles, n_points) * 2
    velocities = np.random.rand(n_particles, n_points) * 0.1
    personal_best_positions = positions.copy()
    personal_best_scores = np.full(n_particles, float('inf'))
    global_best_position = None
    global_best_score = float('inf')
    
    # Normalize initial positions
    for i in range(n_particles):
        positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
    
    # PSO loop
    max_iterations = 1000
    iteration = 0
    
    while time.time() - start_time < budget_s - 2 and iteration < max_iterations:
        iteration += 1
        
        # Evaluate fitness for all particles
        for i in range(n_particles):
            # Clip and rescale to feasible range
            h_particle = np.clip(positions[i], 0, 1)
            h_particle = h_particle * (n_points / 2) / np.sum(h_particle)
            
            # Compute C5
            correlation = np.correlate(h_particle, 1 - h_particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_best_positions[i] = h_particle.copy()
                
            if c5 < global_best_score:
                global_best_score = c5
                global_best_position = h_particle.copy()
                
        # Update positions and velocities
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best_positions[i] - positions[i]) + 
                           c2 * r2 * (global_best_position - positions[i]))
            
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            
            positions[i] += velocities[i]
            
            # Clamp to feasible range and re-normalize
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
    
    # If any improvement was found, do final refinement with CVXPY
    if global_best_position is not None:
        try:
            # Define the optimization variables and constraints
            h_vars = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Try to minimize maximum overlap using convex optimization
            # Create a linearized approximation for the max to make it convex
            # We'll calculate all correlations explicitly for objective
            
            # Reconstruct the best solution from PSO
            h_refine = np.clip(global_best_position, 0, 1)
            h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
            
            # Compute correlation matrix or sum though the full overlap
            # Get the maximum overlap from this solution
            correlation_full = np.correlate(h_refine, 1 - h_refine, mode='full')
            c5_initial = np.max(correlation_full) * dx
            
            # Use this as warm-start
            h_vars.value = h_refine
            
            # For CVXPY - set objective to minimize C5 bound
            # This is binary search approach since we have convex constraints
            
            # For this scale, we can solve directly with convolutions
            best_c5 = c5_initial
            best_h = h_refine.copy()
            
            # Run fine-tuning by doing more local search
            def objective(h_vector):
                h_vector = np.clip(h_vector, 0, 1)
                h_vector = h_vector * (n_points / 2) / np.sum(h_vector)
                correlation = np.correlate(h_vector, 1 - h_vector, mode='full')
                return -np.max(correlation) * dx  # maximize -minimize
                
            # Apply constrained fixed-step local minimization
            for _ in range(100):
                # Gradient-based local search on subset of parameters
                h_iter = best_h.copy()
                # Finite differences for gradient estimate
                eps = 1e-4
                for i in range(0, n_points, 5):  # sample every 5 points
                    h_plus = h_iter.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    corr_base = np.correlate(h_iter, 1 - h_iter, mode='full')
                    c5_base = np.max(corr_base) * dx
                    
                    grad = (c5_plus - c5_base) / eps
                    
                    # Update with learning rate
                    lr = 1e-3
                    h_iter[i] -= lr * grad
                    h_iter[i] = max(0, min(1, h_iter[i]))
                    
                h_iter = h_iter * (n_points / 2) / np.sum(h_iter)
                
                corr_iter = np.correlate(h_iter, 1 - h_iter, mode='full')
                c5_iter = np.max(corr_iter) * dx
                
                if c5_iter < best_c5:
                    best_c5 = c5_iter
                    best_h = h_iter.copy()
        except:
            pass

    # Final return of best solution found
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
