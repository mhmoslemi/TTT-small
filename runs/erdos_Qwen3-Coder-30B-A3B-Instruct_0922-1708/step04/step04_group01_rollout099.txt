import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Gradient-based optimization with perturbations
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Coarse grid search to find promising areas
    coarse_points = 20
    coarse_grid = np.linspace(0, 1, coarse_points)
    best_candidates = []
    
    # Sample coarse candidates
    for _ in range(50):
        if time.time() - start_time > budget_s - 20:
            break
        h_coarse = np.random.choice(coarse_grid, size=coarse_points)
        h_coarse = h_coarse * (coarse_points / 2) / np.sum(h_coarse)
        
        # Interpolate to full resolution
        x_coarse = np.linspace(0, 2, coarse_points)
        x_full = np.linspace(0, 2, n_points)
        h_full = np.interp(x_full, x_coarse, h_coarse)
        
        # Ensure it's feasible
        h_full = np.clip(h_full, 0, 1)
        h_full = h_full * (n_points / 2) / np.sum(h_full)
        
        c5 = compute_c5(h_full)
        best_candidates.append((h_full.copy(), c5))
    
    # Sort by C5 value and take top 10
    best_candidates.sort(key=lambda x: x[1])
    top_candidates = best_candidates[:10]
    
    # Refine top candidates
    for h_candidate, _ in top_candidates:
        if time.time() - start_time > budget_s - 20:
            break
            
        h_refined = h_candidate.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 20:
                break
            # Compute gradient using 10-step finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c5_plus = compute_c5(h_plus)
                c5 = compute_c5(h_refined)
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update step
            lr = 0.01
            h_new = h_refined - lr * grad
            
            # Maintain constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_refined = h_new
        
        c5_final = compute_c5(h_refined)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_refined.copy()
    
    # Final refinement using CVXPY if time allows
    try:
        if time.time() - start_time < budget_s - 50:
            # At this point, h_best should be a good solution
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Use finite difference approximations to construct objective
            # Approximate max_k int h(x)(1-h(x+k)) dx using empirical maximum
            # This is a rough approximation, we'll minimize max over few computed lags            
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (n_points, 1)))))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_cvx = h_vars.value
                if h_cvx is not None:
                    h_cvx = np.clip(h_cvx, 0, 1)
                    h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                    
                    correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                    c5_cvx = np.max(correlation) * dx
                    
                    if c5_cvx < best_c5:
                        best_h = h_cvx
                        best_c5 = c5_cvx
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
