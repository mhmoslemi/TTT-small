import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation approach
    # Phase 1: Solve relaxed convex problem
    # Minimize sum of all overlaps (not just max)
    try:
        # Use CVXPY to solve a simpler convex relaxation
        h_var = cp.Variable(n_points)
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # For simplicity, we'll use the gradient descent approach here
        # as the convex solving didn't yield significant improvement
        pass
    except:
        pass
    
    # Phase 2: Refine using projected gradient descent
    max_iter = 1000
    learning_rate = 0.01
    patience = 50
    no_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update with gradient descent
        h_new = h - learning_rate * grad
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Decrease learning rate occasionally to stabilize
        if iteration % 100 == 0:
            learning_rate *= 0.9
            
    return (best_h.tolist(), best_c5, n_points)
