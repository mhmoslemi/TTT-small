import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient-based optimization with restarts
    max_iter = 1000
    max_no_improvement = 100
    patience = 0
    
    # Try different initialization strategies
    initial_strategies = [
        lambda: h.copy(),
        lambda: np.random.uniform(0, 1, n_points),
        lambda: np.ones(n_points) * 0.5,
    ]
    
    for strategy_idx, strategy in enumerate(initial_strategies):
        if time.time() - start_time > budget_s - 1:
            break
            
        h = strategy()
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        
        for iter in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                patience = 0
            else:
                patience += 1
                
            if patience > max_no_improvement:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update step
            lr = 0.001 / (1.0 + iter / 100.0)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
    # Try some additional modifications
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random small perturbations
        h_perturb = best_h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h_perturb[i] += np.random.normal(0, 0.01)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
        
        correlation = np.correlate(h_perturb, 1 - h_perturb, mode='full')
        c5_new = np.max(correlation) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_perturb.copy()
    
    return (best_h.tolist(), best_c5, n_points)
