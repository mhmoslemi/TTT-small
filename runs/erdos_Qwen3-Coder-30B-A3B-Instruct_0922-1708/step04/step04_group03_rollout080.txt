import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation using CVXPY
    # Define variables
    h_var = cp.Variable(n_points)
    
    # Objective: minimize sum of overlaps (approximation for first stage)
    # This is a simplified relaxation to get a good starting point
    objective = cp.Minimize(cp.sum_squares(h_var))
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Solve relaxation
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = h_var.value.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        # If convex relaxation fails, continue with existing h
        pass
    
    # Phase 2: Projected gradient descent
    def compute_overlap(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(h_vec, eps=1e-5):
        grad = np.zeros_like(h_vec)
        base_val = compute_overlap(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            grad[i] = (compute_overlap(h_plus) - base_val) / eps
        return grad
    
    # Main optimization loop with projected gradient descent
    max_no_improvement = 50
    no_improvement_count = 0
    learning_rate = 0.01
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        # Compute current overlap
        current_c5 = compute_overlap(h)
        
        # Store best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            improved = True
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update with gradient descent
        h_new = h - learning_rate * grad
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update h for next iteration
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            learning_rate *= 0.9
            
        # Early stopping
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
    # Final check for best solution
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
