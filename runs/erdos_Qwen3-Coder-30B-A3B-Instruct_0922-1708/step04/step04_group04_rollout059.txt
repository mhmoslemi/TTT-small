```python
import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to correct integral
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Iterative convex optimization approach
    max_iter = 100
    tol = 1e-5
    step_size = 0.5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current correlations to get best C5 so far
        corr = correlate(h, 1-h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Formulate convex approximation
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            C >= 0
        ]
        
        # Add linearized cross-correlation constraints
        # For each k, sum h[i] * (1 - h[i+k]) <= C
        for k in range(1, n_points):
            # Compute linearized cross-term constraint
            # We approximate h[i+k] with h_old[i+k] 
            # So constraint becomes sum(h[i] * (1 - h_old[i+k])) <= C
            # Which is sum(h[i]) - sum(h[i] * h_old[i+k]) <= C
            # Which is sum(h[i]) - sum(h_old[i+k] * h[i]) <= C
            # Or in vector form: (1 - h_old_k) * h <= C for valid k
            # For positive k: i range from 0 to n_points - k - 1
            # h_i * (1 - h_{i+k}) <= C becomes h_i - h_i * h_{i+k} <= C
            # Linearized version: h_i - h_old[i+k] * h_i <= C
            
            # But that's still non-convex. So we linearize by using known h_old[i+k],
            # So we want to enforce h_i * (1 - h_old[i+k]) <= C
            # This would be h_i - h_i * h_old[i+k] <= C
            # Then, note that h_old[i+k] is constant, so it's linear.
            # But we have to make sure i+k is within bounds.
            
            #y = np.concatenate([np.zeros(k), h_old[:-k]])
            #correlation = np.correlate(h, y, 'valid')
            #unsafe_code_blocked
            
            # Instead, implement it more carefully:
            # sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k]) <= C
            
            # So we have
            # h[i] - h[i] * h_old[i+k] <= C
            # That's a linear constraint in h if h_old[i+k] is known.
            # Making it a vector: 
            # For all k from 1 to n-1
            #  sum (h[i] - h[i] * h_old[i+k]) for i=0..n-k-1 <= C  
            #  sum h[i] - sum h[i] * h_old[i+k] <= C
            
            if k < n_points:
                # This is valid for index i from 0 to n_points-k-1
                # Corresponding indices in l = i+k from k to n_points-1
                # We break this into vector form
                #  sum(h[0:n-k] * (1 - h_old[k:n_points])) <= C
                # Is that same as:
                # sum(h_acc) - sum(h[i] * h_old[i+k] for i=0..n-k-1) <= C
                
                # Make vectorized:
                weight_vec = np.ones(n_points - k)
                cross_term = h_var[:n_points - k] * h[:n_points - k]  # Mimic h[i] * h_old[i+k]
                # Actually, use h_old[i+k]: 
                const_part = h_var[:n_points - k]  # h[i]
                # For constraint const_part - cross_term <= C
                # Is equivalent to const_part - cross_term - C <= 0 
                # Sum over valid i: 
                # Let's get that from the (0..n-k-1) indices,
                
                # For i=0 to n-k-1: h[i] * (1-h_old[i+k]) = h[i] - h[i] * h_old[i+k]
                # => sum h[i] - sum h[i] * h_old[i+k] <= C
                forward_indices = np.arange(n_points - k)
                backward_indices = np.arange(k, n_points)
                # sum(h_var[i] - h_var[i] * h[backward_indices[i]] for i in 0..n-k-1) <= C
                # Can be written as sum(h_var[i]) - sum(h_var[i] * h[backward_indices[i]]) <= C
                # This is a linear constraint, just that h_var(i) * h_old(i+k) is not linear.
                #
                # So we keep old approach: fix h_old there, linearize:
                # const_term = h[backward_indices]
                # h_term = h_var[forward_indices] 
                # However, cvxpy handles expressions with same variable multiple times as non-convex
                
                # So simply define:
                # Use sum(1 - h_old[i+k] for i in 0..n-k-1) * h_var[i] <= C + constant  
                # But no, not quite.
                
                # Let's not get too invalid
                # We just minimize the overlap by direct formulation:
                # For a fixed h (must be same when we compute corr >= 0)
                # sum(h[i] * (1 - h[i + k])) <= C for all k, i
                # = sum(h[i]) - sum(h[i] * h[i+k]) <= C for all k
                
                # But we are solving for h and trying to minimize C.
                # Indicator that it's still non-convex - the h[i]*h[i+k] inner products.
                
                # Worth trying a convex relaxation:
                # For each pair of indices (i, i+k) with k in [1, n-1]:
                # h[i] * (1 - h_old[i+k]) <= h[i] - h[i] * h_old[i+k]
                # For all k in [1, n-1]:
                # sum_{i=0}^{n-k-1} h[i] - h[i] * h_old[i+k] <= C
                # That would be:
                # sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i] * h_old[i+k] <= C
                # Let's approximate both terms via angle? No need.
                
                # In CVXPY we can have the full form along with scalar variables:
                # A more correct translation:
                # sum_{i=0}^{n-k-1} (h[i] - h[i] * h_old[i+k]) <= C
                # sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i] * h_old[i+k] <= C 
                # sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i] * h_old[i+k] <= C
                # Let's not have h in both sides - instead, use 
                # sum_{i=0}^{n-k-1} h[i] - \sum h[i]*h_old[i+k] <= C
                # C >= sum h[i] - sum h[i]*h_old[i+k]
                # i goes from 0 to n-k-1.
                # Thus:
                # P[i] = 1 if i in [0, n-k-1], 0 otherwise
                # Q[i] = h_old[i+k] if i in [0, n-k-1], 0 otherwise is more clever.
                # Or better yet:
                # Note: x_i = h[i] but h_old[i+k] is fixed, 
                # so this constraint is: \sum_{i=0}^{n-k-1} x_i * (1 - h[i+k])  
                # Will be read back easily by CVXPY: h_var[i] * (1 - h[k+i])
                # But that conflicts with writing h_var[i] * h_var[i+k]
                # But even in Python, (x[i] * (1 - h_old[i+k]))[i=0..n-k-1] is fine.
                
                # Let's avoid two variations of h completely:
                # Instead, a trick:
                # In CVXPY, we define h_var and think of that as the new h.
                # We are approximating the constraint as sum(h_var[i] * (1 - h_old[i+k])) <= C.
                # This is a linear relaxation of sum(h_var[i] * (1 - h_var[i+k])) but we use a known h_old
                
                h_part = h_var[forward_indices]  # i up to n-k-1 
                h_old_part = h[backward_indices]
                cross = h_part * (1 - h_old_part)
                total_corr = cp.sum(h_part) - cp.sum(cross)
                constraints.append(total_corr <= C)
                
                # Old comment trying to inspect the philosophy:
                # We compute a good first estimate to get constraint vals
                # then iterate by changing h while tolerating some fluctuation.
        
        # Try to solve the convex problem:
        obj = cp.Minimize(C)
        prob = cp.Problem(obj, constraints)
        
        # Try Solve with Schulze solver with tolerance
        try:
            prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-4, reltol=1e-4)
        except:
            # Fallback if convex approximation fails
            # Either just continue or try a line-search refinement
            pass
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = h_var.value
            if h_new is not None:
                # Validate feasibility and compute actual C5
                # Project back (non-negative, bounded)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Check if this is an actual improvement
                corr = correlate(h_new, 1 - h_new, mode='full')
                new_c5 = np.max(corr) * dx
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                
                h = h_new
                
                # termination check
                if np.abs(current_c5 - new_c5) < tol:
                    break
            else:
                break  # No solution found from CVXPY, exit
        else:
            # If infeasible or failed to solve, use line search
            # Slightly perturb h
            pert = np.random.normal(0, 0.01, n_points)
            new_h = h + step_size * pert
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            corr = correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(corr) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
            h = new_h
            
            if step_size > 1e-8 and np.abs(current_c5 - new_c5) < tol:
                break
            step_size *= 0.5  # Reduce adaptive step size
    
    # Final evaluation and return
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
```
