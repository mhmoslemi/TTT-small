import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    T_start = 1.0
    decay_factor = 0.999
    perturb_freq = 1000
    perturb_rate = 0.05
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    h = (h_init >= 0.5).astype(int)
    
    # Adjust to have exactly n_points/2 ones
    count_ones = np.sum(h)
    target_ones = n_points // 2
    
    if count_ones > target_ones:
        # Turn some 1s to 0s
        zero_indices = np.where(h == 1)[0]
        indices_to_flip = np.random.choice(zero_indices, count_ones - target_ones, replace=False)
        h[indices_to_flip] = 0
    elif count_ones < target_ones:
        # Turn some 0s to 1s
        one_indices = np.where(h == 0)[0]
        indices_to_flip = np.random.choice(one_indices, target_ones - count_ones, replace=False)
        h[indices_to_flip] = 1
    
    # Compute initial correlation
    corr_full = np.correlate(h, 1 - h, mode='full')
    c5_current = np.max(corr_full) * dx
    best_h = h.copy()
    best_c5 = c5_current
    T = T_start
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select a random index to flip
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]  # Flip the bit
        
        # Project to ensure exactly n_points/2 ones
        count_ones_new = np.sum(h_new)
        if count_ones_new != target_ones:
            if count_ones_new > target_ones:
                # Need to flip some 1s back to 0s
                one_indices = np.where(h_new == 1)[0]
                indices_to_flip = np.random.choice(one_indices, count_ones_new - target_ones, replace=False)
                h_new[indices_to_flip] = 0
            else:
                # Need to flip some 0s to 1s
                zero_indices = np.where(h_new == 0)[0]
                indices_to_flip = np.random.choice(zero_indices, target_ones - count_ones_new, replace=False)
                h_new[indices_to_flip] = 1
        
        # Compute new correlation
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept or reject the change
        if c5_new < c5_current:
            h = h_new
            c5_current = c5_new
        else:
            # Accept with probability based on temperature
            delta_c5 = c5_new - c5_current
            if np.random.rand() < np.exp(-delta_c5 / T):
                h = h_new
                c5_current = c5_new
                
        # Update best solution
        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current
            
        # Decrease temperature
        T *= decay_factor
        
        # Periodic perturbation
        if iteration % perturb_freq == 0 and iteration > 0:
            # Reset a random portion of the bits
            num_reset = int(n_points * perturb_rate)
            reset_indices = np.random.choice(n_points, num_reset, replace=False)
            for idx in reset_indices:
                h[idx] = np.random.randint(0, 2)
                
            # Re-project to exact count
            count_ones = np.sum(h)
            if count_ones > target_ones:
                zero_indices = np.where(h == 1)[0]
                indices_to_flip = np.random.choice(zero_indices, count_ones - target_ones, replace=False)
                h[indices_to_flip] = 0
            elif count_ones < target_ones:
                one_indices = np.where(h == 0)[0]
                indices_to_flip = np.random.choice(one_indices, target_ones - count_ones, replace=False)
                h[indices_to_flip] = 1
    
    # Final computation of the correlation for the best solution
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
