import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure bounds and sum
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated annealing
    temperature = 1.0
    cooling_rate = 0.99
    iterations = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 2:
        iterations += 1
        if iterations % 100 == 0:
            temperature *= cooling_rate
            
        # Generate candidate
        h_new = h.copy()
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for i in indices:
            perturbation = np.random.uniform(-0.1, 0.1)
            h_new[i] += perturbation
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute overlap integral
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Compute current overlap
        correlation_curr = np.correlate(h, 1 - h, mode='full')
        c5_curr = np.max(correlation_curr) * dx
        
        # Accept or reject
        if c5_new < c5_curr or np.random.random() < np.exp(-(c5_new - c5_curr) / temperature):
            h = h_new.copy()
            c5_curr = c5_new
            
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()
    
    # Final refinement with local search around best solution
    h_final = best_h.copy()
    current_c5 = best_c5
    
    # Local search with small random perturbations
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_test = h_final.copy()
        # Make small changes
        for i in range(n_points // 20):
            idx = np.random.randint(0, n_points)
            h_test[idx] += np.random.uniform(-0.01, 0.01)
            
        # Ensure feasibility
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Evaluate
        correlation = np.correlate(h_test, 1 - h_test, mode='full')
        c5_test = np.max(correlation) * dx
        
        if c5_test < current_c5:
            h_final = h_test
            current_c5 = c5_test
    
    return (h_final.tolist(), current_c5, n_points)
