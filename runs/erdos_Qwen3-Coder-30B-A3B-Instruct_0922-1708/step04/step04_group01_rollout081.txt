import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    coarse_grid_size = 20
    top_candidates = 10
    max_refinement_steps = 500
    learning_rate = 0.01
    
    start_time = time.time()
    
    # Initialize the best solution
    best_h = None
    best_c5 = float('inf')
    
    # Start from initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Phase 1: Grid search with coarse resolution
    if time.time() - start_time < budget_s * 0.3:
        # Generate a coarse grid around a few good starting points
        candidates = []
        
        # Add the current solution as a candidate
        candidates.append(h.copy())
        
        # Add random perturbations
        for _ in range(5):
            h_pert = h + np.random.normal(0, 0.1, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            candidates.append(h_pert)
        
        # Add some structured functions
        # Block function
        h_block = np.zeros(n_points)
        h_block[:n_points//2] = 2.0
        h_block = h_block * (n_points / 2) / np.sum(h_block)
        candidates.append(h_block)
        
        # Gaussian-like function
        x = np.linspace(0, 2, n_points)
        h_gauss = np.exp(-((x - 1)**2) / 0.1)
        h_gauss = h_gauss * (n_points / 2) / np.sum(h_gauss)
        candidates.append(h_gauss)
        
        # Evaluate candidates
        for i, candidate_h in enumerate(candidates):
            correlation = np.correlate(candidate_h, 1 - candidate_h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate_h.copy()
    
    # Phase 2: Refinement with high resolution
    if time.time() - start_time < budget_s * 0.8:
        # Refine top candidates from grid search
        for _ in range(top_candidates):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Select a candidate with higher probability to the current best
            if best_h is not None:
                candidate_h = best_h.copy()
            else:
                candidate_h = h.copy()
                
            for step in range(max_refinement_steps):
                if time.time() - start_time > budget_s - 10:
                    break
                
                # Compute gradient manually using finite differences
                grad = np.zeros_like(candidate_h)
                eps = 1e-4
                
                for i in range(n_points):
                    h_plus = candidate_h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    # This function is not smooth, so we compute gradient with respect to one variable at a time.
                    grad[i] = (c5_plus - best_c5) / eps
                
                # Update with gradient step
                candidate_h = candidate_h - learning_rate * grad
                
                # Clip and rescale
                candidate_h = np.clip(candidate_h, 0, 1)
                candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
                
                # Check the new solution
                correlation = np.correlate(candidate_h, 1 - candidate_h, mode='full')
                c5 = np.max(correlation) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = candidate_h.copy()
    
    # Phase 3: Final CVX optimization check
    if best_h is not None and time.time() - start_time < budget_s - 10:
        try:
            h_vars = cp.Variable(n_points)
            
            # Constraints for CVX
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Objective function (minimize maximum overlap)
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_cvx = h_vars.value
                if h_cvx is not None:
                    h_cvx = np.clip(h_cvx, 0, 1)
                    h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                    
                    correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                    c5_cvx = np.max(correlation) * dx
                    
                    if c5_cvx < best_c5:
                        best_h = h_cvx
                        best_c5 = c5_cvx
                        
        except Exception as e:
            pass

    # Final check to make sure we have a valid solution
    if best_h is None:
        best_h = h
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), best_c5, n_points)
