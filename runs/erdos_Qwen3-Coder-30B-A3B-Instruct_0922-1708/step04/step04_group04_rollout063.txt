```python
import numpy as np
import time
from heapq import heappush, heappop
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Initialize priority queue for shifts
    def compute_overlaps(h_vec):
        overlaps = {}
        for k in range(1 - n_points, n_points):
            if k == 0:
                continue
            corr = correlate(h_vec, 1 - h_vec, mode='full')
            overlap = corr[n_points - 1 + k] * dx
            overlaps[k] = overlap
        return overlaps
    
    def compute_c5(h_vec):
        overlaps = compute_overlaps(h_vec)
        return max(overlaps.values()) if overlaps else float('inf')
    
    # Helper to update priority queue with new overlaps
    def update_priority_queue(overlaps, pq):
        pq.clear()
        for k, overlap in overlaps.items():
            heappush(pq, (-overlap, k))
    
    # Core optimization loop
    start_time = time.time()
    
    # Initialize overlaps and priority queue
    overlaps = compute_overlaps(h)
    pq = []
    update_priority_queue(overlaps, pq)
    
    # Control parameters:
    window_width = 5
    perturb_scale = 0.02
    min_improvement = 1e-6
    random_seeds = 5
    max_no_improvement = 10
    refresh_interval = 50
    
    no_improvement_count = 0
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Extract worst shift
        if not pq:
            break
        neg_overlap, k = heappop(pq)
        current_overlap = -neg_overlap
        
        # Find the indices causing maximum overlap for this k
        # Compute correlation in a more precise way
        corr = correlate(h, 1 - h, mode='full')
        # The overlap is corr[n_points - 1 + k] * dx
        overlap_index = n_points - 1 + k
        if overlap_index < 0 or overlap_index >= len(corr):
            # skip invalid index
            continue
            
        # Parameter to control how much to adjust
        adjustment_step = 0.01
        
        # Identify indices that receive adjustment
        # For k>0, h[i] overlaps with (1-h[i+k])
        # For k<0, h[i+k] overlaps with (1-h[i])
        indices = []
        if k > 0:
            indices = list(range(n_points - k))
        elif k < 0:
            indices = list(range(-k, n_points))
        else:
            continue
        # Focus on the region around maximum overlap
        # We can do max projected sum over region
        # For efficiency, just adjust a few key indices
        max_idx = overlap_index - (n_points - 1) if k > 0 else overlap_index - (n_points + k - 1)
        max_idx = max(0, min(max_idx, len(indices) - 1)) if indices else 0
        
        # Narrow focus to window around max overlap
        start_idx = max(0, max_idx - window_width)
        end_idx = min(len(indices), max_idx + window_width + 1)
        focus_indices = indices[start_idx:end_idx]
        
        # Modify h to reduce overlap
        h_new = h.copy()
        for i in focus_indices:
            if k > 0:
                if i + k < n_points:
                    # h[i] overlaps with (1 - h[i+k]) -> pull down h[i]
                    h_new[i] = max(0, h_new[i] - adjustment_step)
                    # Also pull up h[i+k] to prevent it from being too big
                    h_new[i+k] = min(1, h_new[i+k] + adjustment_step*0.5)
            elif k < 0:
                # h[i+k] overlaps with (1 - h[i])
                if i + k >= 0:
                    h_new[i+k] = max(0, h_new[i+k] - adjustment_step)
                    h_new[i] = min(1, h_new[i] + adjustment_step*0.5)
        
        # Ensure constraints are met
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Add some randomness 
        if np.random.random() < 0.7:
            for i in range(n_points):
                if np.random.random() < 0.1:
                    h_new[i] += np.random.normal(0, perturb_scale)
            
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update overlap and queue
        overlaps = compute_overlaps(h_new)
        pq = []
        update_priority_queue(overlaps, pq)
        
        # Check if we should adjust hyperparameters
        if no_improvement_count > max_no_improvement:
            window_width = min(20, window_width + 2)
            perturb_scale *= 2
            no_improvement_count = 0
            
        if iteration % refresh_interval == 0:
            # Recompute overlaps every so often
            overlaps = compute_overlaps(h_new)
            pq = []
            update_priority_queue(overlaps, pq)
            
    # Final validation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
```
