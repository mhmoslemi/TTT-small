import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # PSO parameters
    n_particles = 30
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    vmax = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals() and np.random.rand() < 0.5:
            # Use parent for some particles
            particle_h = h.copy()
        else:
            particle_h = np.random.uniform(0, 1, n_points)
            particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
            particle_h = np.clip(particle_h, 0, 1)
        particles.append(particle_h)
        velocities.append(np.random.uniform(-vmax, vmax, n_points))
        pbest.append(particle_h.copy())
        pbest_c5.append(float('inf'))
    
    gbest = h.copy()
    gbest_c5 = float('inf')
    
    # PSO iterations
    max_iter = 200
    no_improvement = 0
    max_no_improvement = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Compute fitness
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particle.copy()
                pbest_c5[i] = c5
                
            # Update global best
            if c5 < gbest_c5:
                gbest = particle.copy()
                gbest_c5 = c5
                
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
                
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            vel = (inertia * vel + 
                   c1 * r1 * (pbest[i] - particle) + 
                   c2 * r2 * (gbest - particle))
            
            # Clip velocity
            vel = np.clip(vel, -vmax, vmax)
            
            # Update position
            particle += vel
            
            # Enforce constraints
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            velocities[i] = vel
            particles[i] = particle
            
        # Early stopping
        if gbest_c5 < best_c5:
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > max_no_improvement:
                break
    
    # Local refinement with Nelder-Mead on best solution
    def objective(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Use the best solution from PSO as starting point for Nelder-Mead
    h_refine = best_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Create simplex with random perturbations around the best solution
    simplex = [h_refine.copy()]
    for _ in range(n_points):
        perturbed = h_refine + np.random.normal(0, 0.01, n_points)
        perturbed = project_h(perturbed)
        simplex.append(perturbed)
    
    # Simplified Nelder-Mead implementation for local refinement
    max_refine_iter = 1000
    for refine_iter in range(max_refine_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort the simplex by objective value
        simplex_values = [objective(s) for s in simplex]
        indices = np.argsort(simplex_values)
        simplex = [simplex[i] for i in indices]
        simplex_values = sorted(simplex_values)
        
        # Compute centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        reflected = centroid + 2 * (centroid - simplex[-1])
        reflected = project_h(reflected)
        f_reflected = objective(reflected)
        
        if simplex_values[0] <= f_reflected < simplex_values[-2]:
            simplex[-1] = reflected
        elif f_reflected < simplex_values[0]:
            # Expansion
            expanded = centroid + 2 * (reflected - centroid)
            expanded = project_h(expanded)
            f_expanded = objective(expanded)
            if f_expanded < f_reflected:
                simplex[-1] = expanded
            else:
                simplex[-1] = reflected
        else:
            # Contraction
            if f_reflected < simplex_values[-1]:
                contracted = centroid + 0.5 * (simplex[-1] - centroid)
            else:
                contracted = 0.5 * (simplex[0] + simplex[-1])
            contracted = project_h(contracted)
            f_contracted = objective(contracted)
            if f_contracted < simplex_values[-1]:
                simplex[-1] = contracted
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = project_h(simplex[i])
        
        # Update best solution
        current_best = simplex[0]
        correlation = np.correlate(current_best, 1 - current_best, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = current_best.copy()
    
    return (best_h.tolist(), best_c5, n_points)
