import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Trust region optimization parameters
    trust_radius = 0.1
    decay_rate = 0.99
    max_iter = 500
    min_trust_radius = 1e-6
    
    # Core optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Project gradient into trust region
        grad_norm = np.linalg.norm(grad)
        if grad_norm > trust_radius:
            grad = grad * trust_radius / grad_norm
            
        # Update using projected gradient
        h_new = h - grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update trust region
        trust_radius = max(min_trust_radius, trust_radius * decay_rate)
        
        h = h_new
    
    # Final refinement with local optimization
    # Flipping to another method for better local search 
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx  # Negative because we minimize
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    def constraint_bounds(h_flat):
        return np.concatenate([h_flat, 1 - h_flat])
    
    # Use scipy minimize with bounds and constraints
    h_flat = best_h.flatten()
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(objective, h_flat, method='SLSQP', bounds=bounds, 
                         constraints=constraints, options={'maxiter': 100})
        if result.success:
            h_final = result.x.reshape(n_points)
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            correlation = np.correlate(h_final, 1 - h_final, mode='full')
            c5_final = np.max(correlation) * dx
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final
    except:
        pass  # Fall back to previous best
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
