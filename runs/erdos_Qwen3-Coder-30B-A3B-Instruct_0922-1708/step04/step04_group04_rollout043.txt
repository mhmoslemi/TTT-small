import numpy as np
from scipy.stats import latinhypercube
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use Latin Hypercube Sampling for initial candidates
    lhs_samples = 50
    candidates = []
    
    # generate lhs samples
    lhs = latinhypercube.LatinHypercube(n_points, seed=seed)
    samples = lhs.random(lhs_samples)
    
    for i in range(lhs_samples):
        sample = samples[i]
        # transform to [0,1] range and rescale to sum to n_points/2
        h_candidate = sample * 1.0  # scale to [0, 1]
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        candidates.append(h_candidate)
    
    # also add the best solution found so far
    candidates.append(h.copy())
    current_candidates = len(candidates)
    
    for i, candidate in enumerate(candidates):
        h_current = candidate.copy()
        # gradient descent on this one
        for iteration in range(500):
            if time.time() - start_time > budget_s - 1:
                break
            
            # compute correlation
            correlation = np.correlate(h_current, 1 - h_current, mode='full')
            c5 = np.max(correlation) * dx
            
            # compute gradients via finite differences
            grad = np.zeros_like(h_current)
            eps = 1e-5
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h_current.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # adaptive learning rate
            lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-4)
            h_new = h_current - lr * grad
            
            # enforce constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # check improvement
            correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(correlation_new) * dx
            
            h_current = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()
                
            # convergence check
            if abs(c5 - c5_new) < 1e-6:
                break
                
    # refine with a second pass using a local optimizer
    # convert to optimization problem with constraints
    def optimize_function(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx  # negative because we want to minimize
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # optimize with scipy
    constraints = {'type': 'eq', 'fun': constraint_func}
    bounds = [(0, 1) for _ in range(n_points)]
    h_opt = best_h.copy()
    
    try:
        # Use SLSQP method with bounds and constraints 
        res = minimize(optimize_function, h_opt, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100})
        if res.success:
            h_opt = res.x
            corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5_opt = -np.max(corr_opt) * dx
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
