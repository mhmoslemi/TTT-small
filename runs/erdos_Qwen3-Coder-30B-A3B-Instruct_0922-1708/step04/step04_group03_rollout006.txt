import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation using CVXPY
    # We'll formulate and solve a relaxed version with a sum of overlaps
    
    # Create a relaxed version to get good initial solution
    h_relaxed = np.copy(h)
    
    # Solver parameters
    max_iter = 500
    learning_rate = 0.01
    decay_rate = 0.99
    patience = 50
    patience_counter = 0
    
    # Gradient descent loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        correlation = np.correlate(h_relaxed, 1 - h_relaxed, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_relaxed.copy()
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h_relaxed)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_relaxed.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update step
        h_new = h_relaxed - learning_rate * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update
        h_relaxed = h_new
        
        # Adaptive learning rate
        if iter_num % 10 == 0:
            learning_rate *= decay_rate
            
        # Early stopping
        patience_counter += 1
        if c5 < best_c5:
            patience_counter = 0  # Reset counter if improvement
            
        if patience_counter >= patience:
            break
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
