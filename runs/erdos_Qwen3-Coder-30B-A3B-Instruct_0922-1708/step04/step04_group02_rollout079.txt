import numpy as np
from scipy.optimize import minimize
import time
import random
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    population_size = 100
    elite_size = 20
    mutation_rate = 0.1
    
    # Initialize population
    def create_individual():
        if 'initial_h_values' in globals():
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        else:
            # Random feasible solution
            h = np.random.rand(n_points)
        # Normalize to meet constraint
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        return h
    
    # Evaluate fitness (C5)
    def evaluate(individual):
        correlation = np.correlate(individual, 1 - individual, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Create initial population
    population = [create_individual() for _ in range(population_size)]
    fitness = [evaluate(ind) for ind in population]
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    
    # Main evolutionary loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        elite = sorted_population[:elite_size]
        next_population = elite[:]
        
        # Generate offspring
        while len(next_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness, 3)
            parent2 = tournament_selection(population, fitness, 3)
            
            # Crossover
            child1, child2 = sbx_crossover(parent1, parent2, 0.5)
            
            # Mutation
            child1 = polymutate(child1, mutation_rate)
            child2 = polymutate(child2, mutation_rate)
            
            # Repair
            child1 = repair_individual(child1, n_points)
            child2 = repair_individual(child2, n_points)
            
            next_population.extend([child1, child2])
        
        # Trim to exact population size
        next_population = next_population[:population_size]
        
        # Evaluate new population
        fitness = [evaluate(ind) for ind in next_population]
        
        # Update best if necessary
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = next_population[best_idx].copy()
        
        # Update population
        population = next_population
        
        # Gradually reduce mutation rate
        mutation_rate = max(0.01, mutation_rate * 0.999)
    
    # Final gradient refinement on best solution
    try:
        extended_h = np.concatenate([best_h, best_h[:10]])
        # Use scipy.optimize to refine
        def cost(h_vec):
            h = np.copy(h_vec)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            correlation = np.correlate(h, 1 - h, mode='full')
            return -np.max(correlation) * dx
            
        def constraint(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        # Optimize with bounds
        h_init = best_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(cost, h_init, bounds=bounds, 
                         constraints={'fun': constraint, 'type': 'eq'},
                         method='SLSQP', options={'maxiter': 50, 'ftol': 1e-6})
        
        if result.success:
            refined_h = result.x
            refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
            refined_h = np.clip(refined_h, 0, 1)
            refined_c5 = evaluate(refined_h)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    except:
        pass
        
    return (best_h.tolist(), best_c5, n_points)

def tournament_selection(population, fitness, tournament_size):
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx]

def sbx_crossover(parent1, parent2, eta):
    # Simulated Binary Crossover
    child1 = np.copy(parent1)
    child2 = np.copy(parent2)
    u = np.random.rand(len(parent1))
    for i in range(len(parent1)):
        if u[i] <= 0.5:
            if abs(parent1[i] - parent2[i]) > 1e-14:
                beta = 2 * u[i]
                if beta <= 1:
                    alpha = (2 * beta) ** (1/(eta + 1))
                else:
                    alpha = (1/(2*(1-beta)))**(1/(eta + 1))
                
                child1[i] = 0.5 * ((parent1[i] + parent2[i]) - alpha * abs(parent1[i] - parent2[i]))
                child2[i] = 0.5 * ((parent1[i] + parent2[i]) + alpha * abs(parent1[i] - parent2[i]))
    
    # Apply bounds and clip
    child1 = np.clip(child1, 0, 1)
    child2 = np.clip(child2, 0, 1)
    return child1, child2

def polymutate(individual, mutation_rate):
    # Polynomial Mutation
    mutated = np.copy(individual)
    eta = 20
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            u = np.random.rand()
            if u <= 0.5:
                delta = (2*u) ** (1/(eta + 1)) - 1
            else:
                delta = 1 - (2*(1-u)) ** (1/(eta + 1))
            
            mutated[i] = individual[i] + delta
            mutated[i] = np.clip(mutated[i], 0, 1)
    return mutated

def repair_individual(h, n_points):
    # Ensure sum constraint
    h_sum = np.sum(h)
    if abs(h_sum - n_points/2) > 1e-6:
        factor = (n_points/2) / h_sum
        h = h * factor
    h = np.clip(h, 0, 1)
    return h
