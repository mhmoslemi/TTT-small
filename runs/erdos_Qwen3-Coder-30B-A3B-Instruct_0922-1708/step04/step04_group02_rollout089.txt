import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    swarm_size = 100
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        particles.append(np.ones(n_points) * 0.5)  # Default initialization
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    while len(particles) < swarm_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Convert to arrays
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # PSO parameters
    omega = 0.7
    phi_p = 2.0
    phi_g = 2.0
    
    # Best positions tracking
    pbest = particles.copy()
    pbest_fitness = np.full(swarm_size, np.inf)
    gbest_idx = 0
    gbest_fitness = np.inf
    
    # Evaluation of initial particles
    for i in range(swarm_size):
        corr = correlate(particles[i], 1 - particles[i], mode='full')
        fitness = np.max(corr) * dx
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest_idx = i
    
    # Main PSO loop
    iteration = 0
    last_improvement = 0
    converged = False
    
    while time.time() - start_time < budget_s - 1 and not converged:
        iteration += 1
        
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r_p = np.random.rand(n_points)
            r_g = np.random.rand(n_points)
            velocities[i] = (omega * velocities[i] + 
                            phi_p * r_p * (pbest[i] - particles[i]) + 
                            phi_g * r_g * (particles[gbest_idx] - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair constraints
            particles[i] = np.clip(particles[i], 0, 1)
            current_sum = np.sum(particles[i])
            target_sum = n_points / 2
            
            if abs(current_sum - target_sum) > 1e-8:
                diff = current_sum - target_sum
                if diff > 0:
                    # Reduce values from largest elements
                    indices = np.argsort(particles[i])[::-1]
                    for idx in indices:
                        if diff <= 0:
                            break
                        reduce_val = min(particles[i][idx], diff)
                        particles[i][idx] -= reduce_val
                        diff -= reduce_val
                else:
                    # Increase values in smallest elements
                    indices = np.argsort(particles[i])
                    for idx in indices:
                        if diff >= 0:
                            break
                        increase_val = min(1 - particles[i][idx], -diff)
                        particles[i][idx] += increase_val
                        diff += increase_val
            
            # Normalize to ensure correct sum
            particles[i] = particles[i] * target_sum / np.sum(particles[i])
            
            # Evaluate fitness
            corr = correlate(particles[i], 1 - particles[i], mode='full')
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest[i] = particles[i].copy()
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_idx = i
                    last_improvement = iteration
                    
        # Local search every 50 iterations
        if iteration % 50 == 0:
            # Refine top 10% of particles
            top_indices = np.argsort(pbest_fitness)[:swarm_size // 10]
            for i in top_indices:
                h_refine = pbest[i].copy()
                base_fitness = pbest_fitness[i]
                for _ in range(10):
                    # Compute finite differences gradient
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h_refine.copy()
                        h_plus[j] = min(1.0, h_plus[j] + eps)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - base_fitness) / eps
                    
                    # Update with learning rate
                    h_new = h_refine - 0.001 * grad
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    
                    corr_new = correlate(h_new, 1 - h_new, mode='full')
                    new_fitness = np.max(corr_new) * dx
                    
                    if new_fitness < base_fitness:
                        h_refine = h_new
                        base_fitness = new_fitness
                    else:
                        break
                
                # Update personal best if improved
                if base_fitness < pbest_fitness[i]:
                    pbest_fitness[i] = base_fitness
                    pbest[i] = h_refine.copy()
                    if base_fitness < gbest_fitness:
                        gbest_fitness = base_fitness
                        gbest_idx = i
                        last_improvement = iteration
        
        # Rejuvenate worst 10%
        if iteration % 20 == 0:
            worst_indices = np.argsort(pbest_fitness)[-swarm_size // 10:]
            for i in worst_indices:
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[i] = h
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                # Update personal best as well
                corr = correlate(particles[i], 1 - particles[i], mode='full')
                fitness = np.max(corr) * dx
                pbest_fitness[i] = fitness
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_idx = i
        
        # Check for convergence
        if iteration - last_improvement > 100:
            converged = True
    
    # Final refinement of best solution
    best_h = particles[gbest_idx]
    for _ in range(1000):
        grad = np.zeros(n_points)
        eps = 1e-5
        base_corr = correlate(best_h, 1 - best_h, mode='full')
        base_fitness = np.max(base_corr) * dx
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] = min(1.0, h_plus[j] + eps)
            h_plus_sum = np.sum(h_plus)
            h_plus = h_plus * (n_points / 2) / h_plus_sum
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - base_fitness) / eps
        
        best_h = best_h - 0.0001 * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        corr_new = correlate(best_h, 1 - best_h, mode='full')
        new_fitness = np.max(corr_new) * dx
        if new_fitness >= base_fitness:
            break
        base_fitness = new_fitness
    
    # Final verification of C5
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
