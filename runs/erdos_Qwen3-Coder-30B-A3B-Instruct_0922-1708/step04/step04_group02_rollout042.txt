import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_segments = 10
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Resample to n_points
        x_orig = np.linspace(0, 2, len(h_init))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_init)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Uniform distribution
        h = np.ones(n_points) * 0.5
        
    # Convert to sparse representation for optimization
    def sparse_to_dense(pos, h_seg, n_points):
        """
        Convert sparse representation back to dense
        pos: [0, pos1, pos2, ..., 2]
        h_seg: [h1, h2, ..., h_m]
        """
        h_dense = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(len(pos) - 1):
            mask = (x >= pos[i]) & (x < pos[i+1])
            h_dense[mask] = h_seg[i]
        return h_dense

    def compute_c5_sparse(pos, h_seg):
        """Compute C5 for sparse representation"""
        n = len(pos) - 1
        x = np.linspace(0, 2, n_points)
        h_dense = sparse_to_dense(pos, h_seg, n_points)
        correlation = np.correlate(h_dense, 1 - h_dense, mode='full')
        c5 = np.max(correlation) * dx
        return c5

    def optimize_sparse():
        """Optimize using sparse representation"""
        # Initialize positions and heights
        pos = np.linspace(0, 2, n_segments + 1)  # [0, ..., 2]
        h_seg = np.ones(n_segments) * (1.0 / n_segments)  # Height normalization
        
        # Combine into optimization vector
        opt_vars = np.concatenate([pos[1:-1], h_seg])  # Exclude first and last pos (fixed)
        
        def objective(vars):
            pos_update = np.concatenate([[0], vars[:n_segments-1], [2]])
            h_update = np.clip(vars[n_segments-1:], 0, 1)
            # Normalize h_update to satisfy integral constraint
            h_update = h_update * (n_points / 2) / np.sum(h_update * np.diff(pos_update))
            c5 = compute_c5_sparse(pos_update, h_update)
            return c5

        def constraint(vars):
            # Ensure h is within bounds and integral is correct
            pos_update = np.concatenate([[0], vars[:n_segments-1], [2]])
            h_update = np.clip(vars[n_segments-1:], 0, 1)
            # Check if positions are increasing
            pos_increasing = np.all(np.diff(pos_update) > 1e-8)
            # Check integral constraint
            integral_check = abs(np.sum(h_update * np.diff(pos_update)) - 1.0)
            return np.array([pos_increasing and integral_check < 1e-6])
        
        bounds = [(0, 2)] * (n_segments - 1) + [(0, 1)] * n_segments
        constraints = {'type': 'eq', 'fun': constraint}
        
        try:
            result = minimize(objective, opt_vars, method='SLSQP', bounds=bounds, constraints=constraints, 
                            options={'maxiter': 1000, 'ftol': 1e-8})
            if result.success:
                final_pos = np.concatenate([[0], result.x[:n_segments-1], [2]])
                final_h = np.clip(result.x[n_segments-1:], 0, 1)
                final_h = final_h * (n_points / 2) / np.sum(final_h * np.diff(final_pos))
                return final_pos, final_h
        except Exception:
            pass
            
        return pos, h_seg

    # Perform optimization
    start_time = time.time()
    best_c5 = float('inf')
    best_pos = None
    best_h = None
    
    # Multi-start
    for i in range(3):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Initialize differently
        pos = np.linspace(0, 2, n_segments + 1)
        pos[1:-1] += np.random.normal(0, 0.1, n_segments - 1)
        pos = np.sort(np.clip(pos, 0, 2))
        
        h_seg = np.random.rand(n_segments)
        h_seg = h_seg * (n_points / 2) / np.sum(h_seg * np.diff(pos))
        
        for _ in range(10):  # Local refinement
            if time.time() - start_time > budget_s - 10:
                break
                
            pos_update = np.concatenate([[0], pos[1:-1], [2]])
            h_update = np.clip(h_seg, 0, 1)
            h_update = h_update * (n_points / 2) / np.sum(h_update * np.diff(pos_update))
            c5 = compute_c5_sparse(pos_update, h_update)
            
            if c5 < best_c5:
                best_c5 = c5
                best_pos = pos_update.copy()
                best_h = h_update.copy()
                
            # Perturb slightly
            pos[1:-1] += np.random.normal(0, 0.01, n_segments - 1)
            pos = np.sort(np.clip(pos, 0, 2))
            
            h_seg += np.random.normal(0, 0.01, n_segments)
            h_seg = np.clip(h_seg, 0, 1)
            
    # Final optimization using sparse representation
    if best_pos is not None and best_h is not None:
        pos, h_seg = optimize_sparse()
        pos_final = np.concatenate([[0], pos[1:-1], [2]])
        h_final = np.clip(h_seg, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final * np.diff(pos_final))
        
        # Compute best c5 found
        final_c5 = compute_c5_sparse(pos_final, h_final)
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_pos = pos_final.copy()
            best_h = h_final.copy()
        
        # Convert back to dense representation
        h_dense = sparse_to_dense(best_pos, best_h, n_points)
        return (h_dense.tolist(), best_c5, n_points)
    
    # Fallback to simple optimization if needed
    h_final = h.copy()
    h_dense = h_final
    correlation = np.correlate(h_dense, 1 - h_dense, mode='full')
    c5 = np.max(correlation) * dx
    
    return (h_dense.tolist(), c5, n_points)
