import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_segments = 8  # Fewer segments for sparsity
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Convert to sparse representation
        # We'll create a piecewise constant approximation
        h_sparse = np.zeros(n_segments)
        pos_sparse = np.linspace(0, 2, n_segments + 1)
        
        # Distribute the integral equally in the sparse form
        for i in range(n_segments):
            segment_width = pos_sparse[i+1] - pos_sparse[i]
            # Simple approximation: uniform height
            h_sparse[i] = 1.0 / n_segments
            
        # Normalize to satisfy constraint
        total_integral = np.sum(h_sparse * (pos_sparse[1:] - pos_sparse[:-1]))
        h_sparse = h_sparse * (1.0 / total_integral)
    else:
        # Uniform initialization
        h_sparse = np.ones(n_segments) / n_segments
        pos_sparse = np.linspace(0, 2, n_segments + 1)
    
    # Combine into flattened variables
    vars_init = np.concatenate([pos_sparse[1:-1], h_sparse])
    
    # Constraints for optimization
    def constraint_integral(vars):
        pos = np.concatenate([[0], vars[:n_segments-1], [2]])
        h = vars[n_segments-1:]
        integral = np.sum(h * (pos[1:] - pos[:-1]))
        return integral - 1.0
    
    def constraint_bounds(vars):
        pos = np.concatenate([[0], vars[:n_segments-1], [2]])
        h = vars[n_segments-1:]
        # Check pos constraints
        pos_diff = np.diff(pos)
        pos_feasible = np.all(pos_diff > 0) and np.all(pos_diff < 2)
        
        # Check h constraints
        h_feasible = np.all(h >= 0) and np.all(h <= 1)
        
        return np.concatenate([np.array([pos_feasible and h_feasible]), pos_diff, 1-h])
    
    # Objective function
    def objective(vars):
        pos = np.concatenate([[0], vars[:n_segments-1], [2]])
        h = vars[n_segments-1:]
        
        # Compute overlaps for all possible lags
        overlap_values = []
        
        # Compute correlation over all possible lags
        # Discretize the lag space
        lags = np.arange(1, n_points)
        max_overlap = 0
        
        # Approximately compute overlaps by forming a piecewise constant approximation
        # We'll evaluate for a subset of lags for speed
        
        # Evaluate C5 by estimating it over a sampled set of lags
        for lag in lags[::max(1, len(lags)//20)]:  # Sample 20 lags max for efficiency
            lag_float = lag * dx
            
            # Compute overlap for this lag
            overlap = 0
            for i in range(n_segments):
                start1 = pos[i]
                end1 = pos[i+1]
                h1 = h[i]
                
                # Find intersecting segments
                for j in range(n_segments):
                    start2 = pos[j]
                    end2 = pos[j+1]
                    h2 = h[j]
                    
                    # Overlap region
                    overlap_start = max(start1, start2 - lag_float)
                    overlap_end = min(end1, end2 - lag_float)
                    
                    if overlap_start < overlap_end:
                        overlap_width = overlap_end - overlap_start
                        overlap += overlap_width * h1 * (1 - h2)
            
            overlap = overlap * dx
            max_overlap = max(max_overlap, overlap)
            
        return max_overlap
                
    # Create bounds
    bounds = []
    # Position bounds: 0 < pos[1] < ... < pos[n_segments-1] < 2
    for i in range(n_segments - 1):
        bounds.append((1e-6, 2 - 1e-6))
    
    # Height bounds: 0 <= h[i] <= 1
    for i in range(n_segments):
        bounds.append((1e-6, 1 - 1e-6))
    
    # Initialize
    best_vars = vars_init.copy()
    best_c5 = float('inf')
    current_vars = vars_init.copy()
    current_c5 = objective(current_vars)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_vars = current_vars.copy()
        
    # Optimization with multiple restarts
    n_restarts = 5
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Randomly perturb
        perturbed = current_vars.copy()
        for i in range(len(perturbed)):
            if np.random.random() < 0.3:
                delta = np.random.normal(0, 0.05)
                if i < n_segments - 1:  # Position
                    new_val = perturbed[i] + delta
                    # Ensure within bounds
                    if 1e-6 < new_val < 2 - 1e-6:
                        perturbed[i] = new_val
                else:  # Height
                    new_val = perturbed[i] + delta
                    # Ensure within bounds
                    if 1e-6 < new_val < 1 - 1e-6:
                        perturbed[i] = new_val
                        
        # Clip bounds
        for i in range(len(perturbed)):
            if i < n_segments - 1:  # Position
                perturbed[i] = np.clip(perturbed[i], 1e-6, 2 - 1e-6)
            else:  # Height
                perturbed[i] = np.clip(perturbed[i], 1e-6, 1 - 1e-6)
        current_vars = perturbed

        try:
            # Local optimization step
            result = minimize(objective, current_vars, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-5})
            if result.success:
                current_vars = result.x
                current_c5 = objective(current_vars)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_vars = current_vars.copy()
        except:
            pass
            
    # Final refinement using a subset of the function space
    pos_final = np.concatenate([[0], best_vars[:n_segments-1], [2]])
    h_final = best_vars[n_segments-1:]
    
    # Ensure integral constraint
    integral = np.sum(h_final * (pos_final[1:] - pos_final[:-1]))
    h_final = h_final * (1.0 / integral)
    
    # Convert final results into full n_points representation
    h_full = np.zeros(n_points)
    for i in range(n_segments):
        start_idx = int(pos_final[i] / dx)
        end_idx = int(pos_final[i+1] / dx)
        if end_idx > start_idx:
            h_full[start_idx:end_idx] = h_final[i]
            
    # Re-normalize
    sum_h = np.sum(h_full) * dx
    h_full = h_full * (n_points / 2) / sum_h
    
    # Final C5 evaluation
    correlation = np.correlate(h_full, 1 - h_full, mode='full')
    c5_final = np.max(correlation) * dx
    
    return (h_full.tolist(), c5_final, n_points)
