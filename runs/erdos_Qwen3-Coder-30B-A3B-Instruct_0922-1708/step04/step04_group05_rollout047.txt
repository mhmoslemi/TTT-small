import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    candidates = []
    num_candidates = 20
    for i in range(num_candidates):
        h = h_init.copy()
        # Add some initial noise
        noise = np.random.normal(0, 0.02, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h.copy())
    
    iteration = 0
    last_improvement = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        current_improvement = iteration - last_improvement
        if current_improvement > 100:
            # Occasionally add new candidates from variation
            for i in range(5):
                h = h_init.copy()
                noise = np.random.normal(0, 0.03, n_points)
                h += noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                candidates.append(h.copy())
        
        # Compute gradients for each candidate
        updated_candidates = []
        for h in candidates:
            # Compute current C5
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Gradient approximation by finite differences
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            
            # Update step
            lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            updated_candidates.append(h_new)
        
        candidates = updated_candidates
        
        # Add some randomness periodically
        if iteration % 10 == 0:
            # Randomly vary existing candidates
            for i in range(len(candidates)):
                if np.random.random() < 0.3:
                    h = candidates[i].copy()
                    noise = np.random.normal(0, 0.005, n_points)
                    h += noise
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    candidates[i] = h
        
        if best_c5 < 0.3807:
            break
    
    # Final test
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
