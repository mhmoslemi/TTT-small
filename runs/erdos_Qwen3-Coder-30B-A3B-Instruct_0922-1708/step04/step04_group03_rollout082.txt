import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation
    # Minimize sum of all overlaps (relaxed objective)
    h_relaxed = h.copy()
    
    # Using CVXPY for convex relaxation
    h_var = cp.Variable(n_points)
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # Build the total overlap expression for relaxation
    total_overlap = 0
    for k in range(n_points):
        for i in range(n_points):
            total_overlap += h_var[i] * (1 - h_var[(i + k) % n_points])
    
    prob = cp.Problem(cp.Minimize(total_overlap), constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_relaxed = h_var.value
            h_relaxed = np.clip(h_relaxed, 0, 1)
            h_relaxed = h_relaxed * (n_points / 2) / np.sum(h_relaxed)
    except:
        pass

    # Phase 2: Local refinement with projected gradient descent
    h = h_relaxed.copy()
    
    # Use a more sophisticated gradient step
    max_iter = 500
    lr = 0.01
    patience = 50
    no_improvement = 0
    
    for iter_count in range(max_iter):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate objective
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= patience:
            break
            
        # Gradient approximated via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update with momentum
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Decrease learning rate if needed
        if iter_count % 100 == 0 and iter_count > 0:
            lr *= 0.9
            
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
