import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use convex optimization approach with CVXPY
    # Define variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1,
        C >= 0
    ]
    
    # Add overlap constraints for all shifts
    for k in range(1, n_points):
        if k <= n_points - 1:
            # Positive shift: h[i] * (1 - h[i+k])
            overlap_pos = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
            constraints.append(overlap_pos <= C)
            
            # Negative shift: h[i+k] * (1 - h[i])
            overlap_neg = cp.sum(cp.multiply(h_var[k:], 1 - h_var[:n_points-k]))
            constraints.append(overlap_neg <= C)

    # Objective: minimize C
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with different solvers
    solvers_to_try = ['ECOS', 'SCS']
    solver_kwargs = {
        'ECOS': {'max_iters': 1000, ' feastol': 1e-6, 'reltol': 1e-6},
        'SCS': {'max_iters': 2000, 'eps': 1e-6}
    }
    
    for solver in solvers_to_try:
        try:
            start_solve = time.time()
            if solver in solver_kwargs:
                prob.solve(solver=solver, **solver_kwargs[solver])
            else:
                prob.solve(solver=solver)
            
            if prob.status in ['optimal', 'optimal_inaccurate']:
                c5_solution = C.value
                h_solution = h_var.value
                
                # Verify and refine
                if c5_solution < best_c5:
                    best_c5 = c5_solution
                    best_h = np.clip(h_solution, 0, 1)
                    # Rescale to match constraint
                    best_h = best_h * (n_points / 2) / np.sum(best_h)
            break
        except:
            # Try next solver if this one fails
            pass
        finally:
            if time.time() - start_time > budget_s - 1:
                break
    
    # Final refinement step with gradient-based method
    h = best_h.copy()
    for iteration in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute maximum overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(correlation) * dx
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - max_overlap) / eps
            
        # Update step with adaptive learning rate
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-4)
        h_new = h - lr * grad
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check convergence
        if np.linalg.norm(h_new - h) < 1e-8:
            break
        
        h = h_new
        max_overlap = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
    
    # Final evaluation
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (best_h.tolist(), float(final_c5), n_points)
