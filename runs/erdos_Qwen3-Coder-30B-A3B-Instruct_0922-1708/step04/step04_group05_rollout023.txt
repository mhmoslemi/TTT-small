import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure integral = 1
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate diverse initial candidates
    candidates = []
    candidates.append(h_init.copy())
    
    # Add some variants
    for i in range(19):
        h = h_init.copy()
        noise = np.random.normal(0, 0.02, n_points)
        h = np.clip(h + noise, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Gradient descent refinement
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # sample one candidate for refinement
        h = candidates[np.random.randint(len(candidates))].copy()
        
        # local optimization with gradient steps
        for step in range(30):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Compute gradient via finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus)
                c5_plus = compute_c5(h_plus)
                c5 = compute_c5(h)
                grad[i] = (c5_plus - c5) / eps
            
            # Update with gradient
            lr = 0.1 / (1 + np.linalg.norm(grad) * 1e-2)
            h_new = h - lr * grad
            h_new = project_to_feasible(h_new)
            h = h_new
            
        # Evaluate and possibly update best
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Add new candidates periodically
        if iter_count % 10 == 0:
            h_new = h.copy()
            noise = np.random.normal(0, 0.005, n_points)
            h_new = np.clip(h_new + noise, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            candidates.append(h_new)
            
        # Remove old candidates occasionally
        if len(candidates) > 20:
            candidates.pop(0)
            
        # Small random perturbation to escape local optima
        if iter_count % 50 == 0:
            for i in range(len(candidates)):
                h_pert = candidates[i].copy()
                noise = np.random.normal(0, 0.001, n_points)
                h_pert = np.clip(h_pert + noise, 0, 1)
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                candidates[i] = h_pert
    
    # Final evaluation of best candidate
    if best_h is not None:
        final_corr = np.correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(final_corr) * dx
        return (best_h.tolist(), final_c5, n_points)
    else:
        return (h_init.tolist(), compute_c5(h_init), n_points)
