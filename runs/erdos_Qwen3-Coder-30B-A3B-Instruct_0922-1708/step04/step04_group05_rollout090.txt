import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h = h_init.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient descent in frequency domain
    gamma = 0.001  # Initial learning rate
    max_iter = 1000
    early_stop_count = 0
    last_improvement = float('inf')
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute overlap using FFT
        h_padded = np.pad(h, (0, n_points - 1), 'constant')
        h_conj_padded = np.pad(1 - h, (0, n_points - 1), 'constant')
        overlap_fft = np.fft.fft(h_padded) * np.conj(np.fft.fft(h_conj_padded))
        overlap = np.fft.ifft(overlap_fft).real[:n_points]
        
        # Find maximum shift
        max_shift_idx = np.argmax(overlap)
        max_shift = max_shift_idx * dx - 1
        
        # Create frequency domain filter to reduce overlap at max_shift
        omega = 2 * np.pi * np.fft.fftfreq(n_points, dx)
        G = 1 - np.exp(-1j * omega * max_shift)
        
        # Apply the filter to the Fourier coefficients
        H = np.fft.fft(h)
        H_filtered = H * G / (np.abs(G) + 1e-10)  # Avoid division by zero
        
        # Inverse transform
        h_new = np.fft.ifft(H_filtered).real
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update with momentum
        h = (1 - gamma) * h + gamma * h_new
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute new C5
        h_padded = np.pad(h, (0, n_points - 1), 'constant')
        h_conj_padded = np.pad(1 - h, (0, n_points - 1), 'constant')
        overlap_fft = np.fft.fft(h_padded) * np.conj(np.fft.fft(h_conj_padded))
        overlap = np.fft.ifft(overlap_fft).real[:n_points]
        new_c5 = np.max(overlap) * dx
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = new_c5
            early_stop_count = 0
        else:
            early_stop_count += 1
            if early_stop_count > 100 and abs(last_improvement - new_c5) < 1e-5:
                break
                
        # Reduce learning rate periodically
        if iter_count % 100 == 0:
            gamma *= 0.9
            
    return (best_h.tolist(), best_c5, n_points)
