import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h using the parent construction or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    temperature = 1.0
    step_size = 0.1
    
    # Main optimization loop
    iteration = 0
    
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate candidate solution
        h_candidate = h.copy()
        
        # Select 10% of points to perturb
        indices_to_change = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        
        # Apply small perturbations
        for i in indices_to_change:
            # Perturb the value
            perturbation = np.random.uniform(-step_size, step_size)
            h_candidate[i] += perturbation
            
        # Clip values to [0, 1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Maintain integral constraint
        current_sum = np.sum(h_candidate)
        target_sum = n_points / 2
        
        if abs(current_sum - target_sum) > 1e-8:
            # Adjust other points to preserve sum
            adjustment = (target_sum - current_sum) / (n_points - len(indices_to_change))
            for i in range(n_points):
                if i not in indices_to_change:
                    h_candidate[i] += adjustment
        
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Normalize sum
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute candidate C5
        correlation_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(correlation_candidate) * dx
        
        # Accept or reject based on temperature
        delta = candidate_c5 - current_c5
        
        if delta < 0 or np.exp(-delta / temperature) > np.random.random():
            h = h_candidate
            current_c5 = candidate_c5
        
        # Cool down
        if iteration % 100 == 0:
            temperature *= 0.99
        
        # Occasionally reset temperature to escape local minima
        if iteration % 1000 == 0 and temperature < 0.01:
            temperature = 1.0
    
    # Final refinement around best solution
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_refine = best_h.copy()
        # Perturb a few points
        for _ in range(5):  # Perturb 5 points
            i = np.random.randint(0, n_points)
            h_refine[i] += np.random.uniform(-0.05, 0.05)
            
        # Ensure bounds and normalize
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
        
        correlation_refine = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5_refine = np.max(correlation_refine) * dx
        
        if c5_refine < best_c5:
            best_c5 = c5_refine
            best_h = h_refine
    
    return (best_h.tolist(), best_c5, n_points)
