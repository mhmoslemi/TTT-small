import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        particle = h + np.random.normal(0, 0.1, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(particle.copy())
        correlation = np.correlate(particle, 1 - particle, mode='full')
        pbest_vals.append(np.max(correlation) * dx)

    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]

    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1

        # Evaluate all particles
        for i in range(n_particles):
            particle = particles[i]
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_vals[i]:
                pbest[i] = particle.copy()
                pbest_vals[i] = c5
            
            if c5 < gbest_val:
                gbest = particle.copy()
                gbest_val = c5
                last_improvement = iter_count

            # Update velocity and position
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particle) + 
                           c2 * r2 * (gbest - particle))
            
            particles[i] = particle + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])

        # Early stopping
        if iter_count - last_improvement > 100:
            break

    # Store best solution from PSO
    if gbest_val < best_c5:
        best_h = gbest.copy()
        best_c5 = gbest_val

    # Try one more refinement with CVX
    try:
        # Define the optimization variables and constraints
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Objective: minimize the maximum overlap using CVX
        # We minimize the maximum of the correlations
        # This requires saddle point formulation or outer approximation
        # For now, try a more direct approach using a reformulation
        # In general, this is a non-convex optimization problem
        # But we can try a convex relaxations or use best PSO solution as starting point
        
        # Use the best PSO solution as starting point
        h_start = best_h.copy()
        
        # Bring back to form that can be used with scipy if needed for last refinement
        h_refined = h_start.copy()
        
        # Simple local search on top of the best solution
        def objective(h_vec):
            correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
            return -np.max(correlation) * dx  # negative because scipy minimizes

        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2

        cons = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Perform local optimization with scipy
        h_opt = minimize(objective, h_refined, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 1000, 'ftol': 1e-8})
        
        if h_opt.success:
            final_h = h_opt.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            correlation = np.correlate(final_h, 1 - final_h, mode='full')
            final_c5 = np.max(correlation) * dx
            
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass

    # Final check - return the best solution found
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
