import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation
    # Minimize sum of all overlaps instead of max
    # This gives us a good starting point
    h_relaxed = h.copy()
    h_prev = h.copy()
    
    # Use CVXPY for the convex relaxation
    h_var = cp.Variable(n_points)
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2]
    
    # For the convex optimization, we minimize the sum of all cross-correlation terms
    # which is an upper bound on the max overlap
    # This is a very simplified approximation
    
    # Let's just do a fast local search approach here
    
    # Phase 2: Local refinement using gradient descent
    max_no_improvement = 50
    no_improvement_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute correlation for current h
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Compute gradient with finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    if best_c5 == float('inf'):
        correlation = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(correlation) * dx
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
