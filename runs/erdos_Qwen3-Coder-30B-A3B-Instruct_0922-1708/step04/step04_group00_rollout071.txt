import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Use a hybrid approach: randomized optimization with local refinement
    n_iter = 1000
    no_improvement = 0
    max_no_improvement = 100
    
    for iteration in range(n_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        h_new = h + np.random.normal(0, 0.02, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5 = np.max(correlation) * dx
        
        # Accept if better
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > max_no_improvement:
                break
                
        h = h_new
    
    # Refine with L-BFGS
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        # Approximate gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - objective(h)) / eps
        return grad
    
    # Try multiple random starting points for L-BFGS
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        h_start = np.random.uniform(0.1, 0.9, n_points)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        # Reshape for scipy
        try:
            result = minimize(objective, h_start, method='L-BFGS-B', 
                             bounds=[(0, 1) for _ in range(n_points)],
                             jac=gradient, options={'maxiter': 100})
            if result.success:
                h_refined = result.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
                c5_refined = np.max(correlation) * dx
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined.copy()
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
