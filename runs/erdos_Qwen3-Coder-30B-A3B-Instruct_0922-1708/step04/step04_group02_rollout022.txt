import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize from parent if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to correct sum
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Current best
    best_h = h.copy()
    current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = current_c5
    
    # Simulated annealing parameters
    temperature = 1.0
    cooling_factor = 0.99
    iter_per_temp = 100
    
    start_time = time.time()
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb a small subset of points
        perturb_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        h_new = h.copy()
        
        # Apply perturbations
        for i in perturb_indices:
            # Add a small random change to h[i] in [0,1]
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
            
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain integral constraint
        current_sum = np.sum(h_new)
        if abs(current_sum - n_points / 2) > 1e-8:
            # Distribute the correction to other elements
            diff = (n_points / 2 - current_sum) / (n_points - len(perturb_indices))
            for i in range(n_points):
                if i not in perturb_indices:
                    h_new[i] += diff
            # Clip and renormalize
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            h = h_new
            current_c5 = new_c5
            
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Decrease temperature
        if (iteration + 1) % iter_per_temp == 0:
            temperature *= cooling_factor
    
    # Final refinement
    try:
        # Try one more gradient-like refinement
        h_refined = best_h.copy()
        lr = 0.001
        improve = True
        for _ in range(100):
            if not improve:
                break
            improve = False
            for i in range(n_points):
                # Sample a small change
                temp_h = h_refined.copy()
                delta = np.random.uniform(-0.01, 0.01)
                temp_h[i] += delta
                
                # Ensure boundaries
                temp_h = np.clip(temp_h, 0, 1)
                
                # Renormalize
                temp_h = temp_h * (n_points / 2) / np.sum(temp_h)
                
                new_c5 = np.max(np.correlate(temp_h, 1 - temp_h, mode='full')) * dx
                
                if new_c5 < current_c5:
                    h_refined = temp_h
                    current_c5 = new_c5
                    improve = True
    except:
        pass

    # Final recompute of C5 for best solution
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    return (best_h.tolist(), final_c5, n_points)
