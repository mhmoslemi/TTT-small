        h[:n_points//2] = 2.0

        h_coarse = np.random.rand(coarse_size)
        h_coarse = h_coarse / np.sum(h_coarse) * (coarse_size / 2)
        h_interp = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, coarse_size), h_coarse)
        candidates.append(h_interp)

        correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(correlation) * dx
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()

    max_refinement_steps = 500
    lr = 0.01
    refinement_steps = 0

            h_vars = cp.Variable(n_points)
            
            # Define constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]

                            best_c5 = c5_cvx

                        break
                except:
                    continue
            
            if not solved:

        except:

        # Gradient descent step

        c5_current = np.max(correlation) * dx

    final_c5 = np.max(correlation) * dx
