import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Iterative convex optimization approach
    max_iter = 100
    step_size = 0.5
    tol = 1e-5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Formulate convex approximation
        h_old = h.copy()
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = []
        constraints.append(cp.sum(h_var) == n_points / 2)
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Linearized shift constraints
        for k in range(1, n_points):
            # Compute shift for k
            shift = np.zeros(n_points)
            if k < n_points:
                shift[k:] = h_old[:-k]
            else:
                shift = np.zeros(n_points)
                
            # Linearize h[i] * h[i+k] as h_old[i+k] * h[i]
            # Constraint: sum(h[i] * (1 - h[i+k])) <= C
            # Linearization: sum(h_old[i+k] * h[i] + (1 - h_old[i+k]) * h[i] - h_old[i+k] * h[i]) <= C
            # Simplifies to: sum(h[i] - h_old[i+k] * h[i]) <= C
            # which is: sum(h[i] * (1 - h_old[i+k])) <= C
            # or equivalently: sum(h[i] * (1 - h_old[i+k]) - h_old[i+k] * h[i]) <= C
            # Which is: sum(h[i] - h_old[i+k] * h[i]) <= C
            # Or: sum(h[i] * (1 - h_old[i+k])) <= C
            
            # In our case the actual correlation is sum(h[i] * (1 - h[i+k]))
            # local_sum = sum(h[i] * (1 - h_old[i+k]) for i in range(n_points-k))
            # But to properly apply MA truncation, we want to "linearize"
            # LocalSum = sum(h_old[i+k] * h[i]) => same as sum(h[i] * h_old[i+k])
            # So we use:
            # Constraint: sum(h[i] * (1 - h_old[i+k])) <= C
            # This becomes: sum(h[i]) - sum(h_old[i+k] * h[i]) <= C
            # That's equivalent to sum(h[i]) - inner_product(h_old[i+k], h[i]) <= C
            
            # But this mixed product is not convex; instead, we will bound it:
            # Let's compute: sum_{i=0}^{n-k-1} h[i] * (1 - h_old[i+k])
            
            # So we do constraint: sum(h[i] - h_old[i+k] * h[i]) <= C for each k
            # That is sum(h[i]) - sum(h_old[i+k] * h[i]) <= C
            # Summarizing a feature at each k: sum(h_old[i+k] * h[i]) <= sum(h[i]) - C
            # The actual contribution is sum(h[i] * (1 - h_old[i+k]))
            # = sum(h[i]) - sum(h_old[i+k] * h[i])
            # Maximize the objective per definition, i.e overall we minimize C such that:
            # The maximum sum over k of sum(h[i] * (1 - h_old[i+k])) <= C
            
            # For each shift k, we linearize the term that would be going into the max:
            # the full sum over i of h[i] * (1 - h_old[i+k]) = sum(h[i]) - sum(h_old[i+k] * h[i])
            
            # We build this linear constraint with the current fixed h_old values:
            left_side = cp.sum(h_var) - cp.sum(cp.multiply(h_old[k:], h_var[:-k]))
            constraints.append(left_side <= C)
            
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-4, reltol=1e-4)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                # Line search and step size adjustment 
                step = step_size
                h_candidate = h_new.copy()
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                
                # Approximate difference
                diff = np.linalg.norm(h_candidate - h_old)
                if diff > 1e-8:  
                    pert = h_candidate - h_old
                    # Backtracking line search
                    while step > 1e-7:
                        h_test = h_old + step * pert
                        h_test = np.clip(h_test, 0, 1)
                        h_test = h_test * (n_points / 2) / np.sum(h_test)
                        corr = np.correlate(h_test, 1 - h_test, mode='full')
                        test_c5 = np.max(corr) * dx
                        if test_c5 < c5:
                            h = h_test.copy()
                            break
                        step *= 0.5
                    else:
                        h = h_candidate
                else:
                    h = h_candidate
            else:
                # Fallback to gradient descent step
                h = h_old + 0.001 * (np.ones(n_points) - 2 * h_old)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            # Fallback on simple update with clipping
            h = h_old + 0.001 * (np.ones(n_points) - 2 * h_old)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Reset step size
        if step_size < 1:
            step_size *= 1.2
        if step_size > 1:
            step_size = 1
    
    # Final evaluation of the best found h
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    c5_final = np.max(correlation) * dx
    
    # Final checks
    if not np.isclose(np.sum(best_h), n_points / 2, atol=1e-5):
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    
    return (best_h.tolist(), c5_final, n_points)
