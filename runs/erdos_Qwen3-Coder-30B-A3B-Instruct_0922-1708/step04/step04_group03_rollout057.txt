import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation using CVXPY to get good starting point
    # Minimize sum of all correlations to relax the max constraint
    h_var = cp.Variable(n_points)
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # Create the correlation sum using matrix operations
    # We'll compute only the correlation terms that contribute to the sum
    # This is a heuristic to get a good starting point
    total_corr = 0
    for k in range(n_points):
        # We only consider a subset of the correlations for speed
        # The key is to avoid full correlation computation for every k
        corr_term = 0
        # Compute term h[i] * (1 - h[(i+k) % n_points])
        indices = np.arange(n_points)
        shifted_indices = (indices + k) % n_points
        corr_term = cp.sum(cp.multiply(h_var, 1 - h_var[shifted_indices]))
        total_corr += corr_term
    
    prob = cp.Minimize(total_corr)
    prob = cp.Problem(prob, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = np.array(h_var.value).flatten()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    # Phase 2: Fine-tuning with projected gradient descent
    max_no_improvement = 50
    no_improvement_count = 0
    iteration = 0
    eps = 1e-5
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        # Compute current value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            improved = True
            
        # Gradient computation with finite differences
        grad = np.zeros_like(h)
        h_plus = h.copy()
        for i in range(n_points):
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
            h_plus[i] = h[i]
            
        # Update using projected gradient descent
        lr = 0.01
        h_new = h - lr * grad
        
        # Apply constraints after update
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Final evaluation of new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx

        # Adaptive learning rate
        if c5_new < c5:
            h = h_new
            improved = True
        else:
            # Try with smaller learning rate
            lr = lr * 0.5
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < c5:
                h = h_new
                improved = True
                
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        if no_improvement_count >= max_no_improvement:
            break

    # Final validation and return
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (best_h.tolist(), best_c5, n_points)
