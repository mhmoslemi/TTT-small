import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Grid search with coarse resolution
    coarse_points = 20
    coarse_dx = 2.0 / coarse_points
    best_coarse = None
    best_c5_coarse = float('inf')
    
    # Generate grid points
    grid_size = 100  # size of grid search
    grid_points = []
    num_grid = 10  # Number of top points to keep
    
    if grid_size <= 1000:  # avoid huge computation
        # Sample grid points
        for i in range(min(grid_size, 1000)):
            h_grid = np.random.rand(coarse_points)
            h_grid = h_grid / np.sum(h_grid) * (coarse_points / 2)
            grid_points.append(h_grid)
    
    # Evaluate grid points
    for h_grid in grid_points:
        corr = np.correlate(h_grid, 1 - h_grid, mode='full')
        c5 = np.max(corr) * coarse_dx
        if c5 < best_c5_coarse:
            best_c5_coarse = c5
            best_coarse = h_grid.copy()
    
    # Phase 2: Refine best coarse solution
    if best_coarse is not None:
        # Upsample to fine resolution
        h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, coarse_points), 
                               best_coarse)
        h = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
    
    # Local optimization with projected gradient
    max_iter = 1000
    lr = 0.01
    
    # Compute gradient by finite differences
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient computation (finite difference)
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
        
        # Update using gradient
        h_new = h - lr * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new

    # Final refinement with CVXPY if possible
    try:
        # Define variables and constraints
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Objective: minimize maximum overlap
        # Evaluate_constraint = cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (1, n_points))))
        # Not directly supported in CVXPY, so work with approximations
        
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (1, n_points)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                corr = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(corr) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass

    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
