import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary
    h = (h_init >= 0.5).astype(int)
    
    # Ensure exactly n_points/2 ones
    current_count = np.sum(h)
    target_count = n_points // 2
    
    if current_count < target_count:
        # Set some zeros to ones
        zeros_indices = np.where(h == 0)[0]
        num_to_flip = target_count - current_count
        flip_indices = np.random.choice(zeros_indices, size=num_to_flip, replace=False)
        h[flip_indices] = 1
    elif current_count > target_count:
        # Set some ones to zeros
        ones_indices = np.where(h == 1)[0]
        num_to_flip = current_count - target_count
        flip_indices = np.random.choice(ones_indices, size=num_to_flip, replace=False)
        h[flip_indices] = 0
    
    # Compute initial C5
    correlation_full = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(correlation_full) * dx
    best_h = h.copy()
    
    # Simulated annealing parameters
    T = 1.0
    decay_rate = 0.999
    perturbation_freq = 1000
    perturbation_rate = 0.05
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    max_stagnation = 1000
    
    while time.time() - start_time < budget_s - 1 and stagnation_count < max_stagnation:
        iteration += 1
        
        # Make a random change
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]  # flip the bit
        
        # Project to ensure exactly target_count ones
        current_count = np.sum(h_new)
        if current_count > target_count:
            # Find ones to flip to zero
            ones_indices = np.where(h_new == 1)[0]
            num_to_flip = current_count - target_count
            flip_indices = np.random.choice(ones_indices, size=num_to_flip, replace=False)
            h_new[flip_indices] = 0
        elif current_count < target_count:
            # Find zeros to flip to one
            zeros_indices = np.where(h_new == 0)[0]
            num_to_flip = target_count - current_count
            flip_indices = np.random.choice(zeros_indices, size=num_to_flip, replace=False)
            h_new[flip_indices] = 1
        
        # Calculate new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject the change
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
        
        # Decrease temperature
        T *= decay_rate
        
        # Periodic random perturbation
        if iteration % perturbation_freq == 0:
            num_perturb = int(n_points * perturbation_rate)
            perturb_indices = np.random.choice(n_points, size=num_perturb, replace=False)
            for idx in perturb_indices:
                h[idx] = 1 - h[idx]
            # Project again
            current_count = np.sum(h)
            if current_count > target_count:
                ones_indices = np.where(h == 1)[0]
                num_to_flip = current_count - target_count
                flip_indices = np.random.choice(ones_indices, size=num_to_flip, replace=False)
                h[flip_indices] = 0
            elif current_count < target_count:
                zeros_indices = np.where(h == 0)[0]
                num_to_flip = target_count - current_count
                flip_indices = np.random.choice(zeros_indices, size=num_to_flip, replace=False)
                h[flip_indices] = 1
    
    return (best_h.tolist(), best_c5, n_points)
