            h = np.interp(x_new, x_orig, h)

    best_c5 = float('inf')

    population_size = 50

    CR = 0.9
    max_no_improvement = 100
    no_improvement_count = 0

    fitness = []

    if 'initial_h_values' in globals():
        population.append(h.copy())
        corr = np.correlate(h, 1 - h, mode='full')
        fitness.append(np.max(corr) * dx)
    else:
        population.append(h.copy())
        corr = np.correlate(h, 1 - h, mode='full')
        fitness.append(np.max(corr) * dx)
        
    # Add random individuals
    for i in range(population_size - 1):
        h_new = np.random.uniform(0, 1, n_points)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        population.append(h_new)
        corr = np.correlate(h_new, 1 - h_new, mode='full')
        fitness.append(np.max(corr) * dx)
    
    # Best individual tracking
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    if best_fitness < best_c5:
        best_c5 = best_fitness
        best_h = best_individual.copy()

    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:

        new_population = []
        new_fitness = []

            idxs = np.random.choice(population_size, 3, replace=False)

            mutant = x1 + F * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, mutant, population[i])
            
            # Rescale to satisfy constraint
            trial = trial * (n_points / 2) / np.sum(trial)
            
            # Evaluate fitness
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_fitness = np.max(corr) * dx
            
            new_population.append(trial)
            new_fitness.append(trial_fitness)

            if trial_fitness < best_fitness:

        population = new_population
        fitness = new_fitness

        if iteration % 20 == 0:
            h = best_individual.copy()
            for _ in range(20):
                corr = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                
                # Gradient approximation via finite differences
                eps = 1e-4
                grad = np.zeros_like(h)
                for j in range(n_points):
                    h_plus = h.copy()

                    h_plus = np.clip(h_plus, 0, 1)

                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5) / eps
                
                # Update with small step size
                lr = 0.001
                h_new = h - lr * grad
                
                # Enforce constraints
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                h = h_new
                
                corr = np.correlate(h, 1 - h, mode='full')
                c5_refine = np.max(corr) * dx
                
                if c5_refine < best_c5:
                    best_c5 = c5_refine
                    best_h = h.copy()
            
            best_individual = h.copy()
            best_fitness = best_c5
            
        no_improvement_count += 1
        
    return (best_h.tolist(), best_c5, n_points)
