import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Trust region optimization
    trust_radius = 0.1
    trust_radius_decay = 0.99
    max_iter = 500
    no_improvement = 0
    max_no_improvement = 100
    eps = 1e-5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > max_no_improvement:
                break
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Trust region step
        # Solve: min_g ||g - grad||^2 s.t. ||g|| <= trust_radius and 0 <= h + g <= 1
        # With projected gradient descent approach
        
        # Try a step
        h_new = h - 0.1 * grad  # initial step
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Also try the constrained step if it improves
        h_test = h.copy()
        for i in range(n_points):
            step = -0.1 * grad[i]
            h_test[i] += step
            h_test[i] = max(0, min(1, h_test[i]))
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Evaluate
        corr_test = np.correlate(h_test, 1 - h_test, mode='full')
        c5_test = np.max(corr_test) * dx
        
        if c5_test < c5:
            h = h_test.copy()
        else:
            h = h_new.copy()
            
        # Update trust radius
        trust_radius *= trust_radius_decay
    
    # Perform additional local refinement
    # Use scipy optimize with bounds 
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx  # negative because we minimize

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Get an initial feasible point
    h_feasible = np.clip(best_h, 0, 1)
    h_feasible = h_feasible * (n_points / 2) / np.sum(h_feasible)
    
    # Optimize with L-BFGS-B 
    try:
        result = minimize(objective, h_feasible, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if result.success:
            h_final = np.clip(result.x, 0, 1)  
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            
            corr_final = np.correlate(h_final, 1-h_final, mode='full')
            c5_final = np.max(corr_final) * dx
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
