import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2 and h in [0,1]
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex optimization approach
    # Variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1,
        C >= 0
    ]
    
    # Process all shifts from 1 to n_points-1 (since we are computing full correlation)
    shifts = range(1, n_points)
    for k in shifts:
        # Compute the overlap integral for shift k
        # This is a quadratic constraint: sum over i of h[i] * (1 - h[i+k]) <= C
        # But we need to be very careful about the indexing
        # For a shift k, we compare h[i] with h[i+k] for valid i
        expr = 0
        for i in range(n_points - k):
            expr += h_var[i] * (1 - h_var[i+k])
        constraints.append(expr <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Problem
    problem = cp.Problem(objective, constraints)
    
    # The solver should be quite efficient for this size
    max_iter = 500
    
    # Calculate rough number of iterations we can run
    # Estimate that one solve takes about 0.1-0.2 seconds
    remaining_budget = budget_s - 5  # leave some time for other computation
    
    iter_count = 0
    while time.time() - start_time < remaining_budget - 1:
        try:
            problem.solve(solver=cp.ECOS, max_iter=max_iter, verbose=False)
            
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INaccurate]:
                h_val = h_var.value
                if h_val is not None:
                    h_val = np.clip(h_val, 0, 1)
                    h_val = h_val * (n_points / 2) / np.sum(h_val)
                    # Compute the actual C5
                    correlation = np.correlate(h_val, 1 - h_val, mode='full')
                    c5 = np.max(correlation) * dx
                    
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_val.copy()
            
            # Perturb solution slightly for next iteration
            # Add small Gaussian noise to encourage exploration
            pert = np.random.normal(0, 0.005, n_points)
            h = np.clip(h_val + pert, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h_var.value = h
            
            iter_count += 1
        except Exception:
            break
    
    # Final computation to make sure we have the correct C5
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
