import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation with exactly n_points/2 ones
    h = (h_init >= 0.5).astype(int)
    count_ones = np.sum(h)
    target_ones = n_points // 2
    
    # Adjust to have exactly target_ones ones
    if count_ones > target_ones:
        zero_indices = np.where(h == 0)[0]
        to_flip = count_ones - target_ones
        flip_indices = np.random.choice(zero_indices, to_flip, replace=False)
        h[flip_indices] = 1
    elif count_ones < target_ones:
        one_indices = np.where(h == 1)[0]
        to_flip = target_ones - count_ones
        flip_indices = np.random.choice(one_indices, to_flip, replace=False)
        h[flip_indices] = 0
    
    # Ensure integral constraint is satisfied (h has exactly n_points/2 ones)
    assert np.sum(h) == target_ones
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Initial evaluation
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    
    start_time = time.time()
    T = 1.0  # Initial temperature
    cooling_rate = 0.999
    perturbation_interval = 1000
    stagnation_limit = 1000
    
    stagnant = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and stagnant < stagnation_limit:
        iteration += 1
        
        # Make a candidate move: flip a random bit
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]
        
        # Adjust to maintain exactly target_ones ones
        count_new = np.sum(h_new)
        if count_new > target_ones:
            # Find zeros to flip to ones
            zero_indices = np.where(h_new == 0)[0]
            flip_indices = np.random.choice(zero_indices, count_new - target_ones, replace=False)
            h_new[flip_indices] = 1
            one_indices = np.where(h_new == 1)[0]
            flip_indices = np.random.choice(one_indices, count_new - target_ones, replace=False)
            h_new[flip_indices] = 0
        elif count_new < target_ones:
            # Find ones to flip to zeros
            one_indices = np.where(h_new == 1)[0]
            flip_indices = np.random.choice(one_indices, target_ones - count_new, replace=False)
            h_new[flip_indices] = 0
            zero_indices = np.where(h_new == 0)[0]
            flip_indices = np.random.choice(zero_indices, target_ones - count_new, replace=False)
            h_new[flip_indices] = 1
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                stagnant = 0
            else:
                stagnant += 1
        else:
            # Probabilistically accept with simulated annealing
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
                current_c5 = new_c5
                stagnant = 0
            else:
                stagnant += 1
        
        # Temperature cooling
        T *= cooling_rate
        
        # Periodic randomization to escape local minima
        if iteration % perturbation_interval == 0:
            # Reset a random 5% of bits
            n_flip = max(1, n_points // 20)
            flip_indices = np.random.choice(n_points, n_flip, replace=False)
            h[flip_indices] = 1 - h[flip_indices]
            
            # Correct counts
            count_new = np.sum(h)
            if count_new > target_ones:
                zero_indices = np.where(h == 0)[0]
                to_flip = count_new - target_ones
                flip_indices = np.random.choice(zero_indices, to_flip, replace=False)
                h[flip_indices] = 1
            elif count_new < target_ones:
                one_indices = np.where(h == 1)[0]
                to_flip = target_ones - count_new
                flip_indices = np.random.choice(one_indices, to_flip, replace=False)
                h[flip_indices] = 0
    
    # Final check of best solution
    corr_best = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_best) * dx
    
    return (best_h.tolist(), final_c5, n_points)
