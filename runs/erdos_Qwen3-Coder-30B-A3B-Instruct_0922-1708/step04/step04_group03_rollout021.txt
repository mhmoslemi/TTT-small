import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Parameters
    learning_rate = 0.1
    decay_factor = 0.99
    max_no_improvement = 100
    no_improvement_count = 0
    epsilon = 1e-4
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    def compute_gradients(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        base_c5 = compute_c5(h_vec)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        return grad
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current objective and gradients
        current_c5 = compute_c5(h)
        grad = compute_gradients(h, epsilon)
        
        # Update h
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        new_c5 = compute_c5(h_new)
        
        # Accept improvement
        if new_c5 < current_c5:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate
        if iteration % 50 == 0:
            learning_rate *= decay_factor
            
        # Early stopping
        if no_improvement_count >= max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
