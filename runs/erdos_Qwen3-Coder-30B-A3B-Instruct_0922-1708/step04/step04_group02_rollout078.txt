import numpy as np
from scipy.optimize import minimize
import time
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize with parent or a simple construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Simple initial construction
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute indices for overlap computation
    lag_indices = np.arange(1, n_points)
    
    # Optimization step function
    def compute_c5(h_vec):
        # Compute convolution for all lags efficiently
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(correlation)
        return max_corr * dx
    
    # Tweak function to make small changes
    def perturb(h_vec, ratio=0.05):
        h_new = h_vec.copy()
        indices = np.random.choice(n_points, size=max(1, int(n_points * ratio)), replace=False)
        for i in indices:
            # Random change in [0.01, 0.05]
            delta = np.random.uniform(-0.05, 0.05)
            h_new[i] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Genetic inspired local search
    pop_size = 30
    population = [h.copy()]
    for _ in range(pop_size-1):
        h_new = perturb(h, 0.1)
        population.append(h_new)
    
    # Evolutionary internal loop
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Evaluate all in population
        fitness_scores = []
        for p in population:
            c5 = compute_c5(p)
            fitness_scores.append(c5)
        
        # Update best
        min_idx = np.argmin(fitness_scores)
        if fitness_scores[min_idx] < best_c5:
            best_c5 = fitness_scores[min_idx]
            best_h = population[min_idx].copy()
        
        # Selection
        sorted_indices = np.argsort(fitness_scores)
        selected = [population[i] for i in sorted_indices[:pop_size//2]]
        
        # Create offspring
        offspring = []
        while len(offspring) < pop_size:
            parent1 = selected[np.random.randint(len(selected))]
            parent2 = selected[np.random.randint(len(selected))]
            
            # Crossover: simple uniform
            child = np.where(np.random.rand(n_points) > 0.5, parent1, parent2)
            
            # Mutation
            child = perturb(child, 0.1)
            
            offspring.append(child)
        
        population = offspring
        
        # Occasionally refine best few with gradient descent
        if iter_count % 50 == 0 and len(selected) > 0:
            elite_h = selected[0]  # Best in current generation
            temp_c5 = compute_c5(elite_h)
            
            # Gradient descent on elite
            for _ in range(10):
                h_grad = elite_h.copy()
                eps = 1e-4
                grad = np.zeros(n_points)
                
                for i in range(n_points):
                    h_plus = h_grad.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c5_plus = compute_c5(h_plus)
                    grad[i] = (c5_plus - temp_c5) / eps
                
                # Step
                updated = h_grad - 0.005 * grad
                updated = np.clip(updated, 0, 1)
                updated = updated * (n_points / 2) / np.sum(updated)
                c5_new = compute_c5(updated)
                
                if c5_new < temp_c5:
                    elite_h = updated
                    temp_c5 = c5_new
                else:
                    break
            
            # Add to population if better
            if temp_c5 < compute_c5(population[0]):
                population[0] = elite_h
    
    # Final polish
    try:
        # Some gradient-based attempts to improve
        h_final = best_h.copy()
        c5_final = best_c5
        
        # Local refinement near best
        for _ in range(20):
            # Simple hill climbing on subset
            indices = np.random.choice(n_points, size=5, replace=False)
            h_test = h_final.copy()
            improved = False
            
            for i in indices:
                for delta in [-0.02, 0.02]:
                    h_probe = h_test.copy()
                    h_probe[i] += delta
                    if 0 <= h_probe[i] <= 1:
                        h_probe = h_probe * (n_points / 2) / np.sum(h_probe)
                        c5_probe = compute_c5(h_probe)
                        if c5_probe < c5_final:
                            h_test = h_probe
                            c5_final = c5_probe
                            improved = True
            
            if not improved:
                break
            
            h_final = h_test
        best_c5 = c5_final
        best_h = h_final
    except Exception:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
