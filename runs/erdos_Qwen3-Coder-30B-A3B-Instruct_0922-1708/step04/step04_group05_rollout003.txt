import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    max_iter = 200
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    n_particles = 20
    max_vel = 0.1
    inertia_w = 0.9
    c1 = 1.5
    c2 = 1.5
    omega_decay = 0.95
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        h = h_init + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        pbest.append(h.copy())
        pbest_vals.append(float('inf'))
    
    gbest_idx = 0
    gbest = particles[gbest_idx].copy()
    gbest_val = float('inf')
    
    start_time = time.time()
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Update inertia weight
        inertia_w *= omega_decay
        
        # Evaluate particles
        for i in range(n_particles):
            h = particles[i]
            
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Compute C5
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < pbest_vals[i]:
                pbest_vals[i] = c5
                pbest[i] = h.copy()
            
            # Update global best
            if c5 < gbest_val:
                gbest_val = c5
                gbest = h.copy()
                gbest_idx = i
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            vel = inertia_w * velocities[i] + \
                  c1 * r1 * (pbest[i] - particles[i]) + \
                  c2 * r2 * (gbest - particles[i])
            
            # Clamp velocity
            vel = np.clip(vel, -max_vel, max_vel)
            velocities[i] = vel
            
            # Update position
            new_pos = particles[i] + vel
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * (n_points / 2) / np.sum(new_pos)
            particles[i] = new_pos
        
        # Adaptive particle refresh if no improvement in last 50 iterations
        if iteration > 50:
            if gbest_val == float('inf') or np.allclose(gbest_val, float('inf'), rtol=1e-6):
                for i in range(min(5, n_particles)):
                    h = h_init + np.random.normal(0, 0.1, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    particles[i] = h.copy()
                    velocities[i] = np.random.uniform(-0.01, 0.01, n_points)
                    pbest[i] = h.copy()
                    pbest_vals[i] = float('inf')
    
    # Refinement with line search
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    refined_h = gbest.copy()
    best_c5 = gbest_val
    
    # Try refining top candidates
    top_candidates = []
    for i in range(min(5, len(pbest))):
        top_candidates.append(pbest[i])
    
    for candidate in top_candidates:
        # Make small perturbation
        perturb = np.random.normal(0, 0.01, n_points)
        h_trial = candidate + perturb
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        corr = np.correlate(h_trial, 1 - h_trial, mode='full')
        trial_c5 = np.max(corr) * dx
        
        if trial_c5 < best_c5:
            best_c5 = trial_c5
            refined_h = h_trial.copy()
    
    # Final validation
    corr_final = np.correlate(refined_h, 1 - refined_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (refined_h.tolist(), final_c5, n_points)
