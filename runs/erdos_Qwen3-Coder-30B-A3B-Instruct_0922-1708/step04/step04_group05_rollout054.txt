import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 150
    dx = 2.0 / n_points
    max_iter = 200
    n_particles = 20
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    max_no_improve = 50
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to satisfy the integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO initialization
    positions = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    # Add small noise to diversify initial particles
    for i in range(n_particles):
        h = h_init.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h = np.clip(h + noise, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        positions.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest_positions.append(h.copy())
        corr = np.correlate(h, 1 - h, mode='full')
        pbest_values.append(np.max(corr) * dx)
    
    gbest_value = min(pbest_values)
    gbest_position = pbest_positions[np.argmin(pbest_values)].copy()
    
    # PSO main loop
    start_time = time.time()
    no_improve_count = 0
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * iter / max_iter
        
        for i in range(n_particles):
            h = positions[i]
            vel = velocities[i]
            
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            cognitive = c1 * r1 * (pbest_positions[i] - h)
            social = c2 * r2 * (gbest_position - h)
            
            vel = w * vel + cognitive + social
            
            # Update position
            h_new = h + vel
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Store new position
            positions[i] = h_new
            
            # Evaluate new solution
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            new_c5 = np.max(corr) * dx
            
            # Update personal best
            if new_c5 < pbest_values[i]:
                pbest_positions[i] = h_new.copy()
                pbest_values[i] = new_c5
                
                # Update global best
                if new_c5 < gbest_value:
                    gbest_value = new_c5
                    gbest_position = h_new.copy()
                    no_improve_count = 0
            else:
                no_improve_count += 1
            
            # Periodically add noise to prevent getting stuck
            if iter % 10 == 0 and np.random.rand() < 0.3:
                noise = np.random.normal(0, 0.01, n_points)
                h_new = np.clip(h_new + noise, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                positions[i] = h_new
            
            # Adaptive particle count reduction
            if no_improve_count > max_no_improve:
                if n_particles > 10:
                    # Remove worst particle
                    worst_idx = np.argmax(pbest_values)
                    positions.pop(worst_idx)
                    velocities.pop(worst_idx)
                    pbest_positions.pop(worst_idx)
                    pbest_values.pop(worst_idx)
                    n_particles -= 1
                    no_improve_count = 0
                    
                    # Repopulate
                    for _ in range(2):
                        h = h_init.copy()
                        noise = np.random.normal(0, 0.1, n_points)
                        h = np.clip(h + noise, 0, 1)
                        h = h * (n_points / 2) / np.sum(h)
                        positions.append(h.copy())
                        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
                        pbest_positions.append(h.copy())
                        corr = np.correlate(h, 1 - h, mode='full')
                        pbest_values.append(np.max(corr) * dx)
                    n_particles += 2
    
    # Final refinement in the neighborhood of top solution
    h_refined = gbest_position.copy()
    best_h = h_refined.copy()
    best_c5 = gbest_value
    
    # Gradient-based line search refinement
    for refinement_iter in range(100):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient estimation
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5 = np.max(corr) * dx
                
            grad[i] = (c5_plus - c5) / eps
        
        # Update with negative gradient
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h_refined - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h_refined = h_new
        
    # Return final results
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
