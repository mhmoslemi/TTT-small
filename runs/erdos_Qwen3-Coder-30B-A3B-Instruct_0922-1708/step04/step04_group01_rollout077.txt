    velocities = []
    pbest = []
    pbest_values = []

        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)

            h_p = h_p / np.sum(h_p) * (n_points / 2)
        else:
            h_p = np.random.rand(n_points)
            h_p = h_p / np.sum(h_p) * (n_points / 2)
        particles.append(h_p.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
        pbest.append(h_p.copy())
        corr = np.correlate(h_p, 1 - h_p, mode='full')
        pbest_values.append(np.max(corr) * dx)
    
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]

    no_improvement_count = 0

    while time.time() - start_time < budget_s - 5 and iter_count < max_iter:

                            c2 * r2 * (gbest - particles[i]))

            particles[i] += velocities[i]

            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Compute fitness
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            fitness = np.max(corr) * dx

                pbest[i] = particles[i].copy()

                    no_improvement_count = 0

                    no_improvement_count += 1

            break

    try:

        h_cvx = gbest.copy()

            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(corr) * dx
            
        # Since smooth convex optimization too complex for this case
        # Use the best PSO solution as initialization for local search
        current_h = h_cvx.copy()
        current_c5 = obj_func(current_h)
        
        # Try to refine further using gradient descent on a smoothed version
        step_size = 0.01
        for _ in range(100):  # Few iterations to get closer to local minimum
            grad = np.zeros_like(current_h)

            for j in range(n_points):
                h_plus = current_h.copy()

                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - current_c5) / eps
                
            # Update
            current_h -= step_size * grad
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            current_c5 = obj_func(current_h)
            
            if current_c5 < gbest_value:
                gbest_value = current_c5
                gbest = current_h.copy()
                
    except Exception:
        pass
    
    # Final best
    return (gbest.tolist(), gbest_value, n_points)
