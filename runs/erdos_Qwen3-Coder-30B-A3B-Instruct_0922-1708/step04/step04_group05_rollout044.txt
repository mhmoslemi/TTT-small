import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Scale to meet integral constraint
    h = h_init * (n_points / 2) / np.sum(h_init)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Frequency domain based approach
    max_iter = 1000
    gamma = 0.001
    patience = 0
    min_improvement = 1e-5
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute overlap via convolution
        overlap = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 100 and gamma > 1e-6:
                gamma *= 0.9
                patience = 0
        
        # Gradient approximation in frequency domain
        # Compute FFTs
        H = np.fft.fft(h)
        G = np.ones(n_points, dtype=complex)
        
        # Find index of maximum overlap
        max_idx = np.argmax(overlap)
        k = max_idx - (n_points - 1)
        
        # Design frequency filter to reduce this overlap
        # G(ω) = 1 - exp(-jωk) 
        omega = 2 * np.pi * np.fft.fftfreq(n_points)
        G = 1 - np.exp(-1j * omega * k)
        G = G / (np.abs(G) + 1e-12)  # Normalize
        
        # Apply frequency filter
        H_filtered = H * G
        h_filtered = np.fft.ifft(H_filtered).real
        
        # Project to feasible set
        h_new = np.clip(h_filtered, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update with momentum
        h = (1 - gamma) * h + gamma * h_new
        
        # Ensure integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Small random perturbations to escape local minima
        if iter_count % 50 == 0:
            noise = np.random.normal(0, 0.0005, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_overlap = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_overlap) * dx
    
    return (best_h.tolist(), final_c5, n_points)
