import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_time = time.time() + budget_s - 1
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # CVXPY approach - optimize using convex optimization
    # Setting up the optimization problem
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2, C >= 0]
    
    # Overlaps for each shift
    for shift in range(1, n_points):
        if time.time() > max_time:
            break
        # Compute correlation for this shift
        # We want sum h[i] * (1 - h[i+shift]) for valid i
        if shift < n_points:
            # Only valid indices for this shift
            corr = cp.sum(cp.multiply(h_var[:n_points-shift], 1 - h_var[shift:]))
            constraints.append(corr <= C)
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers_to_try = ['SCS', 'ECOS']
    for solver in solvers_to_try:
        if time.time() > max_time:
            break
        try:
            prob.solve(solver=solver, max_iters=1000, verbose=False)
            if prob.status == cp.OPTIMAL:
                sol_h = h_var.value
                if sol_h is not None:
                    sol_h = np.clip(sol_h, 0, 1)
                    sol_h = sol_h * (n_points / 2) / np.sum(sol_h)
                    # Compute actual C5
                    correlation = np.correlate(sol_h, 1 - sol_h, mode='full')
                    c5 = np.max(correlation) * dx
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = sol_h.copy()
        except:
            continue
    
    # If no optimization worked, use the initialization
    if best_c5 == float('inf'):
        correlation = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(correlation) * dx
        best_h = h.copy()
    
    # Refinement: local gradient-descent with bounds
    def safe_projection(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    lr = 0.01
    for i in range(500):
        if time.time() > max_time:
            break
        h_current = best_h.copy()
        
        # Estimate gradients via finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for j in range(n_points):
            if time.time() > max_time:
                break
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = safe_projection(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = safe_projection(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update
        h_new = h_current - lr * grad
        h_new = safe_projection(h_new)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
