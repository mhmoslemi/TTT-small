import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    # PSO parameters
    n_particles = 30
    max_iter = 1000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        particle = h + np.random.normal(0, 0.1, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_c5.append(compute_c5(particle, dx))
    
    gbest = pbest[np.argmin(pbest_c5)].copy()
    gbest_c5 = min(pbest_c5)
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (gbest - particles[i]))
            
            # Clip velocities (example bounds)
            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Update personal best
            current_c5 = compute_c5(particles[i], dx)
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
    
    # Refine with CVX optimization
    try:
        # Define the optimization variables and constraints
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Reformulate the objective: approximate it via quadratic constraints (simplified)
        # This is a challenging problem to solve exactly, so use the PSO solution as a good initial guess
        initial_guess = gbest
        if cp.sum(initial_guess).value == n_points / 2:
            h_vars.value = initial_guess
        
        # Create matrix for convolution-like constraints if possible, otherwise, partial approach
        # For quadratic formulation of overlap, it's still though; approximate with direct evaluation
        # For this specific form, can only optimize locally in CVXPY after PSO
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1-h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_refined = h_vars.value
            if h_refined is not None:
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                
                c5_refined = compute_c5(h_refined, dx)
                
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    except:
        pass

    # Final check with PSO best if no better CVX result
    if compute_c5(gbest, dx) < best_c5:
        best_h = gbest.copy()
        best_c5 = compute_c5(gbest, dx)
        
    return (best_h.tolist(), best_c5, n_points)
