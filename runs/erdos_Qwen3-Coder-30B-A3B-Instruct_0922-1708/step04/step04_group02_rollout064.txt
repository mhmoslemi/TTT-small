import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 900
    swarm_size = 100
    inertia_w = 0.7
    phi_p = 2.0
    phi_g = 2.0
    local_search_freq = 50
    rejuvenate_freq = 200
    rejuvenate_rate = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        swarm.append(h_init.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        swarm.append(h_init.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Fill rest of swarm
    for i in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests and global best
    pbest = [swarm[0].copy()]
    pbest_fitness = [compute_c5(swarm[0], dx)]
    gbest_idx = 0
    gbest_fitness = pbest_fitness[0]
    
    start_time = time.time()
    
    # Main PSO Loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all particles
        for i in range(swarm_size):
            fitness = compute_c5(swarm[i], dx)
            if fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fitness
                if fitness < gbest_fitness:
                    gbest_idx = i
                    gbest_fitness = fitness
        
        # Update global best
        gbest = swarm[gbest_idx].copy()
        
        # Update velocities and positions
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia_w * velocities[i] +
                phi_p * r1 * (pbest[i] - swarm[i]) +
                phi_g * r2 * (gbest - swarm[i])
            )
            
            # Update position
            swarm[i] += velocities[i]
            
            # Repair constraints
            swarm[i] = np.clip(swarm[i], 0, 1)
            current_sum = np.sum(swarm[i])
            diff = current_sum - n_points / 2
            
            if abs(diff) > 1e-8:
                # Adjust to enforce sum = n_points / 2
                if diff > 0:
                    # Excess - reduce largest elements
                    indices = np.argsort(swarm[i])[::-1]
                    for idx in indices:
                        if diff <= 0:
                            break
                        reduction = min(diff, swarm[i][idx])
                        swarm[i][idx] -= reduction
                        diff -= reduction
                else:
                    # Deficit - increase smallest elements
                    indices = np.argsort(swarm[i])
                    for idx in indices:
                        if diff >= 0:
                            break
                        increase = min(-diff, 1 - swarm[i][idx])
                        swarm[i][idx] += increase
                        diff += increase
        
        # Local Search
        if iteration % local_search_freq == 0 and iteration > 0:
            top_k = min(10, swarm_size // 10)
            fitness_indices = np.argsort(pbest_fitness)[:top_k]
            
            for idx in fitness_indices:
                # Gradient descent around candidates
                h_trial = pbest[idx].copy()
                lr = 0.001
                for _ in range(10):
                    grad = compute_gradient(h_trial, dx)
                    h_trial = h_trial - lr * grad
                    h_trial = np.clip(h_trial, 0, 1)
                    h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                    new_fitness = compute_c5(h_trial, dx)
                    if new_fitness >= pbest_fitness[idx]:
                        break
                    pbest_fitness[idx] = new_fitness
                    pbest[idx] = h_trial.copy()
        
        # Rejuvenation
        if iteration % rejuvenate_freq == 0 and iteration > 0:
            num_rejuvenate = int(swarm_size * rejuvenate_rate)
            worst_indices = np.argsort(pbest_fitness)[-num_rejuvenate:]
            
            for i in worst_indices:
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
                swarm[i] = h.copy()
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                pbest[i] = h.copy()
                pbest_fitness[i] = compute_c5(h, dx)
    
    # Final refinement
    best_candidate = pbest[gbest_idx]
    best_c5 = gbest_fitness
    best_h = best_candidate.copy()
    
    # Local optimization around best solution
    h_best = best_h.copy()
    for _ in range(1000):
        grad = compute_gradient(h_best, dx)
        h_test = h_best - 0.0001 * grad
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        c5_test = compute_c5(h_test, dx)
        if c5_test < best_c5:
            h_best = h_test.copy()
            best_c5 = c5_test
        else:
            break
    
    return (h_best.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def compute_gradient(h, dx):
    eps = 1e-6
    grad = np.zeros_like(h)
    
    h_original = h.copy()
    c5_base = compute_c5(h, dx)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (len(h) / 2) / np.sum(h_plus)
        
        c5_plus = compute_c5(h_plus, dx)
        grad[i] = (c5_plus - c5_base) / eps
    
    return grad
