import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation using CVXPY
    # Minimize sum of all overlaps (not maximum)
    h_relaxed = cp.Variable(n_points)
    constraints = [0 <= h_relaxed, h_relaxed <= 1, cp.sum(h_relaxed) == n_points / 2]
    
    # Create the objective that is a convex relaxation of minimizing sum of overlaps
    # For i in [0, n_points-1], we want to compute sum over all k of h[i] * (1 - h[(i+k)%n_points])
    # sum of overlaps = sum_{i,k} h[i] * (1 - h[(i+k)%n_points]) 
    # = sum_i h[i] * (n_points - sum_k h[(i+k)%n_points]) 
    # = sum_i h[i] * (n_points - sum_i h[i]) = sum_i h[i]*n_points - sum_i h[i]^2
    
    objective = cp.Minimize(cp.sum(h_relaxed) * n_points - cp.sum(cp.power(h_relaxed, 2)))
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-6)
        h = h_relaxed.value
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    # Phase 2: Projected gradient descent
    max_iter = 1000
    lr = 0.01
    patience = 50
    
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        if no_improvement_count >= patience:
            break
            
        # Compute gradient using finite differences  
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps

        # Gradient step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new

        # Decrease learning rate
        if iteration % 100 == 0:
            lr *= 0.9
                
    return (best_h.tolist(), best_c5, n_points)
