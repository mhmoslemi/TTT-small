import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Coarse grid search
    coarse_size = 20
    coarse_candidates = []
    for _ in range(50):
        if time.time() - start_time > budget_s - 100:
            break
        # Generate a random candidate
        h_coarse = np.random.rand(coarse_size)
        h_coarse = h_coarse / np.sum(h_coarse) * (coarse_size / 2)
        # Interpolate to fine grid
        x_coarse = np.linspace(0, 2, coarse_size)
        x_fine = np.linspace(0, 2, n_points)
        h_fine = np.interp(x_fine, x_coarse, h_coarse)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        # Compute C5
        c5 = np.max(np.correlate(h_fine, 1 - h_fine, mode='full')) * dx
        coarse_candidates.append((h_fine.copy(), c5))
    
    # Sort top candidates
    coarse_candidates.sort(key=lambda x: x[1])
    top_candidates = coarse_candidates[:10]
    
    # Phase 2: Fine refinement of top candidates
    for h_start, _ in top_candidates:
        if time.time() - start_time > budget_s - 100:
            break
        h_refined = h_start.copy()
        
        # CVXPY optimization for refinement
        try:
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            # For minimization we'll need to take care of the max operation manually
            # We use a trick: minimize max_k int h(x)(1-h(x+k)) dx 
            # We compute all contribute to find max overlap.
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_cvx = h_vars.value
                if h_cvx is not None:
                    h_cvx = np.clip(h_cvx, 0, 1)
                    h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                    c5_cvx = np.max(np.correlate(h_cvx, 1 - h_cvx, mode='full')) * dx
                    if c5_cvx < best_c5:
                        best_c5 = c5_cvx
                        best_h = h_cvx.copy()
        except:
            pass
            
        # Gradient-based refinement
        for _ in range(500):
            if time.time() - start_time > budget_s - 100:
                break
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-4
            c5_current = np.max(np.correlate(h_refined, 1 - h_refined, mode='full')) * dx
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c5_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode='full')) * dx
                grad[i] = (c5_plus - c5_current) / eps
            
            # Update using gradient
            lr = 0.01
            h_new = h_refined - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_refined = h_new
            c5_refined = np.max(np.correlate(h_refined, 1 - h_refined, mode='full')) * dx
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
                
        # Final check
        c5_final = np.max(np.correlate(h_refined, 1 - h_refined, mode='full')) * dx
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_refined.copy()
            
    # Final optimized step
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        # Use finite difference to calculate max_v {int h(x) * (1 - h(x+v)) dx} over all v
        # but with max_k over all shifts -> we compute the full convolution
        # For now, perform optimized grid search
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_final = h_vars.value
            if h_final is not None:
                h_final = np.clip(h_final, 0, 1)
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                c5_final = np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
    except:
        pass

    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
