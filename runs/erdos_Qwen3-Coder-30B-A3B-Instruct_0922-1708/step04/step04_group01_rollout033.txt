import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Grid search with coarse resolution
    coarse_points = 20
    coarse_dx = 2.0 / coarse_points
    coarse_h = np.zeros(coarse_points)
    for i in range(coarse_points // 2):
        coarse_h[i] = 2.0
    
    # Get all combinations of choices (2 states per point)
    top_candidates = []
    for i in range(2**coarse_points):
        candidate = np.zeros(coarse_points)
        for j in range(coarse_points):
            if i & (1 << j):
                candidate[j] = 2.0
        # Normalize to sum = coarse_points / 2
        candidate = candidate * (coarse_points / 2) / np.sum(candidate)
        
        # Evaluate C5
        correlation = np.correlate(candidate, 1 - candidate, mode='full')
        c5 = np.max(correlation) * coarse_dx
        
        top_candidates.append((c5, candidate))
    
    # Sort top candidates
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:10]  # Take top 10
    
    # Phase 2: Refine candidates
    for _, coarse_candidate in top_candidates:
        # Upsample to fine resolution
        h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, coarse_points), 
                               coarse_candidate)
        h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
        
        # Refine using gradient descent
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros_like(h_upsampled)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_upsampled.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr_base = np.correlate(h_upsampled, 1 - h_upsampled, mode='full')
                c5_base = np.max(corr_base) * dx
                
                grad[i] = (c5_plus - c5_base) / eps
            
            # Update with gradient descent
            lr = 0.01
            h_new = h_upsampled - lr * grad
            
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_upsampled = h_new
            
            # Check current C5
            correlation = np.correlate(h_upsampled, 1 - h_upsampled, mode='full')
            c5_current = np.max(correlation) * dx
            
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_upsampled.copy()
                
    # Final optimization using CVXPY if time allows
    if time.time() - start_time < budget_s - 1:
        try:
            # Define variables and constraints
            h_vars = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Objective: minimize the maximum overlap
            # We approximate this by computing correlations
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            obj_value = cp.max(corr) * dx
            objective = cp.Minimize(obj_value)
            
            # Try to solve with a quadratic approximation
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_vars.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                correlation = np.correlate(h_new, 1 - h_new, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_new
                    best_c5 = c5_cvx
        except:
            pass

    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
