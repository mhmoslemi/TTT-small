import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Coarse grid search
    coarse_size = 20
    coarse_candidates = []
    
    # Generate diversified initial candidates
    for _ in range(100):
        h_c = np.random.rand(coarse_size)
        h_c = h_c * (coarse_size / 2) / np.sum(h_c)
        coarse_candidates.append(h_c)
    
    # Also add some structured candidates
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, coarse_size)
        h_c = np.interp(x_target, x_orig, h_orig)
        h_c = h_c * (coarse_size / 2) / np.sum(h_c)
        coarse_candidates.append(h_c)
    
    # Convert to fine resolution and evaluate
    for h_coarse in coarse_candidates:
        # Upsample to fine resolution
        h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, coarse_size), h_coarse)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        # Evaluate and store
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fine.copy()
    
    # Fine-tune the best solutions with constrained optimization
    def eval_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
        
    # Local refinement with scipy.optimize in a loop
    for i in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Take a random direction and perturb
        perturb = np.random.normal(0, 0.01, n_points)
        h_new = h + perturb
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = eval_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        elif c5_new < eval_c5(h):
            h = h_new.copy()
            
    # Run final rounds with CVXPY optimization if time allows
    try:
        if time.time() - start_time < budget_s - 1:
            # Define the optimization variables and constraints
            h_vars = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Objective is to minimize the maximum overlap
            # This is hard to represent directly in CVXPY, so we compute correlation
            # and solve using an iterative approach
            
            # Get current best h as starting point
            h_start = best_h.copy()
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
            
            # Use scipy's minimize with bounds for the challenge
            def objective(h_vec):
                correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
                return -np.max(correlation) * dx  # we minimize negative because it's max problem

            result = minimize(objective, h_start, method='L-BFGS-B', bounds=[(0, 1)] * n_points, 
                            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6})
            
            if result.success:
                h_opt = result.x.copy()
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_opt = eval_c5(h_opt)
                
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    
    except:
        pass

    # Final verification
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    # Ensure the final solution satisfies the demand
    if not (0 <= np.min(best_h) and np.max(best_h) <= 1 and np.abs(np.sum(best_h) - n_points/2) < 1e-8):
        # This should not happen due to the regulation in the code, but to make sure
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(final_correlation) * dx

    return (best_h.tolist(), final_c5, n_points)
