import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex optimization approach
    # Variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Objective
    objective = cp.Minimize(C)
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        0 <= h_var,
        h_var <= 1
    ]
    
    # Compute all possible correlations
    # For each shift k from -n_points+1 to n_points-1
    for k in range(-n_points+1, n_points):
        if k == 0:
            continue
        # Compute the correlation sum for shift k
        if k > 0:
            corr_sum = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
        else:
            corr_sum = cp.sum(cp.multiply(h_var[-k:], 1 - h_var[:-k]))
        constraints.append(corr_sum <= C)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Iterate with different tolerances and refinement
    max_iter = 1000
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-5, reltol=1e-5)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                
                # Compute actual C5
                correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                c5 = np.max(correlation) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_sol.copy()
            else:
                # Use a simple gradient descent fallback
                h = best_h.copy()
                lr = 0.01
                for _ in range(100):
                    if time.time() - start_time > budget_s - 1:
                        break
                    # Compute gradient numerically
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
                    for i in range(n_points):
                        h_plus = h.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        c5_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode='full')) * dx
                        grad[i] = (c5_plus - c5) / eps
                    
                    h_new = h - lr * grad
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    h = h_new
                    c5_new = np.max(np.correlate(h, 1 - h, mode='full')) * dx
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
        except Exception as e:
            # Fallback to simple optimization
            h = best_h.copy()
            lr = 0.001
            for _ in range(1000):
                if time.time() - start_time > budget_s - 1:
                    break
                # Compute gradient
                grad = np.zeros(n_points)
                eps = 1e-4
                c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c5_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode='full')) * dx
                    grad[i] = (c5_plus - c5) / eps
                
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                c5_new = np.max(np.correlate(h, 1 - h, mode='full')) * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
    
    # Use the best solution found so far
    return (best_h.tolist(), best_c5, n_points)
