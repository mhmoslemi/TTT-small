import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    max_iter = 10000
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate diverse initial candidates
    candidates = []
    for i in range(20):
        if i == 0:
            h = h_init.copy()
        else:
            # Add noise and ensure sum constraint
            noise = np.random.normal(0, 0.05, n_points)
            h = h_init + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        candidates.append(h.copy())
    
    def compute_c5(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx
    
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    # Optimization
    iter_count = 0
    stagnation_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Try multiple candidate solutions
        for candidate in candidates:
            if time.time() - start_time > budget_s - 2:
                break
                
            h = candidate.copy()
            
            # Gradient descent
            for _ in range(10):
                if time.time() - start_time > budget_s - 2:
                    break
                
                grad = gradient(h)
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-2)
                h_new = h - lr * grad
                h_new = project(h_new)
                
                # Check if it leads to improvement
                c5_new = compute_c5(h_new)
                c5_old = compute_c5(h)
                
                if c5_new < c5_old:
                    h = h_new
                    c5 = c5_new
                else:
                    break
            
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = iter_count
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Add new diversity periodically
        if iter_count % 50 == 0:
            for _ in range(3):
                if time.time() - start_time > budget_s - 2:
                    break
                h = h_init + np.random.normal(0, 0.03, n_points)
                h = project(h)
                candidates.append(h)
                if len(candidates) > 20:
                    candidates.pop(0)
                    
        if stagnation_count > 200:
            # Add diversity by creating new candidate
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = project(h)
            candidates.append(h)
            if len(candidates) > 20:
                candidates.pop(0)
            stagnation_count = 0
            
    if best_h is None:
        best_h = h_init.copy()
    
    return (best_h.tolist(), best_c5, n_points)
