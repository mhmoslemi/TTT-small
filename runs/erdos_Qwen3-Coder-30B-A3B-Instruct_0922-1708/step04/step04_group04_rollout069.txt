import numpy as np
from scipy.stats import latinhypercube
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # LHS Sampling approach
    lhs_samples = 50
    max_grad_steps = 500
    
    # Generate Latin hypercube samples
    lhs = latinhypercube.LatinHypercube(n_points, seed=seed)
    samples = lhs.random(n=lhs_samples)
    
    # Convert LHS samples to h values
    for i in range(lhs_samples):
        # Transform samples to [0, 1]
        candidate_h = samples[i]
        # Normalize to meet sum constraint
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Refine using gradient descent
        h_current = candidate_h.copy()
        for step in range(max_grad_steps):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute correlation for all shifts
            correlation = np.correlate(h_current, 1 - h_current, mode='full')
            c5 = np.max(correlation) * dx
            
            # Compute gradient via finite differences
            grad = np.zeros_like(h_current)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - c5) / eps
            
            # Update with adaptive learning rate
            lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
            if lr < 1e-4:
                lr = 1e-4
                
            h_new = h_current - lr * grad
            
            # Maintain constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_current = h_new
            
            # Update best if better
            correlation = np.correlate(h_current, 1 - h_current, mode='full')
            c5_current = np.max(correlation) * dx
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
                
            # Very simple convergence check
            if np.abs(c5 - c5_current) < 1e-8:
                break
    
    # Final refinement
    h_refine = best_h.copy()
    for step in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5 = np.max(correlation) * dx
        
        grad = np.zeros_like(h_refine)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h_refine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adaptive learning rate
        lr = 0.005 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h_refine - lr * grad
        
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_refine = h_new
        
        correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_refine.copy()
            
        if np.abs(c5 - c5_new) < 1e-10:
            break
    
    return (best_h.tolist(), best_c5, n_points)
