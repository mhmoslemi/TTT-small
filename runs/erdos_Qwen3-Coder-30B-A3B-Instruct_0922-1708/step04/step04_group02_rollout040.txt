import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_segments = 10
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize positions and heights
    def initialize_sparse():
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            # Find significant segments
            h_init = h_init / np.sum(h_init) * (n_points / 2)
            # Approximate with segments
            pos = [0.0]
            h_seg = []
            # Group significant regions
            start = 0
            while start < n_points:
                while start < n_points and h_init[start] < 1e-6:
                    start += 1
                if start >= n_points:
                    break
                end = start
                while end < n_points and h_init[end] > 1e-6:
                    end += 1
                pos.append(end * dx)
                h_seg.append(np.mean(h_init[start:end]))
                start = end
            # Make sure we have n_segments
            while len(pos) < n_segments + 1:
                pos.append(2.0)
                h_seg.append(0.0)
            pos = pos[:n_segments+1]
            h_seg = h_seg[:n_segments]
            h_seg = np.array(h_seg)
            # Normalize area to 1
            area = sum((pos[i+1] - pos[i]) * h_seg[i] for i in range(n_segments))
            h_seg = h_seg * (1.0 / area)
            return np.array(pos), h_seg
        else:
            # Default: uniform distribution
            pos = np.linspace(0, 2, n_segments + 1)
            h = np.ones(n_segments) / n_segments
            return pos, h
    
    def compute_overlap(pos, h, dx):
        # Create full h(x) discretized
        h_full = np.zeros(n_points)
        for i in range(n_segments):
            start_idx = int(pos[i] / dx)
            end_idx = int(pos[i+1] / dx)
            h_full[start_idx:end_idx] = h[i]
        
        # Compute overlaps using full correlation
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        # Max over all lags
        max_corr = np.max(corr)
        return max_corr * dx
    
    def compute_sparse_overlap(pos, h, dx):
        # Find all critical points
        critical_lags = set()
        # Compute over full domain using segment interaction
        max_val = 0
        for k in range(n_points):  # Lag values to test
            val = 0
            for i in range(n_segments):
                # Compute shift by k*dx
                for j in range(n_segments):
                    # Segment i--pos[i],pos[i+1] shifted by k*dx
                    pos_i_start = pos[i]
                    pos_i_end = pos[i+1]
                    pos_j_start = pos[j] - k*dx
                    pos_j_end = pos[j+1] - k*dx
                    # Intersection
                    if pos_i_end <= pos_j_start or pos_j_end <= pos_i_start:
                        continue
                    overlap_start = max(pos_i_start, pos_j_start)
                    overlap_end = min(pos_i_end, pos_j_end)
                    overlap_len = overlap_end - overlap_start
                    if overlap_len > 0:
                        val += h[i] * (1 - h[j]) * overlap_len
            max_val = max(max_val, val)
        return max_val
    
    def obj_func(x):
        # x is [pos[1], ..., pos[n_segments-1], h[0], ..., h[n_segments-1]]
        pos = np.concatenate([[0.0], x[:n_segments-1], [2.0]])
        h = x[n_segments-1:]
        h = np.clip(h, 0, 1)
        # Normalize area to 1
        area = sum((pos[i+1] - pos[i]) * h[i] for i in range(n_segments))
        h = h * (1.0 / area)
        return compute_sparse_overlap(pos, h, dx)
    
    def constraint_func(x):
        pos = np.concatenate([[0.0], x[:n_segments-1], [2.0]])
        h = x[n_segments-1:]
        return np.sum(h * (pos[1:] - pos[:-1])) - 1.0
    
    def optimize_sparse():
        pos, h = initialize_sparse()
        # Combine variables
        x0 = np.concatenate([pos[1:-1], h])
        bounds = [(0, 2) for _ in range(n_segments-1)] + [(0, 1) for _ in range(n_segments)]
        
        # Define constraints
        cons = {'type': 'eq', 'fun': constraint_func}
        
        start_time = time.time()
        best_val = float('inf')
        best_x = x0.copy()
        
        # Multi-start optimization
        for _ in range(3):
            if time.time() - start_time > budget_s - 10:
                break
            # Perturb initial
            x_init = x0.copy()
            x_init += np.random.normal(0, 0.01, len(x_init))  
            x_init = np.clip(x_init, 0, 2)
            try:
                res = minimize(obj_func, x_init, bounds=bounds, constraints=cons, 
                               method='SLSQP', options={'maxiter': 200, 'ftol': 1e-8,
                                                        'gtol': 1e-6, 'disp': False})
                if not res.failed and res.fun < best_val:
                    best_val = res.fun
                    best_x = res.x.copy()
            except:
                continue
                
            # Refine with local search
            for _ in range(100):
                if time.time() - start_time > budget_s - 5:
                    break
                try:
                    # Local refinement with fixed number of iterations
                    x_test = best_x + np.random.normal(0, 0.005, len(best_x))
                    x_test = np.clip(x_test, 0, 2)
                    new_val = obj_func(x_test)
                    if new_val < best_val:
                        best_val = new_val
                        best_x = x_test
                except:
                    continue
                        
        # Final adjust
        pos_final = np.concatenate([[0.0], best_x[:n_segments-1], [2.0]])
        h_final = best_x[n_segments-1:]
        h_final = np.clip(h_final, 0, 1)
        area_final = sum((pos_final[i+1] - pos_final[i]) * h_final[i] for i in range(n_segments))
        h_final = h_final * (1.0 / area_final)
        
        return pos_final, h_final, best_val
    
    # Main optimization process
    start_time = time.time()
    best_pos, best_h, best_c5 = optimize_sparse()
    
    # Convert back to discretized form
    h_discrete = np.zeros(n_points)
    for i in range(n_segments):
        start_idx = int(best_pos[i] / dx)
        end_idx = int(best_pos[i+1] / dx)
        h_discrete[start_idx:end_idx] = best_h[i]
    
    # Final recomputation to verify
    final_overlap = np.correlate(h_discrete, 1 - h_discrete, mode='full')
    final_c5 = np.max(final_overlap) * dx
    
    return (h_discrete.tolist(), final_c5, n_points)
