import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # Default initialization
        
    # Ensuring the sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Bit flipping optimization part
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx

    def enforce_sum_constraint(h_vec):
        current_sum = np.sum(h_vec)
        if abs(current_sum - n_points / 2) > 1e-6:
            diff = current_sum - n_points / 2
            if diff > 0:
                # Remove some mass
                indices = np.argsort(h_vec)[::-1][:int(diff)]
                h_vec[indices] = 0
            else:
                # Add mass
                indices = np.argsort(h_vec)[:int(-diff)]
                h_vec[indices] = 1
        return h_vec

    # Evolutionary approach with bit-valued h
    def evolve_population(population, fitness_scores):
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        top_candidates = population[sorted_indices[:len(population)//4]]
        
        # Generate new population
        new_population = []
        new_population.extend(top_candidates)
        
        # Crossover and Mutation
        for _ in range(len(population) - len(top_candidates)):
            if np.random.rand() < 0.5:
                # Crossover
                parent1, parent2 = np.random.choice(top_candidates, 2, replace=False)
                crossover_point = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                # Mutation
                parent = np.random.choice(top_candidates)
                child = parent.copy()
                # Flip bits
                mutations = np.random.choice(n_points, max(1, n_points // 20), replace=False)
                for i in mutations:
                    child[i] = 1 - child[i]
            
            # Ensure that sum is correct
            child = enforce_sum_constraint(child)
            
            new_population.append(child)
        
        return np.array(new_population[:len(population)])

    # Initialize population
    population_size = 100
    population = []
    for _ in range(population_size):
        h_trial = h.copy()
        # Add some randomness
        if np.random.rand() < 0.5:
            num_changes = np.random.randint(1, n_points // 10)
            indices = np.random.choice(n_points, num_changes, replace=False)
            for i in indices:
                h_trial[i] = 1 - h_trial[i]
        h_trial = enforce_sum_constraint(h_trial)
        population.append(h_trial)
    
    population = np.array(population)
    
    gen = 0
    max_generations = 500
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and gen < max_generations:
        # Evaluate fitness
        fitness_scores = []
        for individual in population:
            c5_val = compute_c5(individual)
            fitness_scores.append(c5_val)
        
        fitness_scores = np.array(fitness_scores)
        
        best_idx = np.argmin(fitness_scores)
        if fitness_scores[best_idx] < best_c5:
            best_c5 = fitness_scores[best_idx]
            best_h = population[best_idx].copy()
            last_improvement = gen
            
        # Check for stagnation
        if gen - last_improvement > 50:
            population = []
            for _ in range(population_size):
                h_trial = h.copy() + np.random.normal(0, 0.01, n_points)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = enforce_sum_constraint(h_trial)
                population.append(h_trial)
            population = np.array(population)
        
        # Evolve population
        population = evolve_population(population, fitness_scores)
        gen += 1
    
    # Post-process with CVXPY
    try:
        # Convert best_h to CVXPY variable
        h_vars = cp.Variable(n_points)
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (n_points, 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=100)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                c5_cvx = compute_c5(h_cvx)
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass
    
    # Final validation and return
    if not (np.all(best_h >= 0) and np.all(best_h <= 1)):
        raise ValueError("Final solution is infeasible")
    if abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = enforce_sum_constraint(best_h)
        
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
