import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Coarse grid search phase
    coarse_grid_size = 20
    coarse_candidates = []
    
    # Generate coarse candidates
    for _ in range(100):
        if time.time() - start_time > budget_s - 200:
            break
            
        h_coarse = np.random.rand(coarse_grid_size)
        h_coarse = h_coarse / np.sum(h_coarse) * (coarse_grid_size / 2)
        
        # Interpolate to fine grid
        x_coarse = np.linspace(0, 2, coarse_grid_size)
        x_fine = np.linspace(0, 2, n_points)
        h_fine = np.interp(x_fine, x_coarse, h_coarse)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full')
        c5 = np.max(correlation) * dx
        
        coarse_candidates.append((c5, h_fine.copy()))
    
    # Sort and keep top candidates
    coarse_candidates.sort(key=lambda x: x[0])
    top_candidates = coarse_candidates[:10] if len(coarse_candidates) >= 10 else coarse_candidates
    
    # Refinement phase
    for c5, h_init in top_candidates:
        if time.time() - start_time > budget_s - 200:
            break
            
        h_refined = h_init.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Try local optimization with gradient descent
        for it in range(500):
            if time.time() - start_time > budget_s - 50:
                break
                
            correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
            current_c5 = np.max(correlation) * dx
            
            # Compute approximate gradient
            grad = np.zeros_like(h_refined)
            eps = 1e-3
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - current_c5) / eps
            
            # Update with gradient descent
            lr = 0.02
            h_new = h_refined - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_refined = h_new
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_refined.copy()
    
    # Try convex optimization with CVXPY
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Define a robust way to compute the maximum overlap
        # Use a different approach: define the constraint properly
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass
    
    # Final check with the best found
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
