import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective_and_gradient(h_vec):
        # Reshape h_vec back to array
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Gradient estimation using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        return c5, grad
    
    # Trust region optimization with TRF method
    max_iter = 500
    current_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize using trust region method
        try:
            res = minimize(
                lambda x: objective_and_gradient(x)[0],
                current_h,
                method='TRF',
                bounds=[(0, 1)] * n_points,
                jac=lambda x: objective_and_gradient(x)[1],
                options={'maxiter': 10}
            )
            
            if res.success:
                current_h = res.x.copy()
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * (n_points / 2) / np.sum(current_h)
                
                # Evaluate objective
                corr = np.correlate(current_h, 1 - current_h, mode='full')
                current_c5 = np.max(corr) * dx
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            else:
                # Fallback to basic projection if optimization fails
                corr = np.correlate(current_h, 1 - current_h, mode='full')
                current_c5 = np.max(corr) * dx
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
                    
        except Exception as e:
            # If optimization fails, try basic projection
            corr = np.correlate(current_h, 1 - current_h, mode='full')
            current_c5 = np.max(corr) * dx
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
