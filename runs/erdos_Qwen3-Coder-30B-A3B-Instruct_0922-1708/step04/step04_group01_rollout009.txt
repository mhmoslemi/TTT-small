import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):

    if 'initial_h_values' in globals():

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a uniform distribution
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Phase 1: Quick exploration with low-resolution grid
    coarse_points = 20
    # Use a simple heuristic: perturb some parts of the function heavily
    # Try a few configurations starting from uniform
    candidates = []
    for _ in range(20):
        if time.time() - start_time > budget_s - 100:
            break
        h_cand = np.random.rand(coarse_points)
        h_cand = h_cand * (coarse_points / 2) / np.sum(h_cand)
        # Upsample to full resolution
        x_orig = np.linspace(0, 2, coarse_points)
        x_target = np.linspace(0, 2, n_points)
        h_upsampled = np.interp(x_target, x_orig, h_cand)
        # Clip and rescale
        h_upsampled = np.clip(h_upsampled, 0, 1)
        h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
        candidates.append(h_upsampled)
        
    # Evaluate all candidates from Phase 1
    for h_cand in candidates:
        c5 = compute_c5(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()
    
    # Phase 2: Refinement using gradient descent from top candidates
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        # Randomly pick from best candidates for next perturbation
        h_candidate = best_h.copy()
        # Add some randomness to prevent getting stuck
        noise = np.random.normal(0, 0.02, n_points)
        h_candidate += noise
        # Clip and rescale 
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Optimization step
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        # Project gradient update
        eps = 1e-4
        
        # Compute approximate gradient using finite differences in a few directions
        grad = np.zeros_like(h_candidate)
        for i in range(min(10, n_points)):  # Sample only a few points for efficiency
            h_plus = h_candidate.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = objective(h_plus)
            grad[i] = (c5_plus - objective(h_candidate)) / eps
        
        # Update with projected gradient descent
        lr = 0.01
        h_new = h_candidate - lr * grad
        
        # Re-project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_candidate = h_new
        
        # Check if this improves the solution
        c5 = compute_c5(h_candidate)
        if c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5
    
    # Final CVX refinement attempt if time allows
    if time.time() - start_time < budget_s - 10:
        try:
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Rearrange to convex optimization
            def loss(h_vec):
                corr = np.correlate(h_vec, 1 - h_vec, mode='full')
                return np.max(corr) * dx
            
            # Create convex approximation or tightened formulation if needed
            # For now, just use the best solution as starting point
            prob = cp.Problem(cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1))))), constraints)
            prob.solve(solver=cp.CBC, verbose=False, feastol=1e-6, abstol=1e-6)
            
            if prob.status == cp.OPTIMAL and h_vars.value is not None:
                h_cvx = np.clip(h_vars.value, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                c5_cvx = compute_c5(h_cvx)
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
        except:
            pass

    # Final assertion
    final_c5 = compute_c5(best_h)
    assert np.isclose(np.sum(best_h), n_points / 2, rtol=1e-3), "Sum constraint violated"
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Bounds violated"
    
    return (best_h.tolist(), final_c5, n_points)
