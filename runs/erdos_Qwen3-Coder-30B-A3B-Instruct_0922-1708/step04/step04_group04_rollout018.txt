import numpy as np
from scipy.stats import latinhypercube
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
        
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Generate Latin Hypercube Samples
    lhs_samples = 50
    lh = latinhypercube.LatinHypercube(n_points, seed=seed)
    lhs = lh.random(lhs_samples)
    
    # Convert LHS to feasible h vectors
    candidate_h_values = []
    for i in range(lhs_samples):
        # Transform [0,1] to [0,1] and normalize
        h_candidate = lhs[i]
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        candidate_h_values.append(h_candidate)
    
    # Add the original if exists
    if 'initial_h_values' in globals():
        candidate_h_values.append(h.copy())
        
    # Refine each candidate
    for i, h_candidate in enumerate(candidate_h_values):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_refined = h_candidate.copy()
        lr = 0.01
        patience = 0
        
        for iteration in range(500): # Refinement steps
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute C5
            correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5 = np.max(correlation) * dx
            
            # Gradient approximation
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h_refined.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
                
            # Update
            h_new = h_refined - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute new C5
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            # Check if improvement
            if c5_new < c5:
                h_refined = h_new
                patience = 0
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_refined.copy()
            else:
                patience += 1
                if patience > 10:
                    lr *= 0.5
                    patience = 0
                    
            if lr < 1e-4:
                break
                
    return (best_h.tolist(), best_c5, n_points)
