import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # CVXPY formulation
    # Variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = []
    constraints.append(cp.sum(h_var) == n_points / 2)
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # For each shift k, we have sum_{i} h[i] * (1 - h[i+k]) <= C
    # We need to loop over all valid shifts
    for k in range(1, n_points):
        # Compute correlation terms based on shift
        if k < n_points:
            # Left part of correlation
            left_sum = cp.sum(cp.multiply(h_var[:n_points-k], (1 - h_var[k:])))
            constraints.append(left_sum <= C)
        else:
            # Handle case when k >= n_points (just zero correlation)
            pass
    
    # Objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=100, eps=1e-5)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h_var.value
            h_sol = np.clip(h_sol, 0, 1)
            h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
            
            # Calculate exact C5
            correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_sol.copy()
                
    except:
        pass
        
    # Local refinement with gradient descent
    h = best_h.copy()
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Use finite differences for gradient estimation
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Update with adaptive learning rate
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-4)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update if better
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Simple convergence check
        if np.abs(c5 - c5_new) < 1e-10:
            break
    
    return (best_h.tolist(), best_c5, n_points)
