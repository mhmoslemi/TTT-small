import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    h = (h_init >= 0.5).astype(int)
    
    # Ensure exactly n_points/2 ones
    count_ones = np.sum(h)
    if count_ones > n_points // 2:
        zero_indices = np.where(h == 0)[0]
        for i in range(count_ones - n_points // 2):
            h[zero_indices[i]] = 1
    elif count_ones < n_points // 2:
        one_indices = np.where(h == 1)[0]
        for i in range(n_points // 2 - count_ones):
            h[one_indices[i]] = 0
            
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Initialize temperature
    T = 1.0
    decay_rate = 0.9995
    perturb_freq = 1000
    
    start_time = time.time()
    
    # Compute initial C5
    corr_full = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr_full) * dx
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    iter_count = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Flip a random bit
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]
        
        # Ensure we maintain exactly n_points//2 ones
        count_ones = np.sum(h_new)
        if count_ones > n_points // 2:
            # Find a 1 to flip to 0
            ones = np.where(h_new == 1)[0]
            flip_idx = np.random.choice(ones)
            h_new[flip_idx] = 0
        elif count_ones < n_points // 2:
            # Find a 0 to flip to 1
            zeros = np.where(h_new == 0)[0]
            flip_idx = np.random.choice(zeros)
            h_new[flip_idx] = 1
            
        # Compute new C5
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject the change
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
        # Apply perturbations
        if iteration % perturb_freq == 0 and iteration > 0:
            # Reset a random 5% of bits
            n_reset = n_points // 20
            reset_indices = np.random.choice(n_points, size=n_reset, replace=False)
            h[reset_indices] = np.random.randint(0, 2, size=n_reset)
            
            # Ensure exactly n_points//2 ones
            count_ones = np.sum(h)
            if count_ones > n_points // 2:
                ones = np.where(h == 1)[0]
                flip_indices = np.random.choice(ones, size=count_ones - n_points // 2, replace=False)
                h[flip_indices] = 0
            elif count_ones < n_points // 2:
                zeros = np.where(h == 0)[0]
                flip_indices = np.random.choice(zeros, size=n_points // 2 - count_ones, replace=False)
                h[flip_indices] = 1
        
        # Update temperature
        T = max(1e-10, T * decay_rate)
        
        iter_count += 1
        
    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
