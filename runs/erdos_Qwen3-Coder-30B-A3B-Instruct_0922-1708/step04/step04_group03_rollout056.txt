import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():

            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation
    # We'll approximate the problem by minimizing sum of overlaps
    # This helps initialize to a better region
    
    # Formulate convex relaxation using CVXPY
    h_var = cp.Variable(n_points)
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Minimize sum of all overlaps (approximation)
    # Using a simpler approach: just minimize the sum of the correlation
    # which we can compute as quadratic form
    # But since we can't easily compute sum of max in cvxpy, we use 
    # an approximation approach: minimize total overlap
    
    # Let us instead do a more practical convex heuristic:
    # Minimize sum of overlaps using eigenvalue decomposition for relaxation
    # This is approximation that provides an initial good point
    
    # This approach uses a direct convex relaxation with a more tractable norm
    # Define matrix to represent the correlation structure
    
    # For simplicity, let's start with a more direct approach:
    # Projected gradient descent with insights from the convex relaxation
    
    # Phase 1: Use convex relaxation (simpler)
    # Since we cannot solve the exact relaxation easily, just do a transformation:
    # Take a simple starting point from the convex relaxation idea
    
    # Phase 2: Refinement with projected gradient descent
    # Compute gradient analytically if possible or use finite differences
    
    max_no_improvement = 50
    no_improvement_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        h_old = h.copy()
        
        # Evaluate current solution
        corr = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(corr) * dx
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            improved = True

        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # Compute gradient via finite differences for each dimension
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5_current) / eps
        
        # Apply gradient descent with decay
        lr = 0.01 * 0.99 ** (iteration / 100)
        h_new = h - lr * grad
        
        # Project to [0,1] and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final evaluation
    corr_best = np.correlate(best_h, 1 - best_h, mode='full')
    c5_final = np.max(corr_best) * dx
    
    if c5_final < best_c5:
        best_c5 = c5_final
        
    return (best_h.tolist(), best_c5, n_points)
