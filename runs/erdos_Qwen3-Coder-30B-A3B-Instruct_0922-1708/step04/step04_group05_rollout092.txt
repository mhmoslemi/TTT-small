import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Scale to satisfy integral constraint
    h = h_init * (n_points / 2) / np.sum(h_init)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # FFT-based convolution function
    def compute_overlap(h):
        h_full = np.concatenate([h, np.zeros(n_points - 1)])
        one_minus_h = np.concatenate([1 - h, np.zeros(n_points - 1)])
        conv_result = np.correlate(h_full, one_minus_h, mode='full')
        return conv_result[:n_points]
    
    # Gradient descent in frequency domain
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Main optimization loop
    iter_count = 0
    last_improvement = 0
    no_improvement_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Evaluate current solution
        overlap = compute_overlap(h)
        current_c5 = np.max(overlap) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 100:
            break
            
        # Compute FFT of h
        H = np.fft.fft(h)
        
        # Identify maximum overlap
        max_idx = np.argmax(overlap)
        shift = max_idx - (n_points - 1)
        
        # Create the frequency domain filter to suppress this overlap
        freqs = np.fft.fftfreq(n_points, d=dx)
        G = 1 - np.exp(-2j * np.pi * freqs * shift)
        G = np.clip(G, 1e-10, 1e10)  # Avoid division by zero
        
        # Update frequency domain
        H_new = H * np.conj(G) / np.abs(G)
        
        # Inverse FFT to spatial domain
        h_new = np.fft.ifft(H_new).real
        
        # Project back to feasible region
        h_new = project_feasible(h_new)
        
        # Update with momentum
        momentum = 0.99
        h = momentum * h + (1 - momentum) * h_new
        
        # Re-project after momentum update
        h = project_feasible(h)
        
        # Regularize occasionally to prevent stagnation
        if iter_count % 20 == 0 and time.time() - start_time < budget_s - 2:
            h += np.random.normal(0, 0.001, n_points)
            h = project_feasible(h)
            
        # Check for early termination
        if time.time() - start_time > budget_s - 2:
            break
            
    # Final evaluation
    final_overlap = compute_overlap(best_h)
    final_c5 = np.max(final_overlap) * dx
    return (best_h.tolist(), final_c5, n_points)
