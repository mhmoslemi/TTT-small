import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_h(h):
    """Evaluate the C5 bound for a given h"""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def repair_h(h, n_points):
    """Repair h to satisfy constraints"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Adjust elements to match target sum
    diff = current_sum - target_sum
    if diff > 0:  # Too much, decrease largest elements
        sorted_indices = np.argsort(h)[::-1]
        for i in sorted_indices:
            if diff <= 0:
                break
            decrease = min(diff, h[i])
            h[i] -= decrease
            diff -= decrease
    else:  # Not enough, increase smallest elements
        sorted_indices = np.argsort(h)
        for i in sorted_indices:
            if diff >= 0:
                break
            increase = min(-diff, 1 - h[i])
            h[i] += increase
            diff += increase
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 900
    inertia_weight = 0.7
    phi_p = 2.0
    phi_g = 2.0
    local_search_freq = 50
    rejuvenation_rate = 0.1
    resolution_steps = [50, 100, 200]
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        swarm.append(h_init)
        velocities.append(np.zeros(n_points))
    else:
        h_init = np.ones(n_points) * 0.5
        swarm.append(h_init)
        velocities.append(np.zeros(n_points))
    
    # Fill up the swarm
    for i in range(swarm_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal and global best
    pbest = [swarm[i].copy() for i in range(swarm_size)]
    pbest_c5 = [evaluate_h(swarm[i]) for i in range(swarm_size)]
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    start_time = time.time()
    current_iter = 0
    
    # Track improvement for early exit
    last_improvement = 0
    best_c5 = gbest_c5
    
    while current_iter < max_iter and (time.time() - start_time) < budget_s - 10:
        # Main PSO loop
        for i in range(swarm_size):
            h = swarm[i]
            v = velocities[i]
            
            # Evaluate current
            c5 = evaluate_h(h)
            
            if c5 < pbest_c5[i]:
                pbest[i] = h.copy()
                pbest_c5[i] = c5
                
                if c5 < gbest_c5:
                    gbest = h.copy()
                    gbest_c5 = c5
                    if c5 < best_c5:
                        best_c5 = c5
                        last_improvement = current_iter
                        
        # Update velocities and positions
        for i in range(swarm_size):
            h = swarm[i]
            v = velocities[i]
            p = pbest[i]
            
            # Update velocity
            r_p = np.random.random(n_points)
            r_g = np.random.random(n_points)
            
            v = (inertia_weight * v 
                 + phi_p * r_p * (p - h) 
                 + phi_g * r_g * (gbest - h))
            
            # Update position
            h_new = h + v
            
            # Repair
            h_new = repair_h(h_new, n_points)
            
            # Update
            swarm[i] = h_new
            velocities[i] = v
            
        current_iter += 1
        
        # Local search
        if current_iter % local_search_freq == 0:
            # Sort by fitness
            sorted_indices = np.argsort(pbest_c5)
            top_indices = sorted_indices[:swarm_size//10]
            
            # Local gradient descent on top candidates
            for idx in top_indices:
                h = pbest[idx].copy()
                for _ in range(10):
                    # Gradient approximation
                    eps = 1e-4
                    grad = np.zeros(n_points)
                    base_c5 = evaluate_h(h)
                    
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = repair_h(h_plus, n_points)
                        
                        c5_plus = evaluate_h(h_plus)
                        grad[j] = (c5_plus - base_c5) / eps
                    
                    # Gradient step
                    h -= 0.001 * grad
                    h = repair_h(h, n_points)
                    
                # Update if better
                new_c5 = evaluate_h(h)
                if new_c5 < pbest_c5[idx]:
                    pbest[idx] = h
                    pbest_c5[idx] = new_c5
                    
                    if new_c5 < gbest_c5:
                        gbest = h.copy()
                        gbest_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            last_improvement = current_iter
        
        # Rejuvenation
        if current_iter % 100 == 0 and current_iter > 0:
            # Replace worst particles
            worst_indices = np.argsort(pbest_c5)[-int(swarm_size * rejuvenation_rate):]
            for idx in worst_indices:
                h_new = np.random.random(n_points)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                swarm[idx] = h_new
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
    
    # Final refinement
    h_final = gbest.copy()
    for i in range(1000):
        eps = 1e-4
        grad = np.zeros(n_points)
        base_c5 = evaluate_h(h_final)
        
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus)
            grad[j] = (c5_plus - base_c5) / eps
            
        h_final -= 0.001 * grad
        h_final = repair_h(h_final, n_points)
        
    final_c5 = evaluate_h(h_final)
    
    # Final validation
    h_final = repair_h(h_final, n_points)
    final_c5 = evaluate_h(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
