import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Frequency domain approach parameters
    gamma = 0.001
    max_iter = 1000
    patience = 100
    patience_count = 0
    prev_c5 = float('inf')
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute overlap using FFT for efficiency
        h_full = np.concatenate([h, np.zeros(n_points - 1)])
        h_conj = np.concatenate([h, np.zeros(n_points - 1)])
        conv_full = np.real(np.fft.ifft(np.fft.fft(h_full) * np.fft.fft(1 - h_conj)))
        overlap = conv_full[:n_points]
        
        # Find maximum overlap
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx]
        
        # Update best solution
        c5 = max_val * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_count = 0
        else:
            patience_count += 1
            
        # Early stopping if improvement is negligible
        if patience_count > patience:
            break
            
        # Gradient step in frequency domain
        # Get frequency domain representation
        H = np.fft.fft(h)
        freqs = np.fft.fftfreq(n_points, dx)
        
        # Design filter to suppress peak correlation
        # Create filter that reduces influence at the problematic shift
        filter_freq = np.ones_like(freqs)
        k = max_idx - n_points + 1  # Shift index (could be negative)
        
        # Apply filter
        # Filter to reduce components at frequency corresponding to shift k
        filter_freq = np.exp(-1j * 2 * np.pi * freqs * k / 2.0)
        filter_freq = np.abs(filter_freq)  # Magnitude-based filter
        
        # Apply filter to frequency domain
        H_filtered = H * filter_freq
        
        # Back to time domain
        h_new = np.real(np.fft.ifft(H_filtered))
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Momentum update
        h = (1 - gamma) * h + gamma * h_new
        
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation
    h_full = np.concatenate([best_h, np.zeros(n_points - 1)])
    h_conj = np.concatenate([best_h, np.zeros(n_points - 1)])
    conv_full = np.real(np.fft.ifft(np.fft.fft(h_full) * np.fft.fft(1 - h_conj)))
    overlap = conv_full[:n_points]
    final_c5 = np.max(overlap) * dx
    
    return (best_h.tolist(), final_c5, n_points)
