import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Grid refinement for optimization
    refinement_levels = 3
    for level in range(refinement_levels):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Start with current best solution
        h = best_h.copy()
        
        # Use a combination of gradient-based and evolutionary approaches
        # Local optimization with constraints
        def objective(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            return c5
            
        def constraint_func(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return np.sum(h) - n_points / 2
            
        # Use COBYLA method which handles constraints well
        cons = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Sample optimization points
        sample_size = min(50, n_points)
        idxs = np.random.choice(n_points, sample_size, replace=False)
        
        # Generate a small perturbed initial guess for local search
        h_perturbed = h.copy()
        for i in idxs:
            h_perturbed[i] = np.clip(h_perturbed[i] + np.random.normal(0, 0.02), 0, 1)
            
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            res = minimize(objective, h_perturbed, method='COBYLA', 
                          constraints=cons, bounds=bounds, 
                          options={'maxiter': 500, 'disp': False})
            if res.success:
                h_opt = res.x.copy()
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    
        except Exception:
            pass
            
        # Also try a few random local perturbations
        for _ in range(20):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_local = h.copy()
            idx = np.random.randint(0, n_points)
            h_local[idx] = np.clip(h_local[idx] + np.random.normal(0, 0.05), 0, 1)
            h_local = h_local * (n_points / 2) / np.sum(h_local)
            c5_local = objective(h_local)
            if c5_local < best_c5:
                best_c5 = c5_local
                best_h = h_local.copy()
                
    # Final deterministic refinement
    h = best_h.copy()
    step_size = 0.01
    min_step = 1e-6
    
    for iter_num in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        # Estimate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
            
        # Step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step_size = min(0.1, step_size * 1.02)
        else:
            step_size = max(min_step, step_size * 0.95)
            
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
