import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from math import ceil

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = float('inf')
    
    # First try LP relaxation approach
    try:
        # Define the variables
        h_var = cp.Variable(n_points)
        # The shift k ranges from 0 to n_points-1 (corresponds to shift of 0 to 2)
        # For the correlation computation, k from 0 to 2*dx*n_points from the full mode
        # Need to handle correlation correctly, so we use the full convolution
        # Let's define a smaller set of shifted copies, but compute correlation correctly
        # We'll compute the full cross-correlation spectrum
        # Since convolution is symmetric, we exploit this to simplify
        # For full mode correlation, there are 2*n - 1 terms, so we have correlation lengths
        # We'll model the part that we're really interested in - the maximum
        
        # We're essentially solving:
        # minimize max_k \int h(x)(1 - h(x+k)) dx
        
        # To use convex relaxation, we compute max of correlation
        # This growths in complexity, so let's use more direct method
        
        # Since we already have gradients from a good solution, we'll
        # also try to run the original method to see if it's not good enough
        # But because it's quite slow, let's add an LP pre-optimization
        # to find a good starting point for the gradient descent
        
        # Try a simple primary vector
        h0 = h.copy()
        
        # Adjusted approach:
        # Use cvxpy to get a good starting point with constraint relaxation
        solver_params = {'max_iters': 1000}
        # Setup the variables and constraints for LP relaxation approach
        h_cvx = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraint 1: h in [0,1]
        constraints = [h_cvx >= 0, h_cvx <= 1]
        
        # Constraint 2: sum(h) = n_points/2
        constraints.append(cp.sum(h_cvx) == n_points/2)
        
        # We will compute correlation series and then use a finite approximation
        # of all correlation coefficients
        # Maximize C5 such that for all k, \int h(x)(1-h(x+k))dx <= C5
        # Since h(x) is discrete with dx, we do this approximately
        
        # Using limitation: if current solution is not improving enough, 
        # reduce complexity and go with local descent only
        
        # For accenting this idea, we apply an approach that will make the model evolution
        # simpler: Lagrangian expectation gradient descent approach.
        
        # Using just the simpler direct optimization with gradient-based descent on the full cost
        # combined with the LP regularization in case it helps
          
        # Try CG solver with default gradient descriptor
        
        # Start with better point from LP solve
        # LP relaxation is not always appropriate - better to fallback to heuristic if fails
        # In case convex relaxation fails, just proceed with local search using
        # generic method - but now with component-wise layering
      
        # Gradient-based approach instead
        # Most robust fix
        
        def compute_c5_cost(h):
            # Compute full cross-correlation of h with (1-h)
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        def gradient_approx(h, eps=1e-5):
            grad = np.zeros_like(h)
            c5_base = compute_c5_cost(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                # Enforce constraints again
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_c5_cost(h_plus)
                grad[i] = (c5_plus - c5_base) / eps
            return grad
        
        # Main gradient descent loop
        h_best = h.copy()
        c5_best = compute_c5_cost(h_best)
        
        iterations = 0
        max_iter = 2000
        lr = 0.01
        patience = 0
        max_patience = 20

        while iterations < max_iter and time.time() - start_time < budget_s - 1:

            h_new = h_best - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = compute_c5_cost(h_new)
            
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
                patience = 0
                lr = max(1e-6, lr * 0.999)  # decay learning rate
            else:
                patience += 1
                lr = max(1e-6, lr * 0.995)  # slow decay
                if patience > max_patience:
                    break
            
            iterations += 1
            
        # Final check of result
        best_h = h_best
        best_c5 = c5_best
        
    except Exception as e:
        # fallback to simple method
        best_h = h.copy()
        best_c5 = compute_c5_cost(h.copy())

            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        h_current = h.copy()
        c5_current = compute_c5_cost(h_current)
        lr = 0.01
        for iter_ in range(2000):
            if time.time() - start_time > budget_s - 1:
                break
                
            grad = np.zeros_like(h_current)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_c5_cost(h_plus)
                grad[i] = (c5_plus - c5_current) / eps
            
            h_new = h_current - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = compute_c5_cost(h_new)
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                lr = max(1e-6, lr * 0.999)
            else:
                lr = max(1e-6, lr * 0.995)
                
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_current.copy()

    return (best_h.tolist(), best_c5, n_points)
