import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy integral constraint exactly
    h = h / np.sum(h) * (n_points / 2)
    
    # Compute initial C5
    correlation = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(correlation) * dx
    
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    temp = 1.0
    cooling_factor = 0.99
    temp_decay_interval = 100
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Generate candidate solution
        h_new = h.copy()
        
        # Select 10% random indices to modify
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        
        # Modify selected indices
        for i in indices:
            old_val = h_new[i]
            # Perturb within [0,1]
            perturbation = np.random.uniform(-0.1, 0.1)
            new_val = old_val + perturbation
            h_new[i] = np.clip(new_val, 0, 1)
        
        # Maintain integral constraint
        sum_diff = np.sum(h_new) - (n_points / 2)
        if abs(sum_diff) > 1e-8:
            # Redistribute to maintain sum
            redistribute_amount = sum_diff / (n_points - len(indices))
            for i in indices:
                h_new[i] -= redistribute_amount
            # Ensure within bounds
            h_new = np.clip(h_new, 0, 1)
            # Adjust remaining points
            remaining_indices = [i for i in range(n_points) if i not in indices]
            if len(remaining_indices) > 0:
                diff_per_point = sum_diff / len(remaining_indices)
                for i in remaining_indices:
                    h_new[i] -= diff_per_point
                # Re-clip to maintain [0,1]
                h_new = np.clip(h_new, 0, 1)
            
        # Recompute
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.exp(-delta / temp) > np.random.random():
            h = h_new
            current_c5 = new_c5
            
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Update temperature
        if iteration % temp_decay_interval == 0:
            temp *= cooling_factor
            
    # Postprocessing - local refinement around best solution
    h_refined = best_h.copy()
    for _ in range(1000):
        h_trial = h_refined.copy()
        idx = np.random.randint(n_points)
        h_trial[idx] += np.random.uniform(-0.02, 0.02)
        h_trial[idx] = np.clip(h_trial[idx], 0, 1)
        
        # Maintain integral constraint
        sum_diff = np.sum(h_trial) - (n_points / 2)
        if abs(sum_diff) > 1e-8:
            redistribute_amount = sum_diff / (n_points - 1)
            h_trial = np.clip(h_trial - redistribute_amount, 0, 1)
        
        correlation_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
        c5_trial = np.max(correlation_trial) * dx
        
        if c5_trial < best_c5:
            h_refined = h_trial
            best_c5 = c5_trial

    # Final verification
    final_correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
