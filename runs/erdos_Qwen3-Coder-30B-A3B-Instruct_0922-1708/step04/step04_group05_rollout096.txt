import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to satisfy integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate multiple initial candidates
    candidates = []
    for i in range(20):
        if i == 0:
            h = h_init.copy()
        else:
            # Perturb initial candidate
            noise = np.random.normal(0, 0.01, n_points)
            h = h_init + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        candidates.append(h.copy())
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project(h):
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def gradient(h):
        # Compute gradient of max correlation wrt h
        # This is a simplified version using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        base_c5 = compute_c5(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    def local_optimize(h):
        lr = 0.01
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            g = gradient(h)
            h = h - lr * g
            h = project(h)
        return h
    
    # Main optimization loop
    for iter_count in range(10000):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Randomly select a candidate
        i = np.random.randint(len(candidates))
        h = candidates[i].copy()
        
        # Local refinement
        h = local_optimize(h)
        
        # Evaluate
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Update the candidate
        candidates[i] = h.copy()
        
        # Occasionally add new diverse candidates
        if iter_count % 100 == 0:
            # Add noise to best_h
            noise = np.random.normal(0, 0.01, n_points)
            h_new = best_h + noise
            h_new = project(h_new)
            candidates.append(h_new)
            
        # Trim candidates to keep 20
        if len(candidates) > 20:
            costs = [compute_c5(c) for c in candidates]
            idx = np.argmin(costs)
            candidates = [candidates[idx]]
    
    # Final check
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
