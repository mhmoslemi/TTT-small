import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Multi-resolution GA with gradient refinement approach
    start_time = time.time()
    resolution_levels = [50, 100, 200]
    population_size = 100
    
    best_h = None
    best_c5 = float('inf')
    
    # Start with coarse resolution
    for n_points in resolution_levels:
        if time.time() - start_time > budget_s - 5:
            break
            
        dx = 2.0 / n_points
        
        # Initialize population
        population = []
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_target = np.linspace(0, 2, n_points)
                h_init = np.interp(x_target, x_orig, h_init)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            population.append(h_init)
        else:
            # Generate random feasible initial solutions
            for _ in range(population_size - 1):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                population.append(h)
        
        # Add a constant distribution as a seed
        h_const = np.ones(n_points) * 0.5
        h_const = h_const * (n_points / 2) / np.sum(h_const)
        population.append(h_const)
        
        # Genetic algorithm on this resolution level
        ga_start = time.time()
        while time.time() - ga_start < (budget_s - (time.time() - start_time)) / len(resolution_levels) and time.time() - start_time < budget_s - 5:
            # Evaluate fitness
            fitness = []
            for h in population:
                # Compute C5 as max of correlations
                correlation = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(correlation) * dx
                fitness.append(c5)
            
            # Tournament selection
            selected = []
            for _ in range(population_size):
                idx1, idx2 = np.random.choice(len(population), 2, replace=False)
                if fitness[idx1] < fitness[idx2]:
                    selected.append(population[idx1].copy())
                else:
                    selected.append(population[idx2].copy())
            
            # Crossover (Simple arithmetic crossover)
            offspring = []
            for i in range(0, population_size, 2):
                h1 = selected[i]
                h2 = selected[i+1] if i+1 < len(selected) else selected[0]
                alpha = np.random.rand()
                child1 = alpha * h1 + (1 - alpha) * h2
                child2 = (1 - alpha) * h1 + alpha * h2
                offspring.append(child1)
                offspring.append(child2)
                
            # Mutation
            for i in range(len(offspring)):
                if np.random.rand() < 0.2:  # 20% mutation rate
                    # Polynomial mutation
                    eta = 20
                    for j in range(n_points):
                        if np.random.rand() < 1 / n_points:
                            r = np.random.rand()
                            if r < 0.5:
                                delta = (2 * r) ** (1 / (eta + 1)) - 1
                            else:
                                delta = 1 - (2 * (1 - r)) ** (1 / (eta + 1))
                            offspring[i][j] += delta
                            
            # Repair and normalize
            for i in range(len(offspring)):
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] * (n_points / 2) / np.sum(offspring[i])
                
            population = offspring
            
            # Gradient refinement of best individuals
            if len(population) > 0:
                # Sort by fitness
                sorted_indices = np.argsort([np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in population])
                top_candidates = [population[i] for i in sorted_indices[:min(10, len(population))]]
                
                # Gradient descent on these
                for h in top_candidates:
                    # Simple gradient descent step
                    lr = 0.01
                    iterations = 30
                    
                    for _ in range(iterations):
                        if time.time() - start_time > budget_s - 1:
                            break
                        # Gradient approximation using finite differences
                        grad = np.zeros_like(h)
                        eps = 1e-4
                        
                        for j in range(n_points):
                            h_plus = h.copy()
                            h_plus[j] += eps
                            h_plus = np.clip(h_plus, 0, 1)
                            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                            c5_plus = np.max(corr_plus) * dx
                            
                            grad[j] = (c5_plus - np.max(np.correlate(h, 1 - h, mode='full')) * dx) / eps
                        
                        h_new = h - lr * grad
                        h_new = np.clip(h_new, 0, 1)
                        h_new = h_new * (n_points / 2) / np.sum(h_new)
                        h = h_new
                    
                    # Check if improved
                    corr = np.correlate(h, 1 - h, mode='full')
                    c5 = np.max(corr) * dx
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        
            # Early termination if good enough
            if best_c5 <= 0.3807:
                break
                
        # Update best solution
        if best_h is not None:
            corr_final = np.correlate(best_h, 1 - best_h, mode='full')
            final_c5 = np.max(corr_final) * (2.0 / len(best_h))
            if final_c5 < best_c5:
                best_c5 = final_c5
            
    if best_h is None:
        # Fallback: just use the parent or uniform if available
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            if len(best_h) < 100:
                x_orig = np.linspace(0, 2, len(best_h))
                x_target = np.linspace(0, 2, 100)
                best_h = np.interp(x_target, x_orig, best_h)
            best_h = best_h * 50 / np.sum(best_h) 
        else:
            best_h = np.ones(100) * 0.5
            best_h = best_h * 50 / np.sum(best_h)
        corr_final = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr_final) * 0.02
        
    return (best_h.tolist(), best_c5, len(best_h))
