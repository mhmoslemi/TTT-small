import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use a different optimization strategy: direct CVXPY optimization
    # We'll solve a sequence of convex relaxations to find a better solution
    
    # Number of variables
    m = n_points
    
    # Setup CVXPY variables and constraints
    h_vars = cp.Variable(m)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == m / 2,
        C >= 0
    ]
    
    # Add overlap constraints for all shifts
    for k in range(1, m):
        # Compute the overlap for shift k
        # This is sum_{i=0}^{m-k-1} h[i] * (1 - h[i+k])
        if k <= m // 2:
            # For positive shift
            overlap = cp.sum(cp.multiply(h_vars[:m-k], 1 - h_vars[k:]))
            constraints.append(overlap <= C)
            
            # For negative shift - use same formula (symmetric)
            overlap_neg = cp.sum(cp.multiply(h_vars[k:], 1 - h_vars[:m-k]))
            constraints.append(overlap_neg <= C)
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Solve using different solvers
    solvers_to_try = ['ECOS', 'SCS']
    for solver in solvers_to_try:
        try:
            if time.time() - start_time > budget_s - 1:
                break
            prob.solve(solver=solver, verbose=False, max_iters=1000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = h_vars.value
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                
                # compute the actual c5
                correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                c5 = np.max(correlation) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_sol.copy()
                break  # Success
        except Exception:
            continue
    
    # If CVXPY did not work, fallback to a local search with corrections
    if best_c5 == float('inf'):
        # Local optimization using gradient descent from a better starting point
        h = best_h.copy()
        
        # Perform a few iterations of gradient descent with projection
        for iter in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute overlap (using full correlation)
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Gradient estimation using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
                
            # Update with gradient descent
            lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            
            # Projection back
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Update best
            correlation = np.correlate(h, 1 - h, mode='full')
            c5_new = np.max(correlation) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
    return (best_h.tolist(), best_c5, n_points)
