import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex optimization approach with iterative refinement
    max_iter = 100
    tol = 1e-5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute shifts for all k in [1, n_points-1]
        shifts = []
        for k in range(1, n_points):
            shift = np.zeros(n_points)
            shift[k:] = h[:-k]
            shifts.append(shift)
        
        # Linearization using previous solution
        linearized_shifts = []
        for k, shift in enumerate(shifts):
            # Simple linearization: h_i * h_{i+k} ~ h_old_{i+k} * h_i
            linearized_shift = h * shift  # This is h_{i+k} * h_i in context of MM
            linearized_shifts.append(linearized_shift)
            
        # Formulate and solve convex program
        h_var = cp.Variable(n_points)
        C_var = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
        ]
        
        # Add linearized overlap constraints
        for k, linear_shift in enumerate(linearized_shifts):
            # w_k = sum_{i=0}^{n-1} h_i (1 - h_{i+k}) ~ sum h_i - sum linear_shift 
            # which should be <= C
            constraints.append(cp.sum(h_var) - cp.sum(linear_shift) <= C_var)
            
        # Objective
        objective = cp.Minimize(C_var)
        
        # Solve with CVXPY
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-4, reltol=1e-4, verbose=False)
            h_new = h_var.value
            c5_new = C_var.value
        except:
            # Fallback to gradient ascent if optimization fails
            h_new = h.copy()
            c5_new = np.max(np.correlate(h, 1-h, mode='full')) * dx
            
        # Convert to feasible solution
        if h_new is not None:
            h_new = np.clip(h_new, 0, 1)
        else:
            h_new = h.copy()
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Line search for stability
        step_size = 0.5
        while True:
            h_test = h_new * step_size + h * (1 - step_size)
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            c5_test = np.max(np.correlate(h_test, 1-h_test, mode='full')) * dx
            
            if c5_test < c5_new:
                h_new = h_test
                c5_new = c5_test
                break
            step_size *= 0.5
            if step_size < 1e-6:
                break
                
        # Update best if improved
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Check for convergence
        if np.linalg.norm(h - h_new) < tol:
            break
            
        h = h_new
        
    # Final validation check
    actual_c5 = np.max(np.correlate(best_h, 1-best_h, mode='full')) * dx
    
    return (best_h.tolist(), actual_c5, n_points)
