import numpy as np
from scipy.stats import qmc
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
        
    # Rescale to meet constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Latin Hypercube Sampling
    sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs_samples = sampler.random(n=50)
    
    # Convert to [0,1] range
    lhs_samples = qmc.scale(lhs_samples, [0]*n_points, [1]*n_points)
    
    # Add the parent solution to the initial
    if 'initial_h_values' in globals():
        lhs_samples = np.vstack([h, lhs_samples])
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    def finite_diff_grad(h_vec, eps=1e-5):
        grad = np.zeros_like(h_vec)
        c5_base = compute_c5(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    # Work with LHS samples
    for i, lhs_h in enumerate(lhs_samples):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Start with the LHS point
        h = lhs_h.copy()
        h = project_to_feasible(h)
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Refinement with gradient descent
        lr = 0.01
        for step in range(500):
            if time.time() - start_time > budget_s - 5:
                break
                
            grad = finite_diff_grad(h)
            
            # Update
            h_new = h - lr * grad
            h_new = project_to_feasible(h_new)
            
            # Compute new c5
            c5_new = compute_c5(h_new)
            
            # Update if better
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                lr *= 0.999  # Decay learning rate
                if lr < 1e-4:
                    break
                    
    # Perform one final fine-tune
    lr = 0.001
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
                        
        grad = finite_diff_grad(best_h)
        
        h_new = best_h - lr * grad
        h_new = project_to_feasible(h_new)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            lr = 0.001  # reset learning rate on improvement
        else:
            lr *= 0.99
            if lr < 1e-5:
                break
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
