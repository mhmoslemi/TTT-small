import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    
    # Start with a reasonable initial solution
    n_points = 100
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Default construction
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    def evaluate(h_vec):
        # Convolution-based computation
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Gradient-based optimization with adaptive learning
    def local_optimize(h0, max_iter=50):
        h = h0.copy()
        c5_prev = evaluate(h)
        lr = 0.01
        
        for i in range(max_iter):
            # Simple gradient approximation using finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for k in range(len(h)):
                h_plus = h.copy()
                h_plus[k] += eps
                h_plus = project_to_feasible(h_plus)
                c5_plus = evaluate(h_plus)
                
                grad[k] = (c5_plus - c5_prev) / eps
            
            # Update
            h_new = h - lr * grad
            h_new = project_to_feasible(h_new)
            
            c5_new = evaluate(h_new)
            
            if c5_new < c5_prev:
                h = h_new
                c5_prev = c5_new
                lr *= 1.05  # Adaptive increase
            else:
                lr *= 0.8  # Adaptive decrease
            
            if abs(c5_prev - evaluate(h)) < 1e-8 or time.time() - start_time > budget_s - 5:
                break
                
        return h, c5_prev
    
    # Main loop
    it = 0
    while time.time() - start_time < budget_s - 5:
        it += 1
        
        # Evaluate current solution
        c5 = evaluate(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Improve with local optimization
        h_local, c5_local = local_optimize(h, max_iter=20)
        if c5_local < c5:
            h = h_local
            c5 = c5_local

        # Random perturbation with probability
        if np.random.random() < 0.3:
            h_perturbed = h.copy()
            # Modify few points
            n_modify = max(1, min(10, n_points // 10))
            indices = np.random.choice(n_points, n_modify, replace=False)
            for i in indices:
                perturb = np.random.normal(0, 0.05)
                h_perturbed[i] += perturb
            h_perturbed = project_to_feasible(h_perturbed)
            
            c5_pert = evaluate(h_perturbed)
            if c5_pert < c5:
                h = h_perturbed
                c5 = c5_pert
        
        # Periodically check for improvement
        if it % 10 == 0:
            h, c5 = local_optimize(h, max_iter=10)
    
    # Final local refinement
    best_h, best_c5 = local_optimize(best_h, max_iter=30)
    
    # Ensure the final solution is valid
    try:
        final_c5 = evaluate(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except:
        pass
        
    return (best_h.tolist(), best_c5, n_points)
