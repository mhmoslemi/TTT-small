import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Try different strategies to find a better solution
    
    # Strategy 1: Convex relaxation approach
    try:
        # Convex relaxation using CVXPY
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - h_var)))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            tmp_h = h_var.value
            tmp_h = np.clip(tmp_h, 0, 1)
            # rescale to sum to n_points/2
            tmp_h = tmp_h * (n_points / 2) / np.sum(tmp_h)
            # compute overlap
            corr = np.correlate(tmp_h, 1 - tmp_h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = tmp_h.copy()
    except:
        pass
    
    # Strategy 2: Gradient-based refinement starting from initial
    h_refined = h.copy()
    
    # Build correlation matrix efficiently
    def compute_overlap(h_in):
        corr = np.correlate(h_in, 1 - h_in, mode='full')
        return np.max(corr) * dx
    
    # Projected gradient descent
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec * (n_points / 2) / np.sum(h_vec)
    
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < 1000:
        iter_count += 1
        
        current_overlap = compute_overlap(h_refined)
        if current_overlap < best_c5:
            best_c5 = current_overlap
            best_h = h_refined.copy()
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h_refined)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            overlap_plus = compute_overlap(h_plus)
            grad[i] = (overlap_plus - current_overlap) / eps
        
        # Adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h_refined - lr * grad
        
        # Project: ensure constraints
        h_new = project_h(h_new)
        
        # Check for improvement
        if compute_overlap(h_new) < current_overlap:
            h_refined = h_new
            last_improvement = iter_count
        else:
            # Decrease learning rate if no improvement
            if iter_count - last_improvement > 10:
                lr *= 0.9
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
    # Final evaluation
    final_overlap = compute_overlap(best_h)
    if final_overlap < best_c5:
        best_c5 = final_overlap
    
    return (best_h.tolist(), best_c5, n_points)
