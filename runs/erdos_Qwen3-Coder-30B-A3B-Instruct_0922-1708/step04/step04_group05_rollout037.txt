import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    h = (h_init >= 0.5).astype(int)
    
    # Adjust to ensure exactly n_points/2 ones
    current_count = np.sum(h)
    target_count = n_points // 2
    
    if current_count > target_count:
        # Turn some 1s into 0s
        zeros_indices = np.where(h == 0)[0]
        ones_indices = np.where(h == 1)[0]
        if len(ones_indices) > target_count:
            # Select indices to flip
            flip_indices = np.random.choice(ones_indices, size=int(current_count - target_count), replace=False)
            h[flip_indices] = 0
    elif current_count < target_count:
        # Turn some 0s into 1s
        zeros_indices = np.where(h == 0)[0]
        ones_indices = np.where(h == 1)[0]
        if len(zeros_indices) > target_count - current_count:
            # Select indices to flip
            flip_indices = np.random.choice(zeros_indices, size=int(target_count - current_count), replace=False)
            h[flip_indices] = 1
    
    # Ensure the constraint of exactly 50 ones
    if np.sum(h) != target_count:
        # Make sure exactly half are 1s
        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        # Adjust if needed
        diff = int(target_count - np.sum(h))
        if diff > 0:
            # Need to add more 1s
            if len(zeros_indices) >= diff:
                to_flip = np.random.choice(zeros_indices, size=diff, replace=False)
                h[to_flip] = 1
        elif diff < 0:
            # Need to remove some 1s
            if len(ones_indices) >= -diff:
                to_flip = np.random.choice(ones_indices, size=-diff, replace=False)
                h[to_flip] = 0
                
    # Projection onto feasible set: ensure exactly target_count ones
    actual_count = np.sum(h)
    if actual_count != target_count:
        # Identify indices to modify
        idx_ones = np.where(h == 1)[0]
        idx_zeros = np.where(h == 0)[0]
        adjustments = target_count - actual_count
        if adjustments > 0:
            new_ones = np.random.choice(idx_zeros, size=adjustments, replace=False)
            h[new_ones] = 1
        else:
            new_zeros = np.random.choice(idx_ones, size=-adjustments, replace=False)
            h[new_zeros] = 0
            
    best_h = h.copy()
    best_c5 = np.inf
    
    def compute_c5(h_vec):
        # Compute full correlation
        corr_full = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value * dx
        return np.max(corr_full) * dx
    
    def project_feasible(h_vec):
        # Ensure exactly half of entries are 1
        if np.sum(h_vec) > target_count:
            # Get indices of 1s
            ones_indices = np.where(h_vec == 1)[0]
            # Select indices to flip back
            to_flip = np.random.choice(ones_indices, size=int(np.sum(h_vec) - target_count), replace=False)
            h_vec[to_flip] = 0
        elif np.sum(h_vec) < target_count:
            # Get indices of 0s
            zeros_indices = np.where(h_vec == 0)[0]
            # Select indices to flip
            to_flip = np.random.choice(zeros_indices, size=int(target_count - np.sum(h_vec)), replace=False)
            h_vec[to_flip] = 1
        return h_vec
    
    # Initialize temperature
    T = 1.0
    cooling_rate = 0.999
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 5000
    
    start_time = time.time()
    
    # Simulated Annealing
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iterations += 1
        
        # Save current solution
        current_c5 = compute_c5(best_h)
        
        # Perturbation: flip a random entry
        idx = np.random.randint(0, n_points)
        new_h = best_h.copy()
        new_h[idx] = 1 - new_h[idx]
        
        # Ensure constraint is still satisfied
        new_h = project_feasible(new_h)
        
        # Evaluate new solution
        new_c5 = compute_c5(new_h)
        
        # Accept or reject
        if new_c5 < current_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            # Probabilistic acceptance
            delta = new_c5 - current_c5
            accept_prob = np.exp(-delta / T)
            if np.random.rand() < accept_prob:
                best_h = new_h.copy()
                best_c5 = new_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        # Update temperature
        T *= cooling_rate
        
        # Random perturbation
        if iterations % 500 == 0:
            # Modify 10% of entries randomly
            num_changes = n_points // 10
            change_indices = np.random.choice(n_points, size=num_changes, replace=False)
            new_h = best_h.copy()
            new_h[change_indices] = 1 - new_h[change_indices]
            new_h = project_feasible(new_h)
            
            # Evaluate new solution
            new_c5 = compute_c5(new_h)
            
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                no_improvement_count = 0
    
    # Final computation of best solution
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
