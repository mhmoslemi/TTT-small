import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective(h):
        # Normalize h to sum to n_points / 2
        h_norm = h * (n_points / 2) / np.sum(h)
        h_norm = np.clip(h_norm, 0, 1)
        correlation = np.correlate(h_norm, 1 - h_norm, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Set up constraints
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Set up bounds (0 to 1 for each variable)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use trust-region reflective method
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize using trust region method
        result = minimize(
            objective,
            h,
            method='TRF',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'init_tr_radius': 0.1}
        )
        
        if result.success:
            h_opt = result.x
            # Project back to feasible region
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            c5 = objective(h_opt)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                
            h = h_opt.copy()
        else:
            # Fallback to random restart if optimization fails
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final refinement using local optimization
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        # Try some random perturbations
        for i in range(n_points):
            h_pert = best_h.copy()
            pert = np.random.normal(0, 0.01)
            h_pert[i] += pert
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            c5_new = objective(h_pert)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_pert.copy()
    
    return (best_h.tolist(), best_c5, n_points)
