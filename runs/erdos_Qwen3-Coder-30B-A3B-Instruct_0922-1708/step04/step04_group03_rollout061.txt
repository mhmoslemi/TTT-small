import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation to get an initial good solution
    # Minimize sum of all overlaps (convex relaxation)
    h_var = cp.Variable(n_points)
    
    # Objective: minimize sum of all overlaps
    # This is a loose relaxation but gives a good starting point
    obj = 0
    for k in range(n_points):
        for i in range(n_points):
            j = (i + k) % n_points
            obj += h_var[i] * (1 - h_var[j])
    
    obj = cp.Minimize(obj)
    
    # Constraints 
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == 'optimal':
            h = h_var.value
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass
        
    # Phase 2: Gradient-based refinement with finite differences
    max_iter = 500
    lr = 0.01
    patience = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update with gradient descent
        h_new = h - lr * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate decay
        if iteration > 0 and iteration % 100 == 0:
            lr *= 0.9
            
        # Early stopping check
        if iteration > 10 and iteration % patience == 0:
            # Check last few iterations
            recent_c5 = []
            for _ in range(5):
                h_test = h.copy()
                grad_test = np.zeros_like(h)
                eps_test = 1e-4
                for i in range(n_points):
                    h_plus = h_test.copy()
                    h_plus[i] += eps_test
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad_test[i] = (c5_plus - c5) / eps_test
                
                h_test_new = h_test - lr * grad_test
                h_test_new = np.clip(h_test_new, 0, 1)
                h_test_new = h_test_new * (n_points / 2) / np.sum(h_test_new)
                corr_test = np.correlate(h_test_new, 1 - h_test_new, mode='full')
                c5_test = np.max(corr_test) * dx
                
                recent_c5.append(c5_test)
                
            if np.std(recent_c5) < 1e-8:
                break
                
    return (best_h.tolist(), best_c5, n_points)
