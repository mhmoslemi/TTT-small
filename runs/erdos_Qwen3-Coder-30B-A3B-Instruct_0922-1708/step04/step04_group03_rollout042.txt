import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():

        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Differential Evolution parameters
    population_size = 50
    mutation_factor = 0.8
    crossover_rate = 0.9
    
    # Initialize population
    population = []
    fitness = []
    
    # Start with the provided solution or uniform distribution
    if 'initial_h_values' in globals():
        population.append(h.copy())
        corr = np.correlate(h, 1 - h, mode='full')
        fitness.append(np.max(corr) * dx)
    else:
        population.append(h.copy())
        corr = np.correlate(h, 1 - h, mode='full')
        fitness.append(np.max(corr) * dx)
        
    # Fill rest of population with random feasible solutions
    for _ in range(population_size - 1):
        individual = np.random.uniform(0, 1, n_points)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        corr = np.correlate(individual, 1 - individual, mode='full')
        fitness.append(np.max(corr) * dx)
        
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    no_improvement_count = 0
    max_no_improvement = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        # DE evolution
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three different individuals
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (n_points / 2) / np.sum(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            jrand = np.random.randint(0, n_points)
            for j in range(n_points):
                if np.random.rand() < crossover_rate or j == jrand:
                    trial[j] = mutant[j]
            
            # Ensure constraints
            trial = np.clip(trial, 0, 1)
            trial = trial * (n_points / 2) / np.sum(trial)
            
            # Evaluate
            corr = np.correlate(trial, 1 - trial, mode='full')
            c5 = np.max(corr) * dx
            
            # Selection
            if c5 < fitness[i]:
                new_population.append(trial)
                new_fitness.append(c5)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = trial.copy()
                    improved = True
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
        
        population = new_population
        fitness = new_fitness
        
        # Local refine every 10 iterations
        if iteration % 10 == 0:
            # Refine the best solution
            h = best_h.copy()
            for refine_iter in range(100):
                if time.time() - start_time > budget_s - 0.5:
                    break
                correlation = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(correlation) * dx
                
                # Compute gradient with finite differences
                grad = np.zeros_like(h)
                eps = 1e-4
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5) / eps
                
                # Adaptive learning rate
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h - lr * grad
                
                # Clip and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                h = h_new
                
                # Check for improvement
                corr = np.correlate(h, 1 - h, mode='full')
                c5_refine = np.max(corr) * dx
                
                if c5_refine < best_c5:
                    best_c5 = c5_refine
                    best_h = h.copy()
                    improved = True
                    
            # Update population and fitness to reflect refinement
            best_idx = np.argmin(fitness)
            if np.max(np.abs(population[best_idx] - best_h)) > 1e-8:
                population[best_idx] = best_h.copy()
                corr = np.correlate(best_h, 1 - best_h, mode='full')
                fitness[best_idx] = np.max(corr) * dx
        
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
