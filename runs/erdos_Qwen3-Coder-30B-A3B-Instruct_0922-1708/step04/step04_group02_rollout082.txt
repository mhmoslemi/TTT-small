import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Multiple initializations
        h_init_list = []
        for i in range(3):
            if i == 0:
                # Uniform
                h = np.ones(n_points) * 0.5
            elif i == 1:
                # Block
                h = np.zeros(n_points)
                h[:n_points//2] = 1.0
            else:
                # Random
                h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * (n_points / 2)
            h_init_list.append(h)
        
        # Evaluate initial points
        best_c5 = float('inf')
        best_h = None
        for h in h_init_list:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        h = best_h
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Relaxation stage
    if time.time() - start_time < budget_s * 0.4:
        try:
            # Convex relaxation with CVXPY
            h_vars = cp.Variable(n_points)
            C = cp.Variable()
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Use a few key correlations to build constraints
            # For example, you might need to derive the possible lags or use 
            # a sampled set of lag values for consistency
            
            # Incorporate some approximations for the overlap constraints
            # Using a subset of lag correlations for the relaxation stage 
            # to reduce computational load  
            max_lag = min(50, n_points)
            for k in range(1, max_lag):  # Use only near neighbors for efficiency
                # Lag k correlation constraint: sum h[i] * (1 - h[i+k])
                term = 0
                for i in range(n_points - k):
                    term += h_vars[i] * (1 - h_vars[i+k])
                constraints.append(term <= C * dx**(-1))
                
            # Solve
            objective = cp.Minimize(C)
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                relaxed_h = h_vars.value
                if relaxed_h is not None:
                    # Rounding to integer
                    t = np.median(relaxed_h)
                    h_rounded = (relaxed_h > t).astype(float)
                    # Adjust sum to match constraint
                    current_sum = np.sum(h_rounded)
                    diff = current_sum - n_points / 2
                    
                    if diff > 0:
                        # Need to decrease ones
                        indices = np.where(h_rounded == 1)[0]
                        flip_indices = np.random.choice(indices, size=int(diff), replace=False)
                        h_rounded[flip_indices] = 0
                    elif diff < 0:
                        # Need to increase zeros
                        indices = np.where(h_rounded == 0)[0]
                        flip_indices = np.random.choice(indices, size=int(-diff), replace=False)
                        h_rounded[flip_indices] = 1
                    
                    h_rounded = h_rounded * (n_points / 2) / np.sum(h_rounded)
                    corr = np.correlate(h_rounded, 1 - h_rounded, mode='full')
                    c5 = np.max(corr) * dx
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_rounded
        except:
            pass
    
    # Local search refinement  
    # Neighborhood exploration
    if time.time() - start_time < budget_s * 0.7:
        # Sample local search around best solution found so far
        best_h_refined = best_h.copy()
        current_c5 = best_c5
        
        # Binary neighborhood adjustment for a subset of indices
        sample_size = max(10, n_points // 15)
        idxs = np.random.choice(n_points, size=sample_size, replace=False)
        
        for idx in idxs:
            for direction in [-1, 1]:
                h_test = best_h_refined.copy()
                if h_test[idx] + direction * 0.05 >= 0 and h_test[idx] + direction * 0.05 <= 1:
                    h_test[idx] += direction * 0.05
                    h_test = np.clip(h_test, 0, 1)
                    # Renormalize
                    h_test = h_test * (n_points / 2) / np.sum(h_test)
                    
                    corr = np.correlate(h_test, 1 - h_test, mode='full')
                    c5 = np.max(corr) * dx
                    
                    if c5 < current_c5:
                        current_c5 = c5
                        best_h_refined = h_test.copy()
        
        best_c5 = current_c5
        best_h = best_h_refined
        
    # Final MC refinement
    if time.time() - start_time < budget_s - 1:
        # Random local refinement for a flexible blend of exploration and exploitation
        h_refined = best_h.copy()
        current_c5 = best_c5
        
        for i in range(1000):  # Simple MC search
            if time.time() - start_time > budget_s - 1:
                break
    
            # Random perturbation
            idx = np.random.randint(0, n_points)
            step = np.random.uniform(-0.05, 0.05)
            
            h_test = h_refined.copy()
            h_test[idx] += step
            
            # Clip
            h_test = np.clip(h_test, 0, 1)
            
            # Renormalize
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            corr = np.correlate(h_test, 1 - h_test, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < current_c5:
                current_c5 = c5
                h_refined = h_test.copy()
                
        best_c5 = current_c5
        best_h = h_refined

    return (best_h.tolist(), best_c5, n_points)
