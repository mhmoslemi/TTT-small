import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 900
    inertia_weight = 0.7
    cognitive_weight = 2.0
    social_weight = 2.0
    local_search_freq = 50
    rejuvenate_freq = 200
    num_rejuvenate = swarm_size // 10
    
    # Initialize swarm
    swarm = []
    pbest_positions = []
    pbest_values = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Generate initial swarm
    swarm.append(h_init.copy())
    pbest_positions.append(h_init.copy())
    pbest_values.append(float('inf'))
    
    for _ in range(swarm_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        swarm.append(h_rand)
        pbest_positions.append(h_rand.copy())
        pbest_values.append(float('inf'))
    
    # Initialize velocities
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(swarm_size)]
    
    # Get initial best
    best_value = float('inf')
    best_position = None
    iter_count = 0
    
    # PSO main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all particles
        for i in range(swarm_size):
            h = swarm[i]
            
            # Compute overlap
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < pbest_values[i]:
                pbest_values[i] = c5
                pbest_positions[i] = h.copy()
            
            # Update global best
            if c5 < best_value:
                best_value = c5
                best_position = h.copy()
        
        # Update swarm
        for i in range(swarm_size):
            h = swarm[i]
            v = velocities[i]
            pbest = pbest_positions[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (inertia_weight * v + 
                 cognitive_weight * r1 * (pbest - h) + 
                 social_weight * r2 * (best_position - h))
            
            # Update position
            h_new = h + v
            
            # Repair constraint
            h_new = np.clip(h_new, 0, 1)
            sum_h = np.sum(h_new)
            diff = sum_h - n_points / 2
            
            if abs(diff) > 1e-8:
                if diff > 0:
                    # Excess - reduce largest values
                    indices = np.argsort(h_new)[::-1]
                    for idx in indices:
                        if diff <= 0:
                            break
                        reduce_amount = min(diff, h_new[idx])
                        h_new[idx] -= reduce_amount
                        diff -= reduce_amount
                else:
                    # Deficit - increase smallest values
                    indices = np.argsort(h_new)
                    for idx in indices:
                        if diff >= 0:
                            break
                        increase_amount = min(-diff, 1 - h_new[idx])
                        h_new[idx] += increase_amount
                        diff += increase_amount
            
            # Normalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update particle
            swarm[i] = h_new
            velocities[i] = v
        
        # Local search
        if iter_count % local_search_freq == 0 and iter_count > 0:
            # Sort by fitness
            fitnesses = [pbest_values[i] for i in range(swarm_size)]
            sorted_indices = np.argsort(fitnesses)
            
            # Perform local search on top 10%
            top_indices = sorted_indices[:swarm_size//10]
            for i in top_indices:
                h = pbest_positions[i].copy()
                for _ in range(10):
                    # Approximate gradient
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - pbest_values[i]) / eps
                    
                    h = h - 0.001 * grad
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    corr_current = correlate(h, 1 - h, mode='full')
                    c5_current = np.max(corr_current) * dx
                    
                    if c5_current < pbest_values[i]:
                        pbest_values[i] = c5_current
                        pbest_positions[i] = h.copy()
        
        # Rejuvenation
        if iter_count % rejuvenate_freq == 0 and iter_count > 0:
            sorted_indices = np.argsort(pbest_values)
            # Replace worst 10% with random particles
            for i in range(num_rejuvenate):
                idx = sorted_indices[-(i+1)]
                h_new = np.random.rand(n_points)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                swarm[idx] = h_new
                pbest_positions[idx] = h_new
                pbest_values[idx] = float('inf')
    
    # Final local refinement on top candidates
    top_candidates = [(pbest_values[i], pbest_positions[i]) for i in range(swarm_size)]
    top_candidates.sort(key=lambda x: x[0])
    
    best_final_value = best_value
    best_final_position = best_position
    
    for c5_val, h_candidate in top_candidates[:5]:
        if time.time() - start_time > budget_s - 10:
            break
        h_refined = h_candidate.copy()
        for _ in range(1000):
            grad = np.zeros(n_points)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - c5_val) / eps
            
            h_refined = h_refined - 0.001 * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            corr_refined = correlate(h_refined, 1 - h_refined, mode='full')
            c5_refined = np.max(corr_refined) * dx
            
            if c5_refined < c5_val:
                c5_val = c5_refined
            else:
                break
            
        if c5_val < best_final_value:
            best_final_value = c5_val
            best_final_position = h_refined.copy()
    
    return (best_final_position.tolist(), best_final_value, n_points)
