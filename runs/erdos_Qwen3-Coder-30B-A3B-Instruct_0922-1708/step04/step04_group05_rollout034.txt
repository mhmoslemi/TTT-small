import numpy as np
import time

def project_feasible(h, n_points):
    """Project h to satisfy bounds and integral constraint."""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, n_points):
    """Compute C5 value for given h."""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_step(h, n_points, eps=1e-4):
    """Compute gradient using finite differences."""
    dx = 2.0 / n_points
    c5_base = compute_c5(h, n_points)
    grad = np.zeros_like(h)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        c5_plus = compute_c5(h_plus, n_points)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

def line_search(h, grad, n_points, max_iter=20):
    """Perform line search to find optimal step size."""
    dx = 2.0 / n_points
    alpha = 1.0
    c5_base = compute_c5(h, n_points)
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n_points)
        c5_new = compute_c5(h_new, n_points)
        if c5_new < c5_base:
            return h_new
        alpha *= 0.5
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 150
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    h = project_feasible(h, n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)

    # Outer loop: grid search with refinement
    N_coarse = 100
    n_low = 50
    n_high = 150
    
    # Coarse grid search
    for _ in range(50):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Generate random candidates
        for i in range(N_coarse):
            if time.time() - start_time > budget_s - 2:
                break
            h_candidate = np.random.rand(n_low)
            h_candidate = project_feasible(h_candidate, n_low)
            # Interpolate to high resolution
            x_orig = np.linspace(0, 2, n_low)
            x_new = np.linspace(0, 2, n_high)
            h_candidate = np.interp(x_new, x_orig, h_candidate)
            h_candidate = project_feasible(h_candidate, n_high)
            
            # Local refinement
            h_refined = h_candidate.copy()
            for _ in range(50):
                if time.time() - start_time > budget_s - 2:
                    break
                grad = gradient_step(h_refined, n_high)
                h_refined = line_search(h_refined, grad, n_high)
                c5 = compute_c5(h_refined, n_high)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_refined.copy()
                    
            # Mutation to escape local minima
            if np.random.rand() < 0.05:
                noise = np.random.normal(0, 0.001, n_high)
                h_refined = np.clip(h_refined + noise, 0, 1)
                h_refined = project_feasible(h_refined, n_high)

    # Final refinement
    for i in range(100):
        if time.time() - start_time > budget_s - 2:
            break
        grad = gradient_step(best_h, n_high)
        best_h = line_search(best_h, grad, n_high)
        c5 = compute_c5(best_h, n_high)
        if c5 < best_c5:
            best_c5 = c5

    # Return best solution
    return (best_h.tolist(), best_c5, n_high)
