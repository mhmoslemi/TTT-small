import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_segments = 10
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Interpolate to n_points
        x_orig = np.linspace(0, 2, len(h_init))
        x_target = np.linspace(0, 2, n_points)
        h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Create sparse representation: segment positions and heights
    def sparse_to_dense(pos, h_vals):
        h = np.zeros(n_points)
        for i in range(len(pos)-1):
            start_idx = int(pos[i] / 2 * n_points)
            end_idx = int(pos[i+1] / 2 * n_points)
            h[start_idx:end_idx] = h_vals[i]
        return h
    
    def dense_to_sparse(h_vals):
        # Find segment boundaries with significant changes
        segments = []
        pos = [0.0]
        heights = []
        prev_val = h_vals[0]
        for i in range(1, len(h_vals)):
            if abs(h_vals[i] - prev_val) > 1e-5:
                pos.append(i * 2.0 / n_points)
                heights.append(prev_val)
                prev_val = h_vals[i]
        pos.append(2.0)
        heights.append(prev_val)
        if len(heights) < n_segments:
            # Fill with uniform heights to reach desired count
            fill = n_segments - len(heights)
            heights.extend([np.mean(heights)] * fill)
            # Adjust positions
            pos = [0.0] + [i * 2.0 / n_segments for i in range(1, n_segments)] + [2.0]
        elif len(heights) > n_segments:
            # Reduce segment count
            pos_reduced = [0.0]
            heights_reduced = []
            step = len(heights) // n_segments
            for i in range(n_segments):
                if i == n_segments - 1:
                    heights_reduced.append(np.mean(heights[i*step:]))
                else:
                    heights_reduced.append(np.mean(heights[i*step:(i+1)*step]))
                pos_reduced.append(pos[min((i+1)*step, len(pos)-1)])
            pos = pos_reduced
            heights = heights_reduced
        return np.array(pos), np.array(heights)
    
    def compute_overlap(pos, h_vals):
        # Non-sparse direct computation for C5 value
        h = sparse_to_dense(pos, h_vals)
        # Compute the overlap integral for all lags using full convolution
        # But use only sufficient lags
        corr = np.correlate(h, 1-h, mode='full')
        c5 = np.max(corr) * dx 
        return c5
    
    # Reset to initial segments
    pos_init, h_init_seg = dense_to_sparse(h_init)
    
    # Objective function for optimization
    def objective(x):
        # x contains [pos[1:-1], h_vals]
        # Since pos[0]=0 and pos[-1]=2, these are fixed
        pos = np.concatenate([[0], x[:n_segments-1], [2]])
        h_values = np.clip(x[n_segments-1:], 0, 1)
        if np.sum((pos[1:] - pos[:-1]) * h_values) != n_points/2:
            # Adjust normalization
            total_width = np.sum(pos[1:] - pos[:-1])
            scale = (n_points/2) / np.sum((pos[1:] - pos[:-1]) * h_values)
            h_values *= scale
    
        return compute_overlap(pos, h_values)
    
    # Constraints
    def constraint_sum(x):
        pos = np.concatenate([[0], x[:n_segments-1], [2]])
        h_values = np.clip(x[n_segments-1:], 0, 1)
        return np.sum((pos[1:] - pos[:-1]) * h_values) - n_points/2
    
    # Bounds and initial guess
    bounds = [(-0.01, 2.01)] * (n_segments-1) + [(0, 1)] * n_segments  # Ensure 0 < pos < 2
    x0 = np.concatenate([pos_init[1:-1], np.clip(h_init_seg, 0, 1)])
    
    # Optimization
    best_x = x0.copy()
    best_c5 = np.inf
    best_pos = None
    best_h = None
    
    # Multi-start
    for _ in range(3):
        # Random perturbations
        x = x0 + np.random.normal(0, 0.05, len(x0))
        # Ensure boundaries
        x = np.clip(x, [b[0] for b in bounds], [b[1] for b in bounds])
        
        try:
            res = minimize(objective, x, method='SLSQP', bounds=bounds, 
                           constraints={'type': 'eq', 'fun': constraint_sum},
                           options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6},
                           callback=lambda x: None)
            if res.success:
                pos = np.concatenate([[0], res.x[:n_segments-1], [2]])
                h_vals = np.clip(res.x[n_segments-1:], 0, 1)
                c5 = compute_overlap(pos, h_vals)
                if c5 < best_c5 and c5 > 0:
                    best_c5 = c5
                    best_pos = pos
                    best_h = h_vals
                    # Update x0 to better solution
                    x0 = res.x
        except:
            continue
            
        if time.time() - start_time > budget_s - 2:
            break
    
    if best_c5 == np.inf:
        # Revert to initial
        best_pos = pos_init
        best_h = h_init_seg
        best_c5 = compute_overlap(pos_init, h_init_seg)
    
    # Final refinement
    final_pos = np.copy(best_pos)
    final_h = np.copy(best_h)
    
    while time.time() - start_time < budget_s - 5:
        # Small local optimization
        h_trial = sparse_to_dense(final_pos, final_h)
        # Refine in one direction
        for i in range(len(final_h)):
            if np.random.rand() < 0.2:  # 20% chance to adjust
                h_mod = final_h.copy()
                delta = np.random.uniform(-0.05, 0.05)
                h_mod[i] = np.clip(h_mod[i] + delta, 0, 1)
                
                pos_mod = final_pos.copy()
                c5_trial = compute_overlap(pos_mod, h_mod)
                if c5_trial < best_c5:
                    final_h = h_mod.copy()
                    best_c5 = c5_trial
                    
        # Check if we're stable
        if np.random.rand() < 0.5:  # Random check
            # Adjacent swaps
            i, j = np.random.choice(len(final_h), size=2, replace=False)
            h_swap = final_h.copy()
            h_swap[i], h_swap[j] = h_swap[j], h_swap[i]
            c5_swap = compute_overlap(final_pos, h_swap)
            if c5_swap < best_c5:
                final_h = h_swap.copy()
                best_c5 = c5_swap
                
        # Ensure boundary conditions
        if time.time() - start_time > budget_s - 5:
            break
    
    # Final validation
    h_final = sparse_to_dense(final_pos, final_h)
    corr_final = np.correlate(h_final, 1-h_final, mode='full')
    c5_final = np.max(corr_final) * dx
    
    return (h_final.tolist(), c5_final, n_points)
