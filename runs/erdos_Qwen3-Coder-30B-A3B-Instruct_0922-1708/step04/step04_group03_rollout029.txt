    ]

        # Therefore direct relaxation doesn't directly help here due to non-convexity. We'll

        pass
    except:
        pass
    
    # Phase 2: Gradient-based optimization with projected updates
    # We'll use projected gradient descent on the actual function
    def evaluate(h):
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Compute initial value
    current_c5 = evaluate(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    iteration = 0
    max_no_improvement = 50
    no_improvement_count = 0
    
    # Projected gradient descent
    step_size = 0.01
    # Start with fine-grained steps and reduce if needed
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Project to valid domain
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update with projected gradient step
        h_new = h - step_size * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new point
        new_c5 = evaluate(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            no_improvement_count = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            no_improvement_count += 1
            
        # Decay step size if no improvements
        if no_improvement_count > 10:
            step_size *= 0.95
            
        # Early stopping
        if no_improvement_count >= max_no_improvement:
            break
            
    return (best_h.tolist(), best_c5, n_points)
