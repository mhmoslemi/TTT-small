import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Genetic Algorithm parameters
    population_size = 30
    generations = 100
    mutation_rate = 0.1
    elite_size = 5
    
    # Initialize population
    population = []
    fitness_scores = []
    
    for _ in range(population_size):
        if 'initial_h_values' in globals() and np.random.rand() < 0.3:
            individual = h.copy()
        else:
            individual = np.random.uniform(0, 1, n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
            individual = np.clip(individual, 0, 1)
        population.append(individual)
        correlation = np.correlate(individual, 1 - individual, mode='full')
        c5 = np.max(correlation) * dx
        fitness_scores.append(c5)
    
    # Genetic Algorithm iterations
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_h = population[0].copy()
        
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = population[np.random.randint(0, population_size // 2)]
            parent2 = population[np.random.randint(0, population_size // 2)]
            
            # Crossover (uniform)
            child = np.where(np.random.rand(n_points) > 0.5, parent1, parent2)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_idx = np.random.randint(0, n_points)
                child[mutation_idx] = np.random.rand()
            
            # Normalize and clip
            child = child * (n_points / 2) / np.sum(child)
            child = np.clip(child, 0, 1)
            
            new_population.append(child)
        
        population = new_population
        
        # Recalculate fitness
        fitness_scores = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness_scores.append(c5)
    
    # Final local refinement using L-BFGS on best solution
    h = best_h.copy()
    
    def objective(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Use scipy.optimize for local refinement
    from scipy.optimize import Bounds, NonlinearConstraint
    
    bounds = Bounds(0, 1)
    nonlinear_constraint = NonlinearConstraint(constraint, 0, 0, jac='2-point')
    
    # Create bounds and constraint for the optimization
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                      constraints=nonlinear_constraint, options={'maxiter': 1000})
    
    if result.success:
        h = result.x
    else:
        # Fallback to simple gradient-based refinement if L-BFGS fails
        h = best_h.copy()
        lr = 0.01
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Compute gradient
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            
            # Update step
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update learning rate
            lr *= 0.999
            h = h_new
            
            # Check improvement
            correlation = np.correlate(h, 1 - h, mode='full')
            new_c5 = np.max(correlation) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    # Final verification
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
