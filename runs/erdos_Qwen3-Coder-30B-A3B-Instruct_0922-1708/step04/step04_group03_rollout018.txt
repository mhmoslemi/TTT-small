import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # DE parameters
    pop_size = 50
    F = 0.8
    CR = 0.9
    max_no_improvement = 100
    no_improvement_count = 0
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            individual = h.copy()
        else:
            individual = np.random.uniform(0, 1, n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        correlation = np.correlate(individual, 1 - individual, mode='full')
        fit = np.max(correlation) * dx
        fitness.append(fit)
        if fit < best_c5:
            best_c5 = fit
            best_h = individual.copy()
    
    # Main DE loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        for i in range(pop_size):
            # Select three different individuals
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (n_points / 2) / np.sum(mutant)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, mutant, population[i])
            
            # Evaluate trial
            correlation = np.correlate(trial, 1 - trial, mode='full')
            trial_fitness = np.max(correlation) * dx
            
            # Selection
            if trial_fitness < fitness[i]:
                population[i] = trial
                fitness[i] = trial_fitness
                improved = True
                if trial_fitness < best_c5:
                    best_c5 = trial_fitness
                    best_h = trial.copy()
        
        # Local refinement on best individual
        if iteration % 20 == 0:
            h = best_h.copy()
            for refine_iter in range(100):
                correlation = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(correlation) * dx
                
                # Compute gradient with finite differences
                grad = np.zeros_like(h)
                eps = 1e-5
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5) / eps
                
                # Adaptive learning rate
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h - lr * grad
                
                # Clip and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                h = h_new
                
                # Check for improvement
                corr = np.correlate(h, 1 - h, mode='full')
                c5_refine = np.max(corr) * dx
                
                if c5_refine < best_c5:
                    best_c5 = c5_refine
                    best_h = h.copy()
                    
            # Update best for DE
            population[0] = best_h.copy()
            correlation = np.correlate(best_h, 1 - best_h, mode='full')
            fitness[0] = np.max(correlation) * dx
        
        # Early stopping
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
    return (best_h.tolist(), best_c5, n_points)
