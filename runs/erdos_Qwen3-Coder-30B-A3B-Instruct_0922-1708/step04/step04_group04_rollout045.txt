import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Coordinate descent approach: optimize one variable at a time
    max_iter = 1000
    tolerance = 1e-6
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        old_h = h.copy()
        improved = False
        
        # Try to improve each variable
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Current value
            current_h = h[i]
            
            # Try small adjustments
            for delta in [-0.01, -0.005, 0.005, 0.01]:
                new_h = current_h + delta
                if new_h < 0 or new_h > 1:
                    continue
                    
                # Test this new value
                h_test = h.copy()
                h_test[i] = new_h
                
                # Normalize
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                h_test = np.clip(h_test, 0, 1)
                
                # Compute C5
                correlation = np.correlate(h_test, 1 - h_test, mode='full')
                c5_test = np.max(correlation) * dx
                
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test.copy()
                
                if c5_test < np.max(np.correlate(h, 1 - h, mode='full')) * dx:
                    h[i] = new_h
                    h = h * (n_points / 2) / np.sum(h)
                    h = np.clip(h, 0, 1)
                    improved = True
                    
        # Check for convergence
        if not improved:
            break
    
    # Final optimization using CVXPY
    # variables and parameters
    h_vars = cp.Variable(n_points)
    C = cp.Variable()
    
    # constraints
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2, C >= 0]
    
    # Build QP constraints for each lag k
    for k in range(1, n_points):
        if time.time() - start_time > budget_s - 1:
            break
        # In the correlation, we're computing h[i] * (1 - h[i+k]) for all valid i
        # For lags since we assume periodicity in correlation calculation, 
        # we compute the sum of h[i] * (1 - h[(i+k)%n_points])
        lag_sum = 0
        for i in range(n_points - k):
            lag_sum += h_vars[i] * (1 - h_vars[i + k])
        for i in range(n_points - k, n_points):
            lag_sum += h_vars[i] * (1 - h_vars[i + k - n_points])
        constraints.append(lag_sum <= C)
    
    # objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # solve
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_solution = h_vars.value
            h = np.clip(h_solution, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            correlation = np.correlate(h, 1 - h, mode='full')
            c5_solution = np.max(correlation) * dx
            
            if c5_solution < best_c5:
                best_c5 = c5_solution
                best_h = h.copy()
    except:
        pass
    
    # Final refinement with gradient descent if not done already
    h = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Estimate gradient numerically
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Update
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-4)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Compute new value
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
