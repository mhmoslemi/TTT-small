import numpy as np
import time

def project_feasible(h, n_points):
    """Project h to satisfy 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    """Compute C5 = max of correlate(h, 1-h) * dx"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_grad(h, dx, eps=1e-5):
    """Gradient of C5 w.r.t. h using finite differences"""
    n = len(h)
    grad = np.zeros(n)
    base_c5 = compute_c5(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, len(h))
        c5_plus = compute_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def line_search(h, grad, dx, max_steps=20):
    """Golden section line search for step size"""
    # Clip gradient to prevent too large steps
    grad = np.clip(grad, -10, 10)
    
    # Initial bounds for step size
    a, b = 0.0, 2.0
    golden_ratio = (3 - np.sqrt(5)) / 2
    
    for _ in range(max_steps):
        # Inner points
        c = a + golden_ratio * (b - a)
        d = b - golden_ratio * (b - a)
        
        h_c = project_feasible(h - c * grad, len(h))
        h_d = project_feasible(h - d * grad, len(h))
        
        c5_c = compute_c5(h_c, dx)
        c5_d = compute_c5(h_d, dx)
        
        if c5_c < c5_d:
            b = d
        else:
            a = c
    
    # Return midpoint of final interval
    step_size = (a + b) / 2
    return step_size
