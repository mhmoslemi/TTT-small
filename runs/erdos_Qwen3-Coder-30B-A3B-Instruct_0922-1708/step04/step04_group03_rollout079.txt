import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation using CVXPY
    # Minimize sum of correlations
    h_var = cp.Variable(n_points)
    # Constraint: 0 <= h[i] <= 1
    constraints = [h_var >= 0, h_var <= 1]
    # Constraint: sum = n_points/2
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # For convex relaxation: minimize sum of overlaps (approximation)
    # Use a simpler formulation to get initial feasible point
    objective = cp.Minimize(cp.sum_squares(h_var))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = h_var.value.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass
    
    # Phase 2: Projected gradient descent
    def correlation(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c_plus = correlation(h_plus)
            grad[i] = (c_plus - correlation(h)) / eps
        return grad
    
    iteration = 0
    max_no_improvement = 50
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        c5_current = correlation(h)
        
        # Gradient computation
        grad = gradient(h)
        
        # Adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        
        # Update
        h_new = h - lr * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        c5_new = correlation(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        h = h_new
        
        # Early stopping
        if no_improvement_count >= max_no_improvement:
            break
            
    return (best_h.tolist(), best_c5, n_points)
