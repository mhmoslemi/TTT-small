import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to meet integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h = h_init.copy()
    
    def evaluate_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def objective(h_vec):
        return evaluate_c5(h_vec)
    
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Use particle swarm optimization approach
    n_particles = 20
    max_iter = 200
    inertia = 0.9
    c1 = 1.5
    c2 = 1.5
    max_vel = 0.1
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest_idx = 0
    gbest_val = float('inf')
    
    for i in range(n_particles):
        # Start near initial solution with some noise
        h = h_init + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_c5(h))
        if pbest_vals[-1] < gbest_val:
            gbest_val = pbest_vals[-1]
            gbest_idx = i
    
    # PSO loop
    start_time = time.time()
    iter_count = 0
    
    # Adaptive population control parameters
    stagnation_count = 0
    max_stagnation = 50
    pop_reduction = 0.5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        iter_count += 1
        # Update inertia weight
        inertia = 0.9 - (0.5 * iteration / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            vel_update = (inertia * velocities[i] +
                         c1 * r1 * (pbest[i] - particles[i]) +
                         c2 * r2 * (pbest[gbest_idx] - particles[i]))
            velocities[i] = np.clip(vel_update, -max_vel, max_vel)
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_val = evaluate_c5(particles[i])
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                if current_val < gbest_val:
                    gbest_val = current_val
                    gbest_idx = i
                    
        stagnation_count += 1
        if stagnation_count >= max_stagnation and n_particles > 5:
            new_n_particles = max(5, int(n_particles * pop_reduction))
            print(f"Reducing particles from {n_particles} to {new_n_particles}")
            n_particles = new_n_particles
            # Retain best particles
            best_indices = np.argsort(pbest_vals)[:n_particles]
            particles = [pbest[i] for i in best_indices]
            velocities = [np.random.uniform(-max_vel, max_vel, n_points) for _ in range(n_particles)]
            pbest = [pbest[i] for i in best_indices]
            pbest_vals = [pbest_vals[i] for i in best_indices]
            gbest_idx = np.argmin(pbest_vals)
            gbest_val = pbest_vals[gbest_idx]
            stagnation_count = 0
            
    # Final refinement using local optimization
    refine_particles = []
    refine_scores = []
    
    for i in range(min(5, len(pbest))):
        refine_particles.append(pbest[i])
        refine_scores.append(pbest_vals[i])
    
    best_h = pbest[gbest_idx].copy()
    best_c5 = gbest_val
    
    # Try to improve the best solution using local optimization
    attempts = 0
    max_attempts = 5
    while attempts < max_attempts and time.time() - start_time < budget_s - 10:
        attempts += 1
        ref_h = best_h.copy()
        # Perturb to find a better solution
        perturb = np.random.normal(0, 0.01, n_points)
        ref_h = np.clip(ref_h + perturb, 0, 1)
        ref_h = ref_h * (n_points / 2) / np.sum(ref_h)
        
        # Use scipy minimize for local refinement
        def obj_func(h_perturb):
            corr = np.correlate(h_perturb, 1 - h_perturb, mode='full')
            return np.max(corr) * dx
        
        # Simple projection method for constraint check
        def constraint_func(x):
            return np.sum(x) - n_points / 2
            
        try:
            result = minimize(obj_func, ref_h, method='L-BFGS-B', 
                             bounds=[(0, 1) for _ in range(n_points)],
                             constraints={'type': 'eq', 'fun': constraint_func})
            if result.success:
                new_val = obj_func(result.x)
                if new_val < best_c5:
                    best_c5 = new_val
                    best_h = result.x.copy()
        except:
            pass
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
