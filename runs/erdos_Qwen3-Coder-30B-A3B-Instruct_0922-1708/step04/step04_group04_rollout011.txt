import numpy as np
import time
from scipy import signal
import heapq

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Priority queue items: (negative_overlap, shift_index)
    shift_queue = []
    
    # Helper to compute overlap for all shifts
    def compute_all_overlaps(h_vec):
        # Use FFT for efficiency
        h_rev = h_vec[::-1]
        corr = signal.fftconvolve(h_vec, h_rev, mode='full')
        corr = corr[len(h_vec)-1:]  # Only positive shifts
        corr = corr * dx
        return corr
    
    # Initialize overlaps
    overlaps = compute_all_overlaps(h)
    max_shift = len(overlaps) // 2
    for i in range(len(overlaps)):
        if i != max_shift:
            heapq.heappush(shift_queue, (-overlaps[i], i-max_shift))
    
    # Adaptive parameters
    w = 5
    perturb_scale = 0.02
    improvement_threshold = 1e-6
    no_improvement_count = 0
    max_no_improvement = 10
    perturb_scale_factor = 1.0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 2000:
        iteration += 1
        
        # Recompute overlaps periodically to prevent staleness
        if iteration % 50 == 0:
            overlaps = compute_all_overlaps(h)
            shift_queue = []
            for i in range(len(overlaps)):
                if i != max_shift:
                    heapq.heappush(shift_queue, (-overlaps[i], i-max_shift))
        
        if not shift_queue:
            break
            
        # Get worst shift
        neg_overlap, shift_k = heapq.heappop(shift_queue)
        overlap_val = -neg_overlap
        
        # Update best if better
        if overlap_val < best_c5:
            best_c5 = overlap_val
            best_h = h.copy()
            
        # Identify regions of maximum impact
        # For shift k, we consider indices where
        # h[i] * (1 - h[i + k]) is maximized (taking k as positive)
        # or (1 - h[i]) * h[i - k] if k is negative
        
        # For now, apply focused adjustment on the affected region
        # If shift is positive (k > 0), we have h[i] and h[i+k]
        # If shift is negative, we have h[i] and h[i+k] (i+k < i) so this
        # can be rewritten as i' = i+k, then h[i'] * (1 - h[i'])
        # But we're covering all overlaps via the full correlation 
        # Let's focus on local adjustment regardless
        
        # Find the index where the overlap is maximum (correlation peak)
        # The corresponding shift corresponds to the overlap
        if shift_k >= 0:
            # h[i] * (1 - h[i + shift_k]) -> use end of h_shrink for this
            # For alignment, we'll just pick a few indices around
            # Actually, for all, we'll adjust at the peak
            peak_idx = shift_k + max_shift
            # Adjust area around peak index to reduce impact
            start_idx = max(0, peak_idx - w)
            end_idx = min(n_points, peak_idx + w)
            
            # Simple adjustment using gradient-like approach:
            # Find pairs where we might reduce overlap
            # Approximation: adjust points near area affected by shift_k
            # Use gradient estimation via finite differences
            
            # Reset the adjustment range
            h_adjust = h.copy()
            for idx in range(start_idx, end_idx):
                # For simplicity, we do averaging with nearby points, and balance
                # Make sure it's within bounds and normalization
                # Do a simple tweak around the indices from shift impact
                
                # Surrounding means to smooth
                center = idx
                left = max(0, center - 1)
                right = min(n_points - 1, center + 1)
                val = h_adjust[center]
                
                # Energy: reduce (val * (1 - val)) where influence is high
                if shift_k > 0:
                    # We have h[i] * (1 - h[i + shift_k])
                    # This is a positive overlap
                    # Reduce h[i] and try to increase h[i + shift_k]
                    if center + shift_k < n_points:
                        loss_val = h_adjust[center + shift_k]
                        # We try to shift some amount from loss_val to center
                        if val > 0.1 and loss_val < 0.9:
                            max_adjust = min(val, 1 - loss_val)
                            adjust = min(0.05, max_adjust * 0.5)
                            h_adjust[center] = val - adjust
                            h_adjust[center + shift_k] = loss_val + adjust
                else:
                    # shift_k < 0
                    # h[i] * (1 - h[i + shift_k]) where i + shift_k < i
                    shifted = center + shift_k
                    if shifted >= 0:
                        loss_val = h_adjust[shifted]
                        if val > 0.1 and loss_val < 0.9:
                            max_adjust = min(val, 1 - loss_val)
                            adjust = min(0.05, max_adjust * 0.5)
                            h_adjust[center] = val - adjust
                            h_adjust[shifted] = loss_val + adjust

            # Clip and rescale
            h_adjust = np.clip(h_adjust, 0, 1)
            h_adjust = h_adjust * (n_points / 2) / np.sum(h_adjust)
        else:
            # For negative shifts, we adjust same principle
            # But generalize to a smoothing rather than very specific
            h_adjust = h.copy()
            # Simple adjustment around where impact is highest (choose small area)
            center_idx = max_shift + shift_k
            start_idx = max(0, center_idx - w)
            end_idx = min(n_points, center_idx + w)
            
            # Apply average to the window
            values = h_adjust[start_idx:end_idx]
            mean_val = np.mean(values)
            for idx in range(start_idx, end_idx):
                # Add randomness to break symmetry in case of plateau
                h_adjust[idx] = mean_val + np.random.normal(0, 0.005) * perturb_scale  
            
            h_adjust = np.clip(h_adjust, 0, 1)
            h_adjust = h_adjust * (n_points / 2) / np.sum(h_adjust)
            
        # Apply some random perturbations at times
        if iteration % 10 == 0:
            for i in range(n_points):
                if np.random.random() < 0.03:
                    change = np.random.normal(0, perturb_scale * perturb_scale_factor)
                    h_adjust[i] += change
            h_adjust = np.clip(h_adjust, 0, 1)
            h_adjust = h_adjust * (n_points / 2) / np.sum(h_adjust)
        
        h = h_adjust.copy()
        
        # Compute new overlaps and update priority queue
        overlaps = compute_all_overlaps(h)
        no_improvement_count += 1
        improvement = False

        for i in range(len(overlaps)):
            if i != max_shift:
                heapq.heappush(shift_queue, (-overlaps[i], i-max_shift))
                
        # Check and adjust parameters based on stagnation
        if no_improvement_count >= max_no_improvement:
            perturb_scale_factor *= 2
            no_improvement_count = 0
            
    # Final evaluation
    final_overlaps = compute_all_overlaps(best_h)
    final_c5 = np.max(final_overlaps)
    
    return (best_h.tolist(), final_c5, n_points)
