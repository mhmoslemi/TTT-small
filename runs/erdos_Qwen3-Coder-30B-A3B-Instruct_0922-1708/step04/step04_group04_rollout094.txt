import numpy as np
from scipy.stats import qmc
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5

    # Rescale to meet sum constraint exactly
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # LHS sampling setup
    sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs_samples = sampler.random(n=50)
    
    # Evaluate and refine each sample
    for i in range(50):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Transform LHS values to [0,1] range and rescale
        candidate_h = lhs_samples[i] * 0.8 + 0.1  # Ensure within [0.1, 0.9]
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Gradient descent refinement
        local_h = candidate_h.copy()
        prev_c5 = float('inf')
        lr = 0.01
        for iter_count in range(500):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Compute current C5
            correlation = np.correlate(local_h, 1 - local_h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = local_h.copy()
            
            # Convergence check
            if abs(prev_c5 - c5) < 1e-6:
                break
            prev_c5 = c5
            
            # Compute gradient via finite differences
            grad = np.zeros_like(local_h)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = local_h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = local_h.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Adaptive learning rate
            local_h -= lr * grad
            local_h = np.clip(local_h, 0, 1)
            local_h = local_h * (n_points / 2) / np.sum(local_h)
            
            # Reduce learning rate if stuck
            if iter_count % 50 == 0 and iter_count > 0:
                lr *= 0.5
                if lr < 1e-4:
                    lr = 1e-4
    
    # Final adjustment with directed optimization
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Gradient descent refinement using finite differences
        grad = np.zeros_like(best_h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Adaptive learning rate
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
        new_h = best_h - lr * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new C5
        correlation = np.correlate(new_h, 1 - new_h, mode='full')
        c5_new = np.max(correlation) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()
            
        if abs(c5 - c5_new) < 1e-8:
            break
    
    return (best_h.tolist(), best_c5, n_points)
