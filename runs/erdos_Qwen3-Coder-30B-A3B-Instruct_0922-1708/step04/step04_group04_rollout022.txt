import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up problem parameters
    n_points = 100
    dx = 2.0 / n_points
    total_sum = n_points / 2
    
    # Create initial guess from parent or uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to meet constraint
    h_init = h_init * total_sum / np.sum(h_init)
    h_init = np.clip(h_init, 0, 1)
    
    # Prepare for optimization
    h = cp.Variable(n_points)
    C = cp.Variable()
    
    # Build constraints for all possible shifts
    constraints = [h >= 0, h <= 1, cp.sum(h) == total_sum, C >= 0]
    
    # For each shift k, add constraint sum(h[i] * (1 - h[i+k])) <= C
    # We compute correlations manually to match the formulation
    for k in range(1, n_points):
        # Create vector for shifted terms (this is a manual computation)
        # The overlap integral at shift k is sum_{i=0}^{n_points-k-1} h[i]*(1-h[i+k])
        expr = 0
        for i in range(n_points - k):
            expr += h[i] * (1 - h[i + k])
        constraints.append(expr <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Setup and solve using ECOS (fast and reliable for this problem)
    prob = cp.Problem(objective, constraints)
    
    # Time tracking
    start_time = time.time()
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-5, reltol=1e-5)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INaccurate]:
            h_opt = h.value
        else:
            h_opt = h_init  # fallback
    except Exception as e:
        h_opt = h_init  # try fallback if solver error
    
    # Post-processing to ensure feasibility
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * total_sum / np.sum(h_opt)
    
    # Compute actual C5 value
    correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_bound = np.max(correlation) * dx
    
    # Return solution
    return (h_opt.tolist(), c5_bound, n_points)
