import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize starting point
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Relaxation with CVXPY
    if time.time() - start_time < budget_s * 0.4:
        try:
            h_relaxed = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points / 2]
            
            # Use smaller subset for effective constraint resolution
            max_lag = min(20, n_points // 2)
            if max_lag > 0:
                for k in range(1, max_lag):
                    corr_expr = 0
                    for i in range(n_points - k):
                        corr_expr += h_relaxed[i] * (1 - h_relaxed[i + k])
                    corr_expr *= dx
                    constraints.append(corr_expr <= 1.0)  # Arbitrary large bound
                
            # Solve convex relaxation
            prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed_sol = h_relaxed.value
                h_candidate = h_relaxed_sol.copy()
                
                # Convert to binary with median thresholding
                threshold = np.median(h_candidate)
                h_binary = (h_candidate > threshold).astype(float)
                
                # Adjust sum to match constraint
                current_sum = np.sum(h_binary)
                diff = current_sum - n_points / 2
                
                if abs(diff) > 1e-6:
                    # Determine how many to flip
                    flip_count = int(abs(diff))
                    flip_indices = np.argsort(np.abs(h_candidate - threshold))[:flip_count]
                    if diff > 0:
                        # Too many 1s; flip some 1s to 0s
                        h_binary[flip_indices] = 0
                    else:
                        # Too many 0s; flip some 0s to 1s
                        h_binary[flip_indices] = 1
                    
                h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
                h = h_binary
                
        except Exception:
            pass
    
    # Local search + refinement using smoother step structure
    while time.time() - start_time < budget_s - 1:
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Perturb current solution for local search
        h_new = h.copy()
        
        # Small random perturbations with adaptive step size
        step = min(0.02, 0.1 * (budget_s - (time.time() - start_time)) / budget_s)
        
        # Perturb random few points
        for i in np.random.choice(n_points, size=max(1, n_points // 20), replace=False):
            direction = np.random.choice([-1, 1])
            h_trial = h_new.copy()
            h_trial[i] += direction * step
            
            # Clip and rescale
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            corr_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
            c5_trial = np.max(corr_trial) * dx
            
            if c5_trial < c5:
                h = h_trial.copy()
                c5 = c5_trial
                
    # Final adjustment - straightforward descent search for a few steps
    if time.time() - start_time < budget_s - 0.5:
        for _ in range(20):
            h_trial = h.copy()
            
            # Sample few indices for adjustment
            sample_indices = np.random.choice(n_points, size=max(5, n_points // 50), replace=False)
            
            for i in sample_indices:
                # Try nearby values for exploration
                candidates = []
                for delta in [-0.02, -0.01, 0.00, 0.01, 0.02]:
                    h_test = h_trial.copy()
                    h_test[i] += delta
                    h_test = np.clip(h_test, 0, 1)
                    h_test = h_test * (n_points / 2) / np.sum(h_test)
                    
                    corr_test = np.correlate(h_test, 1 - h_test, mode='full')
                    c5_test = np.max(corr_test) * dx
                    
                    candidates.append((c5_test, h_test.copy()))
                
                # Take best among candidates
                best_cand = min(candidates, key=lambda x: x[0])
                if best_cand[0] < np.max(np.correlate(h_trial, 1 - h_trial, mode='full')) * dx:
                    h_trial = best_cand[1]
            
            h = h_trial
            
    # Final check if we improved anything
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
