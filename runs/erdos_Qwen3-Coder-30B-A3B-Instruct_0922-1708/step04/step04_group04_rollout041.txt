import numpy as np
from scipy.stats import qmc
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Latin Hypercube Sampling with refinement
    lhs_sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs_samples = lhs_sampler.random(n=50)
    
    # Convert LHS samples to valid h vectors
    valid_h_samples = []
    for sample in lhs_samples:
        # Transform to [0,1] range
        h_sample = sample
        # Rescale to sum to n_points/2
        h_sample = h_sample * (n_points / 2) / np.sum(h_sample)
        h_sample = np.clip(h_sample, 0, 1)
        valid_h_samples.append(h_sample)
    
    # Add the parent solution if available
    if 'initial_h_values' in globals():
        valid_h_samples.append(h)
    
    # Evaluate samples and refine
    for sample_idx, h_candidate in enumerate(valid_h_samples):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Refinement loop
        h_refined = h_candidate.copy()
        current_c5 = float('inf')
        
        for iteration in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute overlap integral
            correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < current_c5:
                current_c5 = c5
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_refined.copy()
            
            # Gradient estimation via finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h_refined.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with adaptive learning rate
            lr = 0.01 / (1 + iteration * 1e-4)
            h_new = h_refined - lr * grad
            
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_refined = h_new
            
            if iteration > 0 and abs(current_c5 - c5) < 1e-6:
                break
                
        # Final check
        correlation_final = np.correlate(h_refined, 1 - h_refined, mode='full')
        final_c5 = np.max(correlation_final) * dx
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_refined.copy()
    
    return (best_h.tolist(), best_c5, n_points)
