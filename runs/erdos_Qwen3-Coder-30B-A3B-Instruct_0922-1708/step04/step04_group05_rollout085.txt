import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation with exactly n_points/2 ones
    h_binary = (h_init >= 0.5).astype(int)
    ones_count = np.sum(h_binary)
    target_ones = n_points // 2
    
    if ones_count > target_ones:
        # Turn some 1s to 0s
        zero_indices = np.where(h_binary == 1)[0]
        np.random.shuffle(zero_indices)
        for idx in zero_indices[:ones_count - target_ones]:
            h_binary[idx] = 0
    elif ones_count < target_ones:
        # Turn some 0s to 1s
        one_indices = np.where(h_binary == 0)[0]
        np.random.shuffle(one_indices)
        for idx in one_indices[:target_ones - ones_count]:
            h_binary[idx] = 1
    
    # Ensure the integral constraint
    h = h_binary.astype(float)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Simulated Annealing parameters
    T = 1.0
    decay = 0.999
    max_iter = 10000
    perturbation_interval = 1000
    perturbation_rate = 0.05
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Make a move: flip a random bit
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]
        
        # Project to ensure exactly target_ones ones
        current_ones = np.sum(h_new)
        if current_ones > target_ones:
            # Turn some ones to zeros
            one_indices = np.where(h_new == 1)[0]
            np.random.shuffle(one_indices)
            for idx in one_indices[:int(current_ones - target_ones)]:
                h_new[idx] = 0
        elif current_ones < target_ones:
            # Turn some zeros to ones
            zero_indices = np.where(h_new == 0)[0]
            np.random.shuffle(zero_indices)
            for idx in zero_indices[:int(target_ones - current_ones)]:
                h_new[idx] = 1
        
        # Compute C5 values
        c5_current = compute_c5(h)
        c5_new = compute_c5(h_new)
        
        # Accept or reject the move
        if c5_new < c5_current:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            # Accept with probability based on temperature
            delta = c5_new - c5_current
            if np.random.random() < np.exp(-delta / T):
                h = h_new
        
        # Update temperature
        T *= decay
        
        # Occasionally perturb the solution
        if iter_count % perturbation_interval == 0:
            num_perturb = int(n_points * perturbation_rate)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for i in indices:
                h[i] = 1 - h[i]
            
            # Re-project to maintain constraint
            current_ones = np.sum(h)
            if current_ones > target_ones:
                one_indices = np.where(h == 1)[0]
                np.random.shuffle(one_indices)
                for idx in one_indices[:int(current_ones - target_ones)]:
                    h[idx] = 0
            elif current_ones < target_ones:
                zero_indices = np.where(h == 0)[0]
                np.random.shuffle(zero_indices)
                for idx in zero_indices[:int(target_ones - current_ones)]:
                    h[idx] = 1
        
        # Early exit if we're stuck
        if iter_count % 1000 == 0 and iter_count > 0:
            if abs(compute_c5(h) - best_c5) < 1e-8:
                break
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
