import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise use uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to satisfy integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    n_particles = 20
    max_iter = 200
    w_start, w_end = 0.9, 0.4
    c1, c2 = 1.5, 1.5
    inertia_decay = (w_start - w_end) / max_iter
    
    # Initialize particles and velocities
    particles = np.random.uniform(0.1, 0.9, (n_particles, n_points))
    for i in range(n_particles):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    # Track best particles
    best_positions = particles.copy()
    best_scores = np.full(n_particles, float('inf'))
    
    # PSO optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate current particles
        for i in range(n_particles):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            score = np.max(corr) * dx
            if score < best_scores[i]:
                best_scores[i] = score
                best_positions[i] = h.copy()
        
        # Global best
        global_best_idx = np.argmin(best_scores)
        global_best = best_positions[global_best_idx]
        
        # Update particles
        w = w_start - inertia_decay * iteration
        for i in range(n_particles):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (best_positions[i] - particles[i]) +
                            c2 * r2 * (global_best - particles[i]))
            particles[i] += velocities[i]
            
            # Projection constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
        # Add noise periodically
        if iteration % 20 == 0:
            noise = np.random.normal(0, 0.005, (n_particles, n_points))
            particles += noise
            for i in range(n_particles):
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Refinement stage
    top_candidates = np.argsort(best_scores)[:5]
    best_c5 = float('inf')
    best_h = global_best.copy()
    
    # Line search refinement for top candidates
    for idx in top_candidates:
        h = best_positions[idx].copy()
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final check
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
