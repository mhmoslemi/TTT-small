import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate diverse starting candidates
    candidates = []
    for i in range(20):
        if i == 0:
            h = h_init.copy()
        else:
            # Add perturbations and systematic shifts
            h = h_init + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        candidates.append(h)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Gradient-based local refinement approach
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # For each candidate, compute and optimize gradient
        for candidate in candidates:
            h = candidate.copy()
            
            # Compute gradients
            grad = np.zeros(n_points)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Gradient descent update with adaptive learning rate
            lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Check if the new solution is better
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            # Update candidate
            candidate[:] = h_new
            
            # Add new candidate periodically
            if iteration % 50 == 0 and time.time() - start_time < budget_s - 1:
                h_new = best_h + np.random.normal(0, 0.01, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                candidates.append(h_new)

    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
