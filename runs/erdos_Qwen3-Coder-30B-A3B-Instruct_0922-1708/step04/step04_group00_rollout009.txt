import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def objective(h_vec):
        # Reshape to column vector for correlation
        h_vec = h_vec.reshape(-1, 1)
        # Compute correlation and find maximum
        corr = np.correlate(h_vec.flatten(), 1 - h_vec.flatten(), mode='full')
        return -np.max(corr) * dx  # Negative because we minimize
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Set up optimization parameters
    max_iter = 500
    optim_time = 0
    
    # Starting point
    current_h = h.copy()
    best_h = current_h.copy()
    best_c5 = float('inf')
    
    # Trust region optimization with multiple initializations
    for i in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random slight perturbations
        if i == 0:
            h_init = current_h.copy()
        else:
            # Small random noise
            h_init = current_h + np.random.normal(0, 0.01, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Optimization with trust-region reflective method
        try:
            res = minimize(
                objective,
                h_init,
                method='TRF',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter // 5, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if res.success:
                optimized_h = res.x
                # Ensure it satisfies constraints
                optimized_h = np.clip(optimized_h, 0, 1)
                optimized_h = optimized_h * (n_points / 2) / np.sum(optimized_h)
                
                # Compute final C5
                corr = np.correlate(optimized_h, 1 - optimized_h, mode='full')
                c5 = np.max(corr) * dx
                
                # Update best
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = optimized_h.copy()
                    
        except Exception:
            continue
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Local refinement
    h_refined = best_h.copy()
    initial_c5 = best_c5
    for refinement_iter in range(200):
        if time.time() - start_time > budget_s - 1:
            break
            
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        c5 = np.max(corr) * dx
        if np.abs(c5 - initial_c5) < 1e-8:
            break
            
        # Finite difference gradient
        grad = np.zeros_like(h_refined)
        eps = 1e-5
        
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Update step
        lr = 0.01
        h_new = h_refined - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_refined = h_new
        
        # Update best if improved
        corr_new = np.correlate(h_refined, 1 - h_refined, mode='full')
        c5_new = np.max(corr_new) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_refined.copy()
            
    return (best_h.tolist(), best_c5, n_points)
