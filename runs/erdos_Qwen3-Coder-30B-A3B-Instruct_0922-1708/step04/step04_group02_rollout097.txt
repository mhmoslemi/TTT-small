import numpy as np
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Clip and normalize
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    temperature = 1.0
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        
        # Generate candidate solution by perturbing 10% of points
        candidate_h = h.copy()
        indices_to_change = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for idx in indices_to_change:
            # Random change in the value within bounds
            change = np.random.uniform(-0.1, 0.1)
            candidate_h[idx] += change
        
        # Ensure bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Adjust to maintain sum constraint
        sum_diff = np.sum(candidate_h) - n_points / 2
        if abs(sum_diff) > 1e-10:
            # Redistribute the difference
            avg_change = sum_diff / n_points
            candidate_h -= avg_change
        
        # Ensure final bounds again
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Ensure normalization
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Compute new C5
        new_c5 = compute_c5(candidate_h)
        
        # Accept with simulated annealing probability
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < math.exp(-delta / temperature):
            h = candidate_h
            current_c5 = new_c5
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
        
        # Cooling schedule
        if iteration % 100 == 0:
            temperature *= 0.99
        
        # Early stopping if no improvement for a while
        if iteration - last_improvement > 2000:
            break
    
    # Final local search
    for _ in range(1000):
        if time.time() - start_time > budget_s - 2:
            break
            
        candidate_h = best_h.copy()
        # Make small random changes
        for _ in range(20):
            idx = np.random.randint(n_points)
            change = np.random.uniform(-0.01, 0.01)
            candidate_h[idx] += change
        candidate_h = np.clip(candidate_h, 0, 1)
        # Maintain sum constraint
        sum_diff = np.sum(candidate_h) - n_points / 2
        if abs(sum_diff) > 1e-10:
            avg_change = sum_diff / n_points
            candidate_h -= avg_change
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        new_c5 = compute_c5(candidate_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = candidate_h
    
    return (best_h.tolist(), best_c5, n_points)
