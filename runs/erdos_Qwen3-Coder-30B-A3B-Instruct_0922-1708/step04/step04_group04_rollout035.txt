import numpy as np
import time
from heapq import heappush, heappop
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_overlap(h):
        # Full correlation (adds padding)
        corr = correlate(h, 1-h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def validate_and_rescale(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Priority queue for shifts
    shifts_queue = []
    
    # Track best
    best_h = h.copy()
    best_c5 = compute_overlap(h)
    
    start_time = time.time()
    
    # Window for localized adjustments
    w = 5
    perturb_scale = 0.02
    improvement_threshold = 1e-6
    
    iterations = 0
    no_improvement_count = 0
    max_iterations = 1000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iterations:
        iterations += 1
        
        # Compute all overlaps
        corr = correlate(h, 1-h, mode='full')
        # The overlap for shift k is at index k + (n-1) in the correlation array
        overlaps = {}
        for k in range(-(n_points-1), n_points):
            if k != 0:
                idx = k + (n_points-1)
                overlaps[k] = corr[idx] * dx
        
        # Update priority queue
        shifts_queue = []
        for k, val in overlaps.items():
            heappush(shifts_queue, (-val, k))  # Negative for max-heap behavior
            
        # Extract most problematic shift
        if not shifts_queue:
            break
            
        neg_max_val, k = heappop(shifts_queue)
        max_val = -neg_max_val
        
        # Determine index range that contributes to this overlap
        start_idx = max(0, -k) if k < 0 else 0
        end_idx = min(n_points, n_points - k) if k > 0 else n_points
        
        # Apply localized gradient-like update
        h_new = h.copy()
        
        # Compute local overlap for this shift
        overlap_indices = range(start_idx, end_idx)
        
        # Start to penalize overlap by adjusting h values
        for i in overlap_indices:
            # The contribution of h[i] to overlap with h[i + k] is h[i] * (1 - h[i + k])
            # To minimize overlap, we want to increase h[i] where (1 - h[i + k]) is large,
            # and decrease h[i] where (1 - h[i + k]) is small
            if k > 0:
                j = i + k
                if j < n_points:
                    if h[i] > 0.5:
                        # Decrease h[i] to reduce overlap
                        h_new[i] -= 0.01
                    else:
                        # Increase h[i] 
                        h_new[i] += 0.01
            else:  # k < 0
                j = i + k
                if j >= 0:
                    if h[i] > 0.5:
                        # Decrease h[i] to reduce overlap
                        h_new[i] -= 0.01
                    else:
                        # Increase h[i] 
                        h_new[i] += 0.01
                        
        # Apply random perturbations within shorter range
        for _ in range(20):
            idx = np.random.randint(0, n_points)
            h_new[idx] += np.random.normal(0, perturb_scale)
            
        h_new = validate_and_rescale(h_new)
        
        # Test new solution
        new_c5 = compute_overlap(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        h = h_new
        
        # Adapt hyperparameters
        if no_improvement_count > 10:
            w = max(1, w - 1)
            perturb_scale *= 1.5
            no_improvement_count = 0
            
    return (best_h.tolist(), best_c5, n_points)
