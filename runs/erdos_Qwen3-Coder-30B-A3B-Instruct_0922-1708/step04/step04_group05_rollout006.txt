import numpy as np
import time

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, dx):
    correlation_full = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation_full) * dx

def gradient_step(h, dx, eps=1e-4):
    grad = np.zeros_like(h)
    base_c5 = compute_c5(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, len(h))
        c5_plus = compute_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

def golden_section_search(f, a, b, tol=1e-5, max_iter=50):
    gr = (np.sqrt(5) - 1) / 2
    c = a + (1 - gr) * (b - a)
    d = a + gr * (b - a)
    for _ in range(max_iter):
        if abs(c - d) < tol:
            break
        if f(c) < f(d):
            b = d
            d = c
            c = a + (1 - gr) * (b - a)
        else:
            a = c
            c = d
            d = a + gr * (b - a)
    return (a + b) / 2

def refine_solution(h, dx, max_iter=100, tol=1e-5):
    h = project_feasible(h, len(h))
    old_c5 = compute_c5(h, dx)
    
    for _ in range(max_iter):
        grad = gradient_step(h, dx)
        if np.linalg.norm(grad) < 1e-8:
            break
            
        # Line search for optimal step size
        def phi(alpha):
            h_new = h - alpha * grad
            h_new = project_feasible(h_new, len(h))
            return compute_c5(h_new, dx)
            
        alpha_opt = golden_section_search(phi, 0, 1)
        h = h - alpha_opt * grad
        h = project_feasible(h, len(h))
        
        c5 = compute_c5(h, dx)
        if abs(old_c5 - c5) < tol:
            break
        old_c5 = c5
        
        if time.time() > time.time() + 50:  # Early exit if near budget
            break
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 150
    dx = 2.0 / n_points
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        
    h_init = project_feasible(h_init, n_points)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, dx)
    
    # Coarse grid search
    coarse_points = 50
    for i in range(100):
        if time.time() - start_time > budget_s - 20:
            break
            
        h = h_init.copy()
        # Add some randomness
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = project_feasible(h, n_points)
        
        # Refine with gradient descent
        h_refined = refine_solution(h, dx)
        c5_refined = compute_c5(h_refined, dx)
        
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    # Final refinement
    final_h = refine_solution(best_h, dx)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
