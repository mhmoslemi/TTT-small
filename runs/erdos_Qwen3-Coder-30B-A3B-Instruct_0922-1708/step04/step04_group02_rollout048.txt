import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize so sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated annealing parameters
    temperature = 1.0
    cooling_factor = 0.99
    steps_per_temp = 100
    perturb_fraction = 0.1
    perturb_magnitude = 0.1
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate candidate solution
        h_new = h.copy()
        n_perturb = max(1, int(n_points * perturb_fraction))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        for i in indices:
            # Perturb value
            delta = np.random.uniform(-perturb_magnitude, perturb_magnitude)
            h_new[i] += delta
        
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2
        diff = target_sum - current_sum
        
        # Redistribute the difference
        if abs(diff) > 1e-10:
            # Distribute the difference equally among all points
            adjustment = diff / n_points
            h_new += adjustment
            # Reclip to keep within bounds
            h_new = np.clip(h_new, 0, 1)
        
        # Compute candidate C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        candidate_c5 = np.max(correlation_new) * dx
        
        # Accept or reject according to simulated annealing criteria
        delta = candidate_c5 - current_c5
        
        if delta < 0 or np.random.random() < np.exp(-delta / temperature):
            h = h_new.copy()
        
        # Decrease temperature
        if iteration % steps_per_temp == 0:
            temperature *= cooling_factor
    
    # Final refinement: local search around best solution
    h_final = best_h.copy()
    best_c5 = float('inf')
    
    # Local search using gradient-like perturbations
    for _ in range(5000):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_test = h_final.copy()
        
        # Select random indices to modify
        n_modify = max(1, n_points // 10)
        indices = np.random.choice(n_points, size=n_modify, replace=False)
        
        # Modify values
        for i in indices:
            h_test[i] += np.random.uniform(-0.01, 0.01)
            
        # Ensure bounds
        h_test = np.clip(h_test, 0, 1)
        
        # Adjust sum
        current_sum = np.sum(h_test)
        target_sum = n_points / 2
        diff = target_sum - current_sum
        
        if abs(diff) > 1e-10:
            adjustment = diff / n_points
            h_test += adjustment
            h_test = np.clip(h_test, 0, 1)
        
        # Evaluate
        correlation = np.correlate(h_test, 1 - h_test, mode='full')
        test_c5 = np.max(correlation) * dx
        
        if test_c5 < best_c5:
            best_c5 = test_c5
            h_final = h_test.copy()
    
    # Final evaluation
    correlation_final = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(correlation_final) * dx
    best_c5 = min(best_c5, final_c5)
    
    return (h_final.tolist(), best_c5, n_points)
