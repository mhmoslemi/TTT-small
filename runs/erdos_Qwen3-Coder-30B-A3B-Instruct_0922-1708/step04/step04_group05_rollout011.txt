import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise create binary
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        # Convert to binary
        h = (h_init >= 0.5).astype(float)
    else:
        # Default binary initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    # Ensure exactly n_points/2 ones
    count_ones = int(n_points / 2)
    current_ones = np.sum(h)
    if current_ones > count_ones:
        # Turn some ones to zeros
        idx_ones = np.where(h == 1)[0]
        flip_indices = np.random.choice(idx_ones, int(current_ones - count_ones), replace=False)
        h[flip_indices] = 0
    elif current_ones < count_ones:
        # Turn some zeros to ones
        idx_zeros = np.where(h == 0)[0]
        flip_indices = np.random.choice(idx_zeros, int(count_ones - current_ones), replace=False)
        h[flip_indices] = 1
    
    # Project to ensure exactly count_ones ones
    ones_indices = np.where(h == 1)[0]
    if len(ones_indices) > count_ones:
        extra_indices = np.random.choice(ones_indices, len(ones_indices) - count_ones, replace=False)
        h[extra_indices] = 0
    elif len(ones_indices) < count_ones:
        zero_indices = np.where(h == 0)[0]
        extra_indices = np.random.choice(zero_indices, count_ones - len(ones_indices), replace=False)
        h[extra_indices] = 1
    
    # Ensure that h is boolean with exactly 50 ones
    h = (h >= 0.5).astype(float)
    h_count = int(np.sum(h))
    if h_count != count_ones:
        to_change = count_ones - h_count
        if to_change > 0:
            zero_indices = np.where(h == 0)[0]
            change_indices = np.random.choice(zero_indices, to_change, replace=False)
            h[change_indices] = 1
        else:
            one_indices = np.where(h == 1)[0]
            change_indices = np.random.choice(one_indices, -to_change, replace=False)
            h[change_indices] = 0
            
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Perform simulated annealing
    T = 1.0
    cooling_factor = 0.999
    iterations = 0
    stagnation_counter = 0
    max_stagnation = 1000
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1 and stagnation_counter < max_stagnation:
        # Save current state
        current_h = h.copy()
        current_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        # Pick a random index to flip
        i = np.random.randint(0, n_points)
        h[i] = 1 - h[i]  # Flip: 0->1 or 1->0

        # Ensure exactly count_ones ones
        h_count = int(np.sum(h))
        if h_count > count_ones:
            # Need to decrement by flipping some 1s to 0s
            ones_indices = np.where(h == 1)[0]
            dec_indices = np.random.choice(ones_indices, h_count - count_ones, replace=False)
            h[dec_indices] = 0
        elif h_count < count_ones:
            # Need to increment by flipping some 0s to 1s
            zeros_indices = np.where(h == 0)[0]
            inc_indices = np.random.choice(zeros_indices, count_ones - h_count, replace=False)
            h[inc_indices] = 1

        if h_count != count_ones:
            # Force correct count
            ones_indices = np.where(h == 1)[0]
            if len(ones_indices) > count_ones:
                extra = len(ones_indices) - count_ones
                flip_indices = np.random.choice(ones_indices, extra, replace=False)
                h[flip_indices] = 0
            else:
                zero_indices = np.where(h == 0)[0]
                extra = count_ones - len(ones_indices)
                flip_indices = np.random.choice(zero_indices, extra, replace=False)
                h[flip_indices] = 1

        # Compute C5 after move
        new_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx

        # Accept with some probability
        if new_c5 <= current_c5:
            # Accept always
            pass
        else:
            # Accept with probability based on temperature
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                # Accept the move
                pass
            else:
                # Reject the move
                h = current_h.copy()

        # Decrease temperature
        T *= cooling_factor
        iterations += 1

        # Periodic perturbation
        if iterations % 1000 == 0 and time.time() - start_time < budget_s - 2:
            # Reset some bits randomly
            num_reset = int(n_points * 0.05)
            reset_indices = np.random.choice(n_points, num_reset, replace=False)
            for idx in reset_indices:
                h[idx] = np.random.randint(0, 2)
            # Ensure correct count
            h_count = int(np.sum(h))
            if h_count > count_ones:
                ones_indices = np.where(h == 1)[0]
                dec_indices = np.random.choice(ones_indices, h_count - count_ones, replace=False)
                h[dec_indices] = 0
            elif h_count < count_ones:
                zeros_indices = np.where(h == 0)[0]
                inc_indices = np.random.choice(zeros_indices, count_ones - h_count, replace=False)
                h[inc_indices] = 1

    # Final check
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
