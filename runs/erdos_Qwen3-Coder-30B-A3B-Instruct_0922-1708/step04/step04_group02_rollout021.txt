import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h using the provided parent construction or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Ensure sum constraint is satisfied
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    temperature = 1.0
    cooling_rate = 0.99
    min_temperature = 1e-6
    
    iteration = 0
    max_iterations = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate a candidate solution
        h_candidate = h.copy()
        
        # Select 10% of points to perturb
        num_perturb = max(1, n_points // 10)
        perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
        
        # Perturb selected points
        for i in perturb_indices:
            # Random perturbation within [0,1]
            perturbation = np.random.uniform(-0.1, 0.1)
            h_candidate[i] += perturbation
        
        # Ensure values are within [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Adjust to maintain sum constraint
        current_sum = np.sum(h_candidate)
        target_sum = n_points / 2
        sum_diff = target_sum - current_sum
        
        if abs(sum_diff) > 1e-8:
            # Distribute the difference evenly among all points
            adjustment = sum_diff / n_points
            h_candidate += adjustment
            # Ensure bounds
            h_candidate = np.clip(h_candidate, 0, 1)
            
        # Normalize again to ensure sum constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute new C5
        correlation_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        new_c5 = np.max(correlation_candidate) * dx
        
        # Accept or reject the candidate
        delta = new_c5 - current_c5
        
        if delta < 0 or np.exp(-delta / temperature) > np.random.random():
            h = h_candidate.copy()
        
        # Update temperature
        if iteration % 100 == 0:
            temperature = max(min_temperature, temperature * cooling_rate)
    
    # Final refinement around the best solution
    h_final = best_h.copy()
    best_c5_final = best_c5
    
    # Local search: try to improve by small changes
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_test = h_final.copy()
        # Perturb one random point
        i = np.random.randint(0, n_points)
        perturbation = np.random.uniform(-0.05, 0.05)
        h_test[i] += perturbation
        h_test = np.clip(h_test, 0, 1)
        
        # Adjust to maintain sum constraint
        current_sum = np.sum(h_test)
        target_sum = n_points / 2
        sum_diff = target_sum - current_sum
        
        if abs(sum_diff) > 1e-8:
            adjustment = sum_diff / n_points
            h_test += adjustment
            h_test = np.clip(h_test, 0, 1)
            
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        correlation_test = np.correlate(h_test, 1 - h_test, mode='full')
        test_c5 = np.max(correlation_test) * dx
        
        if test_c5 < best_c5_final:
            best_c5_final = test_c5
            h_final = h_test.copy()
    
    return (h_final.tolist(), best_c5_final, n_points)
