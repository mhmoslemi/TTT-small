import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate multiple initial candidates
    candidates = [h_init.copy()]
    for _ in range(19):
        h_perturbed = h_init + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        candidates.append(h_perturbed)
    
    # Store best solution
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Gradient-based local search
    max_iters = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iters:
        iter_count += 1
        
        # Evaluate all candidates
        for h in candidates:
            # Compute overlap integral
            corr_full = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr_full) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sample a candidate for refinement
        idx = np.random.randint(len(candidates))
        h = candidates[idx].copy()
        
        # Gradient computation
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update h using gradient descent
        lr = 0.01 / (1.0 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update candidate
        candidates[idx] = h_new
        
        # Add new candidate periodically
        if iter_count % 100 == 0 and time.time() - start_time < budget_s - 2:
            h_new = h_init + np.random.normal(0, 0.02, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            candidates.append(h_new)
    
    # Final check
    if best_h is None:
        best_h = h_init
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
