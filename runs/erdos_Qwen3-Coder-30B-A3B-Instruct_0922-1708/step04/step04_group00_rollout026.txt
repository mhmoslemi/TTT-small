import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # PSO parameters
    n_particles = 30
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    vmax = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals() and np.random.rand() < 0.5:
            # Use parent for some particles
            particle_h = h.copy()
        else:
            particle_h = np.random.uniform(0, 1, n_points)
            particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
            particle_h = np.clip(particle_h, 0, 1)
        particles.append(particle_h)
        velocities.append(np.random.uniform(-vmax, vmax, n_points))
        pbest.append(particle_h.copy())
        pbest_c5.append(float('inf'))
    
    gbest = h.copy()
    gbest_c5 = float('inf')
    
    # PSO iterations
    max_iter = 300
    no_improvement = 0
    max_no_improvement = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Compute fitness
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particle.copy()
                pbest_c5[i] = c5
                
            # Update global best
            if c5 < gbest_c5:
                gbest = particle.copy()
                gbest_c5 = c5
                
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
                
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            vel = (inertia * vel + 
                   c1 * r1 * (pbest[i] - particle) + 
                   c2 * r2 * (gbest - particle))
            
            # Clip velocity
            vel = np.clip(vel, -vmax, vmax)
            
            # Update position
            particle += vel
            
            # Enforce constraints
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            velocities[i] = vel
            particles[i] = particle
            
        # Early stopping
        if gbest_c5 < best_c5:
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > max_no_improvement:
                break
    
    # Local refinement with Nelder-Mead on best solution
    h = best_h.copy()
    max_refine_iter = 1000
    restarts = 0
    max_restarts = 5
    
    while restarts <= max_restarts:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create simplex starting from current best
        simplex_points = []
        simplex_points.append(h.copy())
        
        # Add random perturbations
        for i in range(5):  # 6 points in simplex 
            pert = np.random.normal(0, 0.01, n_points)
            pert_h = h + pert
            pert_h = np.clip(pert_h, 0, 1)
            pert_h = pert_h * (n_points / 2) / np.sum(pert_h)
            simplex_points.append(pert_h.copy())
            
        # Use scipy minimize with Nelder-Mead
        try:
            # Define objective function that takes flattened h
            def objective_flattened(h_flat):
                h_vec = h_flat.copy()
                h_vec = np.clip(h_vec, 0, 1)
                h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
                correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
                return np.max(correlation) * dx
            
            # Flatten h for Nelder-Mead
            h_flattened = h.copy()
            res = minimize(objective_flattened, 
                          h_flattened, 
                          method='Nelder-Mead',
                          options={'maxiter': 500, 'disp': False})
            
            if res.success or res.nit > 0:
                h_new = res.x.copy()
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate new solution
                correlation = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(correlation) * dx
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    restarts = 0  # reset restart counter if improved
                else:
                    restarts += 1  # increase restarts if no improvement
            else:
                restarts += 1
                
        except Exception:
            restarts += 1
            
        if restarts <= max_restarts:
            # Restart by small random perturbations
            pert = np.random.normal(0, 0.01, n_points)
            h = best_h + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final constraint check and refinement
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Compute final C5
    correlation = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Ensure we return the best result found
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_final.copy()
        
    return (best_h.tolist(), best_c5, n_points)
