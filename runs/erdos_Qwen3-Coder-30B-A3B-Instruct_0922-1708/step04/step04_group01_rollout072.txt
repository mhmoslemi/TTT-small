import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Coarse grid search to find promising regions
    coarse_size = 20
    candidates = []
    
    # Generate random candidates
    np.random.seed(42)
    for _ in range(50):
        h_temp = np.random.rand(coarse_size)
        h_temp = h_temp * (coarse_size / 2) / np.sum(h_temp)
        # Upsample to full resolution
        x_coarse = np.linspace(0, 2, coarse_size)
        x_full = np.linspace(0, 2, n_points)
        h_upsampled = np.interp(x_full, x_coarse, h_temp)
        h_upsampled = np.clip(h_upsampled, 0, 1)
        h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
        candidates.append(h_upsampled)
    
    # Add the initial solution
    if 'initial_h_values' in globals():
        candidates.append(best_h)
    
    # Evaluate coarse candidates
    for i, candidate in enumerate(candidates):
        h_eval = candidate.copy()
        correlation = np.correlate(h_eval, 1 - h_eval, mode='full')
        c5 = np.max(correlation) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_eval.copy()
    
    # Fine-grained optimization
    fine_iter = 500
    for iter_num in range(fine_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient ascent with small steps
        eps = 1e-4
        grad = np.zeros_like(best_h)
        
        # Compute approximate gradient using finite differences
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
            
        # Update with adaptive step size
        lr = 0.01
        h_new = best_h - lr * grad
        
        # Re-project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    # Try a more refined CVXPY optimization
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Objective: minimize the maximum overlap using a different approach
        # We'll approach it by minimizing a smooth approximation of max
        temp_corr = cp.Variable(2 * n_points - 1)  # Full correlation length
        # Build a matrix representation of correlation (this is a non-smooth constraint)
        objective = cp.Minimize(cp.max(temp_corr))
        prob = cp.Problem(objective, constraints)  # Only simple constraints
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass

    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
