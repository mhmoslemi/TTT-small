import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define optimization variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2,
        C >= 0
    ]
    
    # Compute overlaps for all shifts
    for k in range(1, n_points):
        # Forward shift
        if k < n_points:
            overlap = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
            constraints.append(overlap <= C)
        # Backward shift
        if k < n_points:
            overlap = cp.sum(cp.multiply(h_var[k:], 1 - h_var[:n_points-k]))
            constraints.append(overlap <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers = ['ECOS', 'SCS']  # Faster ones first
    
    for solver in solvers:
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            prob.solve(solver=solver, max_iters=1000, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_sol = h_var.value
                c5_val = C.value
                
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_sol.copy()
                    
            if prob.status == 'optimal':
                break
        except Exception:
            continue
    
    # If no solver worked, use gradient ascent with refinement
    if best_c5 == float('inf'):
        # Start with the parent solution or uniform
        h = best_h.copy()
        lr = 0.01
        eps = 1e-4
        max_iter = 5000
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute overlap
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Gradient computation via finite difference
            grad = np.zeros_like(h)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - c5) / eps
            
            # Update using gradient descent
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Check improvement
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
            h = h_new
            
            if np.abs(c5 - c5_new) < 1e-6:
                break
    
    # Final validation
    final_h = best_h.copy()
    final_corr = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (final_h.tolist(), float(final_c5), n_points)
