import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    h = (h_init >= 0.5).astype(int)
    one_count = np.sum(h)
    target_ones = n_points // 2
    
    # Adjust to ensure exactly target_ones ones
    if one_count > target_ones:
        zero_indices = np.where(h == 0)[0]
        np.random.shuffle(zero_indices)
        for idx in zero_indices[:one_count - target_ones]:
            h[idx] = 1
    elif one_count < target_ones:
        one_indices = np.where(h == 1)[0]
        np.random.shuffle(one_indices)
        for idx in one_indices[:target_ones - one_count]:
            h[idx] = 0
    
    # Precompute all possible correlations for all shifts
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    
    # Simulated annealing parameters
    T = 1.0
    decay = 0.999
    perturbation_freq = 1000
    stagnation_limit = 1000
    stagnation_count = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 2:
        iteration += 1
        # Make a random change by flipping one bit
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]
        
        # Ensure the constraint of exactly half ones
        ones_count = np.sum(h_new)
        if ones_count > target_ones:
            # Find all 1s and randomly set one to 0
            one_indices = np.where(h_new == 1)[0]
            idx_to_flip = np.random.choice(one_indices)
            h_new[idx_to_flip] = 0
        elif ones_count < target_ones:
            # Find all 0s and randomly set one to 1
            zero_indices = np.where(h_new == 0)[0]
            idx_to_flip = np.random.choice(zero_indices)
            h_new[idx_to_flip] = 1
        
        # Compute new C5
        new_c5 = compute_c5(h_new)
        delta = new_c5 - current_c5
        
        # Accept or reject based on simulated annealing
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h = h_new
            current_c5 = new_c5
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Apply temperature decay
        if iteration % 10 == 0:
            T *= decay
        
        # Periodic random reset
        if iteration % perturbation_freq == 0:
            # Randomly reset some bits (5%)
            num_reset = max(1, n_points // 20)
            reset_indices = np.random.choice(n_points, num_reset, replace=False)
            for idx in reset_indices:
                h[idx] = np.random.randint(0, 2)
            
            # Fix the constraint
            ones_count = np.sum(h)
            if ones_count > target_ones:
                one_indices = np.where(h == 1)[0]
                np.random.shuffle(one_indices)
                for idx in one_indices[:ones_count - target_ones]:
                    h[idx] = 0
            elif ones_count < target_ones:
                zero_indices = np.where(h == 0)[0]
                np.random.shuffle(zero_indices)
                for idx in zero_indices[:target_ones - ones_count]:
                    h[idx] = 1
        
        # Early termination if stagnant
        if stagnation_count > stagnation_limit:
            break
    
    return (best_h.tolist(), best_c5, n_points)
