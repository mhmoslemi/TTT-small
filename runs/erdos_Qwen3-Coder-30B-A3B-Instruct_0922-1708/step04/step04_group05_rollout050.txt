import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    max_iterations = 200
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5

    # Normalize to integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    num_particles = 20
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    max_inertia_decrease = 50
    
    # Initialize swarm
    particles = np.random.uniform(0, 1, (num_particles, n_points)) * 0.2 + h_init * 0.8
    velocities = np.random.uniform(-0.01, 0.01, (num_particles, n_points))
    personal_best = particles.copy()
    personal_best_scores = np.full(num_particles, np.inf)
    global_best_idx = 0
    global_best_score = np.inf
    last_improvement = 0
    
    start_time = time.time()
    
    # Swarm size adaptation
    def adapt_swarm_size(iteration):
        nonlocal num_particles
        if iteration > max_inertia_decrease and iteration % 50 == 0:
            num_particles = max(10, num_particles - 5)
    
    # Evaluate function
    def evaluate(h):
        h_clipped = np.clip(h, 0, 1)
        h_normalized = h_clipped * (n_points / 2) / np.sum(h_clipped)
        corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        return np.max(corr) * dx
    
    # PSO iteration
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 2:
            break
            
        adapt_swarm_size(iteration)
        w = w_start - (w_start - w_end) * iteration / max_iterations
        
        for i in range(num_particles):
            if time.time() - start_time > budget_s - 2:
                break
            # Clip and normalize particles
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate current particle
            current_score = evaluate(particles[i])
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best[i] = particles[i].copy()
            
            # Update global best
            if current_score < global_best_score:
                global_best_score = current_score
                global_best_idx = i
                last_improvement = iteration
                
            if iteration - last_improvement > max_inertia_decrease:
                # Add noise to particles
                noise = np.random.normal(0, 0.005, n_points)
                particles[i] += noise
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                
        # Update velocities and positions
        for i in range(num_particles):
            if time.time() - start_time > budget_s - 2:
                break
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            particles[i] += velocities[i]
            
    # Final best solution
    final_h = personal_best[global_best_idx]
    final_c5 = evaluate(final_h)
    
    # Refinement by gradient descent on top solutions
    top_solutions = []
    for i in range(min(5, num_particles)):
        candidate = personal_best[i].copy()
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        candidate_score = evaluate(candidate)
        top_solutions.append((candidate, candidate_score))
    
    top_solutions.sort(key=lambda x: x[1])
    best_refined_h = top_solutions[0][0]
    best_refined_score = top_solutions[0][1]
    
    # Iterate gradient-based refinement
    for particle, score in top_solutions[:2]:  # Try top 2 for gradient refinement
        if time.time() - start_time > budget_s - 2:
            break
        # Simplified gradient estimation
        step_size = 0.001
        for _ in range(100):
            if time.time() - start_time > budget_s - 2:
                break
            grad = np.zeros(n_points)
            eps = 1e-4
            for idx in range(n_points):
                h_plus = particle.copy()
                h_plus[idx] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[idx] = (c5_plus - score) / eps
            lr = step_size / (1 + np.linalg.norm(grad) * 1e-3)
            particle -= lr * grad
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            corr = np.correlate(particle, 1 - particle, mode='full')
            score = np.max(corr) * dx
            if score < best_refined_score:
                best_refined_score = score
                best_refined_h = particle.copy()
    
    # Final verification
    actual_corr = np.correlate(best_refined_h, 1 - best_refined_h, mode='full')
    final_c5 = np.max(actual_corr) * dx
    
    return (best_refined_h.tolist(), final_c5, n_points)
