import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5

    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = float('inf')

    # PSO parameters
    n_particles = 30
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    vmax = 0.1

    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []

    for _ in range(n_particles):
        if 'initial_h_values' in globals() and np.random.rand() < 0.5:
            # Use parent for some particles
            particle_h = h.copy()
        else:
            particle_h = np.random.uniform(0, 1, n_points)
            particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
            particle_h = np.clip(particle_h, 0, 1)
        particles.append(particle_h)
        velocities.append(np.random.uniform(-vmax, vmax, n_points))
        pbest.append(particle_h.copy())
        pbest_c5.append(float('inf'))

    gbest = h.copy()
    gbest_c5 = float('inf')
    
    # PSO iterations
    max_iter = 300
    no_improvement = 0
    max_no_improvement = 60

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Update each particle
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]

            # Compute fitness
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx

            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particle.copy()
                pbest_c5[i] = c5

            # Update global best
            if c5 < gbest_c5:
                gbest = particle.copy()
                gbest_c5 = c5

            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()

            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            vel = (inertia * vel + 
                   c1 * r1 * (pbest[i] - particle) + 
                   c2 * r2 * (gbest - particle))

            # Clip velocity
            vel = np.clip(vel, -vmax, vmax)

            # Update position
            particle += vel

            # Enforce constraints
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)

            velocities[i] = vel
            particles[i] = particle

        # Early stopping
        if gbest_c5 < best_c5:
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > max_no_improvement:
                break

    # Local refinement with Nelder-Mead
    def objective(h_vec):
        h = h_vec.reshape(-1)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5

    # Use best solution as starting point for Nelder-Mead
    x0 = best_h.copy()
    
    # Set bounds for Nelder-Mead
    bounds = [(0, 1) for _ in range(n_points)]
    # Add sum constraint as penalty to objective for inner optimizer
    
    # Normalize initial guess
    x0 = x0 * (n_points / 2) / np.sum(x0)
    x0 = np.clip(x0, 0, 1)
    
    # Nelder-Mead optimization with bounds
    result = minimize(objective, x0, method='Nelder-Mead', options={'maxiter': 1000, 'xatol': 1e-6, 'fatol': 1e-6})
    
    if result.success:
        refined_h = result.x.reshape(-1)
        refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
        refined_h = np.clip(refined_h, 0, 1)
        
        correlation = np.correlate(refined_h, 1 - refined_h, mode='full')
        refined_c5 = np.max(correlation) * dx
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h.copy()

    # Final verification 
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
