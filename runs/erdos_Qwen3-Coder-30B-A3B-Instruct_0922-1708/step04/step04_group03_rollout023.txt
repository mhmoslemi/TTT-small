import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint ∫h(x)dx = 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Iteration counter and early stopping
    iteration = 0
    max_no_improvement = 100
    no_improvement_count = 0
    
    # Learning rate schedule
    learning_rate = 0.1
    decay_every = 50
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute correlations for all lags
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Track best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Early stopping
        if no_improvement_count >= max_no_improvement:
            break
        
        # Compute gradient using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            # Increase h[i] slightly
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Decrease h[i] slightly
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Compute directional derivative
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update h using gradient descent
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adjust learning rate
        if iteration % decay_every == 0:
            learning_rate *= 0.99
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
