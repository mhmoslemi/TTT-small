import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Use a simple initial construction
        h = np.ones(n_points) * 0.5
    
    # Keep track of best solution so far
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_c5(h_vec):
        # Compute overlap integral
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Grid-based exploration for initial candidates
    def explore_grid():
        nonlocal h, best_h, best_c5
        grid_size = 20
        candidates = []
        n_samples = 10  # Pick top candidates
        
        for _ in range(50):  # Generate diverse samples
            # Random sample with sum constraint
            trial_h = np.random.rand(grid_size)
            # Scale to target sum
            trial_h = trial_h * (grid_size / 2) / np.sum(trial_h)
            
            # Interpolate to full resolution
            x_old = np.linspace(0, 2, grid_size)
            x_new = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_new, x_old, trial_h)
            
            # Compute overlap
            c5 = compute_c5(h_interp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_interp.copy()
            
            candidates.append((h_interp, c5))
        
        # Sort and select top candidates
        candidates.sort(key=lambda x: x[1])
        return candidates[:n_samples]
    
    # Try grid search first
    try:
        grid_candidates = explore_grid()
        for candidate_h, _ in grid_candidates:
            h_init = candidate_h.copy()
            # Local refinement
            for _ in range(200):
                # Simple gradient descent
                eps = 1e-5
                grad = np.zeros_like(h_init)
                base_c5 = compute_c5(h_init)
                
                for i in range(n_points):
                    h_plus = h_init.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c5_plus = compute_c5(h_plus)
                    grad[i] = (c5_plus - base_c5) / eps
                
                h_new = h_init - 0.01 * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_init = h_new
                
                c5_new = compute_c5(h_init)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_init.copy()
    except:
        pass
        
    # Final check with small CVX optimization on best solution
    try:
        # Slightly refine with CVXPY to see if we can improve
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        # For the constraint, we compute correlation for all lags
        # For efficiency, we use an efficient formulation to get max overlap
        # Approximate data by computing correlation manually for a few lags and
        # using max over all lags with constraints
        
        # For now let's just do a simple local update
        h_final = best_h.copy()
        # Do one more round of local refinement if time remains
        if time.time() - start_time < budget_s - 5:
            for _ in range(500):
                eps = 1e-5
                grad = np.zeros(n_points)
                base_c5 = compute_c5(h_final)
                for i in range(n_points):
                    h_plus = h_final.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c5_plus = compute_c5(h_plus)
                    grad[i] = (c5_plus - base_c5) / eps
                
                h_new = h_final - 0.005 * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_final = h_new
                
                c5 = compute_c5(h_final)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_final.copy()
    except:
        pass
        
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
