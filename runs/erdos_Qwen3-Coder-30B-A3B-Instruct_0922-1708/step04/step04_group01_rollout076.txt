import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Particle Swarm Optimization parameters
    n_particles = 30
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            if len(particle) != n_points:
                h_orig = np.array(initial_h_values)
                x_orig = np.linspace(0, 2, len(h_orig))
                x_target = np.linspace(0, 2, n_points)
                particle = np.interp(x_target, x_orig, h_orig)
            particle = particle / np.sum(particle) * (n_points / 2)
        else:
            particle = np.random.rand(n_points)
            particle = particle * (n_points / 2) / np.sum(particle)
        
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(particle.copy())
        pbest_c5.append(float('inf'))
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Compute current particle's overlap
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = particles[i].copy()
        
        # Update particles
        gbest_idx = np.argmin(pbest_c5)
        gbest = pbest[gbest_idx]
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize sum
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Final check after PSO
    if best_c5 == float('inf'):
        # If no improvement in PSO, do a local search from starting point
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Try CVX refinement
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        # This is a complex constraint to handle maximum correlation 
        # Since CVXPY cannot directly minimize max, we try an alternative:
        # We'll solve by formulating it as a minimax optimization problem
        
        # Instead, try to solve using local descent from best PSO solution
        # This is a heuristic since direct CVX formulation is complex
        h_refine = best_h.copy()
        
        # Flatten the problem for local optimization
        x0 = h_refine.copy()
        
        def obj_func(x):
            # Normalize x
            x = x * (n_points / 2) / np.sum(x)
            # Compute overlap
            correlation = np.correlate(x, 1 - x, mode='full')
            c5 = np.max(correlation) * dx
            return c5
        
        # Simple local search
        for _ in range(50):  # Limit local search steps
            if time.time() - start_time > budget_s - 1:
                break
            grad = np.zeros_like(x0)
            eps = 1e-4
            for i in range(n_points):
                x_plus = x0.copy()
                x_plus[i] += eps
                x_plus = np.clip(x_plus, 0, 1)
                x_plus = x_plus * (n_points / 2) / np.sum(x_plus)
                c5_plus = obj_func(x_plus)
                grad[i] = (c5_plus - obj_func(x0)) / eps
            
            # Descent step
            lr = 0.01
            x0 = x0 - lr * grad
            x0 = np.clip(x0, 0, 1)
            x0 = x0 * (n_points / 2) / np.sum(x0)
            
            # Update best if improved
            current_c5 = obj_func(x0)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = x0.copy()
    except:
        pass

    # Final validation 
    validation_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(validation_corr) * dx
    
    # Ensure the final h satisfies constraints strictly
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
