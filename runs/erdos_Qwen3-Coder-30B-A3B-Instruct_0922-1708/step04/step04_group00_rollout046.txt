import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def evaluate_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # Gradient-based optimization
    def objective(h_vals):
        return evaluate_c5(h_vals)
    
    def constraint_fn(h_vals):
        return np.sum(h_vals) * dx - 1.0
    
    # Use scipy.optimize.minimize with SLSQP
    cons = {'type': 'eq', 'fun': constraint_fn}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple restarts with different initializations
    for restart in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate a different initialization
        if 'initial_h_values' in globals() and restart > 0:
            # Start from a perturbed version of the parent
            h0 = h.copy() + np.random.normal(0, 0.05, n_points)
            h0 = np.clip(h0, 0, 1)
            h0 = h0 * (n_points / 2) / np.sum(h0)
        else:
            # Random initialization
            h0 = np.random.uniform(0.1, 0.9, n_points)
            h0 = h0 * (n_points / 2) / np.sum(h0)
            
        try:
            # Optimize using SLSQP
            res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9})
            
            if res.success:
                c5_res = evaluate_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = res.x.copy()
        except:
            pass
            
        # Also try some local search with random perturbations
        h_test = h0.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            try:
                # Small random perturbations
                pert = np.random.normal(0, 0.01, n_points)
                h_new = h_test + pert
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_test = h_new.copy()
                else:
                    h_test = h_new.copy()
            except:
                break
    
    # Additional refinement using gradient descent
    h_refine = best_h.copy()
    lr = 0.01
    max_iter = 5000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_current = evaluate_c5(h_refine)
        
        # Compute approximate gradient via finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for j in range(n_points):
            h_plus = h_refine.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_c5(h_plus)
            grad[j] = (c5_plus - c5_current) / eps
            
        # Update
        h_new = h_refine - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h_refine = h_new.copy()
        lr = max(1e-6, lr * 0.9999)
    
    return (best_h.tolist(), best_c5, n_points)
