```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time
import cvxpy as cp

def evaluate_h(h, n_points):
    """Evaluate the overlap integral for given h."""
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def project_h(h, n_points):
    """Project h to maintain sum constraint."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    h_best = None
    c5_best = float('inf')
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            # Interpolate to target resolution
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
        h_best = h_init.copy()
        c5_best = evaluate_h(h_init, n_points)
    else:
        # Start with a block function
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
        h_init = project_h(h_init, n_points)
        h_best = h_init.copy()
        c5_best = evaluate_h(h_init, n_points)
    
    # Phase 1: Grid search with coarse resolution
    coarse_points = 20
    if coarse_points <= n_points:
        # Generate coarse grid points
        grid_points = []
        if coarse_points < 10:
            # Small grid for fast exploration
            step = 1.0 / (coarse_points - 1) if coarse_points > 1 else 1.0
            for i in range(coarse_points):
                h = np.full(coarse_points, i * step)
                h = project_h(h, coarse_points)
                grid_points.append(h)
        else:
            # Use a random grid in the feasible set
            for _ in range(200):
                h = np.random.rand(coarse_points)
                h = project_h(h, coarse_points)
                grid_points.append(h)
        
        # Evaluate coarse grid points and keep best
        for h_coarse in grid_points:
            # Upsample
            h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                                    np.linspace(0, 2, coarse_points), h_coarse)
            h_upsampled = project_h(h_upsampled, n_points)
            c5 = evaluate_h(h_upsampled, n_points)
            if c5 < c5_best:
                c5_best = c5
                h_best = h_upsampled.copy()
                
            if time.time() - start_time > budget_s - 10:
                break
    
    # Phase 2: Local refinement with high resolution
    if time.time() - start_time < budget_s - 20:
        # Refine the best found solution
        h_refine = h_best.copy()
        max_iter = 500
        
        for it in range(max_iter):
            # Compute the gradient numerically
            eps = 1e-4
            grad = np.zeros_like(h_refine)
            c5_base = evaluate_h(h_refine, n_points)
            
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, n_points)
                grad[i] = (c5_plus - c5_base) / eps
            
            # Update with gradient descent
            lr = 0.01
            h_new = h_refine - lr * grad
            h_new = project_h(h_new, n_points)
            
            # Check if we improved
            c5_new = evaluate_h(h_new, n_points)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
            
            h_refine = h_new.copy()
            
            if time.time() - start_time > budget_s - 10:
                break
    
    # Final CVX check if time allows
    if time.time() - start_time < budget_s - 50:
        try:
            # Define optimization using CVXPY to verify
            h_vars = cp.Variable(n_points)
            constraints = [
                h_vars >= 0,
                h_vars <= 1,
                cp.sum(h_vars) == n_points / 2
            ]
            
            # For small number of points, use correlation-based formulation
            # We attempt to find an upper bound that's actually smaller 
            corr_matrix = np.zeros((n_points, n_points))
            for i in range(n_points):
                for j in range(n_points):
                    k = (j - i) % n_points
                    if k < n_points:
                        corr_matrix[i, j] = 1.0 if k == 0 else 0.0  # Approximate
            
            # Directly compute using a sample fine approach instead of conv
            # Try aggressive local search with projected gradient using scipy
            
            def objective(h):
                return evaluate_h(h, n_points)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            h_final = minimize(objective, h_best, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 200, 'ftol': 1e-8})
            
            if h_final.success:
                h_cvx = project_h(h_final.x, n_points)
                c5_cvx = evaluate_h(h_cvx, n_points)
                if c5_cvx < c5_best:
                    c5_best = c5_cvx
                    h_best = h_cvx.copy()
                    
        except Exception:
            pass
    
    return (h_best.tolist(), c5_best, n_points)
```
