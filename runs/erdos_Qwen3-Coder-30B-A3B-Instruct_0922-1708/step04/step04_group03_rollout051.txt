import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # First phase: Convex relaxation
    # We'll solve a relaxed version to get a good starting point
    
    # Build the constraint matrix for the integral constraint
    # We want sum(h) * dx = 1, so sum(h) = n_points / 2
    integral_constraint = np.ones(n_points)
    
    # Since we are trying to minimize the overlap over all possible shifts,
    # for now we'll constrain using a proxy objective: minimize the sum of overlaps
    # as a relaxation. This often helps in finding good starting points.
    
    # Create variable for projection
    h_var = cp.Variable(n_points)
    # Constraint: sum h = n_points/2 (average value is 0.5)
    constraints = [cp.sum(h_var) == n_points / 2,
                   h_var >= 0,
                   h_var <= 1]
    
    # Objective: approximate sum of overlaps
    # Compute as sum of h[i]*(1-h[i]) for all i, which is roughly the sum of overlaps with k=0
    # This is for relaxation only, not final objective
    objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - h_var)))
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_relaxed = h_var.value
            if np.all(h_relaxed >= 0) and np.all(h_relaxed <= 1):
                # Rescale to maintain integral constraint
                h = h_relaxed * (n_points / 2) / np.sum(h_relaxed)
    except:
        pass  # fall back to current h if convex relaxation fails
    
    # Second phase: Local refinement with projected gradient descent
    max_no_improvement = 50
    no_improvement_count = 0
    iteration = 0
    max_iterations = 2000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Compute overlap at current point
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlation) * dx
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        h_current = h.copy()
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5_current) / eps
        
        # Projected gradient descent
        lr = 0.01
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
