import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation
    # Create a simpler convex problem to get a good starting point
    def solve_convex_relaxation():
        h_vars = cp.Variable(n_points)
        
        # The convex relaxation tries to minimize sum of overlaps
        # This is a simplification to get a better starting point
        
        # Objective: minimize sum of all correlations
        # This is an approximation to find a good feasible solution
        objective = cp.Minimize(
            cp.sum([h_vars[i] * (1 - h_vars[(i+j) % n_points]) 
                   for i in range(n_points) for j in range(n_points)])
        )
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Solve the convex problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            return np.maximum(0, np.minimum(1, h_vars.value))
        except:
            return h.copy()  # fallback to current h if convex fails
            
    # Try convex relaxation for initial solution
    try:
        h_init = solve_convex_relaxation()
        if h_init is not None:
            # Make sure solution is feasible
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            h = h_init
    except:
        pass
    
    # Phase 2: Projected Gradient Descent with adaptive learning rate
    def compute_correlation(h_vec):
        # Compute all correlations (full mode)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
        
    def compute_gradient(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        base_val = compute_correlation(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            val_plus = compute_correlation(h_plus)
            grad[i] = (val_plus - base_val) / eps
        return grad
        
    # Main optimization loop
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 50
    lr = 0.05  # Initial learning rate
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current correlation and gradient
        current_c5 = compute_correlation(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update h with projected gradient descent
        h_new = h - lr * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        new_c5 = compute_correlation(h_new)
        if new_c5 < current_c5:
            h = h_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate decay
        if iteration % 100 == 0:
            lr *= 0.9
            
        # Early stopping condition
        if no_improvement_count >= max_no_improvement:
            break
            
    # Final check
    final_c5 = compute_correlation(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (best_h.tolist(), best_c5, n_points)
