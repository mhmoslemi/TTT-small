import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Try a few different solvers
    solvers_to_try = ['ECOS', 'SCS']
    
    for solver in solvers_to_try:
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            # Model using CVXPY
            h_var = cp.Variable(n_points)
            C = cp.Variable()
            
            # Constraints
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2,
                C >= 0
            ]
            
            # Compute overlaps for all shifts
            for k in range(1, n_points):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # For positive shift
                if k < n_points:
                    # Overlap = sum h[i] * (1 - h[i+k])
                    overlap = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
                    constraints.append(overlap <= C)
                
                # For negative shift
                if k < n_points:
                    # Overlap = sum h[i+k] * (1 - h[i])
                    overlap = cp.sum(cp.multiply(h_var[k:], 1 - h_var[:n_points-k]))
                    constraints.append(overlap <= C)
            
            # Objective
            objective = cp.Minimize(C)
            prob = cp.Problem(objective, constraints)
            
            # Set solver parameters
            prob.solve(solver=solver, abstol=1e-6, reltol=1e-6, time_limit=budget_s - (time.time() - start_time) - 1)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h_var.value
                c5_solution = C.value
                
                if c5_solution < best_c5:
                    best_c5 = c5_solution
                    best_h = np.copy(h_solution)
                    
        except Exception:
            continue
    
    # Final refinement: gradient-based
    h = best_h.copy()
    
    for i in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Use finite differences for gradient estimation
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
        
        # Update with adaptive learning rate
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update if better
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Convergence check
        if np.abs(c5 - c5_new) < 1e-10:
            break
    
    return (best_h.tolist(), best_c5, n_points)
