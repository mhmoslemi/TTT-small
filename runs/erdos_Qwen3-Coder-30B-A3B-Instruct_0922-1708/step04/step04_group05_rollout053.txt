import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    max_iter = 200
    n_particles = 20
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    inertia_decay = (w_start - w_end) / max_iter
    early_stop_patience = 50
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO initialization
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    g_best = h_init.copy()
    g_best_val = float('inf')
    
    for i in range(n_particles):
        h = h_init + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.normal(0, 0.01, n_points))
        val = np.max(np.correlate(h, 1-h, mode='full')) * dx
        p_best.append(h.copy())
        p_best_vals.append(val)
        if val < g_best_val:
            g_best = h.copy()
            g_best_val = val
    
    # PSO algorithm
    start_time = time.time()
    last_improvement = 0
    w = w_start
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (p_best[i] - particles[i]) +
                             c2 * r2 * (g_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            val = np.max(np.correlate(particles[i], 1-particles[i], mode='full')) * dx
            
            # Update personal best
            if val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = val
                
                if val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = val
                    last_improvement = iter_num
        
        # Adaptive inertia weight
        w = max(w_end, w - inertia_decay)
        
        # Adapt swarm size if no improvement
        if iter_num - last_improvement > early_stop_patience:
            # Remove worst particle
            worst_idx = np.argmax(p_best_vals)
            particles.pop(worst_idx)
            velocities.pop(worst_idx)
            p_best.pop(worst_idx)
            p_best_vals.pop(worst_idx)
            
            # Add new random particle
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.normal(0, 0.01, n_points))
            p_best.append(h.copy())
            p_best_vals.append(np.max(np.correlate(h, 1-h, mode='full')) * dx)
            
            last_improvement = iter_num
    
    # Final refinement with line search for top solutions
    top_candidates = [p_best[i] for i in np.argsort(p_best_vals)[:5]]
    best_h = g_best.copy()
    best_c5 = g_best_val
    
    for candidate in top_candidates:
        # Simple gradient-based refinement
        h = candidate.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 5:
                break
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            lr = 1e-3 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
