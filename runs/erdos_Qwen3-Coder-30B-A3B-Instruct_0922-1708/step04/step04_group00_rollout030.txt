import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective_and_gradient(h_vec):
        # Project h to [0,1] and normalize
        h_proj = np.clip(h_vec, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        
        # Compute correlation
        correlation = np.correlate(h_proj, 1 - h_proj, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences
        eps = 1e-4
        grad = np.zeros_like(h_proj)
        for i in range(n_points):
            h_plus = h_proj.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        return c5, grad
    
    # Trust-region optimization
    max_iter = 500
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Optimize with trust region method
        res = minimize(
            fun=lambda x: objective_and_gradient(x)[0],
            x0=h,
            method='TRF',
            jac=lambda x: objective_and_gradient(x)[1],
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 10, 'initial_tr_radius': 0.1}
        )
        
        if res.success:
            h = res.x
            # Project to feasible region
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Fallback: random perturbation
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            
        # Check current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        iteration += 1
            
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
