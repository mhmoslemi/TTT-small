import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Use a hybrid approach: Genetic Algorithm + Local Search
    
    # GA parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.05
    elite_size = 5
    
    # Initialize population
    population = []
    fitness = []
    
    for _ in range(population_size):
        if 'initial_h_values' in globals() and np.random.rand() < 0.5:
            individual = h.copy()
        else:
            individual = np.random.uniform(0, 1, n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
            individual = np.clip(individual, 0, 1)
        population.append(individual)
        
    # Evaluate initial population
    for i, individual in enumerate(population):
        correlation = np.correlate(individual, 1 - individual, mode='full')
        c5 = np.max(correlation) * dx
        fitness.append(c5)
    
    # Best solution
    min_fit_idx = np.argmin(fitness)
    if fitness[min_fit_idx] < best_c5:
        best_c5 = fitness[min_fit_idx]
        best_h = population[min_fit_idx].copy()
    
    # GA iterations
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        elite = [population[i] for i in sorted_idx[:elite_size]]
        fitness = [fitness[i] for i in sorted_idx]
        population = [population[i] for i in sorted_idx]
        
        # Create new population
        new_population = elite[:]
        
        # Tournament selection
        def tournament_selection():
            k = 3
            idxs = np.random.choice(len(population), k)
            best = idxs[0]
            for i in idxs[1:]:
                if fitness[i] < fitness[best]:
                    best = i
            return population[best].copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] = np.random.uniform(0, 1)
                if np.random.rand() < mutation_rate:
                    child2[i] = np.random.uniform(0, 1)
            
            # Normalize and clip
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child1 = np.clip(child1, 0, 1)
            
            child2 = child2 * (n_points / 2) / np.sum(child2)
            child2 = np.clip(child2, 0, 1)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        # Update population
        population = new_population[:]
        
        # Evaluate new population
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = individual.copy()
    
    # Local refinement with gradient descent on best solution
    h = best_h.copy()
    max_refine_iter = 1000
    lr = 0.01
    
    for refine_iter in range(max_refine_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update learning rate
        lr = max(1e-7, lr * 0.999)
        
        h = h_new
        
        # Update best if improved
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Early stopping based on improvement rate  
        if refine_iter > 100 and np.abs(c5 - c5_new) < 1e-8:
            break
            
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
