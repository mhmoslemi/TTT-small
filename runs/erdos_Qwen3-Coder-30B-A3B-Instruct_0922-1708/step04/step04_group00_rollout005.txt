import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective_and_gradient(h):
        # Normalize h to sum to n_points/2
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute the overlap integral using correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        return c5, grad
    
    # Trust Region Reflective optimization
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Define bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize with trust region method
        try:
            result = minimize(objective_and_gradient, h, 
                            method='TRF', 
                            jac=True,
                            bounds=bounds,
                            options={'maxiter': 10, 'gtol': 1e-6})
            h = result.x
        except:
            # Fallback if TRF fails
            try:
                result = minimize(lambda x: objective_and_gradient(x)[0], h,
                                method='L-BFGS-B',
                                bounds=bounds,
                                options={'maxiter': 20})
                h = result.x
            except:
                pass
                
        # Ensure constraints are satisfied
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early stopping if no improvement
        if iteration > 10 and abs(c5 - best_c5) < 1e-8:
            break
    
    # Final refinement with local optimization
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient again
        c5, grad = objective_and_gradient(best_h)
        
        # Update with small step
        step_size = 0.001
        h_new = best_h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate and accept if better
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
