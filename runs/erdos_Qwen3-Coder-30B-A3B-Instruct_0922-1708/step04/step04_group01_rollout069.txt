import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Guided random search
    scale = 0.1
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        perturbation = np.random.normal(0, scale, n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Compute overlap
        correlation = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()
            
        # Decrease scale
        scale *= 0.99
            
    # Gradient descent refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute finite difference gradient
        grad = np.zeros(n_points)
        eps = 1e-4
        c5_ref = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        
        for i in range(n_points):
            h_eps = h.copy()
            h_eps[i] += eps
            h_eps = np.clip(h_eps, 0, 1)
            h_eps = h_eps * (n_points / 2) / np.sum(h_eps)
            
            corr = np.correlate(h_eps, 1 - h_eps, mode='full')
            c5_eps = np.max(corr) * dx
            
            grad[i] = (c5_eps - c5_ref) / eps
            
        # Update with gradient
        lr = 0.01
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlap
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new
    
    return (best_h.tolist(), best_c5, n_points)
