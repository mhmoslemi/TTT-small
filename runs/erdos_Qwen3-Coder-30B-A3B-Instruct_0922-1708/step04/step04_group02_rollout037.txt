import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize particles
    swarm_size = 100
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
    
    # Add 30 more random initial particles
    for _ in range(30):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
    
    # Initialize velocities
    for i in range(len(particles)):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize personal bests
    pbest = [particles[i].copy() for i in range(len(particles))]
    pbest_values = [evaluate_h(particles[i], dx) for i in range(len(particles))]
    
    # Find initial global best
    best_idx = np.argmin(pbest_values)
    gbest = pbest[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    # PSO parameters
    omega = 0.7
    phi_p = 2.0
    phi_g = 2.0
    max_iter = 900
    local_search_freq = 50
    restart_freq = 200
    
    # Main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update particle positions
        for i in range(len(particles)):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (omega * velocities[i] + 
                            phi_p * r1 * (pbest[i] - particles[i]) + 
                            phi_g * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair and normalize
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate fitness
            current_value = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
            
            # Update global best
            if current_value < gbest_value:
                gbest = particles[i].copy()
                gbest_value = current_value
                
        # Local search every local_search_freq iterations
        if iter_count % local_search_freq == 0:
            # Apply gradient descent on top 10% of particles
            top_k = min(10, len(particles) // 10)
            indices = np.argsort(pbest_values)[:top_k]
            
            for idx in indices:
                h = pbest[idx].copy()
                current_c5 = pbest_values[idx]
                
                # Gradient descent with small steps
                for _ in range(10):
                    eps = 1e-5

                        grad[j] = (c5_plus - current_c5) / eps
                    
                    # Update using gradient
                    h = h - 0.001 * grad
                    h = repair_h(h, n_points)
                    corr = correlate(h, 1 - h, mode='full')
                    current_c5 = np.max(corr) * dx
                    
                    if current_c5 < pbest_values[idx]:

                        pbest_values[idx] = current_c5

                            gbest = h.copy()
                            gbest_value = current_c5
                    
        # Restart with new particles occasionally
        if iter_count % restart_freq == 0 and iter_count > 0:
            # Replace worst particles with new random ones
            worst_idx = np.argsort(pbest_values)[-10:]
            for idx in worst_idx:
                new_h = np.random.rand(n_points)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                particles[idx] = new_h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                pbest[idx] = new_h.copy()
                pbest_values[idx] = evaluate_h(new_h, dx)
                
    # Final refinement with gradient descent on the best solution
    best_h = gbest.copy()
    
    # More intensive gradient descent
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - gbest_value) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        gbest_value = np.max(corr) * dx
    
    return (best_h.tolist(), gbest_value, n_points)
