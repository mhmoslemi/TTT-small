import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return np.clip(h, 0, 1)

def trust_region_optimize(h0, n_points, dx, budget_s):
    start_time = time.time()
    h = h0.copy()
    c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5
    
    # Trust region parameters
    trust_radius = 0.1
    trust_radius_min = 1e-6
    decay_factor = 0.99
    max_iter = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - c5) / eps
        
        # Solve trust region subproblem
        # We'll use a simplified approach: limit gradient step by trust radius
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            step = -trust_radius * grad / grad_norm
        else:
            step = np.zeros_like(h)
            
        # Apply step and project
        h_new = h + step
        h_new = project_h(h_new, n_points)
        
        # Evaluate new point
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept step if improvement
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Reduce trust region
            trust_radius *= decay_factor
            if trust_radius < trust_radius_min:
                trust_radius = trust_radius_min
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = project_h(h, n_points)
    
    # Optimize using trust region method
    h_opt, c5 = trust_region_optimize(h, n_points, dx, budget_s)
    
    # Final evaluation
    final_c5 = evaluate_c5(h_opt, dx)
    
    return (h_opt.tolist(), final_c5, n_points)
