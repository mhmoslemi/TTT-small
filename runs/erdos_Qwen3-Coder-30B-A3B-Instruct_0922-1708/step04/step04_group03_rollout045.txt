import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters
    learning_rate = 0.1
    decay_rate = 0.99
    max_no_improvement = 100
    no_improvement_count = 0
    eps = 1e-4
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlaps for all lags
        corr = correlate(h, 1 - h, mode='full')
        max_overlap_val = np.max(corr) * dx
        max_overlap_k = np.argmax(corr)
        
        # Early stopping if no improvement
        if max_overlap_val < best_c5:
            best_c5 = max_overlap_val
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Compute directional derivatives for each point
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            # Increase h[i]
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Decrease h[i]
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Compute gradient component
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Update h using gradient descent
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Decrease learning rate with time
        if iteration % 50 == 0:
            learning_rate *= decay_rate
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
