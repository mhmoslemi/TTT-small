import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Trust region optimization
    trust_radius = 0.1
    max_iter = 500
    decay_rate = 0.99
    min_radius = 1e-6
    
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    def project_h(h_vals):
        # Clip and normalize
        h_proj = np.clip(h_vals, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj
    
    # Gradient estimation using finite differences
    def estimate_gradient(h_vals, eps=1e-4):
        grad = np.zeros_like(h_vals)
        base_c5 = compute_c5(h_vals)
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Trust region subproblem
    def solve_trust_region_subproblem(grad, trust_radius):
        # Simple trust region: find x that minimizes 0.5 * ||x||^2 subject to ||x|| <= trust_radius
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-12:
            return np.zeros_like(grad)
        
        # Step is in direction of negative gradient with bounded norm
        step = -trust_radius * grad / norm_grad
        return step
    
    # Core iteration
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current value
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Estimate gradient
        grad = estimate_gradient(h)
        
        # Solve trust region subproblem
        step = solve_trust_region_subproblem(grad, trust_radius)
        
        # Update h
        h_new = h + step
        h_new = project_h(h_new)
        
        # Accept step if it improves, otherwise decrease trust radius
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5:
            h = h_new
            trust_radius = min(2 * trust_radius, 0.5)  # Increase trust region
        else:
            trust_radius *= decay_rate
            if trust_radius < min_radius:
                trust_radius = min_radius
        
        if trust_radius < min_radius:
            break
            
    # Final refinement with L-BFGS if we have time
    if time.time() - start_time < budget_s - 5:
        # Re-initialize with the best solution
        h = best_h.copy()
        
        def objective(h_vals):
            return compute_c5(h_vals)
        
        def constraint(h_vals):
            return np.sum(h_vals) - n_points / 2

        # Optimize with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint}
        
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, 
                             options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6})
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                opt_c5 = compute_c5(h)
                if opt_c5 < best_c5:
                    best_c5 = opt_c5
                    best_h = h.copy()
        except Exception:
            pass
            
    # Final check on best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
