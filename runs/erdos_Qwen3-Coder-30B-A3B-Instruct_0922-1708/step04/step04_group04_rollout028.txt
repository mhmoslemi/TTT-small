import numpy as np

from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Newton-based optimization
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    def gradient(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_idx = np.argmax(correlation)
        max_val = correlation[max_idx]
        
        # Use finite differences for gradient
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - objective(h_vec)) / eps
        return grad

    # Alternating optimization with gradient descent and bounds
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient descent step
        grad = gradient(h)
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-5)
        h_new = h - lr * grad
        
        # Project back to feasible space
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update if better
        c5_new = objective(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new.copy()
        
        # Early termination based on convergence
        if iteration > 10 and np.abs(objective(h) - objective(h_new)) < 1e-10:
            break

    return (best_h.tolist(), best_c5, n_points)
