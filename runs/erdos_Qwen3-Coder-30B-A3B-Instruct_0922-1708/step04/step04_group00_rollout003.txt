import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # PSO parameters
    n_particles = 30
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    vmax = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals() and np.random.rand() < 0.5:
            # Use parent for some particles
            particle_h = h.copy()
        else:
            particle_h = np.random.uniform(0, 1, n_points)
            particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
            particle_h = np.clip(particle_h, 0, 1)
        particles.append(particle_h)
        velocities.append(np.random.uniform(-vmax, vmax, n_points))
        pbest.append(particle_h.copy())
        pbest_c5.append(float('inf'))
    
    gbest = h.copy()
    gbest_c5 = float('inf')
    
    # PSO iterations
    max_iter = 200
    no_improvement = 0
    max_no_improvement = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Compute fitness
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particle.copy()
                pbest_c5[i] = c5
                
            # Update global best
            if c5 < gbest_c5:
                gbest = particle.copy()
                gbest_c5 = c5
                
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
                
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            vel = (inertia * vel + 
                   c1 * r1 * (pbest[i] - particle) + 
                   c2 * r2 * (gbest - particle))
            
            # Clip velocity
            vel = np.clip(vel, -vmax, vmax)
            
            # Update position
            particle += vel
            
            # Enforce constraints
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            velocities[i] = vel
            particles[i] = particle
            
        # Early stopping
        if gbest_c5 < best_c5:
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > max_no_improvement:
                break
    
    # Local refinement with Nelder-Mead on best solution
    def objective(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx
    
    def constraint(h):
        return np.sum(h) - n_points / 2

    simplex_points = []
    simplex_points.append(best_h)
    
    for i in range(n_points):
        h_pert = best_h.copy()
        h_pert[i] += 0.01
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        simplex_points.append(h_pert)

        # Evaluate simplex
        if len(simplex.shape) == 1:
            simplex = simplex.reshape(1, -1)

        for point in simplex:
            correlation = np.correlate(point, 1 - point, mode='full')
            value = np.max(correlation) * dx
            simplex_values.append(value)

        max_iter_refine = 100

            # Sort simplex by value
            idx_sorted = np.argsort(simplex_values)

            values_sorted = simplex_values[idx_sorted]
            
            # Calculate centroid excluding worst
            centroid = np.mean(simplex_sorted[:-1], axis=0)
            
            # Reflection
            x_r = centroid + 2 * (centroid - simplex_sorted[-1])

            x_r = x_r * (n_points / 2) / np.sum(x_r)
            correlation = np.correlate(x_r, 1 - x_r, mode='full')
            value_r = np.max(correlation) * dx

            if values_sorted[0] <= value_r < values_sorted[-2]:

                values_sorted[-1] = value_r

                x_e = centroid + 2 * (x_r - centroid)
                x_e = np.clip(x_e, 0, 1)
                x_e = x_e * (n_points / 2) / np.sum(x_e)

                value_e = np.max(correlation) * dx
                if value_e < value_r:
                    simplex_sorted[-1] = x_e
                    values_sorted[-1] = value_e
                else:
                    simplex_sorted[-1] = x_r
                    values_sorted[-1] = value_r
            else:
                # Contraction
                if value_r < values_sorted[-1]:
                    x_c = centroid + 0.5 * (x_r - centroid)
                    x_c = np.clip(x_c, 0, 1)

                        simplex_sorted[-1] = x_c
                        values_sorted[-1] = value_c

                            simplex_sorted[j] = np.clip(simplex_sorted[j], 0, 1)
                            simplex_sorted[j] = simplex_sorted[j] * (n_points / 2) / np.sum(simplex_sorted[j])
                        # Recalculate all values
                        values_sorted = []
                        for point in simplex_sorted:
                            correlation = np.correlate(point, 1 - point, mode='full')
                            value = np.max(correlation) * dx
                            values_sorted.append(value)

                    # Recalculate all values
                    values_sorted = []

                        values_sorted.append(value)
                    values_sorted = np.array(values_sorted)
            
            simplex = simplex_sorted
            simplex_values = values_sorted
            
            # Update best
            if values_sorted[0] < best_c5:

    except:

    return (best_h.tolist(), best_c5, n_points)
