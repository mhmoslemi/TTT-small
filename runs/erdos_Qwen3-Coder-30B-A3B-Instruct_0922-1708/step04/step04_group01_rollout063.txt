        h = np.zeros(n_points)
        h[:n_points//2] = 2.0

    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def convert_to_binary(h_vals):
        # Convert to binary by thresholding
        h_binary = (h_vals > 0.5).astype(float)
        # Adjust to maintain sum = n_points/2
        current_sum = np.sum(h_binary)
        diff = int(n_points/2 - current_sum)
        if diff > 0:
            # Need to flip diff zeros to ones
            zeros = np.where(h_binary == 0)[0]
            flip_indices = np.random.choice(zeros, diff, replace=False)
            h_binary[flip_indices] = 1
        elif diff < 0:
            # Need to flip |diff| ones to zeros
            ones = np.where(h_binary == 1)[0]
            flip_indices = np.random.choice(ones, -diff, replace=False)
            h_binary[flip_indices] = 0
        return h_binary
    
    def mutate_binary(h_binary, rate=0.05):
        # Flip some bits
        mask = np.random.random(len(h_binary)) < rate
        h_binary = h_binary.copy()
        h_binary[mask] = 1 - h_binary[mask]
        # Adjust to maintain sum
        current_sum = np.sum(h_binary)
        diff = int(n_points/2 - current_sum)
        if diff > 0:
            zeros = np.where(h_binary == 0)[0]
            flip_indices = np.random.choice(zeros, diff, replace=False)
            h_binary[flip_indices] = 1
        elif diff < 0:
            ones = np.where(h_binary == 1)[0]
            flip_indices = np.random.choice(ones, -diff, replace=False)
            h_binary[flip_indices] = 0
        return h_binary
    
    def crossover_binary(h1, h2):
        # Arithmetic crossover follow by adjustment
        alpha = np.random.random()
        h_child = alpha * h1 + (1 - alpha) * h2
        # Convert to binary
        h_child = convert_to_binary(h_child)
        return h_child
    
    # Initialize population
    population_size = 100
    population = []

        h_init = np.random.random(n_points) * 2
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)

        fitness.sort()

        if fitness[0][0] < best_c5:
            best_c5 = fitness[0][0]

            parent2 = np.random.choice(top_population)
            child = crossover_binary(parent1, parent2)

        for i in range(population_size//3):
            ind = np.random.choice(top_population)
            mut_ind = mutate_binary(ind, rate=0.05)
            new_population.append(mut_ind)
        
        # Trim to population_size
        population = new_population[:population_size]
    
    # Convex optimization on the best solution
    try:
        # First convert best_h to actual values (not binary) and rescale
        h_finetune = best_h.copy()
        if np.sum(h_finetune) > 0:
            h_finetune = h_finetune * (n_points / 2) / np.sum(h_finetune)
        
        # Define the optimization variables
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]

        # Use the computed overlap directly
        correlation = np.correlate(h_finetune, 1 - h_finetune, mode='full')

        # This is a quadratic program approximating the cost 
        
        # Modified warmstart approach
        h_opt = h_finetune.copy()
        # Refine with gradient-descent or other means
        for _ in range(20):
            correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5 = np.max(correlation) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                break
                
            # Compute approximate gradient using finite differences
            grad = np.zeros_like(h_opt)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_opt.copy()

                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)

            # Update h

            h_new = h_opt - lr * grad

    except Exception as e:

    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Ensure feasibility
    if np.sum(best_h) < n_points/2 - 1e-6 or np.sum(best_h) > n_points/2 + 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
