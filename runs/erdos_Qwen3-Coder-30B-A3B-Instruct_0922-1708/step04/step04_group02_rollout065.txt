import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_segments = 10
    n_points = 100
    dx = 2.0 / n_points
    
    # Build initial sparse representation
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Convert to piecewise segments
        # Use a simple approach: find significant regions
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        # Discretize and build sparse representation
        segments = []
        pos = [0]
        heights = []
        for i in range(n_segments):
            start = i * n_points // n_segments
            end = (i + 1) * n_points // n_segments
            avg_h = np.mean(h_init[start:end])
            heights.append(avg_h)
            pos.append(end * 2.0 / n_points)
        pos = np.array(pos)
        heights = np.array(heights)
    else:
        # Uniform segment initialization
        pos = np.linspace(0, 2, n_segments + 1)
        heights = np.ones(n_segments) * (1.0 / n_segments)
    
    # Convert to optimization variables (positions and heights)
    x0 = np.concatenate([pos[1:-1], heights])
    
    def evaluate_sparse_h(x, n_segments=n_segments):
        # Reconstruct h from sparse representation
        positions = np.concatenate([[0], x[:n_segments-1], [2]])
        heights = x[n_segments-1:]
        
        # Ensure valid segment structure
        for i in range(1, len(positions)):
            positions[i] = max(positions[i-1] + 1e-8, positions[i])
        heights = np.clip(heights, 0, 1)
        
        # Normalize to satisfy integral constraint
        total_width = np.diff(positions)
        integral = np.sum(total_width * heights)
        heights = heights * (n_points / 2) / integral
        
        # Build dense h for overlap calculations
        h = np.zeros(n_points)
        for i in range(n_segments):
            start_idx = int(positions[i] / 2 * n_points)
            end_idx = int(positions[i+1] / 2 * n_points)
            h[start_idx:end_idx] = heights[i]
        
        return h
    
    def objective(x):
        h = evaluate_sparse_h(x)
        correlation = np.correlate(h, 1 - h, mode='full')
        max_correlation = np.max(correlation)
        return max_correlation * dx
    
    def constraint_integral(x, n_segments=n_segments):
        h = evaluate_sparse_h(x)
        integral = np.sum(h) * dx
        return integral - 1.0
    
    def constraint_pos(x, n_segments=n_segments):
        positions = np.concatenate([[0], x[:n_segments-1], [2]])
        diffs = np.diff(positions)
        return np.minimum(diffs, 0)  # This should be negative if positions are not increasing
    
    # Constraints
    constraints = [
        {'type': 'eq', 'fun': constraint_integral},
        {'type': 'ineq', 'fun': lambda x: np.diff(np.concatenate([[0], x[:n_segments-1], [2]]))},
    ]
    
    # Optimize using SLSQP
    best_x = x0.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Random seed for initial points
    seed_points = []
    for _ in range(3):
        # Sample perturbed initial solution
        x_pert = x0 + np.random.normal(0, 0.05, len(x0))
        seed_points.append(x_pert)
    
    # Add some local searches
    for i in range(10):
        x0_local = np.copy(best_x)
        x0_local += np.random.normal(0, 0.02, len(x0_local))
        try:
            res = minimize(objective, x0_local, method='SLSQP', constraints=constraints, 
                          options={'maxiter': 500, 'ftol': 1e-6}, bounds=[(0, 2)] * (n_segments-1) + [(0, 1)] * n_segments)
            if res.success:
                c5_temp = objective(res.x)
                if c5_temp < best_c5:
                    best_x = res.x
                    best_c5 = c5_temp
        except:
            pass
            
        if time.time() - start_time > budget_s - 5:
            break
    
    # Convert final best solution
    final_h = evaluate_sparse_h(best_x)
    
    # Final refinement with optimization - use once more 
    # to try to lower C5 if we're close to target
    try:
        res = minimize(objective, best_x, method='SLSQP', constraints=constraints,
                      options={'maxiter': 100, 'ftol': 1e-8}, bounds=[(0, 2)] * (n_segments-1) + [(0, 1)] * n_segments)
        if res.success:
            c5_final = objective(res.x)
            if c5_final < best_c5:
                final_h = evaluate_sparse_h(res.x)
                best_c5 = c5_final
    except:
        pass
    
    # Fast computation of final C5 directly using correlation
    correlation = np.correlate(final_h, 1 - final_h, mode='full')
    best_c5_final = np.max(correlation) * dx
    
    return (final_h.tolist(), best_c5_final, n_points)
