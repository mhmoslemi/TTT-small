import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    h_binary = (h_init >= 0.5).astype(int)
    ones_count = np.sum(h_binary)
    target_ones = n_points // 2
    
    # Adjust to exactly target_ones
    if ones_count > target_ones:
        # Need to turn some 1s into 0s
        zero_indices = np.where(h_binary == 0)[0]
        one_indices = np.where(h_binary == 1)[0]
        num_to_flip = ones_count - target_ones
        flip_indices = np.random.choice(one_indices, size=num_to_flip, replace=False)
        h_binary[flip_indices] = 0
    elif ones_count < target_ones:
        # Need to turn some 0s into 1s
        zero_indices = np.where(h_binary == 0)[0]
        num_to_flip = target_ones - ones_count
        flip_indices = np.random.choice(zero_indices, size=num_to_flip, replace=False)
        h_binary[flip_indices] = 1
    
    best_h = h_binary.copy()
    best_c5 = float('inf')
    
    # Temperature schedule
    T = 1.0
    decay_rate = 0.999
    max_iter = 10000
    
    start_time = time.time()
    
    current_h = h_binary.copy()
    current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Make a random change
        idx = np.random.randint(0, n_points)
        new_h = current_h.copy()
        new_h[idx] = 1 - new_h[idx]  # Flip the bit
        
        # Project to ensure exactly target_ones ones
        old_ones = np.sum(current_h)
        new_ones = np.sum(new_h)
        if new_ones != target_ones:
            # Find indices to flip to balance
            if new_ones > target_ones:
                flip_indices = np.where(new_h == 1)[0]
                num_to_flip = new_ones - target_ones
                to_flip = np.random.choice(flip_indices, size=num_to_flip, replace=False)
                new_h[to_flip] = 0
            else:
                flip_indices = np.where(new_h == 0)[0]
                num_to_flip = target_ones - new_ones
                to_flip = np.random.choice(flip_indices, size=num_to_flip, replace=False)
                new_h[to_flip] = 1
        
        # Compute new C5
        new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        
        # Accept or reject
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        # Update temperature
        T *= decay_rate
        
        # Random perturbation every 500 iterations
        if iter_count % 500 == 0:
            # Reset 10% of values randomly
            reset_indices = np.random.choice(n_points, size=n_points//10, replace=False)
            for idx in reset_indices:
                new_h[idx] = np.random.randint(0, 2)
            # Project to target ones
            current_ones = np.sum(new_h)
            if current_ones != target_ones:
                if current_ones > target_ones:
                    one_indices = np.where(new_h == 1)[0]
                    num_to_flip = current_ones - target_ones
                    flip_indices = np.random.choice(one_indices, size=num_to_flip, replace=False)
                    new_h[flip_indices] = 0
                else:
                    zero_indices = np.where(new_h == 0)[0]
                    num_to_flip = target_ones - current_ones
                    flip_indices = np.random.choice(zero_indices, size=num_to_flip, replace=False)
                    new_h[flip_indices] = 1
            current_h = new_h
            current_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
        
        # Early termination if no improvement
        if no_improvement_count >= max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
