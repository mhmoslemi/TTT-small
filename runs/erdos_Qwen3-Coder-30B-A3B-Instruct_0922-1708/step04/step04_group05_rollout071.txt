import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise use a structured initialization
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        # Structured initialization: piecewise constant with some smooth transitions
        h_init = np.ones(n_points)
        # Create a few segments with varying heights
        segments = 4
        segment_size = n_points // segments
        for i in range(segments):
            start_idx = i * segment_size
            end_idx = (i + 1) * segment_size if i < segments - 1 else n_points
            if i % 2 == 0:
                h_init[start_idx:end_idx] = 0.6
            else:
                h_init[start_idx:end_idx] = 0.4
        h_init = h_init * (n_points / 2) / np.sum(h_init)

    # Ensure integral constraint
    h = h_init.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Number of initial candidates to generate
    num_candidates = 20
    candidates = [h.copy()]
    
    # Generate diverse starting points
    for i in range(num_candidates - 1):
        h_perturbed = h.copy()
        # Random perturbations
        noise = np.random.normal(0, 0.01, n_points)
        h_perturbed += noise
        # Clip and renormalize
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        candidates.append(h_perturbed)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Evaluate each candidate and find best so far
        for candidate in candidates:
            current_corr = np.correlate(candidate, 1 - candidate, mode='full')
            current_c5 = np.max(current_corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = candidate.copy()
        
        # Generate better candidates by refining the best solution
        if iter_count % 10 == 0 and time.time() - start_time < budget_s - 2:
            new_candidates = []
            h_refined = best_h.copy()
            # Gradient ascent to improve this specific candidate
            for _ in range(10):
                if time.time() - start_time > budget_s - 2:
                    break
                grad = np.zeros(n_points)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    corr = np.correlate(h_refined, 1 - h_refined, mode='full')
                    c5 = np.max(corr) * dx
                    grad[i] = (c5_plus - c5) / eps
                # Update step
                lr = 0.005 / (1 + np.linalg.norm(grad) * 1e-2)
                h_refined = h_refined - lr * grad
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                
                # Check termination condition
                new_c5 = np.max(np.correlate(h_refined, 1-h_refined, mode='full')) * dx
                if new_c5 > c5:
                    break
            candidates.append(h_refined)
            # Keep only the best candidates
            candidates.sort(key=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx)
            candidates = candidates[:num_candidates]
        
        # Apply random mutations to candidates
        for i, candidate in enumerate(candidates):
            if np.random.rand() < 0.3 and time.time() - start_time < budget_s - 2:
                new_candidate = candidate.copy()
                # Add Gaussian noise
                noise = np.random.normal(0, 0.005, n_points)
                new_candidate += noise
                # Clip and renormalize
                new_candidate = np.clip(new_candidate, 0, 1)
                new_candidate = new_candidate * (n_points / 2) / np.sum(new_candidate)
                candidates[i] = new_candidate
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
