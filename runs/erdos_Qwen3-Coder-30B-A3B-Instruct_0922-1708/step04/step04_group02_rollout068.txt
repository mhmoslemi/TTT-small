import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    n_segments = 10  # Number of segments in sparse representation
    
    # Initialize sparse representation
    pos = np.linspace(0, 2, n_segments + 1)
    h = np.ones(n_segments) * (1.0 / n_segments)
    
    # Constraint: integral should be 1
    integral = np.sum((pos[1:] - pos[:-1]) * h)
    h = h * (1.0 / integral)
    
    # If parent construction exists, use it for initialization
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Interpolate to n_points
        x_orig = np.linspace(0, 2, len(h_init))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_orig, h_init)
        # Reconstruct sparse segments from this
        h_sparse = np.zeros(n_segments)
        for i in range(n_segments):
            start_idx = int(i * n_points / n_segments)
            end_idx = int((i + 1) * n_points / n_segments)
            h_sparse[i] = np.mean(h_interp[start_idx:end_idx])
        h = h_sparse * (1.0 / np.sum(h_sparse) * n_segments / 2)
    
    best_h = None
    best_c5 = float('inf')
    best_pos = None
    start_time = time.time()
    
    # Define function to compute overlap for given pos and h
    def compute_c5(pos_vals, h_vals):
        # Compute the full cross-correlation
        h_full = np.zeros(n_points)
        for i in range(n_segments):
            start_idx = int(pos_vals[i] / dx)
            end_idx = int(pos_vals[i+1] / dx)
            h_full[start_idx:end_idx] = h_vals[i]
        
        # Compute overlaps
        c5_vals = []
        for k in range(1, n_points):
            # Run convolution at lags from 1 to n_points
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5_k = np.max(corr)
            c5_vals.append(c5_k * dx)
        
        return max(c5_vals)
    
    # Function for optimization
    def objective(vars):
        # vars[0:n_segments] = h_vals
        # vars[n_segments:] = pos_vals
        h_vals = vars[:n_segments]
        pos_vals = vars[n_segments:]
        
        # Ensure pos_vals are increasing
        pos_vals = np.clip(pos_vals, 0, 2)
        pos_vals = np.sort(pos_vals)
        pos_vals = np.concatenate([[0], pos_vals, [2]])
        
        # Ensure h_vals are in [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        
        # Normalize h_vals to satisfy integral constraint
        integral = np.sum((pos_vals[1:] - pos_vals[:-1]) * h_vals)
        if integral > 0:
            h_vals = h_vals * (1.0 / integral)
        
        # Compute C5
        return compute_c5(pos_vals, h_vals)
    
    # Constraints
    def constraint_integral(vars):
        h_vals = vars[:n_segments]
        pos_vals = vars[n_segments:]
        pos_vals = np.sort(pos_vals)
        pos_vals = np.concatenate([[0], pos_vals, [2]])
        integral = np.sum((pos_vals[1:] - pos_vals[:-1]) * h_vals)
        return 1.0 - integral
    
    def constraint_pos_order(vars):
        pos_vals = vars[n_segments:]
        pos_vals = np.sort(pos_vals)
        pos_vals = np.concatenate([[0], pos_vals, [2]])
        diff = np.diff(pos_vals)
        return np.min(diff)  # Should be > 0
    
    # Optimization
    def optimize_sparse():
        nonlocal best_h, best_c5, best_pos
        last_best_c5 = float('inf')
        max_stagnation = 10
        stagnation_count = 0
        
        # Initial variables: [h_vals, pos_vals]
        x0 = np.concatenate([
            h, 
            np.linspace(0.1, 1.9, n_segments-1)  # Initial positions
        ])
        
        bounds = [(0, 1) for _ in range(n_segments)] + [(0, 2) for _ in range(n_segments-1)]
        
        # Optimize
        constraints = [
            {'type': 'eq', 'fun': constraint_integral},
            {'type': 'ineq', 'fun': constraint_pos_order}
        ]
        
        # Optimization with monitoring
        while time.time() - start_time < budget_s - 20 and stagnation_count < max_stagnation:
            try:
                # Randomize starting point for different search paths
                x0 = np.concatenate([
                    np.random.uniform(0, 1, n_segments),
                    np.random.uniform(0.1, 1.9, n_segments-1)
                ])
                x0 = np.clip(x0, 0, 1)
                
                # Solve optimization subproblem
                result = minimize(
                    objective,
                    x0,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8},
                    callback=lambda x: None  # No callback for now
                )
                
                if result.success and result.fun < last_best_c5:
                    last_best_c5 = result.fun
                    if result.fun < best_c5:
                        best_c5 = result.fun
                        # Reconstruct best_h
                        h_vals = result.x[:n_segments]
                        pos_vals = np.sort(result.x[n_segments:])
                        pos_vals = np.concatenate([[0], pos_vals, [2]])
                        
                        # Update final reconstruction if possible
                        best_h = np.zeros(n_points)
                        for i in range(n_segments):
                            start_idx = int(pos_vals[i] / dx)
                            end_idx = int(pos_vals[i+1] / dx)
                            if end_idx > start_idx:
                                best_h[start_idx:end_idx] = h_vals[i]
                        best_pos = pos_vals
                        
                # Check for stagnation
                if abs(last_best_c5 - best_c5) < 1e-6 and best_c5 != float('inf'):
                    stagnation_count += 1
                else:
                    stagnation_count = 0
                    
            except Exception as e:
                if 'nan' in str(e) or 'inf' in str(e):
                    pass
                else:
                    continue  # Continue loop if error
                
            # Try local refinement on best solution
            try:
                if best_h is not None:
                    # Create adapted function
                    h_refine = np.interp(np.linspace(0, 2, n_points), 
                                       np.linspace(0, 2, len(best_h)), 
                                       best_h)
                    
                    # Evaluation
                    correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
                    c5_val = np.max(correlation) * dx
                    
                    if c5_val < best_c5:
                        best_c5 = c5_val
            except Exception:
                pass
        
        if best_h is None:
            # Fallback to initial segment distribution
            best_h = np.zeros(n_points)
            for i in range(n_segments):
                start_idx = int(pos[i] / dx)
                end_idx = int(pos[i+1] / dx)
                if end_idx > start_idx:
                    best_h[start_idx:end_idx] = h[i]
        
        return best_c5
    
    # Perform optimization
    last_c5 = optimize_sparse()
    total_time = time.time() - start_time
    
    # Final adjustment if time permits
    while time.time() - start_time < budget_s - 10:
        try:
            # Fine-tune final function
            if best_h is not None:
                h_fine = np.interp(np.linspace(0, 2, n_points), 
                                  np.linspace(0, 2, len(best_h)), 
                                  best_h)
                
                correlation = np.correlate(h_fine, 1 - h_fine, mode='full')
                c5_val = np.max(correlation) * dx
                
                if c5_val < best_c5:
                    best_c5 = c5_val
                    
        except Exception as e:
            break
    
    # Return final result
    if best_h is None:
        # Fallback to uniform distribution
        best_h = np.ones(n_points) * 0.5
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), best_c5, n_points)
