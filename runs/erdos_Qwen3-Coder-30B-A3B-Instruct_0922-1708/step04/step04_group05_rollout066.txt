import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Scale to meet integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Gradient-based iterative refinement with FFT convolution
    max_iter = 2000
    gamma = 0.001
    decay_every = 100
    patience = 100
    patience_count = 0
    prev_c5 = float('inf')
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute overlap using fft-based convolution
        h_full = np.concatenate([h, np.zeros(n_points - 1)])
        one_minus_h_full = np.concatenate([1 - h, np.zeros(n_points - 1)])
        overlap_full = np.real(np.fft.ifft(np.fft.fft(h_full) * np.fft.fft(one_minus_h_full)))
        overlap_full = overlap_full[:n_points]
        
        # Find maximum overlap
        max_idx = np.argmax(overlap_full)
        max_overlap = overlap_full[max_idx]
        
        # Compute C5 value
        current_c5 = max_overlap * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_count = 0
        else:
            patience_count += 1
            
        if patience_count > patience:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_full_plus = np.concatenate([h_plus, np.zeros(n_points - 1)])
            one_minus_h_full_plus = np.concatenate([1 - h_plus, np.zeros(n_points - 1)])
            overlap_plus_full = np.real(np.fft.ifft(np.fft.fft(h_full_plus) * np.fft.fft(one_minus_h_full_plus)))
            overlap_plus_full = overlap_plus_full[:n_points]
            
            grad[i] = (np.max(overlap_plus_full) - np.max(overlap_full)) / eps
        
        # Apply gradient step (with learning rate decay)
        if iter_num % decay_every == 0 and iter_num > 0:
            gamma *= 0.9
            
        h_new = h - gamma * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Momentum step
        h = 0.9 * h + 0.1 * h_new
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation
    h_full = np.concatenate([best_h, np.zeros(n_points - 1)])
    one_minus_h_full = np.concatenate([1 - best_h, np.zeros(n_points - 1)])
    overlap_full = np.real(np.fft.ifft(np.fft.fft(h_full) * np.fft.fft(one_minus_h_full)))
    overlap_full = overlap_full[:n_points]
    final_c5 = np.max(overlap_full) * dx
    
    return (best_h.tolist(), final_c5, n_points)
