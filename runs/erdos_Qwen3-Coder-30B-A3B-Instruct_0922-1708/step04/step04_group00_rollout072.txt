import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Use a hybrid approach: gradient-based refinement with random restarts
    max_iter = 500
    no_improvement = 0
    max_no_improvement = 30
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Start with current best
        current_h = best_h.copy()
        current_c5 = best_c5
        
        # Gradient-based optimization on a subset of points
        # Sample subset for gradient computation to speed up
        sample_indices = np.random.choice(n_points, min(30, n_points), replace=False)
        
        # Define objective for optimization
        def objective(params):
            h_local = current_h.copy()
            h_local[sample_indices] = params
            h_local = np.clip(h_local, 0, 1)
            h_local = h_local * (n_points / 2) / np.sum(h_local)
            
            correlation = np.correlate(h_local, 1 - h_local, mode='full')
            c5 = np.max(correlation) * dx
            return c5
            
        # Gradient computation using finite differences with low samples
        def gradient(params):
            grad = np.zeros_like(params)
            eps = 1e-4
            base_c5 = objective(params)
            for i in range(len(params)):
                perturbed = params.copy()
                perturbed[i] += eps
                perturbed_c5 = objective(perturbed)
                grad[i] = (perturbed_c5 - base_c5) / eps
            return grad
        
        # optimization
        try:
            # Start with simple gradient descent on subset
            lr = 0.02
            for _ in range(50):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Compute gradient
                current_subset = current_h[sample_indices]
                grad = gradient(current_subset)
                
                # Update step
                updated_subset = current_subset - lr * grad
                updated_subset = np.clip(updated_subset, 0, 1)
                
                # Place updated subset back
                new_h = current_h.copy()
                new_h[sample_indices] = updated_subset
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                # Evaluate new C5
                correlation = np.correlate(new_h, 1 - new_h, mode='full')
                new_c5 = np.max(correlation) * dx
                
                if new_c5 < current_c5:
                    current_c5 = new_c5
                    current_h = new_h.copy()
                    lr = min(lr * 1.05, 0.2)
                else:
                    lr = max(lr * 0.9, 1e-5)
                    if lr < 1e-5:
                        break
                        
            # Update global best if improved
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
                
        except Exception:
            no_improvement += 1
            
        # Random restart if no progress
        if no_improvement > max_no_improvement:
            # Reinitialize with small perturbation of best solution
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            best_h = h_perturbed.copy()
            correlation = np.correlate(best_h, 1 - best_h, mode='full')
            best_c5 = np.max(correlation) * dx
            no_improvement = 0
    
    # Final validation
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
