import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use CVXPY for optimization
    # Variable for step function
    h_var = cp.Variable(n_points)
    
    # Variable for the maximum overlap
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1,
        C >= 0
    ]
    
    # Build the overlap constraints for all shifts
    # For shifts k in [-n_points+1, n_points-1] excluding 0
    for k in range(-n_points+1, n_points):
        if k == 0:
            continue
            
        # Compute correlation for shift k
        if k > 0:
            # h[i] * (1 - h[i+k]) for i = 0 to n_points-k-1
            expr = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
        else:
            # h[i+k] * (1 - h[i]) for i = -k to n_points-1 (i.e. i = 0 to n_points+k-1)
            # But ignore those beyond the bounds
            expr = cp.sum(cp.multiply(h_var[-k:], 1 - h_var[:n_points+k]))
            
        # Add constraint: overlap <= C
        constraints.append(expr <= C)
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers and take the best solution
    solvers = ['ECOS', 'SCS']
    
    for solver in solvers:
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            prob.solve(solver=solver, verbose=False, max_iters=5000)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                c5_val = C.value
                
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_sol.copy()
                    
        except Exception:
            continue
    
    # Final refinement with gradient estimates using simple descent
    # If best solution found but needs further refinement, do that
    h = best_h.copy()
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_value = np.max(correlation) * dx
        
        # Compute gradient with finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5_value) / eps
            
        # Update with gradient descent
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-4)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
