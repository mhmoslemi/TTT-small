import numpy as np
import time
from scipy.signal import correlate
import heapq

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Helper to compute overlap
    def compute_overlap(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Helper to check constraints
    def validate_h(h):
        return np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n_points/2) < 1e-8
    
    # Multi-scale optimization
    start_time = time.time()
    w = 5  # window width
    perturbation_scale = 0.02
    max_iter = 10000
    improvement_threshold = 1e-6
    
    # Initial overlap
    c5 = compute_overlap(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Iterative improvement
    iter_no_improve = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps for prioritization
        overlaps = []
        for k in range(-(n_points-1), n_points):
            if k == 0:
                continue
            # Compute shift correlation
            k_abs = abs(k)
            if k > 0:
                corr = np.sum(h[k:] * (1 - h[:-k_abs]))
            else:
                corr = np.sum(h[:-k_abs] * (1 - h[k:]))
            overlaps.append((corr * dx, k))
        
        # Sort by overlap in descending order
        sorted_overlaps = sorted(overlaps, key=lambda x: x[0], reverse=True)
        
        # Select a problematic shift
        if sorted_overlaps:
            max_overlap, shift = sorted_overlaps[0]
            if max_overlap > 0.9 * best_c5 or iteration % 5 == 0:
                # Apply localized adjustments
                h_new = h.copy()
                # Find regions contributing to the max overlap
                k_abs = abs(shift)
                if shift > 0:
                    # h[shift:] overlaps with 1 - h[:-shift]
                    regions = list(range(shift, n_points))
                    # Modify region centered on overlap
                    if len(regions) > 0:
                        center = (regions[0] + regions[-1]) // 2
                        start_idx = max(0, center - w)
                        end_idx = min(n_points, center + w + 1)
                        # Adjust
                        for i in range(start_idx, end_idx):
                            if i < n_points - k_abs:
                                h_new[i] = min(1.0, h_new[i] + 0.01 * (1 - h_new[i]))
                            if i + k_abs < n_points:
                                h_new[i + k_abs] = max(0.0, h_new[i + k_abs] - 0.01 * h_new[i + k_abs])
                else:
                    # h[:-shift] overlaps with 1 - h[shift:]
                    regions = list(range(n_points - k_abs))
                    # Modify region centered on overlap
                    if len(regions) > 0:
                        center = (regions[0] + regions[-1]) // 2
                        start_idx = max(0, center - w)
                        end_idx = min(n_points, center + w + 1)
                        # Adjust
                        for i in range(start_idx, end_idx):
                            if i + k_abs < n_points:
                                h_new[i] = min(1.0, h_new[i] + 0.01 * (1 - h_new[i]))
                            if i < n_points - k_abs:
                                h_new[i + k_abs] = max(0.0, h_new[i + k_abs] - 0.01 * h_new[i + k_abs])
                
                # Rescale
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                # Clip
                h_new = np.clip(h_new, 0, 1)
                
                # Perturb around the problematic region
                if len(regions) > 0:
                    center = (regions[0] + regions[-1]) // 2
                    start_idx = max(0, center - w)
                    end_idx = min(n_points, center + w + 1)
                    for i in range(start_idx, end_idx):
                        if np.random.random() < 0.3:
                            noise = np.random.normal(0, perturbation_scale)
                            h_new[i] = np.clip(h_new[i] + noise, 0, 1)
                
                # Normalize
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Test new solution
                c5_new = compute_overlap(h_new)
                if c5_new < c5 - improvement_threshold or np.random.random() < 0.1:
                    h = h_new
                    c5 = c5_new
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    iter_no_improve = 0
                else:
                    iter_no_improve += 1
                    if iter_no_improve > 10:
                        # Increase window and perturbation if stuck
                        w = min(10, w + 1)
                        perturbation_scale *= 1.1
        else:
            # Random exploration
            h_new = h.copy()
            for i in range(n_points):
                if np.random.random() < 0.5:
                    noise = np.random.normal(0, 0.01)
                    h_new[i] = np.clip(h_new[i] + noise, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = compute_overlap(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
            
    return (best_h.tolist(), best_c5, n_points)
