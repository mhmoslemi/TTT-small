import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    h = (h_init >= 0.5).astype(int)
    num_ones = np.sum(h)
    target_ones = n_points // 2
    
    # Ensure correct number of ones
    if num_ones > target_ones:
        # Turn some 1s to 0s
        zero_indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(zero_indices, size=int(num_ones - target_ones), replace=False)
        h[flip_indices] = 0
    elif num_ones < target_ones:
        # Turn some 0s to 1s
        one_indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(one_indices, size=int(target_ones - num_ones), replace=False)
        h[flip_indices] = 1
    
    # Track best solution
    best_h = h.copy()
    current_h = h.copy()
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    current_c5 = compute_c5(current_h)
    best_c5 = current_c5
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 1.0
    decay_rate = 0.999
    perturbation_interval = 1000
    stagnation_threshold = 10000
    
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Random perturbation
        if iteration % perturbation_interval == 0:
            # Randomly flip 5% of the entries
            n_flip = int(n_points * 0.05)
            flip_indices = np.random.choice(n_points, size=n_flip, replace=False)
            for idx in flip_indices:
                current_h[idx] = 1 - current_h[idx]
            
            # Project to maintain exactly half ones
            num_ones = np.sum(current_h)
            target_ones = n_points // 2
            if num_ones > target_ones:
                zero_indices = np.where(current_h == 1)[0]
                flip_indices = np.random.choice(zero_indices, size=int(num_ones - target_ones), replace=False)
                current_h[flip_indices] = 0
            elif num_ones < target_ones:
                one_indices = np.where(current_h == 0)[0]
                flip_indices = np.random.choice(one_indices, size=int(target_ones - num_ones), replace=False)
                current_h[flip_indices] = 1
                
        # Select random index to flip
        idx = np.random.randint(0, n_points)
        new_h = current_h.copy()
        new_h[idx] = 1 - new_h[idx]
        
        # Project to maintain exactly half ones
        num_ones = np.sum(new_h)
        target_ones = n_points // 2
        if num_ones > target_ones:
            # Turn one 1 to 0
            zero_indices = np.where(new_h == 1)[0]
            flip_idx = np.random.choice(zero_indices)
            new_h[flip_idx] = 0
        elif num_ones < target_ones:
            # Turn one 0 to 1
            one_indices = np.where(new_h == 0)[0]
            flip_idx = np.random.choice(one_indices)
            new_h[flip_idx] = 1
        
        new_c5 = compute_c5(new_h)
        
        # Accept or reject the new solution
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                last_improvement = iteration
        else:
            # Accept with probability based on temperature
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                current_h = new_h
                current_c5 = new_c5
        
        # Decrease temperature
        T = T * decay_rate
        
        # Early termination if no improvement for a long time
        if iteration - last_improvement > stagnation_threshold:
            break
    
    return (best_h.tolist(), best_c5, n_points)
