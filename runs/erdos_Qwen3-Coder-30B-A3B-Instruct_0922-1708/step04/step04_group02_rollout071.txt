import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Current best
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Annealing parameters
    temp = 1.0
    cooling_rate = 0.99
    temp_drop_interval = 100
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Generate candidate
        h_new = h.copy()
        
        # Select random points (10%)
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        
        # Perturb selected points
        for i in indices:
            # Random perturbation
            perturb = np.random.uniform(-0.1, 0.1)
            h_new[i] += perturb
            
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain sum = n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5 for candidate
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.exp(-delta / temp) > np.random.random():
            h = h_new
            
        # Cooling schedule
        if iteration % temp_drop_interval == 0:
            temp *= cooling_rate
    
    # Local refinement around best solution
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_refine = best_h.copy()
        # Select few points for perturbation
        indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
        for i in indices:
            perturb = np.random.uniform(-0.01, 0.01)
            h_refine[i] += perturb
            
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
        
        correlation_refine = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5_refine = np.max(correlation_refine) * dx
        
        if c5_refine < best_c5:
            best_c5 = c5_refine
            best_h = h_refine
    
    return (best_h.tolist(), best_c5, n_points)
