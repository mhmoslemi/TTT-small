import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        # Random perturbation around base solution
        pert = np.random.normal(0, 0.1, n_points)
        h_p = h + pert
        h_p = np.clip(h_p, 0, 1)
        h_p = h_p * (n_points / 2) / np.sum(h_p)
        particles.append(h_p.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h_p.copy())
        corr = np.correlate(h_p, 1 - h_p, mode='full')
        pbest_vals.append(np.max(corr) * dx)
    
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    if gbest_val < best_c5:
        best_c5 = gbest_val
        best_h = gbest.copy()
    
    # PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = fitness
                
                # Update global best
                if fitness < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = fitness
                    
                    if gbest_val < best_c5:
                        best_c5 = gbest_val
                        best_h = gbest.copy()
    
    # CVX refinement of best particle
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Attempt to compute maximum overlap via CVX
        # Direct use of max in CVX is tricky, so we work with a scalar variable
        # We minimize a scalar y such that y >= max_k integral of h*(1-h_k)
        
        # Since direct approach isn't supported, we'll instead do simple refinement
        
        # Try a more direct approach, but still keep the optimized particle
        h_cvx = best_h.copy()
        
        # Refinement using gradient descent
        def obj_val(h_vec):
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(corr) * dx
            
        def grad_eval(h_vec):
            grad = np.zeros_like(h_vec)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - obj_val(h_vec)) / eps
                
            return grad
        
        # Iterative refinement
        current_h = h_cvx.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            g = grad_eval(current_h)
            lr = 0.01
            new_h = current_h - lr * g
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            current_h = new_h
            
        h_cvx = current_h
        
        # Evaluate final result
        correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
        c5_cvx = np.max(correlation) * dx
        
        if c5_cvx < best_c5:
            best_c5 = c5_cvx
            best_h = h_cvx.copy()
            
    except Exception as e:
        pass

    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
