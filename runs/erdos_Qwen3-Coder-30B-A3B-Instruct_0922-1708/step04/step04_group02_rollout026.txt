import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
        h = h_init / np.sum(h_init) * (n_points / 2)
    else:
        # Use simple alternating pattern
        n_points = 100
        h = np.ones(n_points) * 0.5
        
    # Flatten the solution to avoid memory issues
    h = np.clip(h, 0, 1) 
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Helper function to compute C5
    def compute_c5(h_vec):
        dx = 2.0 / len(h_vec)
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Adaptive mutation controller
    def adaptive_mutation_rate(generation, max_generations):
        # Start with higher rate, decay over time
        return max(0.05, 0.3 * (1 - generation / max_generations))
    
    # Mutation operator
    def mutate(h_vec, rate):
        h_new = h_vec.copy()
        n_points = len(h_vec)
        for i in range(n_points):
            if random.random() < rate:
                # Small random perturbation
                delta = np.random.normal(0, 0.02)
                h_new[i] = h_new[i] + delta
        # Repair to bounds
        h_new = np.clip(h_new, 0, 1)
        # Repair to sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Crossover operator (arithmetic)
    def crossover(h1, h2):
        alpha = random.random()
        h_new = alpha * h1 + (1 - alpha) * h2
        # Repair
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (len(h1) / 2) / np.sum(h_new)
        return h_new
    
    # Genetic algorithm with adaptive resolution
    max_generations = 1000
    population_size = 50
    generation = 0
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:
            population.append(h.copy())
        else:
            # Perturb base soln slightly
            h_perturbed = h.copy()
            for j in range(len(h_perturbed)):
                if random.random() < 0.1:
                    h_perturbed[j] += random.uniform(-0.05, 0.05)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            population.append(h_perturbed)
    
    # Main loop
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for i, ind in enumerate(population):
            c5 = compute_c5(ind)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50%
        keep_count = population_size // 2
        population = population[:keep_count]
        fitness = fitness[:keep_count]
        
        # Fill rest with offspring
        while len(population) < population_size:
            # Tournament selection
            p1_idx = random.randint(0, len(population) - 1)
            p2_idx = random.randint(0, len(population) - 1)
            
            # Crossover
            child = crossover(population[p1_idx], population[p2_idx])
            
            # Mutation
            rate = adaptive_mutation_rate(generation, max_generations)
            child = mutate(child, rate)
            
            population.append(child)
        
        generation += 1
    
    # Final gradient refinement on best solution
    h_refined = best_h.copy()
    
    # Local search around best
    def obj_func(h_vec):
        dx = 2.0 / len(h_vec)
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Gradient approximation and small improvements
    eps = 1e-6
    patience = 0
    old_c5 = obj_func(h_refined)
    
    while patience < 10 and time.time() - start_time < budget_s - 1:
        improved = False
        for i in range(len(h_refined)):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = obj_func(h_plus)
            
            # Gradient approximation
            grad = (c5_plus - old_c5) / eps
            
            # Update with small learning rate
            lr = 0.001
            h_new = h_refined - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = obj_func(h_new)
            
            if new_c5 < old_c5:
                h_refined = h_new
                old_c5 = new_c5
                improved = True
                
        if not improved:
            patience += 1
        else:
            patience = 0
    
    final_c5 = obj_func(h_refined)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_refined
    
    return (best_h.tolist(), best_c5, n_points)
