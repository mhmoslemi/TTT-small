import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Sliding window parameters
    window_width = 5
    perturb_scale = 0.02
    improvement_thresh = 1e-6
    
    def overlap_for_shift(h, k):
        """compute overlap for given shift k using convolution."""
        # ensures 1D array is passed to correlate
        if k > 0:
            # h * (1 - h right-shifted by k)
            corr = np.correlate(h, np.concatenate([np.ones(k), 1 - h[:-k]]), mode='valid')
        elif k < 0:
            # h * (1 - h left-shifted by |k|)
            corr = np.correlate(h, np.concatenate([1 - h[-k:], np.ones(-k)]), mode='valid')
        else:
            # Special case for k=0 overlap is not meaningful
            return 0
        return np.max(corr) * dx
    
    def compute_all_overlaps(h):
        """compute overlaps for all valid shifts."""
        overlaps = []
        for k in range(1 - n_points, n_points):
            if k == 0:
                continue
            c = overlap_for_shift(h, k)
            overlaps.append((c, k))
        return sorted(overlaps, key=lambda x: x[0], reverse=True)
            
    def apply_adjustment(h, k, w):
        """apply localized adjustment to reduce overlap for shift k."""
        h_new = h.copy()
        if k > 0:
            # h_i * (1 - h_{i+k})
            # Target: minimize sum over shifts
            # reduce values where both h_i and (1-h_{i+k}) are large.
            # Simple reverse: decrease h where both are large
            # We'll reduce weights of h in the window around i where overlap occurs.
            idx_start = max(0, k)
            idx_end = min(len(h_new), len(h_new) - k)
            for i in range(max(0, idx_start - w), min(len(h_new), idx_end + w)):
                if i + k < len(h_new):
                    if h_new[i] + (1 - h_new[i + k]) > 1.0:
                        h_new[i] = max(0, h_new[i] - 0.001)
        elif k < 0:
            k_abs = -k
            idx_start = max(0, k_abs)
            idx_end = min(len(h_new), len(h_new) - k_abs)
            for i in range(max(0, idx_start - w), min(len(h_new), idx_end + w)):
                if i + k_abs < len(h_new):
                    if h_new[i] + (1 - h_new[i + k_abs]) > 1.0:
                        h_new[i] = max(0, h_new[i] - 0.001)
        return np.clip(h_new, 0, 1)
    
    def random_perturb(h, w, scale):
        """add small random perturbation."""
        h_new = h.copy()
        for i in range(max(0, w), min(len(h_new), len(h_new) - w)):
            if np.random.random() < 0.1:
                delta = np.random.normal(0, scale)
                h_new[i] += delta
        return np.clip(h_new, 0, 1)
    
    # Main loop
    iter_count = 0
    no_improvement = 0
    refresh_frequency = 50
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # compute all possible overlaps
        all_overlaps = compute_all_overlaps(h)
        
        # extract largest overlap
        if not all_overlaps:
            break
        max_overlap, best_k = all_overlaps[0]
        
        # Check if improvement due to algorithm complexity
        if max_overlap <= best_c5 * 0.9999:
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > 10:
            # increase window
            window_width = min(10, window_width + 1)
        else:
            window_width = max(3, window_width)
            
        # apply adjustment to reduce max_overlap
        h = apply_adjustment(h, best_k, window_width)
        h = np.clip(h, 0, 1)
        # rescale to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # add small random perturbation
        h = random_perturb(h, window_width, perturb_scale)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute new overlap
        all_overlaps_new = compute_all_overlaps(h)
        if all_overlaps_new:
            new_max = all_overlaps_new[0][0]
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h.copy()
        
        # Break if composite minimum improvement not achieved over 100 steps
        if iter_count % refresh_frequency == 0:
            # Recompute and update best
            all_overlaps = compute_all_overlaps(h)
            if all_overlaps:
                new_max = all_overlaps[0][0]
                if new_max < best_c5:
                    best_c5 = new_max
                    best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
