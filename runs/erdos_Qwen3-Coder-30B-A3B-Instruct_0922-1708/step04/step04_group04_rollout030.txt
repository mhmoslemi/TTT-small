import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Iterative convex optimization with line search
    max_iter = 100
    tol = 1e-5
    step_size = 0.5
    backtracking_ratio = 0.5
    solver_tolerances = 1e-4
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 via full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(h_var) == n_points / 2]
        constraints += [0 <= h_var, h_var <= 1]
        
        # Add linearized cross-correlation constraints
        # For each k, we have sum_{i} h[i] * (1 - h[i+k]) <= C
        # Linearize h[i+k] using h_old[i+k]
        h_old = h.copy()
        for k in range(1, n_points):
            # Compute left and right parts for the constraint h[i] * (1 - h[i+k]) <= C
            # Linearized: h[i] * (1 - h_old[i+k]) + h_old[i+k] * (h[i] - h_old[i]) <= C
            # i.e., h[i] - h[i]*h_old[i+k] + h_old[i+k]*h[i] - h_old[i+k]*h_old[i] <= C
            # Simplified: h[i] - h_old[i+k]*h_old[i] <= C
            # Actually, let's compute: h[i]*h_old[i+k] from h_old, linearize it:
            constraint = cp.sum(cp.multiply(h_var, 1 - h_old[k:])) <= C
            constraints.append(constraint)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, abstol=solver_tolerances, reltol=solver_tolerances)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = np.array(h_var.value).flatten()
                # Line search
                alpha = step_size
                while alpha > 1e-10:
                    h_candidate = h + alpha * (h_new - h)
                    h_candidate = np.clip(h_candidate, 0, 1)
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    
                    corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                    c5_candidate = np.max(corr) * dx
                    if c5_candidate <= current_c5:
                        h = h_candidate
                        break
                    alpha *= backtracking_ratio
                else:
                    h = h_new
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            else:
                # If solver fails, use gradient descent step
                grad = np.zeros_like(h)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - current_c5) / eps
                
                # Gradient descent step
                lr = 0.001
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except Exception:
            # Fallback to gradient descent
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - current_c5) / eps

            lr = 0.001
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
        
        # Early termination
        if np.abs(current_c5 - best_c5) < tol:
            break
    
    # Final check of best solution
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h.tolist(), final_c5, n_points)
