import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure the integral constraint sum(h) * dx = 1
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h = h_init.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Frequency domain approach with gradient descent
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation using FFT
        h_full = np.concatenate([h, np.zeros(n_points - 1)])
        one_minus_h_full = np.concatenate([1 - h, np.zeros(n_points - 1)])
        conv_result = np.correlate(h_full, one_minus_h_full, mode='full')
        conv_result = conv_result[:n_points]  # Keep only valid convolution
        
        current_c5 = np.max(conv_result) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient in frequency domain
        H = np.fft.fft(h)
        G = np.zeros(n_points, dtype=complex)
        max_shift = np.argmax(conv_result)
        # Phase shift that decreases overlap at the peak
        omega = 2 * np.pi * np.arange(n_points) / n_points
        if max_shift < n_points:
            G = 1 - np.exp(-1j * omega * max_shift)
        else:
            G = 1 - np.exp(-1j * omega * (max_shift - n_points))
        G = np.where(np.abs(G) > 1e-10, G, 1.0)
        
        H_new = H * G / np.abs(G)
        h_new = np.real(np.fft.ifft(H_new))
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update with momentum
        momentum = 0.9
        h = momentum * h + (1 - momentum) * h_new
        
        # Project again
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodically add noise to escape local minima
        if iteration % 50 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final check
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
