import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, n_points, dx):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_step(h, n_points, dx, step_size=0.001):
    """Perform a gradient descent step to reduce C5"""
    eps = 1e-5
    grad = np.zeros(n_points)
    
    # Compute finite difference gradient
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        c5_plus = compute_c5(h_plus, n_points, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_feasible(h_minus, n_points)
        c5_minus = compute_c5(h_minus, n_points, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    # Update h with gradient step
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n_points)
    
    return h_new

    # Phase 2: Fine-tune with local refinement
    fine_refinement_time = 400
