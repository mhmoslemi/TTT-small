import numpy as np
from scipy.optimize import minimize
import time
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 1000
    inertia_w = 0.7
    phi_p = 2.0
    phi_g = 2.0
    local_search_freq = 50
    rejuvenate_freq = 200
    max_stagnation = 50
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
    else:
        # Start with block function
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
    
    # Initialize other particles randomly
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    
    # Initialize velocities
    for _ in range(swarm_size):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize personal bests and global best 
    pbest = [particles[i].copy() for i in range(swarm_size)]
    pbest_fitness = [float('inf')] * swarm_size
    gbest = particles[0].copy()
    gbest_fitness = float('inf')
    
    # Compute initial fitness for particles and find best
    for i in range(swarm_size):
        h = particles[i]
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        pbest_fitness[i] = c5
        
        if c5 < gbest_fitness:
            gbest = h.copy()
            gbest_fitness = c5
    
    # PSO main loop
    iteration = 0
    stagnation_count = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update particles
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            
            # Compute fitness for current particle
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_fitness[i]:
                pbest[i] = h.copy()
                pbest_fitness[i] = c5
                
                if c5 < gbest_fitness:
                    gbest = h.copy()
                    gbest_fitness = c5
                    stagnation_count = 0  # Reset stagnation on new best
            
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            v = (inertia_w * v + 
                 phi_p * r1 * (pbest[i] - h) + 
                 phi_g * r2 * (gbest - h))
            
            # Update position
            h_new = h + v
            
            # Repair feasibility
            h_new = np.clip(h_new, 0, 1)
            sum_h = np.sum(h_new)
            
            if abs(sum_h - n_points/2) > 1e-6:
                # Rescale to maintain constraint
                h_new = h_new * (n_points / 2) / sum_h
                
            # Update particle and velocity
            particles[i] = h_new
            velocities[i] = v
        
        # Local search on top particles
        if iteration % local_search_freq == 0 or iteration == 1:
            # Sort particles by fitness
            sorted_indices = np.argsort([pbest_fitness[i] for i in range(swarm_size)])
            top_indices = sorted_indices[:swarm_size // 10]
            
            # Apply local search
            for idx in top_indices:
                h_current = pbest[idx].copy()
                current_fitness = pbest_fitness[idx]
                
                # Simple gradient descent with small steps
                for _ in range(10):
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    
                    for j in range(n_points):
                        h_plus = h_current.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - current_fitness) / eps
                    
                    h_new = h_current - 0.001 * grad
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    
                    corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                    new_fitness = np.max(corr_new) * dx
                    
                    if new_fitness < current_fitness:
                        h_current = h_new
                        current_fitness = new_fitness
                    else:
                        break
                        
                # Update pbest if improved
                if current_fitness < pbest_fitness[idx]:
                    pbest[idx] = h_current
                    pbest_fitness[idx] = current_fitness
                    
                    if current_fitness < gbest_fitness:
                        gbest = h_current
                        gbest_fitness = current_fitness
                        stagnation_count = 0
        
        # Rejuvenation of worst particles
        if iteration % rejuvenate_freq == 0 and iteration > 0:
            # Sort and get worst particles
            sorted_indices = np.argsort(pbest_fitness)[::-1]
            worst_indices = sorted_indices[:swarm_size // 10]
            
            for idx in worst_indices:
                # Replace with new random particle
                h_new = np.random.rand(n_points)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                particles[idx] = h_new
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                pbest[idx] = h_new.copy()
                pbest_fitness[idx] = float('inf')
        
        stagnation_count += 1
        if stagnation_count > max_stagnation:
            break
    
    # Final refinement
    best_h = gbest.copy()
    best_c5 = gbest_fitness
    
    # Intensive local search around best solution
    h_refined = best_h.copy()
    for _ in range(1000):
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
            
        h_new = h_refined - 0.0005 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < best_c5:
            h_refined = h_new
            best_c5 = new_c5
        else:
            break
    
    return (h_refined.tolist(), best_c5, n_points)
