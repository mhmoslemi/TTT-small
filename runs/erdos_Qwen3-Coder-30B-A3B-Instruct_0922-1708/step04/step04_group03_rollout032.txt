import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation using CVXPY
    # We minimize sum of all correlations over the full convolution
    print("Starting convex relaxation phase...")
    
    # Create a convex relaxation of the optimization problem
    # Create variable matrix for each point
    h_var = cp.Variable(n_points)
    
    # Make all h[i] between 0 and 1
    constraints = [h_var >= 0, h_var <= 1]
    
    # Constraint: sum h = n_points/2
    constraints.append(cp.sum(h_var) == n_points/2)
    
    # Try to minimize total (not max) overlap using a convex relaxation
    # which is something like summing all convolutions together
    total_overlap = 0
    
    # Attempt to create an approximate convex form of the objective
    # Specifically, with uniform h, correlation is simpler
    # Now we'll add a penalty to only the maximum
    
    # Constraints are simple bounds and sum
    
    # convexify by minimizing the sum of overlapping terms
    # This initial relaxation provides a good starting solution
    prob = cp.Problem(cp.Minimize(cp.sum(h_var)), constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        # Use CVXPY solution as initial point
        if prob.status == 'optimal':
            h = h_var.value.copy()
            # Normalize to ensure integral constraint
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
        else:
            print("Solving convex relaxation failed, using original initialization")
    except:
        print("CVXPY error, using original initialization")
    
    # Phase 2: Local search on the initial solution
    print("Starting local search...")
    
    # Use projected gradient descent with finite differences
    max_no_improvement = 50
    no_improvement_count = 0
    iteration = 0
    lr = 0.01
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 or iteration < max_iter:
        iteration += 1
        if iteration > max_iter:
            break
            
        # Compute current correlation maximum
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.9
            
        # Update with gradient
        h_new = h - lr * grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)  # enforce sum constraint
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
