```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
import random

def evaluate_h(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def ensure_sum_constraint(h, target_sum, dx):
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) < 1e-6:
        return h
    diff = int((target_sum - current_sum) / dx)
    if diff == 0:
        return h
    h = h.copy()
    if diff > 0:
        # Need to add more ones
        zero_indices = np.where(h == 0)[0]
        if len(zero_indices) >= diff:
            for i in range(diff):
                idx = zero_indices[i]
                h[idx] = 1.0
    else:
        # Need to remove some ones
        one_indices = np.where(h == 1)[0]
        if len(one_indices) >= -diff:
            for i in range(-diff):
                idx = one_indices[i]
                h[idx] = 0.0
    return h

def mutate_binary(individual, mutation_rate):
    individual = individual.copy()
    n = len(individual)
    for i in range(n):
        if random.random() < mutation_rate:
            individual[i] = 1 - individual[i]
    return ensure_sum_constraint(individual, 50.0, 2.0/n)

def crossover_binary(parent1, parent2):
    # Simple uniform crossover
    child = parent1.copy()
    n = len(parent1)
    for i in range(n):
        if random.random() < 0.5:
            child[i] = parent2[i]
    return ensure_sum_constraint(child, 50.0, 2.0/n)

def binary_to_float(binary_h, dx):
    # Convert binary vector to float vector with same sum
    return binary_h * (50.0 / np.sum(binary_h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = 50.0
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        # Use initial_h_values as starting point
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        # Make binary
        h0_binary = (h0 > 0.5).astype(int)
        h0_binary = ensure_sum_constraint(h0_binary, target_sum, dx)
        population.append(h0_binary)
    else:
        # Start with uniform binary distribution
        h0 = np.zeros(n_points)
        h0[:50] = 1
        h0 = ensure_sum_constraint(h0, target_sum, dx)
        population.append(h0)
        
    # Fill rest of population randomly
    for _ in range(population_size - len(population)):
        h = np.zeros(n_points)
        ones = np.random.choice(n_points, 50, replace=False)
        h[ones] = 1
        h = ensure_sum_constraint(h, target_sum, dx)
        population.append(h)
    
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    mutation_rate = 0.05
    crossover_rate = 0.5
    
    max_generations = 100
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append((c5, h))
            
        fitness.sort(key=lambda x: x[0])
        if fitness[0][0] < best_c5:
            best_c5 = fitness[0][0]
            best_h = fitness[0][1].copy()
            
        # Selection: take top 10%
        top_indices = [i for i in range(int(0.1 * population_size))]
        
        # Make offspring
        offspring = []
        for _ in range(int(0.5 * population_size)):
            if random.random() < crossover_rate:
                parent1 = population[random.choice(top_indices)]
                parent2 = population[random.choice(top_indices)]
                child = crossover_binary(parent1, parent2)
            else:
                parent = population[random.choice(top_indices)]
                child = mutate_binary(parent, mutation_rate)
            offspring.append(child)
            
        # Mutation
        mutated = []
        for _ in range(int(0.3 * population_size)):
            parent = population[random.choice(top_indices)]
            child = mutate_binary(parent, mutation_rate)
            mutated.append(child)
        
        # Create new population
        population = [fitness[i][1] for i in range(int(0.1 * population_size))]
        population.extend(offspring)
        population.extend(mutated)
        
        # Adjust to fit population size
        if len(population) > population_size:
            population = population[:population_size]
            
    # Try convex refinement on best candidates
    top_candidates = []
    for h in population:
        c5 = evaluate_h(h, dx)
        top_candidates.append((c5, h))
        
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = [x[1] for x in top_candidates[:10]]
    
    for h in top_candidates:
        try:
            # Relax constraint to continuous case
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == target_sum]
            
            # Convex optimization for better smoothing
            h_binary = binary_to_float(h, dx)
            
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL and h_vars.value is not None:
                h_refined = np.clip(h_vars.value, 0, 1)
                h_refined = h_refined * (target_sum / np.sum(h_refined))
                
                c5_refined = evaluate_h(h_refined, dx)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
        except:
            pass
            
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
