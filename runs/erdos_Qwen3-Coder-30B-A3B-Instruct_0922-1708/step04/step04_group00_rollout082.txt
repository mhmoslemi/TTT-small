import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective_and_gradient(h):
        # Ensure h is within bounds and sum constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute overlap integral using convolution (full mode)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        return c5, grad
    
    # Optimization using trust-region reflective method
    max_iter = 500
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        # Ensure we have a valid h
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        c5, _ = objective_and_gradient(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Optimization step using trust-region
        try:
            # Create bounds
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Use trust-region reflective method that handles bounds
            result = minimize(
                lambda x: objective_and_gradient(x)[0],
                h,
                method='TRF',
                bounds=bounds,
                options={'maxiter': 50}
            )
            
            if result.success:
                h = result.x
            else:
                # Fallback: stochastic local search if TRF fails
                h = best_h.copy()
                for _ in range(10):
                    idx = np.random.randint(0, n_points)
                    h[idx] = np.clip(h[idx] + np.random.normal(0, 0.01), 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    
        except:
            # Fallback in case of numerical issues
            h = best_h.copy()
            for _ in range(10):
                idx = np.random.randint(0, n_points)
                h[idx] = np.clip(h[idx] + np.random.normal(0, 0.01), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        iteration += 1
    
    # Final check of best solution
    h = np.clip(best_h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    correlation = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (h.tolist(), float(final_c5), n_points)
