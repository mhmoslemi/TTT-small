import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Initialize population for differential evolution
    population_size = 50
    population = []
    fitness = []
    
    # Create initial population with constraint satisfaction
    for i in range(population_size):
        if i == 0 and 'initial_h_values' in globals():
            individual = h.copy()
        else:
            individual = np.random.uniform(0, 1, n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        corr = np.correlate(individual, 1 - individual, mode='full')
        fitness.append(np.max(corr) * dx)
    
    # Best solution tracking
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # DE parameters
    F = 0.8
    CR = 0.9
    max_no_improvement = 100
    no_improvement_count = 0
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        new_population = []
        new_fitness = []
        improved = False
        
        for i in range(population_size):
            # DE mutation: select three distinct individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + F * (b - c)
            
            # Crossover
            trial = np.empty(n_points)
            jrand = np.random.randint(n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == jrand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            
            # Ensure constraints
            trial = np.clip(trial, 0, 1)
            trial = trial * (n_points / 2) / np.sum(trial)
            
            # Evaluate
            corr = np.correlate(trial, 1 - trial, mode='full')
            c5_trial = np.max(corr) * dx
            
            # Selection
            if c5_trial < fitness[i]:
                new_population.append(trial)
                new_fitness.append(c5_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = trial.copy()
                    improved = True
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
                
        population = new_population
        fitness = new_fitness
        
        # Local refinement every 20 iterations
        if iteration % 20 == 0:
            h = best_h.copy()
            lr = 0.01
            
            # Gradient-like local search: adjust each point
            for _ in range(50):
                corr = np.correlate(h, 1 - h, mode='full')
                c5_current = np.max(corr) * dx
                
                # Simple finite difference gradient estimate
                grad = np.zeros(n_points)
                eps = 1e-4
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5_current) / eps
                
                # Update with gradient descent
                h_new = h - lr * grad
                
                # Enforce constraints
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Accept improvement
                corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr_new) * dx
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                else:
                    h = h_new
                    
        # Early stopping
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        if no_improvement_count >= max_no_improvement:
            break
        
    return (best_h.tolist(), best_c5, n_points)
