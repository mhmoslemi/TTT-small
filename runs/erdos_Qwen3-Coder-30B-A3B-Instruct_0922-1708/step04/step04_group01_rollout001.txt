import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Implement PSO approach
    n_particles = 30
    max_iter = 1000
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_pos = []
    pbest_val = []
    
    for _ in range(n_particles):
        # Initialize particle with random values
        particle_h = np.random.rand(n_points)
        particle_h = np.clip(particle_h, 0, 1)
        particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
        particles.append(particle_h.copy())
        velocities.append(np.zeros(n_points))
        pbest_pos.append(particle_h.copy())
        pbest_val.append(float('inf'))
    
    # Initialize global best
    gbest_pos = h.copy()
    gbest_val = float('inf')
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Compute fitness for current particle
            particle = particles[i]
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_val[i]:
                pbest_val[i] = c5
                pbest_pos[i] = particle.copy()
            
            # Update global best
            if c5 < gbest_val:
                gbest_val = c5
                gbest_pos = particle.copy()
                
        # Update global best again to be sure
        if gbest_val < best_c5:
            best_c5 = gbest_val
            best_h = gbest_pos.copy()
            
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            vel = w * velocities[i] + c1 * r1 * (pbest_pos[i] - particles[i]) + c2 * r2 * (gbest_pos - particles[i])
            velocities[i] = np.clip(vel, -0.5, 0.5)
            particles[i] += velocities[i]
            
            # Ensure feasibility
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # After PSO, try CVX refinement using the best solution
    try:
        # Define the optimization variables and constraints
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Objective: minimize the maximum overlap using CVX
        # We'll use a more direct optimization approach
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass

    # Final check with the best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
