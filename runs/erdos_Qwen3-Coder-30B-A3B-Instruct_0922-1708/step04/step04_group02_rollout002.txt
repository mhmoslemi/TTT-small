import numpy as np
from scipy.optimize import minimize
import time
import random
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 500
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 20
    
    # Initialize population
    def create_individual():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def evaluate(individual):
        correlation = np.correlate(individual, 1 - individual, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def mutate(individual, rate):
        mutated = individual.copy()
        for i in range(len(mutated)):
            if random.random() < rate:
                mutated[i] += np.random.normal(0, 0.05)
                mutated[i] = max(0, min(1, mutated[i]))
        # Rescale to maintain sum constraint
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        child1 = parent1.copy()
        child2 = parent2.copy()
        
        # Single-point crossover
        point = random.randint(1, len(parent1) - 1)
        child1[point:] = parent2[point:]
        child2[point:] = parent1[point:]
        
        # Rescale children
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        return child1, child2
    
    def repair(individual):
        # Ensure individual is valid
        individual = np.clip(individual, 0, 1)
        # Normalize to sum to n_points/2
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual
    
    # Create initial population
    population = [create_individual() for _ in range(population_size)]
    evaluated_population = [(ind, evaluate(ind)) for ind in population]
    evaluated_population.sort(key=lambda x: x[1])
    best_individual = evaluated_population[0][0]
    best_c5 = evaluated_population[0][1]
    
    # Evolution loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current population
        evaluated_population = [(ind, evaluate(ind)) for ind in population]
        evaluated_population.sort(key=lambda x: x[1])
        
        # Update best solution
        if evaluated_population[0][1] < best_c5:
            best_c5 = evaluated_population[0][1]
            best_individual = evaluated_population[0][0]
        
        # Selection: top elite_size individuals
        elite = [ind for ind, _ in evaluated_population[:elite_size]]
        
        # Generate new population
        new_population = deepcopy(elite)
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(evaluated_population)
            parent2 = tournament_selection(evaluated_population)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    # Final refinement with gradient descent
    best_individual = np.array(best_individual)
    best_c5 = evaluate(best_individual)
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient approximation
        eps = 1e-4
        grad = np.zeros_like(best_individual)
        
        for i in range(len(best_individual)):
            h_plus = best_individual.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
        
        # Update with gradient descent
        step_size = 0.01
        new_individual = best_individual - step_size * grad
        new_individual = np.clip(new_individual, 0, 1)
        new_individual = new_individual * (n_points / 2) / np.sum(new_individual)
        
        # Evaluate new solution
        new_c5 = evaluate(new_individual)
        
        if new_c5 < best_c5:
            best_individual = new_individual
            best_c5 = new_c5
        else:
            break
    
    return (best_individual.tolist(), best_c5, n_points)

def tournament_selection(evaluated_population, tournament_size=3):
    selected = random.sample(evaluated_population, tournament_size)
    selected.sort(key=lambda x: x[1])
    return selected[0][0]
