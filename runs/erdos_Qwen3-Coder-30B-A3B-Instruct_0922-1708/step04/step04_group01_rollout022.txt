import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # Uniform initialization
        
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    # Coarse grid search to find promising regions
    coarse_size = 20
    n_coarse = coarse_size ** 2
    coarse_h_all = []
    
    # Generate coarse grid samples
    for i in range(coarse_size):
        for j in range(coarse_size):
            h_coarse = np.ones(coarse_size) * 0.5
            h_coarse[:i] = 1.0
            h_coarse[i:j] = 0.0
            h_coarse[j:] = 1.0
            h_coarse = h_coarse * (coarse_size / 2) / np.sum(h_coarse)
            coarse_h_all.append(h_coarse)
    
    # Evaluate coarse candidates
    coarse_scores = []
    for h_c in coarse_h_all:
        # Upsample to fine resolution
        h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, coarse_size), h_c)
        h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
        c5 = np.max(np.correlate(h_upsampled, 1 - h_upsampled, mode='full')) * dx
        coarse_scores.append((c5, h_upsampled))
        
    coarse_scores.sort(key=lambda x: x[0])
    top_candidates = coarse_scores[:10]
    
    # Refine using different techniques
    refinement_iterations = 500
    for c5, h_coarse in top_candidates:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Refinement with CVXPY
        try:
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_cvx = h_vars.value
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                c5_cvx = np.max(np.correlate(h_cvx, 1 - h_cvx, mode='full')) * dx
                
                if c5_cvx < best_c5:
                    best_c5 = c5_cvx
                    best_h = h_cvx.copy()
        except:
            pass

        # Gradient-based refinement
        h_refine = h_coarse.copy()
        for _ in range(refinement_iterations):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute approximate gradient
            grad = np.zeros_like(h_refine)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr = np.correlate(h_refine, 1 - h_refine, mode='full')
                c5_ref = np.max(corr) * dx
                
                grad[i] = (c5_plus - c5_ref) / eps
            
            # Update with gradient descent
            lr = 0.01
            h_new = h_refine - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_refine = h_new
            
            # Check for improvement
            corr_refine = np.correlate(h_refine, 1 - h_refine, mode='full')
            c5_refine = np.max(corr_refine) * dx
            
            if c5_refine < best_c5:
                best_c5 = c5_refine
                best_h = h_refine.copy()

    # Final check
    if best_h is None:
        best_h = h.copy()
    
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    return (best_h.tolist(), final_c5, n_points)
