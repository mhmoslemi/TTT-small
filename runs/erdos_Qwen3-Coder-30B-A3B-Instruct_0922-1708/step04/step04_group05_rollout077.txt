import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure the integral constraint sum(h) * dx = 1
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    n_particles = 20
    max_iter = 200
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    inertia_weight = w_start
    
    # Initialize PSO swarm
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for i in range(n_particles):
        h = h_init + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest_positions.append(h.copy())
        pbest_values.append(float('inf'))
    
    gbest_position = h_init.copy()
    gbest_value = float('inf')
    
    start_time = time.time()
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        inertia_weight = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < pbest_values[i]:
                pbest_values[i] = c5
                pbest_positions[i] = h.copy()
                
            if c5 < gbest_value:
                gbest_value = c5
                gbest_position = h.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2), np.random.random(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - particles[i]) + 
                            c2 * r2 * (gbest_position - particles[i]))
            particles[i] += velocities[i]
            
            # Clip particles to [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            # Renormalize integral
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Refinement stage
    top_candidates = []
    for i, (h, c5) in enumerate(zip(pbest_positions, pbest_values)):
        top_candidates.append((c5, h))
        
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:5]  # Top 5 candidates
    
    best_h = gbest_position.copy()
    best_c5 = gbest_value
    
    max_refine_iter = 500
    refine_iter = 0
    
    # Refine each top candidate
    for c5, candidate_h in top_candidates:
        if refine_iter >= max_refine_iter or time.time() - start_time > budget_s - 5:
            break
            
        h = candidate_h.copy()
        h_old = h.copy()
        
        # Line search in one direction
        search_dir = np.random.normal(0, 0.01, n_points)
        search_dir = search_dir * (n_points / 2) / np.sum(search_dir)
        
        # Golden section line search
        a, b = -1.0, 1.0
        phi = (1 + np.sqrt(5)) / 2
        resphi = 2 - phi
        x1 = a + resphi * (b - a)
        x2 = b - resphi * (b - a)
        
        for _ in range(100):
            if refine_iter >= max_refine_iter or time.time() - start_time > budget_s - 5:
                break
                
            h1 = h + x1 * search_dir
            h1 = np.clip(h1, 0, 1)
            h1 = h1 * (n_points / 2) / np.sum(h1)
            corr1 = np.correlate(h1, 1 - h1, mode='full')
            c5_1 = np.max(corr1) * dx
            
            h2 = h + x2 * search_dir
            h2 = np.clip(h2, 0, 1)
            h2 = h2 * (n_points / 2) / np.sum(h2)
            corr2 = np.correlate(h2, 1 - h2, mode='full')
            c5_2 = np.max(corr2) * dx
            
            if c5_1 < c5_2:
                b = x2
                x2 = x1
                x1 = a + resphi * (b - a)
            else:
                a = x1
                x1 = x2
                x2 = b - resphi * (b - a)
                
            refine_iter += 1
            
        h_opt = h + x1 * search_dir
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
        c5_opt = np.max(corr_opt) * dx
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Verify final result
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
