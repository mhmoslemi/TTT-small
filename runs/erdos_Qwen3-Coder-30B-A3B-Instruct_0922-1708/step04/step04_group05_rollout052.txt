import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    max_iter = 200
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h = h_init.copy()
    
    # PSO parameters
    n_particles = 20
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    max_vel = 0.1
    
    # Initialize PSO
    particles = np.random.rand(n_particles, n_points) * 0.2 + h_init.reshape(1, -1)
    velocities = (np.random.rand(n_particles, n_points) - 0.5) * 0.01
    best_positions = particles.copy()
    best_values = np.full(n_particles, np.inf)
    global_best = h.copy()
    global_best_value = np.inf
    
    start_time = time.time()
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Normalize particles and evaluate
        for i in range(n_particles):
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            value = np.max(corr) * dx
            
            if value < best_values[i]:
                best_values[i] = value
                best_positions[i] = particles[i].copy()
            
            if value < global_best_value:
                global_best_value = value
                global_best = particles[i].copy()
        
        # Update velocities and positions
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (best_positions[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            particles[i] += velocities[i]
            
            # Adaptive noise addition
            if iteration % 20 == 0 and np.random.rand() < 0.3:
                noise = np.random.normal(0, 0.001, n_points)
                particles[i] += noise
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Refinement stage using line search
    top_candidates = np.argsort(best_values)[:5]
    for idx in top_candidates:
        h_candidate = best_positions[idx].copy()
        best_candidate = h_candidate.copy()
        best_c5 = best_values[idx]
        
        # Perform line search in a few directions
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Generate a random perturbation that preserves integral
            pert = np.random.randn(n_points)
            pert = pert - np.mean(pert)  # Ensures no net change in mean
            pert = pert * (n_points / 2) / np.sum(np.abs(pert))  # Scale appropriately
            
            # Try several stepsizes
            for stepsize in [0.001, 0.005, 0.01]:
                h_new = h_candidate + stepsize * pert
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)  # Rescale
                
                corr = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr) * dx
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_candidate = h_new.copy()
                    h_candidate = h_new.copy()
        
        if best_c5 < global_best_value:
            global_best_value = best_c5
            global_best = best_candidate.copy()
    
    final_corr = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (global_best.tolist(), final_c5, n_points)
