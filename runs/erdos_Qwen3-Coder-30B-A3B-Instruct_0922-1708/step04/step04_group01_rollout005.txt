import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Store all candidates from grid search
    candidates = []
    grid_size = 20
    
    start_time = time.time()
    elapsed = 0
    
    # Phase 1: Grid Search
    grid_points = np.linspace(0, 1, grid_size)
    
    # Create all grid points (very inefficient - but saves memory)
    total_combinations = grid_size ** grid_size
    
    # Use random sampling to avoid full enumeration
    np.random.seed(seed)
    
    # Sample random points in the grid space
    n_samples = min(10000, total_combinations)
    if total_combinations <= 10000:
        # Small enough to enumerate all
        grid_indices = np.array(np.meshgrid(*[range(grid_size)]*grid_size, indexing='ij')).T.reshape(-1, grid_size)
    else:
        # Sample randomly
        grid_indices = np.random.randint(0, grid_size, (n_samples, grid_size))
    
    for i in range(min(n_samples, 1000)):
        if elapsed > budget_s - 5:
            break
        indices = grid_indices[i]
        h_candidate = grid_points[indices]
        h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        try:
            correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_val = np.max(correlation) * dx
            candidates.append((h_candidate.copy(), c5_val))
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_candidate.copy()
        except:
            continue
        elapsed = time.time() - start_time
        
    # Sort and keep top candidates
    candidates.sort(key=lambda x: x[1])
    top_candidates = candidates[:min(10, len(candidates))]
    
    # Phase 2: Refinement with.top candidates
    for _, h_init in top_candidates:
        h_refined = h_init.copy()
        lr = 0.01
        for step in range(500):
            if elapsed > budget_s - 2:
                break
            # Gradient estimation
            grad = np.zeros_like(h_refined)
            eps = 1e-4
            for j in range(n_points):
                h_pert = h_refined.copy()
                h_pert[j] += eps
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                
                corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
                c5_pert = np.max(corr_pert) * dx
                
                corr_ref = np.correlate(h_refined, 1 - h_refined, mode='full')
                c5_ref = np.max(corr_ref) * dx
                
                grad[j] = (c5_pert - c5_ref) / eps
            
            h_refined = h_refined - lr * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            elapsed = time.time() - start_time
            
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5_refined = np.max(corr) * dx
            
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
    
    # Final optimization with CVXPY if still time
    if time.time() - start_time < budget_s - 2:
        try:
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Compute the maximum overlap efficiently
            def compute_max_overlap(h_vec):
                corr = np.correlate(h_vec, 1 - h_vec, mode='full')
                return np.max(corr) * dx
            
            # For large number of variables, using a more direct formulation
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_cvx = h_vars.value
                if h_cvx is not None:
                    h_cvx = np.clip(h_cvx, 0, 1)
                    h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                    
                    correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                    c5_cvx = np.max(correlation) * dx
                    
                    if c5_cvx < best_c5:
                        best_h = h_cvx
                        best_c5 = c5_cvx
        except:
            pass
    
    # Final evaluation of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
