import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Define objective function
    def objective(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Define constraint function for sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint dictionary
    constraint_dict = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization parameters
    max_iter = min(500, int(budget_s / 0.2))
    no_improvement = 0
    max_no_improvement = 20
    
    # Trust region optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use trust region reflective method
        result = minimize(
            objective,
            h,
            method='TRF',
            bounds=bounds,
            constraints=constraint_dict,
            options={'maxiter': 10, 'initial_tr_radius': 0.1}
        )
        
        if result.success:
            h = result.x
            # Project to feasible region
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Compute C5
            c5 = objective(h)
            
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
                
            if no_improvement > max_no_improvement:
                break
        else:
            # If optimization fails, try random perturbation
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            
    # Final refinement with local search
    max_refine_iter = 100
    for refine_iter in range(max_refine_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(best_h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update with gradient descent
        lr = 0.01
        h_new = best_h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update best if improved
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
