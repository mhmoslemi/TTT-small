import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def evaluate_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def compute_gradient(h, n_points):
    eps = 1e-4
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus)
        grad[i] = (c5_plus - base_c5) / eps
    
    return grad

def line_search(h, grad, n_points, max_iter=20):
    alpha = 1.0
    lr = 0.001
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for _ in range(max_iter):
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n_points)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            lr *= 2
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
                
    return best_h, best_c5

    else:
        h = np.ones(n_points) * 0.5
    
    h = project_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()

    coarse_grid_size = 100
    fine_grid_size = 50
    total_iterations = 0
    
    for coarse_iter in range(20):
        if time.time() - start_time > budget_s - 10:

        # Generate candidates using random perturbations
        candidates = []
        for _ in range(coarse_grid_size):
            h_candidate = h + np.random.normal(0, 0.05, n_points)
            h_candidate = project_feasible(h_candidate, n_points)
            c5 = evaluate_c5(h_candidate)
            candidates.append((c5, h_candidate))
            
        candidates.sort(key=lambda x: x[0])
        
        # Refine top candidates
        for _, h_candidate in candidates[:int(coarse_grid_size * 0.2)]:
            # Gradient-based refinement
            for _ in range(100):
                if time.time() - start_time > budget_s - 5:
                    break
                    
                grad = compute_gradient(h_candidate, n_points)
                h_candidate, c5_new = line_search(h_candidate, grad, n_points)
                
                if c5_new < best_c5:
                    best_c5 = c5_new

                # Early stopping based on small gradient or time
