import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Integration function
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def project_to_feasible(h):
        """Project h to feasible region: [0,1] and sum = n_points/2"""
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Generate Latin Hypercube Samples for exploration
    def generate_lhs_samples(n_samples, n_dim):
        samples = np.random.random((n_samples, n_dim))
        for i in range(n_dim):
            samples[:, i] = np.sort(samples[:, i])
            samples[:, i] = (samples[:, i] * n_samples + np.arange(n_samples)) / n_samples
        return samples
    
    # Multi-start refinement with improved start points
    lhs_samples = generate_lhs_samples(50, n_points)
    for i in range(50):
        h_candidate = lhs_samples[i]
        # Ensure sum constraint
        h_candidate = project_to_feasible(h_candidate)
        
        # Gradient descent
        h_current = h_candidate.copy()
        learning_rate = 0.01
        for iteration in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            c5_current = compute_c5(h_current)
            eps = 1e-5
            grad = np.zeros_like(h_current)
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus)
                
                h_minus = h_current.copy()
                h_minus[j] -= eps
                h_minus = project_to_feasible(h_minus)
                
                c5_plus = compute_c5(h_plus)
                c5_minus = compute_c5(h_minus)
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Adaptive learning rate
            h_new = h_current - learning_rate * grad
            h_new = project_to_feasible(h_new)
            
            # Check improvement
            c5_new = compute_c5(h_new)
            if c5_new < c5_current:
                h_current = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_current.copy()
            else:
                learning_rate *= 0.95  # Reduce learning rate if no improvement
                if learning_rate < 1e-6:
                    break
        
        # Also try original initial vector as candidate
        if i == 0 and 'initial_h_values' in globals():
            h_candidate = project_to_feasible(h.copy())
            
            # Gradient descent on original
            h_current = h_candidate.copy()
            learning_rate = 0.01
            for iteration in range(500):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                c5_current = compute_c5(h_current)
                eps = 1e-5
                grad = np.zeros_like(h_current)
                
                for j in range(n_points):
                    h_plus = h_current.copy()
                    h_plus[j] += eps
                    h_plus = project_to_feasible(h_plus)
                    
                    h_minus = h_current.copy()
                    h_minus[j] -= eps
                    h_minus = project_to_feasible(h_minus)
                    
                    c5_plus = compute_c5(h_plus)
                    c5_minus = compute_c5(h_minus)
                    
                    grad[j] = (c5_plus - c5_minus) / (2 * eps)
                
                # Adaptive learning rate
                h_new = h_current - learning_rate * grad
                h_new = project_to_feasible(h_new)
                
                # Check improvement
                c5_new = compute_c5(h_new)
                if c5_new < c5_current:
                    h_current = h_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_current.copy()
                else:
                    learning_rate *= 0.95  # Reduce learning rate if no improvement
                    if learning_rate < 1e-6:
                        break
    
    # Final refinement of the best solution
    h_final = best_h.copy()
    for iteration in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_current = compute_c5(h_final)
        eps = 1e-5
        grad = np.zeros_like(h_final)
        
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus)
            
            h_minus = h_final.copy()
            h_minus[j] -= eps
            h_minus = project_to_feasible(h_minus)
            
            c5_plus = compute_c5(h_plus)
            c5_minus = compute_c5(h_minus)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        h_new = h_final - 0.001 * grad
        h_new = project_to_feasible(h_new)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_final = h_new
        
    # Ensure final projection
    best_h = project_to_feasible(best_h)
    # Final computation of C5
    best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
