import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Trust region parameters
    trust_radius = 0.1
    min_radius = 1e-6
    radius_decay = 0.99
    max_iter = 500
    
    # Core optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Estimate gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Apply trust region method
        # Determine step direction
        step = -grad
        step_norm = np.linalg.norm(step)
        
        if step_norm > trust_radius:
            step = step * trust_radius / step_norm
            
        # Apply step
        h_new = h + step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept or reject step
        if c5_new < c5:
            h = h_new
            trust_radius = min(2 * trust_radius, 0.5)
        else:
            trust_radius *= radius_decay
            if trust_radius < min_radius:
                trust_radius = min_radius
                
        # Early stopping
        if trust_radius < min_radius and iteration > 10:
            break
            
    # Final refinement: local gradient descent with cooling
    h = best_h.copy()
    lr = 0.02
    for refine_iter in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        lr *= 0.998
        
        # Accept step if improvement
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        if c5_new < c5:
            h = h_new
            
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
