import numpy as np
import time

def project_feasible(h, n_points):
    """Projects h to satisfy bounds and integral constraint"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h):
    """Computes C5 from h values"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def gradient_step(h, eps=1e-4):
    """Computes gradient of C5 w.r.t. h"""
    n_points = len(h)
    grad = np.zeros(n_points)
    h_base = h.copy()
    c5_base = compute_c5(h_base)
    
    for i in range(n_points):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        c5_plus = compute_c5(h_plus)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

def line_search(h, grad, max_iter=20):
    """Performs line search to find optimal step size"""
    alpha_max = 1.0
    best_alpha = 0.0
    best_c5 = compute_c5(h)
    
    for i in range(max_iter):
        alpha = alpha_max * (0.9 ** i)
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, len(h))
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_alpha = alpha
    return best_alpha
