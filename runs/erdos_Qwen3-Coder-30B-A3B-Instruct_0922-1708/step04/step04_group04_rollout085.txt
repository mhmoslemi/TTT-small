import numpy as np
import heapq
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters
    window_width = 5
    perturbation_scale = 0.02
    min_improvement = 1e-6
    patience = 10
    
    def compute_overlap(h):
        # Compute cross-correlation for all shifts
        corr = signal.correlate(h, 1 - h, mode='full')
        # Extract valid overlaps (exclude zero shift)
        n_shifts = len(corr)
        valid_overlaps = corr[n_points-1:]
        return np.max(valid_overlaps) * dx
    
    def apply_adjustment(h, k, window_width):
        # Create a copy to avoid modifying during computation
        h_new = h.copy()
        # Determine affected regions based on shift k
        if k < 0:
            start_idx = -k
            end_idx = n_points
        else:
            start_idx = 0
            end_idx = n_points - k
        
        # Apply adjustment in window around the critical region
        for i in range(max(0, start_idx - window_width), min(n_points, end_idx + window_width)):
            if i < n_points - abs(k):
                # Adjust based on overlapping regions
                if k > 0:
                    # For positive k, h[i] overlaps with (1-h[i+k])
                    val = h_new[i] * (1 - h_new[i + k])
                else:
                    # For negative k, h[i-k] overlaps with (1-h[i])
                    val = h_new[i - k] * (1 - h_new[i])
                
                # Reduce overlap by adjusting h
                if val > 0.1:  # Only adjust when overlap is significant
                    # Decrease h in overlapping regions
                    if i < n_points - abs(k):
                        h_new[i] = max(0, h_new[i] - 0.01)
                        if k > 0:
                            h_new[i + k] = max(0, h_new[i + k] - 0.01)
                        else:
                            h_new[i - k] = max(0, h_new[i - k] - 0.01)
        
        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return np.clip(h_new, 0, 1)
    
    def random_perturb(h, window_width, scale):
        h_new = h.copy()
        # Select random points and perturb them
        num_perturbations = max(1, n_points // 20)
        indices = np.random.choice(n_points, num_perturbations, replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, scale)
            # Apply window-based perturbation
            start_idx = max(0, i - window_width)
            end_idx = min(n_points, i + window_width + 1)
            for j in range(start_idx, end_idx):
                if np.random.random() < 0.3:
                    h_new[j] += np.random.normal(0, scale / 2)
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Iteration loop
    improvement_count = 0
    last_improvement = 0
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = compute_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            improvement_count = 0
            last_improvement = iteration
        else:
            improvement_count += 1
            
        # Randomly perturb sometimes
        if np.random.random() < 0.3 or improvement_count > patience:
            h = random_perturb(h, window_width, perturbation_scale)
        else:
            # Apply most effective adjustments
            corr = signal.correlate(h, 1 - h, mode='full')
            n_shifts = len(corr)
            valid_overlaps = corr[n_points-1:]
            max_idx = np.argmax(valid_overlaps)
            best_k = max_idx - (n_points - 1)
            
            # Apply adjustment for the worst shift
            h = apply_adjustment(h, best_k, window_width)
            
            # Occasionally increase window size if stuck
            if improvement_count >= patience / 2:
                window_width = min(window_width + 1, 20)
    
    # Ensure final constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    final_c5 = compute_overlap(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
