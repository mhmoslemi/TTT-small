import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            part = np.array(initial_h_values)
            if len(part) != n_points:
                h_orig = np.array(initial_h_values)
                x_orig = np.linspace(0, 2, len(h_orig))
                x_target = np.linspace(0, 2, n_points)
                part = np.interp(x_target, x_orig, h_orig)
            part = part / np.sum(part) * (n_points / 2)
        else:
            part = np.zeros(n_points)
            part[:n_points//2] = 2.0
        part = np.clip(part, 0, 1)
        part = part * (n_points / 2) / np.sum(part)
        particles.append(part)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(part.copy())
        correlation = np.correlate(part, 1 - part, mode='full')
        pbest_vals.append(np.max(correlation) * dx)
    
    # Track global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = c5
                
                # Update global best
                if c5 < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = c5
        
        # Check if we should early stop (no improvement in last 100 iterations)
        if iter_num > 100:
            if gbest_val == best_c5:
                # Stuck, try alternative approach
                break
                
        if gbest_val < best_c5:
            best_c5 = gbest_val
            best_h = gbest.copy()
    
    # Try CVX optimization
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        # Create overlap matrix and minimize max of it
        # This works with cvxpy in this setup
        corr_expr = None
        for i in range(1, n_points):
            h_lag = np.roll(1 - best_h, i)
            corr_expr_i = cp.sum(cp.multiply(h_vars, h_lag))
            if corr_expr is None:
                corr_expr = corr_expr_i
            else:
                corr_expr = cp.vstack([corr_expr, corr_expr_i])
        
        objective = cp.Minimize(cp.max(corr_expr) * dx)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
                    
    except:
        pass
        
    # Final check with more precise calculation  
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
