import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale h to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure feasibility
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Iterative gradient descent in frequency domain
    max_iter = 1000
    gamma = 0.001  # Initial learning rate
    conv_threshold = 1e-5
    patience = 100
    no_improvement_count = 0
    
    # Precompute frequency grid
    freqs = np.fft.fftfreq(n_points, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap using FFT
        h_fft = np.fft.fft(h)
        one_minus_h_fft = np.fft.fft(1 - h)
        overlap_fft = h_fft * np.conj(one_minus_h_fft)
        overlap = np.fft.ifft(overlap_fft).real
        
        # Find maximum overlap location
        max_idx = np.argmax(overlap)
        max_shift = (max_idx - n_points + 1) * dx
        
        # Design frequency domain filter to reduce overlap at max_shift
        # G(omega) = exp(-j*omega*max_shift)
        G = np.exp(-1j * 2 * np.pi * freqs * max_shift)
        
        # Apply the filter in frequency domain
        h_new_fft = h_fft * G  # Apply filter to improve convergence
        
        # Inverse transform to get new spatial domain h
        h_new = np.fft.ifft(h_new_fft).real
        
        # Project to feasible region: [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to enforce integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update with momentum
        h = (1 - gamma) * h + gamma * h_new
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current C5
        current_overlap = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(current_overlap) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Early stopping if no improvement
        if no_improvement_count >= patience:
            break
        
        # Adaptive learning rate
        if iter_num % 100 == 0:
            gamma *= 0.9
    
    # Final evaluation
    final_overlap = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_overlap) * dx
    
    return (best_h.tolist(), final_c5, n_points)
