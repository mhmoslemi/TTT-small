import numpy as np
import time
from scipy.signal import correlate
import heapq

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simple gradient-based local search with adaptive windowing
    max_iter = 5000
    window_width = 5
    perturb_scale = 0.02
    min_improvement = 1e-6
    
    # Helper functions
    def compute_overlap(h):
        # Compute full correlation using FFT
        corr = correlate(h, 1-h, mode='full')
        return np.max(corr) * dx
    
    def validate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def adaptive_adjustment(h, window_width):
        # Compute overlap for all shifts
        corr = correlate(h, 1-h, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # We need to pick the index in the full correlation array
        # which corresponds to shift k = max_k - len(h) + 1
        shift_k = max_k - (len(h) - 1)
        
        # Apply windowed adjustment
        h_new = h.copy()
        
        # Find where correlation is maximized
        start_idx = max_k - window_width if max_k >= window_width else 0
        end_idx = min(max_k + window_width + 1, len(h))
        
        # Modify values near where overlap is max
        for i in range(start_idx, end_idx):
            # Simple heuristic: if in "bad" region, adjust
            h_new[i] += np.random.normal(0, 0.005)
        
        h_new = validate(h_new)
        return h_new, shift_k
    
    def random_perturb(h, window_width, scale):
        h_new = h.copy()
        # Perturb a random subset of window_width around a random position
        pert_start = np.random.randint(0, len(h))
        for i in range(window_width):
            if i + pert_start >= len(h):
                break
            if np.random.random() < 0.3:
                h_new[pert_start + i] += np.random.normal(0, scale)
        h_new = validate(h_new)
        return h_new
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_c5 = compute_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Adaptive random perturbation
        if np.random.random() < 0.5 or iteration < 100:
            h_new = random_perturb(h, window_width, perturb_scale)
            new_c5 = compute_overlap(h_new)
            if new_c5 < current_c5:
                h = h_new
        else:
            # Local adaptive adjustment
            h_new, shift_k = adaptive_adjustment(h, window_width)
            new_c5 = compute_overlap(h_new)
            if new_c5 < current_c5:
                h = h_new
            
            # Decrease window size if no improvement
            if new_c5 >= current_c5 and window_width > 1:
                window_width = max(1, window_width - 1)
            elif new_c5 < current_c5:
                window_width = min(10, window_width + 1)
            else:
                # If convergence stall, increase perturb scale
                perturb_scale *= 1.1
                
        # Occasionally reset to best if stagnates
        if iteration % 1000 == 0 and iteration > 0:
            h = best_h.copy()
            perturb_scale *= 0.9
            
        # Medium-strength random reset every few iterations to avoid local optima
        if iteration % 500 == 0 and iteration > 0:
            h = np.random.uniform(0.05, 0.95, n_points)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
