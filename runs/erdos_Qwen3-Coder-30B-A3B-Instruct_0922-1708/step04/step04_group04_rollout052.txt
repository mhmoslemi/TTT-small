import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # CVXPY approach
    # Define variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # Compute overlap terms
    max_c = 0
    for k in range(1, n_points):
        # For positive shifts (k > 0)
        if k < n_points:
            overlap = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
            constraints.append(overlap <= C)
            max_c = max(max_c, k)
            
        # For negative shifts (k < 0), we need to be careful about indexing
        # Actually, we can handle this more directly with reflect and mirror
        # but it's easier to consider shifts as shifting in the array
    
    # Add constraint from both positive and negative shifts
    # We consider all possible lags from 1 to n_points-1
    for k in range(1, n_points):
        # For k shifts to the right, we have overlap sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
        if k < n_points:
            # The i-th term in correlation is h[i] * (1 - h[i+k]) for valid i
            # For this to be valid, we need i + k < n_points => i < n - k
            # So i ranges from 0 to n - k - 1 inclusive, giving us n - k terms
            # Therefore, both h and (1-h) have n-k terms in the product
            terms = cp.multiply(h_var[:n_points-k], 1 - h_var[k:])
            overlap = cp.sum(terms)
            constraints.append(overlap <= C)  # We could also add it once for the positive
            max_c = max(max_c, k)
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    
    # Optimization problem
    problem = cp.Problem(objective, constraints)
    
    # Set solver options
    solver_options = {}
    
    # Try solving with solver
    try:
        # Use ECOS or SCS solver for speed
        problem.solve(solver=cp.ECOS, **solver_options, max_iters=1000)
        
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = h_var.value
            c5_sol = C.value
            if c5_sol is not None:
                # Ensure solution respects constraints
                h_solution = np.clip(h_solution, 0, 1)
                # Scale to meet sum constraint
                h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
                
                # Compute actual C5 value and check if it's better
                correlation = np.correlate(h_solution, 1 - h_solution, mode='full')
                actual_c5 = np.max(correlation) * dx
                
                if actual_c5 < best_c5:
                    best_c5 = actual_c5
                    best_h = h_solution.copy()
    except Exception:
        pass
    
    # Refinement using gradient-based optimization if needed
    if time.time() - start_time > budget_s - 10:
        return (best_h.tolist(), best_c5, n_points)
    
    # Simple local gradient-based refinement of best solution
    h = best_h.copy()
    
    # Function to compute C5 from h
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # Function to compute gradient
    def compute_gradient(h_vals, eps=1e-5):
        grad = np.zeros_like(h_vals)
        c5_base = compute_c5(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            # Rescale to sum = m/2
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_plus = np.clip(h_plus, 0, 1)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    # Gradient descent
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        grad = compute_gradient(h)
        lr = 0.005 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new.copy()
        
        # Simple convergence check
        if np.abs(c5_base - c5_new) < 1e-8:
            break
    
    return (best_h.tolist(), best_c5, n_points)
