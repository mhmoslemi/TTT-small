import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h using provided construction or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Try 3 different random initializations to avoid local minima
        h_candidates = []
        for _ in range(3):
            h_init = np.random.rand(n_points)
            h_init = h_init / np.sum(h_init) * (n_points / 2)
            h_candidates.append(h_init)
        
        # Evaluate which one gives the best starting C5
        best_c5 = float('inf')
        best_h = None
        for h_init in h_candidates:
            corr = np.correlate(h_init, 1 - h_init, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_init.copy()
        h = best_h
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Relaxation stage using CVXPY
    if time.time() - start_time < budget_s * 0.4:
        try:
            # Define the relaxed convex problem
            h_var = Variable(n_points)
            C = Variable()
            
            # Constraints
            constraints = [h_var >= 0, h_var <= 1, sum(h_var) == n_points / 2]
            
            # Add overlap constraints
            C_threshold = 0.5  # Initial guess
            # Instead of full correlation constraint, we use a relaxation approach
            
            # Objective (minimize C) - but we'll use a placeholder since it's handled via the convex solver
            prob = Problem(Minimize(C),
                            constraints)
            
            # Attempt solving with a relaxed approach - use more direct method
            # For now, we'll skip solving with full convex optimization
            
        except:
            pass
    
    # Heuristic optimization: guided local search
    while time.time() - start_time < budget_s - 1:
        # Compute C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Local neighborhood search
        improvements = 0
        for _ in range(10):
            if time.time() - start_time > budget_s - 2:
                break
            # Small adaptive perturbation
            h_new = h.copy()
            idx = np.random.randint(0, n_points)
            
            # Random change (in range [0,1], clip if needed)
            delta = np.random.uniform(-0.05, 0.05)
            h_new[idx] += delta
            h_new[idx] = np.clip(h_new[idx], 0, 1)
            
            # Rescale to meet integral constraint
            scale_factor = (n_points / 2) / np.sum(h_new)
            h_new = h_new * scale_factor
            
            # Evaluate new C5
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                improvements += 1
                
        if improvements < 1:  # no improvements, try different strategy
            # Try a more structured change
            # Find a good direction
            h_new = h.copy()
            # Find index with max variance
            idx = np.argmax(h)
            # Reduce it, increase others
            h_new[idx] *= 0.9
            h_new[(idx + 1) % n_points] = np.clip(h_new[(idx + 1) % n_points] + 0.01, 0, 1)
            h_new[(idx - 1) % n_points] = np.clip(h_new[(idx - 1) % n_points] + 0.01, 0, 1)
            
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < c5:
                h = h_new
                c5 = c5_new
    
    # Final polishing with Coarse-to-Fine search
    if time.time() - start_time < budget_s - 1:
        # Convert to binary and enhance
        h_binary = h.copy()
        threshold = np.median(h_binary)
        h_binary = (h_binary > threshold).astype(float)
        h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
        
        # Binary constraints adjustment
        while abs(np.sum(h_binary) - n_points / 2) > 1e-6:
            if np.sum(h_binary) > n_points / 2:
                idx_to_decrease = np.random.choice(np.where(h_binary == 1)[0])
                h_binary[idx_to_decrease] = 0
            else:
                idx_to_increase = np.random.choice(np.where(h_binary == 0)[0])
                h_binary[idx_to_increase] = 1
                
        # Evaluate binary solution
        corr_binary = np.correlate(h_binary, 1 - h_binary, mode='full')
        c5_binary = np.max(corr_binary) * dx
        
        if c5_binary < best_c5:
            best_c5 = c5_binary
            best_h = h_binary
            
    return (best_h.tolist(), best_c5, n_points)
