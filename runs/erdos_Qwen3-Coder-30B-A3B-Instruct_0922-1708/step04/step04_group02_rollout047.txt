import numpy as np
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    # Initial correlation computation
    correlation = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(correlation) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    # Annealing parameters
    temp = 1.0
    cooling_factor = 0.99
    temp_decay_interval = 100
    perturb_ratio = 0.1
    perturb_mag = 0.1
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Adaptive perturbation
        n_perturb = max(1, int(n_points * perturb_ratio))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        h_new = h.copy()
        sum_diff = 0.0
        
        for i in indices:
            old_val = h_new[i]
            # Apply perturbation in [0,1]
            new_val = old_val + np.random.uniform(-perturb_mag, perturb_mag)
            new_val = np.clip(new_val, 0, 1)
            
            sum_diff += (new_val - old_val)
            h_new[i] = new_val
            
        # Redistribute to maintain constraint
        if abs(sum_diff) > 1e-10:
            # Distribute the difference evenly
            redistribute = sum_diff / (n_points - n_perturb)
            for i in range(n_points):
                if i not in indices:
                    h_new[i] -= redistribute
                    h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept with probability
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < math.exp(-delta / temp):
            h = h_new
            current_c5 = new_c5
            
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Temperature update
        if iteration % temp_decay_interval == 0:
            temp *= cooling_factor
            
    # Final refinement around the best solution
    h_refined = best_h.copy()
    best_c5_refined = best_c5
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Local search by small uniform perturbations
        h_test = h_refined.copy()
        i = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.05, 0.05)
        
        h_test[i] += delta
        h_test[i] = np.clip(h_test[i], 0, 1)
        
        # Adjust rest to keep sum
        diff = np.sum(h_test) - n_points / 2
        if abs(diff) > 1e-10:
            redistribute = diff / (n_points - 1)
            for j in range(n_points):
                if i != j:
                    h_test[j] -= redistribute
                    h_test[j] = np.clip(h_test[j], 0, 1)
        
        # Normalize
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Evaluate
        correlation_test = np.correlate(h_test, 1 - h_test, mode='full')
        test_c5 = np.max(correlation_test) * dx
        
        if test_c5 < best_c5_refined:
            h_refined = h_test
            best_c5_refined = test_c5
            
    # Final validation
    final_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
