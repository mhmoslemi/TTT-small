import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Optimization with CVXPY
    # Define the optimization problem using CVXPY
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1]
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Overlap constraints for all shifts k from -n_points+1 to n_points-1, excluding k=0
    for k in range(-n_points+1, n_points):
        if k == 0:
            continue
        # Compute overlap for shift k
        if k > 0:
            # h[i] * (1 - h[i+k]) for i = 0,...,n_points-k-1
            overlap = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
        else:
            # h[i+|k|] * (1 - h[i]) for i = 0,...,n_points+|k|-1
            overlap = cp.sum(cp.multiply(h_var[-k:], 1 - h_var[:-k]))
        constraints.append(overlap <= C)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Try multiple solvers
    solvers = ['ECOS', 'SCS']
    for solver in solvers:
        if time.time() - start_time > budget_s - 1:
            break
        try:
            prob.solve(solver=solver, eps=1e-6, max_iters=5000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    # Compute C5 for this solution
                    correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                    c5 = np.max(correlation) * dx
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_sol.copy()
        except Exception:
            continue
    
    # if we haven't improved, use a gradient-based refinement
    if best_c5 == float('inf'):
        # If no CVXPY solution, fallback to gradient descent with clipped updates
        h = best_h.copy()
        for i in range(3000):  # Limited iterations
            if time.time() - start_time > budget_s - 1:
                break
            # Compute overlap
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Estimate gradients
            grad = np.zeros_like(h)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - c5) / eps
            
            # Update
            lr = 0.0002 / (1 + np.linalg.norm(grad) * 1e-4)
            h_new = h - lr * grad
            
            # Project back
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Update best if better
            correlation = np.correlate(h, 1 - h, mode='full')
            c5_new = np.max(correlation) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
