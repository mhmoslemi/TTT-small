import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    lr = 0.1
    decay = 0.99
    
    while time.time() - start_time < budget_s - 1:
        if iteration % 50 == 0 and iteration > 0:
            lr *= decay
            
        # Compute overlap for all lags
        correlation = np.correlate(h, 1 - h, mode='full')
        max_overlap_val = np.max(correlation) * dx
        max_overlap_k = np.argmax(correlation)
        
        if max_overlap_val < best_c5:
            best_c5 = max_overlap_val
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        epsilon = 1e-4
        
        for i in range(n_points):
            # Increase h[i]
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus) * dx
            
            # Decrease h[i]
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            max_minus = np.max(corr_minus) * dx
            
            # Approximate gradient
            grad[i] = (max_plus - max_minus) / (2 * epsilon)
            
        # Update h
        h = h - lr * grad
        
        # Enforce constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        iteration += 1
        
    return (best_h.tolist(), best_c5, n_points)
