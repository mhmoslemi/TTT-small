import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use CVXPY convex optimization
    # Variables: h (step function values) and C (upper bound)
    h_var = cp.Variable(n_points)
    C_var = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1,
        C_var >= 0
    ]
    
    # Add overlap constraints for all shifts
    # For shift k: sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k]) <= C
    for k in range(1, n_points):
        # Compute correlation for shift k
        shift_sum = 0
        for i in range(n_points - k):
            shift_sum += h_var[i] * (1 - h_var[i + k])
        constraints.append(shift_sum <= C_var)
    
    # Objective
    objective = cp.Minimize(C_var)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with default settings
    try:
        # Set solver parameters to improve performance
        prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-5, reltol=1e-5)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h_solution = h_var.value
            c5_solution = C_var.value
            if c5_solution < best_c5 and h_solution is not None:
                best_c5 = c5_solution
                best_h = h_solution.copy()
    except Exception as e:
        # Fallback to simpler approach if CVXPY fails
        pass
    
    # Final refinement with local search
    h = best_h.copy()
    max_iter = 200
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps

        # Update using gradient descent
        lr = 0.001
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update if better
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Convergence check
        if np.abs(c5 - c5_new) < 1e-10:
            break

    # Post-processing check
    if best_c5 == float('inf'):
        # Fallback to initial solution
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(correlation) * dx

    return (best_h.tolist(), best_c5, n_points)
