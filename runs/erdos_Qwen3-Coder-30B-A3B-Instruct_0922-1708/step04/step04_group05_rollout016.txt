import time

    dx = 2.0 / n_points

            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5

    h = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Initialize with some small random noise
    h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    
    # Gradient-based optimization with projection
    max_iter = 10000
    lr = 0.002
    patience = 0
    min_improvement = 1e-6
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute overlap integral
        overlap_full = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(overlap_full) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Check for early termination
        if patience > 200 and iter_count > 1000:
            break
            
        # Compute gradient approximation via finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()

            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            overlap_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(overlap_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update h using gradient descent with learning rate
        h_new = h - lr * grad
        
        # Project to feasible region [0,1] and maintain integral constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Apply momentum-like update
        h = 0.9 * h + 0.1 * h_new
        
        # Project again
        h = np.clip(h, 0, 1)

        # Reduce learning rate occasionally
        if iter_count % 200 == 0 and lr > 1e-4:

    # Final evaluation of best solution

    final_c5 = np.max(final_overlap) * dx
