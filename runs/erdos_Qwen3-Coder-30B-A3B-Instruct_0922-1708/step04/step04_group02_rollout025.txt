import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_h(h, n_points):
    """Evaluate the C5 bound for a given h function."""
    dx = 2.0 / n_points
    # Compute correlation for all lags
    correlation = np.correlate(h, 1 - h, mode='full')
    # Maximum correlation (C5)
    c5 = np.max(correlation) * dx
    return c5

def initialize_h(n_points):
    """Initialize h with the parent construction if available."""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a reasonably good step function
        h = np.ones(n_points) * 0.5
        # Make it a block-like function to start 
        h[:n_points//3] = 1.0
        h[n_points//3:2*n_points//3] = 0.5
        h[2*n_points//3:] = 0.0
        # Normalize to correct sum
        h = h * (n_points / 2) / np.sum(h)
    return h

def mutate_h(h, mutation_rate):
    """Perform bounded mutation on h values."""
    h_new = h.copy()
    for i in range(len(h_new)):
        if random.random() < mutation_rate:
            # Random change with bounded magnitude
            delta = random.uniform(-0.1, 0.1)
            h_new[i] += delta
    # Ensure bounds and normalization
    h_new = np.clip(h_new, 0, 1)
    h_new = h_new * (len(h_new) / 2) / np.sum(h_new)
    return h_new

def crossover_h(h1, h2):
    """Uniform crossover between two h functions."""
    h_new = np.zeros_like(h1)
    for i in range(len(h1)):
        if random.random() < 0.5:
            h_new[i] = h1[i]
        else:
            h_new[i] = h2[i]
    # Normalize
    h_new = h_new * (len(h_new) / 2) / np.sum(h_new)
    return h_new

def genetic_search(seed, budget_s, n_points, num_generations=500):
    """Multi-resolution genetic search for optimal h function."""
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    population = [initialize_h(n_points) for _ in range(population_size)]
    
    best_h = initialize_h(n_points)
    best_c5 = float('inf')
    
    mutation_rate = 0.1
    max_mutation_rate = 0.3
    
    for gen in range(num_generations):
        if time.time() - start_time > budget_s * 0.9:  # Leave time for final refinement
            break
            
        # Evaluate population
        fitness_scores = [evaluate_h(individual, n_points) for individual in population]
        
        # Update best
        for i, fitness in enumerate(fitness_scores):
            if fitness < best_c5:
                best_c5 = fitness
                best_h = population[i].copy()
        
        # Selection (tournament selection)
        selected = []
        for _ in range(population_size):
            tournament_size = 3
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Create new generation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            child1 = crossover_h(parent1, parent2)
            child2 = crossover_h(parent1, parent2)
            child1 = mutate_h(child1, mutation_rate)
            child2 = mutate_h(child2, mutation_rate)
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        
        # Adaptive mutation rate
        if gen % 10 == 0 and gen > 0:
            mutation_rate = min(max_mutation_rate, mutation_rate * 1.1)
        
        # Occasionally add new diverse individuals
        if gen % 20 == 0:
            for _ in range(5):
                population.append(initialize_h(n_points))
                population.pop(0)  # Remove worst
    

    # Final refinement with gradient-like approach  
    h_refined = best_h.copy()
    c5_refined = best_c5
    
    # Perform local search by small perturbations
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try perturbing each point slightly
        for i in range(len(h_refined)):
            h_test = h_refined.copy()
            delta = random.uniform(-0.05, 0.05)
            h_test[i] += delta
            
            # Clip and rescale 
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (len(h_test) / 2) / np.sum(h_test)
            
            c5_test = evaluate_h(h_test, n_points)
            if c5_test < c5_refined:
                h_refined = h_test
                c5_refined = c5_test
    
    return h_refined, c5_refined

def run(seed=42, budget_s=600, **kwargs):
    # Start with modest resolution 
    n_points = 100
    h, c5 = genetic_search(seed, budget_s, n_points)
    
    return (h.tolist(), c5, n_points)
