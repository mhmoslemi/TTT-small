import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Parameters for gradient-based approach
    learning_rate = 0.1
    epsilon = 1e-4
    max_no_improvement = 100
    
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute correlation for current h
        correlation = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(correlation) * dx
        
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Perturb h[i] upwards
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus) * dx
            
            # Perturb h[i] downwards
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            max_minus = np.max(corr_minus) * dx
            
            # Gradient estimate (forward difference)
            grad[i] = (max_plus - max_minus) / (2 * epsilon)
        
        # Update h
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate decay
        if iteration % 50 == 0:
            learning_rate *= 0.99
            
        # Early stopping
        if current_max < best_c5:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
