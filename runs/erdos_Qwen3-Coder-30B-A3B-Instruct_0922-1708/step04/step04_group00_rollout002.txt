import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # PSO parameters
    n_particles = 30
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    vmax = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals() and np.random.rand() < 0.5:
            # Use parent for some particles
            particle_h = h.copy()
        else:
            particle_h = np.random.uniform(0, 1, n_points)
            particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
            particle_h = np.clip(particle_h, 0, 1)
        particles.append(particle_h)
        velocities.append(np.random.uniform(-vmax, vmax, n_points))
        pbest.append(particle_h.copy())
        pbest_c5.append(float('inf'))
    
    gbest = h.copy()
    gbest_c5 = float('inf')
    
    # PSO iterations
    max_iter = 300
    no_improvement = 0
    max_no_improvement = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Compute fitness
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particle.copy()
                pbest_c5[i] = c5
                
            # Update global best
            if c5 < gbest_c5:
                gbest = particle.copy()
                gbest_c5 = c5
                
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
                
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            vel = (inertia * vel + 
                   c1 * r1 * (pbest[i] - particle) + 
                   c2 * r2 * (gbest - particle))
            
            # Clip velocity
            vel = np.clip(vel, -vmax, vmax)
            
            # Update position
            particle += vel
            
            # Enforce constraints
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            velocities[i] = vel
            particles[i] = particle
            
        # Adaptive inertia
        if gbest_c5 < best_c5:
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > 20:
            inertia = min(0.9, inertia + 0.05)
            
        if no_improvement > max_no_improvement:
            break
    
    # Local refinement with Nelder-Mead on best solution
    h = best_h.copy()
    
    # Construct the objective function for Nelder-Mead optimization
    def objective(h_vals):
        h_local = h_vals.copy()
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * (n_points / 2) / np.sum(h_local)
        correlation = np.correlate(h_local, 1 - h_local, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Fit simplex around current solution
    simplex_points = [h.copy()]
    for i in range(n_points):
        perturbed = h.copy()
        perturbed[i] += 0.01 * (1 - 2 * np.random.rand())
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        simplex_points.append(perturbed)
        
    # Nelder-Mead optimization with explicit constraints
    try:
        res = minimize(
            objective, 
            h, 
            method='Nelder-Mead', 
            options={
                'maxiter': 1000,
                'xtol': 1e-6,
                'ftol': 1e-6
            }
        )
        if res.success:
            final_h = res.x.copy()
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            correlation = np.correlate(final_h, 1 - final_h, mode='full')
            c5_final = np.max(correlation) * dx
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = final_h.copy()
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
