import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Ensure it sums to exactly n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    temp = 1.0
    cooling_rate = 0.99
    temp_decay_interval = 100
    
    # Main optimization loop
    iterations = 0
    while time.time() - start_time < budget_s - 5:
        iterations += 1

        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Generate candidate solution
        h_new = h.copy()
        
        # Select around 10% of points to perturb
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        # Perturb each selected point
        for i in indices:
            # Random perturbation within [0,1]
            perturbation = np.random.uniform(-0.1, 0.1)
            h_new[i] += perturbation
            
        # Ensure bounds and constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute candidate C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        candidate_c5 = np.max(correlation_new) * dx

        # Accept with simulated annealing probability
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            h = h_new
            
        # Adjust temperature
        if iterations % temp_decay_interval == 0:
            temp *= cooling_rate
            
    # Final refinement
    final_h = best_h.copy()
    final_c5 = best_c5
    
    # Try a few more refinements
    for _ in range(500):
        h_test = final_h.copy()
        
        # Random small perturbation
        idx = np.random.randint(0, n_points)
        perturbation = np.random.uniform(-0.02, 0.02)
        h_test[idx] += perturbation
        
        # Adjust to maintain constraints
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        # Compute and evaluate
        corr_test = np.correlate(h_test, 1 - h_test, mode='full')
        c5_test = np.max(corr_test) * dx
        
        if c5_test < final_c5:
            final_h = h_test
            final_c5 = c5_test
            
    return (final_h.tolist(), final_c5, n_points)
