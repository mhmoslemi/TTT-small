import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to satisfy integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    # Generate diverse starting points
    candidates = [h_init.copy()]
    for i in range(19):
        h_pert = h_init + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        candidates.append(h_pert)
    
    # Try many random starting configs
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h_random = np.random.rand(n_points)
        h_random = h_random * (n_points / 2) / np.sum(h_random)
        candidates.append(h_random)
    
    # Main optimization loop
    iter_count = 0
    stagnant = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and stagnant < 100:
        iter_count += 1
        if iter_count % 5 == 0:
            # Periodically check for best solution
            corr_best = np.correlate(best_h, 1 - best_h, mode='full')
            c5_best = np.max(corr_best) * dx
            if c5_best < best_c5:
                best_c5 = c5_best
                last_improvement = iter_count
        
        # Select a random candidate
        idx = np.random.randint(0, len(candidates))
        h_current = candidates[idx].copy()
        
        # Compute gradient steepest ascent
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr_current = np.correlate(h_current, 1 - h_current, mode='full')
            c5_current = np.max(corr_current) * dx
            
            grad[i] = (c5_plus - c5_current) / eps
        
        # Gradient step
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-2)
        h_new = h_current - lr * grad
        
        # Project to [0,1] and integral constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Add to candidates
        candidates.append(h_new)
        
        # Quick evaluation and update
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stagnant = 0
        else:
            stagnant += 1
            
        # Occasionally add new random candidate
        if np.random.rand() < 0.05:
            h_random = np.random.rand(n_points)
            h_random = h_random * (n_points / 2) / np.sum(h_random)
            candidates.append(h_random)
    
    # Final optimization
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    return (best_h.tolist(), final_c5, n_points)
