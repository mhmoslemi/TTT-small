import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    start_time = time.time()
    
    # Set up optimization problem using CVXPY
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1,
    ]
    
    # Add overlap constraints
    for k in range(1, n_points):
        # Compute the correlation for shift k
        # h[i] * (1 - h[i+k]) for valid i
        expr = 0
        for i in range(n_points - k):
            expr += h_var[i] * (1 - h_var[i+k])
        constraints.append(expr <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-5, reltol=1e-5)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_result = h_var.value
            c5_result = C.value
        else:
            h_result = h
            c5_result = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    except:
        # Fallback to simple optimization if CVXPY fails
        h_result = h
        c5_result = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    # Ensure feasibility
    h_result = np.clip(h_result, 0, 1)
    h_result = h_result * (n_points / 2) / np.sum(h_result)
    
    # Evaluate final solution
    correlation = np.correlate(h_result, 1 - h_result, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (h_result.tolist(), final_c5, n_points)
