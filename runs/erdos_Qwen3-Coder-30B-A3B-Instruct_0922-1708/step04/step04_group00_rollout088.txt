import numpy as np
from cvxpy import *
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Try to solve LP relaxation
    try:
        # CVXPY variables
        h_vars = Variable(n_points)
        C5 = Variable()
        
        # Objective
        objective = Minimize(C5)
        
        # Constraints
        constraints = [0 <= h_vars, h_vars <= 1]
        constraints.append(sum(h_vars) == n_points / 2)
        
        # For each shift k, we have a constraint on C5 via correlation
        # We can use auxiliary variables to make this linear in the relaxation
        # But for simplicity in LP, we'll handle it differently
        
        # In LP relaxation, approximate the non-linear terms
        # Create auxiliary variables for terms `h[i] * h[i+k]` 
        # This is a complex constraint - instead, start with good heuristics
        
        # Let's try the following: use finite differences to set up gradient descent
        # but first solve LP to get a good starting point
        h_lp = h.copy()
        best_h = h.copy()
        best_c5 = float('inf')
        
        # Simple heuristic: smear h to get lower overlap
        h_try = np.ones(n_points) * 0.5
        h_try[:n_points//2] = 1.0
        h_try[n_points//2:] = 0.0
        h_try = h_try * (n_points / 2) / np.sum(h_try)
        h_try = np.clip(h_try, 0, 1)
        
        correlation = np.correlate(h_try, 1 - h_try, mode='full')
        c5_try = np.max(correlation) * dx
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            
        # Also try another simple construction
        h_try2 = np.ones(n_points) * 0.5
        h_try2[10:90] = 1.0  # increasing ramp
        h_try2 = h_try2 * (n_points / 2) / np.sum(h_try2)
        h_try2 = np.clip(h_try2, 0, 1)
        
        correlation = np.correlate(h_try2, 1 - h_try2, mode='full')
        c5_try2 = np.max(correlation) * dx
        if c5_try2 < best_c5:
            best_c5 = c5_try2
            best_h = h_try2.copy()
        
        # Generative sampling
        h_try3 = np.random.uniform(0, 1, n_points)
        h_try3 = h_try3 * (n_points / 2) / np.sum(h_try3)
        h_try3 = np.clip(h_try3, 0, 1)
        
        correlation = np.correlate(h_try3, 1 - h_try3, mode='full')
        c5_try3 = np.max(correlation) * dx
        if c5_try3 < best_c5:
            best_c5 = c5_try3
            best_h = h_try3.copy()
            
        h = best_h.copy()
        
    except:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        best_c5 = float('inf')
    
    # Gradient descent refinement 
    max_iter = 10000
    lr = 0.01
    last_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Recompute C5 and correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if abs(c5 - last_c5) < 1e-8:
            break
            
        last_c5 = c5
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update step with projection back to feasible domain
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update learning rate
        lr = max(1e-6, lr * 0.9999)
        h = h_new
        
        # Early termination if target is met
        if best_c5 <= 0.3807:
            break
    
    # Final refinement
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
