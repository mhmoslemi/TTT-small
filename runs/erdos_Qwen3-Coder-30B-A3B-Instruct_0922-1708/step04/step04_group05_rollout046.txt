import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5

    # Ensure integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate diverse candidates
    candidates = []
    for i in range(20):
        h = h_init.copy()
        # Add different perturbations
        noise = np.random.normal(0, 0.01, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h.copy())
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    # Local refinement with gradient descent
    max_iter = 5000
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current population
        for h in candidates:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Gradient descent for each candidate
        new_candidates = []
        for h in candidates:
            # Compute gradient approximation
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                grad[i] = (c5_plus - c5) / eps
            
            # SGD update
            lr = 0.01 / (1 + np.linalg.norm(grad) * 0.01)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_candidates.append(h_new)
        
        candidates = new_candidates
        
        # Add new diversifying candidates periodically
        if iter_count % 50 == 0:
            for _ in range(5):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                candidates.append(h)
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
