import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Define objective function
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Gradient computation via finite differences
    def compute_gradient(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        base_obj = objective(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            grad[i] = (objective(h_plus) - base_obj) / eps
        return grad
    
    # Trust region optimization with TRF method
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current solution
        current_c5 = objective(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            
        # Compute gradient at current point
        grad = compute_gradient(best_h)
        
        # Trust region method with bounds
        try:
            # Optimization step
            result = minimize(
                objective, 
                best_h, 
                method='TRF',
                bounds=[(0, 1)] * n_points,
                options={'maxiter': 10, 'initial_trust_radius': 0.1}
            )
            if result.success:
                new_h = result.x
                # Re-project to satisfy sum constraint
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                new_h = np.clip(new_h, 0, 1)
                best_h = new_h
        except:
            # Fall back to gradient descent
            lr = 0.01
            new_h = best_h - lr * grad
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            best_h = new_h
    
    # Final computation of best solution
    final_c5 = objective(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
