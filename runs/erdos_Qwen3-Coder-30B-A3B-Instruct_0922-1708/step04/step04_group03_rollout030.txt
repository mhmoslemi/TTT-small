import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation
    # We minimize sum of all correlations (not max), which is a convex relaxation
    # This gives us a better starting point
    
    # Create the convex optimization problem
    h_var = cp.Variable(n_points)
    
    # Objective: minimize sum of correlations (relaxed version)
    # We compute sum of all terms (not just max) 
    # to get a convex problem  
    objective = cp.Minimize(cp.sum_entries(h_var) * (n_points / 2) - 0.5 * cp.sum_entries(h_var * h_var))
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum_entries(h_var) == n_points / 2
    ]
    
    # Solve relaxation
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = h_var.value
            # Ensure it's within bounds
            h = np.clip(h, 0, 1)
    except:
        pass  # Fall back to previous initialization if Convex relaxation fails
    
    # Phase 2: Refinement with projected gradient descent 
    max_iter = 1000
    lr = 0.01
    patience = 50
    no_improvement_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute the current correlation (add small epsilon for numerical stability)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update step
        h_new = h - lr * grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check convergence
        if np.linalg.norm(h_new - h) < 1e-6:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        h = h_new
        
        # Reduce learning rate occasionally
        if iteration % 100 == 0 and iteration > 0:
            lr *= 0.9
            
        iteration += 1
        
        if no_improvement_count >= patience:
            break
            
    # Return final best solution
    return (best_h.tolist(), best_c5, n_points)
