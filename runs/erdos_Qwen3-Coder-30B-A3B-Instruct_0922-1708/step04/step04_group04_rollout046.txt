import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    total_sum = n_points / 2
    
    # Initialize h from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to correct sum
    h = h * total_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Try multiple random restarts to escape local optima
    n_restarts = 20
    
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Start with current best or random perturbation
        if restart == 0:
            h_current = best_h.copy()
        else:
            # Add noise for diversity
            h_current = best_h + np.random.normal(0, 0.02, n_points)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * total_sum / np.sum(h_current)
            
        # Convex optimization approach
        # Variables
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == total_sum, C >= 0]
        
        # For each possible shift, add constraint
        # Shifts: -n_points+1 to n_points-1 (excluding 0)
        for k in range(-n_points+1, n_points):
            if k == 0:
                continue
            # Compute the correlation part: sum h[i] * (1 - h[i+k])
            # This is the overlap at shift k
            shifted_h = cp.vstack([h_var[-k:], h_var[:-k]]) if k > 0 else cp.vstack([h_var[-k:], h_var[:-k]])
            constraint_expr = cp.sum(cp.multiply(h_var, 1-shifted_h)) <= C
            constraints.append(constraint_expr)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, feastol=1e-5)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h_var.value
                # Discretize the solution to be actual step function
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * total_sum / np.sum(h_opt)
                
                # Evaluate C5
                correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
                c5_val = np.max(correlation) * dx
                
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
        except:
            pass  # If optimization fails, continue
        
        # Simple gradient descent refinement
        h_refine = h_current.copy()
        lr = 0.001
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute overlap
            correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
            c5_refine = np.max(correlation) * dx
            
            # Compute gradient via finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * total_sum / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5_refine) / eps
            
            # Update
            h_new = h_refine - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * total_sum / np.sum(h_new)
            
            h_refine = h_new
            
            # Update best
            correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
            c5_new = np.max(correlation) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_refine.copy()
                
            # Early stopping
            if np.abs(c5_refine - c5_new) < 1e-10:
                break
                
    return (best_h.tolist(), best_c5, n_points)
