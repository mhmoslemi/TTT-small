import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a block function
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    start_time = time.time()
    
    # Early stopping criteria
    last_improvement = 0
    patience = 50
    
    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h.copy() + np.random.normal(0, 0.01, n_points)
        else:
            particle = np.random.rand(n_points) * 2
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.random.rand(n_points) * 0.1)
        pbest.append(particle.copy())
        pbest_vals.append(evaluate_overlap(particle, dx))
    
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update particles
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            vel = (w * vel + 
                   c1 * r1 * (pbest[i] - particle) + 
                   c2 * r2 * (gbest - particle))
            
            # Update position
            particle += vel
            
            # Clip and rescale
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Store updated particle
            particles[i] = particle
            
            # Evaluate fitness
            fitness = evaluate_overlap(particle, dx)
            
            # Update personal best
            if fitness < pbest_vals[i]:
                pbest[i] = particle.copy()
                pbest_vals[i] = fitness
                
                # Update global best
                if fitness < gbest_val:
                    gbest = particle.copy()
                    gbest_val = fitness
                    last_improvement = iteration
        
        # Early stopping
        if iteration - last_improvement > patience:
            break
    
    # Final refinement using CVXPY
    try:
        h_vars = cp.Variable(n_points)
        
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Directly compute correlation through a quadratic form
        # This is more complex, so we approximate the convex constrained optimization
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        
        # Try to solve with different solvers
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except:
                pass
                
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                c5_cvx = evaluate_overlap(h_cvx, dx)
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass
    
    # Final check
    final_c5 = evaluate_overlap(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
