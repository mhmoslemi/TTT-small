import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Differential Evolution parameters
    population_size = 50
    F = 0.8
    CR = 0.9
    max_no_improvement = 100
    
    # Initialize population
    population = []
    fitness = []
    
    # Start with initial solution
    population.append(h.copy())
    corr = np.correlate(h, 1 - h, mode='full')
    fitness.append(np.max(corr) * dx)
    
    # Fill rest with random feasible solutions
    for _ in range(population_size - 1):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
        corr = np.correlate(h_rand, 1 - h_rand, mode='full')
        fitness.append(np.max(corr) * dx)
    
    # Find initial best
    best_idx = np.argmin(fitness)
    best_f = fitness[best_idx]
    if best_f < best_c5:
        best_c5 = best_f
        best_h = population[best_idx].copy()
    
    no_improvement_count = 0
    
    # DE optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three different individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (n_points / 2) / np.sum(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            
            # Evaluation
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_c5 = np.max(corr) * dx
            
            # Selection
            if trial_c5 < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_c5)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
        
        # Update population and fitness
        population = new_population
        fitness = new_fitness
        
        # Update best
        current_best_idx = np.argmin(fitness)
        current_best_val = fitness[current_best_idx]
        if current_best_val < best_c5:
            best_c5 = current_best_val
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Local refinement on best every 20 steps
        if iteration % 20 == 0 and time.time() - start_time < budget_s - 1:
            h_refine = best_h.copy()
            for refine_iter in range(100):
                # Compute gradient using finite differences
                grad = np.zeros_like(h_refine)
                eps = 1e-5
                for j in range(n_points):
                    h_plus = h_refine.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - best_c5) / eps
                
                # Adaptive learning rate
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h_refine - lr * grad
                
                # Re-enforce constraints
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate
                corr = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr) * dx
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_refine = h_new.copy()
                else:
                    break
                    
    return (best_h.tolist(), best_c5, n_points)
