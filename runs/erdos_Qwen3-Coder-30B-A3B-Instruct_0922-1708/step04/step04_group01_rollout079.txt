import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                h_orig = np.array(initial_h_values)
                x_orig = np.linspace(0, 2, len(h_orig))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h_orig)
            h = h / np.sum(h) * (n_points / 2)
        else:
            h = np.zeros(n_points)
            h[:n_points//2] = 2.0
        
        # Add some random perturbation
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_c5.append(float('inf'))
    
    gbest = particles[0].copy()
    gbest_c5 = float('inf')
    
    # PSO Loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        for i in range(n_particles):
            h = particles[i]
            # Compute fitness
            c5 = compute_c5(h, dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = h.copy()
                pbest_c5[i] = c5
            
            # Update global best
            if c5 < gbest_c5:
                gbest = h.copy()
                gbest_c5 = c5
                
        # Update global best if needed
        if gbest_c5 < best_c5:
            best_c5 = gbest_c5
            best_h = gbest.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            v = w * v + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)
            
            # Limit velocity
            max_vel = 0.5
            v = np.clip(v, -max_vel, max_vel)
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            velocities[i] = v
    
    # Try CVX refinement on the best solution found
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Reconstruct the C5 maximization problem in convex form
        # Since the original problem is non-convex we attempt to solve a related convex relaxation
        
        # Initialize with best solution
        h_cvx = best_h.copy()
        h_cvx = np.clip(h_cvx, 0, 1)
        h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
        
        # Refinement: try to optimize the fit after PSO
        # Use simpler optimization for final touch
        def obj_func(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return -np.max(corr) * dx
        
        # Minimize -C5 (equivalent to maximizing C5)
        result = minimize(obj_func, h_cvx, method='L-BFGS-B', bounds=[(0,1)]*n_points, 
                          options={'maxiter': 500})
        
        if result.success:
            h_cvx = np.clip(result.x, 0, 1)
            h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
            corr = np.correlate(h_cvx, 1 - h_cvx, mode='full')
            c5_cvx = -np.max(corr) * dx
            
            if c5_cvx < best_c5:
                best_h = h_cvx
                best_c5 = c5_cvx
                
    except Exception as e:
        pass

    # Final computation of the best C5
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
