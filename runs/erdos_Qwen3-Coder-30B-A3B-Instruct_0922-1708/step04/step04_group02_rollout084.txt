import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Relaxation stage using CVXPY
    if time.time() - start_time < budget_s * 0.3:
        try:
            # Define the relaxed problem
            h_vars = cp.Variable(n_points)
            h_relaxed = np.zeros(n_points)
            
            # Constraints
            constraints = [h_vars[i] >= 0 for i in range(n_points)] + \
                         [h_vars[i] <= 1 for i in range(n_points)] + \
                         [cp.sum(h_vars) == n_points / 2]
            
            # Objective
            C = cp.Variable()
            
            # Add C5 constraint for all lags
            for k in range(1, n_points):
                lag_sum = 0
                for i in range(n_points - k):
                    lag_sum += h_vars[i] * (1 - h_vars[i + k])
                constraints.append(lag_sum <= C * dx**(-1))
            
            # Objective: minimize C
            objective = cp.Minimize(C)
            problem = cp.Problem(objective, constraints)
            
            # Solve with CVXOPT
            problem.solve(solver=cp.CVXOPT, verbose=False)
            
            if problem.status == cp.OPTIMAL:
                h_relaxed = h_vars.value
                c5_relaxed = compute_c5(h_relaxed)
                
                if c5_relaxed < best_c5:
                    best_c5 = c5_relaxed
                    best_h = h_relaxed.copy()
        except:
            pass
    
    # Integer rounding stage
    if time.time() - start_time < budget_s * 0.6 and best_c5 >= 0.3805: 
        # Thresholding approach
        threshold = np.median(best_h) if len(best_h) > 0 else 0.5
        h_binary = (best_h > threshold).astype(float)
        h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
        
        if np.sum(h_binary) < n_points / 2:
            # Need to add more ones
            diff = int(n_points / 2 - np.sum(h_binary))
            indices = np.argsort(best_h)[::-1][:diff]
            for i in indices:
                if h_binary[i] == 0:
                    h_binary[i] = 1
        elif np.sum(h_binary) > n_points / 2:
            # Need to remove ones
            diff = int(np.sum(h_binary) - n_points / 2)
            indices = np.argsort(best_h)[:diff]
            for i in indices:
                if h_binary[i] == 1:
                    h_binary[i] = 0
        
        h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
        c5_binary = compute_c5(h_binary)
        
        if c5_binary < best_c5:
            best_c5 = c5_binary
            best_h = h_binary.copy()
    
    # Local search to improve
    if time.time() - start_time < budget_s - 1 and best_c5 >= 0.3805:
        h_local = best_h.copy()
        for _ in range(200):
            if time.time() - start_time > budget_s - 0.5:
                break
                
            # Try small random adjustments
            idx = np.random.randint(0, n_points)
            eps = np.random.uniform(-0.02, 0.02)
            
            h_test = h_local.copy()
            if h_test[idx] + eps >= 0 and h_test[idx] + eps <= 1:
                h_test[idx] += eps
                
                # Maintain normalization
                sum_diff = np.sum(h_test) - n_points / 2
                if np.abs(sum_diff) > 1e-8:
                    indices = [i for i in range(n_points) if i != idx]
                    diff = sum_diff / len(indices)
                    h_test[indices] -= diff
                    h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                
                c5_test = compute_c5(h_test)
                if c5_test < best_c5:
                    h_local = h_test
                    best_c5 = c5_test
        
        best_h = h_local
    
    # Final evaluation
    c5_final = compute_c5(best_h)
    
    return (best_h.tolist(), c5_final, n_points)
