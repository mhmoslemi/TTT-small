import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a default construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Default construction
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Relaxation stage
    if time.time() - start_time < budget_s * 0.4:
        # Define the relaxed optimization problem
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Impose constraint on correlations
        for i in range(1, n_points):
            corr_constraint = cp.sum(cp.multiply(h_vars[:n_points-i], 
                                                1-h_vars[i:])) <= C * dx**(-1) 
            constraints.append(corr_constraint)
        
        # Objective
        objective = cp.Minimize(C)
        
        # Solve through CVXPY
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                relaxed_h = h_vars.value
                h = np.clip(relaxed_h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            pass
    
    # Integer rounding stage
    if time.time() - start_time < budget_s * 0.6:
        # Using median-based rounding
        threshold = np.median(h)
        h_binary = (h > threshold).astype(float)
        
        # Adjust to match constraint
        current_sum = np.sum(h_binary)
        needed = n_points / 2 - current_sum
        
        if needed > 0:
            # Set lowest values to 1
            indices = np.argsort(h)
            for i in range(int(needed)):
                h_binary[indices[i]] = 1.0
        elif needed < 0:
            # Set highest values to 0
            indices = np.argsort(h)[::-1]
            for i in range(int(-needed)):
                h_binary[indices[i]] = 0.0
                
        h = h_binary * (n_points / 2) / np.sum(h_binary)
        
    # Post-processing phase
    best_c5 = evaluate_solution(h, dx)
    best_h = h.copy()
    
    # Neighborhood search variation
    if time.time() - start_time < budget_s * 0.9:
        for _ in range(100):
            h_test = h.copy()
            # Generate small variations
            for i in range(n_points):
                if np.random.random() < 0.1:  # 10% chance of change
                    change = np.random.uniform(-0.05, 0.05)
                    h_test[i] = np.clip(h_test[i] + change, 0, 1)
            
            # Re-normalize
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            c5_test = evaluate_solution(h_test, dx)
            
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_solution(h, dx):
    # Create the convolution-like overlap integral
    corr = correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
