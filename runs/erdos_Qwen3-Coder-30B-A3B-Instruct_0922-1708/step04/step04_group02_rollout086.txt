import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 100
    max_iterations = 1000
    inertia_weight = 0.7
    cognitive_param = 2.0
    social_param = 2.0
    rejuvenation_rate = 0.1
    local_search_freq = 50
    resolution_steps = [50, 100, 200]
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
    else:
        # Use block construction as default
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
    
    # Add random particles to fill swarm
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    
    # Initialize velocities
    for _ in range(swarm_size):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize personal bests
    pbest = [p.copy() for p in particles]
    pbest_fitness = []
    for h in particles:
        corr = np.correlate(h, 1 - h, mode='full')
        fitness = np.max(corr) * dx
        pbest_fitness.append(fitness)
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    start_time = time.time()
    
    # PSO Main Loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Update particles
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (inertia_weight * v + 
                 cognitive_param * r1 * (pbest[i] - h) + 
                 social_param * r2 * (gbest - h))
            
            # Update position
            h = h + v
            
            # Repair constraints
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            target_sum = n_points / 2
            
            if abs(current_sum - target_sum) > 1e-6:
                diff = current_sum - target_sum
                
                # Adjust values to maintain sum constraint
                if diff > 0:
                    sorted_indices = np.argsort(h)[::-1]  # Descending order
                    for idx in sorted_indices:
                        if diff <= 0:
                            break
                        available = h[idx]
                        reduce = min(diff, available)
                        h[idx] -= reduce
                        diff -= reduce
                else:
                    sorted_indices = np.argsort(h)  # Ascending order
                    for idx in sorted_indices:
                        if diff >= 0:
                            break
                        available = 1.0 - h[idx]
                        increase = min(-diff, available)
                        h[idx] += increase
                        diff += increase
            
            # Normalize to target sum
            h = h * target_sum / np.sum(h)
            
            particles[i] = h.copy()
            velocities[i] = v.copy()
            
            # Evaluate fitness
            corr = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = h.copy()
                pbest_fitness[i] = fitness
                
                # Check if this is new global best
                if fitness < gbest_fitness:
                    gbest = h.copy()
                    gbest_fitness = fitness
        
        # Local search every `local_search_freq` iterations
        if iteration % local_search_freq == 0:
            # Refine top 10% particles
            top_indices = np.argsort(pbest_fitness)[:int(swarm_size * 0.1)]
            for idx in top_indices:
                h = pbest[idx].copy()
                # Gradient-based refinement
                for _ in range(10):
                    # Estimate gradient via finite differences
                    grad = np.zeros_like(h)
                    eps = 1e-4
                    base_corr = np.correlate(h, 1 - h, mode='full')
                    base_fitness = np.max(base_corr) * dx
                    
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        fitness_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (fitness_plus - base_fitness) / eps
                    
                    # Update
                    h -= 0.001 * grad
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                
                # Check final refinement
                corr = np.correlate(h, 1 - h, mode='full')
                fitness = np.max(corr) * dx
                
                if fitness < pbest_fitness[idx]:
                    pbest[idx] = h.copy()
                    pbest_fitness[idx] = fitness
                    
                    if fitness < gbest_fitness:
                        gbest = h.copy()
                        gbest_fitness = fitness
        
        # Rejuvenation
        if iteration % (local_search_freq * 2) == 0:
            num_rejuvenate = int(swarm_size * rejuvenation_rate)
            worst_indices = np.argsort(pbest_fitness)[-num_rejuvenate:]
            for idx in worst_indices:
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[idx] = h.copy()
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                # Update personal best
                corr = np.correlate(h, 1 - h, mode='full')
                fitness = np.max(corr) * dx
                pbest[idx] = h.copy()
                pbest_fitness[idx] = fitness
                
                if fitness < gbest_fitness:
                    gbest = h.copy()
                    gbest_fitness = fitness
    
    # Final refinement using intensive local search
    top_candidates = []
    top_fitness = []
    
    # Take best particles for intensive refinement
    sorted_indices = np.argsort(pbest_fitness)
    for i in range(min(5, len(sorted_indices))):
        top_candidates.append(pbest[sorted_indices[i]].copy())
        top_fitness.append(pbest_fitness[sorted_indices[i]])
    
    # Intensive gradient descent on top candidates
    best_final_h = gbest.copy()
    best_final_c5 = gbest_fitness
    
    for i, h in enumerate(top_candidates):
        # Gradient descent
        h_refined = h.copy()
        for _ in range(500):
            # Estimate gradient with finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            base_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            base_fitness = np.max(base_corr) * dx
            
            improved = False
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                
                grad[j] = (fitness_plus - base_fitness) / eps
                
                # Check improvement
                if fitness_plus < base_fitness:
                    improved = True
                    
            if not improved:
                break
                
            # Update
            h_refined -= 0.001 * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            new_fitness = np.max(corr) * dx
            
            if new_fitness < base_fitness:
                base_fitness = new_fitness
            else:
                break
        
        corr_final = np.correlate(h_refined, 1 - h_refined, mode='full')
        final_fitness = np.max(corr_final) * dx
        
        if final_fitness < best_final_c5:
            best_final_c5 = final_fitness
            best_final_h = h_refined.copy()
    
    return (best_final_h.tolist(), best_final_c5, n_points)
