import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure constraint ∫ h dx = 1 (i.e., sum(h) * dx = 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Main optimization loop
    max_iter = 100
    step_size = 0.5
    tol = 1e-5
    solver_opts = {"eps": 1e-4, "max_iters": 1000}
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute maximum overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Construct convex subproblem
        h_old = h.copy()
        
        # Variables
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        constraints = [cp.sum(h_var) == n_points / 2]
        constraints += [0 <= h_var[i] for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        
        # Linearize the constraint terms
        # For each shift k, we have sum h[i] * (1 - h[i+k]) <= C
        # Using first-order Taylor approx: h[i+k] ≈ h_old[i+k]
        # So constraint becomes sum h[i] * (1 - h_old[i+k]) <= C
        for k in range(1, n_points):
            lhs = cp.sum(cp.multiply(h_var, 1 - h_old[k:]))
            rhs = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_old[:n_points-k]))
            constraints.append(lhs <= C)
            constraints.append(rhs <= C)
            
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, **solver_opts)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_var.value
            else:
                # Fallback to gradient descent if optimization fails
                h_new = h.copy()
        except Exception:
            # Fallback to gradient descent
            h_new = h.copy()
        
        # Line search: backtracking
        alpha = step_size
        for _ in range(20):  # Max line search steps
            h_candidate = h - alpha * (h - h_new)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(correlation) * dx
            
            if c5_candidate <= current_c5 or alpha < 1e-8:
                break
            alpha *= 0.5
            
        h = h_candidate
        
        # Check for convergence
        if np.linalg.norm(h - h_old) < tol:
            break
        
    # Final validation and refinement
    correlation = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
