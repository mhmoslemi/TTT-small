import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0

    best_h = h.copy()
    best_c5 = float('inf')

    # Coarse grid search
    coarse_size = 20
    coarse_h_list = []
    for _ in range(20):
        h_coarse = np.random.rand(coarse_size)
        h_coarse = h_coarse / np.sum(h_coarse) * (coarse_size / 2)
        # Interpolate to full resolution
        h_full = np.interp(np.linspace(0, 2, n_points), 
                          np.linspace(0, 2, coarse_size), h_coarse)
        coarse_h_list.append(h_full)
    
    # Add the initial solution
    coarse_h_list.append(h.copy())
    
    # Evaluate coarse candidates
    for h_candidate in coarse_h_list:
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5 = np.max(correlation) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()

    # Fine refinement with constrained optimization
    iterations = 0
    max_iterations = 500
    
    while time.time() - start_time < budget_s - 5 and iterations < max_iterations:
        # Mini-batch gradient descent refinement
        h_refined = best_h.copy()
        lr = 0.01
        
        # Compute gradient
        grad = np.zeros_like(h_refined)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / np.sum(h_plus) * (n_points / 2)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            h_minus = h_refined.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus / np.sum(h_minus) * (n_points / 2)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)

        # Update with gradient
        h_new = h_refined - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate new solution
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        iterations += 1
    
    # CVX optimization at the end
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        # Objective: Maximize the negative of correlation for minimization
        corr = cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))
        objective = cp.Minimize(cp.max(corr))
        
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL and h_vars.value is not None:
            h_cvx = np.clip(h_vars.value, 0, 1)
            h_cvx = h_cvx / np.sum(h_cvx) * (n_points / 2)
            
            correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
            c5_cvx = np.max(correlation) * dx
            
            if c5_cvx < best_c5:
                best_h = h_cvx
                best_c5 = c5_cvx
    except:
        pass
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
