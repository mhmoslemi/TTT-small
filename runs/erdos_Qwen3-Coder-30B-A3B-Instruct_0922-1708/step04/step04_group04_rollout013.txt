import numpy as np
import time
from scipy.signal import correlate
import heapq

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters
    w = 5  # Window width for local adjustments
    perturbation_scale = 0.02
    improvement_threshold = 1e-6
    max_no_improvement = 20
    max_iterations = 1000
    
    # Helper function to compute overlap
    def compute_overlaps(h):
        # Compute all overlaps via correlation
        corr = correlate(h, 1 - h, mode='full')
        # Slice to get valid shifts [-(n-1), n-1] excluding 0
        overlaps = []
        for k in range(-(n_points-1), n_points):
            if k != 0:
                overlaps.append((corr[n_points-1+k] * dx, k))
        return overlaps
    
    # Helper function to apply adjustments
    def apply_adjustment(h, k, window_width=5):
        h_new = h.copy()
        # Determine affected indices
        if k > 0:
            # h[i] * (1 - h[i+k]) for i in range(n-k)
            start_eff = 0
            end_eff = n_points - k
        elif k < 0:
            # h[i+k] * (1 - h[i]) for i in range(-k, n)
            start_eff = -k
            end_eff = n_points
        else:
            start_eff = 0
            end_eff = n_points
            
        # Create adjustment array
        adjustment = np.zeros_like(h_new)
        
        # Local adjustment: reduce overlap at the highest overlap regions
        # This is a simplified approach, value approximation may be refined
        for i in range(start_eff, end_eff):
            if k > 0:
                # h[i] * (1 - h[i+k])
                # We want to reduce this contribution
                # So we ideally decrease h[i] and increase h[i+k] if the overlap is problematic
                if i < n_points - 1 and i + k >= 0:
                    if h[i] > 0.5 and h[i+k] < 0.5:
                        adjustment[i] -= 0.002
                        adjustment[i+k] += 0.002
                    elif h[i] < 0.5 and h[i+k] > 0.5:
                        adjustment[i] += 0.002
                        adjustment[i+k] -= 0.002
                        
        # Apply adjustment
        h_new += adjustment 
        # Project back to feasible space: [0,1] and sum=h_sum
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Random perturb 
    def random_perturb(h, window_width=5, scale=0.02):
        h_new = h.copy()
        # Randomly perturb a few locations
        num_perts = max(1, n_points // 20)
        indices = np.random.choice(n_points, num_perts, replace=False)
        for i in indices:
            delta = np.random.normal(0, scale)
            h_new[i] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Initialize
    overlaps = compute_overlaps(h)
    heap = [(-overlap, k) for overlap, k in overlaps]  # Negative for max-heap mimic
    heapq.heapify(heap)
    
    no_improvement_count = 0
    for iter_count in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Reassess overlaps every 50 iterations
        if iter_count % 50 == 0:
            overlaps = compute_overlaps(h)
            heap = [(-overlap, k) for overlap, k in overlaps]
            heapq.heapify(heap)
        
        if not heap:
            continue
            
        # Get k with max overlap
        max_overlap, k = heapq.heappop(heap)
        max_overlap = -max_overlap  # Convert back
        
        # Apply adjustments to h
        h = apply_adjustment(h, k, window_width=w)
        
        # Perturb a bit randomly around problem shift
        h = random_perturb(h, window_width=w, scale=perturbation_scale)
        
        # Evaluate
        overlaps = compute_overlaps(h)
        c5 = max(overlap for overlap, k in overlaps)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Increase perturbation scale  
            perturbation_scale *= 1.2
            no_improvement_count = 0
        
        # Rebuild heap
        heap = [(-overlap, k) for overlap, k in overlaps] 
        heapq.heapify(heap)
    
    # Final refinement
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        # Small perturbation and evaluate
        h_perturb = random_perturb(h, window_width=w, scale=0.005)
        corr = correlate(h_perturb, 1 - h_perturb, mode='full')
        c5_perturb = np.max(corr) * dx
        if c5_perturb < best_c5:
            best_c5 = c5_perturb
            best_h = h_perturb.copy()
        h = h_perturb
    
    return (best_h.tolist(), best_c5, n_points)
