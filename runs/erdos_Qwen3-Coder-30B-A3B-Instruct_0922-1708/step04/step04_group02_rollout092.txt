import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    # Define number of segments for sparse representation
    n_segments = 10
    
    def get_h_from_segments(segments):
        # segments: [pos1, pos2, ..., pos_{n_segments-1}, h1, h2, ..., h_{n_segments}]
        pos = np.zeros(n_segments + 1)
        pos[0] = 0
        pos[1:-1] = segments[:n_segments-1]
        pos[-1] = 2.0
        h_vals = segments[n_segments-1:]
        # Ensure bounds
        pos = np.clip(pos, 0, 2)
        h_vals = np.clip(h_vals, 0, 1)
        
        # Enforce increasing positions
        for i in range(1, n_segments):
            pos[i] = max(pos[i], pos[i-1] + 1e-8)
            
        return pos, h_vals
    
    def segments_to_h(segments):
        pos, h_vals = get_h_from_segments(segments)
        h = np.zeros(n_points)
        for i in range(n_segments):
            start_idx = int(pos[i] / 2.0 * n_points)
            end_idx = int(pos[i+1] / 2.0 * n_points)
            if end_idx > start_idx:
                h[start_idx:end_idx] = h_vals[i]
        return h
    
    def compute_c5(segments):
        h = segments_to_h(segments)
        
        # Fast computation with correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        # We only compute max over non-zero lags
        # But start from index of the middle to include all
        max_idx = len(correlation) // 2
        c5 = np.max(correlation[max_idx:]) * dx
        
        return c5
    
    # Optimization with local search around best solution
    best_c5 = float('inf')
    best_segments = None
    
    start_time = time.time()
    
    def objective(segments):
        return compute_c5(segments)
        
    def constraint_sum(segments):
        pos, h_vals = get_h_from_segments(segments)
        integral = 0
        for i in range(n_segments):
            integral += (pos[i+1] - pos[i]) * h_vals[i]
        return integral - 1.0
    
    # Initial setup
    pos_guess = np.linspace(0, 2, n_segments+1)
    h_guess = np.ones(n_segments) * 0.5
    
    # Create initial segments
    initial_segments = np.concatenate([pos_guess[1:-1], h_guess])
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 2)] * (n_segments - 1) + [(0, 1)] * n_segments
    
    # Start with multi-start approach
    for i in range(5):
        # Randomly perturb the starting solution
        segments = initial_segments.copy()
        if i > 0:
            # Add some variation to avoid local minima
            segments[:n_segments-1] += np.random.normal(0, 0.05, n_segments-1)
            segments[n_segments-1:] += np.random.normal(0, 0.05, n_segments)
        
        try:
            # Optimize
            result = minimize(objective, segments, method='SLSQP', bounds=bounds,
                              constraints=constraints, options={'maxiter': 100})
            if result.success:
                segments_result = result.x
                c5_val = compute_c5(segments_result)
                
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_segments = segments_result.copy()
        except:
            pass
            
        if time.time() - start_time > budget_s - 5:
            break
    
    # Likely haven't found a better solution by this point, use final refinement
    if best_c5 >= 0.3807:
        # Try gradient-based refinement on best found
        segments_final = best_segments.copy()
        for step in range(50):
            if time.time() - start_time > budget_s - 1:
                break
                
            h = segments_to_h(segments_final)
            
            # Simple gradient approximation based on local search
            step_size = 0.01
            
            # Perturb positions and heights to compute approximate gradient
            grad = np.zeros_like(segments_final)
            
            for j in range(len(segments_final)):
                segments_plus = segments_final.copy()
                segments_plus[j] += step_size
                
                c5_plus = compute_c5(segments_plus)
                grad[j] = (c5_plus - best_c5) / step_size
                
            # Update
            segments_final -= 0.005 * grad
            
            # Enforce constraints
            pos, h_vals = get_h_from_segments(segments_final)
            pos = np.clip(pos, 0, 2)
            h_vals = np.clip(h_vals, 0, 1)
            
            segments_final = np.concatenate([pos[1:-1], h_vals])
            
            h_new = segments_to_h(segments_final)
            new_c5 = compute_c5(segments_final)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_segments = segments_final.copy()
    
    # Final conversion back to h values
    if best_segments is not None:
        h_fin = segments_to_h(best_segments)
    else:
        h_fin = h  # Fallback if no optimization worked
        
    return (h_fin.tolist(), best_c5, n_points)
