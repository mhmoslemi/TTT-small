import numpy as np
import time
from heapq import heappush, heappop
from scipy import fft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    time_start = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Compute initial overlap
    def overlap_for_shift(h, k):
        if k > 0:
            return np.sum(h[:-k] * (1 - h[k:])) * dx
        elif k < 0:
            return np.sum(h[-k:] * (1 - h[:-k])) * dx
        else:
            return 0
    
    # Initial overlap for all shifts
    overlap = []
    for k in range(-n_points + 1, n_points):
        if k != 0:
            overlap.append((overlap_for_shift(h, k), k))
    
    # Build max heap (negate for max heap behavior)
    overlap_heap = [(-val, k) for val, k in overlap]
    np.random.shuffle(overlap_heap)
    heap = []
    for val, k in overlap_heap:
        heappush(heap, (val, k))
    
    best_h = h.copy()
    best_c5 = -heappop(heap)[0]  # Correct to positive again
    
    # Parameters
    window_width = 5
    perturb_scale = 0.02
    improvement_threshold = 1e-6
    max_no_improvement = 10
    refresh_interval = 50
    
    # Main optimization loop
    iteration = 0
    no_improvement = 0
    
    while time.time() - time_start < budget_s - 1:
        if iteration % refresh_interval == 0:
            # Re-compute overlaps
            overlap_new = []
            for k in range(-n_points + 1, n_points):
                if k != 0:
                    overlap_new.append((overlap_for_shift(h, k), k))
            
            # Rebuild heap
            overlap_heap = [(-val, k) for val, k in overlap_new]
            heap = []
            for val, k in overlap_heap:
                heappush(heap, (val, k))
        
        # Get worst shift
        if not heap:
            break
        neg_overlap, k = heappop(heap)
        current_overlap = -neg_overlap
        
        # Apply localized adjustment around k
        h_new = h.copy()
        
        # Determine tail / head for shift k
        if k > 0:
            head = h_new[:-k]
            tail = h_new[k:]
        elif k < 0:
            head = h_new[-k:]
            tail = h_new[:-k]
        else:
            head = tail = None
        
        if head is not None and tail is not None:
            # Find maximum contribution
            contribution = head * (1 - tail)
            max_idx = np.argmax(contribution)
            
            # Adjust region
            adjust_start = max(0, max_idx - window_width)
            adjust_end = min(n_points, max_idx + window_width + 1)
            
            # Apply small changes in both areas
            for i in range(adjust_start, adjust_end):
                if i + k >= 0 and i + k < n_points:
                    # Local adjustment
                    delta = 0.001 * (1 - 2 * h_new[i])  # Tiny move towards improving
                    if 0 <= i < n_points and 0 <= i + k < n_points:
                        if h_new[i] + delta >= 0 and h_new[i] + delta <= 1:
                            h_new[i] += delta
                        if h_new[i + k] - delta >= 0 and h_new[i + k] - delta <= 1:
                            h_new[i + k] -= delta
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Random perturbation
        for i in range(n_points):
            if np.random.rand() < 0.1:
                delta = np.random.normal(0, perturb_scale)
                h[i] += delta
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Check if improvement
        new_c5 = -heappop(heap)[0] if heap else float('inf')
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Adjustment makes no sense to continue if no improvement
        if no_improvement > max_no_improvement:
            window_width = min(20, window_width + 1)
            perturb_scale *= 2
            no_improvement = 0
        
        iteration += 1
    
    # Final optimization round with direct computation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
