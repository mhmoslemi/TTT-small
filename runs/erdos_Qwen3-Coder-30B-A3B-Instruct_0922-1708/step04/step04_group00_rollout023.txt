import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simpler approach: gradient-based optimization with constraints
    def objective(h):
        # Normalize h
        h_norm = h * (n_points / 2) / np.sum(h)
        h_norm = np.clip(h_norm, 0, 1)
        # Compute correlation
        correlation = np.correlate(h_norm, 1 - h_norm, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def constraint_func(h):
        return np.sum(h) * dx - 1.0
    
    # Define constraints for scipy optimization
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple initializations with different strategies
    for i in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Different initialization strategies
        if i == 0:
            # Use parent if available
            if 'initial_h_values' in globals():
                h_init = h.copy()
            else:
                h_init = np.random.uniform(0.3, 0.7, n_points)
        elif i == 1:
            # Small random perturbations
            h_init = np.random.uniform(0.2, 0.8, n_points)
        elif i == 2:
            # Biased towards 0 and 1
            h_init = np.random.choice([0.0, 0.5, 1.0], n_points, p=[0.3, 0.4, 0.3])
        elif i == 3:
            # Peaks in middle
            h_init = np.zeros(n_points)
            center = n_points // 2
            width = n_points // 5
            h_init[max(0, center - width):min(n_points, center + width)] = 1.0
        else:  # i == 4
            # Smooth
            x = np.linspace(0, 2, n_points)
            h_init = np.exp(-4*(x-1)**2) + 0.1
            h_init = h_init / np.sum(h_init) * (n_points / 2)
            
        # Normalize and clip
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Optimize with bounds and constraints
        try:
            result = minimize(objective, h_init, method='SLSQP', 
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 1000, 'ftol': 1e-6})
            
            if result.success:
                c5_val = objective(result.x)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = result.x.copy()
                    
        except:
            pass
    
    # Local refinement with coordinate descent
    h_local = best_h.copy()
    max_refine_iter = 2000
    
    for refine_iter in range(max_refine_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Keep track of current best
        c5_current = objective(h_local)
        improved = False
        
        # Try small perturbations for each coordinate
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Store old value
            old_val = h_local[i]
            
            # Try small changes in both directions
            for delta in [0.01, -0.01]:
                h_local[i] = old_val + delta
                h_local = np.clip(h_local, 0, 1)
                h_local = h_local * (n_points / 2) / np.sum(h_local)
                
                c5_new = objective(h_local)
                if c5_new < c5_current:
                    c5_current = c5_new
                    improved = True
                else:
                    h_local[i] = old_val  # revert
                    
        if not improved:
            break
            
        # Update best if improved
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_local.copy()
    
    # Final check
    final_c5 = objective(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
