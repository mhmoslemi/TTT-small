import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Coarse grid search
    start_time = time.time()
    coarse_size = 20
    coarse_candidates = []
    
    for _ in range(500):
        if time.time() - start_time > budget_s * 0.3:
            break
        # Generate a random candidate
        h_coarse = np.random.rand(coarse_size)
        h_coarse = h_coarse / np.sum(h_coarse) * (coarse_size / 2)
        
        # Map to fine grid
        h_fine = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, coarse_size), h_coarse)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        # Evaluate
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full')
        c5 = np.max(correlation) * dx
        coarse_candidates.append((h_fine.copy(), c5))
    
    # Sort candidates by C5
    coarse_candidates.sort(key=lambda x: x[1])
    top_candidates = coarse_candidates[:10]
    
    # Refine each top candidate
    for candidate_h, _ in top_candidates:
        h_refined = candidate_h.copy()
        
        # Gradient descent refinement
        for i in range(1000):
            if time.time() - start_time > budget_s * 0.8:
                break
                
            # Compute gradient
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr = np.correlate(h_refined, 1 - h_refined, mode='full')
                c5_current = np.max(corr) * dx
                
                grad[j] = (c5_plus - c5_current) / eps
            
            # Update
            lr = 0.01
            h_new = h_refined - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_refined = h_new
            
            # Evaluate
            correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_refined.copy()
    
    # Final CVX optimization
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass

    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
