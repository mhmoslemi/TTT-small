import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    # Ensure the sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Evolutionary algorithm approach
    population_size = 100
    generations = 100
    mutation_rate = 0.05
    crossover_rate = 0.5
    selection_rate = 0.1
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            ind = np.array(initial_h_values)
            if len(ind) != n_points:
                x_orig = np.linspace(0, 2, len(ind))
                x_target = np.linspace(0, 2, n_points)
                ind = np.interp(x_target, x_orig, ind)
        else:
            ind = np.random.rand(n_points)
        ind = np.clip(ind, 0, 1)
        ind = ind * (n_points / 2) / np.sum(ind)
        population.append(ind)
    
    # Main evolutionary loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all individuals
        fitness = []
        for ind in population:
            correlation = np.correlate(ind, 1 - ind, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Selection
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:int(population_size * selection_rate)]]
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover - uniform crossover
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(selected) >= 2:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            else:
                # Mutate
                parent = np.random.choice(selected)
                child = parent.copy()
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        # Flip bit in binary-like interpretation
                        child[i] = 1 - child[i]
                # Adjust to sum constraint
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        population = new_population[:population_size]
        
        # Update best solution
        for i, ind in enumerate(population):
            if fitness[i] < best_c5:
                best_c5 = fitness[i]
                best_h = ind.copy()
    
    # Final convex refinement on the best solution found
    try:
        # Define the optimization variables and constraints
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Objective
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass

    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
