import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Evaluate initial function
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Evolutionary algorithm parameters
    pop_size = 100
    generations = 100
    mutation_rate = 0.05
    crossover_rate = 0.5
    selection_rate = 0.1

    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                h_orig = np.array(initial_h_values)
                x_orig = np.linspace(0, 2, len(h_orig))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h_orig)
            h = h / np.sum(h) * (n_points / 2)
        else:
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
        population.append(h)

    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate population
        fitness = []
        for ind in population:
            correlation = np.correlate(ind, 1 - ind, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:int(pop_size * selection_rate)]
        population_sorted = [population[i] for i in sorted_indices]
        
        # Keep best solution so far
        if fitness[sorted_indices[0]] < best_c5:
            best_c5 = fitness[sorted_indices[0]]
            best_h = population_sorted[0].copy()
        
        # Selection: keep top performers
        selected = [population_sorted[i] for i in range(len(top_indices))]
        
        # Crossover and mutation to create new population
        new_population = selected[:]
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate and len(selected) >= 2:
                i1, i2 = np.random.choice(len(selected), 2, replace=False)
                parent1, parent2 = selected[i1], selected[i2]
                # Blend crossover
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
                # Ensure sum constraint
                child = child / np.sum(child) * (n_points / 2)
            else:
                # Mutation with small random changes
                parent = selected[np.random.randint(0, len(selected))]
                child = parent.copy()
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, 0.05)
                child = np.clip(child, 0, 1)
                child = child / np.sum(child) * (n_points / 2)
            
            new_population.append(child)
            
        population = new_population[:pop_size]
    
    # Convex refinement on best candidates
    top_candidates = []
    fitness = []
    for ind in population:
        correlation = np.correlate(ind, 1 - ind, mode='full')
        c5 = np.max(correlation) * dx
        fitness.append(c5)
    sorted_indices = np.argsort(fitness)
    for i in range(min(10, len(sorted_indices))):
        top_candidates.append(population[sorted_indices[i]])
    
    for ind in top_candidates:
        try:
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Use a slightly modified approach to the convex problem by relaxing:
            # Convert the max_overlap problem into a constraint satisfaction task
            # One common approach is using semi-definite relaxation
            # For this simplified version, we'll take a direct least-squares approach:

            # Use convex optimization on a single candidate without guards for speed
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
            prob = cp.Problem(objective, constraints)
            
            # Try a few solvers; use fallback if needed
            solvers = [cp.CBC, cp.CVXOPT, cp.GLPK]
            solver_used = None
            for solver in solvers:
                try:
                    prob.solve(solver=solver, verbose=False)
                    if prob.status == cp.OPTIMAL:
                        solver_used = solver
                        break
                except:
                    continue

            if solver_used is not None:
                h_refined = h_vars.value
                if h_refined is not None:
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    
                    correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
                    c5_refined = np.max(correlation) * dx
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h_refined.copy()
        except:
            pass

    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
