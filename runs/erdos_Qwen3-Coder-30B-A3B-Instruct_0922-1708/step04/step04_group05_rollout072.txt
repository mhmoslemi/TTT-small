import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize from global initial_h_values or create a structured one
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        # Create structured initialization
        h_init = np.ones(n_points) * 0.5
        # Add a small perturbation for diversity
        h_init += np.random.normal(0, 0.01, n_points) * (np.sin(np.linspace(0, 4*np.pi, n_points)) + 1) * 0.5

    # Ensure the integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iter_count = 0

    # Generate diverse starting candidates
    candidates = []
    for i in range(20):
        h = h_init.copy()
        # Add noise and shift
        noise = np.random.normal(0, 0.02, n_points)
        h = np.clip(h + noise, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h.copy())
    
    # Gradient-based refinement loop
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        # Select a random candidate
        h = candidates[np.random.randint(len(candidates))].copy()
        
        # Project h to satisfy the integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute the correlation to get current C5
        correlation_full = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation_full) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update h using gradient descent
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Add new diverse candidate periodically
        if iter_count % 20 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_temp = h + noise
            h_temp = np.clip(h_temp, 0, 1)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            candidates.append(h_temp.copy())
        
        # Maintain diverse candidates by trimming
        if len(candidates) > 30:
            candidates = candidates[1:]  # Remove oldest

    # Final check and return best solution
    correlation_full = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation_full) * dx
    
    return (best_h.tolist(), final_c5, n_points)
