import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Hyperparameters
    learning_rate = 0.1
    decay_rate = 0.99
    max_iter_without_improvement = 100
    epsilon = 1e-5
    
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_iter_without_improvement:
        # Compute overlap for all lags
        correlation = correlate(h, 1 - h, mode='full')
        max_overlap_val = np.max(correlation) * dx
        
        if max_overlap_val < best_c5:
            best_c5 = max_overlap_val
            best_h = h.copy()
            
        # Compute gradients for each point
        gradients = np.zeros(n_points)
        
        for i in range(n_points):
            # Increase h[i] slightly
            h_plus = h.copy()
            h_plus[i] += epsilon
            # Clip and rescale
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Decrease h[i] slightly
            h_minus = h.copy()
            h_minus[i] -= epsilon
            # Clip and rescale
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Approximate gradient
            gradients[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update h
        h_new = h - learning_rate * gradients
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Set h to new value
        h = h_new
        
        # Decay learning rate
        learning_rate *= decay_rate
        
        # Check for improvement
        iteration += 1
        if max_overlap_val < best_c5:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
    
    # Final evaluation
    correlation = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
