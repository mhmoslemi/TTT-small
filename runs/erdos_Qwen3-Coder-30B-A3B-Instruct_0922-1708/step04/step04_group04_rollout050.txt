import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from provided parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    
    # Use CVXPY to define and solve optimization problem
    # Variables
    h_var = cp.Variable(n_points, nonneg=True)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var <= 1
    ]
    
    # Compute overlaps for all possible shifts
    for k in range(1, n_points):
        # This part computes the correlation for shift k
        # We will compute the overlap for k shifts forward
        # and k shifts backward
        if k < n_points:
            # For positive shift
            corr_forward = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
            constraints.append(corr_forward <= C)
            
            # For negative shift (same as positive in reverse)
            corr_backward = cp.sum(cp.multiply(h_var[k:], 1 - h_var[:n_points-k]))
            constraints.append(corr_backward <= C)
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    
    # Formulate problem
    prob = cp.Problem(objective, constraints)
    
    # Try multiple solvers if needed
    solvers = ['ECOS', 'SCS']
    best_h = h.copy()
    best_c5 = float('inf')
    
    for solver_name in solvers:
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            prob.solve(solver=getattr(cp, solver_name), verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = h_var.value
                if h_sol is not None:
                    # Ensure bounds
                    h_sol = np.clip(h_sol, 0, 1)
                    # Rescale to match the constraint
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    
                    # Evaluate the maximum overlap
                    correlations = []
                    for k in range(1, n_points):
                        if k < n_points:
                            corr_val = np.sum(h_sol[:n_points-k] * (1 - h_sol[k:]))
                            correlations.append(corr_val)
                    
                    max_correlation = max(correlations) if correlations else 0
                    c5_val = max_correlation * dx
                    
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_sol.copy()
        except Exception:
            continue
            
    # Final refinement
    h_current = best_h.copy()
    max_iter = 100
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        correlations_list = []
        for k in range(1, n_points):
            if k < n_points:
                corr_val = np.sum(h_current[:n_points-k] * (1 - h_current[k:]))
                correlations_list.append(corr_val)
                
        max_overlap = max(correlations_list) if correlations_list else 0
        c5_current = max_overlap * dx
        
        # Gradient descent step on a simple approximating function
        # This is a basic gradient estimation approach
        grad = np.zeros_like(h_current)
        eps = 1e-4
        
        # finite differences to estimate gradient
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = 0
            for k in range(1, n_points):
                if k < n_points:
                    corr_plus += np.sum(h_plus[:n_points-k] * (1 - h_plus[k:]))
            c5_plus = corr_plus * dx
            
            grad[i] = (c5_plus - c5_current) / eps
            
        # Update step
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h_current - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_current = h_new
        
        # Evaluate new value
        correlations_new = []
        for k in range(1, n_points):
            if k < n_points:
                corr_val = np.sum(h_current[:n_points-k] * (1 - h_current[k:]))
                correlations_new.append(corr_val)
                
        max_correlation_new = max(correlations_new) if correlations_new else 0
        c5_new = max_correlation_new * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_current.copy()
            
        if abs(c5_current - c5_new) < 1e-10:
            break
    
    return (best_h.tolist(), best_c5, n_points)
