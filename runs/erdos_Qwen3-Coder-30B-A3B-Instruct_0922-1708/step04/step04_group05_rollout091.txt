import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure integral constraint sum(h) * dx = 1
    h = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize tracking variables
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Parameters
    gamma = 0.001
    max_iter = 1000
    patience = 100
    patience_counter = 0
    prev_c5 = float('inf')
    
    # FFT-based convolution helper
    def convolve_h_1h(h_vec):
        # Pad to avoid circular convolution effects
        pad = 2 * n_points
        h_padded = np.pad(h_vec, (0, pad), mode='constant')
        h1_padded = np.pad(1 - h_vec, (0, pad), mode='constant')
        conv = np.fft.irfft(np.fft.rfft(h_padded) * np.fft.rfft(h1_padded))
        return conv[:2*n_points-1]
    
    # Gradient-based refinement loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate current C5
        overlap = convolve_h_1h(h)
        current_c5 = np.max(overlap) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            break
            
        # Compute gradient (simplified finite difference approach)
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            overlap_plus = convolve_h_1h(h_plus)
            c5_plus = np.max(overlap_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps

        # Update with projected gradient descent
        h_new = h - gamma * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Add momentum
        h = 0.9 * h + 0.1 * h_new
        
        # Project again
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodic perturbation to escape local optima
        if iter_count % 100 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Adapt learning rate
        if iter_count % 100 == 0:
            gamma *= 0.9
            
    # Final evaluation
    final_overlap = convolve_h_1h(best_h)
    final_c5 = np.max(final_overlap) * dx
    
    return (best_h.tolist(), final_c5, n_points)
