import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from itertools import combinations

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_generations = 100
    population_size = 100
    mutation_rate = 0.05
    crossover_rate = 0.5
    
    # Initialize population
    def create_individual():
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h[indices] = 1
        return h
    
    def create_population():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
            h = (h > 0.5).astype(float)
            if np.sum(h) != n_points / 2:
                # Adjust to sum to exactly n_points/2
                diff = int(n_points / 2 - np.sum(h))
                if diff > 0:
                    zero_indices = np.where(h == 0)[0]
                    if len(zero_indices) >= diff:
                        flip_indices = np.random.choice(zero_indices, size=diff, replace=False)
                        h[flip_indices] = 1
                else:
                    one_indices = np.where(h == 1)[0]
                    if len(one_indices) >= -diff:
                        flip_indices = np.random.choice(one_indices, size=-diff, replace=False)
                        h[flip_indices] = 0
        else:
            h = create_individual()
        
        # Create population
        population = [h.copy()]
        for _ in range(population_size - 1):
            population.append(create_individual())
        
        return population
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def mutate(h, rate):
        h_mut = h.copy()
        num_mutations = int(rate * n_points)
        flip_indices = np.random.choice(n_points, size=num_mutations, replace=False)
        for idx in flip_indices:
            h_mut[idx] = 1 - h_mut[idx]
        
        # Fix sum to be exactly n_points / 2
        current_sum = np.sum(h_mut)
        diff = int(n_points / 2 - current_sum)
        if diff != 0:
            if diff > 0:
                zero_indices = np.where(h_mut == 0)[0]
                if len(zero_indices) >= diff:
                    flip_indices = np.random.choice(zero_indices, size=diff, replace=False)
                    h_mut[flip_indices] = 1
            else:
                one_indices = np.where(h_mut == 1)[0]
                if len(one_indices) >= -diff:
                    flip_indices = np.random.choice(one_indices, size=-diff, replace=False)
                    h_mut[flip_indices] = 0
        return h_mut
    
    def crossover(h1, h2):
        # Arithmetic crossover
        r = np.random.random(n_points)
        h_new = r * h1 + (1 - r) * h2
        # Binary regularization
        h_new = (h_new > 0.5).astype(float)
        # Enforce sum
        if np.sum(h_new) != n_points / 2:
            diff = int(n_points / 2 - np.sum(h_new))
            if diff > 0:
                zero_indices = np.where(h_new == 0)[0]
                if len(zero_indices) >= diff:
                    flip_indices = np.random.choice(zero_indices, size=diff, replace=False)
                    h_new[flip_indices] = 1
            else:
                one_indices = np.where(h_new == 1)[0]
                if len(one_indices) >= -diff:
                    flip_indices = np.random.choice(one_indices, size=-diff, replace=False)
                    h_new[flip_indices] = 0
        return h_new
    
    def refine_with_cvx(h):
        # Use convex relaxation
        try:
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            # Convex formulation of maximizing the overlap
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (n_points, 1)))))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            if prob.status == cp.OPTIMAL and h_vars.value is not None:
                h_refined = np.clip(h_vars.value, 0, 1)
                if abs(np.sum(h_refined) - n_points / 2) < 1e-6:
                    return h_refined
        except:
            pass
        return h
    
    # Start optimization
    population = create_population()
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind)
            fitness.append((c5, ind))
        
        # Sort by fitness
        fitness.sort(key=lambda x: x[0])
        
        # Update best
        if fitness[0][0] < best_c5:
            best_c5 = fitness[0][0]
            best_h = fitness[0][1].copy()
        
        # Select top performers
        top_k = max(10, population_size // 10)
        top_individuals = [ind for _, ind in fitness[:top_k]]
        
        # Crossover
        new_population = []
        for _ in range(population_size):
            if np.random.random() < crossover_rate:
                parent1, parent2 = np.random.choice(top_individuals, 2, replace=False)
                child = crossover(parent1, parent2)
                new_population.append(child)
            else:
                child = mutate(np.random.choice(top_individuals), mutation_rate)
                new_population.append(child)
        
        # Population refinement with convex optimization
        top_candidates = [ind for _, ind in fitness[:10]]
        for i, ind in enumerate(top_candidates):
            refined = refine_with_cvx(ind)
            c5_refined = compute_c5(refined)
            if c5_refined < fitness[i][0]:
                new_population[i] = refined
        
        population = new_population
        
        # Periodic reinitialization to avoid premature convergence
        if gen % 10 == 0:
            for i in range(population_size // 2):
                if np.random.random() < 0.1:
                    population[i] = create_individual()
    
    # Final convex refinement of best solution
    if best_h is not None:
        best_h_refined = refine_with_cvx(best_h)
        final_c5 = compute_c5(best_h_refined)
        if final_c5 < best_c5:
            best_h = best_h_refined
            best_c5 = final_c5
    
    if best_h is None:
        # Fallback
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            if len(best_h) != n_points:
                best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        else:
            best_h = np.ones(n_points) * 0.5
        best_h = best_h / np.sum(best_h) * (n_points / 2)
        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
