import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Constraint function for sum constraint
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Trust region optimization with TRF method
    max_iter = 200
    current_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Objective function
        def objective(x):
            # Ensure x is within bounds
            x = np.clip(x, 0, 1)
            # Renormalize to satisfy constraint
            x = x * (n_points / 2) / np.sum(x)
            correlation = np.correlate(x, 1 - x, mode='full')
            return -np.max(correlation) * dx  # Negate for minimization
        
        # Gradient estimation via finite differences
        def gradient(x):
            eps = 1e-4
            grad = np.zeros_like(x)
            for i in range(n_points):
                x_plus = x.copy()
                x_plus[i] += eps
                x_plus = np.clip(x_plus, 0, 1)
                x_plus = x_plus * (n_points / 2) / np.sum(x_plus)
                corr_plus = np.correlate(x_plus, 1 - x_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                x_minus = x.copy()
                x_minus[i] -= eps
                x_minus = np.clip(x_minus, 0, 1)
                x_minus = x_minus * (n_points / 2) / np.sum(x_minus)
                corr_minus = np.correlate(x_minus, 1 - x_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            return grad
        
        # Optimization step
        try:
            res = minimize(objective, current_h, method='TRF', bounds=bounds, 
                          options={'maxiter': 10, 'init_tr_radius': 0.1})
            if res.success:
                current_h = res.x
            else:
                # If optimization fails, use a simple gradient descent step
                grad = gradient(current_h)
                current_h = current_h - 0.001 * grad
        except:
            # Fallback to simple gradient descent step if optimization fails
            try:
                grad = gradient(current_h)
                current_h = current_h - 0.001 * grad
            except:
                # If gradient estimation fails, perturb randomly
                current_h = current_h + np.random.normal(0, 0.001, n_points)
        
        # Maintain constraints
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # Evaluate current solution
        correlation = np.correlate(current_h, 1 - current_h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
    
    # Final evaluation of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
