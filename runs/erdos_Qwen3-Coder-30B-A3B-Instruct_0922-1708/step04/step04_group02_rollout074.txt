import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Temperature schedule
    temperature = 1.0
    cooling_factor = 0.99
    cooling_interval = 100
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C₅
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate candidate solution
        h_new = h.copy()
        
        # Select random subset of indices to modify
        n_modify = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, size=n_modify, replace=False)
        
        # Modify selected points with small random changes
        for i in indices:
            # Add small random change
            change = np.random.uniform(-0.1, 0.1)
            h_new[i] += change
            
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain integral constraint
        sum_diff = np.sum(h_new) - n_points / 2
        if abs(sum_diff) > 1e-8:
            # Redistribute to maintain sum = n_points/2
            diff_per_point = sum_diff / n_points
            h_new -= diff_per_point
            # Re-clip to maintain bounds
            h_new = np.clip(h_new, 0, 1)
            
        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C₅ of new solution
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject based on simulated annealing
        delta = new_c5 - current_c5
        if delta < 0 or np.exp(-delta / temperature) > np.random.random():
            h = h_new.copy()
        
        # Update temperature
        if iteration % cooling_interval == 0:
            temperature *= cooling_factor
    
    # Final refinement with local search
    refine_h = best_h.copy()
    refine_c5 = best_c5
    
    for _ in range(500):
        h_new = refine_h.copy()
        i = np.random.randint(n_points)
        # Small perturbation
        change = np.random.uniform(-0.05, 0.05)
        h_new[i] += change
        
        # Ensure bounds and integral constraint
        h_new = np.clip(h_new, 0, 1)
        sum_diff = np.sum(h_new) - n_points / 2
        if abs(sum_diff) > 1e-8:
            diff_per_point = sum_diff / n_points
            h_new -= diff_per_point
            h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        if new_c5 < refine_c5:
            refine_h = h_new.copy()
            refine_c5 = new_c5
    
    # Final evaluation
    final_correlation = np.correlate(refine_h, 1 - refine_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (refine_h.tolist(), final_c5, n_points)
