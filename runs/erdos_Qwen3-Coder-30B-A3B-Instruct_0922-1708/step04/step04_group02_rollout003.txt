import time
import random

    def evaluate_h(h):
        """Evaluate the C5 bound for a given h"""
        dx = 2.0 / len(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def repair_h(h):
        """Ensure h sums to n_points/2 and is in [0,1]"""
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * (len(h) / 2) / current_sum
        return h
    
    def crossover(parent1, parent2, eta=20):
        """Simulated Binary Crossover"""
        if random.random() > 0.6:
            return parent1.copy(), parent2.copy()
            
        child1, child2 = parent1.copy(), parent2.copy()
        for i in range(len(parent1)):
            if random.random() < 0.5:
                continue
            if parent1[i] == parent2[i]:
                continue
                
            if parent1[i] > parent2[i]:
                parent1[i], parent2[i] = parent2[i], parent1[i]
                
            rand = random.random()
            if rand <= 0.5:
                beta = 2 * rand
            else:
                beta = 1 / (2 * (1 - rand))
                
            alpha = (2 - beta) ** (1 / (eta + 1))
            child1[i] = 0.5 * ((parent1[i] + parent2[i]) - beta * (parent2[i] - parent1[i]))
            child2[i] = 0.5 * ((parent1[i] + parent2[i]) + beta * (parent2[i] - parent1[i]))
            
        return child1, child2
        
    def mutate(h, ind_mutation_rate):
        """Polynomial mutation"""
        h = h.copy()
        for i in range(len(h)):
            if random.random() < ind_mutation_rate:
                if random.random() < 0.5:
                    delta = 1 - h[i]
                else:
                    delta = h[i]
                    
                rand = random.random()
                if rand <= 0.5:
                    sigma = (2 * rand) ** (1 / (20 + 1)) - 1
                else:
                    sigma = 1 - (2 * (1 - rand)) ** (1 / (20 + 1))
                    
                h[i] = h[i] + sigma * delta
                
        return h

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    
    # Genetic Algorithm parameters
    pop_size = 100
    max_gen = 200
    mutation_rate = 0.1
    elite_size = 20
    
    population = []
    for _ in range(pop_size):
        h_rand = repair_h(np.random.rand(n_points) * 2)
        population.append(h_rand)
    
    population = [repair_h(h) for h in population]
    population = sorted(population, key=evaluate_h)
    
    start_time = time.time()
    
    # Evolution loop
    for gen in range(max_gen):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate the population
        fitness = [evaluate_h(h) for h in population]
        
        # Selection using tournament
        selected = []
        for _ in range(pop_size):
            tournament_size = 3
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            tournament.sort(key=lambda x: x[1])  # Sort by fitness (lower is better)
            selected.append(tournament[0][0])
        
        # Crossover and mutation
        new_population = []
        
        # Elitism: keep the best solutions
        elite_indices = sorted(range(len(fitness)), key=lambda i: fitness[i])[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            child1 = repair_h(child1)
            child2 = repair_h(child2)

            h_test = h_refined.copy()
            h_test[j] += eps
            h_test = repair_h(h_test)
            
            c5_test = evaluate_h(h_test)
            grad[j] = (c5_test - h_c5) / eps
            
        # Update with gradient descent
        lr = 0.005
        h_refined -= lr * grad
        h_refined = repair_h(h_refined)
        
        # Early stopping
        if evaluate_h(h_refined) < best_c5:
            best_h = h_refined.copy()
            best_c5 = evaluate_h(h_refined)
    
    return (best_h.tolist(), best_c5, n_points)
