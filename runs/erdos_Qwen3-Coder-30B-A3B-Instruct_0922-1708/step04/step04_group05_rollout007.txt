import numpy as np
import time

def project_feasible(h, n_points):
    """Project h to satisfy bounds and integral constraint."""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_C5(h, n_points):
    """Compute the C5 bound."""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_step(h, n_points, eps=1e-4):
    """Compute gradient of C5 with finite differences."""
    dx = 2.0 / n_points
    base_corr = np.correlate(h, 1 - h, mode='full')
    base_c5 = np.max(base_corr) * dx
    grad = np.zeros_like(h)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        grad[i] = (c5_plus - base_c5) / eps
    
    return grad

def line_search(h, grad, n_points, max_iter=20):
    """Perform line search for step size using golden section."""
    # Golden ratio
    phi = (1 + np.sqrt(5)) / 2
    rho = 1 / phi
    
    # Initial points
    a, b = 0, 10.0  # Reasonable step size range
    c = a + (b - a) * (1 - 1/phi)
    d = a + (b - a) * 1/phi
    
    for _ in range(max_iter):
        # Try c
        h_new_c = project_feasible(h - c * grad, n_points)
        c5_c = compute_C5(h_new_c, n_points)
        
        # Try d
        h_new_d = project_feasible(h - d * grad, n_points)
        c5_d = compute_C5(h_new_d, n_points)
        
        if c5_c < c5_d:
            b = d
            d = c
            c = a + (b - a) * (1 - 1/phi)
        else:
            a = c
            c = d
            d = a + (b - a) * 1/phi
    
    # Choose best step
    step_size = (a + b) / 2
    return step_size

        # Evaluate candidates and find best
        for h_cand in h_candidates:
