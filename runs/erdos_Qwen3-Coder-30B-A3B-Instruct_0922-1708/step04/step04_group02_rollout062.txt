import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # PSO parameters
    n_points = 100
    swarm_size = 100
    max_iter = 900
    omega = 0.7
    phi_p = 2.0
    phi_g = 2.0
    local_search_freq = 50
    rejuvenation_rate = 0.1
    resolution_steps = [50, 100, 200]
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Interpolate to target resolution
        x_orig = np.linspace(0, 2, len(h_init))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_init)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Generate random starting solution
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        
    # Initialize swarm
    particles = [h.copy() for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(swarm_size)]
    
    # Initialize personal bests
    pbest = [p.copy() for p in particles]
    pbest_fitness = [float('inf')] * swarm_size
    
    # Global best
    gbest = h.copy()
    gbest_fitness = float('inf')
    
    start_time = time.time()
    
    # PSO main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness for all particles
        for i in range(swarm_size):
            h = particles[i]
            # Compute correlation
            correlation = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(correlation) * (2.0 / n_points)
            
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest[i] = h.copy()
                
            if fitness < gbest_fitness:
                gbest_fitness = fitness
                gbest = h.copy()
        
        # Update swarm
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r_p = np.random.random(n_points)
            r_g = np.random.random(n_points)
            
            v = (omega * v + 
                 phi_p * r_p * (pbest[i] - h) + 
                 phi_g * r_g * (gbest - h))
            
            # Update position
            h_new = h + v
            
            # Repair
            h_new = np.clip(h_new, 0, 1)
            sum_h = np.sum(h_new)
            if abs(sum_h - n_points / 2) > 1e-8:
                diff = sum_h - n_points / 2
                if diff > 0:
                    # Excess, decrease largest elements
                    indices = np.argsort(h_new)[::-1]
                    for idx in indices:
                        if diff <= 0:
                            break
                        decrease = min(diff, h_new[idx])
                        h_new[idx] -= decrease
                        diff -= decrease
                else:
                    # Deficit, increase smallest elements
                    indices = np.argsort(h_new)
                    for idx in indices:
                        if diff >= 0:
                            break
                        increase = min(-diff, 1 - h_new[idx])
                        h_new[idx] += increase
                        diff += increase
            
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            velocities[i] = v
            
        # Local search every local_search_freq iterations
        if iter_count % local_search_freq == 0:
            # Get top 10% of particles
            top_indices = np.argsort([pbest_fitness[i] for i in range(swarm_size)])[:swarm_size//10]
            for i in top_indices:
                h = pbest[i]
                # Gradient-based local search
                for _ in range(10):
                    # Compute gradient using finite differences
                    grad = np.zeros(n_points)
                    eps = 1e-5
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus_sum = np.sum(h_plus)
                        if abs(h_plus_sum - n_points / 2) > 1e-8:
                            h_plus = h_plus * (n_points / 2) / h_plus_sum
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * (2.0 / n_points)
                        
                        corr = np.correlate(h, 1 - h, mode='full')
                        c5 = np.max(corr) * (2.0 / n_points)
                        
                        grad[j] = (c5_plus - c5) / eps
                    
                    # Update with gradient
                    h_new = h - 0.001 * grad
                    
                    # Repair and normalize
                    h_new = np.clip(h_new, 0, 1)
                    sum_h = np.sum(h_new)
                    if abs(sum_h - n_points / 2) > 1e-8:
                        diff = sum_h - n_points / 2
                        if diff > 0:
                            indices = np.argsort(h_new)[::-1]
                            for idx in indices:
                                if diff <= 0:
                                    break
                                decrease = min(diff, h_new[idx])
                                h_new[idx] -= decrease
                                diff -= decrease
                        else:
                            indices = np.argsort(h_new)
                            for idx in indices:
                                if diff >= 0:
                                    break
                                increase = min(-diff, 1 - h_new[idx])
                                h_new[idx] += increase
                                diff += increase
                    
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    
                    # Check improvement
                    corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                    c5_new = np.max(corr_new) * (2.0 / n_points)
                    
                    if c5_new < pbest_fitness[i]:
                        pbest[i] = h_new.copy()
                        pbest_fitness[i] = c5_new
                        if c5_new < gbest_fitness:
                            gbest = h_new.copy()
                            gbest_fitness = c5_new
                    else:
                        break
                    
                    h = h_new
        
        # Rejuvenation
        if iter_count % (local_search_freq * 2) == 0:
            worst_indices = np.argsort(pbest_fitness)[-int(swarm_size * rejuvenation_rate):]
            for idx in worst_indices:
                h = np.random.uniform(0, 1, n_points)
                h = h / np.sum(h) * (n_points / 2)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
        
        # Resolution stepping
        if iter_count in [50, 100, 200]:
            new_n_points = resolution_steps[resolution_steps.index(iter_count) + 1] \
                           if iter_count < resolution_steps[-1] else n_points
            
            # Interpolate best solution to new resolution
            if new_n_points < n_points:
                x_target = np.linspace(0, 2, new_n_points)
                x_ref = np.linspace(0, 2, n_points)
                h_ref = gbest
                h_interp = np.interp(x_target, x_ref, h_ref)
                h_interp = h_interp * (new_n_points / 2) / np.sum(h_interp)
                
                # Reinitialize swarm around this
                particles = [h_interp + np.random.normal(0, 0.05, new_n_points) for _ in range(swarm_size)]
                velocities = [np.random.uniform(-0.1, 0.1, new_n_points) for _ in range(swarm_size)]
                n_points = new_n_points
            else:
                # Increase resolution
                x_target = np.linspace(0, 2, new_n_points)
                x_ref = np.linspace(0, 2, n_points)
                h_ref = gbest
                h_interp = np.interp(x_target, x_ref, h_ref)
                h_interp = h_interp * (new_n_points / 2) / np.sum(h_interp)
                
                # Reinitialize swarm
                particles = [h_interp + np.random.normal(0, 0.05, new_n_points) for _ in range(swarm_size)]
                velocities = [np.random.uniform(-0.1, 0.1, new_n_points) for _ in range(swarm_size)]
                n_points = new_n_points
    
    # Final comprehensive refinement
    top_candidates = [pbest[i] for i in np.argsort(pbest_fitness)[:5]]
    final_fitness = float('inf')
    final_h = gbest.copy()
    
    # Intensive gradient descent on top candidates
    for h_candidate in top_candidates:
        h_refined = h_candidate.copy()
        for _ in range(1000):
            grad = np.zeros(n_points)
            eps = 1e-7
            # Compute gradient
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1) 
                h_plus_sum = np.sum(h_plus)
                if abs(h_plus_sum - n_points / 2) > 1e-8:
                    h_plus = h_plus * (n_points / 2) / h_plus_sum
                    
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * (2.0 / n_points)
                
                corr = np.correlate(h_refined, 1 - h_refined, mode='full')
                c5 = np.max(corr) * (2.0 / n_points)
                
                grad[j] = (c5_plus - c5) / eps
            
            # Update with gradient
            h_new = h_refined - 0.0005 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if abs(h_new_sum - n_points / 2) > 1e-8:
                diff = h_new_sum - n_points / 2
                if diff > 0:
                    indices = np.argsort(h_new)[::-1]
                    for idx in indices:
                        if diff <= 0:
                            break
                        decrease = min(diff, h_new[idx])
                        h_new[idx] -= decrease
                        diff -= decrease
                else:
                    indices = np.argsort(h_new)
                    for idx in indices:
                        if diff >= 0:
                            break
                        increase = min(-diff, 1 - h_new[idx])
                        h_new[idx] += increase
                        diff += increase
            
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * (2.0 / n_points)
            
            if c5_new < final_fitness:
                final_fitness = c5_new
                final_h = h_new.copy()
                h_refined = h_new.copy()
            else:
                break
    
    return (final_h.tolist(), final_fitness, n_points)
