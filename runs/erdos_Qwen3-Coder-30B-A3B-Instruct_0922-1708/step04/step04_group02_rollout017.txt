import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Multi-start optimization
    for start_iter in range(5):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Initialize with perturbed version of best solution or uniform
        if start_iter == 0:
            h = best_h.copy()
        else:
            h = best_h.copy()
            # Add small random perturbations
            for i in range(n_points):
                if np.random.random() < 0.3:
                    h[i] += (np.random.random() - 0.5) * 0.1
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Optimization using L-BFGS-B
        def objective(h_flat):
            # Reshape
            h_vec = h_flat.reshape(-1)
            # Ensure sum constraint
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            
            # Compute correlation
            correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
            c5 = np.max(correlation) * dx
            return c5
            
        def constraint_sum(h_flat):
            h_vec = h_flat.reshape(-1)
            return np.sum(h_vec) - n_points / 2
            
        # Bound constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        # Initial guess
        x0 = h.copy()
        
        # Optimize
        try:
            result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                            constraints=constraints, options={'maxiter': 500})
            
            if result.success:
                h_opt = result.x.reshape(-1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                
                # Check the objective
                corr = np.correlate(h_opt, 1 - h_opt, mode='full')
                c5_opt = np.max(corr) * dx
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
                    
        except:
            pass
            
        # Local refinement
        for refinement_iter in range(50):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_new = best_h.copy()
            
            # Small perturbations
            for i in range(n_points):
                if np.random.random() < 0.1:
                    step = (np.random.random() - 0.5) * 0.05
                    h_new[i] += step
                    
            # Clip and normalize
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Final computation of C₅
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
