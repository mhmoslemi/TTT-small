import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Compute initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    temperature = 1.0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Decrease temperature
        if iteration % 100 == 0:
            temperature *= 0.99
            
        # Generate candidate solution
        h_candidate = h.copy()
        
        # Perturb 10% of points
        indices_to_change = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        
        for i in indices_to_change:
            # Random perturbation in [0, 1] with magnitude 0.1
            perturbation = (random.random() - 0.5) * 0.2
            h_candidate[i] += perturbation
            
        # Clamp to [0, 1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Adjust to maintain sum = n_points/2
        current_sum = np.sum(h_candidate)
        diff = (n_points / 2) - current_sum
        
        if abs(diff) > 1e-8:
            # Distribute the difference among other points
            other_indices = [j for j in range(n_points) if j not in indices_to_change]
            if len(other_indices) > 0:
                avg_adjust = diff / len(other_indices)
                for j in other_indices:
                    h_candidate[j] += avg_adjust
                    h_candidate[j] = np.clip(h_candidate[j], 0, 1)
        
        # Normalize sum to exactly n_points/2
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute C5 for candidate
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(corr_candidate) * dx
        
        # Accept or reject
        delta = candidate_c5 - current_c5
        
        if delta < 0 or np.exp(-delta / temperature) > random.random():
            h = h_candidate.copy()
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        # Force normalization occasionally
        if iteration % 1000 == 0:
            h = h * (n_points / 2) / np.sum(h)
    
    # Refinement phase with local search
    if best_c5 > 0.3807:
        # Try 100 iterations of focused local search
        h_refined = best_h.copy()
        corr_refined = np.correlate(h_refined, 1 - h_refined, mode='full')
        c5_refined = np.max(corr_refined) * dx
        
        for _ in range(100):
            h_test = h_refined.copy()
            i = np.random.randint(0, n_points)
            
            # Perturb this index
            perturbation = (random.random() - 0.5) * 0.05
            h_test[i] += perturbation
            
            h_test = np.clip(h_test, 0, 1)
            
            # Normalize
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            c5_test = np.max(corr_test) * dx
            
            if c5_test < c5_refined:
                h_refined = h_test.copy()
                c5_refined = c5_test
                
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Compute final C5 for return
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
