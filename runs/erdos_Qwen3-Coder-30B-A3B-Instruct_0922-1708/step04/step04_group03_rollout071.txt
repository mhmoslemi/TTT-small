import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    epsilon = 1e-4
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to ensure sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Learning rate and decay
    lr = 0.1
    decay_rate = 0.99
    decay_interval = 50
    
    # Early stopping
    max_no_improvement = 100
    no_improvement_count = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap for all lags
        overlap = np.correlate(h, 1 - h, mode='full')
        max_overlap_val = np.max(overlap) * dx
        
        if max_overlap_val < best_c5:
            best_c5 = max_overlap_val
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Compute gradient for each point
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Increase h[i] slightly
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Decrease h[i] slightly
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Estimate derivative
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update h
        h = h - lr * grad
        
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Decay learning rate
        if iteration % decay_interval == 0:
            lr *= decay_rate
            
    return (best_h.tolist(), best_c5, n_points)
