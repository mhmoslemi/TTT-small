import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    population_size = 50
    elite_size = 10
    mutation_rate = 0.1
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    population = []
    for _ in range(population_size):
        h_test = h + np.random.normal(0, 0.05, n_points)
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        population.append(h_test)
        
    best_h = h.copy()
    best_c5 = float('inf')
    t = 0
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Genetic algorithm
    while time.time() - start_time < budget_s - 1 and t < max_iter:
        t += 1
        # Evaluate fitness
        fitness = []
        for h_ind in population:
            c5 = compute_c5(h_ind)
            fitness.append(c5)
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best found solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        # Selection
        selected = population[:elite_size]
        
        # Crossover and mutation
        offspring = []
        while len(offspring) < population_size - elite_size:
            # Tournament selection
            parent1 = population[np.random.randint(0, elite_size)]
            parent2 = population[np.random.randint(0, elite_size)]
            
            # SBX crossover
            alpha = np.random.rand(n_points)
            child1 = alpha * parent1 + (1 - alpha) * parent2
            child2 = (1 - alpha) * parent1 + alpha * parent2
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child1[i] += np.random.normal(0, 0.05)
                        
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child2[i] += np.random.normal(0, 0.05)
            
            # Repair
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            offspring.extend([child1, child2])
            
        # Combine elites and offspring
        population = selected + offspring[:population_size - elite_size]
        
        # Adaptive mutation rate
        if t % 100 == 0:
            mutation_rate *= 0.95
        
    # Final refinement
    h_refined = best_h.copy()
    best_c5_refined = best_c5
    
    # Line search around good solution
    for _ in range(200):
        h_test = h_refined + np.random.normal(0, 0.02, n_points)
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        c5_test = compute_c5(h_test)
        if c5_test < best_c5_refined:
            best_c5_refined = c5_test
            h_refined = h_test.copy()
            
    return (h_refined.tolist(), best_c5_refined, n_points)
