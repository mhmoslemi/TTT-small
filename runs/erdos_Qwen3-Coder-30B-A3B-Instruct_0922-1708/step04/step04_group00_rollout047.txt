import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Use scipy optimization for more sophisticated approach.
    # Since we're in the frequency domain, we'll try a few strategies
    
    def compute_c5(h_values):
        h = np.array(h_values)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def objective(h_values):
        return compute_c5(h_values)
    
    # Constraints
    def constraint_integral(h_values):
        return np.sum(h_values) * dx - 1.0
    
    def constraint_bounds(h_values):
        # This ensures that bounds on h are satisfied
        return np.array([h_values[i] - 1.0 for i in range(len(h_values))])  # h <= 1
    
    def constraint_bounds_low(h_values):
        return np.array([-h_values[i] for i in range(len(h_values))])  # h >= 0
    
    # Initial optimization with trust-constr method
    constraints = [
        {'type': 'eq', 'fun': constraint_integral},
        {'type': 'ineq', 'fun': constraint_bounds_low},
        {'type': 'ineq', 'fun': constraint_bounds}
    ]
    
    # Start time tracking for the main optimization loop
    optimize_start_time = time.time()
    
    # Initial run
    res = minimize(
        objective,
        h,
        method='trust-constr',
        constraints=constraints,
        options={'maxiter': 100},
        tol=1e-6
    )
    
    if res.success:
        optimized_h = res.x
        optimized_c5 = compute_c5(optimized_h)
        
        if optimized_c5 < best_c5:
            best_c5 = optimized_c5
            best_h = optimized_h.copy()
    
    # Try a few PSO iterations with the optimized result as starting point
    n_particles = 30
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    vmax = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Start from best solution or a small random perturbation
    base_h = best_h.copy() if best_c5 < float('inf') else h
    base_h = np.clip(base_h, 0, 1)
    base_h = base_h * (n_points / 2) / np.sum(base_h)
    
    for _ in range(n_particles):
        if np.random.rand() < 0.3 and 'initial_h_values' in globals():
            particle_h = base_h.copy()
        else:
            particle_h = np.random.uniform(0, 1, n_points)
            particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
            particle_h = np.clip(particle_h, 0, 1)
        particles.append(particle_h)
        velocities.append(np.random.uniform(-vmax, vmax, n_points))
    
    gbest_c5 = float('inf')
    gbest = base_h.copy()
    
    max_iter = 200
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Ensure particle is valid
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Compute fitness
            c5 = compute_c5(particle)
            
            # Update global best
            if c5 < gbest_c5:
                gbest = particle.copy()
                gbest_c5 = c5
                
            # Update best solution overall
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
                
            # Update velocity and position for real particles as in PSO
            r1, r2 = np.random.rand(2, n_points)
            vel = (inertia * vel + 
                   c1 * r1 * (gbest - particle) + 
                   c2 * r2 * (gbest - particle))
            
            vel = np.clip(vel, -vmax, vmax)
            particle += vel
            
            # Enforce constraints again
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            velocities[i] = vel
            particles[i] = particle
    
    # Final refinement using a local optimization
    refinement_method = 'L-BFGS-B'
    
    res = minimize(
        objective,
        best_h,
        method=refinement_method,
        constraints=constraints,
        options={'maxiter': 100},
        tol=1e-7
    )
    
    if res.success:
        refined_h = res.x
        refined_c5 = compute_c5(refined_h)
        
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
