        new_individual = individual.copy()
        for i in range(len(new_individual)):
            if np.random.rand() < mutation_rate:
                new_individual[i] = np.clip(new_individual[i] + np.random.normal(0, 0.05), 0, 1)
        # Renormalize
        new_individual = new_individual * (n_points / 2) / np.sum(new_individual)
        return new_individual

        correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(correlation) * dx
