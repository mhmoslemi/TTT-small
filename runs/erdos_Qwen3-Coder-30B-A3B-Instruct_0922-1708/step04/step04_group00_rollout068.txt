import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Trust region optimization
    max_iter = 500
    trust_radius = 0.1
    trust_radius_decay = 0.99
    min_trust_radius = 1e-6
    
    # Function to compute C5
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # Function to compute gradient using finite differences
    def compute_gradient(h_vals, eps=1e-4):
        grad = np.zeros_like(h_vals)
        base_c5 = compute_c5(h_vals)
        
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
            
        return grad
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = compute_c5(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Solve trust region subproblem
        # Quadratic model: q(p) = grad^T * p + 0.5 * p^T * H * p
        # We will use a simplified trust region method without Hessian
        
        # Simple gradient descent with projected step
        step = -grad * 0.01
        h_new = h + step
        
        # Project step to stay within trust region
        step_norm = np.linalg.norm(step)
        if step_norm > trust_radius:
            step = step * trust_radius / step_norm
            h_new = h + step
        
        # Enforce constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept step if improves objective
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5:
            h = h_new.copy()
        else:
            # Decrease trust region
            trust_radius *= trust_radius_decay
            if trust_radius < min_trust_radius:
                trust_radius = min_trust_radius
                
        # Continue until improvement or trust radius small
        if trust_radius < min_trust_radius:
            break
    
    # Final refinement with simple gradient descent
    h_final = best_h.copy()
    lr = 0.005
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        grad = compute_gradient(h_final)
        h_new = h_final - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < compute_c5(h_final):
            h_final = h_new
            lr *= 0.99
        else:
            lr *= 0.9
            
        if np.abs(new_c5 - compute_c5(h_final)) < 1e-10:
            break
    
    # Final check
    c5_final = compute_c5(h_final)
    return (h_final.tolist(), c5_final, n_points)
