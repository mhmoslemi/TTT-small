import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Objective function and gradient
    def obj_and_grad(h_vec):
        # Normalize
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        h_vec = np.clip(h_vec, 0, 1)
        
        # Compute correlation
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5, h_vec
    
    # Gradient estimation using finite differences
    def compute_gradient(h_vec, epsilon=1e-4):
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        h_vec = np.clip(h_vec, 0, 1)
        c5, _ = obj_and_grad(h_vec)
        grad = np.zeros_like(h_vec)
        
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += epsilon
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_plus = np.clip(h_plus, 0, 1)
            
            c5_plus, _ = obj_and_grad(h_plus)
            grad[i] = (c5_plus - c5) / epsilon
            
        return grad
    
    # Optimization loop
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 and gradient
        c5_val, _ = obj_and_grad(h)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            
        # Estimate gradient
        grad = compute_gradient(h)
        
        # Use trust-region reflective method for optimization
        # We regard each point as an independent variable with bounds [0,1]
        # and the constraint that sum = n_points/2
        
        # Form constraints for scipy.optimize
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Define bounds for each variable
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refine with scipy.optimize.minimize using trust-constr
        try:
            result = minimize(
                lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
                x0=h,
                method='trust-constr',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 10, 'disp': False}
            )
            
            if result.success:
                h = result.x
                # Clamp values and normalize
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # If optimization failed, do simple gradient descent step with small step
                lr = 0.005
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
        except:
            # Fallback: simple gradient descent
            lr = 0.01
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
    # Final validation
    c5_val, _ = obj_and_grad(best_h)
    
    return (best_h.tolist(), c5_val, n_points)
