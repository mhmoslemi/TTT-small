import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():

        if len(h) != n_points:

            x_orig = np.linspace(0, 2, len(h))

            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # DE parameters
    pop_size = 50
    F = 0.8
    CR = 0.9
    max_iter = 500
    no_improvement_limit = 100
    
    # Initialize population
    population = []
    fitness = []
    
    # Create initial population with constraint enforcement
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            candidate = h.copy()
        else:
            candidate = np.random.uniform(0, 1, n_points)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
        population.append(candidate)
        # Evaluate fitness
        corr = np.correlate(candidate, 1 - candidate, mode='full')
        fit_val = np.max(corr) * dx
        fitness.append(fit_val)
    
    # Find best in initial population
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Track no improvement
    no_improvement = 0
    
    start_time = time.time()
    
    # Main DE loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        new_population = []
        new_fitness = []
        
        for i in range(pop_size):
            # Select three distinct individuals
            candidates = list(range(pop_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[a] + F * (population[b] - population[c])
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.random() < CR:
                    trial[j] = mutant[j]
            
            # Enforce constraints
            trial = np.clip(trial, 0, 1)
            trial = trial * (n_points / 2) / np.sum(trial)
            
            # Evaluate trial
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_c5 = np.max(corr) * dx
            
            # Selection
            if trial_c5 < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_c5)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
            
            # Update best if improved
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = trial.copy()
                no_improvement = 0
        
        # Update population
        population = new_population
        fitness = new_fitness
        
        # Local refinement every 20 iterations
        if iteration % 20 == 0:
            # Refine best
            h_refined = best_h.copy()
            for ref_iter in range(50):
                if time.time() - start_time > budget_s - 1:
                    break
                corr = np.correlate(h_refined, 1 - h_refined, mode='full')
                c5 = np.max(corr) * dx
                
                # Gradient estimation via finite differences
                grad = np.zeros(n_points)
                eps = 1e-4
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5) / eps
                
                # Adaptive learning rate
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h_refined - lr * grad
                
                # Clip and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                corr_test = np.correlate(h_new, 1 - h_new, mode='full')
                c5_test = np.max(corr_test) * dx
                
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_new.copy()
                
                h_refined = h_new
                
        # Early stopping
        no_improvement += 1
        if no_improvement >= no_improvement_limit:
            break
    
    return (best_h.tolist(), best_c5, n_points)
