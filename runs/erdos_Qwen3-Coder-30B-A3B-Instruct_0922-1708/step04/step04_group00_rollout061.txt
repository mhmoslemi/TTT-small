import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Solve using LP relaxation
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Create variables for the convex optimization formulation
    h_var = cp.Variable(n_points)
    C5_var = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, 
                   cp.sum(h_var) * dx == 1, 
                   C5_var >= 0]
    
    # Build shifted correlation using auxiliary variables
    # Create auxiliary variables for h[i] * h[i+k]
    for k in range(1, 2*n_points-1):
        # Create variable w representing h[i] * h[i+k]
        if k <= n_points-1:
            # Positive shift: w[i] = h[i] * h[i+k]
            # So h[i] - w[i] >= 0 and w[i] <= h[i], w[i] <= h[i+k]
            # Add constraint: sum(h[i] - w[i]) * dx <= C5
            # But since correlations are full, handle them strategically
            # We'll compute this differently - full convolution of h and (1-h)
            
            # So instead, solve the optimization problem directly
            # For now, use gradient descent refinement of a good starting solution
            
            pass
    
    # Use LP for lower bound estimation (simpler approach)
    # We can solve this simpler formulation with CVXPY
    
    # For a simpler and more robust approach, solve with local search only,
    # using the previous good solution as initialization

    def compute_c5(h_vec):
        # Compute the full cross-correlation 
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # First try the simplest refinement approach from initial solution
    h_trial = h.copy()
    
    try:
        # Use scipy minimize to refine the solution
        def objective(h_vec):
            return compute_c5(h_vec)
        
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_func}
        
        # bounds = [(0, 1) for _ in range(n_points)]
        
        # Perform optimization with bounds
        result = minimize(objective, h_trial, 
                         method='SLSQP', 
                         bounds=[(0,1) for _ in range(n_points)],
                         constraints=cons,
                         options={'maxiter': 500, 'ftol': 1e-9})
        
        # If successful, extract result
        if result.success:
            h_sol = result.x.copy()
            h_sol = np.clip(h_sol, 0, 1)
            # Normalize 
            h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
            
            # Compute C5
            c5_sol = compute_c5(h_sol)
            
            if c5_sol < best_c5:
                best_c5 = c5_sol
                best_h = h_sol.copy()
        
    except Exception as e:
        # If optimization fails, keep the original
        pass
    
    # Also do a final gradient-based refinement
    h_final = best_h.copy()
    max_iter = 1000
    lr = 0.01
    
    current_c5 = compute_c5(h_final)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute gradient using finite differences
        eps = 1e-4
        grad = np.zeros_like(h_final)
        
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[j] = (c5_plus - current_c5) / eps
            
        # Update
        h_new = h_final - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        if new_c5 < current_c5:
            h_final = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_final.copy()
            
        # Reduce learning rate
        lr = max(1e-6, lr * 0.999)
        
        # Early stopping
        if abs(current_c5 - best_c5) < 1e-9:
            break

    return (best_h.tolist(), best_c5, n_points)
