import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Iterative convex optimization approach
    max_iter = 100
    tol = 1e-5
    step_size = 0.5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Try to solve convex relaxation
        h_old = h.copy()
        
        # Create variables for convex optimization
        h_var = cp.Variable(n_points)
        C_var = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
        ]
        
        # Add linearized constraints for each shift
        for k in range(1, n_points):
            # Linear approximation of h[i] * (1 - h[i+k]) 
            # around h_old[i], h_old[i+k]
            shift_constraint = 0
            # Compute actual shift indices
            for i in range(n_points):
                if i + k < n_points:
                    shift_constraint += h_var[i] * (1 - h_old[i+k]) + h_old[i] * (1 - h_var[i+k]) - h_old[i] * (1 - h_old[i+k])
                else:
                    # This should be handled when we do correlation shifting correctly
                    # For now, ignore improperly indexed terms or handle as zero
                    pass
                    
            # Only add meaningful constraint:
            if k < n_points:
                # For proper correlation, the terms are:
                # ∑_{i=0}^{n-k-1} h[i] * (1 - h[i+k]) 
                # We create a proxy patch for these terms by computing 
                # for shift k, we have indices from 0 to n-k-1 for first h,
                # from k to n-1 for second h
                approx = 0
                for i in range(n_points - k):
                    approx += h_var[i] * (1 - h_old[i+k]) + h_old[i] * (1 - h_var[i+k]) - h_old[i] * (1 - h_old[i+k])
                constraints.append(approx <= C_var)
        
        # Objective
        objective = cp.Minimize(C_var)
        
        # Solve with ECOS (or SCS)
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, abstol=1e-4, reltol=1e-4, feastol=1e-4)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array(h_var.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            else:
                # fallback to gradient descent-based update with small step
                h_new = h + step_size * (h_old - h)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
        except:
            # Fallback to gradient-descent like update
            h_new = h + step_size * (h_old - h)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
        h = h_new
        
        # Line search portion if needed
        # Try multiple steps of linesearch with backtracking
        step = 0.5
        for _ in range(5):
            h_test = h + step * (h_old - h)
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            corr = np.correlate(h_test, 1 - h_test, mode='full')
            c5_test = np.max(corr) * dx
            if c5_test < current_c5 and c5_test < best_c5:
                h = h_test
                break
            step *= 0.5
            
        # Companion check for convergence
        if np.linalg.norm(h - h_old) < tol:
            break
            
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
