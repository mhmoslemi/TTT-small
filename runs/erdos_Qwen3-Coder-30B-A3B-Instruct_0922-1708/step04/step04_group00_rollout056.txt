import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Project h to satisfy constraints
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Objective function
    def objective(h_vec):
        h_vec = project_h(h_vec)
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Gradient computation via finite differences
    def compute_gradient(h_vec, eps=1e-5):
        grad = np.zeros_like(h_vec)
        base_obj = objective(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            grad[i] = (objective(h_plus) - base_obj) / eps
        return grad
    
    # Trust region optimization with TRF method
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current objective and gradient
        current_obj = objective(h)
        
        # Update best solution
        if current_obj < best_c5:
            best_c5 = current_obj
            best_h = h.copy()
        
        # Compute gradient (optional, may need refinement)
        try:
            grad = compute_gradient(h)
        except:
            grad = np.zeros_like(h)
        
        # Use scipy's trust-region reflective method
        try:
            # Construct bounds
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Optimize with TRF method
            res = minimize(
                objective, 
                h, 
                method='TRF',
                bounds=bounds,
                options={'maxiter': 10, 'ftol': 1e-6, 'gtol': 1e-6}
            )
            
            if res.success:
                h = project_h(res.x)
            else:
                h = project_h(h - 0.01 * grad)
                h = project_h(h)
                
        except:
            # Fallback to gradient descent step
            h = project_h(h - 0.01 * grad)
    
    # Final projection and validation
    h = project_h(h)
    final_obj = objective(h)
    if final_obj < best_c5:
        best_c5 = final_obj
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
