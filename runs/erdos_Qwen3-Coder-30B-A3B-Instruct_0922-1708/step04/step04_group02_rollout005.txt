import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or generate random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Relaxation stage - Convex optimization
    while time.time() - start_time < budget_s * 0.4 and best_c5 > 0.3807:
        try:
            h_relaxed = cp.Variable(n_points)
            constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points / 2]
            # We'll approximate the overlap constraint by considering a set of large enough lags
            # and using a relaxed formulation of the maximum over all k to be tightened later
            objective = cp.Minimize(cp.sum(cp.multiply(h_relaxed, 1 - h_relaxed)))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_relaxed = np.array(h_relaxed.value).flatten()
                # Rounding to binary via threshold median
                t = np.median(h_relaxed)
                h_candidate = (h_relaxed > t).astype(float)
                # Adjust sum to required value
                current_sum = np.sum(h_candidate)
                diff = n_points / 2 - current_sum
                if diff > 0:
                    indices = np.where(h_candidate == 0)[0]
                    np.random.shuffle(indices)
                    for i in range(int(diff)):
                        if i < len(indices):
                            h_candidate[indices[i]] = 1
                elif diff < 0:
                    indices = np.where(h_candidate == 1)[0]
                    np.random.shuffle(indices)
                    for i in range(int(-diff)):
                        if i < len(indices):
                            h_candidate[indices[i]] = 0
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                
                corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5 = np.max(corr) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_candidate.copy()
                    
                # Post-processing - neighborhood search
                if time.time() - start_time < budget_s * 0.7:
                    neighborhood_size = max(1, int(n_points * 0.02))
                    for _ in range(10):
                        if time.time() - start_time > budget_s * 0.7:
                            break
                        h_test = h_candidate.copy()
                        indices = np.random.choice(n_points, neighborhood_size, replace=False)
                        for i in indices:
                            h_test[i] = np.clip(h_test[i] + np.random.normal(0, 0.01), 0, 1)
                        h_test = h_test * (n_points / 2) / np.sum(h_test)
                        corr_test = np.correlate(h_test, 1 - h_test, mode='full')
                        c5_test = np.max(corr_test) * dx
                        if c5_test < best_c5:
                            best_c5 = c5_test
                            best_h = h_test.copy()
        except:
            pass
    
    # If we haven't improved enough, do a standard gradient-based descent
    if best_c5 > 0.3807 and time.time() - start_time < budget_s * 0.8:
        h = best_h.copy()
        # Use scipy.optimize minimize with L-BFGS-B for constrained optimization
        def objective(h_vec):
            # Rescale h_vec to sum to n_points / 2
            h_rescaled = h_vec * (n_points / 2) / np.sum(h_vec)
            corr = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
            return -np.max(corr) * dx  # Negative for minimization
        
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        # Keep bounds on h_vec
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_func}
        
        try:
            result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraints,
                              options={'maxiter': 100})
            if result.success:
                h_opt = result.x * (n_points / 2) / np.sum(result.x)
                corr_opt = np.correlate(h_opt, 1 - h_opt, mode='full')
                c5_opt = np.max(corr_opt) * dx
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
    
    # Finally a grid search refinement towards best solution
    if time.time() - start_time < budget_s:
        h_final = best_h.copy()
        c5_final = best_c5
        for _ in range(200):
            if time.time() - start_time > budget_s - 1:
                break
            h_test = h_final.copy()
            # Random small perturbations
            try:
                indices = np.random.choice(n_points, size=min(20, n_points//5), replace=False)
                for i in indices:
                    h_test[i] = np.clip(h_test[i] + np.random.normal(0, 0.01), 0, 1)
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                corr_test = np.correlate(h_test, 1 - h_test, mode='full')
                c5_test = np.max(corr_test) * dx
                if c5_test < c5_final:
                    c5_final = c5_test
                    h_final = h_test.copy()
            except:
                continue
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    return (best_h.tolist(), best_c5, n_points)
