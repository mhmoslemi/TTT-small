import numpy as np
import time

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def golden_section_search(f, a, b, tol=1e-5, max_iter=100):
    gr = (1 + np.sqrt(5)) / 2
    res = gr - 1
    c = a + res * (b - a)
    d = b - res * (b - a)
    for _ in range(max_iter):
        if np.abs(c - d) < tol:
            break
        if f(c) < f(d):
            b = d
            d = c
            c = a + res * (b - a)
        else:
            a = c
            c = d
            d = b - res * (b - a)
    return (a + b) / 2

def gradient_step(h, eps=1e-4):
    n = len(h)
    grad = np.zeros(n)
    c5_base = compute_c5(h)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n)
        c5_plus = compute_c5(h_plus)
        grad[i] = (c5_plus - c5_base) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 150
    dx = 2.0 / n_points

    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h) != n_points:

            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = project_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Outer loop: grid search
    N_coarse = 100
    n_low = 50
    n_high = 150
    
    # Coarse sampling
    coarse_h = []
    for _ in range(N_coarse):
        if time.time() - start_time > budget_s - 10:
            break
        h_temp = np.random.rand(n_low)
        h_temp = project_feasible(h_temp, n_low)
        c5 = compute_c5(h_temp)
        coarse_h.append((h_temp, c5))
    
    # Sort and take top performers
    coarse_h.sort(key=lambda x: x[1])
    top_coarse = coarse_h[:N_coarse // 10]
    
    # Refine top performers
    for h_temp, _ in top_coarse:
        if time.time() - start_time > budget_s - 5:
            break
        # Interpolate to fine resolution
        x_orig = np.linspace(0, 2, n_low)
        x_new = np.linspace(0, 2, n_high)
        h_fine = np.interp(x_new, x_orig, h_temp)
        h_fine = project_feasible(h_fine, n_high)
        
        # Local gradient descent
        lr = 0.001
        for _ in range(100):
            if time.time() - start_time > budget_s - 2:
                break
            grad = gradient_step(h_fine)
            h_fine -= lr * grad
            h_fine = project_feasible(h_fine, n_high)
            new_c5 = compute_c5(h_fine)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_fine.copy()
            lr *= 0.995  # decay learning rate

        # Random noise injection to escape local minima
        if time.time() - start_time < budget_s - 2:
            noise = np.random.normal(0, 0.001, n_high)
            h_fine = np.clip(h_fine + noise, 0, 1)
            h_fine = project_feasible(h_fine, n_high)
            new_c5 = compute_c5(h_fine)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_fine.copy()

    # Final validation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
