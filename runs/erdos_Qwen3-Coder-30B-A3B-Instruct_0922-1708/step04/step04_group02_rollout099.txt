import numpy as np
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h using available parent or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Initialize temperature
    temp = 1.0
    cooling_rate = 0.99
    iterations = 0
    max_iter = 10000
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        iterations += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate perturbed solution
        h_new = h.copy()
        
        # Select random points to perturb (10% of total)
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Perturb selected points
        for i in indices:
            # Random perturbation with bounded magnitude
            pert = np.random.uniform(-0.1, 0.1)
            h_new[i] += pert
            
        # Maintain bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain integral constraint
        sum_diff = np.sum(h_new) - n_points / 2
        if abs(sum_diff) > 1e-8:
            # Redistribute the difference among non-perturbed indices
            non_perturbed = set(range(n_points)) - set(indices)
            if len(non_perturbed) > 0:
                adjust_per_point = sum_diff / len(non_perturbed)
                for i in non_perturbed:
                    h_new[i] -= adjust_per_point

        # Clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept or reject new solution based on simulated annealing
        new_c5 = compute_c5(h_new)
        delta = new_c5 - current_c5
        
        # Accept if better or with some probability
        if delta < 0 or np.random.random() < math.exp(-delta / temp):
            h = h_new.copy()
            
        # Cool down temperature
        if iterations % 100 == 0:
            temp *= cooling_rate
            
    # Final local refinement
    try:
        refinement_h = best_h.copy()
        refinement_c5 = best_c5
        for _ in range(500):
            # Make small step changes
            h_trial = refinement_h.copy()
            idx = np.random.randint(0, n_points)
            perturbation = np.random.uniform(-0.01, 0.01)
            h_trial[idx] += perturbation
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            trial_c5 = compute_c5(h_trial)
            if trial_c5 < refinement_c5:
                refinement_h = h_trial
                refinement_c5 = trial_c5
        best_h = refinement_h
        best_c5 = refinement_c5
    except:
        pass
        
    return (best_h.tolist(), best_c5, n_points)
