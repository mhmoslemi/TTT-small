import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation using CVXPY
    # Formulate a relaxed problem to get a good initial solution
    # Try to minimize sum of all overlaps (this is convex)
    try:
        # Create the matrix for overlap sums
        # C[i,j] = h[i] * (1 - h[j]) for all valid i,j
        # Since we're linearizing in convex approach, we'll be more nuanced
        
        # For the convex relaxation, minimize the sum of all overlaps
        # Not directly applicable as constraints define a complex non-convex objective
        # So we'll proceed with projected gradient method on the best current solution
        
        # Phase 2: Projected gradient descent
        max_iter = int(budget_s * 10)  # Rough estimate of iterations
        patience = 50
        no_improvement = 0
        prev_best = float('inf')
        
        iteration = 0
        # Account for initial time
        iteration_time = time.time() - start_time
        while iteration_time < budget_s - 2 and iteration < max_iter:
            iteration += 1
            if iteration % 100 == 0:
                # Gradually decrease learning rate
                lr = 0.01 / (1 + iteration / 1000.0) 
            else:
                lr = 0.01
            
            # Compute current objective value and gradient
            # Correlation gives us the overlaps for different shifts
            correlation = np.correlate(h, 1 - h, mode='full')
            c5_current = np.max(correlation) * dx
            
            # Gradient computation 
            grad = np.zeros_like(h)
            eps = 1e-6  # Slightly smaller epsilon for better approximation
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                # Respect bounds
                h_plus = np.clip(h_plus, 0, 1)
                # Rescale to maintain constraint sum == n_points / 2
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5_current) / eps
                
            # Update step
            h_new = h - lr * grad
            
            # Project to bounds
            h_new = np.clip(h_new, 0, 1)
            # Rescale to satisfy the integral constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update h
            h = h_new
            
            # Evaluate new solution
            corr = np.correlate(h, 1 - h, mode='full')
            c5_new = np.max(corr) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
                
            if no_improvement >= patience:
                break
                
            iteration_time = time.time() - start_time
                
    except Exception as e:
        pass  # If we cannot do convex or gradient methods, fall back to the initial
        
    # Final validation 
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Make sure it's a valid solution 
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
