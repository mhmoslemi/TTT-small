import numpy as np
from scipy.stats import qmc
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Generate Latin Hypercube samples
    sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs_samples = sampler.random(n=50)
    
    # Convert LHS to feasible solutions
    feasible_samples = []
    for sample in lhs_samples:
        # Transform to [0,1] range using inverse CDF
        sample = np.sort(sample)  # Sort for better distribution
        # Map to [0,1] with proper scaling
        sample = qmc.scale(sample, 0, 1)
        # Scale to satisfy sum constraint
        scaled_sample = sample * (n_points / 2) / np.sum(sample)
        # Clip to [0,1]
        scaled_sample = np.clip(scaled_sample, 0, 1)
        # Final rescale to ensure sum constraint exactly
        scaled_sample = scaled_sample * (n_points / 2) / np.sum(scaled_sample)
        feasible_samples.append(scaled_sample)
    
    # Evaluate all initial samples and refine
    for candidate in feasible_samples:
        h = candidate.copy()
        # Gradient descent refinement
        for iter_num in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute overlap integral
            c5 = 0
            max_corr = -np.inf
            
            # Instead of full correlation, we compute the max shift correlation directly
            for k in range(-n_points+1, n_points):
                if k == 0:
                    continue
                shift = abs(k)
                if k > 0:
                    # Left shift
                    corr = np.sum(h[:-shift] * (1 - h[shift:]))
                else:
                    # Right shift
                    corr = np.sum(h[shift:] * (1 - h[:-shift]))
                max_corr = max(max_corr, corr)
                
            c5 = max_corr * dx
            
            # Finite difference gradient
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = 0
                max_corr_plus = -np.inf
                
                for k in range(-n_points+1, n_points):
                    if k == 0:
                        continue
                    shift = abs(k)
                    if k > 0:
                        corr = np.sum(h_plus[:-shift] * (1 - h_plus[shift:]))
                    else:
                        corr = np.sum(h_plus[shift:] * (1 - h_plus[:-shift]))
                    max_corr_plus = max(max_corr_plus, corr)
                corr_plus = max_corr_plus * dx
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                corr_minus = 0
                max_corr_minus = -np.inf
                
                for k in range(-n_points+1, n_points):
                    if k == 0:
                        continue
                    shift = abs(k)
                    if k > 0:
                        corr = np.sum(h_minus[:-shift] * (1 - h_minus[shift:]))
                    else:
                        corr = np.sum(h_minus[shift:] * (1 - h_minus[:-shift]))
                    max_corr_minus = max(max_corr_minus, corr)
                corr_minus = max_corr_minus * dx
                
                grad[i] = (corr_plus - corr_minus) / (2 * eps)
            
            # Update with adaptive learning rate
            lr = 0.01 / (1.0 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Update best if improvement
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
    return (best_h.tolist(), best_c5, n_points)
