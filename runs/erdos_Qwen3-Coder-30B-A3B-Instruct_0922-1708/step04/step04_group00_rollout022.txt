import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient-based optimization using scipy
    def objective(h_vec):
        # Reshape h_vec
        h = h_vec.reshape(-1)
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute correlation using numpy's correlate
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def constraint_func(h_vec):
        h = h_vec.reshape(-1)
        return np.sum(h) * dx - 1.0
    
    # Define bounds for optimization (h in [0,1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    max_iter = 1000
    max_no_improvement = 50
    no_improvement = 0
    prev_c5 = float('inf')
    
    # Initial guess
    x0 = h.copy()
    
    # Apply scipy optimization
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            result = minimize(
                objective, 
                x0, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            
            if result.success:
                x0 = result.x
                c5 = objective(x0)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = x0.reshape(-1).copy()
                    no_improvement = 0
                else:
                    no_improvement += 1
                    
                # Early stopping if no improvement for too long
                if no_improvement >= max_no_improvement:
                    break
                
                # Check if we've already beaten the target
                if best_c5 <= 0.3807:
                    break
                    
        except Exception as e:
            pass  # Skip this iteration if optimization fails
        
        # Add small random perturbation sometimes to escape local minima
        if iter_num % 20 == 0 and iter_num > 0:
            perturb = np.random.normal(0, 0.01, n_points)
            x0 = np.clip(x0 + perturb, 0, 1)
            x0 = x0 * (n_points / 2) / np.sum(x0)
    
    # Final evaluation
    final_h = best_h.reshape(-1)
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (final_h.tolist(), final_c5, n_points)
