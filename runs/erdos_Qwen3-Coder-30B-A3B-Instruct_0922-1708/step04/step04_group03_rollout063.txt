import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Convex relaxation approach
    # We minimize sum of overlaps instead of max overlap for a starting point
    # Create constraint matrix for sum(h) = n_points/2 and h >= 0
    h = h.copy()
    
    # Relax to minimize sum of overlaps, not max
    def sum_overlaps(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.sum(corr) * dx

    # Use CVXPY for initial convex relaxation
    try:
        h_cvx = cp.Variable(n_points)
        constraints = [0 <= h_cvx, h_cvx <= 1, cp.sum(h_cvx) == n_points/2]
        objective = cp.Minimize(cp.sum(cp.convolve(h_cvx, 1-h_cvx, mode='full')) * dx)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = np.array(h_cvx.value).flatten()
    except:
        pass
    
    # Project to feasible region
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def eval_c5(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx
    
    def eval_grad(h, eps=1e-4):
        grad = np.zeros_like(h)
        c5_base = eval_c5(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = eval_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    # Projected gradient descent
    start_time = time.time()
    max_iter = 10000
    lr = 0.01
    patience = 50
    no_improvement_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= patience:
                break
                
        grad = eval_grad(h)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Adaptive learning rate
        if i % 100 == 0 and i > 0:
            lr *= 0.95
    
    # Final check
    final_c5 = eval_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (best_h.tolist(), best_c5, n_points)
