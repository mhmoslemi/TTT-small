import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Current best solution
    best_h = h.copy()
    correlation = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(correlation) * dx
    
    # Simulated annealing parameters
    temperature = 1.0
    cooling_rate = 0.99
    min_temp = 1e-6
    iterations = 0
    max_iterations = 10000
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iterations:
        # Generate candidate solution
        h_candidate = h.copy()
        
        # Select random points for perturbation (10%)
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        
        for i in indices:
            # Perturb h[i] by small amount
            perturbation = np.random.uniform(-0.1, 0.1)
            h_candidate[i] += perturbation
            
        # Clip values to [0, 1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Maintain sum constraint
        sum_diff = np.sum(h_candidate) - n_points / 2
        if abs(sum_diff) > 1e-8:
            # Adjust all points to maintain sum
            adjustment = sum_diff / n_points
            h_candidate -= adjustment
            # Ensure bounds
            h_candidate = np.clip(h_candidate, 0, 1)
        
        # Normalize to exact sum
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute new C5
        correlation_cand = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(correlation_cand) * dx
        
        # Accept or reject new solution
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / temperature):
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
        
        # Decrease temperature
        if iterations % 100 == 0:
            temperature = max(min_temp, temperature * cooling_rate)
            
        iterations += 1
    
    # Local refinement around best solution
    h_refined = best_h.copy()
    best_refined_c5 = best_c5
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_test = h_refined.copy()
        # Perturb a few random points
        for _ in range(5):
            i = np.random.randint(0, n_points)
            noise = np.random.uniform(-0.01, 0.01)
            h_test[i] += noise
        
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        corr = np.correlate(h_test, 1 - h_test, mode='full')
        c5_test = np.max(corr) * dx
        
        if c5_test < best_refined_c5:
            h_refined = h_test.copy()
            best_refined_c5 = c5_test
    
    # Final validation
    correlation_final = np.correlate(h_refined, 1 - h_refined, mode='full')
    c5_final = np.max(correlation_final) * dx
    
    return (h_refined.tolist(), c5_final, n_points)
