import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Set up CVXPY problem
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1]
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Add overlap constraints for all shifts
    for k in range(1, n_points):
        # Create shifted version of h_var
        shifted = cp.hstack([h_var[k:], cp.zeros(k)])
        # Compute the correlation sum
        correlation_sum = cp.sum(cp.multiply(h_var, 1 - shifted))
        constraints.append(correlation_sum <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    prob = cp.Problem(objective, constraints)
    
    start_time = time.time()
    
    # Solve with ECOS solver
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        else:
            h_opt = h
    except:
        h_opt = h
    
    # Final computation of C5
    h_final = h_opt
    correlation = np.correlate(h_final, 1 - h_final, mode='full')
    c5_bound = np.max(correlation) * dx
    
    return (h_final.tolist(), c5_bound, n_points)
