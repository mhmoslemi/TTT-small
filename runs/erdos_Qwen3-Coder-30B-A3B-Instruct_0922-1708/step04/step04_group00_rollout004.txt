import numpy as np
from scipy.optimize import minimize
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Differential Evolution approach
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with DE
    try:
        result = differential_evolution(
            objective, 
            bounds, 
            maxiter=500,
            popsize=20,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            callback=lambda x, convergence: None,
            disp=False
        )
        
        if not result.success:
            xopt = h
        else:
            xopt = result.x
            
        # Ensure feasibility
        xopt = np.clip(xopt, 0, 1)
        xopt = xopt * (n_points / 2) / np.sum(xopt)
        
        # Evaluate final
        correlation = np.correlate(xopt, 1 - xopt, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_h = xopt.copy()
            best_c5 = c5
            
    except Exception:
        pass
    
    # Post-processing: Local search with Nelder-Mead on good candidates
    # Prepare starting points
    test_h = []
    if 'initial_h_values' in globals():
        test_h.append(np.array(initial_h_values))
    
    # Add a few random points
    for _ in range(5):
        h_random = np.random.uniform(0, 1, n_points)
        h_random = h_random * (n_points / 2) / np.sum(h_random)
        h_random = np.clip(h_random, 0, 1)
        test_h.append(h_random)
    
    # Add the previous result if it's good
    test_h.append(best_h)
    
    # Local refinement with Nelder-Mead
    for i, h_start in enumerate(test_h):
        if time.time() - start_time > budget_s - 1:
            break
            
        def local_objective(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            return c5
        
        # Ensure itSum constraint is met
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        try:
            # Local search using Nelder-Mead with limit of 200 iterations
            res = minimize(
                local_objective, 
                h_start, 
                method='Nelder-Mead',
                options={'maxiter': 200, 'adaptive': True},
                tol=1e-6
            )
            
            if res.success:
                h_best = res.x
                h_best = np.clip(h_best, 0, 1)
                h_best = h_best * (n_points / 2) / np.sum(h_best)
                
                correlation = np.correlate(h_best, 1 - h_best, mode='full')
                c5 = np.max(correlation) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_best.copy()
                    
        except Exception:
            pass
    
    # Final check with more modeling
    # Gradient-assisted local search
    h = best_h.copy()
    max_iter = 500
    lr = 1e-3
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation and maximum C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Gradient estimation
        eps = 1e-4
        grad = np.zeros_like(h)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5) / eps
        
        # Update using gradient
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update learning rate
        lr *= 0.999
        
        h = h_new
        
        # Update best if better
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    return (h_final.tolist(), best_c5, n_points)
