import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Trust region optimization with adaptive radius
    max_iter = 500
    trust_radius = 0.1
    decay_rate = 0.99
    min_radius = 1e-6
    
    # Iterative trust region with adaptive radius
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient approximation
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Gradient descent with trust region
        direction = -grad
        if np.linalg.norm(direction) > trust_radius:
            direction = direction * trust_radius / np.linalg.norm(direction)
            
        h_new = h + direction
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update trust region
        trust_radius *= decay_rate
        trust_radius = max(trust_radius, min_radius)
        
        h = h_new
        
    # Final refinement with local optimization
    refined_h = best_h.copy()
    max_refine = 500
    for _ in range(max_refine):
        if time.time() - start_time > budget_s - 1:
            break
            
        correlation = np.correlate(refined_h, 1 - refined_h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Gradient estimation
        grad = np.zeros_like(refined_h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = refined_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Take gradient step
        step_size = 0.01
        refined_h = refined_h - step_size * grad
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
        
        # Test improvement
        correlation = np.correlate(refined_h, 1 - refined_h, mode='full')
        new_c5 = np.max(correlation) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = refined_h.copy()
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
