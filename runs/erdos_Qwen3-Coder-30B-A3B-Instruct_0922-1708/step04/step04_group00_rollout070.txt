import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient-based optimization with constraints
    def objective(h_vec):
        # Ensure h is in [0,1]
        h = np.clip(h_vec, 0, 1)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute correlation
        correlation = np.correlate(h, 1-h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def constraint_func(h_vec):
        # Integral constraint
        h = np.clip(h_vec, 0, 1)
        return np.sum(h) * dx - 1.0
    
    # Use scipy.optimize with trust-constr
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Optimization parameters
    max_iter = 1000
    lr = 0.01
    patience = 20
    no_improvement = 0
    
    # Start with a smoothed version of initial
    h = best_h.copy()
    
    # Local gradient descent
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_prev = objective(h)
        
        # Compute finite differences gradient
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = objective(h_plus)
            
            grad[j] = (c5_plus - c5_prev) / eps
            
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update learning rate
        lr = max(1e-6, lr * 0.9995)
        
        h = h_new
        
        # Update best
        c5_new = objective(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > patience:
            break
    
    # Final verification and refinement
    h_final = best_h.copy()
    
    # Use constrained optimization as final refinement
    try:
        # Using approximate method to optimize with constraints
        h_cand = h_final.copy()
        for _ in range(50):
            c5_current = objective(h_cand)
            # Perturb a bit more carefully
            for j in range(n_points):
                h_test = h_cand.copy()
                h_test[j] = min(1.0, max(0.0, h_test[j] + np.random.normal(0, 0.01)))
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                c5_test = objective(h_test)
                if c5_test < c5_current:
                    h_cand = h_test.copy()
                    c5_current = c5_test
            if abs(objective(h_cand) - c5_current) < 1e-6:
                break
        best_h = h_cand
        best_c5 = objective(best_h)
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
