import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation approach: minimize sum of all overlaps
    # This provides a good initial guess
    h_relaxed = h.copy()
    try:
        # Use cvxpy for convex relaxation to minimize total overlap
        # This is slightly different from the original problem but should be close
        h_cvx = cp.Variable(n_points)
        constraints = [
            h_cvx >= 0,
            h_cvx <= 1,
            cp.sum(h_cvx) == n_points / 2
        ]
        
        # Minimize sum of all pairwise overlaps
        # This facilitates better generalization in the space
        total_overlap = 0
        for i in range(n_points):
            for j in range(n_points):
                k = (j - i) % n_points
                total_overlap += h_cvx[i] * (1 - h_cvx[j])
                
        objective = cp.Minimize(total_overlap)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_relaxed = np.array(h_cvx.value)
            h_relaxed = np.clip(h_relaxed, 0, 1)
            h_relaxed = h_relaxed * (n_points / 2) / np.sum(h_relaxed)
    except:
        pass  # Fall back to previous method if needed
    
    # Use the relaxed solution as starting point
    h = h_relaxed.copy()
    
    # Projected gradient descent with automatic learning rate
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-2)
        
        # Update
        h_new = h - lr * grad
        
        # Project back to constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final check of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
