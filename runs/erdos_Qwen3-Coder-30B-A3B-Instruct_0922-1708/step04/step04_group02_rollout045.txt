import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize using parent construction or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.full(n_points, 0.5)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimized evaluation function
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated annealing
    temp = 1.0
    cooling_factor = 0.99
    temp_drop_interval = 100
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb 10% of the points
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Create candidate solution
        h_candidate = h.copy()
        
        # Perturb selected points
        for i in indices:
            # Random perturbation within [0,1]
            perturbation = np.random.uniform(-0.1, 0.1)
            h_candidate[i] += perturbation
            
        # Ensure values stay in [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Fix the integral constraint
        current_sum = np.sum(h_candidate)
        adjustment = (n_points / 2) - current_sum
        if abs(adjustment) > 1e-10:
            # Distribute adjustment across all points
            h_candidate += adjustment / n_points
            
        # Ensure still in bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Final normalization
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute objective
        candidate_c5 = compute_c5(h_candidate)
        
        # Accept or reject based on simulated annealing
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_candidate
            current_c5 = candidate_c5
            
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Temperature schedule
        if iteration % temp_drop_interval == 0:
            temp *= cooling_factor
    
    # Final refinement around best solution
    refinement_attempts = 1000
    for _ in range(refinement_attempts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Small perturbation to random point
        i = np.random.randint(0, n_points)
        h_test = best_h.copy()
        h_test[i] += np.random.uniform(-0.01, 0.01)
        h_test = np.clip(h_test, 0, 1)
        
        # Keep the sum
        current_sum = np.sum(h_test)
        adjustment = (n_points / 2) - current_sum
        h_test += adjustment / n_points
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        test_c5 = compute_c5(h_test)
        if test_c5 < best_c5:
            best_c5 = test_c5
            best_h = h_test.copy()
    
    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
