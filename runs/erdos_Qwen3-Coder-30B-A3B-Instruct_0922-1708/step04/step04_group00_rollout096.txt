import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Use genetic algorithm with mutations
    population_size = 30
    generations = 100
    mutation_rate = 0.1
    elite_size = 5
    
    def evaluate_individual(individual):
        # Normalize the individual
        norm_individual = individual * (n_points / 2) / np.sum(individual)
        norm_individual = np.clip(norm_individual, 0, 1)
        
        correlation = np.correlate(norm_individual, 1 - norm_individual, mode='full')
        c5 = np.max(correlation) * dx
        return c5, norm_individual
    
    # Initialize population
    population = []
    fitnesses = []
    
    for _ in range(population_size):
        if 'initial_h_values' in globals() and np.random.rand() < 0.3:
            individual = h.copy()
        else:
            individual = np.random.uniform(0, 1, n_points)
        fitness, norm_individual = evaluate_individual(individual)
        population.append(norm_individual)
        fitnesses.append(fitness)
    
    # Sort by fitness
    sorted_indices = np.argsort(fitnesses)
    population = [population[i] for i in sorted_indices]
    fitnesses = [fitnesses[i] for i in sorted_indices]
    
    if fitnesses[0] < best_c5:
        best_c5 = fitnesses[0]
        best_h = population[0].copy()
    
    # Genetic algorithm iterations
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Elitism
        new_population = population[:elite_size]
        
        # Generate offspring through crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = np.random.randint(0, population_size // 2)
            parent2_idx = np.random.randint(0, population_size // 2)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover (uniform)
            child = np.where(np.random.rand(n_points) > 0.5, parent1, parent2)
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    # Mutate by adding small random change
                    child[i] += np.random.normal(0, 0.05)
            
            # Ensure bounds and normalization
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        # Evaluate new population
        new_fitnesses = []
        for individual in new_population:
            fitness, norm_individual = evaluate_individual(individual)
            new_fitnesses.append(fitness)
        
        # Sort by fitness
        sorted_indices = np.argsort(new_fitnesses)
        population = [new_population[i] for i in sorted_indices]
        fitnesses = [new_fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
    
    # Local refinement with trust-region optimization
    def obj_func(x):
        norm_x = x * (n_points / 2) / np.sum(x)
        norm_x = np.clip(norm_x, 0, 1)
        correlation = np.correlate(norm_x, 1 - norm_x, mode='full')
        return -np.max(correlation) * dx
    
    def constraint_func(x):
        return np.sum(x) - n_points / 2
    
    res = minimize(obj_func, best_h, 
                   method='SLSQP', 
                   constraints={'type': 'eq', 'fun': constraint_func},
                   bounds=[(0, 1) for _ in range(n_points)],
                   options={'maxiter': 100, 'ftol': 1e-9})
    
    if not res.success:
        # If optimization failed, revert to best found
        pass
    else:
        refined_h = np.clip(res.x * (n_points / 2) / np.sum(res.x), 0, 1)
        correlation = np.correlate(refined_h, 1 - refined_h, mode='full')
        c5_refined = np.max(correlation) * dx
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = refined_h
    
    return (best_h.tolist(), best_c5, n_points)
