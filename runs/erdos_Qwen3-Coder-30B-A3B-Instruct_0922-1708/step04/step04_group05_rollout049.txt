import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_c5(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx
    
    def project_feasible(h_values):
        h_proj = np.clip(h_values, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj
    
    def evaluate_gradient(h_values):
        grad = np.zeros_like(h_values)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_values.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h_values.copy()
            h_minus[i] -= eps
            h_minus = project_feasible(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    # Main optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Gradient descent step
        grad = evaluate_gradient(h)
        lr = 0.005 / (1.0 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        h_new = project_feasible(h_new)
        h = h_new
        
        # Occasionally add noise to escape local minima
        if iter_count % 30 == 0:
            noise = np.random.normal(0, 0.002, n_points)
            h = project_feasible(h + noise)
            
        # Occasionally generate new candidates from better solutions
        if iter_count % 100 == 0 and best_c5 < 0.5:
            # Take current best and slightly perturb
            h_candidate = best_h + np.random.normal(0, 0.005, n_points)
            h_candidate = project_feasible(h_candidate)
            if compute_c5(h_candidate) < best_c5:
                h = h_candidate
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
