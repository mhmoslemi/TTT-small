from scipy.optimize import minimize
import time

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize so that sum(h) * dx = 1, i.e., sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Helper function to compute C5
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Helper function to project h to feasible region
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Objective function with gradient estimation using finite differences
    def objective_and_gradient(h_vec):
        c5 = compute_c5(h_vec)
        eps = 1e-4
        
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps
            
        return c5, grad
    
    # Trust region optimization with projected gradients
    max_iter = 500
    opt_result = None
    try:
        # Use trust-region reflective method for constrained optimization  
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current value and gradient
            c5, grad = objective_and_gradient(h)
            
            # Update if better
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Optimization step with projection
            # Update using gradient descent with stepsize
            step_size = 0.1
            h_new = h - step_size * grad
            
            h = project_h(h_new)
            
    except Exception as e:
        pass  # fallback to best found

    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
