import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Solve with LP relaxation
    try:
        # LP relaxation approach
        h_vars = Variable(n_points)
        C5 = Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1]
        constraints.append(sum(h_vars) == n_points / 2)
        
        # Auxiliary variables for w(x, k) terms
        # For each k, we need sum(h(x) - w(x, k)) <= C5*dx
        # w(x, k) <= h(x), w(x, k) <= h(x + k), w(x, k) >= 0
        # We'll linearize this:
        # minimize C5
        # subject to:
        # sum(h) = n_points / 2
        # h >= 0, h <= 1
        # For each valid k, sum(h[x] - h[x]*h[x+k]) <= C5*dx (approximated)
        
        # A more direct approach, we directly solve this with CVXPY
        # Use variable approach directly
        h_val = h.copy()
        
        # Transformation approach: Isolate overlap term
        # Let's compute each shift and minimize maximum
        # We can try to solve a slightly different but equivalent relaxation
        
        # Try a simpler approach using cvxpy 
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Initial constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Compute all correlations
        # This mathematically complex, but we can solve with very coarse model
        # Using structure if h is piecewise constant
        # Redefine h values to work better with CVXPY
        h_init = h.copy()
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Call the maximizer directly on the original problem but solve with LP-warm start
        def compute_c5(h_vec):
            if len(h_vec) != n_points:
                h_vec = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_vec)), h_vec)
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(corr) * dx
            
        # Simple local search with successive refinement
        h_current = h_init.copy()
        best_c5 = compute_c5(h_current)
        
        # Gradient descent refinement
        def gradient_step(h_current, lr=0.005, max_iter=200):
            h_refined = h_current.copy()
            old_c5 = compute_c5(h_refined)
            
            for i in range(max_iter):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Finite difference gradient
                grad = np.zeros(n_points)
                eps = 1e-5
                
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    c5_plus = compute_c5(h_plus)
                    grad[j] = (c5_plus - old_c5) / eps
                
                # Update step
                h_refined = h_refined - lr * grad
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                
                new_c5 = compute_c5(h_refined)
                if new_c5 < old_c5:
                    old_c5 = new_c5
                else:
                    # Reduce learning rate
                    lr = max(1e-8, lr * 0.9)
                
            return h_refined, old_c5
            
        h_opt, c5_opt = gradient_step(h_current, lr=0.01)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            
        # Additional tuning with other heuristics
        if time.time() - start_time < budget_s - 1:
            # Heuristic LP instantiation
            try:
                h_vars = cp.Variable(n_points)
                C5 = cp.Variable()
                
                # Simplified LP formulation
                constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
                
                # More direct strategy: L-BFGS on C5
                def fun(h_vec):
                    correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
                    return np.max(correlation) * dx
                
                def jac(h_vec):
                    # Approximate gradient via finite differences (since no analytical form)
                    eps = 1e-6
                    grad = np.zeros(n_points)
                    base_val = fun(h_vec)
                    for i in range(n_points):
                        h_plus = h_vec.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        grad[i] = (fun(h_plus) - base_val) / eps
                    return grad
                
                # L-BFGS-B
                res = minimize(fun, h_current, method='L-BFGS-B', jac=jac,
                             bounds=[(0, 1)] * n_points, options={'maxiter': 100})
                if not res.failed and fun(res.x) < best_c5:
                    best_h = res.x
                    best_c5 = fun(res.x)
                    
            except:
                pass  # Fall back to existing method if LP fails
                
    except Exception as e:
        pass  # Proceed with basic methods if something fails
        
    return (best_h.tolist(), best_c5, n_points)
