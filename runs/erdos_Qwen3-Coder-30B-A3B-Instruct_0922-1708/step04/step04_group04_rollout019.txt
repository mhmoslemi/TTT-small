import numpy as np
from scipy.stats import qmc
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Generate Latin Hypercube Samples
    sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs_samples = sampler.random(n=50)
    
    # Convert LHS to valid h vectors
    h_candidates = []
    for sample in lhs_samples:
        # Transform to [0,1]^n_points
        h_sample = sample
        # Rescale to satisfy sum constraint
        h_sample = h_sample * (n_points / 2) / np.sum(h_sample)
        # Clip to [0,1]
        h_sample = np.clip(h_sample, 0, 1)
        h_candidates.append(h_sample)
    
    # Add the initial solution if available
    if 'initial_h_values' in globals():
        h_candidates.append(h.copy())
    
    # Evaluate and refine each candidate
    for candidate in h_candidates:
        temp_h = candidate.copy()
        
        # Gradient descent refinement
        for iter_count in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute the overlap integral
            correlation = np.correlate(temp_h, 1 - temp_h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Compute gradient via finite differences
            grad = np.zeros_like(temp_h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = temp_h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = temp_h.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with adaptive learning rate
            lr = 0.01 / (1 + iter_count * 1e-4)
            temp_h_new = temp_h - lr * grad
            
            # Clip and rescale
            temp_h_new = np.clip(temp_h_new, 0, 1)
            temp_h_new = temp_h_new * (n_points / 2) / np.sum(temp_h_new)
            
            temp_h = temp_h_new
            
            # Update best solution if improved
            correlation = np.correlate(temp_h, 1 - temp_h, mode='full')
            c5_new = np.max(correlation) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = temp_h.copy()
                
            # Convergence check - stop early if no improvement
            if iter_count > 10 and abs(c5 - c5_new) < 1e-8:
                break
    
    # Final check with evaluator
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
