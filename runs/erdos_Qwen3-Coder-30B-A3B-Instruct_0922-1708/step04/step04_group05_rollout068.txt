import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    lr = 0.001
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Project to [0,1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute fft for efficiency
    fft_size = 2 * n_points - 1
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute overlap integral
        h_conv = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(h_conv) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient computation using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode='full')) * dx
            grad[j] = (c5_plus - c5) / eps
        
        # Update h
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Adaptive learning rate
        lr = max(1e-6, lr * 0.9999)
        
        h = h_new
    
    # Final check
    final_conv = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_conv) * dx
    
    return (best_h.tolist(), final_c5, n_points)
