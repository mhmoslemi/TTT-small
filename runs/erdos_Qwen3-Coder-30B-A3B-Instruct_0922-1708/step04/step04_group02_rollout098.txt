import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure feasibility
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    temperature = 1.0
    cooling_factor = 0.99
    temp_steps = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Generate candidate
        h_new = h.copy()
        
        # Select random points to modify
        n_modify = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_modify, replace=False)
        
        for i in indices:
            # Add noise within [0,1]
            h_new[i] += np.random.normal(0, 0.1)
        
        # Ensure feasible
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / temperature):
            h = h_new
            
        # Cool temperature
        if iteration % temp_steps == 0:
            temperature *= cooling_factor
            
        # Occasionally try more aggressive updates
        if iteration % 500 == 0:
            h_new = h.copy()
            # Try larger perturbations
            n_modify = max(1, n_points // 20)
            indices = np.random.choice(n_points, n_modify, replace=False)
            for i in indices:
                # More aggressive perturbation
                h_new[i] += np.random.normal(0, 0.2)
                
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            new_c5 = np.max(corr_new) * dx
            
            if new_c5 < current_c5 or np.random.random() < np.exp(-new_c5 / temperature):
                h = h_new
                
    # Final refinement
    h_refined = best_h.copy()
    prev_c5 = best_c5
    
    # Local search
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_refined.copy()
        
        # Small random modification
        idx = np.random.randint(0, n_points)
        h_new[idx] += np.random.normal(0, 0.01)
        
        # Clamp
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < prev_c5:
            h_refined = h_new
            prev_c5 = new_c5
    
    # Double-check final C5
    corr_final = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
