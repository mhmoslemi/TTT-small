import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Phase 1: Convex relaxation using CVXPY
    # Minimize sum of all correlations (relaxed version)
    h_var = cp.Variable(n_points)
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # Construct the relaxation objective (approximate)
    # We'll minimize the sum of all correlation terms for a rough starting point
    relaxation_objective = 0
    for k in range(n_points):
        # Approximate the correlation sum with a few terms
        for i in range(n_points):
            j = (i + k) % n_points
            relaxation_objective += h_var[i] * (1 - h_var[j])
    
    prob = cp.Problem(cp.Minimize(relaxation_objective), constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = np.array(h_var.value).reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass  # Fallback to previous initialization if convex optimization fails
    
    # Phase 2: Projected gradient descent refinement
    max_no_improvement = 50
    no_improvement_count = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        # Compute current correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            improved = True
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update step with adaptive learning rate
        h_new = h - lr * grad
        
        # Apply constraints: clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Check for improvement
        corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            improved = True
            
        # Update learning rate
        if iteration % 100 == 0:
            lr *= 0.9
            
        # Early stopping
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
    return (best_h.tolist(), best_c5, n_points)
