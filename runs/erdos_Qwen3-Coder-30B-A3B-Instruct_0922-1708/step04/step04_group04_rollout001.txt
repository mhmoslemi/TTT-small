import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Use CVXPY to solve our optimization problem
    # Variables
    h_var = cp.Variable(n_points)
    C_var = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1]
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # For each shift k, form the overlap expression and constraint
    # We'll compute this manually for now - the constraints are: 
    # sum h[i] * (1 - h[i+k]) for valid i 
    # This is a non-convex constraint but CVXPY will handle it when we use a solver
    
    # Semidefinite relaxation approach - not straightforward for this
    # Instead, we'll simply iterate with a convex approximation or use numerical approach
    
    # Idea: Use CVXPY with appropriate solver settings on a surrogate problem
    
    # Since the actual formulation is non-convex, we'll iterate with a greedy approach
    # First, define the usual overlap computation function
    def compute_overlap(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Our approach: update h directly using gradient-based method with constraints
    # This mimics structured optimization
    
    # Let's do one optimization approach: L-BFGS style optimizers, but constrained
    from scipy.optimize import minimize
    
    def loss_func(h_vec):
        # Scale back to have sum(h) = n_points/2
        h_scale = h_vec * (n_points/2) / np.sum(h_vec)
        return compute_overlap(h_scale)
    
    def constraint_func(h_vec):
        # Return the constraint violation
        h_scale = h_vec * (n_points/2) / np.sum(h_vec)
        return np.sum(h_scale) - n_points/2
    
    # Setup the optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    max_iter = 500
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        try:
            # Using SLSQP for the optimization
            res = minimize(
                loss_func,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            if res.success:
                h = res.x
            else:
                # fallback to gradient descent with projected updates
                # Compute approximate gradient via finite differences
                eps = 1e-5
                grad = np.zeros(n_points)
                base_loss = loss_func(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    loss_plus = loss_func(h_plus)
                    grad[i] = (loss_plus - base_loss) / eps
                
                # Gradient step
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h - lr * grad
                
                # Project and re-normalize
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                h = h_new

            # Update best if better
            c5_val = compute_overlap(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
                
        except Exception as e:
            if time.time() - start_time > budget_s - 1:
                break
            # If something goes wrong, simplest fallback
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5_val = compute_overlap(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()

    # Final computation for best solution
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
