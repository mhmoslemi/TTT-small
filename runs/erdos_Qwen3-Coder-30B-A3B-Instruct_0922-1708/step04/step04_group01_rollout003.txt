import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Warm up: try some simple optimizations first
    for _ in range(50):
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try a smart perturbation based on gradient
        grad = np.zeros_like(h)
        eps = 1e-4
        
        # Compute approximate gradient using finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update h using gradient descent
        lr = 0.01
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new

    # PSO-based approach for global search
    num_particles = 30
    max_iter = 1000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles
    particles = []
    pbests = []
    pbest_values = []
    
    for _ in range(num_particles):
        # Start from the best solution found so far
        h_init = best_h + np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        pbests.append(h_init.copy())
        corr = np.correlate(h_init, 1 - h_init, mode='full')
        pbest_values.append(np.max(corr) * dx)
    
    gbest_value = min(pbest_values)
    gbest_index = np.argmin(pbest_values)
    gbest = pbests[gbest_index].copy()
    
    # Initialize velocities
    velocities = [np.random.normal(0, 0.1, n_points) for _ in range(num_particles)]
    
    iter_count = 0
    no_improvement_count = 0
    patience = 100
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Update particles
        for i in range(num_particles):
            h = particles[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbests[i] - h) + 
                            c2 * r2 * (gbest - h))
            
            # Update position
            h_new = h + velocities[i]
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Check if better than personal best
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            new_value = np.max(corr) * dx
            
            if new_value < pbest_values[i]:
                pbest_values[i] = new_value
                pbests[i] = h_new.copy()
                
                if new_value < gbest_value:
                    gbest_value = new_value
                    gbest = h_new.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
                
            particles[i] = h_new
        
        # Early termination
        if no_improvement_count > patience:
            break

    # CVX refinement of best solution
    try:
        # Use the best particle from PSO as warm start
        h_cvx = gbest.copy()
        
        # Define the optimization variables with proper constraints
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Objective: minimize the maximum overlap
        # Build the correlation matrix efficiently
        # For simplicity, we manually optimize the constraint
        # Since CPX has issues with max of convolution, we introduce slack variables
        
        # Use a projection-based refinement approach
        # Let's try a more direct quadratic formulation
    
        # Refine using optimization
        h_current = h_cvx * (n_points / 2) / np.sum(h_cvx)
        h_current = np.clip(h_current, 0, 1)
        
        # Run additional local optimization with scipy
        def objective(h_vec):
            h_binary = np.clip(h_vec, 0, 1)
            h_normalized = h_binary * (n_points / 2) / np.sum(h_binary)
            correlation = np.correlate(h_normalized, 1 - h_normalized, mode='full')
            return -np.max(correlation) * dx  # Negative for maximization
            
        # However, Since this is non-convex, use scipy's L-BFGS-B
        result = minimize(objective, h_current, method='L-BFGS-B', bounds=[(0,1)]*n_points)
        
        if result.success or result.fun < best_c5:
            h_final = np.clip(result.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            
            correlation = np.correlate(h_final, 1 - h_final, mode='full')
            c5_final = np.max(correlation) * dx
            
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception as e:
        pass

    # Final validation and return
    if best_c5 == float('inf'):
        best_h = np.zeros(n_points)
        best_h[:n_points//2] = 2.0
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
