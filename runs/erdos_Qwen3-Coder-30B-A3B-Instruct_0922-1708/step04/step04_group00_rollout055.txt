import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h with the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy the constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective_and_gradient(h):
        # Normalize h to satisfy constraint
        h_normalized = h * (n_points / 2) / np.sum(h)
        h_normalized = np.clip(h_normalized, 0, 1)
        
        # Compute correlation
        correlation = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_plus = np.clip(h_plus, 0, 1)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        return c5, grad
    
    # Optimization settings
    max_iter = 500
    initial_tr_radius = 0.1
    lr = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Re-normalize h to satisfy constraint
        h_norm = h * (n_points / 2) / np.sum(h)
        h_norm = np.clip(h_norm, 0, 1)
        
        # Evaluate current solution
        c5, grad = objective_and_gradient(h_norm)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_norm.copy()
            
        # Update step using gradient descent
        h_new = h_norm - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        lr = max(1e-6, lr * 0.999)
        
        # Early stopping based on improvement rate
        if iteration > 10 and np.abs(c5 - best_c5) < 1e-8:
            break
    
    # Final validation
    h_final = best_h * (n_points / 2) / np.sum(best_h)
    h_final = np.clip(h_final, 0, 1)
    
    correlation = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (h_final.tolist(), final_c5, n_points)
