import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation: minimize sum of overlaps
    # This provides a good starting point
    def solve_convex_relaxation(h_init):
        # Variables
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize sum of all overlap terms
        # Note: This is a relaxation; we minimize sum rather than max
        obj = 0
        for k in range(n_points):
            for i in range(n_points):
                obj += h_vars[i] * (1 - h_vars[(i+k) % n_points])
        
        obj = cp.Minimize(obj)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Solve
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return np.array(h_vars.value)
        except:
            pass
        return h_init.copy()
    
    # First phase: solve convex relaxation
    h = solve_convex_relaxation(h)
    
    # Second phase: projected gradient descent
    max_iter = 1000
    lr = 0.01
    decay_rate = 0.9
    patience = 50
    no_improvement = 0
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integral and gradient
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Update with gradient descent
        h_new = h - lr * grad
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update h
        h = h_new
        
        # Adaptive learning rate
        if iter_num % 100 == 0 and iter_num > 0:
            lr *= decay_rate
            
        # Check for convergence
        if c5 < best_c5:
            no_improvement = 0
        else:
            no_improvement += 1
        if no_improvement >= patience:
            break
            
    # Final check for best value
    if np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx < best_c5:
        corr_final = np.correlate(best_h, 1 - best_h, mode='full') * dx
        best_c5 = np.max(corr_final)
        
    return (best_h.tolist(), best_c5, n_points)
