import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_segments = 8
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize segments
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Convert to sparse representation by finding peaks
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        # Create piecewise segments from initial values
        segments = []
        # Simple approach: find roughly n_segments peaks
        peak_positions = np.argsort(h_init)[-n_segments:]
        peak_positions.sort()
        heights = h_init[peak_positions]
        # Normalize heights so integral = 1
        seg_starts = np.linspace(0, 2, n_segments+1)
        seg_heights = np.zeros(n_segments)
        # Assign values to segments
        for i in range(n_segments):
            seg_heights[i] = heights[i] if i < len(heights) else 1.0
        seg_heights = seg_heights * (n_points/2) / np.sum(seg_heights * np.diff(seg_starts))
    else:
        seg_starts = np.linspace(0, 2, n_segments+1)
        seg_heights = np.ones(n_segments) * (1.0 / n_segments)
    
    # Optimization variables: segment start positions (except 0 and 2) and heights
    initial_vars = np.concatenate([seg_starts[1:-1], seg_heights])
    
    def segments_from_vars(vars):
        starts = np.concatenate([[0], vars[:n_segments-1], [2]])
        heights = vars[n_segments-1:]
        return starts, heights
    
    def compute_overlap_integral(segments):
        starts, heights = segments
        # Normalize heights to satisfy integral constraint ∫h(x)dx = 1
        integral = np.sum(heights * np.diff(starts))
        heights = heights * (1.0 / integral)
        
        # For each lag k, compute overlap
        max_overlap = 0
        
        # Sample lags for more efficiency
        lags = np.arange(1, n_points)  # Use fewer lags since calculation is expensive
        lags = lags[::max(1, len(lags)//20)]  # Reduce for performance
        
        for k_index in lags:
            # Convert k_index to actual shift k
            k = k_index * dx
            overlap = 0.0
            
            # Compute overlap integral ∫ h(x) * (1 - h(x+k)) dx
            for i in range(len(starts)-1):
                a, b = starts[i], starts[i+1]
                h = heights[i]
                # Check how much of this segment overlaps with (x+k) in [a,b]
                # So we check segments [a,b] vs [a+k, b+k]
                for j in range(len(starts)-1):
                    a2, b2 = starts[j], starts[j+1]
                    h2 = heights[j]
                    # Overlap of intervals [a,b] and [a+k, b+k]
                    overlap_start = max(a, a+k)
                    overlap_end = min(b, b+k)
                    if overlap_start < overlap_end:
                        length = overlap_end - overlap_start
                        overlap += h * (1 - h2) * length
                
            overlap *= dx
            max_overlap = max(max_overlap, overlap)
            
        return max_overlap
    
    def objective(vars):
        segments = segments_from_vars(vars)
        return compute_overlap_integral(segments)
    
    def constraint(vars):
        starts, heights = segments_from_vars(vars)
        integral = np.sum(heights * np.diff(starts))
        return integral - 1.0
    
    # Set up constraints
    constraints = {'type': 'eq', 'fun': constraint}
    
    bounds = []
    # Bounds for starts (except first and last)
    for i in range(n_segments-1):
        bounds.append((0.001, 1.999))
    # Bounds for heights (0 to 1)
    for i in range(n_segments):
        bounds.append((1e-6, 1.0))
    
    # Optimization loop
    best_vars = initial_vars.copy()
    best_c5 = float('inf')
    
    # Initialize with standard greedy approach if needed
    vars = initial_vars.copy()
    try:
        # Apply bounds
        for i, (lb, ub) in enumerate(bounds):
            vars[i] = np.clip(vars[i], lb, ub)
        
        for i in range(100):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Solve optimization step
            res = minimize(objective, vars, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 10})
            
            if res.success:
                vars = res.x
                # Check constraints
                starts, heights = segments_from_vars(vars)
                integral_violation = abs(np.sum(heights * np.diff(starts)) - 1.0)
                if integral_violation < 1e-3:
                    obj_val = objective(vars)
                    if obj_val < best_c5:
                        best_c5 = obj_val
                        best_vars = vars.copy()
            else:
                break
                
            # Perturb slightly
            for i in range(len(vars)):
                if i < n_segments - 1:  # Position variables
                    vars[i] += np.random.normal(0, 0.01)
                    vars[i] = np.clip(vars[i], 0.001, 1.999)
                else:  # Height variables
                    vars[i] += np.random.normal(0, 0.01)
                    vars[i] = np.clip(vars[i], 1e-6, 1.0)
                    
    except Exception:
        pass
    
    # Final conversion to step function
    starts, heights = segments_from_vars(best_vars)
    h_final = np.zeros(n_points)
    
    # Interpolate to actual points
    x_points = np.linspace(0, 2, n_points)
    for i in range(len(starts)-1):
        mask = (x_points >= starts[i]) & (x_points < starts[i+1])
        h_final[mask] = heights[i]
    
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Recompute the final C5 value for accuracy
    corr = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_final.tolist(), final_c5, n_points)
