import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Solve using convex optimization approach
    # We'll use CVXPY for convex relaxation
    iter_count = 0
    max_iter = 50
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Define CVXPY variables
        h_vars = cp.Variable(n_points, name="h")
        C = cp.Variable(name="C")
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,  # sum(h) = 1
            h_vars >= 0,                     # h >= 0
            h_vars <= 1                      # h <= 1
        ]
        
        # Add constraints for all possible overlaps
        # This includes the convolution constraint C >= \max_k \int h(x)(1-h(x+k)) dx
        # We compute this approximately by checking maximum correlation
        for shift in range(1, n_points):
            # Create the shifted correlation manually
            # Only compute correlation when there's overlap
            if shift < n_points:
                correlation_sum = 0
                for i in range(n_points - shift):
                    correlation_sum += h_vars[i] * (1 - h_vars[i + shift])
                constraints.append(correlation_sum <= C)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.ECOS, max_iter=100, feastol=1e-5)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                # Rescale to meet sum constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_new = np.clip(h_new, 0, 1)
                
                # Check actual C5
                correlation = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(correlation) * dx
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    
                # Update for next iteration
                h = h_new.copy()
            else:
                # Fall back to gradient descent if optimization fails
                break
        except Exception as e:
            # Fall back to simple gradient descent if optimization fails
            break
            
        iter_count += 1
        
    # If CVXPY failed, do gradient descent approach to refine
    if best_c5 == float('inf'):
        h = h.copy()
        for i in range(50):
            if time.time() - start_time > budget_s - 1:
                break
                
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Estimate gradient via finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - c5) / eps
                
            # Update with adaptive learning rate
            lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-4)
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Check improvement
            correlation = np.correlate(h, 1 - h, mode='full')
            c5_new = np.max(correlation) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
    return (best_h.tolist(), best_c5, n_points)
