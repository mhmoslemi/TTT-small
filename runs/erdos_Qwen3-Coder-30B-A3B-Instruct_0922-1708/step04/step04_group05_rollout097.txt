import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Generate diverse initial candidates
    candidates = []
    candidates.append(h_init.copy())
    for i in range(19):
        h_pert = h_init + np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        candidates.append(h_pert)
    
    # Use the candidates to guide the optimization
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Gradient-based refinement for a randomly selected candidate
        idx = np.random.randint(0, len(candidates))
        h = candidates[idx].copy()
        
        # Local optimization using gradient descent
        for _ in range(50):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            max_corr = corr[max_idx]
            
            # Compute gradient
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                grad[i] = (corr_plus[max_idx] - max_corr) / eps
            
            # Update step
            lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Early termination if change is small
            if np.max(np.abs(h - h_new)) < 1e-6:
                break
        
        # Evaluate this optimized candidate
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Add new candidates periodically
        if iter_count % 10 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_new = h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            candidates.append(h_new)
            
            # Remove oldest candidate to keep size reasonable
            if len(candidates) > 20:
                candidates.pop(0)
    
    # Final check
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
