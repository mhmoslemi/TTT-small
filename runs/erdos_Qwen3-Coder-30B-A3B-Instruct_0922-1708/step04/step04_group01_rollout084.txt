import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Uniform distribution
        h = np.ones(n_points) * 0.5
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Phase 1: Grid Search on a coarse resolution
    if time.time() - start_time < budget_s * 0.2:
        coarse_points = 20
        coarse_dx = 2.0 / coarse_points
        best_coarse_c5 = float('inf')
        best_coarse_h = None
        
        # Generate a few random candidates with different structure
        for _ in range(50):
            if time.time() - start_time > budget_s * 0.2:
                break
            h_coarse = np.random.random(coarse_points)
            h_coarse = h_coarse / np.sum(h_coarse) * (coarse_points / 2)
            
            # Try some block structures
            if np.random.random() > 0.5:
                # Simple block function
                block = np.random.randint(1, coarse_points - 1)
                h_coarse[:block] = 1.0
                h_coarse[block:] = 0.0
                h_coarse = h_coarse / np.sum(h_coarse) * (coarse_points / 2)
            
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, coarse_points), h_coarse)
            c5 = compute_c5(h_interp)
            if c5 < best_coarse_c5:
                best_coarse_c5 = c5
                best_coarse_h = h_interp.copy()
        
        if best_coarse_h is not None:
            h = best_coarse_h

    # Phase 2: Refinement using gradient-based methods
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    if time.time() - start_time < budget_s * 0.8:
        max_iter = 500
        lr = 0.01
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute current loss and its subgradient
            c5 = compute_c5(h)
            
            # Estimate gradient via finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = project_h(h_plus)
                c5_plus = compute_c5(h_plus)
                grad[j] = (c5_plus - c5) / eps
            
            # Update with gradient descent
            h_new = h - lr * grad
            h = project_h(h_new)

            # Check if we're improving
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

    # Phase 3: Convex optimization refinement
    try:
        # Use simpler convex relaxation for better stability
        if time.time() - start_time < budget_s - 2:
            # Define variables
            h_vars = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Objective: approximate quadratic form that encourages smaller max overlap
            # Since we can't directly minimize the max, we try using an approximate formulation
            # But also since it's a fixed value we just optimize over a good local region
            
            # Use CVX to solve as a QP - approximate the maximum by using a proxy
            # We penalize large values of the correlation at all bins
            correlation_proxy = 0
            for k in range(1, n_points):
                corr_kernel = np.concatenate([np.zeros(k), np.ones(1), np.zeros(n_points - k - 1)])
                corr_max = cp.sum(cp.multiply(h_vars, cp.conv(h_vars, cj - h_vars)))
                correlation_proxy += corr_max
            
            prob = cp.Problem(cp.Minimize(cp.max(cp.conv(h_vars, 1 - h_vars))), constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL and prob.value < best_c5:
                h_cvx = h_vars.value
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                c5_cvx = compute_c5(h_cvx)
                if c5_cvx < best_c5:
                    best_c5 = c5_cvx
                    best_h = h_cvx
    except:
        pass

    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
