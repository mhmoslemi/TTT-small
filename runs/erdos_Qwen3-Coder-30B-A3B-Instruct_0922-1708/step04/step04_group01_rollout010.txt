import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize using parent or create block function
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init / np.sum(h_init) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Mutate a binary version while maintaining sum
    def binary_mutation(h_bin, rate=0.05):
        h_new = h_bin.copy()
        for i in range(len(h_new)):
            if np.random.random() < rate:
                h_new[i] = 1 - h_new[i]
        # Ensure sum stays constant
        ones_count = np.sum(h_new)
        if ones_count < 50:
            # Add more ones
            indices = np.where(h_new == 0)[0]
            np.random.shuffle(indices)
            add_count = int(50 - ones_count)
            for i in range(add_count):
                h_new[indices[i]] = 1
        elif ones_count > 50:
            # Remove some ones
            indices = np.where(h_new == 1)[0]
            np.random.shuffle(indices)
            remove_count = int(ones_count - 50)
            for i in range(remove_count):
                h_new[indices[i]] = 0
        return h_new
    
    # Create a population of binary individuals
    def create_population(size, n_points):
        population = []
        for i in range(size):
            h_candidate = np.zeros(n_points)
            ones_indices = np.random.choice(n_points, size=50, replace=False)
            h_candidate[ones_indices] = 1
            population.append(h_candidate)
        return population
    
    # Evaluate a population and return best
    def evaluate_population(population):
        results = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            results.append((c5, h))
        return sorted(results, key=lambda x: x[0])
    
    # Genetic algorithm part
    pop_size = 100
    generations = 100
    mutation_rate = 0.05
    crossover_rate = 0.5
    
    population = create_population(pop_size, n_points)
    
    # Evaluate initial population
    evaluated = evaluate_population(population)
    best_in_gen = evaluated[0][0]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection (top 20%)
        selected = evaluated[:20]
        
        # Create new population
        new_pop = [h for _, h in selected]  # Keep best
        
        # Crossover
        while len(new_pop) < pop_size:
            if np.random.random() < crossover_rate:
                parent1, parent2 = selected[np.random.randint(0, 20)], selected[np.random.randint(0, 20)]
                # Single point crossover
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                # Repair to maintain sum
                ones_count = np.sum(child)
                if ones_count < 50:
                    indices = np.where(child == 0)[0]
                    np.random.shuffle(indices)
                    add_count = int(50 - ones_count)
                    for i in range(add_count):
                        child[indices[i]] = 1
                elif ones_count > 50:
                    indices = np.where(child == 1)[0]
                    np.random.shuffle(indices)
                    remove_count = int(ones_count - 50)
                    for i in range(remove_count):
                        child[indices[i]] = 0
                new_pop.append(child)
            else:
                new_pop.append(selected[np.random.randint(0, 20)][1])  # Copy random parent
        
        # Mutation
        for i in range(len(new_pop)):
            if np.random.random() < mutation_rate:
                new_pop[i] = binary_mutation(new_pop[i], mutation_rate)
        
        # Evaluate new population
        evaluated = evaluate_population(new_pop)
        
        # Track global best
        if evaluated[0][0] < best_c5:
            best_c5 = evaluated[0][0]
            best_h = evaluated[0][1].copy()
    
    # Post-process with CVX optimization for final refinement
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL and h_vars.value is not None:
            h_cvx = np.clip(h_vars.value, 0, 1)
            h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
            corr = np.correlate(h_cvx, 1 - h_cvx, mode='full')
            c5_cvx = np.max(corr) * dx
            if c5_cvx < best_c5:
                best_c5 = c5_cvx
                best_h = h_cvx
    except:
        pass

    # Final validation and computation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h.tolist(), final_c5, n_points)
