import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use CVXPY to solve a relaxed version of the problem
    # We'll iterate over several starting points to improve chances of finding global minimum
    num_starts = 5  # Number of random starts
    
    for start_idx in range(num_starts):
        if time.time() - start_time > budget_s - 2:
            break
            
        if start_idx == 0:  # Use initial guess
            h_start = h.copy()
        else:  # Random initialization with same constraints
            h_start = np.random.uniform(0, 1, n_points)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
            
        # Solve with CVXPY
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            C >= 0
        ]
        
        # Compute overlaps for all shifts
        for k in range(1, n_points):
            if k <= n_points // 2:
                # Positive shift
                overlap = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
                constraints.append(overlap <= C * dx)
                # Negative shift
                overlap = cp.sum(cp.multiply(h_var[k:], 1 - h_var[:n_points-k]))
                constraints.append(overlap <= C * dx)
        
        # Objective: minimize C
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solved = h_var.value
                # Compute actual C5 value
                corr = np.correlate(h_solved, 1 - h_solved, mode='full')
                c5_value = np.max(corr) * dx
                
                if c5_value < best_c5:
                    best_c5 = c5_value
                    best_h = h_solved.copy()
        except:
            pass
        
        # Local refinement using gradient-based optimization
        h_refined = best_h.copy()
        lr = 0.01
        for _ in range(100):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Compute gradient
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            max_idx = np.argmax(corr)
            max_corr = corr[max_idx]
            c5_refined = max_corr * dx
            
            # Simple gradient descent update
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h_refined.copy()
                eps = 1e-5
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                max_corr_plus = np.max(corr_plus)
                grad[i] = (max_corr_plus - max_corr) / eps
                
            # Update with learning rate
            h_new = h_refined - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_refined = h_new
            
            # Check for improvement
            corr_new = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5_new = np.max(corr_new) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_refined.copy()
                
    return (best_h.tolist(), best_c5, n_points)
