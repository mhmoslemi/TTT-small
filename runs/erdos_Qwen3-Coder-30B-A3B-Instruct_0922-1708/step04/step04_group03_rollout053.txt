import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation using CVXPY
    # Create variables
    h_var = cp.Variable(n_points)
    
    # Objective: minimize sum of all overlaps (relaxed)
    # We don't directly minimize max overlap, but we can do a smoother approximation
    # For now, we'll use a two-phase approach:
    # 1. Convex relaxation to get reasonable initial solution
    # 2. Direct optimization of max overlap
    
    # Use a simple heuristic for convex relaxation: minimize sum of all overlaps
    # Total overlap = sum_{i,k} h[i]*(1-h[i+k]) * dx
    # = sum h[i] * dx - sum h[i]*h[i+k] * dx
    
    # Let's create a simple heuristic solution by trying to pick a solution that 
    # locally minimizes the max overlap
    
    # Phase 2: Optimization with projected gradient descent
    # Initialize with some precomputed good values from previous good solution
    h = h.copy()
    
    def compute_c5(h_vec):
        # Compute correlation manually
        h_vec = np.array(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_val = np.max(corr)
        return max_val * dx
    
    def project_and_normalize(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Apply projected gradient descent
    max_iter = 1000
    lr = 0.01
    patience = 50
    no_improvement = 0
    
    c5_old = compute_c5(h)
    if c5_old < best_c5:
        best_c5 = c5_old
        best_h = h.copy()
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_and_normalize(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_old) / eps
        
        # Update step
        h_new = h - lr * grad
        h_new = project_and_normalize(h_new)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_old:
            h = h_new
            c5_old = c5_new
            if c5_old < best_c5:
                best_c5 = c5_old
                best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            lr *= 0.995  # Adaptive learning rate
            
        if no_improvement >= patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
