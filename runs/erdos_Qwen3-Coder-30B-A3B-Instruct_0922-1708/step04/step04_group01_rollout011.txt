import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()

    # Speed-up: reduce number of iterations in the main loop
    max_iter = 5000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try a smarter approach: Simulated Annealing with adaptive cooling
        # Compute gradient with finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update h using gradient descent with adaptive learning rate
        lr = 0.005 * np.exp(-iter_count / 1000) # Cooling schedule
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept new solution with probability based on Metropolis criterion
        delta_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx - c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / (0.01 * (1 + iter_count/1000))):
            h = h_new
        
        iter_count += 1

    # Try convex optimization to refine
    try:
        h_vars = cp.Variable(n_points)

        # This is a more difficult convex problem to solve directly so we use a relaxation technique
        
        # Re-define objective using approximate relaxation
        # Solving a better quadratic programming formulation
        # This should be done with proper covariance constraints
        
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')

                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:

    # Final verification step using another method
    # Try a Newton-type method with a small number of iterations to further refine
    h_refined = best_h.copy()

        c5 = np.max(correlation) * dx
        
        # Compute approximate gradient
        grad = np.zeros_like(h_refined)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h_refined.copy()

            h_plus = np.clip(h_plus, 0, 1)

            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx

        # Update
        lr = 0.01 * np.exp(-i / 50)
        h_refined = h_refined - lr * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)

    # Final check
    final_correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
