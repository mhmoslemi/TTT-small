        correlation = np.correlate(h, 1 - h, mode='full')

            alpha = np.random.random()
            h_new = alpha * h1 + (1 - alpha) * h2

            # Make a more targeted formulation
            # Since we're solving this in a relaxed setting, the best we can do is use
            # evolutionary candidates as starting points
