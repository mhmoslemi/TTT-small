import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with parent construction or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Grid search with a smaller resolution to quickly find promising areas
    coarse_points = 20
    dx_coarse = 2.0 / coarse_points
    feasible_points = []
    
    # Generate grid points in low resolution space
    grid_values = np.linspace(0, 1, 5)  # Just for initial exploration
    
    # Try various step functions from grid
    for i in range(4):
        h_test = np.zeros(coarse_points)
        h_test[:coarse_points//2] = grid_values[i]
        h_test = np.tile(h_test, n_points // coarse_points)
        h_test = h_test[:n_points]
        h_test = h_test / np.sum(h_test) * (n_points / 2)
        feasible_points.append(h_test)
    
    # Add some random samples for diversity
    for _ in range(10):
        h_test = np.random.rand(n_points)
        h_test = h_test / np.sum(h_test) * (n_points / 2)
        feasible_points.append(h_test)
    
    # Evaluate coarse points and select best
    candidates = []
    for h_test in feasible_points:
        # Ensure constraints
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test / np.sum(h_test) * (n_points / 2)
        
        correlation = np.correlate(h_test, 1 - h_test, mode='full')
        c5 = np.max(correlation) * dx
        candidates.append((h_test.copy(), c5))
    
    # Sort by C5 and take top 5
    candidates.sort(key=lambda x: x[1])
    top_candidates = candidates[:5]
    
    # Refine top candidates using a more robust approach
    max_refinement_steps = 500
    for h_start, _ in top_candidates:
        h_refined = h_start.copy()
        best_candidate = h_refined.copy()
        best_c5_candidate = float('inf')
        
        for step in range(max_refinement_steps):
            if time.time() - start_time > budget_s - 1:
                break
            # Compute gradient approximation using finite differences
            eps = 1e-4
            grad = np.zeros_like(h_refined)
            c5_current = float('inf')
            
            # Compute current value
            correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5_current = np.max(correlation) * dx
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5_current) / eps
                
            # Gradient descent step
            lr = 0.01
            h_new = h_refined - lr * grad
            
            # Project back onto feasible set and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            if np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx < best_c5_candidate:
                best_c5_candidate = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
                best_candidate = h_new.copy()
                
            h_refined = h_new
            
        if best_c5_candidate < best_c5:
            best_c5 = best_c5_candidate
            best_h = best_candidate.copy()

    # Final check with cvxpy for potentially better solution
    try:
        # Use CVXPY with constrained optimization
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points/2]
        
        # Define objective using max constraint
        # Since direct max cannot be used in CVX, approximate with a smoothing
        # Let's collect all the overlaps and try a quadratic approach
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1-h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                if c5_cvx < best_c5:
                    best_c5 = c5_cvx
                    best_h = h_cvx.copy()
    except Exception as e:
        pass

    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
