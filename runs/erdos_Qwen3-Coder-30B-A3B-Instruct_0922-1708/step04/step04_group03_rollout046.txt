import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iterations = 0
    no_improvement = 0
    max_no_improvement = 100
    lr = 0.1
    
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        iterations += 1
        
        # Compute overlaps
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Compute gradient via finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            # Perturb up
            h_up = h.copy()
            h_up[i] += eps
            h_up = np.clip(h_up, 0, 1)
            h_up = h_up * (n_points / 2) / np.sum(h_up)
            
            corr_up = np.correlate(h_up, 1 - h_up, mode='full')
            c5_up = np.max(corr_up) * dx
            
            # Perturb down
            h_down = h.copy()
            h_down[i] -= eps
            h_down = np.clip(h_down, 0, 1)
            h_down = h_down * (n_points / 2) / np.sum(h_down)
            
            corr_down = np.correlate(h_down, 1 - h_down, mode='full')
            c5_down = np.max(corr_down) * dx
            
            # Estimate gradient
            grad[i] = (c5_up - c5_down) / (2 * eps)
        
        # Update h
        h = h - lr * grad
        
        # Enforce constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Decay learning rate
        if iterations % 50 == 0:
            lr *= 0.99
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
