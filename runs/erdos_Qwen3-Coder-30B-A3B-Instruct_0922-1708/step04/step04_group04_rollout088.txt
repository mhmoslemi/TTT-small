import numpy as np
import time
from heapq import heappush, heappop
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_overlap(h):
        # Compute all overlaps efficiently using convolution
        h_pad = np.pad(h, (0, n_points-1), mode='constant')
        h_rev = h[::-1]
        corr = signal.convolve(h_pad, 1 - h_rev, mode='full')
        corr = corr[n_points-1:]
        return np.max(corr) * dx
    
    def overlap_for_shift(h, k):
        # Compute overlap for specific shift k
        if k > 0:
            # h[i] * (1 - h[i+k])
            overlap = np.sum(h[:-k] * (1 - h[k:])) * dx
        elif k < 0:
            # h[i] * (1 - h[i+k]) where i+k >= 0
            overlap = np.sum(h[-k:] * (1 - h[:-k])) * dx
        else:
            overlap = 0
        return overlap
    
    def get_worst_shifts(h, num=20):
        # Get the most problematic shifts
        shifts = []
        for k in range(-n_points+1, n_points):
            if k == 0:
                continue
            overlap = overlap_for_shift(h, k)
            shifts.append((-overlap, k))  # negative for max-heap behavior
        
        # Get top problematic shifts
        shifts.sort()
        return [shift[1] for shift in shifts[:num]]
    
    def adjust_h(h, k, w=5):
        h_new = h.copy()
        # Adjust values around the problematic shift k
        gain = 0.02
        if k > 0:
            # Apply local adjustments based on overlap analysis
            for i in range(max(0, k-w), min(len(h), len(h)-k+w)):
                if h_new[i] > 0 and h_new[i+k] < 1:
                    # Reduce h[i] to decrease overlap
                    h_new[i] = max(0, h_new[i] - gain)
                    h_new[i+k] = min(1, h_new[i+k] + gain)
        elif k < 0:
            for i in range(max(0, -k-w), min(len(h), len(h)+k+w)):
                if h_new[i] > 0 and h_new[i+k] < 1:
                    h_new[i] = max(0, h_new[i] - gain)
                    h_new[i+k] = min(1, h_new[i+k] + gain)
        else:
            return h_new
        
        # Rescale to maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return np.clip(h_new, 0, 1)
    
    def random_perturb(h, w=3):
        h_new = h.copy()
        # Randomly perturb close to some shifts
        indices = np.random.choice(n_points, size=min(w*2, n_points//2), replace=False)
        pert = np.random.normal(0, 0.01, len(indices))
        for i, idx in enumerate(indices):
            h_new[idx] += pert[i]
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    start_time = time.time()
    best_c5 = compute_overlap(best_h)
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Sample several worst shifts
        worst_shifts = get_worst_shifts(h, num=min(10, n_points//2))
        
        # Try to fix each of the worst shifts
        improved = False
        for k in worst_shifts:
            h_new = adjust_h(h, k, w=3)
            c5_new = compute_overlap(h_new)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                improved = True
                no_improvement_count = 0
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
            else:
                # Maybe try a random perturbation around this shift
                h_new = random_perturb(h, w=2)
                c5_new = compute_overlap(h_new)
                if c5_new < best_c5:
                    h = h_new
                    best_c5 = c5_new
                    improved = True
                    no_improvement_count = 0
                    best_h = h.copy()
                    best_c5 = c5_new
        
        # Occasionally random perturbation
        if iteration % 20 == 0 or not improved:
            h = random_perturb(h, w=5)
            
        # Check convergence conditions
        no_improvement_count += 1
        if no_improvement_count > max_no_improvement:
            break
    
    # Final validation and refinement
    c5_final = compute_overlap(best_h)
    if c5_final < best_c5:
        best_c5 = c5_final
    
    return (best_h.tolist(), best_c5, n_points)
