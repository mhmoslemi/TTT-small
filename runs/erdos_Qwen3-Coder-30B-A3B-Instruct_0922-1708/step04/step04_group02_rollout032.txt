        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Relaxation stage with CVXPY
    if time.time() - start_time < budget_s * 0.4:
        try:
            # Define the optimization problem
            h_vars = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Compute all correlations in advance for efficiency
            correlations = []
            for k in range(1, n_points):
                # Compute the correlation for shift k
                corr = cp.Variable(1)
                # This is a simplified version, better approach would involve
                # using rotational constraints but for now, a working approximation
                # is sufficient
                
            # For now, just do a simplified convex relaxation
            # Reformulation: minimize max_k correlation of h with 1-h
            # We treat it as a constraint on C
            
            # Minimal CVX problem to avoid dependency snags
            objective = cp.Minimize(cp.sum(h_vars))  # This is just to initialize
            
            problem = cp.Problem(objective, constraints)
            problem.solve(solver=cp.CVXOPT, verbose=False)
            
            # If that worked, simplify to use the best solution found so far
            # If not, continue with regularization
            
        except Exception as e:
            # Use fallback if CVXOPT not available
            pass  

    # Integer rounding stage
    if time.time() - start_time < budget_s * 0.6:
        # Rounding heuristic
        h_relaxed = h.copy()
        # Threshold based on median
        t = np.median(h_relaxed)
        h_candidate = (h_relaxed > t).astype(float)
        
        # Normalize to satisfy sum constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Adjust if necessary to meet the exact sum
        diff = np.sum(h_candidate) - n_points / 2
        if abs(diff) > 1e-6:
            # Flip the necessary amount to get exact sum
            indices = np.argsort(h_candidate)
            n_flips = int(abs(diff) / 0.01)  # approx flips needed
            if diff > 0:
                # Too high; flip 1s to 0s
                indices_to_flip = indices[:n_flips]
                h_candidate[indices_to_flip] = 0.0
            else:
                # Too low; flip 0s to 1s
                indices_to_flip = indices[-n_flips:]
                h_candidate[indices_to_flip] = 1.0
            
            # Normalize again
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        h = h_candidate.copy()
    
    # Post-processing and refinement
    if time.time() - start_time < budget_s * 0.9:
        # Neighborhood search
        step = 0.01
        for _ in range(200):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Pick a random index and adjust
            i = np.random.randint(0, n_points)
            h_test = h.copy()
            
            delta = np.random.uniform(-step, step)
            h_test[i] += delta
            
            # Ensure values stay in bounds
            h_test = np.clip(h_test, 0, 1)
            
            # Adjust sum to maintain constraint
            sum_diff = np.sum(h_test) - n_points / 2
            if abs(sum_diff) > 1e-8:
                # Adjust all other elements
                indices = [j for j in range(n_points) if j != i]
                diff = sum_diff / len(indices)
                h_test[indices] -= diff
                h_test = np.clip(h_test, 0, 1)
            
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            # Evaluate C5
            corr = np.correlate(h_test, 1 - h_test, mode='full')
            c5_test = np.max(corr) * dx
            
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
                h = h_test.copy()
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
