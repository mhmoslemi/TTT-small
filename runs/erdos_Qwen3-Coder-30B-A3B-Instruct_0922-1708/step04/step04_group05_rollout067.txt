import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Scale to meet constraint sum(h) * dx = 1
    h = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient descent with frequency domain modification
    gamma = 0.001
    max_iter = 10000
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute overlap via FFT
        h_full = np.concatenate([h, np.zeros(n_points)])
        h_conj = np.concatenate([h, np.zeros(n_points)])
        conv_fft = np.fft.fft(h_full) * np.fft.fft(1 - h_conj)
        overlap = np.fft.ifft(conv_fft).real[:2*n_points-1]
        
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx]
        
        # Update best solution
        current_c5 = max_overlap * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient in frequency domain
        # Direction that reduces the maximum overlap 
        k = max_idx - n_points + 1
        if k < 0:
            k = k + 2*n_points - 1  # wrap around
        k = k % (2*n_points-1)
        
        # Generate frequency domain filter to reduce maximum overlap
        omega = np.arange(2*n_points-1)
        # Create a filter that tries to reduce overlap at position k
        # Simple projector in frequency domain to reduce the most violating shift
        if k <= n_points-1:
            # Use a smooth filter function to reduce that specific shift
            filter_g = np.exp(-1j * omega * k / (n_points * 10))
        else:
            # wrap negative shifts
            k_wrapped = (2*n_points-1) - k
            filter_g = np.exp(-1j * omega * k_wrapped / (n_points * 10))
        
        # Apply filter
        h_freq = np.fft.fft(h)
        def apply_filter(f):
            return f * filter_g / (np.abs(filter_g) + 1e-10)  # Prevent division by zero
        h_smooth = apply_filter(h_freq)
        h_new = np.fft.ifft(h_smooth).real[:n_points]
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update with momentum
        h = (1 - gamma) * h + gamma * h_new
        
        # Project again
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate periodically
        if iter_count % 500 == 0 and gamma > 1e-6:
            gamma *= 0.95
            
    # Final evaluation
    h_full = np.concatenate([best_h, np.zeros(n_points)])
    h_conj = np.concatenate([best_h, np.zeros(n_points)])
    conv_fft = np.fft.fft(h_full) * np.fft.fft(1 - h_conj)
    overlap = np.fft.ifft(conv_fft).real[:2*n_points-1]
    final_c5 = np.max(overlap) * dx
    
    return (best_h.tolist(), final_c5, n_points)
