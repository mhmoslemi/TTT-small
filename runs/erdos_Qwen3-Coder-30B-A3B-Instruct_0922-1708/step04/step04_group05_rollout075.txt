import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize initial h to satisfy integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_integral(h):
        return np.sum(h) - n_points / 2

    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])

    # PSO Parameters
    n_particles = 20
    max_iter = 200
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    
    # Initialize PSO
    particles = np.random.uniform(0, 1, (n_particles, n_points)) * 0.5
    for i in range(n_particles):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    pbest = particles.copy()
    pbest_vals = np.array([objective(p) for p in particles])
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO loop
    start_time = time.time()
    iter_count = 0
    no_improve_count = 0
    
    while time.time() - start_time < budget_s - 2 and iter_count < max_iter:
        iter_count += 1
        w = w_start - (w_start - w_end) * iter_count / max_iter
        
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize particle to satisfy integral constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate particle
            curr_val = objective(particles[i])
            
            # Update personal best
            if curr_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = curr_val
                
                # Update global best
                if curr_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = curr_val
                    no_improve_count = 0
                else:
                    no_improve_count += 1
        
        # Adaptative swarm size: reduce if no improvement
        if no_improve_count > 50:
            n_particles = max(5, n_particles - 1)
            no_improve_count = 0
    
    # Refinement with line search
    def line_search(h0, direction, max_alpha=1.0):
        def obj(alpha):
            h_new = h0 + alpha * direction
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            return objective(h_new)
        
        # Golden section search
        phi = (1 + np.sqrt(5)) / 2
        resphi = 2 - phi
        a, b = 0.0, max_alpha
        v = a + resphi * (b - a)
        w = b - resphi * (b - a)
        fv = obj(v)
        fw = obj(w)
        
        for _ in range(50):
            if fv < fw:
                b = w
                w = v
                v = a + resphi * (b - a)
                fw = fv
                fv = obj(v)
            else:
                a = v
                v = w
                w = b - resphi * (b - a)
                fv = fw
                fw = obj(w)
        
        return (a + b) / 2
    
    # Collect top particles for refinement
    top_indices = np.argsort(pbest_vals)[:min(5, len(pbest_vals))]
    refined_particles = []
    
    for idx in top_indices:
        h_refine = pbest[idx].copy()
        
        for _ in range(50):
            # Generate random direction
            direction = np.random.normal(0, 0.01, n_points)
            # Project direction to satisfy constraints
            direction = direction - np.mean(direction)
            
            alpha = line_search(h_refine, direction)
            h_new = h_refine + alpha * direction
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            if objective(h_new) < objective(h_refine):
                h_refine = h_new
            else:
                break
        
        refined_particles.append(h_refine)
    
    # Final evaluation and return
    best_h = gbest
    final_c5 = gbest_val
    
    for h in refined_particles:
        c5_candidate = objective(h)
        if c5_candidate < final_c5:
            final_c5 = c5_candidate
            best_h = h.copy()
    
    return (best_h.tolist(), final_c5, n_points)
