import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    # Generate diverse initial candidates
    candidates = []
    candidates.append(h_init.copy())
    
    # Add perturbed versions
    for _ in range(19):
        h_pert = h_init + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        candidates.append(h_pert)
        
    # Optimization loop
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Evaluate all candidates
        current_best_c5 = float('inf')
        current_best_h = None
        
        for h in candidates:
            # Compute overlap integral
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < current_best_c5:
                current_best_c5 = c5
                current_best_h = h.copy()
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
            last_improvement = iter_count
            
        # Gradient-based refinement for the best candidate
        h = current_best_h.copy()
        
        # Gradient computation using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update with gradient descent
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update candidates with new refined version
        candidates[0] = h_new
        
        # Add new candidate occasionally to maintain diversity
        if iter_count % 10 == 0 and time.time() - start_time < budget_s - 2:
            noise = np.random.normal(0, 0.01, n_points)
            h_div = h_new + noise
            h_div = np.clip(h_div, 0, 1)
            h_div = h_div * (n_points / 2) / np.sum(h_div)
            candidates.append(h_div)
            
            # Remove the oldest candidate if we have too many
            if len(candidates) > 20:
                candidates.pop(0)
        
        # Early termination if no improvement in a while
        if iter_count - last_improvement > 50 and time.time() - start_time > budget_s * 0.7:
            break
    
    # Final optimization on best found
    h = best_h.copy()
    for _ in range(50):
        if time.time() - start_time > budget_s - 2:
            break
            
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new

    # Final check
    final_corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (h.tolist(), final_c5, n_points)
