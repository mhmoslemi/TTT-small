import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # Phase 1: Convex relaxation using CVXPY
    print("Starting convex relaxation phase...")
    try:
        # Create variables
        h_vars = cp.Variable(n_points)
        
        # Sum constraint
        sum_constraint = cp.Equality(cp.sum(h_vars), n_points / 2)
        
        # Bound constraints
        bound_constraints = [0 <= h_vars, h_vars <= 1]
        
        # Objective: minimize sum of all overlaps
        # This is an approximation - we'll use a different approach for actual minimization
        # Add a dummy objective to create the problem
        obj = cp.Minimize(cp.sum(h_vars))  # dummy objective, we'll ignore this
        
        # Create problem
        prob = cp.Problem(obj, bound_constraints + [sum_constraint])
        
        # Solve (this is just for getting an initial feasible solution)
        prob.solve(solver=cp.SCS, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_init = np.array(h_vars.value).flatten()
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            h = h_init.copy()
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    except Exception as e:
        print(f"Convex relaxation failed: {e}")
        
    # Phase 2: Projected gradient descent
    print("Starting gradient descent phase...")
    
    # Optimized finite difference gradient computation
    def compute_gradient(h_vals, eps=1e-4):
        grad = np.zeros_like(h_vals)
        c5_base = compute_c5(h_vals)
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    # Simplified gradient descent with adaptive learning rate
    max_iter = 5000
    patience = 50
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        if iteration % 100 == 0:
            print(f"Iteration {iteration} - C5: {best_c5:.6f}")
        
        # Compute gradient
        grad = compute_gradient(h)
        
        # Adaptive learning rate
        lr = 0.01 / (1 + iteration * 1e-4)
        
        # Update
        h_new = h - lr * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new value
        new_c5 = compute_c5(h_new)
        
        # If improvement, update h
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Check for early stopping
        if no_improvement_count >= patience:
            break
            
        h = h_new.copy()
    
    # Final check with the best solution found
    if best_c5 == float('inf'):
        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
