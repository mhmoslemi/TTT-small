import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution levels to try
    resolutions = [50, 100, 200]
    
    # Best solution found
    best_h = None
    best_c5 = float('inf')
    
    # For each resolution level
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # Initialize with provided parent if available
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h)
            h = h / np.sum(h) * (n_points / 2)
        else:
            # Start with a block function
            h = np.zeros(n_points)
            h[:n_points//2] = 2.0
            h = h / np.sum(h) * (n_points / 2)
        
        # Store the best at current resolution
        current_best_h = h.copy()
        current_best_c5 = float('inf')
        
        # Use a population-based approach with local refinement
        population_size = 50
        
        # Create initial population
        population = []
        for _ in range(population_size):
            if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
                individual = h.copy()
            else:
                individual = np.random.rand(n_points)
                individual = individual / np.sum(individual) * (n_points / 2)
                individual = np.clip(individual, 0, 1)
            population.append(individual)
            
        # Genetic algorithm within this resolution
        gen_count = 0
        max_gens = 200 if n_points == 50 else 100 if n_points == 100 else 50
        
        while gen_count < max_gens and time.time() - start_time < budget_s - 1:
            # Evaluate population
            fitness = []
            for individual in population:
                corr = correlate(individual, 1 - individual, mode='full')
                c5 = np.max(corr) * dx
                fitness.append(c5)
            
            # Sort by fitness
            sorted_indices = np.argsort(fitness)
            population = [population[i] for i in sorted_indices]
            fitness = [fitness[i] for i in sorted_indices]
            
            # Update best found so far
            if fitness[0] < current_best_c5:
                current_best_c5 = fitness[0]
                current_best_h = population[0].copy()
                
            if fitness[0] < best_c5:
                best_c5 = fitness[0]
                best_h = population[0].copy()
            
            # Keep top 30%
            keep_size = max(10, int(0.3 * population_size))
            selected = population[:keep_size]
            
            # Generate new population
            new_population = selected.copy()
            
            # Crossover and mutation for rest
            while len(new_population) < population_size:
                # Pick two parents
                parent1 = selected[np.random.randint(0, len(selected))]
                parent2 = selected[np.random.randint(0, len(selected))]
                
                # SBX crossover
                if np.random.random() < 0.6:
                    alpha = np.random.random()
                    child1 = alpha * parent1 + (1 - alpha) * parent2
                    child2 = (1 - alpha) * parent1 + alpha * parent2
                else:
                    child1 = parent1.copy()
                    child2 = parent2.copy()
                
                # Mutation (polynomial mutation)
                eta_m = 20
                for i in range(n_points):
                    if np.random.random() < 0.1:
                        r = np.random.random()
                        if r < 0.5:
                            delta = (2 * r) ** (1.0 / (eta_m + 1)) - 1
                        else:
                            delta = 1 - (2 * (1 - r)) ** (1.0 / (eta_m + 1))
                        
                        child1[i] += delta
                        child2[i] += delta
                
                # Repair to make valid
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
                
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            
            population = new_population
            gen_count += 1
        
        # Local refinement for the best found at this resolution
        refine_h = current_best_h.copy()
        refine_n = 200  # Final refinement to higher resolution if needed
        
        # Refine on a finer grid
        if refine_n > n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_refine = np.linspace(0, 2, refine_n)
            refine_h = np.interp(x_refine, x_orig, refine_h)
            refine_h = refine_h * (refine_n / 2) / np.sum(refine_h)
            
            corr = correlate(refine_h, 1 - refine_h, mode='full')
            refined_c5 = np.max(corr) * (2.0 / refine_n)
            
            if refined_c5 < current_best_c5:
                current_best_c5 = refined_c5
                current_best_h = refine_h
        
        # Keep best overall
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
    
    # Final gradient descent refinement if time allows 
    if best_h is not None and time.time() - start_time < budget_s - 1:
        # Create refined copy for local optimization
        h_ref = best_h.copy()
        dx = 2.0 / len(h_ref)
        
        # Gradient descent refinement
        def objective(h):
            # Compute C5 using convolution
            corr = correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
            
        def gradient(h):
            # Approximate gradient using finite differences:
            # Since the gradient computation can be complex, here's a simple approx:
            eps = 1e-6
            grad = np.zeros_like(h)
            base_val = objective(h)
            
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (len(h_ref) / 2) / np.sum(h_plus)
                grad[i] = (objective(h_plus) - base_val) / eps
                
            return grad
            
        # Newton-style gradient descent
        lr = 0.005
        tol = 1e-6
        for _ in range(200):
            if time.time() - start_time > budget_s - 1:
                break
                
            grad = gradient(h_ref)
            h_new = h_ref - lr * grad
            
            # Repair for constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (len(h_ref) / 2) / np.sum(h_new)
            
            if np.allclose(h_ref, h_new, atol=tol):
                break
                
            h_ref = h_new
            
        # Check final C5
        corr_final = correlate(h_ref, 1 - h_ref, mode='full')
        final_c5 = np.max(corr_final) * dx
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_ref.copy()
    
    if best_h is None:
        # Fallback: use a uniform random solution
        n_points = 100
        best_h = np.random.rand(n_points)
        best_h = best_h / np.sum(best_h) * (n_points / 2)
        dx = 2.0 / n_points
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    # Ensure parameters are correct one last time
    # Final check
    assert len(best_h) == 100  # standard return size
    dx = 2.0 / len(best_h)
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, 100)
