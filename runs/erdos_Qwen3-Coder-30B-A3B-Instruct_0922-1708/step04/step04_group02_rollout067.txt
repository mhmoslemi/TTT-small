from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_init)
        h = h_init.copy()

        h = np.ones(n_points) * 0.5
    
    # Normalize initial h to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Discretization
    dx = 2.0 / n_points
    
    def compute_c5(h_values):
        """Compute C5 value for given h"""
        correlation = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(correlation) * dx
    
    def evaluate(h_vals):
        """Evaluate objective function"""
        return compute_c5(h_vals)
    
    # Define the constraints
    def constraint_sum(h_vals):
        return np.sum(h_vals) * dx - 1.0
    
    # Bounds for variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using scipy minimize
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start method
    max_starts = 3
    for start in range(max_starts):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Initialize with either initial_h or random
        if start == 0 and 'initial_h_values' in globals():
            h_current = h.copy()
        else:
            h_current = np.random.uniform(0, 1, n_points)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Ensure bounds
        h_current = np.clip(h_current, 0, 1)
        
        # Optimize with constraints
        try:
            # Use SLSQP method which handles constraints well
            result = minimize(
                evaluate,
                h_current,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 500, 'ftol': 1e-8}
            )
            
            if result.success:
                c5_val = evaluate(result.x)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = result.x.copy()
        except:
            pass
    
    # Local refinement with gradient method if time allows
    if time.time() - start_time < budget_s - 10:
        # Simple gradient-based local search
        step_size = 0.005
        iterations = 100
        h_refined = best_h.copy()
        
        for i in range(iterations):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Estimate gradient by finite differences
            grad = np.zeros(n_points)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_c5(h_plus)
                
                h_minus = h_refined.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                c5_minus = compute_c5(h_minus)
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new = h_refined - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Accept only if improvement
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h_refined):
                h_refined = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_refined.copy()
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
