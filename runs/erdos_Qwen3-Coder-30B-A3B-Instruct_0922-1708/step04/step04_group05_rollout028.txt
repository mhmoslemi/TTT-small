import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    # PSO initialization
    n_particles = 20
    particles = []
    velocities = []
    
    for _ in range(n_particles):
        h = h_init + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.normal(0, 0.01, n_points))
    
    pbest = [p.copy() for p in particles]
    pbest_c5 = [float('inf')] * n_particles
    
    gbest = h_init.copy()
    gbest_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO parameters
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    max_iter = 200
    no_improve_iter = 0
    max_no_improve = 50
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate current particle
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
            if c5 < gbest_c5:
                gbest = particles[i].copy()
                gbest_c5 = c5
                
                if gbest_c5 < best_c5:
                    best_c5 = gbest_c5
                    best_h = gbest.copy()
                    
        # Check for no improvement
        if gbest_c5 >= best_c5 or (iter_num > 0 and iter_num % 10 == 0):
            no_improve_iter += 1
            if no_improve_iter > max_no_improve:
                # Reduce particles if no improvement
                n_particles = max(10, n_particles - 2)
                if n_particles < 10:
                    n_particles = 10
                no_improve_iter = 0
                # Reinitialize particles
                particles = []
                velocities = []
                for _ in range(n_particles):
                    h = gbest + np.random.normal(0, 0.05, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    particles.append(h.copy())
                    velocities.append(np.random.normal(0, 0.01, n_points))
                pbest = [p.copy() for p in particles]
                pbest_c5 = [float('inf')] * n_particles
        else:
            no_improve_iter = 0
            
        # Add noise occasionally to maintain diversity
        if iter_num % 20 == 0:
            for i in range(n_particles):
                if np.random.random() < 0.3:
                    noise = np.random.normal(0, 0.02, n_points)
                    particles[i] = np.clip(particles[i] + noise, 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Refinement with line search for top candidates
    top_candidates = []
    for i in range(n_particles):
        corr = np.correlate(particles[i], 1 - particles[i], mode='full')
        c5 = np.max(corr) * dx
        top_candidates.append((c5, particles[i].copy()))
    
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:5]
    
    refined_candidates = []
    for c5, h in top_candidates:
        h_refined = h.copy()
        # Perform gradient descent
        for _ in range(50):
            if time.time() - start_time > budget_s - 2:
                break
            grad = np.zeros(n_points)
            eps = 1e-6
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus) 
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - c5) / eps
            lr = 0.01 / (1.0 + np.linalg.norm(grad) * 1e-3)
            h_refined = h - lr * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5 = np.max(corr) * dx
            if c5 >= c5 - 1e-8:  # No improvement
                break
            h = h_refined.copy()
        refined_candidates.append((c5, h_refined))
        
    if refined_candidates:
        refined_candidates.sort(key=lambda x: x[0])
        final_c5, final_h = refined_candidates[0]
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
    
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    c5_final = np.max(corr) * dx
    return (best_h.tolist(), c5_final, n_points)
