import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Use LP relaxation approach - define variables
    h_vars = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Define constraints for LP relaxation
    constraints = [h_vars >= 0, h_vars <= 1]
    constraints.append(cp.sum(h_vars) == n_points / 2)
    
    # Add auxiliary variables for correlations
    # We will build them iteratively in gradient descent
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # First, use a simple gradient descent refinement from start
    max_refine_iter = 500
    lr = 0.01
    iters_no_improvement = 0
    max_no_improvement = 50
    
    for refine_iter in range(max_refine_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5
        correlation = np.correlate(h, 1 - h, mode='full') 
        c5 = np.max(correlation) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            iters_no_improvement = 0
        else:
            iters_no_improvement += 1
            if iters_no_improvement > max_no_improvement:
                break
                
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update learning rate
        lr = max(1e-6, lr * 0.999)
        
        h = h_new
        
    # Use simple local search around best solution
    h = best_h.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation
        h_pert = h.copy()
        idx = np.random.randint(0, n_points)
        h_pert[idx] += np.random.uniform(-0.02, 0.02)
        h_pert[idx] = np.clip(h_pert[idx], 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        
        correlation = np.correlate(h_pert, 1 - h_pert, mode='full')
        c5_pert = np.max(correlation) * dx
        
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
            h = h_pert.copy()
            
    # Final check of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
