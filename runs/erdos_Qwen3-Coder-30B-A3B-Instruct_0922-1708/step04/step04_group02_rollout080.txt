import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation approach
    def solve_convex_relaxation(h_init):
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Estimate maximum correlation
        # For now, just create a simple convex constraint approximation
        # We'll discretize the lag space
        lags = list(range(1, n_points))
        for k in lags[:20]:  # Limit for faster solving
            # Approximate by linear constraint
            term = 0
            for i in range(n_points - k):
                term += h_vars[i] * (1 - h_vars[i + k])
            constraints.append(term <= C * dx**(-1))
            
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h_vars.value
        except:
            pass
        return h_init
    
    # Try multiple different analysis points
    for i in range(3):
        time_remaining = budget_s - (time.time() - start_time)
        if time_remaining < 5:
            break
            
        # Init different versions
        h_init = h.copy()
        if i == 1:
            # Rotate and normalize
            h_init = np.roll(h_init, 10)
        elif i == 2:
            # Random noise + normalization
            h_init += np.random.normal(0, 0.05, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
                
        # Convex Relaxation
        h_relaxed = solve_convex_relaxation(h_init)
        if h_relaxed is not None:
            h_test = h_relaxed.copy()
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            # Integer rounding to binary
            threshold = np.median(h_test)
            h_binary = (h_test > threshold).astype(float)
            
            # Adjust if needed to maintain sum constraint
            count = np.sum(h_binary)
            if count != n_points / 2:
                diff = int(n_points/2 - count)
                if diff > 0:
                    indices = np.where(h_binary == 0)[0]
                    if len(indices) >= diff:
                        flip_indices = np.random.choice(indices, diff, replace=False)
                        h_binary[flip_indices] = 1
                else:
                    indices = np.where(h_binary == 1)[0]
                    if len(indices) >= -diff:
                        flip_indices = np.random.choice(indices, -diff, replace=False)
                        h_binary[flip_indices] = 0
                        
            # Evaluate final C5
            correlation = np.correlate(h_binary, 1 - h_binary, mode='full')
            c5_binary = np.max(correlation) * dx
            
            if c5_binary < best_c5:
                best_c5 = c5_binary
                best_h = h_binary.copy()
        
        # Fine-tune with local search if sufficient time
        if time.time() - start_time < budget_s - 10 and best_c5 > 0.3807:
            h_current = best_h.copy()
            for _ in range(100):
                if time.time() - start_time > budget_s - 5:
                    break
                h_test = h_current.copy()
                
                # Modify one point
                idx = np.random.randint(0, n_points)
                h_test[idx] = np.clip(h_test[idx] + np.random.normal(0, 0.05), 0, 1)
                
                # Rescale
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                
                # Evaluate
                correlation = np.correlate(h_test, 1 - h_test, mode='full')
                c5_test = np.max(correlation) * dx
                
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test.copy()
                    h_current = h_test.copy()
        
    return (best_h.tolist(), best_c5, n_points)
