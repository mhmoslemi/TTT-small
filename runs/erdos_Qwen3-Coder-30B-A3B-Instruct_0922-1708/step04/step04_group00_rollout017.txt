import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Store best solution found so far
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Trust region optimization using projected gradient
    max_iter = 500
    trust_radius = 0.1
    decay_rate = 0.99
    min_trust_radius = 1e-6
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    def compute_gradient(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        base_c5 = compute_c5(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    def project_and_scale(h_vec):
        h_new = np.clip(h_vec, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def solve_trust_region_subproblem(h_vec, grad, trust_radius):
        # Simple trust region subproblem solver (projected gradient)
        h_new = h_vec - 0.01 * grad  # Initial step
        h_new = project_and_scale(h_new)
        
        # Ensure we don't move too far from current point
        diff = h_new - h_vec
        if np.linalg.norm(diff) > trust_radius:
            diff = diff * trust_radius / np.linalg.norm(diff)
            h_new = h_vec + diff
        
        return project_and_scale(h_new)
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h)
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Solve trust region problem
        h_new = solve_trust_region_subproblem(h, grad, trust_radius)
        
        # Accept new point if it improves
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5:
            h = h_new.copy()
        else:
            trust_radius = max(min_trust_radius, trust_radius * decay_rate)
            if trust_radius < min_trust_radius:
                break
    
    # Final refinement with improved trust region logic
    trust_radius = 0.05
    for refinement_iter in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        grad = compute_gradient(h)
        h_new = solve_trust_region_subproblem(h, grad, trust_radius)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5:
            h = h_new.copy()
        else:
            trust_radius *= 0.95
            if trust_radius < min_trust_radius:
                break
                
    # Final check and return
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
