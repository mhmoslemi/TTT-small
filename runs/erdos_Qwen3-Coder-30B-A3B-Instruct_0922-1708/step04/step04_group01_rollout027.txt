import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    # PSO Parameters
    n_particles = 30
    max_iter = 1000
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles
    particles = [h.copy() for _ in range(n_particles)]
    velocities = [np.random.rand(n_points) * 0.1 for _ in range(n_particles)]
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [compute_c5(p, dx) for p in particles]
    global_best = best_h.copy()
    global_best_score = best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Clamp position to [0, 1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to sum = n_points / 2
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            score = compute_c5(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                
                # Update global best
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
                    
                    # Early update of best solution
                    if global_best_score < best_c5:
                        best_h = global_best.copy()
                        best_c5 = global_best_score

    # Try a refinement step with CVXPY
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Use a different approach for CVX optimization
        # Create a proxy for the maximum overlap using auxiliary variables
        # Note: We actually solve for smooth approximation rather than exact maximum due to CVXPY limitations
        
        h_refined = global_best.copy()
        
        # Try to refine using scipy optimization with constraints
        def obj(h_vec):
            return compute_c5(h_vec, dx)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        def constraint_bounds(h_vec):
            return np.concatenate([h_vec, 1-h_vec])
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints_obj = [
            {'type': 'eq', 'fun': constraint_sum},
        ]
        
        # Optimize using SLSQP
        result = minimize(
            obj, 
            h_refined, 
            method='SLSQP', 
            bounds=bounds,
            constraints=constraints_obj,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success and result.fun < global_best_score:
            global_best = result.x
            global_best_score = result.fun
            if global_best_score < best_c5:
                best_h = global_best.copy()
                best_c5 = global_best_score

    except:
        pass

    # Final check with exact evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
