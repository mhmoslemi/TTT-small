import numpy as np
import time
from scipy.optimize import minimize_scalar

def project_feasible(h, n_points):
    """Project h to satisfy bounds and integral constraint"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, n_points):
    """Compute C5 value for given h"""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_step(h, n_points, eps=1e-4):
    """Compute finite difference gradient for h"""
    dx = 2.0 / n_points
    base_corr = np.correlate(h, 1 - h, mode='full')
    base_c5 = np.max(base_corr) * dx
    grad = np.zeros_like(h)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        grad[i] = (c5_plus - base_c5) / eps
        
    return grad

def line_search(h, grad, n_points, max_iter=20):
    """Perform line search to find optimal step size"""
    def phi(alpha):
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n_points)
        return compute_c5(h_new, n_points)
    
    # Golden section search for alpha
    phi_min = float('inf')
    best_alpha = 0.0
    
    # Reduced golden ratio search
    a, b = 0, 1.0
    resphi = 2 - (1 + np.sqrt(5)) / 2
    x1 = a + resphi * (b - a)
    x2 = b - resphi * (b - a)
    
    f1 = phi(x1)
    f2 = phi(x2)
    
    for _ in range(max_iter):
        if f1 < f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = a + resphi * (b - a)
            f1 = phi(x1)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = b - resphi * (b - a)
            f2 = phi(x2)
            
    return (a + b) / 2.0

    # Outer loop: Adaptive grid search
    n_coarse = 100
    n_low = 50

    coarse_candidates = []

        h_interpolated = np.interp(x_new, x_orig, h)
        coarse_candidates.append(h_interpolated)
    
    # Evaluate coarse candidates
    coarse_scores = []
