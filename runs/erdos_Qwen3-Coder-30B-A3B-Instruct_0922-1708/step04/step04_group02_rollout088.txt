import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    swarm_size = 100
    max_iterations = 900
    inertia_weight = 0.7
    cognitive_weight = 2.0
    social_weight = 2.0
    local_search_freq = 50
    rejuvenation_freq = 200
    resolution_steps = [50, 100, 200]
    n_refine = resolution_steps[-1]
    
    dx = 2.0 / n_points
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
    else:
        # Initialize with random feasible solutions
        for _ in range(30):  # Start with some random particles
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
    
    # Add more diverse initial particles
    for i in range(30, swarm_size):
        # Create some structured initial solutions
        h = np.zeros(n_points)
        # Try to e.g., have some blocks with specific heights
        if i % 3 == 0:
            # Uniforms
            h[:] = 1.0
        elif i % 3 == 1:
            # Piecewise constant
            mid = n_points // 2
            h[:mid] = 2.0
            h[mid:] = 0.5
        else:
            # Random with sum constraint
            h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
    
    # Initialize velocities
    for _ in range(len(particles)):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize personal best
    pbest = [h.copy() for h in particles]
    pbest_c5 = [float('inf')] * len(particles)
    
    # Initialize global best
    gbest_idx = 0
    gbest_c5 = float('inf')
    
    # Evaluate initial particles
    for i, h in enumerate(particles):
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        pbest_c5[i] = c5
        if c5 < gbest_c5:
            gbest_c5 = c5
            gbest_idx = i
    
    # Initialize optimizer
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 10 and iteration < max_iterations:
        iteration += 1
        
        # Update each particle
        for i in range(len(particles)):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            v = (inertia_weight * v 
                 + cognitive_weight * r1 * (pbest[i] - h) 
                 + social_weight * r2 * (particles[gbest_idx] - h))
            
            # Update position
            h_new = h + v
            
            # Repair constraint
            h_new = np.clip(h_new, 0, 1)
            sum_h = np.sum(h_new)
            target_sum = n_points / 2
            
            if abs(sum_h - target_sum) > 1e-6:
                diff = sum_h - target_sum
                if diff > 0:
                    # Excess, reduce largest elements
                    idxs = np.argsort(h_new)[::-1]
                    for j in idxs:
                        if diff <= 0:
                            break
                        reduction = min(h_new[j], diff)
                        h_new[j] -= reduction
                        diff -= reduction
                else:
                    # Deficit, increase smallest elements
                    idxs = np.argsort(h_new)
                    for j in idxs:
                        if diff >= 0:
                            break
                        addition = min(1 - h_new[j], -diff)
                        h_new[j] += addition
                        diff += addition
            
            # Normalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            velocities[i] = v
            
            # Evaluate fitness
            corr = correlate(h_new, 1 - h_new, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest_c5[i] = c5
                pbest[i] = h_new.copy()
            
            # Update global best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_idx = i
                
        # Local search every "local_search_freq"
        if iteration % local_search_freq == 0:
            # Perform local search on top 10%
            top_indices = np.argsort(pbest_c5)[:swarm_size//10]
            for i in top_indices:
                h = pbest[i].copy()
                # Gradient-based refinement
                for _ in range(10):
                    eps = 1e-4
                    grad = np.zeros_like(h)
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - pbest_c5[i]) / eps
                    
                    h_new = h - 0.001 * grad
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    corr_new = correlate(h_new, 1 - h_new, mode='full')
                    c5_new = np.max(corr_new) * dx
                    
                    if c5_new < pbest_c5[i]:
                        h = h_new
                        pbest_c5[i] = c5_new
                        pbest[i] = h.copy()
                    else:
                        break
        
        # Rejuvenation every "rejuvenation_freq"
        if iteration % rejuvenation_freq == 0:
            # Replace worst 10% with new random particles
            worst_indices = np.argsort(pbest_c5)[-swarm_size//10:]
            for i in worst_indices:
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[i] = h
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                # Check if new solution is better than personal best
                corr = correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                if c5 < pbest_c5[i]:
                    pbest_c5[i] = c5
                    pbest[i] = h.copy()
                if c5 < gbest_c5:
                    gbest_c5 = c5
                    gbest_idx = i
                            
    # Final refinement with more gradient descent
    final_candidates = [pbest[i] for i in np.argsort(pbest_c5)[:5]]
    best_final_h = None
    best_final_c5 = float('inf')
    
    for h_trial in final_candidates:
        h_refined = h_trial.copy()
        for _ in range(1000):
            # Estimate gradient
            eps = 1e-5
            grad = np.zeros_like(h_refined)
            c5_current = float('inf')
            
            corr_current = correlate(h_refined, 1 - h_refined, mode='full')
            c5_current = np.max(corr_current) * dx
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - c5_current) / eps
            
            h_refined = h_refined - 0.001 * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            corr_test = correlate(h_refined, 1 - h_refined, mode='full')
            c5_test = np.max(corr_test) * dx
            
            if c5_test < best_final_c5:
                best_final_c5 = c5_test
                best_final_h = h_refined.copy()
            else:
                break
        if best_final_c5 < gbest_c5:
            gbest_c5 = best_final_c5
            gbest_idx = -1  # Not actually persistent
            
        if best_final_c5 < gbest_c5:
            gbest_c5 = best_final_c5
            gbest_idx = -1  # not really used but holds best
            
    # Use the best found solution to round it properly
    if best_final_h is not None:
        final_h = best_final_h
    else:
        final_h = particles[gbest_idx]
    
    # Recompute final C5 to make sure
    corr_final = correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (final_h.tolist(), final_c5, n_points)
