import numpy as np
import time

def project_feasible(h, n_points):
    """Project h to feasible region: clip to [0,1] and normalize sum to n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, n_points):
    """Compute the C5 value from h"""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_step(h, n_points, eps=1e-4):
    """Compute gradient of C5 w.r.t. h using finite differences"""
    grad = np.zeros_like(h)
    h_orig = h.copy()
    c5_orig = compute_c5(h, n_points)
    
    for i in range(n_points):
        h_plus = h_orig.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        c5_plus = compute_c5(h_plus, n_points)
        grad[i] = (c5_plus - c5_orig) / eps
    
    return grad

def line_search(h, grad, n_points, max_iter=20):
    """Perform line search to find optimal step size"""
    alpha_max = 1.0
    alpha = alpha_max
    h_new = project_feasible(h - alpha * grad, n_points)
    c5_new = compute_c5(h_new, n_points)
    
    # Golden section search
    phi = (1 + np.sqrt(5)) / 2
    resphi = 2 - phi
    
    a, b = 0.0, alpha_max
    v = a + resphi * (b-a)
    w = b - resphi * (b-a)
    fv = compute_c5(project_feasible(h - v * grad, n_points), n_points)
    fw = compute_c5(project_feasible(h - w * grad, n_points), n_points)
    
    for _ in range(max_iter):
        if abs(b-a) < 1e-8:
            break
        if fv < fw:
            b, w, fw = w, v, fv
            v = a + resphi * (b-a)
            fv = compute_c5(project_feasible(h - v * grad, n_points), n_points)
        else:
            a, v, fv = v, w, fw
            w = b - resphi * (b-a)
            fw = compute_c5(project_feasible(h - w * grad, n_points), n_points)
    
    return project_feasible(h - v * grad, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Get initial h if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    h = project_feasible(h_init, n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    
    start_time = time.time()
    
    # Adaptive grid search with refinement
    # Generate coarse candidates
    N_coarse = 100
    coarse_samples = []
    for i in range(N_coarse):
        h_sample = np.random.rand(n_points)
        h_sample = project_feasible(h_sample, n_points)
        c5_val = compute_c5(h_sample, n_points)
        coarse_samples.append((c5_val, h_sample))
    
    coarse_samples.sort(key=lambda x: x[0])
    top_candidates = coarse_samples[:10]  # Top 10% for refinement
    
    # Refine top candidates
    for _, h_start in top_candidates:
        h_refined = h_start
        h_refined = project_feasible(h_refined, n_points)
        
        for iter_count in range(100):  # Max iterations per refinement
            if time.time() - start_time > budget_s - 5:
                break
                
            grad = gradient_step(h_refined, n_points)
            h_refined = line_search(h_refined, grad, n_points)
            
            c5_val = compute_c5(h_refined, n_points)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_refined.copy()
                
            # Escape local minima with small noise
            if iter_count % 50 == 0:
                noise = np.random.normal(0, 0.001, n_points)
                h_refined = project_feasible(h_refined + noise, n_points)
                
            if iter_count % 100 == 0 and time.time() - start_time > budget_s - 5:
                break
    
    # Final validation
    final_c5 = compute_c5(best_h, n_points)
    return (best_h.tolist(), final_c5, n_points)
