    h0 = h.copy()

    # Quadratic programming approach via convex relaxation
    # We're trying to minimize sum of overlaps using a continuous 
    # variable approach

    # We'll warm start with the initial solution and perform a local search
    h = h0.copy()
    
    # Phase 2: Projected gradient descent
    max_iter = 1000
    proj_grad_iter = 100
    
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_grad(h, eps=1e-4):
        grad = np.zeros_like(h)
        h_base = h.copy()
        for i in range(len(h)):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c_plus = np.max(corr_plus) * dx
            
            corr_base = np.correlate(h_base, 1 - h_base, mode='full')
            c_base = np.max(corr_base) * dx
            
            grad[i] = (c_plus - c_base) / eps
        return grad

    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:

        # Compute the current value
        current_c5 = compute_overlap(h)

            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = compute_grad(h)
        
        # Adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project onto constraints: [0,1] and sum = n_points/2
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check for improvement in the main loop
        new_c5 = compute_overlap(h_new)
        
        # Check if the new solution is better
        if new_c5 < current_c5:
            h = h_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping if no improvement for many iterations
        if no_improvement_count > 50:
            break
            
    return (best_h.tolist(), best_c5, n_points)
