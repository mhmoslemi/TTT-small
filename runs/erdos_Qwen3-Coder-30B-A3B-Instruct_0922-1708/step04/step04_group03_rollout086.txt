import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation using CVXPY to get a good starting point
    # Try to minimize sum of all correlations (not max)
    h_var = cp.Variable(n_points)
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    objective = cp.Minimize(cp.sum(cp.conv(cp.reshape(h_var, (1, -1)), 
                                          cp.reshape(1 - h_var, (-1, 1)), 
                                          'full')))
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.ECOS, verbose=False)
    
    if problem.status == cp.OPTIMAL:
        h_initial = np.array(h_var.value).flatten()
        # Ensure proper sum
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h = h_initial.copy()
    else:
        # fallback: use the parent or uniform
        h = np.ones(n_points) * 0.5
        
    # Gradient descent refinement
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
        
    def compute_gradient(h, eps=1e-4):
        grad = np.zeros_like(h)
        c5_base = compute_c5(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Ensure constraints after perturbation
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad

    # Gradient descent iterations
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Optimize learning rate
        lr = 0.01 / (1.0 + np.linalg.norm(grad) * 1e-3)
        
        # Update step
        h_new = h - lr * grad
        
        # Re-project to constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update best solution
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
