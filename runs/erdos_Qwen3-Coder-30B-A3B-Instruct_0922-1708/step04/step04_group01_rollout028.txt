import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_overlap(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or block function
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    start_time = time.time()
    
    # Particle Swarm Optimization parameters
    n_particles = 30
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            p = np.array(initial_h_values)
            p = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), p)
            p = p / np.sum(p) * (n_points / 2)
        else:
            p = np.zeros(n_points)
            p[:n_points//2] = 2.0
            # Add small random noise
            noise = np.random.normal(0, 0.1, n_points)
            p = np.clip(p + noise, 0, 1)
            p = p / np.sum(p) * (n_points / 2)
        particles.append(p)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    pbest = [p.copy() for p in particles]
    pbest_vals = [evaluate_overlap(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO main loop
    iter_count = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints - clip to [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize sum to n_points/2
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            fitness = evaluate_overlap(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = fitness
                
                # Update global best if needed
                if fitness < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = fitness
        
        iter_count += 1
    
    # Final refinement using CVXPY
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Reconstruct the optimization problem for better solution
        # It's an optimization of a maximum over lags, which can be challenging
        # We'll use a relaxation approach
        
        # For a more direct approach, simply use the PSO result for the CVXPY warm start
        h_cvx = gbest.copy()
        h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)  # Ensure sum constraint
        
        # Try a simple refinement using scipy's minimize
        def obj(h_vec):
            return evaluate_overlap(h_vec, dx)
        
        # Use the best solution found so far as initial guess
        result = minimize(obj, h_cvx, method='L-BFGS-B', bounds=[(0,1)]*n_points,
                          options={'maxiter': 200, 'ftol': 1e-8})
        if result.success:
            h_refined = result.x
            # Normalize and validate
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            c5_refined = evaluate_overlap(h_refined, dx)
            
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
    except:
        pass

    # Final assessment
    final_c5 = evaluate_overlap(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
