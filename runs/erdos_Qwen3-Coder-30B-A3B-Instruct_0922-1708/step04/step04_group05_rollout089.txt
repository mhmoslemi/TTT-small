import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    h = (h_init >= 0.5).astype(int)
    
    # Ensure exactly n_points/2 ones
    count_ones = np.sum(h)
    if count_ones > n_points // 2:
        # Turn some ones into zeros
        zero_indices = np.where(h == 1)[0]
        to_flip = zero_indices[:count_ones - n_points // 2]
        h[to_flip] = 0
    elif count_ones < n_points // 2:
        # Turn some zeros into ones
        one_indices = np.where(h == 0)[0]
        to_flip = one_indices[:n_points // 2 - count_ones]
        h[to_flip] = 1
    
    # Evaluate initial C5
    correlation_full = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(correlation_full) * dx
    best_h = h.copy()
    
    # Simulated Annealing parameters
    temp = 1.0
    cooling_rate = 0.999
    perturb_frequency = 1000
    perturb_fraction = 0.05
    
    start_time = time.time()
    iterations = 0
    stagnation_count = 0
    max_stagnation = 1000
    
    while time.time() - start_time < budget_s - 1 and stagnation_count < max_stagnation:
        iterations += 1
        # Randomly flip one element
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]
        
        # Maintain exactly n_points/2 ones
        count_new = np.sum(h_new)
        if count_new > n_points // 2:
            # Flip back one of the 1s to 0
            ones_idx = np.where(h_new == 1)[0]
            flip_back = np.random.choice(ones_idx)
            h_new[flip_back] = 0
        elif count_new < n_points // 2:
            # Flip back one of the 0s to 1
            zeros_idx = np.where(h_new == 0)[0]
            flip_back = np.random.choice(zeros_idx)
            h_new[flip_back] = 1
        
        # Evaluate new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject based on simulated annealing
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            stagnation_count = 0
        else:
            # Accept with probability based on temperature
            delta = new_c5 - best_c5
            accept_prob = np.exp(-delta / temp)
            if np.random.rand() < accept_prob:
                h = h_new
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Update temperature and perturb occasionally
        temp *= cooling_rate
        if iterations % perturb_frequency == 0:
            # Randomly reset a fraction of the values
            num_reset = int(n_points * perturb_fraction)
            reset_indices = np.random.choice(n_points, num_reset, replace=False)
            for idx in reset_indices:
                h[idx] = np.random.choice([0, 1])
            
            # Maintain the constraint
            count_new = np.sum(h)
            if count_new > n_points // 2:
                ones_idx = np.where(h == 1)[0]
                flip_back = np.random.choice(ones_idx, count_new - n_points // 2, replace=False)
                h[flip_back] = 0
            elif count_new < n_points // 2:
                zeros_idx = np.where(h == 0)[0]
                flip_back = np.random.choice(zeros_idx, n_points // 2 - count_new, replace=False)
                h[flip_back] = 1
        
        # Update best if needed
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
