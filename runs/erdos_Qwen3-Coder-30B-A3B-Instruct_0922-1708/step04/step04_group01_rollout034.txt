import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    coarse_n = 20
    coarse_dx = 2.0 / coarse_n
    top_candidates = 10
    max_iter = 500
    lr = 0.01
    
    # Initialize with parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Rescale to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Explore coarse grid
    if time.time() - start_time < budget_s * 0.3:
        # Generate coarse resolution candidate points
        coarse_points = []
        # Simple Latin Hypercube sampling approach for coarse search
        for _ in range(100):
            h_coarse = np.random.rand(coarse_n)
            h_coarse = h_coarse / np.sum(h_coarse) * (coarse_n / 2)
            coarse_points.append(h_coarse)
        
        # Grid sampling approach to improve quality
        coarse_points = []
        # Sample with a few strategic inputs
        base_points = [
            np.ones(coarse_n) * 0.5,
            np.concatenate([np.ones(coarse_n//2), np.zeros(coarse_n//2)]),
            np.concatenate([np.zeros(coarse_n//2), np.ones(coarse_n//2)]),
            np.concatenate([np.ones(coarse_n//4), np.zeros(coarse_n//2), np.ones(coarse_n//4)]),
            np.ones(coarse_n) * 0.25,
            np.ones(coarse_n) * 0.75
        ]
        
        for h_base in base_points:
            coarse_points.append(h_base)
            
        # Add random mid-range points
        for _ in range(50):
            h_rand = np.random.rand(coarse_n) * 0.8 + 0.1
            h_rand = h_rand / np.sum(h_rand) * (coarse_n / 2)
            coarse_points.append(h_rand)
            
        # Evaluate all coarse points
        coarse_evals = []
        for h_c in coarse_points:
            # Upsample to fine grid
            h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                                  np.linspace(0, 2, coarse_n), h_c)
            if np.isfinite(np.sum(h_upsampled)):
                correlation = np.correlate(h_upsampled, 1 - h_upsampled, mode='full')
                c5 = np.max(correlation) * dx
                coarse_evals.append((h_upsampled, c5))
        
        # Sort and keep best candidates
        coarse_evals.sort(key=lambda x: x[1])
        top_coarse = [x[0] for x in coarse_evals[:top_candidates]]
    else:
        top_coarse = [h.copy()]
        
    # Phase 2: Refinement
    for i, h_init in enumerate(top_coarse):
        h_refine = h_init.copy()
        c5_refine = float('inf')
        
        # Local refinement with projected gradient descent
        for it in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient via finite differences
            grad = np.zeros_like(h_refine)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_refine.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h_refine.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with gradient
            h_refine = h_refine - lr * grad
            
            # Projection step (clip and rescale)
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
            
            # Check progress
            correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
            c5_update = np.max(correlation) * dx
            
            if c5_update < c5_refine:
                c5_refine = c5_update
                if c5_update < best_c5:
                    best_c5 = c5_update
                    best_h = h_refine.copy()
                    
    # Final CVX optimization if time allows
    if time.time() - start_time < budget_s - 10:
        try:
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            # Using a heuristic convex relaxation approach
            correlation_expr = cp.conv(h_vars, cp.reshape(1 - h_vars, (n_points, 1)))
            objective = cp.Minimize(cp.max(correlation_expr))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False, max_iters=1000)
            
            if prob.status == cp.OPTIMAL and h_vars.value is not None:
                h_cvx = h_vars.value
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
        except:
            pass
            
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
