import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Ensure constraints are met
    h = h * (n_points / 2) / np.sum(h)
    
    # LHS sampling approach
    lhs_samples = []
    lhs_count = 50
    
    # Generate Latin hypercube samples
    for _ in range(lhs_count):
        # Generate uniform samples for each dimension
        sample = np.random.uniform(0, 1, n_points)
        # Sort and assign to ensure LHS property
        sorted_indices = np.argsort(sample)
        # Reassign to LHS positions
        lhs_sample = np.zeros(n_points)
        for i in range(n_points):
            lhs_sample[sorted_indices[i]] = i / n_points
        # Rescale to sum constraint
        lhs_sample = lhs_sample * (n_points / 2) / np.sum(lhs_sample)
        lhs_samples.append(lhs_sample)
    
    # Add the initial h to the samples
    lhs_samples.append(h)
    
    # Refine each sample
    for i, h_candidate in enumerate(lhs_samples):
        h_current = h_candidate.copy()
        max_iter = 500
        lr = 0.01
        patience = 0
        best_val = float('inf')
        
        for iter in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute the overlap integral
            correlation = np.correlate(h_current, 1 - h_current, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_val:
                best_val = c5
                if best_val < best_c5:
                    best_c5 = best_val
                    best_h = h_current.copy()
                patience = 0
            else:
                patience += 1
                
            # Use finite differences for simple gradient estimation
            grad = np.zeros_like(h_current)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h_current.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with adaptive learning rate
            h_new = h_current - lr * grad
            
            # Clip to [0, 1] and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_current = h_new
            
            # Adaptive learning rate
            if patience > 20:
                lr *= 0.5
                patience = 0
                if lr < 1e-4:
                    lr = 1e-4
            
            if patience > 50:
                break
                
        # Make sure best solution is kept
        if best_val < best_c5:
            best_c5 = best_val
            best_h = h_current.copy()
            
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
