from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points

            h = np.interp(x_new, x_orig, h)

        h = h * (n_points / 2) / np.sum(h)

        h = np.ones(n_points) * 0.5

    best_c5 = float('inf')

    population_size = 50
    F = 0.8
    CR = 0.9
    max_no_improvement = 100
    no_improvement_count = 0
    
    # Initialize population
    population = []
    fitness = []

    if 'initial_h_values' in globals():
        population.append(h.copy())
        corr = np.correlate(h, 1 - h, mode='full')
        fitness.append(np.max(corr) * dx)
    else:
        fitness.append(float('inf'))
        
    # Fill population with random feasible solutions
    for i in range(population_size - len(population)):
        particle = np.random.uniform(0, 1, n_points)
        particle = particle * (n_points / 2) / np.sum(particle)
        population.append(particle)
        corr = np.correlate(particle, 1 - particle, mode='full')
        fitness.append(np.max(corr) * dx)
    
    # Find initial best
    best_idx = np.argmin(fitness)
    gbest = population[best_idx].copy()
    gbest_val = fitness[best_idx]
    
    if gbest_val < best_c5:
        best_c5 = gbest_val
        best_h = gbest.copy()
    
    start_time = time.time()
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False

        new_population = []
        new_fitness = []
        
        for i in range(population_size):

            idxs = np.random.choice(population_size, 3, replace=False)

            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)

            trial = np.copy(population[i])

                    trial[j] = mutant[j]

            trial = trial * (n_points / 2) / np.sum(trial)

            corr = np.correlate(trial, 1 - trial, mode='full')

            if c5 < fitness[i]:

                new_fitness.append(c5)

                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                h = h_new
                
                # Check for early termination
                if np.linalg.norm(h_new - h) < 1e-6:
                    break
            
            # Update global best from locally refined solution
            corr = np.correlate(h, 1 - h, mode='full')
            c5_refine = np.max(corr) * dx
            if c5_refine < best_c5:
                best_c5 = c5_refine
                best_h = h.copy()
            gbest = h.copy()
            gbest_val = best_c5
        
        # Early stopping
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
