import numpy as np
import time
from heapq import heappush, heappop
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure h is properly normalized
    h = h * (n_points / 2) / np.sum(h)
    
    def overlap_computation(h, k):
        # Compute overlap for shift k using FFT
        # We want integral of h(x) * (1 - h(x + k)) dx
        # Using convolution theorem: integrate h * (1 - h_shifted)
        # This is equivalent to computing correlation of h with (1-h) at shift k
        h1 = h.copy()
        h2 = 1 - h.copy()
        
        # Pad arrays to avoid circular convolution effects
        pad = 2 * n_points - 1
        h1_padded = np.pad(h1, (0, pad - len(h1)), mode='constant', constant_values=0)
        h2_padded = np.pad(h2, (0, pad - len(h2)), mode='constant', constant_values=0)
        
        # FFT-based convolution
        H1 = fft(h1_padded)
        H2 = fft(h2_padded)
        correlation = ifft(H1 * np.conj(H2)).real
        return correlation[k + n_points - 1] * dx
    
    def compute_all_overlaps(h):
        overlaps = []
        for k in range(-(n_points-1), n_points):
            if k == 0: continue
            overlap = overlap_computation(h, k)
            overlaps.append((overlap, k))
        return overlaps
    
    def update_priority_queue(overlaps):
        # Create max heap (negate values to use min heap)
        heap = []
        for overlap, k in overlaps:
            heappush(heap, (-overlap, k))
        return heap
    
    def apply_adjustment(h, k, w=5):
        # Local adjustment to reduce overlap in shift k
        # Create a temporary copy to modify
        h_new = h.copy()
        # Compute the regions where overlap occurs
        if k > 0:  # h(x) overlaps with (1 - h(x+k))
            # Adjust on interval [0, n-k] where h_i * (1 - h_{i+k}) contributes
            for i in range(max(0, k-w), min(n_points-k, k+w)):
                if i >= 0 and i+k < n_points:
                    # Prefer to decrease values with large h_i * (1 - h_{i+k}) terms
                    # Simple gradient ascent towards reducing overlap in term i
                    h_new[i] = max(0, h_new[i] - 0.0001)
                    h_new[i+k] = min(1, h_new[i+k] + 0.0001)
        elif k < 0:  # h(x) overlaps with (1 - h(x+k)) where k is negative
            # h_i * (1 - h_{i-k}) contributes
            abs_k = -k
            for i in range(max(0, abs_k-w), min(n_points, abs_k+w)):
                if i >= 0 and i-abs_k < n_points:
                    h_new[i] = max(0, h_new[i] - 0.0001)
                    h_new[i-abs_k] = min(1, h_new[i-abs_k] + 0.0001)
        return h_new
    
    def random_perturb(h, w=5, scale=0.02):
        h_new = h.copy()
        # Randomly perturb values around the shifts most impacting overlap
        idx = np.random.choice(n_points, size=min(w*2, n_points), replace=False)
        for i in idx:
            perturbation = np.random.normal(0, scale)
            h_new[i] = np.clip(h_new[i] + perturbation, 0, 1)
        # Maintain constraint: sum(h_new) = n/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Initialize using current configuration
    overlaps = compute_all_overlaps(h)
    pq = update_priority_queue(overlaps)
    
    num_iter = 0
    no_improvement_count = 0
    min_improvement = 1e-6
    w = 5
    scale = 0.02
    
    while time.time() - start_time < budget_s - 1:
        if num_iter >= 10000:  # Safety break
            break
        num_iter += 1
        
        # Sample a few high overlap shifts to consider
        indices = [np.random.randint(0, len(pq)) for _ in range(min(20, len(pq)))]
        candidates = []
        temp_pq = pq[:]
        for _ in range(10):
            if temp_pq:
                val, shift = heappop(temp_pq)
                candidates.append((val, shift))
        
        # Select best one from candidates
        if not candidates:
            break
            
        _, k = max(candidates, key=lambda x: x[0])
        k = -k  # Due to negation in the heap
        
        # Apply localized adjustment
        h_temp = apply_adjustment(h, k, w)
        
        # Add random perturbations to escape local minima
        h_temp = random_perturb(h_temp, w, scale)
        
        # Compute new overlap for the chosen shift
        new_overlap = overlap_computation(h_temp, k)
        
        # Check if we did improve
        improvement = False
        # Need to be careful here - we only update if we ended up with a better {shift, k}
        # But we are optimizing objective max of overlaps
        new_overlaps = compute_all_overlaps(h_temp)
        max_new_overlap = max(o for o, _ in new_overlaps) if new_overlaps else 0
        
        # Main focus is on global overlap value (which is max of overlaps over all shifts)
        curr_c5 = max(o for o, _ in overlaps) if overlaps else float('inf')
        new_c5 = max_new_overlap
        
        if new_c5 < curr_c5 - min_improvement:
            improvement = True
            h = h_temp.copy()
            overlaps = new_overlaps
            pq = update_priority_queue(overlaps)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive changes
        if no_improvement_count > 10:
            w = min(w + 1, 15)
            scale *= 2
            no_improvement_count = 0
            
        if no_improvement_count > 50:
            # Start over with random exploration
            h[...] = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            
    # Final validation
    final_overlaps = compute_all_overlaps(best_h)
    final_c5 = max(o for o, _ in final_overlaps) if final_overlaps else 0
    
    return (best_h.tolist(), final_c5, n_points)
