import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Generate diverse initial candidates
    candidates = [h_init.copy()]
    for i in range(19):
        h_pert = h_init.copy()
        noise = np.random.normal(0, 0.01, n_points)
        h_pert += noise
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        candidates.append(h_pert)
    
    for i, h in enumerate(candidates):
        h = h.copy()
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute correlation
            corr_full = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr_full) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Compute gradient numerically
            eps = 1e-4
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - c5) / eps
            
            # Update step
            lr = 0.005 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new

            # Add diversity occasionally
            if iter_count % 500 == 0 and time.time() - start_time < budget_s - 1:
                noise = np.random.normal(0, 0.002, n_points)
                h = np.clip(h + noise, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
    
    # Final check
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
