    # This gives a reasonable starting point

    A = np.zeros((n_points, n_points))
    for i in range(n_points):
        for j in range(n_points):
            k = (j - i) % n_points
            if k <= n_points // 2:
                A[i, j] = 1
    
    # The problem is complex to set up directly in CVXPY
    # Instead, we'll explore a different approach directly

    # For robust convergence, we'll use a multi-start strategy
    max_no_improvement = 50
    no_improvement_count = 0

    for start_iter in range(3):

            # Enforce constraints

    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
