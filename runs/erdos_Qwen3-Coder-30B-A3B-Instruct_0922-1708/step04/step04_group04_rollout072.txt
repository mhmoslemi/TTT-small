import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use CVXPY to solve the convex relaxation
    # Variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2, C >= 0]
    
    # For all shifts
    for k in range(1, n_points):
        # Correlation contribution
        corr = 0
        for i in range(n_points):
            if i + k < n_points:
                corr += h_var[i] * (1 - h_var[i + k])
        constraints.append(corr <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with ECOS
    prob.solve(solver=cp.ECOS, max_iters=100, feastol=1e-5)
    
    # Extract solution
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        h_opt = h_var.value
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        # Evaluate the actual C5
        correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
        c5_val = np.max(correlation) * dx
        
        if c5_val < best_c5:
            best_h = h_opt
            best_c5 = c5_val
    else:
        # Fallback to gradient descent refinement
        h = best_h.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Gradient estimation
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update
            lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-4)
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Update best
            correlation = np.correlate(h, 1 - h, mode='full')
            c5_new = np.max(correlation) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
