import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Get all valid k values for correlation
    max_k = n_points - 1
    
    # Iterative convex optimization with line search
    max_iter = 100
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute current correlation to get reference value
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(h_var) == n_points / 2]
        constraints += [0 <= h_var, h_var <= 1]

            # Simplify to: h[i] + h_old[i+k] - h_old[i+k]*h[i] - h[i]*h_old[i+k] = h[i] + h_old[i+k] - h_old[i+k]*h[i] - h[i]*h_old[i+k]

            # ≈ h[i] - h_old[i+k]*h[i] - h[i]*h_old[i+k] + h_old[i+k]^2
            # But that is wrong. Let's linearize around h_old:
            # L(h[i], h[i+k]) = h_old[i] * (1 - h_old[i+k]) + (h[i] - h_old[i]) * (1 - h_old[i+k]) + (h[i+k] - h_old[i+k]) * (-h_old[i])
            # = h_old[i] - h_old[i]*h_old[i+k] + (h[i] - h_old[i])(1 - h_old[i+k]) - h_old[i](h[i+k] - h_old[i+k])
            # = h_old[i] - h_old[i]*h_old[i+k] + h[i] - h_old[i] - h[i]*h_old[i+k] + h_old[i]*h_old[i+k] - h_old[i]*h[i+k] + h_old[i]*h_old[i+k]
            # Simplify:
            # = h[i] - h_old[i] - h[i]*h_old[i+k] + h_old[i]*h_old[i+k] - h_old[i]*h[i+k] + h_old[i]*h_old[i+k]
            # = h[i] - h_old[i] + h_old[i]*h_old[i+k] - h[i]*h_old[i+k] - h_old[i]*h[i+k] + h_old[i]*h_old[i+k]
            # = h[i] - h_old[i] + h_old[i]*h_old[i+k] - h_old[i]*h[i+k] - h[i]*h_old[i+k] + h_old[i]*h_old[i+k]
            # = h[i] - h_old[i] + h_old[i]*(h_old[i+k] - h[i+k] - h_old[i+k] + h_old[i+k])

            # h[i] - h_old[i]*h_old[i+k] - h[i]*h_old[i+k] + h_old[i]*h_old[i+k] - h_old[i]*h[i+k] + h_old[i]*h_old[i+k] <= C * dx
            # h[i] - h[i]*h_old[i+k] - h_old[i]*h[i+k] + h_old[i]*h_old[i+k] <= C * dx
            # h[i] - h[i]*h_old[i+k] - h_old[i]*h[i+k] + h_old[i]*h_old[i+k] <= C * dx
            # Using h_old[i]*h_old[i+k] = h_old[i]*h_old[i+k]:
            # Let’s take another approach: use h_old[i]*h_old[i+k] for the cross-product part
            # h[i] - h_old[i]*h_old[i+k] - h_old[i] * h[i+k] + h_old[i]*h_old[i+k] - h[i]*h_old[i+k] + h_old[i]*h_old[i+k]
            # = h[i] - h[i]*h_old[i+k] - h_old[i]*h[i+k]
            # Thus: h[i] - h[i]*h_old[i+k] - h_old[i]*h[i+k] + h_old[i]*h_old[i+k] <= C*dx
            # == h[i] + h_old[i]*h_old[i+k] - h[i]*h_old[i+k] - h_old[i]*h[i+k] <= C*dx

            # h[i]*(1-h[i+k]) - (h_old[i] - h_old[i]*h_old[i+k]) = h[i]*1 - h[i]*h[i+k] - h_old[i] + h_old[i]*h_old[i+k] = h[i] - h[i]*h[i+k] - h_old[i] + h_old[i]*h_old[i+k]
            #设用 h[i] 拟合前项，以当前 h 为基准线
