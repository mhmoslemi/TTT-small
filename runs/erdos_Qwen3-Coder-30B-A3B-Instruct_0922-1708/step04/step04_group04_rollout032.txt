import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100
    tol = 1e-5
    alpha = 0.5
    line_search_ratio = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 value
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # CVXPY approach
        
        # Variables
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(h_var) == n_points / 2]
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Constraint using linearized terms
        # For each k from 1 to n_points-1
        # sum_{i=0}^{n_points-k-1} h_var[i] * (1 - h_var[i+k]) <= C
        # Linearize h_var[i+k] around h[i+k]
        for k in range(1, n_points):
            # We approximate h_var[i+k] ~ h[i+k] (current values)
            # So the constraint becomes: sum h_var[i] * (1 - h[i+k]) <= C
            # But for each fixed i, we know h[i+k] is constant
            # We aggregate terms in the full correlate
            # We can do this by precomputing valid pairs
            idx = k
            constraint_expr = 0
            
            # Doing a loop over all valid indices where shift k is possible
            for i in range(n_points - k):
                constraint_expr += h_var[i] * (1 - h[i+k])
            constraints.append(constraint_expr <= C)
            
        objective = cp.Minimize(C)
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, abstol=1e-4, reltol=1e-4)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                # Line search steps
                step_size = alpha
                converged = False
                for _ in range(20):  # Limit line search steps
                    if time.time() - start_time > budget_s - 1:
                        break
                    trial_h = h + step_size * (new_h - h)
                    trial_h = np.clip(trial_h, 0, 1)
                    trial_h = trial_h * (n_points / 2) / np.sum(trial_h)
                    trial_correlation = np.correlate(trial_h, 1 - trial_h, mode='full')
                    trial_c5 = np.max(trial_correlation) * dx
                    
                    if trial_c5 < current_c5:
                        h = trial_h
                        converged = True
                        break
                    else:
                        step_size *= line_search_ratio
                        
                if not converged:
                    h = np.clip(new_h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    
            else:
                # Fallback if not optimal, maybe just randomly perturb slightly
                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            # Fallback if CVXPY fails, just use gradient step
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # One more validation check
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    # Final validation
    if time.time() - start_time > budget_s - 1:
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(correlation) * dx

    return (best_h.tolist(), best_c5, n_points)
