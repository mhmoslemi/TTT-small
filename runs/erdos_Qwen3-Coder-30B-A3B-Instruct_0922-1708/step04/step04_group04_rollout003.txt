import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Helper function to evaluate correlation
    def evaluate_c5(h_values):
        correlation = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(correlation) * dx
    
    # Start with the initial solution
    current_c5 = evaluate_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Try to improve with a combination of local search and convex relaxation
    max_evaluations = 500
    eval_count = 0
    
    for iteration in range(max_evaluations):
        if time.time() - start_time > budget_s - 1:
            break
            
        eval_count += 1
        # Random perturbation
        h_perturbed = h.copy()
        perturb_idx = np.random.randint(0, n_points)
        h_perturbed[perturb_idx] += np.random.normal(0, 0.02)
        
        # Ensure feasibility
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        c5_perturbed = evaluate_c5(h_perturbed)
        
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
            h = h_perturbed.copy()
        elif c5_perturbed < current_c5:
            # Accept with some probability (simulated annealing)
            if np.random.random() < 0.1:
                h = h_perturbed.copy()
                current_c5 = c5_perturbed
    
    # Final convex optimization approach
    # Make the shifted correlation constraints using CVXPY
    try:
        # Define optimization variables
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            C >= 0
        ]
        
        # For each shift k (from 1 to n_points-1)
        for k in range(1, n_points):
            # Compute overlap term for shift k
            # This is sum_{i=0}^{n_points-1-k} h_var[i] * (1 - h_var[i+k])
            # For indices 0 to n_points-1-k, but in CVXPY we define directly
            # For all i, i+k < n_points: h_var[i] * (1 - h_var[i+k])
            if k < n_points:
                # Sum over valid indices
                inner_sum = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
                constraints.append(inner_sum <= C)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        # Set solver parameters appropriately
        prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-6, reltol=1e-6)
        
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_optimal = np.array(h_var.value).flatten()
            h_optimal = np.clip(h_optimal, 0, 1)
            # Rescale to correct sum
            h_optimal = h_optimal * (n_points / 2) / np.sum(h_optimal)
            c5_optimal = evaluate_c5(h_optimal)
            if c5_optimal < best_c5:
                best_c5 = c5_optimal
                best_h = h_optimal.copy()
                
    except Exception as e:
        pass  # Continue with best found so far
    
    return (best_h.tolist(), best_c5, n_points)
