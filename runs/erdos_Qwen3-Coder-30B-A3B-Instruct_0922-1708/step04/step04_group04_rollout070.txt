import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use CVXPY convex optimization
    while time.time() - start_time < budget_s - 1:
        # Reset h to current best
        h = best_h.copy()
        
        # Formulate as convex optimization problem
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
        ]
        
        # Add overlap constraints for all shifts
        for k in range(1, n_points):
            # For each shift k, we want to constrain the overlap
            # sum(h[i] * (1 - h[i+k])) <= C
            # This is equivalent to sum(h[i] - h[i] * h[i+k]) <= C
            # Which is sum(h[i]) - sum(h[i] * h[i+k]) <= C
            # => sum(h[i] * h[i+k]) >= sum(h[i]) - C
            # But we want the maximum overlap integral, which is equivalent to
            # sum(h[i] * (1 - h[i+k])) <= C 
            # This is a quadratic constraint. This dual formulation: 
            # max_{k} sum h[i] * (1 - h[i+k]) <= C  
            # can be made as: 
            # max_k sum h[i] * (1 - h[i+k]) <= C
            # Using a more direct computation approach for the functional,
            # we calculate the full correlation manually.
            
            # Actually, we can model it as: for each shift k:
            #  sum_{i=0}^{n-k-1} h_i * (1 - h_{i+k}) <= C
            # We'll compute explicit constraints for known valid indices.
            expr = 0
            for i in range(n_points - k):
                expr += h_var[i] * (1 - h_var[i+k])
            constraints.append(expr <= C) 
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, feastol=1e-5)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol) 
                
                # compute actual C5
                correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                c5 = np.max(correlation) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_sol.copy()
        except Exception:
            pass  # Fall back to simpler approach if optimization fails
            break
            
        if time.time() - start_time > budget_s - 1:
            break
            
    # Final refinement for best solution
    h = best_h.copy()
    for _ in range(1000):  # Simple local refinement
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Finite difference gradient estimation
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-4)
        h_new = h - lr * grad
        
        # Projection
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Check if better
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
                
        if np.abs(c5 - c5_new) < 1e-10:
            break
            
    # Final check
    h = best_h.copy()
    correlation = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (h.tolist(), final_c5, n_points)
