    else:
        # Start with a block function that approaches 1 in the first half
        h = np.zeros(n_points)

    h = h * (n_points / 2) / np.sum(h)

        # Compute correlation using convolution
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        # We only care about the non-negative lags (as those represent the overlap)
        # But we want the maximum value among all lags
        return np.max(correlation) * dx

    # Try to make small modifications to existing solutions
    iteration = 0
