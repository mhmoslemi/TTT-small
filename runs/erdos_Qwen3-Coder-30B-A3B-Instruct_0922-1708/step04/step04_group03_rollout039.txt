import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Convex relaxation
    h_relaxed = cp.Variable(n_points)
    objective = cp.Minimize(cp.sum(h_relaxed * (1 - h_relaxed)))
    
    constraints = [
        h_relaxed >= 0,
        h_relaxed <= 1,
        cp.sum(h_relaxed) == n_points / 2
    ]
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CVXOPT, verbose=False)
    
    if prob.status == cp.OPTIMAL:
        h = h_relaxed.value.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Helper to evaluate C5
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Gradient descent with finite differences
    def gradient_descent_step(h, lr=0.01):
        c5 = evaluate_c5(h)
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps
        
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Optimization loop
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Improve the current solution
        h = gradient_descent_step(h, lr=0.01)
        
        # Evaluate current solution
        c5 = evaluate_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early stopping if no improvement in last 50 iterations
        if iteration - last_improvement >= 50:
            break
            
    return (best_h.tolist(), best_c5, n_points)
