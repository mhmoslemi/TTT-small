import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
        
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Optimize with gradient-free methods
    def objective(x):
        # x contains h values
        h_vals = np.clip(x, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)  # Rescale to match constraint
        return compute_c5(h_vals)
    
    # Use a hybrid approach
    method = 'Nelder-Mead'  # Simple and robust for this problem
    max_iter = 5000
    iter_count = 0
    
    # Use Levenberg-Marquardt style local search
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Random perturbation for exploration
        h_test = h.copy()
        idx = np.random.randint(0, n_points)
        perturb = np.random.normal(0, 0.01)
        h_test[idx] += perturb
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        test_c5 = compute_c5(h_test)
        if test_c5 < current_c5:
            h = h_test.copy()

    # Final local optimization
    try:
        # Perform trust-region optimization
        result = minimize(objective, h, method='trust-constr', 
                          options={'maxiter': 1000, 'disp': False})
        if result.success:
            final_h = np.clip(result.x, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
