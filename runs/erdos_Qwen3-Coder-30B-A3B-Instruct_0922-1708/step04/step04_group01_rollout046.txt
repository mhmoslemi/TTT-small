import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Coarse search
    coarse_grid_size = 20
    coarse_candidates = []
    
    for _ in range(50):
        if time.time() - start_time > budget_s - 200:
            break
        # Generate random coarse candidate
        h_c = np.random.rand(coarse_grid_size)
        h_c = h_c / np.sum(h_c) * (coarse_grid_size / 2)
        
        # Up sample to fine grid
        h_up = np.interp(np.linspace(0, 2, n_points), 
                         np.linspace(0, 2, coarse_grid_size), h_c)
        
        # Ensure sum constraint
        h_up = h_up * (n_points / 2) / np.sum(h_up)
        
        # Evaluate
        corr = np.correlate(h_up, 1 - h_up, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_up.copy()
            
        coarse_candidates.append((h_up.copy(), c5))
    
    # Sort coarse candidates by C5 value
    coarse_candidates.sort(key=lambda x: x[1])
    top_candidates = coarse_candidates[:10] if len(coarse_candidates) >= 10 else coarse_candidates
    
    # Refine top candidates
    for h_c, c5_c in top_candidates:
        if time.time() - start_time > budget_s - 100:
            break
            
        # Use CVXPY for refinement
        try:
            # Create CVXPY variable
            h_vars = cp.Variable(n_points)
            
            # CVXPY constraints
            constraints = [
                h_vars >= 0,
                h_vars <= 1,
                cp.sum(h_vars) == n_points / 2
            ]
            
            # Objective: maximize C5 by minimizing -C5
            # We'll do this by working with the correlation threshold
            
            # Simple gradient step refinement
            h_refine = h_c.copy()
            lr = 0.01
            for _ in range(500):
                if time.time() - start_time > budget_s - 50:
                    break
                    
                # Compute correlation 
                corr = np.correlate(h_refine, 1 - h_refine, mode='full')
                current_c5 = np.max(corr) * dx
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_refine.copy()
                
                # Compute gradient by finite differences
                grad = np.zeros_like(h_refine)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_refine.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[i] = (c5_plus - current_c5) / eps
                
                # Update
                h_refine = h_refine - lr * grad
                
                # Re-project
                h_refine = np.clip(h_refine, 0, 1)
                h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
                
        except:
            continue
            
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
