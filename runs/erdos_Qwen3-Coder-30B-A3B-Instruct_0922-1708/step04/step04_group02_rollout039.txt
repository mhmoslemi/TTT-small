import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 100
    inertia_weight = 0.7
    phi_p = 2.0
    phi_g = 2.0
    local_search_freq = 50
    rejuvenation_rate = 0.1
    max_iterations = int(budget_s / 0.5)  # Rough estimate
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Interpolate initial solution to our resolution
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h_interpolated = np.interp(x_target, x_orig, h_orig)
        h_interpolated = h_interpolated * (n_points / 2) / np.sum(h_interpolated)
        swarm.append(h_interpolated)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        # Generate random feasible particles
        swarm.append(np.ones(n_points) * 0.5)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Fill the rest
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Personal best and global best
    pbest = [h.copy() for h in swarm]
    pbest_fitness = [float('inf')] * swarm_size
    
    # Initialize global best
    gbest_idx = 0
    gbest = swarm[gbest_idx].copy()
    gbest_fitness = float('inf')
    
    start_time = time.time()
    
    # Main PSO loop
    iteration = 0
    
    while time.time() - start_time < budget_s - 10 and iteration < max_iterations:
        # Evaluate fitness for all particles
        for i in range(swarm_size):
            h = swarm[i]
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_fitness[i]:
                pbest_fitness[i] = c5
                pbest[i] = h.copy()
                
            if c5 < gbest_fitness:
                gbest_fitness = c5
                gbest = h.copy()
                gbest_idx = i
                
        # Update velocities and positions
        for i in range(swarm_size):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (inertia_weight * v + 
                 phi_p * r1 * (pbest[i] - h) + 
                 phi_g * r2 * (gbest - h))
            
            # Update position
            h_new = h + v
            
            # Repair solution
            h_new = np.clip(h_new, 0, 1)
            h_sum = np.sum(h_new)
            
            if abs(h_sum - n_points / 2) > 1e-8:
                # Adjust to meet sum constraint
                diff = h_sum - n_points / 2
                if diff > 0:
                    # Reduce largest elements
                    indices = np.argsort(h_new)[::-1][:int(diff/dx)]
                    h_new[indices] -= dx
                    h_new = np.clip(h_new, 0, 1)
                else:
                    # Increase smallest elements
                    indices = np.argsort(h_new)[:int(-diff/dx)]
                    h_new[indices] += dx
                    h_new = np.clip(h_new, 0, 1)
                
                # Final adjustment
                h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            swarm[i] = h_new
            velocities[i] = v
        
        # Local search and rejuvenation
        if iteration % local_search_freq == 0:
            # Perform local search on top 10% particles
            top_indices = np.argsort([pbest_fitness[i] for i in range(swarm_size)])[:max(1, int(swarm_size * 0.1))]
            
            for idx in top_indices:
                h = pbest[idx].copy()
                # Gradient descent with small steps
                for _ in range(5):
                    # Compute gradient approximation
                    grad = np.zeros(n_points)
                    eps = 1e-5
                    
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        h_minus = h.copy()
                        h_minus[j] -= eps
                        h_minus = np.clip(h_minus, 0, 1)
                        h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                        
                        corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                        c5_minus = np.max(corr_minus) * dx
                        
                        grad[j] = (c5_plus - c5_minus) / (2 * eps)
                    
                    h = h - 0.001 * grad
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                
                # Check if better
                corr_h = np.correlate(h, 1 - h, mode='full')
                c5_h = np.max(corr_h) * dx
                if c5_h < pbest_fitness[idx]:
                    pbest_fitness[idx] = c5_h
                    pbest[idx] = h.copy()
                    if c5_h < gbest_fitness:
                        gbest_fitness = c5_h
                        gbest = h.copy()
        
        # Rejuvenation
        if iteration % (local_search_freq * 2) == 0:
            worst_indices = np.argsort([pbest_fitness[i] for i in range(swarm_size)])[-int(swarm_size * rejuvenation_rate):]
            for idx in worst_indices:
                # Replace with random feasible individual
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                swarm[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                pbest[idx] = h.copy()
                pbest_fitness[idx] = float('inf')
        
        iteration += 1
    
    # Final refinement with gradient-based method on best candidates
    candidates = [pbest[i] for i in range(swarm_size) if pbest_fitness[i] < gbest_fitness + 0.001]
    candidates = candidates[:5]  # Top 5
    
    final_best_h = gbest.copy()
    final_best_c5 = gbest_fitness
    
    # Intensive gradient descent
    for h in candidates:
        h_refined = h.copy()
        
        for _ in range(100):
            grad = np.zeros(n_points)
            eps = 1e-6
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h_refined.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            h_refined = h_refined - 0.0005 * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            # Check improvement
            corr_refined = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5_refined = np.max(corr_refined) * dx
            
            if c5_refined < final_best_c5:
                final_best_c5 = c5_refined
                final_best_h = h_refined.copy()
    
    return (final_best_h.tolist(), final_best_c5, n_points)
