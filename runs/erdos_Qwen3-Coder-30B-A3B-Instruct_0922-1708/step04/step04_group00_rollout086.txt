        # But start with gradient-based approach first for efficiency

        best_h = h.copy()
        best_c5 = float('inf')

        corr = np.correlate(h, 1 - h, mode='full')
        initial_c5 = np.max(corr) * dx

        def compute_c5_and_gradient(h, eps=1e-4):
            # Use finite differences for gradient
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx

                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)

                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps

    final_c5 = np.max(final_corr) * dx
    
    return (final_h.tolist(), float(final_c5), n_points)
