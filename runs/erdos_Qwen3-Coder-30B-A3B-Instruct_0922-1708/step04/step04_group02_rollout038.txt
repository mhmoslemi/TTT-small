import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 100
    inertia = 0.7
    phi_p = 2.0
    phi_g = 2.0
    local_search_freq = 50
    rejuvenate_freq = 200
    max_iter = int(budget_s * 1.5)  # more iterations than time to ensure we don't miss anything
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Load or create initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5  # uniform start
    
    # Create initial swarm around central solution
    for _ in range(swarm_size):
        particle = h_init + np.random.normal(0, 0.05, n_points)
        # Repair
        particle = np.clip(particle, 0, 1)
        sum_h = np.sum(particle)
        if sum_h > 0:
            particle = particle * (n_points / 2) / sum_h
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(particle.copy())
    
    # Evaluate initial
    gbest_idx = 0
    gbest_c5 = float('inf')
    for i, particle in enumerate(particles):
        corr = np.correlate(particle, 1 - particle, mode='full')
        c5 = np.max(corr) * dx
        pbest_values.append(c5)
        if c5 < gbest_c5:
            gbest_c5 = c5
            gbest_idx = i
    
    # Main PSO loop
    start_time = time.time()
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 5:
        # Optimization loop for each particle
        for i in range(swarm_size):
            particle = particles[i]
            v = velocities[i]
            
            # Compute overlap
            corr = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < pbest_values[i]:
                pbest[i] = particle.copy()
                pbest_values[i] = c5
                
            # Update global best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_idx = i
                last_improvement = iter_count
        
        # Update swarm
        for i in range(swarm_size):
            particle = particles[i]
            v = velocities[i]
            pbest_i = pbest[i]
            gbest = particles[gbest_idx]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (inertia * v + 
                 phi_p * r1 * (pbest_i - particle) + 
                 phi_g * r2 * (gbest - particle))
            
            # Update position
            particle = particle + v
            
            # Repair to feasible region
            particle = np.clip(particle, 0, 1)
            sum_h = np.sum(particle)
            if sum_h > 0:
                particle = particle * (n_points / 2) / sum_h
            
            particles[i] = particle
            velocities[i] = v
            
        iter_count += 1
        
        # Local search periodically
        if iter_count % local_search_freq == 0:
            for i in range(int(swarm_size * 0.1)):  # top 10%
                idx = np.argsort(pbest_values)[i]
                particle = pbest[idx]
                # Gradient-based refinement
                for _ in range(10):
                    corr = np.correlate(particle, 1 - particle, mode='full')
                    c5 = np.max(corr) * dx
                    # Compute gradient
                    grad = np.zeros_like(particle)
                    eps = 1e-5
                    for j in range(len(particle)):
                        particle_plus = particle.copy()
                        particle_plus[j] += eps
                        particle_plus = np.clip(particle_plus, 0, 1)
                        sum_plus = np.sum(particle_plus)
                        if sum_plus > 0:
                            particle_plus = particle_plus * (n_points / 2) / sum_plus
                        corr_plus = np.correlate(particle_plus, 1 - particle_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - c5) / eps
                    # Gradient step
                    particle = particle - 0.001 * grad
                    particle = np.clip(particle, 0, 1)
                    sum_h = np.sum(particle)
                    if sum_h > 0:
                        particle = particle * (n_points / 2) / sum_h
                # Update best
                corr = np.correlate(particle, 1 - particle, mode='full')
                c5 = np.max(corr) * dx
                if c5 < pbest_values[idx]:
                    pbest[idx] = particle
                    pbest_values[idx] = c5
                    if c5 < gbest_c5:
                        gbest_c5 = c5
                        gbest_idx = idx
        
        # Rejuvenate worst particles periodically
        if iter_count % rejuvenate_freq == 0:
            worst_indices = np.argsort(pbest_values)[-int(swarm_size * 0.1):]
            for idx in worst_indices:
                particle = np.random.rand(n_points)
                particle = np.clip(particle, 0, 1)
                sum_h = np.sum(particle)
                if sum_h > 0:
                    particle = particle * (n_points / 2) / sum_h
                particles[idx] = particle
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                pbest[idx] = particle.copy()
                pbest_values[idx] = float('inf')
    
    # Final refinement of best solution using gradient method
    best_particle = particles[gbest_idx].copy()
    best_corr = np.correlate(best_particle, 1 - best_particle, mode='full')
    best_c5 = np.max(best_corr) * dx
    
    # Fine-tune the best solution
    for _ in range(100):
        corr = np.correlate(best_particle, 1 - best_particle, mode='full')
        c5 = np.max(corr) * dx
        
        # Compute gradient with finite differences
        grad = np.zeros_like(best_particle)
        eps = 1e-5
        for i in range(len(best_particle)):
            particle_plus = best_particle.copy()
            particle_plus[i] += eps
            particle_plus = np.clip(particle_plus, 0, 1)
            sum_plus = np.sum(particle_plus)
            if sum_plus > 0:
                particle_plus = particle_plus * (n_points / 2) / sum_plus
            corr_plus = np.correlate(particle_plus, 1 - particle_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Step
        best_particle = best_particle - 0.001 * grad
        best_particle = np.clip(best_particle, 0, 1)
        sum_h = np.sum(best_particle)
        if sum_h > 0:
            best_particle = best_particle * (n_points / 2) / sum_h
    
    # Final validation
    final_corr = np.correlate(best_particle, 1 - best_particle, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_particle.tolist(), final_c5, n_points)
