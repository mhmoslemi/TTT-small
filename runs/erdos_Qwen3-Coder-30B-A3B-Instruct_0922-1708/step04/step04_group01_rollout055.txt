import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Try to use initial_h_values if available, otherwise create something reasonable
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function that's slightly refined
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def compute_c5(h_vector):
        correlation = np.correlate(h_vector, 1 - h_vector, mode='full')
        return np.max(correlation) * dx
    
    # Phase 1: Grid search on a coarse resolution
    coarse_n = 20
    coarse_dx = 2.0 / coarse_n
    best_coarse = None
    best_coarse_c5 = float('inf')
    
    if time.time() - start_time < budget_s - 100:
        # Simple grid search on smaller grid
        num_samples = 1000
        for _ in range(num_samples):
            if time.time() - start_time > budget_s - 50:
                break
            h_sample = np.random.rand(coarse_n)
            h_sample = h_sample / np.sum(h_sample) * (coarse_n / 2)
            
            c5 = compute_c5(h_sample)
            if c5 < best_coarse_c5:
                best_coarse_c5 = c5
                best_coarse = h_sample.copy()
    
    # If we got a good coarse solution, refine it
    if best_coarse is not None:
        # Upsample and refine
        h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, coarse_n), best_coarse)
        h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
        h = h_upsampled
    
    # Phase 2: Refinement with gradient-based approach
    def loss_func(h_vector):
        correlation = np.correlate(h_vector, 1 - h_vector, mode='full')
        return -np.max(correlation) * dx  # Negative because we minimize
    
    def constraint_sum(h_vector):
        return np.sum(h_vector) - n_points / 2
    
    # Refinement using scipy.optimize
    constr = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    refinement_steps = 0
    max_refinement_steps = 500
    
    while time.time() - start_time < budget_s - 100 and refinement_steps < max_refinement_steps:
        # Apply minimization step
        result = minimize(loss_func, h, method='SLSQP', bounds=bounds, constraints=constr, 
                          options={'maxiter': 10, 'ftol': 1e-6})
        
        if result.success:
            new_h = result.x
            # Project back to valid range
            new_h = np.clip(new_h, 0, 1)
            # Rescale to meet integral constraint
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            h = new_h
            
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
        refinement_steps += 1
    
    # Final refinement with CVXPY if possible
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL and h_vars.value is not None:
            h_cvx = h_vars.value
            h_cvx = np.clip(h_cvx, 0, 1)
            h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
            
            c5_cvx = compute_c5(h_cvx)
            if c5_cvx < best_c5:
                best_h = h_cvx
                best_c5 = c5_cvx
    except:
        pass
    
    # Final evaluation
    c5_final = compute_c5(best_h)
    
    return (best_h.tolist(), c5_final, n_points)
