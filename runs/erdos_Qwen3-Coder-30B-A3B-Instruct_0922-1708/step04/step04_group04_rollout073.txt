import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Start timing
    start_time = time.time()
    
    # Precompute shift indices for correlation
    max_shift = n_points - 1
    shifts = list(range(-max_shift, max_shift + 1))
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    max_iters = 500
    for iter_num in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Formulate and solve convex relaxation
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1]
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Add overlap constraints: sum(h_i * (1 - h_{i+k})) <= C for k != 0
        for k in shifts:
            if k == 0:
                continue
            # Indices for h_i and h_{i+k}
            if k > 0:
                # h[i] * (1 - h[i+k]) for i = 0..(n_points-k-1)
                expr = 0
                for i in range(n_points - k):
                    expr += h_var[i] * (1 - h_var[i+k])
            else:
                # h[i] * (1 - h[i+k]) for i = -k..(n_points-1)
                expr = 0
                for i in range(-k, n_points):
                    expr += h_var[i] * (1 - h_var[i+k])
                    
            constraints.append(expr <= C)
            
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100, feastol=1e-5)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                c5_sol = C.value
                if c5_sol < best_c5:
                    best_c5 = c5_sol
                    best_h = h_sol.copy()
            else:
                # Use current h as fallback if optimization fails
                evaluate_c5 = max(np.correlate(h, 1-h, mode='full')) * dx
                if evaluate_c5 < best_c5:
                    best_c5 = evaluate_c5
                    best_h = h.copy()
        except:
            # Fallback evaluation of current solution
            evaluate_c5 = max(np.correlate(h, 1-h, mode='full')) * dx
            if evaluate_c5 < best_c5:
                best_c5 = evaluate_c5
                best_h = h.copy()

        # Update h with the solution
        h = best_h.copy()
        
        # Apply small random perturbations to maintain exploration
        if iter_num % 10 == 0:
            pert = np.random.normal(0, 0.01, n_points)
            h += pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final refinement and validation
    h_final = best_h.copy()
    c5_final = max(np.correlate(h_final, 1-h_final, mode='full')) * dx
    
    return (h_final.tolist(), c5_final, n_points)
