import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective_and_gradient(h):
        # Ensure h is within bounds and normalized
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Gradient computation using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
            
        return c5, grad
    
    # Initial computation
    c5, _ = objective_and_gradient(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Trust region optimization
    max_iter = 500
    trust_radius = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize with trust region method
        # We work with the flattened version
        h_flat = h.copy()
        
        # Function to minimize
        def fun(h_flat):
            h = h_flat.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            return c5
        
        # Gradient function (approximated)
        def jac(h_flat):
            h = h_flat.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            eps = 1e-4
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - fun(h_flat)) / eps
            return grad
        
        # Optimization step
        try:
            res = minimize(fun, h_flat, method='TRF', jac=jac, bounds=[(0, 1)]*n_points, options={'maxiter': 50, 'initial_tr_radius': trust_radius})
            h = res.x.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        except:
            # fallback to simple gradient descent if TRF fails
            h = h_flat.copy()
            lr = 0.01
            for _ in range(20):
                _, grad = objective_and_gradient(h)
                h = h - lr * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                lr *= 0.99
        
        # Update best solution
        c5, _ = objective_and_gradient(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
