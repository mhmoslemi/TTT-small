import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective(h):
        # Compute correlation and return max value (scaled by dx)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define constraints and bounds
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Trust region reflective optimization
    max_iter = 500
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Optimize with trust region reflective method
        res = minimize(objective, h, method='TRF', bounds=bounds, constraints=cons, 
                       options={'maxiter': 10, 'initial_tr_radius': 0.1})
        
        if res.success:
            h = res.x
            # Project to ensure sum constraint and bounds
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate current solution
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # If optimization failed, try a different approach
            # Perturb slightly and recompute
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        iter_count += 1
    
    # Final refinement with gradient-based method if needed
    if time.time() - start_time < budget_s - 1:
        # Try local search
        def obj_func(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlation = np.correlate(h, 1 - h, mode='full')
            return np.max(correlation) * dx
        
        # Finite difference to estimate gradient on best solution
        eps = 1e-5
        std_grads = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            std_grads[i] = (c5_plus - obj_func(h)) / eps
        
        # Update using steepest descent
        lr = 0.001
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            h = h - lr * std_grads
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = obj_func(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            lr *= 0.995  # Decay learning rate
    
    return (best_h.tolist(), best_c5, n_points)
