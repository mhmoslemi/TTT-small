import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise use uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
        h = h_init.copy()
    else:
        h = np.ones(n_points) * 0.5
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # PSO parameters
    n_particles = 20
    max_iter = 200
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    inertia_decay = (w_start - w_end) / max_iter
    
    # Initialize PSO
    positions = np.array([h + np.random.normal(0, 0.05, n_points) for _ in range(n_particles)])
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    personal_best_positions = positions.copy()
    personal_best_scores = [float('inf')] * n_particles
    global_best_position = h.copy()
    global_best_score = float('inf')
    
    # PSO main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - inertia_decay * iter_num
        
        for i in range(n_particles):
            # Clip positions
            positions[i] = np.clip(positions[i], 0, 1)
            # Normalize to maintain sum constraint
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
            
            # Evaluate current position
            corr = np.correlate(positions[i], 1 - positions[i], mode='full')
            current_c5 = np.max(corr) * dx
            
            # Update personal best
            if current_c5 < personal_best_scores[i]:
                personal_best_scores[i] = current_c5
                personal_best_positions[i] = positions[i].copy()
            
            # Update global best
            if current_c5 < global_best_score:
                global_best_score = current_c5
                global_best_position = positions[i].copy()
                if global_best_score < best_c5:
                    best_c5 = global_best_score
                    best_h = global_best_position.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (personal_best_positions[i] - positions[i])
                           + c2 * r2 * (global_best_position - positions[i]))
            positions[i] += velocities[i]
        
        # Add periodic random noise for exploration
        if iter_num % 30 == 0:
            for i in range(n_particles):
                noise = np.random.normal(0, 0.01, n_points)
                positions[i] += noise
                positions[i] = np.clip(positions[i], 0, 1)
                positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
    
    # Refinement with local optimization
    top_candidates = []
    for i in range(n_particles):
        corr = np.correlate(personal_best_positions[i], 1 - personal_best_positions[i], mode='full')
        score = np.max(corr) * dx
        top_candidates.append((score, personal_best_positions[i]))
    top_candidates.sort()
    
    # Refine top candidates with scipy minimize
    for score, h_candidate in top_candidates[:5]:
        if time.time() - start_time > budget_s - 5:
            break
            
        def objective(h_vec):
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return -np.max(corr) * dx
            
        def constraint_h_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_h_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_candidate, bounds=bounds, constraints=cons, 
                          method='SLSQP', options={'maxiter': 200})
            if res.success:
                h_new = res.x.copy()
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                corr = np.correlate(h_new, 1 - h_new, mode='full')
                refined_c5 = np.max(corr) * dx
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = h_new.copy()
        except:
            pass
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
