import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation using CVXPY
    # This provides a good initialization for the non-convex problem
    h_init = h.copy()
    
    # Create the convex relaxation problem
    h_var = cp.Variable(n_points)
    # Minimize sum of all correlations (simplified)
    # We approximate by minimizing sum of squares of overlaps
    objective = cp.Minimize(cp.sum_squares(h_var @ (np.eye(n_points) - np.eye(n_points, k=1))))
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Solve convex problem
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        h = np.maximum(0, np.minimum(1, h_var.value))
        h = h * (n_points / 2) / np.sum(h)
    except:
        # Fallback to initial
        h = h_init.copy()
    
    # Track best solution
    try:
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    except:
        pass
        
    # Phase 2: Refine using projected gradient descent + local search
    def compute_overlap(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    max_iter = 5000
    lr = 0.01
    patience = 50
    no_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient by finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        current_c5 = compute_overlap(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Check improvement
        try:
            new_c5 = compute_overlap(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improvement = 0
                lr = max(1e-5, lr * 0.995)
            else:
                no_improvement += 1
                if no_improvement > patience:
                    break
        except:
            pass
            
    return (best_h.tolist(), best_c5, n_points)
