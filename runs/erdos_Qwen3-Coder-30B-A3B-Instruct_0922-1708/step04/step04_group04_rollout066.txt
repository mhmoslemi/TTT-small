import numpy as np
from scipy.stats import qmc
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use Latin Hypercube Sampling for better exploration
    sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs_samples = sampler.random(n=50)
    
    # Scale samples to [0,1] range for each dimension
    lhs_samples = qmc.scale(lhs_samples, 0, 1)
    
    # Adjust each sample to meet sum constraint and bound constraints
    for i in range(len(lhs_samples)):
        sample = lhs_samples[i]
        # Rescale to sum = n_points/2
        sample = sample * (n_points / 2) / np.sum(sample)
        # Clip to [0,1]
        sample = np.clip(sample, 0, 1)
        lhs_samples[i] = sample
    
    # Include the parent solution if available
    if 'initial_h_values' in globals():
        lhs_samples = np.vstack([lhs_samples, h])
    
    # Gradient descent refinement function
    def refine_solution(h_start, time_limit):
        h = h_start.copy()
        best_h_refined = h.copy()
        best_c5_refined = float('inf')
        lr = 0.01
        iter_count = 0
        
        while time.time() - start_time < time_limit - 1 and iter_count < 1000:
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5_refined:
                best_c5_refined = c5
                best_h_refined = h.copy()
                
            # Compute gradient via finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] = min(1.0, h_plus[i] + eps)
                # Rescale to meet sum constraint
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h.copy()
                h_minus[i] = max(0.0, h_minus[i] - eps)
                # Rescale to meet sum constraint
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[i] = (c5_plus - c5_minus) / (2*eps)
                
            # Update with adaptive learning rate
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            iter_count += 1
            
            # Decrease learning rate if no improvement
            if iter_count % 50 == 0:
                lr = max(1e-4, lr * 0.5)
                
            if iter_count > 10 and abs(c5 - np.max(np.correlate(h, 1-h, mode='full') * dx)) < 1e-6:
                break

        return best_h_refined, best_c5_refined
    
    # Evaluate all LHS samples and refine the best ones
    for i in range(len(lhs_samples)):
        if time.time() - start_time > budget_s - 1:
            break
            
        candidate = lhs_samples[i]
        
        # Refine the sample
        refined_h, refined_c5 = refine_solution(candidate, budget_s)
        
        # Update best solution
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h.copy()
        
    # Final refinement on the best solution
    if time.time() - start_time < budget_s - 1:
        final_h, final_c5 = refine_solution(best_h, budget_s)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
