import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Variation 1: Use binary representation in evolutionary approach
    def binary_h(h):
        """Convert h to binary representation"""
        return (h > 0.5).astype(float)
    
    def enforce_sum(h, target_sum):
        """Adjust h to enforce sum = target_sum, by flipping values"""
        current_sum = np.sum(h)
        diff = int(target_sum - current_sum)
        if abs(diff) < 1e-3:
            return h
        indices = np.arange(len(h))
        mask = h > 0.5
        if diff > 0:
            # Want to increase sum, flip zeros to ones
            zero_indices = indices[~mask]
            flip_indices = np.random.choice(zero_indices, diff, replace=False)
            h[flip_indices] = 1.0
        elif diff < 0:
            # Want to decrease sum, flip ones to zeros
            one_indices = indices[mask]
            flip_indices = np.random.choice(one_indices, -diff, replace=False)
            h[flip_indices] = 0.0
        return h

    # Evolutionary algorithm
    def evaluate_c5(h_values):
        h = np.array(h_values)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5

    population_size = 100
    generations = 100
    mutation_rate = 0.05
    crossover_rate = 0.5
    elite_size = 10
    
    # Initialize population
    population = []
    for i in range(population_size):
        if 'initial_h_values' in globals() and i == 0:
            h = np.array(initial_h_values)
            if len(h) != n_points:
                h_orig = np.array(initial_h_values)
                x_orig = np.linspace(0, 2, len(h_orig))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h_orig)
        else:
            # Initialize randomly with sum=50
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
        h = enforce_sum(h, n_points / 2)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Evolutionary optimization loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = []
        for individual in population:
            c5 = evaluate_c5(individual)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Selection (elitism)
        selected = population[:elite_size]
        
        # Crossover and mutation
        new_population = selected[:]
        
        # Crossover: blend two parents (simple average with rescaling)
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate:
                parent1_idx = np.random.randint(0, elite_size)
                parent2_idx = np.random.randint(0, elite_size)
                parent1 = population[parent1_idx]
                parent2 = population[parent2_idx]
                child = (parent1 + parent2) / 2.0
                child = enforce_sum(child, n_points / 2)
                new_population.append(child)
            else:
                # Mutation: flip some bits
                parent_idx = np.random.randint(0, elite_size)
                parent = population[parent_idx].copy()
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        if parent[i] > 0.5:
                            parent[i] = 0.0
                        else:
                            parent[i] = 1.0
                parent = enforce_sum(parent, n_points / 2)
                new_population.append(parent)
        
        population = new_population[:population_size]
    
    # Refinement step with convex optimization on best individuals
    try:
        # Create a more sophisticated refinement approach
        top_individuals = []
        fitness = []
        for individual in population:
            c5 = evaluate_c5(individual)
            top_individuals.append(individual)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        top_individuals = [top_individuals[i] for i in sorted_indices[:10]]
        
        # Try to improve with CVX optimization on top candidates
        for candidate in top_individuals:
            try:
                h_vars = cp.Variable(n_points)
                constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
                
                # Compute the maximum overlap using conv
                h_tmp = np.array(candidate)
                h_tmp = np.clip(h_tmp, 0, 1)
                h_tmp = h_tmp * (n_points / 2) / np.sum(h_tmp)
                
                h_vars.value = h_tmp
                
                # Use a convex relaxation of the solvable problem
                objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.CBC, verbose=False)
                
                if prob.status == cp.OPTIMAL and h_vars.value is not None:
                    h_refined = np.clip(h_vars.value, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = evaluate_c5(h_refined)
                    
                    if c5_refined < best_c5:
                        best_c5 = c5_refined
                        best_h = h_refined.copy()
            except:
                continue
    except:
        pass
    
    # Final evaluation
    if best_h is None:
        best_h = h
    
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
