import numpy as np
import time
from heapq import heappush, heappop
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Precompute shift indices
    shift_indices = list(range(-(n_points-1), n_points))
    
    # Helper to compute overlaps
    def compute_all_overlaps(h):
        # Correlation using FFT
        corr = signal.correlate(h, 1-h, mode='full')
        # Convert to numpy array and extract valid shifts
        corr_array = np.array(corr)
        overlaps = {}
        for k in shift_indices:
            overlaps[k] = corr_array[k + (n_points - 1)] * dx
        return overlaps
    
    # Helper to compute max overlap
    def compute_max_overlap(h):
        return max(compute_all_overlaps(h).values())
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_max_overlap(h)
    
    start_time = time.time()
    
    # Key parameters
    window_width = 5
    perturb_scale = 0.02
    max_stall = 10
    stall_counter = 0
    improvement_threshold = 1e-6
    
    # Main loop
    iterations = 0
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        
        # Compute overlaps
        overlaps = compute_all_overlaps(h)
        
        # Create priority queue of problematic shifts (max heap)
        pq = [(-v, k) for k, v in overlaps.items() if k != 0]
        pq.sort()  # Simple sort to simulate max heap
        
        if not pq:
            break
            
        # Get worst shift
        max_val, k = pq[0]
        max_val = -max_val
        k_start = k + (n_points - 1)

        if max_val < best_c5 - improvement_threshold:
            stall_counter = 0

            stall_counter += 1
            if stall_counter > max_stall:
                # Introduce randomness
                perturbation = np.random.normal(0, perturb_scale, n_points)
                h = h + perturbation
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                stall_counter = 0  # Reset stall counter
                if compute_max_overlap(h) < best_c5:
                    best_h = h.copy()
                    best_c5 = compute_max_overlap(h)
                continue
        
        # Adjust h based on the worst shift k
        # Find indices where h_i and h_{i+k} overlap in the sum
        adjusted = h.copy()
        adjust_indices = []
        
        # Exception handling: empty adjustment may occur
        if k >= 0:
            adjust_indices = list(range(n_points - k))
        elif k < 0:
            adjust_indices = list(range(-k, n_points))
            
        # Determine whether to increase or decrease h based on current behavior
        if abs(k) <= n_points:
            # Decide on adjustment
            adjust_amount = min(0.1, 0.01 + max_val * 0.1)
            if k > 0:
                # Adjust forward indices (k positive)
                for i in adjust_indices:
                    # Prefer to decrease areas where h overlaps heavily with (1-h)
                    if i + k < n_points:
                        if h[i] > 0.5 and h[i+k] < 0.5:
                            adjusted[i] = max(0, h[i] - adjust_amount)
                        elif h[i] < 0.5 and h[i+k] > 0.5:
                            adjusted[i] = min(1, h[i] + adjust_amount)
            else:  # k < 0
                # Adjust backward indices
                for i in adjust_indices:
                    if h[i] > 0.5 and h[i+k] < 0.5:
                        adjusted[i] = max(0, h[i] - adjust_amount)
                    elif h[i] < 0.5 and h[i+k] > 0.5:
                        adjusted[i] = min(1, h[i] + adjust_amount)
        
        # Apply the adjustment
        new_h = adjusted.copy()
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # If improvement, update
        if compute_max_overlap(new_h) < best_c5:
            best_h = new_h.copy()
            best_c5 = compute_max_overlap(new_h)
            
        h = new_h.copy()
        
        # Periodic full overlap recomputation to manage drift
        if iterations % 50 == 0:
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Periodic reset if not improving for a while
        if stall_counter > max_stall * 2:
            if 'initial_h_values' in globals():

                x_orig = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_orig, h)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                h = np.ones(n_points) * 0.5
            stall_counter = 0
            
    return (best_h.tolist(), best_c5, n_points)
