import numpy as np
from scipy.stats import qmc
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Latin Hypercube Sampling approach
    sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs_samples = sampler.random(n=50)
    
    # Create initial candidates: LHS samples + parent
    candidates = []
    if 'initial_h_values' in globals():
        candidates.append(h)
    
    for i in range(lhs_samples.shape[0]):
        h_sample = lhs_samples[i, :]
        # Rescale to sum to n_points/2
        h_sample = h_sample * (n_points / 2) / np.sum(h_sample)
        # Clip to [0,1]
        h_sample = np.clip(h_sample, 0, 1)
        candidates.append(h_sample)
    
    # Refine each candidate with gradient descent
    max_grad_steps = 300
    for candidate in candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        h = candidate.copy()
        c5_prev = float('inf')
        lr = 0.01
        stagnation_counter = 0
        
        for step in range(max_grad_steps):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Compute gradient via finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
                
            # Update
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Check improvement
            correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(correlation_new) * dx
            
            if c5_new < c5:
                h = h_new
                c5_prev = c5_new
                stagnation_counter = 0
            else:
                stagnation_counter += 1
                if stagnation_counter > 20:
                    lr *= 0.5
                    stagnation_counter = 0
                    if lr < 1e-4:
                        break
                        
    # Final evaluation of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
