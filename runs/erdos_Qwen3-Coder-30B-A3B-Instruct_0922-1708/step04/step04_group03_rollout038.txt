import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    max_no_improvement = 50
    no_improvement_count = 0
    
    # Two-phase approach: convex relaxation + gradient descent
    # Phase 1: Use CVXPY for convex relaxation
    try:
        # Create variables for the convex problem
        h_vars = cp.Variable(n_points)
        # Minimize sum of overlap terms
        objective = cp.Minimize(cp.sum(h_vars))
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = np.array(h_vars.value).flatten()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass  
    
    # Phase 2: Refine with gradient descent
    while time.time() - start_time < budget_s - 1: 
        iteration += 1
        improved = False
        
        # Gradient descent with finite differences
        h_new = h.copy()
        for refine_iter in range(50):
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Compute gradient with finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - c5) / eps
            
            # Adaptive learning rate
            lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Check for improvement
            corr = np.correlate(h, 1 - h, mode='full')
            c5_refine = np.max(corr) * dx
            
            if c5_refine < best_c5:
                best_c5 = c5_refine
                best_h = h.copy()
                improved = True
        
        # Early stopping
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
