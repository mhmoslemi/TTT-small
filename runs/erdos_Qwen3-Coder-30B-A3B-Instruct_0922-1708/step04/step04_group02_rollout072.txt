import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.full(n_points, 0.5)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure valid bounds
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    temperature = 1.0
    cooling_rate = 0.99
    iterations = 0
    max_iter = 9000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        iterations += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate candidate
        h_new = h.copy()
        
        # Select random points to perturb (10% of points)
        perturb_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        
        for i in perturb_indices:
            # Perturb value
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
            h_new[i] = max(0, min(1, h_new[i]))
        
        # Maintain sum constraint
        sum_diff = np.sum(h_new) - n_points / 2
        if abs(sum_diff) > 1e-8:
            # Redistribute among other points
            indices = list(set(range(n_points)) - set(perturb_indices))
            if len(indices) > 0:
                diff = sum_diff / len(indices)
                h_new[indices] -= diff
                # Clamp to bounds
                h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / temperature):
            h = h_new
        
        # Cool down
        if iterations % 100 == 0:
            temperature *= cooling_rate
            
    # Final refinement
    # Try small local adjustments around best solution
    h_refined = best_h.copy()
    for i in range(n_points):
        if np.random.random() < 0.1:
            # Systematic perturbation
            for delta in [-0.01, 0.01]:
                h_test = h_refined.copy()
                h_test[i] += delta
                h_test[i] = max(0, min(1, h_test[i]))
                
                # Maintain integral
                sum_diff = np.sum(h_test) - n_points / 2
                if abs(sum_diff) > 1e-8:
                    indices = [j for j in range(n_points) if j != i]
                    diff = sum_diff / len(indices)
                    h_test[indices] -= diff
                    h_test = np.clip(h_test, 0, 1)
                
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                
                correlation_test = np.correlate(h_test, 1 - h_test, mode='full')
                c5_test = np.max(correlation_test) * dx
                
                if c5_test < best_c5:
                    h_refined = h_test
                    best_c5 = c5_test
    
    return (best_h.tolist(), best_c5, n_points)
