import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    h = (h_init >= 0.5).astype(int)
    
    # Ensure exactly n_points/2 ones
    count_ones = np.sum(h)
    if count_ones < n_points / 2:
        indices = np.where(h == 0)[0]
        np.random.shuffle(indices)
        h[indices[:int(n_points/2 - count_ones)]] = 1
    elif count_ones > n_points / 2:
        indices = np.where(h == 1)[0]
        np.random.shuffle(indices)
        h[indices[:int(count_ones - n_points/2)]] = 0
    
    # Ensure feasibility with exact count
    h = h.astype(float)
    count_ones = int(n_points / 2)
    ones_indices = np.where(h == 1)[0]
    if len(ones_indices) > count_ones:
        # Turn some ones to zeros
        to_zero = len(ones_indices) - count_ones
        for i in range(to_zero):
            idx = np.random.choice(ones_indices)
            h[idx] = 0
            ones_indices = np.where(h == 1)[0]
    elif len(ones_indices) < count_ones:
        # Turn some zeros to ones
        to_one = count_ones - len(ones_indices)
        zeros_indices = np.where(h == 0)[0]
        for i in range(to_one):
            idx = np.random.choice(zeros_indices)
            h[idx] = 1
            zeros_indices = np.where(h == 0)[0]
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    T = 1.0
    decay = 0.999
    iter_count = 0
    no_improvement = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        iter_count += 1
        
        # Make a random change
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]
        
        # Ensure exactly n_points/2 ones
        ones_count = np.sum(h_new)
        if ones_count != count_ones:
            if ones_count < count_ones:
                # Need to add more ones
                zero_indices = np.where(h_new == 0)[0]
                if len(zero_indices) > 0:
                    to_flip = count_ones - ones_count
                    flip_indices = np.random.choice(zero_indices, size=min(to_flip, len(zero_indices)), replace=False)
                    h_new[flip_indices] = 1
            else:
                # Need to remove ones
                one_indices = np.where(h_new == 1)[0]
                if len(one_indices) > 0:
                    to_flip = ones_count - count_ones
                    flip_indices = np.random.choice(one_indices, size=min(to_flip, len(one_indices)), replace=False)
                    h_new[flip_indices] = 0
        
        new_c5 = compute_c5(h_new)
        delta = new_c5 - current_c5
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_c5 = new_c5
            
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                no_improvement = 0
        else:
            no_improvement += 1
            
        T *= decay
        
        # Periodic perturbation
        if iter_count % 1000 == 0:
            # Flip 5% of the bits randomly
            n_flips = int(0.05 * n_points)
            indices = np.random.choice(n_points, size=n_flips, replace=False)
            h[indices] = 1 - h[indices]
            
            # Ensure exactly count_ones
            ones_count = np.sum(h)
            if ones_count != count_ones:
                if ones_count < count_ones:
                    zero_indices = np.where(h == 0)[0]
                    to_flip = count_ones - ones_count
                    flip_indices = np.random.choice(zero_indices, size=min(to_flip, len(zero_indices)), replace=False)
                    h[flip_indices] = 1
                else:
                    one_indices = np.where(h == 1)[0]
                    to_flip = ones_count - count_ones
                    flip_indices = np.random.choice(one_indices, size=min(to_flip, len(one_indices)), replace=False)
                    h[flip_indices] = 0
    
    return (best_h.tolist(), best_c5, n_points)
