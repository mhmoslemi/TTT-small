import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():

        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation phase
    # Minimize sum of overlap terms
    h_vars = cp.Variable(n_points)
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
    
    # Build the overlap sum using the actual correlations
    # Since we can't directly compute full correlation in CVXPY, we'll use 
    # a convex approximation involving pairwise products
    
    # For now, let's use a simple convex relaxation:
    # minimize sum_{i,k} h[i]*h[(i+k)%n] over all valid k
    # This would require introducing additional variables, but that's complex
    
    # Instead, we'll do a more direct approach with projected gradient descent
    # We'll improve from the initial solution
    
    # Phase 1: Convex relaxation using CVXPY to get a better starting point
    # Actually, since we can't easily specify the full overlap matrix in CVXPY,
    # let's just proceed with projected gradient descent
    
    # Phase 2: Gradient-based refinement
    iteration = 0
    max_no_improvement = 50
    no_improvement_count = 0
    
    # Gradient computation using finite differences
    def compute_c5_and_grad(h):
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences  
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        return c5, grad
    
    def project_and_rescale(h):
        # Clip values
        h = np.clip(h, 0, 1)
        # Rescale to meet constraint
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        # Compute current value and gradient
        c5, grad = compute_c5_and_grad(h)
        
        # Update with gradient descent step
        lr = 0.01 / (1 + iteration * 1e-4)
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = project_and_rescale(h_new)
        
        # Evaluate new point
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept if improvement
        if c5_new < c5 and c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
            improved = True
        else:
            h = h_new.copy()  # Still update for next iteration
            
        # Early stopping
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
    # Final validation
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
