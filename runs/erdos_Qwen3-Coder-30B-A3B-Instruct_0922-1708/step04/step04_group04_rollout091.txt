import numpy as np
from scipy.stats import qmc
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use Latin Hypercube Sampling for diverse initialization
    sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs_samples = sampler.random(n=50)
    lhs_samples = lhs_samples * 2  # Scale to [0,2]
    
    # Generate candidates using LHS samples
    candidates = []
    for sample in lhs_samples:
        # Convert LHS sample to feasible h
        h_sample = np.clip(sample, 0, 1)
        h_sample = h_sample * (n_points / 2) / np.sum(h_sample)
        candidates.append(h_sample)
    
    # Add the parent if it exists
    if 'initial_h_values' in globals():
        candidates.append(h)
    
    # Gradient descent with adaptive learning rate
    max_iterations = 1000
    lr = 0.01
    patience = 0
    patience_limit = 20
    
    for iter_count in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try each candidate with gradient descent
        for i, candidate in enumerate(candidates):
            h = candidate.copy()
            prev_c5 = float('inf')
            
            for step in range(500):  # Max gradient steps per candidate
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Compute overlap integral C₅
                corr = np.correlate(h, 1-h, mode='full')
                c5 = np.max(corr) * dx
                
                # Gradient estimation using finite differences
                grad = np.zeros_like(h)
                eps = 1e-5
                
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1-h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5) / eps
                    
                # Update with adaptive learning rate
                h_new = h - lr * grad
                
                # Clip and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Track improvement
                corr_new = np.correlate(h_new, 1-h_new, mode='full')
                c5_new = np.max(corr_new) * dx
                
                if c5_new < c5:
                    h = h_new.copy()
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                    prev_c5 = c5_new
                    patience = 0
                else:
                    patience += 1
                    if patience >= patience_limit:
                        break
                        
            # Adjust learning rate
            if lr > 1e-4:
                lr *= 0.999
                
            # Early stopping if not improving
            if patience >= patience_limit:
                break
                
    return (best_h.tolist(), best_c5, n_points)
