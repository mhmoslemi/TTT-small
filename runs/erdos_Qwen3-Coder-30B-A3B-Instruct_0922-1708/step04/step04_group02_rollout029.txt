import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    
    # Parameters
    n_points_list = [50, 100, 200]
    max_iter = 1000
    pop_size = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_ratio = 0.2
    
    best_h = None
    best_c5 = float('inf')
    
    # Try different resolutions
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # Initialize population
        population = []
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                h_orig = np.array(initial_h_values)
                x_orig = np.linspace(0, 2, len(h_orig))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h_orig)
            h = h / np.sum(h) * (n_points / 2)
        else:
            h = np.ones(n_points) * 0.5
        population.append(h.copy())
        
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
            population.append(h.copy())
        
        # GA loop
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Evaluate fitness
            fitness = []
            for h in population:
                correlation = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(correlation) * dx
                fitness.append(c5)
            
            # Sort by fitness
            sorted_indices = np.argsort(fitness)
            population = [population[i] for i in sorted_indices]
            fitness = [fitness[i] for i in sorted_indices]
            
            # Update best
            if fitness[0] < best_c5:
                best_c5 = fitness[0]
                best_h = population[0].copy()
            
            # Select elite
            elite_count = int(pop_size * elite_ratio)
            elite = population[:elite_count]
            
            # Create new population
            new_population = elite[:]
            
            # Crossover and mutation
            while len(new_population) < pop_size:
                # Tournament selection
                a, b = random.sample(range(elite_count), 2)
                parent1 = elite[a] if fitness[a] < fitness[b] else elite[b]
                
                a, b = random.sample(range(elite_count), 2)
                parent2 = elite[a] if fitness[a] < fitness[b] else elite[b]
                
                if random.random() < crossover_rate:
                    # SBX crossover
                    child1, child2 = sbx_crossover(parent1, parent2)
                else:
                    child1, child2 = parent1.copy(), parent2.copy()
                
                # Mutation
                if random.random() < mutation_rate:
                    child1 = polynomial_mutation(child1, n_points)
                if random.random() < mutation_rate:
                    child2 = polynomial_mutation(child2, n_points)
                
                # Repair
                child1 = repair_constraints(child1, n_points, dx)
                child2 = repair_constraints(child2, n_points, dx)
                
                new_population.extend([child1, child2])
            
            population = new_population[:pop_size]
            
            # Adaptive mutation rate
            if iteration % 50 == 0:
                if iteration > 0 and iteration % 100 == 0:
                    mutation_rate = min(0.3, mutation_rate * 1.1)
                else:
                    mutation_rate = max(0.01, mutation_rate * 0.95)
    
    # Final gradient refinement on best solution
    if best_h is not None and time.time() - start_time < budget_s - 1:
        best_h = refine_with_gradients(best_h, n_points, dx, start_time, budget_s)
        
    # Final evaluation
    if best_h is not None:
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(correlation) * dx
    
    if best_h is None:
        # Fallback to simple construction if no search worked
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
        best_h = h.copy()
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), best_c5, n_points)


def sbx_crossover(parent1, parent2, eta=20):
    """Simulated Binary Crossover"""
    child1 = np.zeros_like(parent1)
    child2 = np.zeros_like(parent2)
    
    for i in range(len(parent1)):
        if random.random() < 0.5:
            if abs(parent1[i] - parent2[i]) > 1e-14:
                y1, y2 = min(parent1[i], parent2[i]), max(parent1[i], parent2[i])
                
                rand = random.random()
                
                # Compute beta
                if rand <= 0.5:
                    beta = 2 * rand
                else:
                    beta = 1.0 / (2 * (1 - rand))
                
                # Compute alpha
                alpha = 2.0 - beta**(-1/eta)
                
                child1[i] = 0.5 * ((y1 + y2) - alpha * (y2 - y1))
                child2[i] = 0.5 * ((y1 + y2) + alpha * (y2 - y1))
            else:
                child1[i] = parent1[i]
                child2[i] = parent2[i]
        else:
            child1[i] = parent1[i]
            child2[i] = parent2[i]
    
    # Bounds clipping
    child1 = np.clip(child1, 0, 1)
    child2 = np.clip(child2, 0, 1)
    
    return child1, child2


def polynomial_mutation(h, n_points, eta=20):
    """Polynomial Mutation"""
    epsilon = 1e-14
    for i in range(len(h)):
        if random.random() < 1.0 / n_points:
            y1 = h[i]
            y2 = 1.0 - h[i] if y1 < 0.5 else h[i]
            
            rand = random.random()
            
            if rand <= 0.5:
                # Compute delta1
                delta1 = (y1 - 0) / (1.0 - 0)
                if delta1 <= 0.5:
                    beta = 2 * delta1
                else:
                    beta = 1.0 / (2 * (1 - delta1))
                alpha = 2.0 - beta**(-1/eta)
                y1 = 0.5 * (0 + 1 - alpha * (1 - 0))
            else:
                # Compute delta2
                delta2 = (1.0 - y1) / (1.0 - 0)
                if delta2 <= 0.5:
                    beta = 2 * delta2
                else:
                    beta = 1.0 / (2 * (1 - delta2))
                alpha = 2.0 - beta**(-1/eta)
                y1 = 0.5 * (0 + 1 + alpha * (1 - 0))
            
            y1 = np.clip(y1, 0, 1)
            
            # Apply mutation
            h[i] = y1
    
    return h


def repair_constraints(h, n_points, dx):
    """Repair solution to satisfy constraints"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points/2) > 1e-8:
        diff = (n_points/2 - h_sum) / n_points
        h = h + diff
        h = np.clip(h, 0, 1)
    return h


def refine_with_gradients(h, n_points, dx, start_time, budget_s):
    """Refine the best solution using gradient descent"""
    max_refine_iter = 1000
    
    # Add small noise to avoid being stuck
    h = h * (1.0 + 1e-6 * np.random.randn(len(h)))
    h = repair_constraints(h, n_points, dx)
    
    # Define objective function and gradient approximation
    def objective(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return -np.max(correlation) * dx
    
    # Gradient approximation is done with finite differences
    def grad_approx(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - (-objective(h_vec))) / eps
        return grad
    
    iter_count = 0
    while iter_count < max_refine_iter and time.time() - start_time < budget_s - 1:
        g = grad_approx(h, 1e-4)
        h = h - 0.001 * g
        h = repair_constraints(h, n_points, dx)
        iter_count += 1
    
    return h
