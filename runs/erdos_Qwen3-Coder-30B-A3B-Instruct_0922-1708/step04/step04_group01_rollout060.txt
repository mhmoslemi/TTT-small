import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    def create_individual():
        # Create a binary array with exactly target_sum ones
        h = np.zeros(n_points)
        ones_indices = np.random.choice(n_points, int(target_sum), replace=False)
        h[ones_indices] = 1
        return h
    
    # Convert to valid continuous representation
    def to_continuous(h_binary):
        h = h_binary.astype(float)
        h = h * (target_sum / np.sum(h))
        return np.clip(h, 0, 1)
    
    # Evaluate C5 for a given h
    def evaluate_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Evolutionary algorithm
    population_size = 100
    generations = 100
    mutation_rate = 0.05
    crossover_rate = 0.5
    elite_size = int(0.1 * population_size)
    
    # Initialize population
    if 'initial_h_values' in globals():
        # Convert and improve the initial solution
        h_init = np.array(initial_h_values)
        h_init = h_init / np.sum(h_init) * target_sum
        pop = [to_continuous(create_individual()) for _ in range(population_size - 1)]
        pop.append(h_init)
    else:
        pop = [to_continuous(create_individual()) for _ in range(population_size)]
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Evaluate fitness
        fitness = [evaluate_c5(h) for h in pop]
        sorted_indices = np.argsort(fitness)
        
        # Keep elite
        new_pop = [pop[i] for i in sorted_indices[:elite_size]]
        
        # Generate offspring
        while len(new_pop) < population_size:
            if np.random.rand() < crossover_rate and len(new_pop) + 1 < population_size:
                # Simple crossover
                i1, i2 = np.random.choice(elite_size, 2, replace=False)
                h1, h2 = pop[i1], pop[i2]
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
                child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
                
                # Mutate
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child1[i] = 1 - child1[i]
                    if np.random.rand() < mutation_rate:
                        child2[i] = 1 - child2[i]
                
                # Repair
                child1 = to_continuous(child1)
                child2 = to_continuous(child2)
                
                new_pop.append(child1)
                if len(new_pop) < population_size:
                    new_pop.append(child2)
            else:
                # Mutation only
                i = np.random.choice(elite_size)
                h = pop[i].copy()
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        h[i] = 1 - h[i]
                h = to_continuous(h)
                new_pop.append(h)
        
        pop = new_pop
    
    # Find best solution from evolution
    final_fitness = [evaluate_c5(h) for h in pop]
    best_idx = np.argmin(final_fitness)
    best_h = pop[best_idx]
    best_c5 = final_fitness[best_idx]
    
    # Refine with convex optimization using the best individual
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == target_sum]
        # Reformulate the overlap maximization problem
        # It will be vulnerable to numerical issues so we use the best solution as initial point
        
        # Use local refinement on top individual
        def obj_func(h_vals):
            h = np.array(h_vals)
            correlation = np.correlate(h, 1 - h, mode='full')
            return -np.max(correlation) * dx  # Because cvxpy minimizes
        
        # Optimize with constraints
        result = minimize(obj_func, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, 
                          constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        
        if result.success:
            h_refined = np.clip(result.x, 0, 1)
            h_refined = h_refined * (target_sum / np.sum(h_refined))
            c5_refined = evaluate_c5(h_refined)
            
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
                
    except Exception:
        # Fall back to original if optimization fails
        pass
    
    # Final validation check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
