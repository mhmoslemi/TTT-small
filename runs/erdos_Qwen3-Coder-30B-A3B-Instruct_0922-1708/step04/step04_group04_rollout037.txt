import numpy as np
import time
from scipy import fft
from heapq import heappush, heappop

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to meet constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Precompute indices for efficient convolution
    def compute_overlap(h_vec):
        # Compute full convolution of h with (1-h) using FFT
        h_conj = 1 - h_vec
        # Use padding to make length sufficient for full convolution
        pad_len = 2 * n_points - 1
        h_padded = np.pad(h_vec, (0, pad_len - len(h_vec)), 'constant')
        h_conj_padded = np.pad(h_conj, (0, pad_len - len(h_conj)), 'constant')
        corr = fft.fft(h_padded) * fft.fft(h_conj_padded).conj()
        corr = fft.ifft(corr).real
        # Take only the overlapping parts
        return corr
    
    # Compression: only compute h and 1-h once, reuse
    def fast_max_overlap(h_vec):
        corr = compute_overlap(h_vec)
        # Take max of positive shifts (corresponds to forward shifts)
        # We care about the most negative convolution value
        max_val = 0
        for i in range(len(corr)):
            if corr[i] > max_val:
                max_val = corr[i]
        return max_val * dx
    
    def validate_h(h_vec):
        return np.all(h_vec >= 0) and np.all(h_vec <= 1) \
               and abs(np.sum(h_vec) - n_points/2) < 1e-6
    
    # Adaptive adjustment with lookup
    def adjust_region(h_vec, start, end, delta):
        h_new = h_vec.copy()
        for i in range(start, min(end, len(h_new))):
            h_new[i] = max(0, min(1, h_new[i] + delta))
        # Maintain the constraint sum(h) = n_points/2
        total = np.sum(h_new)
        factor = (n_points/2) / total
        h_new = h_new * factor
        return h_new

    start_time = time.time()
    
    # Adaptive parameters
    window_width = 5
    perturb_scale = 0.02
    min_improvement = 1e-6
    iter_no_improvement = 0
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        overlap = fast_max_overlap(h)
        
        if overlap < best_c5:
            best_c5 = overlap
            best_h = h.copy()
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
        
        if iter_no_improvement > 30:
            window_width = min(window_width + 1, 10)
            perturb_scale *= 1.5

        # Compute all overlaps for prioritization
        overlaps = []
        for shift in range(-(n_points-1), n_points):
            if shift == 0:
                continue
            h_shifted = np.roll(h, shift)
            if shift >= 0:
                corr = np.sum(h[:len(h)-shift] * (1 - h_shifted[shift:]))
            else:
                corr = np.sum(h[-shift:] * (1 - h_shifted[:-shift]))
            overlaps.append((abs(corr), shift))
        
        overlaps.sort(reverse=True)
        if not overlaps:
            break
        
        # Take top shift to fix
        max_overlap, shift = overlaps[0]
        
        # Modify the values that cause max overlap
        if shift > 0:
            a, b = 0, len(h)-shift
            adjust_start = max(0, a - window_width)
            adjust_end = min(len(h), b + window_width)
            delta = -0.005 * np.sign(shift)
            h = adjust_region(h, adjust_start, adjust_end, delta)
        else:
            a, b = -shift, len(h)
            adjust_start = max(0, a - window_width)
            adjust_end = min(len(h), b + window_width)
            delta = 0.005 * np.sign(shift)
            h = adjust_region(h, adjust_start, adjust_end, delta)
        
        # Perturb randomly in region
        if np.random.random() < 0.3:
            perturb_idx = np.random.randint(0, len(h))
            h[perturb_idx] += np.random.normal(0, perturb_scale)
            h[perturb_idx] = max(0, min(1, h[perturb_idx]))
            
            # Maintain constraint sum(h) = n_points/2
            total = np.sum(h)
            factor = (n_points/2) / total
            h = h * factor

    return (best_h.tolist(), best_c5, n_points)
