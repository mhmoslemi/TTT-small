import numpy as np
from scipy.optimize import minimize
import time
import random
from typing import Tuple, List

def evaluate_h(h: np.ndarray, dx: float) -> float:
    """Evaluate the C5 bound for a given h function."""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def repair_h(h: np.ndarray, n_points: int) -> np.ndarray:
    """Repair h to ensure it's feasible."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate_h(h: np.ndarray, mutation_rate: float) -> np.ndarray:
    """Mutate h with given rate."""
    h_mut = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            # Apply small random perturbation
            perturbation = random.gauss(0, 0.05)
            h_mut[i] += perturbation
    return repair_h(h_mut, len(h))

def crossover_h(h1: np.ndarray, h2: np.ndarray) -> Tuple[np.ndarray, np.ndarray]:
    """Simulated binary crossover."""
    alpha = random.uniform(0, 1)
    h1_child = alpha * h1 + (1 - alpha) * h2
    h2_child = (1 - alpha) * h1 + alpha * h2
    return repair_h(h1_child, len(h1)), repair_h(h2_child, len(h1))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_population = 50
    generations = 200
    mutation_rate = 0.1
    
    # Initialize population using parent if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_parent))
        x_target = np.linspace(0, 2, n_points)
        h_initial = np.interp(x_target, x_orig, h_parent)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    else:
        # Start with a reasonable initialization
        h_initial = np.ones(n_points) * 0.5  # uniform distribution
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Generate initial population
    population = [h_initial]
    for _ in range(max_population - 1):
        h_mut = mutate_h(h_initial, 0.1)
        population.append(h_mut)
    
    best_h = None
    best_c5 = float('inf')
    
    # Track generation without getting hung up on time
    gen_count = 0
    
    while time.time() - start_time < budget_s - 1 and gen_count < generations:
        gen_count += 1
        
        # Evaluate population
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Find best in current generation
        gen_min_fitness = min(fitnesses)
        gen_best_idx = fitnesses.index(gen_min_fitness)
        gen_best_h = population[gen_best_idx]
        
        if gen_min_fitness < best_c5:
            best_c5 = gen_min_fitness
            best_h = gen_best_h.copy()
        
        # Selection: tournament selection
        selected = []
        for _ in range(max_population):
            idx1, idx2 = random.randint(0, max_population-1), random.randint(0, max_population-1)
            if fitnesses[idx1] < fitnesses[idx2]:
                selected.append(population[idx1].copy())
            else:
                selected.append(population[idx2].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, max_population, 2):
            h1 = selected[i]
            h2 = selected[i+1] if i+1 < max_population else selected[0]
            h1_child, h2_child = crossover_h(h1, h2)
            h1_child = mutate_h(h1_child, mutation_rate)
            h2_child = mutate_h(h2_child, mutation_rate)
            new_population.extend([h1_child, h2_child])
        
        # Keep valid population size
        population = new_population[:max_population]
        
        # Adaptive mutation rate
        if gen_count % 10 == 0:
            mutation_rate = max(0.05, mutation_rate * 0.99)
    
    # Final refinement with local search on best solution
    if best_h is not None:
        h_refined = best_h.copy()
        min_c5 = best_c5
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            # Make small perturbations
            h_trial = h_refined.copy()
            idx = random.randint(0, n_points - 1)
            h_trial[idx] += random.uniform(-0.05, 0.05)
            h_trial = repair_h(h_trial, n_points)
            c5_trial = evaluate_h(h_trial, dx)
            if c5_trial < min_c5:
                min_c5 = c5_trial
                h_refined = h_trial

        best_h = h_refined
    
    if best_h is None:
        best_h = h_initial
    
    return (best_h.tolist(), best_c5, n_points)
