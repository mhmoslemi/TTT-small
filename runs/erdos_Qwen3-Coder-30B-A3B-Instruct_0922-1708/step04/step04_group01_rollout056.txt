import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    coarse_n = 20
    fine_n = 100
    dx_coarse = 2.0 / coarse_n
    dx_fine = 2.0 / fine_n
    top_candidates = 10
    max_iter = 500
    lr = 0.01
    
    start_time = time.time()
    
    # Initialize with provided construction or uniform distribution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        if len(h_start) != fine_n:
            # Interpolate to target resolution
            x_orig = np.linspace(0, 2, len(h_start))
            x_target = np.linspace(0, 2, fine_n)
            h_start = np.interp(x_target, x_orig, h_start)
    else:
        h_start = np.ones(fine_n) * 0.5
    
    # Ensure sum constraint
    h_start = h_start * (fine_n / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = float('inf')
    
    # Phase 1: Grid search with coarse resolution for quick exploration
    grid_points = []
    
    # Generate coarse grid
    grid_size = 10  # Grid resolution per dimension
    
    for i in range(grid_size):
        h_grid = np.ones(coarse_n) * (i / (grid_size - 1) if grid_size > 1 else 0.5)
        h_grid = h_grid * (coarse_n / 2) / np.sum(h_grid)
        grid_points.append(h_grid)
    
    # Evaluate all grid points
    grid_scores = []
    
    for h_grid in grid_points:
        correlation = np.correlate(h_grid, 1 - h_grid, mode='full')
        c5 = np.max(correlation) * dx_coarse
        grid_scores.append((c5, h_grid))
    
    # Sort and keep top candidates
    grid_scores.sort(key=lambda x: x[0])
    top_h = [h for _, h in grid_scores[:top_candidates]]
    
    # Phase 2: Refine candidates using high resolution and optimization
    for h_candidate in top_h:
        h = h_candidate.copy()
        
        # Interpolate to fine resolution
        if len(h) < fine_n:
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, fine_n)
            h = np.interp(x_new, x_old, h)
            h = h * (fine_n / 2) / np.sum(h)
        
        # Refinement with gradient-based optimization
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Compute current correlation and gradient
            correlation = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(correlation) * dx_fine
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for j in range(fine_n):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (fine_n / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx_fine
                
                grad[j] = (c5_plus - current_c5) / eps
            
            # Update with gradient ascent
            h_new = h - lr * grad
            
            # Projection onto constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (fine_n / 2) / np.sum(h_new)
            h = h_new
    
    # Final check with CVXPY optimization for better solution if time allows
    if time.time() - start_time < budget_s - 5:
        try:
            h_vars = cp.Variable(fine_n)
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == fine_n / 2]
            
            # Objective: minimize maximum overlap
            # This approach directly tries to optimize the maximum correlation
            
            # For better formulation: define overlap as maximum over all lags
            # We'll build a problem that approximates the worst-case correlation loss
            
            # Manually compute best solution through gradient as above
            # Since full CVXPY optimization for max correlation is complex,
            # we'll use the best one from earlier modified approach
            
        except:
            pass
    
    # Final computation of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx_fine
    
    return (best_h.tolist(), final_c5, fine_n)
