import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Grid search on coarse resolution
    coarse_n = 20
    coarse_dx = 2.0 / coarse_n
    coarse_h = np.ones(coarse_n) * 0.5
    if 'initial_h_values' in globals():
        coarse_h = np.array(initial_h_values)
        if len(coarse_h) != coarse_n:
            x_orig = np.linspace(0, 2, len(initial_h_values))
            x_target = np.linspace(0, 2, coarse_n)
            coarse_h = np.interp(x_target, x_orig, initial_h_values)
    
    # Try a couple of grid points around the best candidates
    candidates = [coarse_h.copy()]
    for i in range(5):
        h_test = np.random.rand(coarse_n)
        h_test = h_test / np.sum(h_test) * (coarse_n / 2)
        candidates.append(h_test)

    top_candidates = []
    for h_test in candidates:
        # Rescale to target n_points
        h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, coarse_n), h_test)
        h_upsampled = h_upsampled / np.sum(h_upsampled) * (n_points / 2)
        
        corr = np.correlate(h_upsampled, 1 - h_upsampled, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_upsampled.copy()
        top_candidates.append((c5, h_upsampled))

    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:5]
    
    # Phase 2: Refinement
    for _, h_refine in top_candidates:
        h = h_refine.copy()
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - best_c5) / eps
            
            lr = 0.01
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Final verification
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
