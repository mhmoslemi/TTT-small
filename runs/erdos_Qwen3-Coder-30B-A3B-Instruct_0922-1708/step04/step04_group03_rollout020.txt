import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Learning rate and decay
    lr = 0.1
    lr_decay = 0.99
    max_iter = 10000
    
    # For early stopping
    patience = 100
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current state
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= patience:
            break
            
        # Compute gradients using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            # Perturb up
            h_up = h.copy()
            h_up[i] += eps
            h_up = np.clip(h_up, 0, 1)
            h_up = h_up * (n_points / 2) / np.sum(h_up)
            corr_up = np.correlate(h_up, 1 - h_up, mode="full")
            c5_up = np.max(corr_up) * dx
            
            # Perturb down
            h_dn = h.copy()
            h_dn[i] -= eps
            h_dn = np.clip(h_dn, 0, 1)
            h_dn = h_dn * (n_points / 2) / np.sum(h_dn)
            corr_dn = np.correlate(h_dn, 1 - h_dn, mode="full")
            c5_dn = np.max(corr_dn) * dx
            
            # Estimate derivative
            grad[i] = (c5_up - c5_dn) / (2 * eps)
            
        # Update h
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Decay learning rate
        if iteration % 50 == 0:
            lr *= lr_decay
            
    return (best_h.tolist(), best_c5, n_points)
