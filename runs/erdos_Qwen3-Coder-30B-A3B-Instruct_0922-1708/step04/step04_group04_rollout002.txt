import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Optimization with CVXPY
    # Variables: h (size n_points), C (scalar)
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints for mean
    constraints = [cp.sum(h_var) == n_points / 2]
    
    # Ensure h is in [0,1]
    constraints += [h_var >= 0, h_var <= 1]
    
    # Constraints for overlap
    for k in range(1, n_points):
        if time.time() - start_time > budget_s - 1:
            break
        # For positive shift k (k > 0)
        # Use a subset of indices for correlation
        # h[i] * (1 - h[i+k]) for all valid i
        # Sum from i=0 to (n-k-1) 
        corr_sum = 0
        for i in range(n_points - k):
            corr_sum += h_var[i] * (1 - h_var[i + k])
        # Scale by dx
        constraints += [corr_sum <= C * dx]
        
    # Objective: minimize C
    objective = cp.Minimize(C)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers = ['ECOS', 'SCS']
    
    for solver in solvers:
        if time.time() - start_time > budget_s - 1:
            break
        try:
            prob.solve(solver=solver, verbose=False)
            
            if prob.status == 'optimal':
                h_sol = h_var.value
                c5_sol = C.value
                
                if c5_sol < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = c5_sol
                    
                    # Re-scale
                    h_norm = best_h * (n_points / 2) / np.sum(best_h)
                    # Ensure bounds
                    h_norm = np.clip(h_norm, 0, 1)
                    best_h = h_norm.copy()
            break  # Success, stop trying other solvers
        except:
            continue
        
    # Final polishing with gradient descent
    h = best_h.copy()
    
    # Safeguard convergence
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Estimate gradient via finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Adaptive stepsize
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-4)
        h_new = h - lr * grad
        
        # Ensure constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update best
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
    # Return final result with constraints satisfied
    return (best_h.tolist(), best_c5, n_points)
