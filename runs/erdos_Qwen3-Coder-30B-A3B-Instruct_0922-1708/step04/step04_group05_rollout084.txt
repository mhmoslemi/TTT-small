import numpy as np
import time

def project_feasible(h, n_points):
    """Project h to satisfy 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h):
    """Compute C5 value from h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * (2.0 / len(h))

def gradient_step(h, eps=1e-4):
    """Compute finite difference gradient of C5 with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    base_c5 = compute_c5(h)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n)
        plus_c5 = compute_c5(h_plus)
        grad[i] = (plus_c5 - base_c5) / eps
    return grad

def line_search(h, grad, max_iter=20):
    """Line search for optimal step size"""
    alpha = 0.001
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, len(h))
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h):
            return h_new
        alpha *= 0.5
    return h

    else:
        h = np.ones(n_points) * 0.5
    
    h = project_feasible(h, n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Adaptive grid search with refinement
    early_stop_count = 0
    min_improvement = 1e-5

            grad = gradient_step(h)
            h = line_search(h, grad)

            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
