def run(seed=42, budget_s=600, **kwargs):

        h = np.ones(n_points) * 0.5

        max_iter = 10000
        lr = 0.01
        decay_factor = 0.9995

            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current value of C5
            corr = np.correlate(h, 1 - h, mode='full')

                best_c5 = current_c5
                best_h = h.copy()
                patience_counter = 0

                patience_counter += 1

            eps = 1e-5

            h = np.clip(h, 0, 1)

            # Update learning rate

    # Final verification
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
