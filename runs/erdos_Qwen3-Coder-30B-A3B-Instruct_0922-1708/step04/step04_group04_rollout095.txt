import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()

    # Convert to CVXPY variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()

    # Constraints
    constraints = [h_var >= 0, h_var <= 1]
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Build the maximum overlap constraint for each shift
    for shift in range(1, n_points):
        # Compute correlation for this shift
        corr = 0
        for i in range(n_points - shift):
            corr += h_var[i] * (1 - h_var[i + shift])
        constraints.append(corr <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with ECOS solver
    try:
        prob.solve(solver=cp.ECOS, max_iters=100, feastol=1e-5)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            if h_opt is not None:
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                # Compute actual C5
                correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
                c5 = np.max(correlation) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
    except:
        pass
    
    # If no improvement from convex optimization, do local refinement
    if time.time() - start_time > budget_s - 10:
        return (best_h.tolist(), best_c5, n_points)

    # Apply local refinement with gradient descent-like steps
    h = best_h.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute current value of C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx

        # Estimate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps

        # Update using adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-2)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update if better
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new

    return (best_h.tolist(), best_c5, n_points)
