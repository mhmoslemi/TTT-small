import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 150
    dx = 2.0 / n_points
    max_pso_iter = 200
    max_refine_iter = 500
    swarm_size = 20
    inertia_start = 0.9
    inertia_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_local_iter = 50
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO initialization
    particles = np.random.uniform(0.1, 0.9, (swarm_size, n_points))
    for i in range(swarm_size):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    velocities = np.random.uniform(-0.01, 0.01, (swarm_size, n_points))
    pbest_positions = particles.copy()
    pbest_scores = np.full(swarm_size, float('inf'))
    gbest_position = h_init.copy()
    gbest_score = float('inf')
    
    # PSO optimization loop
    for iteration in range(max_pso_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Inertia weight schedule
        inertia = inertia_start - (inertia_start - inertia_end) * iteration / max_pso_iter
        
        # Evaluate all particles
        for i in range(swarm_size):
            h = particles[i].copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate overlap integral
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < pbest_scores[i]:
                pbest_scores[i] = c5
                pbest_positions[i] = h.copy()
                
            if c5 < gbest_score:
                gbest_score = c5
                gbest_position = h.copy()
                
        # Update global best if needed
        if gbest_score < best_c5:
            best_c5 = gbest_score
            best_h = gbest_position.copy()
            
        # Update particles
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (gbest_position - particles[i]))
            
            particles[i] += velocities[i]
            
            # Keep in bounds and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Refinement using gradient descent
    start_refinement = time.time()
    refine_particles = [gbest_position.copy()]
    for _ in range(4):
        idx = np.random.randint(0, swarm_size)
        refine_particles.append(pbest_positions[idx].copy())
    
    for h in refine_particles:
        if time.time() - start_time > budget_s - 20:
            break
            
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        for local_iter in range(max_local_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Numerical gradient estimation
            grad = np.zeros(n_points)
            eps = 1e-5
            
            for j in range(n_points):
                h_eps = h.copy()
                h_eps[j] += eps
                h_eps = np.clip(h_eps, 0, 1)
                h_eps = h_eps * (n_points / 2) / np.sum(h_eps)
                corr_plus = np.correlate(h_eps, 1 - h_eps, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_neg = h.copy()
                h_neg[j] -= eps
                h_neg = np.clip(h_neg, 0, 1)
                h_neg = h_neg * (n_points / 2) / np.sum(h_neg)
                corr_neg = np.correlate(h_neg, 1 - h_neg, mode='full')
                c5_neg = np.max(corr_neg) * dx
                
                grad[j] = (c5_plus - c5_neg) / (2 * eps)
                
            # Gradient descent step
            lr = 0.001 / (1 + np.linalg.norm(grad)*1e-3)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new solution
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr) * dx
            
            # Accept or reject step
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
            else:
                # Sometimes accept bad steps to escape local minima
                if np.random.rand() < 0.01:
                    h = h_new
    
    # Final computation of best solution
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
