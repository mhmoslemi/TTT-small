import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary
    h = (h_init >= 0.5).astype(int)
    
    # Ensure we have exactly n_points/2 ones
    count_ones = np.sum(h)
    if count_ones > n_points // 2:
        # Set some ones to zeros
        zero_indices = np.where(h == 1)[0]
        to_flip = zero_indices[:count_ones - n_points // 2]
        h[to_flip] = 0
    elif count_ones < n_points // 2:
        # Set some zeros to ones
        one_indices = np.where(h == 0)[0]
        to_flip = one_indices[:n_points // 2 - count_ones]
        h[to_flip] = 1
    
    # Compute initial C5
    correlation_full = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(correlation_full) * dx
    best_h = h.copy()
    
    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.999
    perturbation_freq = 1000
    perturb_ratio = 0.05
    
    start_time = time.time()
    
    # Main optimization loop
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 10000
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iter_count += 1
        
        # Make a random change
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]  # Flip the bit
        
        # Project to ensure exactly n_points/2 ones
        count_new = np.sum(h_new)
        if count_new > n_points // 2:
            # Need to set some ones to zeros
            flip_indices = np.where(h_new == 1)[0]
            to_flip = flip_indices[:count_new - n_points // 2]
            h_new[to_flip] = 0
        elif count_new < n_points // 2:
            # Need to set some zeros to ones
            flip_indices = np.where(h_new == 0)[0]
            to_flip = flip_indices[:n_points // 2 - count_new]
            h_new[to_flip] = 1
        
        # Compute the new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject the change
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            # Accept with probability based on simulated annealing
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                h = h_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Decrease temperature
        temp *= cooling_rate
        
        # Perturb the solution periodically
        if iter_count % perturbation_freq == 0:
            num_to_flip = int(n_points * perturb_ratio)
            indices_to_flip = np.random.choice(n_points, num_to_flip, replace=False)
            h[indices_to_flip] = 1 - h[indices_to_flip]
            
            # Project again
            count_new = np.sum(h)
            if count_new > n_points // 2:
                flip_indices = np.where(h == 1)[0]
                to_flip = flip_indices[:count_new - n_points // 2]
                h[to_flip] = 0
            elif count_new < n_points // 2:
                flip_indices = np.where(h == 0)[0]
                to_flip = flip_indices[:n_points // 2 - count_new]
                h[to_flip] = 1
                
        # Early termination if no improvement for a long time
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
