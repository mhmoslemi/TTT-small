import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Differential Evolution parameters
    population_size = 50
    F = 0.8
    CR = 0.9
    max_no_improvement = 100
    
    # Initialize population
    population = []
    population_values = []
    
    # Add initial solution to population
    population.append(h.copy())
    correlation = np.correlate(h, 1 - h, mode='full')
    population_values.append(np.max(correlation) * dx)
    
    # Fill rest of population with random feasible solutions
    for i in range(population_size - 1):
        new_h = np.random.uniform(0, 1, n_points)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        population.append(new_h)
        correlation = np.correlate(new_h, 1 - new_h, mode='full')
        population_values.append(np.max(correlation) * dx)
        
    # Global best
    best_idx = np.argmin(population_values)
    gbest = population[best_idx].copy()
    gbest_val = population_values[best_idx]
    
    if gbest_val < best_c5:
        best_c5 = gbest_val
        best_h = gbest.copy()
        
    start_time = time.time()
    
    # Optimization loop
    no_improvement_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        # DE evolution
        new_population = []
        new_population_values = []
        
        for i in range(population_size):
            # Select three different individuals
            indices = np.random.choice(population_size, 3, replace=False)
            x1, x2, x3 = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = x1 + F * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.copy(population[i])
            j_rand = np.random.randint(0, n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                    
            # Rescale to satisfy constraint
            trial = trial * (n_points / 2) / np.sum(trial)
            
            # Evaluate
            correlation = np.correlate(trial, 1 - trial, mode='full')
            trial_val = np.max(correlation) * dx
            
            # Selection
            if trial_val < population_values[i]:
                new_population.append(trial)
                new_population_values.append(trial_val)
                if trial_val < best_c5:
                    best_c5 = trial_val
                    best_h = trial.copy()
                    improved = True
            else:
                new_population.append(population[i])
                new_population_values.append(population_values[i])
        
        population = new_population
        population_values = new_population_values
        
        # Update global best
        current_best_idx = np.argmin(population_values)
        current_best_val = population_values[current_best_idx]
        if current_best_val < gbest_val:
            gbest = population[current_best_idx].copy()
            gbest_val = current_best_val
            improved = True
            
        # Local search every 20 iterations on the global best
        if iteration % 20 == 0:
            h = gbest.copy()
            # Gradient-based local search
            for refine_iter in range(30):
                correlation = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(correlation) * dx
                
                # Finite differences gradient estimation
                grad = np.zeros_like(h)
                eps = 1e-4
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5) / eps
                
                # Adaptive learning rate
                lr = 0.02 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h - lr * grad
                
                # Clip and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                h = h_new
                
                # Re-evaluate
                corr = np.correlate(h, 1 - h, mode='full')
                c5_refine = np.max(corr) * dx
                
                if c5_refine < best_c5:
                    best_c5 = c5_refine
                    best_h = h.copy()
                    
            gbest = h.copy()
            gbest_val = best_c5
            
        # Early stopping
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
    return (best_h.tolist(), best_c5, n_points)
