import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    initial_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary representation
    h = (h_init >= 0.5).astype(int)
    
    # Ensure exactly n_points/2 ones
    count_ones = np.sum(h)
    if count_ones > n_points // 2:
        # Turn some ones into zeros
        zeros_indices = np.where(h == 0)[0]
        ones_indices = np.where(h == 1)[0]
        for i in range(count_ones - n_points // 2):
            if len(ones_indices) > 0:
                idx = np.random.choice(ones_indices)
                h[idx] = 0
                ones_indices = np.delete(ones_indices, np.where(ones_indices == idx)[0])
    elif count_ones < n_points // 2:
        # Turn some zeros into ones
        zeros_indices = np.where(h == 0)[0]
        for i in range(n_points // 2 - count_ones):
            if len(zeros_indices) > 0:
                idx = np.random.choice(zeros_indices)
                h[idx] = 1
                zeros_indices = np.delete(zeros_indices, np.where(zeros_indices == idx)[0])
    
    # Ensure feasibility
    h = h.astype(float)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T = 1.0
    decay = 0.999
    perturbation_rate = 0.05
    perturbation_interval = 1000
    
    # Evaluate initial solution
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    current_c5 = compute_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Main optimization loop
    max_iter = budget_s * 1000  # Rough estimate
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 10000
    
    while iter_count < max_iter and (time.time() - initial_time) < budget_s - 1:
        iter_count += 1
        # Select a random index to flip
        i = np.random.randint(0, n_points)
        
        # Save current state
        old_h = h.copy()
        old_c5 = current_c5
        
        # Flip the bit
        h[i] = 1 - h[i]
        
        # Project to ensure exactly n_points/2 ones
        count_ones = np.sum(h)
        if count_ones > n_points // 2:
            # Find ones and turn some into zeros
            ones_indices = np.where(h == 1)[0]
            for _ in range(count_ones - n_points // 2):
                if len(ones_indices) > 0:
                    idx = np.random.choice(ones_indices)
                    h[idx] = 0
                    ones_indices = np.delete(ones_indices, np.where(ones_indices == idx)[0])
        elif count_ones < n_points // 2:
            # Find zeros and turn some into ones
            zeros_indices = np.where(h == 0)[0]
            for _ in range(n_points // 2 - count_ones):
                if len(zeros_indices) > 0:
                    idx = np.random.choice(zeros_indices)
                    h[idx] = 1
                    zeros_indices = np.delete(zeros_indices, np.where(zeros_indices == idx)[0])
        
        # Compute new C5
        new_c5 = compute_c5(h)
        delta = new_c5 - old_c5
        
        # Accept or reject based on simulated annealing
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            # Restore previous state
            h = old_h.copy()
            current_c5 = old_c5
            no_improvement_count += 1
        
        # Decrease temperature
        T *= decay
        
        # Periodic random perturbation
        if iter_count % perturbation_interval == 0:
            # Randomly reset a fraction of bits
            num_reset = int(n_points * perturbation_rate)
            reset_indices = np.random.choice(n_points, num_reset, replace=False)
            h[reset_indices] = np.random.randint(0, 2, num_reset)
            
            # Project to feasibility
            count_ones = np.sum(h)
            if count_ones > n_points // 2:
                ones_indices = np.where(h == 1)[0]
                for _ in range(count_ones - n_points // 2):
                    if len(ones_indices) > 0:
                        idx = np.random.choice(ones_indices)
                        h[idx] = 0
                        ones_indices = np.delete(ones_indices, np.where(ones_indices == idx)[0])
            elif count_ones < n_points // 2:
                zeros_indices = np.where(h == 0)[0]
                for _ in range(n_points // 2 - count_ones):
                    if len(zeros_indices) > 0:
                        idx = np.random.choice(zeros_indices)
                        h[idx] = 1
                        zeros_indices = np.delete(zeros_indices, np.where(zeros_indices == idx)[0])
            
            # Recompute C5
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Early termination on no improvement
        if no_improvement_count > max_no_improvement:
            break
            
    return (best_h.tolist(), best_c5, n_points)
