import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Initialize population for DE
    population_size = 50
    population = []
    fitness = []
    
    # Generate initial population
    for i in range(population_size):
        if i == 0 and 'initial_h_values' in globals():
            individual = h.copy()
        else:
            individual = np.random.uniform(0, 1, n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        corr = np.correlate(individual, 1 - individual, mode='full')
        fitness.append(np.max(corr) * dx)
    
    # Find best individual in initial population
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    no_improvement_count = 0
    max_no_improvement = 100
    
    # DE parameters
    F = 0.8
    CR = 0.9
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three different individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.copy(population[i])
            j_rand = np.random.randint(n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
            
            # Ensure constraint
            trial = trial * (n_points / 2) / np.sum(trial)
            
            # Evaluate
            corr = np.correlate(trial, 1 - trial, mode='full')
            c5 = np.max(corr) * dx
            
            # Selection
            if c5 < fitness[i]:
                new_population.append(trial)
                new_fitness.append(c5)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = trial.copy()
                    no_improvement_count = 0
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
                
        population = new_population
        fitness = new_fitness
        no_improvement_count += 1
        
        # Local refinement every 20 iterations
        if iteration % 20 == 0:
            # Refine best solution
            h_refine = best_h.copy()
            for refine_iter in range(10):
                corr = np.correlate(h_refine, 1 - h_refine, mode='full')
                c5 = np.max(corr) * dx
                
                # Compute gradient with finite differences
                grad = np.zeros_like(h_refine)
                eps = 1e-5
                for j in range(n_points):
                    h_plus = h_refine.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5) / eps
                
                # Adaptive learning rate
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h_refine - lr * grad
                
                # Clip and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Accept improvement
                corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr_new) * dx
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_refine = h_new.copy()
                else:
                    h_refine = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
