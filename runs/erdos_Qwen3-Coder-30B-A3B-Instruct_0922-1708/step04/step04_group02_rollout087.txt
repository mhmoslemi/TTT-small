import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 900  # 900 iterations with 500s budget
    inertia = 0.7
    phi_p = 2.0
    phi_g = 2.0
    local_search_freq = 50
    random_reinit_freq = 200
    rejuvenate_rate = 0.1
    max_local_steps = 10
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        particles.append(np.ones(n_points) * 0.5)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        
    # Add more random particles
    for _ in range(swarm_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Personal bests and global best
    pbest = [p.copy() for p in particles]
    pbest_fitness = [float('inf')] * len(particles)
    
    # Compute initial fitness
    for i, h in enumerate(particles):
        corr = correlate(h, 1 - h, mode='full')
        fitness = np.max(corr) * dx
        pbest_fitness[i] = fitness
        
    # Global best tracking
    best_idx = np.argmin(pbest_fitness)
    gbest = pbest[best_idx].copy()
    gbest_fitness = pbest_fitness[best_idx]
    
    # Optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update particle positions
        for i in range(len(particles)):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r_p = np.random.random(n_points)
            r_g = np.random.random(n_points)
            v_new = (inertia * v + 
                    phi_p * r_p * (pbest[i] - h) +
                    phi_g * r_g * (gbest - h))
            
            # Update position
            h_new = h + v_new
            
            # Repair constraint: clip and rescale to make sum = n_points/2
            h_new = np.clip(h_new, 0, 1)
            current_sum = np.sum(h_new)
            if abs(current_sum - n_points/2) > 1e-8:
                # Adjust for sum constraint
                diff = current_sum - n_points/2
                if diff > 0:  # Too large
                    # Reduce elements
                    indices = np.argsort(h_new)[::-1]
                    for idx in indices:
                        if diff <= 0:
                            break
                        decrease = min(diff, h_new[idx])
                        h_new[idx] -= decrease
                        diff -= decrease
                else:  # Too small
                    # Increase elements
                    indices = np.argsort(h_new)
                    for idx in indices:
                        if diff >= 0:
                            break
                        increase = min(-diff, 1 - h_new[idx])
                        h_new[idx] += increase
                        diff += increase
                        
            # Final scaling
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update particle
            particles[i] = h_new.copy()
            velocities[i] = v_new.copy()
            
        # Evaluate and update personal bests
        for i in range(len(particles)):
            h = particles[i]
            corr = correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest[i] = h.copy()
                
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest = h.copy()
                    
        # Local search
        if iter_num % local_search_freq == 0:
            # Get top 10% particles
            top_indices = np.argsort(pbest_fitness)[:int(0.1 * swarm_size)]
            for idx in top_indices:
                h = pbest[idx].copy()
                for _ in range(max_local_steps):
                    # Compute gradient
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    base_fitness = pbest_fitness[idx]
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                        grad[j] = (np.max(corr_plus) * dx - base_fitness) / eps
                    
                    # Update with gradient
                    h = h - 0.001 * grad
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    
                # Evaluate new solution
                corr_new = correlate(h, 1 - h, mode='full')
                fitness_new = np.max(corr_new) * dx
                
                if fitness_new < pbest_fitness[idx]:
                    pbest_fitness[idx] = fitness_new
                    pbest[idx] = h.copy()
                    if fitness_new < gbest_fitness:
                        gbest_fitness = fitness_new
                        gbest = h.copy()
                        
        # Rejuvenation
        if iter_num % random_reinit_freq == 0:
            num_rejuv = int(rejuvenate_rate * swarm_size)
            for _ in range(num_rejuv):
                idx = np.random.randint(swarm_size)
                h = np.random.random(n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[idx] = h.copy()
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                # Update personal best
                corr = correlate(h, 1 - h, mode='full')
                fitness = np.max(corr) * dx
                if fitness < pbest_fitness[idx]:
                    pbest_fitness[idx] = fitness
                    pbest[idx] = h.copy()
                    if fitness < gbest_fitness:
                        gbest_fitness = fitness
                        gbest = h.copy()
    
    # Final refinement on top solutions
    top_indices = np.argsort(pbest_fitness)[:5]
    best_h = gbest.copy()
    best_c5 = gbest_fitness
    
    # Try finishing with further gradient descent on best candidates
    for idx in top_indices:
        h = pbest[idx].copy()
        for _ in range(1000):  # More iterations here
            if time.time() - start_time > budget_s - 5:
                break
            grad = np.zeros(n_points)
            eps = 1e-4
            base_fitness = pbest_fitness[idx]
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                grad[j] = (np.max(corr_plus) * dx - base_fitness) / eps
            
            h = h - 0.0005 * grad  # Lower LR for more stable descent
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        corr_final = correlate(h, 1 - h, mode='full')
        fitness_final = np.max(corr_final) * dx
        if fitness_final < best_c5:
            best_c5 = fitness_final
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
