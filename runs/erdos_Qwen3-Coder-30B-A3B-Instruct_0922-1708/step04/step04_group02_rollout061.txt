import numpy as np
import time

def evaluate_h(h, n_points):
    """Evaluate the C5 value for given h"""
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def repair_h(h, n_points):
    """Repair h to satisfy constraints"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Adjust values to meet sum constraint
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        indices = np.argsort(h)
        for i in range(len(indices)):
            idx = indices[i]
            add = min(diff, 1 - h[idx])
            h[idx] += add
            diff -= add
            if diff <= 1e-10:
                break
    else:
        # Need to decrease values
        indices = np.argsort(h)[::-1]
        for i in range(len(indices)):
            idx = indices[i]
            subtract = min(-diff, h[idx])
            h[idx] -= subtract
            diff += subtract
            if diff >= -1e-10:
                break
    
    return h
