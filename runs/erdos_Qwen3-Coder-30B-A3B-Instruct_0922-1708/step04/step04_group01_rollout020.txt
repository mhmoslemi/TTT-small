import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Perform grid search on coarse version
    coarse_size = 30
    coarse_h = np.linspace(0, 1, coarse_size)
    
    # Grid search
    coarse_candidates = []
    for i in range(1, coarse_size):
        h_coarse = np.zeros(coarse_size)
        h_coarse[:i] = 1.0
        h_coarse = h_coarse * (coarse_size / 2) / np.sum(h_coarse)
        h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, coarse_size), h_coarse)
        correlation = np.correlate(h_upsampled, 1 - h_upsampled, mode='full')
        c5 = np.max(correlation) * dx
        coarse_candidates.append((h_upsampled, c5))
        
    coarse_candidates.sort(key=lambda x: x[1])
    top_candidates = coarse_candidates[:8]
    
    # Refine each candidate
    for candidate_h, _ in top_candidates:
        h = candidate_h.copy()
        
        # Use gradient descent refinement
        for _ in range(300):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Compute overlap
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Compute approximate gradient via finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            h_base = h.copy()
            
            for i in range(n_points):
                h_plus = h_base.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update h using gradient descent
            lr = 0.05
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
    
    # Final optimization using CVXPY for better results
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Objective is to minimize maximum correlation
        # We need to use an iterative approach or form a concrete optimization
        # Let's just do one final improvement with CVXPY if time allows
        if time.time() - start_time < budget_s - 2:
            # Approximate correlations appropriately
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (n_points, 1)))))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_cvx = h_vars.value
                if h_cvx is not None:
                    h_cvx = np.clip(h_cvx, 0, 1)
                    h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                    
                    correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                    c5_cvx = np.max(correlation) * dx
                    
                    if c5_cvx < best_c5:
                        best_h = h_cvx
                        best_c5 = c5_cvx
    except:
        pass
    
    # Final check of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
