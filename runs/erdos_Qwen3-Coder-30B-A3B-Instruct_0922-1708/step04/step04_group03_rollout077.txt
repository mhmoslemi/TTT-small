import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation with CVXPY
    # Minimize sum of all overlaps (approximation)
    if n_points > 10:
        # Use a simpler method for small arrays or when cvxpy isn't available
        h_cvx = h.copy()
    else:
        # For larger n_points, solve a relaxed convex problem
        try:
            h_cvx = np.zeros(n_points)
            # Relaxation: minimize the sum of all correlations
            h_vars = cp.Variable(n_points)
            
            # Constraint: sum == n_points/2
            constraints = [cp.sum(h_vars) == n_points / 2]
            
            # Bound constraints
            constraints += [h_vars >= 0, h_vars <= 1]
            
            # Simplified objective: sum of all h[i]*(1-h[i+k])
            # This is a rough approximation of the actual problem
            obj = cp.Minimize(cp.sum_squares(h_vars))  # dummy objective
            
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_cvx = np.array(h_vars.value).reshape(-1)
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
        except Exception:
            h_cvx = h.copy()
    
    # Use the convex solution as starting point for gradient descent
    h = h_cvx.copy()
    
    iteration = 0
    max_no_improvement = 50
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        # Gradient-based update
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        corr = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            improved = True
            
        h = h_new
        
        # Early stopping
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
    return (best_h.tolist(), best_c5, n_points)
