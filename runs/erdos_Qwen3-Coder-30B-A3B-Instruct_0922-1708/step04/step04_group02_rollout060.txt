import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = current_sum - target_sum
    if diff > 0:
        # Need to decrease some values
        sorted_indices = np.argsort(h)[::-1]  # Descending order
        for i in sorted_indices:
            if diff <= 0:
                break
            reduce = min(h[i], diff)
            h[i] -= reduce
            diff -= reduce
    else:
        # Need to increase some values
        sorted_indices = np.argsort(h)  # Ascending order
        for i in sorted_indices:
            if diff >= 0:
                break
            increase = min(1 - h[i], -diff)
            h[i] += increase
            diff += increase
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    if 'initial_h_values' in globals():

        n_points = 100

        x_orig = np.linspace(0, 2, len(h_init))
        x_target = np.linspace(0, 2, n_points)
        h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(100) * 0.5
        dx = 2.0 / 100
    
    swarm_size = 100
    n_points = len(h_init)
    dx = 2.0 / n_points
    
    h_swarm = np.array([h_init.copy() for _ in range(swarm_size)])
    v_swarm = (np.random.rand(swarm_size, n_points) - 0.5) * 0.2
    
    # Initialize personal bests and global best
    pbest = h_swarm.copy()
    pbest_c5 = np.array([evaluate_c5(h, dx) for h in h_swarm])
    
    gbest_idx = np.argmin(pbest_c5)
    gbest = h_swarm[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    iteration = 0
    stagnation_count = 0
    best_c5 = float('inf')
    best_h = None
    
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Evaluate all particles
        for i in range(swarm_size):
            h = h_swarm[i]
            c5 = evaluate_c5(h, dx)
            if c5 < pbest_c5[i]:
                pbest_c5[i] = c5
                pbest[i] = h.copy()
                if c5 < gbest_c5:
                    gbest_c5 = c5
                    gbest = h.copy()
                    stagnation_count = 0
        
        # Update global best
        current_best_idx = np.argmin(pbest_c5)
        if pbest_c5[current_best_idx] < best_c5:
            best_c5 = pbest_c5[current_best_idx]
            best_h = pbest[current_best_idx].copy()
        
        # Update stagnation counter
        stagnation_count += 1
        if stagnation_count > 50:
            # Reinitialize worst 10% of particles
            worst_indices = np.argsort(pbest_c5)[-swarm_size//10:]
            for idx in worst_indices:
                h_swarm[idx] = np.random.rand(n_points) * 0.5
                h_swarm[idx] = h_swarm[idx] * (n_points / 2) / np.sum(h_swarm[idx])
                v_swarm[idx] = (np.random.rand(n_points) - 0.5) * 0.2
            stagnation_count = 0
        
        # Update particles
        for i in range(swarm_size):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            v = w * v_swarm[i] + c1 * r1 * (pbest[i] - h_swarm[i]) + c2 * r2 * (gbest - h_swarm[i])
            h_swarm[i] = h_swarm[i] + v
            h_swarm[i] = repair_h(h_swarm[i], n_points)
            v_swarm[i] = v
            
        # Local search on top particles every 50 iterations
        if iteration % 50 == 0:
            top_indices = np.argsort(pbest_c5)[:swarm_size//10]
            for i in top_indices:
                h = h_swarm[i].copy()
                for _ in range(10):
                    grad = np.zeros(n_points)
                    eps = 1e-5
                    corr_base = np.correlate(h, 1 - h, mode='full')
                    c5_base = np.max(corr_base) * dx
                    for j in range(n_points):
                        h_pert = h.copy()
                        h_pert[j] += eps
                        h_pert = repair_h(h_pert, n_points)
                        corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
                        c5_pert = np.max(corr_pert) * dx
                        grad[j] = (c5_pert - c5_base) / eps
                    h -= 0.001 * grad
                    h = repair_h(h, n_points)
                h_swarm[i] = h.copy()
                corr_test = np.correlate(h, 1 - h, mode='full')
                c5_new = np.max(corr_test) * dx
                if c5_new < pbest_c5[i]:
                    pbest_c5[i] = c5_new
                    pbest[i] = h.copy()
        
        # Periodic refinement
        if iteration % 200 == 0:
            # Simple refinement strategy
            top_candidates = np.argsort(pbest_c5)[:10]
            if len(top_candidates) > 0:
                best_idx = top_candidates[0]
                h_refined = pbest[best_idx].copy()
                # Local gradient search refinement
                h_best = h_refined.copy()
                c5_best = evaluate_c5(h_refined, dx)
                for _ in range(20):
                    grad = np.zeros(n_points)
                    eps = 1e-5
                    corr_base = np.correlate(h_best, 1 - h_best, mode='full')
                    c5_base = np.max(corr_base) * dx
                    improve = False
                    for j in range(n_points):
                        h_pert = h_best.copy()
                        h_pert[j] += eps
                        h_pert = repair_h(h_pert, n_points)
                        corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
                        c5_pert = np.max(corr_pert) * dx
                        grad[j] = (c5_pert - c5_base) / eps
                        if c5_pert < c5_base:
                            improve = True
                    if not improve:
                        break
                    h_best -= 0.001 * grad
                    h_best = repair_h(h_best, n_points)
                    c5 = evaluate_c5(h_best, dx)
                    if c5 < c5_best:
                        c5_best = c5
                        h_refined = h_best.copy()
                for j in range(swarm_size//10):
                    idx = top_candidates[j]
                    h_swarm[idx] = h_refined.copy()
    
    # Final evaluation
    final_best_idx = np.argmin([evaluate_c5(h, dx) for h in pbest])
    final_h = pbest[final_best_idx]
    final_c5 = evaluate_c5(final_h, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
