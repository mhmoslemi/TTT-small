import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure normalization
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    n_particles = 20
    max_iter = 150
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    inertia_weight = w_start
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest_idx = 0
    gbest_val = float('inf')
    
    for i in range(n_particles):
        # Initialize position near h_init
        pos = h_init + np.random.normal(0, 0.05, n_points)
        pos = np.clip(pos, 0, 1)
        pos = pos * (n_points / 2) / np.sum(pos)
        particles.append(pos.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(pos.copy())
        val = np.max(np.correlate(pos, 1 - pos, mode='full')) * dx
        pbest_vals.append(val)
        if val < gbest_val:
            gbest_val = val
            gbest_idx = i
    
    iter_count = 0
    no_improvement_count = 0
    
    # PSO main loop
    while time.time() - start_time < budget_s - 5 and iter_count < max_iter:
        iter_count += 1
        inertia_weight = w_start - (w_start - w_end) * iter_count / max_iter
        
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (
                inertia_weight * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (pbest[gbest_idx] - particles[i])
            )
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            val = np.max(np.correlate(particles[i], 1 - particles[i], mode='full')) * dx
            
            # Update personal best
            if val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = val
                if val < gbest_val:
                    gbest_val = val
                    gbest_idx = i
                    no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Adapt swarm if no improvement
        if no_improvement_count > 50:
            n_particles = max(10, n_particles // 2)
            no_improvement_count = 0
            
            # Respawn particles
            for i in range(n_particles, 20):
                pos = h_init + np.random.normal(0, 0.1, n_points)
                pos = np.clip(pos, 0, 1)
                pos = pos * (n_points / 2) / np.sum(pos)
                particles.append(pos.copy())
                velocities.append(np.random.uniform(-0.1, 0.1, n_points))
                pbest.append(pos.copy())
                val = np.max(np.correlate(pos, 1 - pos, mode='full')) * dx
                pbest_vals.append(val)
                if val < gbest_val:
                    gbest_val = val
                    gbest_idx = len(particles) - 1
    
    # Refinement using line search on best particles
    top_candidates = np.argsort(pbest_vals)[:5]
    best_h = particles[gbest_idx].copy()
    best_c5 = gbest_val
    
    # Try simple line search for top candidates
    for idx in top_candidates:
        h = particles[idx].copy()
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            # Gradient approximation
            grad = np.zeros_like(h)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * dx
                
                grad[j] = (c5_plus - c5) / eps
            
            # Update with gradient
            lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-2)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Check improvement
            c5_new = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
            if c5_new < np.max(np.correlate(h, 1 - h, mode='full')) * dx:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                else:
                    break
            else:
                break
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
