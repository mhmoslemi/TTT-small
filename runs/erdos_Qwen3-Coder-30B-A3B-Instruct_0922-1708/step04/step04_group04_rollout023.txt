import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize h to satisfy constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    
    # Use CVXPY to solve the convex relaxation
    # Variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2, C >= 0]
    
    # For each lag k, we compute the correlation
    # This is constraint: sum h[i] * (1 - h[i+k]) <= C for all valid k
    for k in range(1, n_points):
        # Convolution for shift k
        corr = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
        constraints.append(corr <= C)
        
    # Objective
    objective = cp.Minimize(C)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with ECOS solver with a reasonable time limit
    try:
        prob.solve(solver=cp.ECOS, max_iters=100, feastol=1e-5, abstol=1e-5, reltol=1e-5)
    except:
        pass
    
    # Extract solution
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        h_sol = h_var.value
    else:
        h_sol = h
    
    # Project back to valid solution
    h_sol = np.clip(h_sol, 0, 1)
    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
    
    # Compute final C5
    correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
    c5_final = np.max(correlation) * dx
    
    # Check all solutions to ensure the best is returned
    return (h_sol.tolist(), c5_final, n_points)
