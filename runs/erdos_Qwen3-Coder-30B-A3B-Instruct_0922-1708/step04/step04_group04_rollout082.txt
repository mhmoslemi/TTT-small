import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Iterative convex optimization with line search approach
    max_iter = 100
    tolerance = 1e-5
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute current correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Solve convex relaxation using CVXPY
        h_old = h.copy()
        
        # Variables
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(h_var) == n_points / 2]
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Linearize the non-convex terms using current values
        # For each k, we have sum h[i] * (1 - h[i+k]) <= C
        # Linearize h[i+k] around h_old[i+k]
        # We get h[i] * (1 - h_old[i+k]) + h[i] * (h_old[i+k] - h[i+k]) <= C
        # Rearranging: h[i] - h[i] * h_old[i+k] + h[i] * h_old[i+k] - h[i] * h[i+k] <= C
        # Simplifies: h[i] - h[i] * h[i+k] <= C
        # Actually for cross-correlation we have: sum h[i] * (1 - h[i+k]) <= C
        
        # Alternate linearization: 
        # The constraint is sum_{i=1}^n h[i] * (1 - h[i+k]) <= C
        # ((1 - h_old[i+k]) * h[i]) + ((h_old[i+k] - h[i+k]) * h[i]) <= C
        # So L = (1 - h_old[i+k]) * h[i] + h_old[i+k] * h[i] - h[i] * h[i+k]
        # => L = h[i] - h[i] * h[i+k]
        # We want to enforce sum L <= C
        
        # In CVXPY we approximate:
        # For each k (i from 0 to n-k-1)
        # Let's compute a matrix constraint as: sum(h[i] * (1 - h_old[i+k])) + sum(h_old[i+k] * h[i]) - sum(h[i] * h[i+k]) <= C
        # But that's wrong, let's re-derive...
        
        # Constraint for correlation term (i+k) over shifted indices:
        for k in range(1, n_points):
            # Indices i=0 to n-k-1
            # We want sum_i h[i] * (1 - h[i+k]) <= C
            # Using majorization: h[i] * (1 - h_old[i+k]) + h_old[i+k] * h[i] - h[i] * h[i+k] <= C
            # Simplifying: h[i] - h[i] * h[i+k] <= C 
            # This is not right...
            
            # Right way to linearize:
            # This should be a scalar constraint for each k:
            # sum_{i=0}^{n-k-1} h[i] (1 - h[i+k]) <= C
            # Linearization:
            # sum_{i=0}^{n-k-1} h[i] - h[i] * h[i+k] <= C
            # sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i] * h[i+k] <= C
            # Maybe we linearize the second term using previous iterate:
            # sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i] * h_old[i+k] + sum_{i=0}^{n-k-1} h[i] * (h_old[i+k] - h[i+k]) <= C
            # sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i] * h_old[i+k] - sum_{i=0}^{n-k-1} h[i] * (h[i+k] - h_old[i+k]) <= C
            # sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i] * h_old[i+k] - sum_{i=0}^{n-k-1} h[i] * h[i+k] + sum_{i=0}^{n-k-1} h[i] * h_old[i+k] <= C
            
            # Simpler but more effective:
            # sum h[i] (1 - h_old[i+k])  <= C (and h_old is fixed for this linearization)
            # etc.
            
            # We approximate the summation term as sum h[i] - h[i] * h_old[i+k] for each k:
            # This gives: sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i] * h_old[i+k] <= C
            # Which is equivalent to sum_{i=0}^{n-k-1} (h[i] - h[i] * h_old[i+k]) <= C
            # Which is equivalent to sum_{i=0}^{n-k-1} h[i] * (1 - h_old[i+k]) <= C
            
            # This is better to enforce individually for each k!
            # We can approximate sum_{i} h[i] (1 - h[i+k]) <= C by:
            # sum_{i} h[i] - sum_{i} h[i] * h_old[i+k] <= C
            # => sum_{i} h[i] - sum_{i} h[i] * h_old[i+k] <= C
            
            # Use sparsely shifted terms with k in the range [1, n_points-1]
            # Slightly different for full correlation, so one constraint per k is fashioned:
            expr = 0
            for i in range(n_points - k):
                expr += h_var[i] * (1 - h_old[i + k])
                
            constraints.append(expr <= C)
                
        # Objective
        objective = cp.Minimize(C)
        
        # Problem formulation
        prob = cp.Problem(objective, constraints)
        
        try:
            # Try solving
            prob.solve(solver=cp.ECOS, abstol=1e-4, reltol=1e-4, feastol=1e-4)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                C_val = C.value
                
                # Line search 
                step = 0.5
                old_c5 = current_c5
                while step > 1e-8:
                    h_test = h_old + step * (h_new - h_old)
                    h_test = np.clip(h_test, 0, 1)
                    h_test = h_test * (n_points / 2) / np.sum(h_test)
                    corr_test = np.correlate(h_test, 1 - h_test, mode='full')
                    c5_test = np.max(corr_test) * dx
                    
                    if c5_test < old_c5:
                        h = h_test
                        break
                    step /= 2
                
                if step <= 1e-8:
                    # If line search failed, stabilize with current good solution
                    h = h_old.copy()
                    
            else:
                # If infeasible or other issue, fallback to previous
                h = h_old.copy()
                
        except:
            # If optimizer fails, use previous step or exit gracefully
            h = h_old.copy()
            
        # Convergence check
        if iter_num > 0 and np.linalg.norm(h - h_old) < tolerance:
            break
            
    # Final fix & validation
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if np.abs(h_sum - n_points / 2) > 1e-5:
        h = h * (n_points / 2) / h_sum

    # Compute final value of C5
    final_corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    # Final update of best if needed
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
