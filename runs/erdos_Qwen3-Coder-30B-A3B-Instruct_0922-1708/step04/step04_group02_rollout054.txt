from scipy.optimize import minimize
import time
import random
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Resolution levels to try
    resolutions = [50, 100, 200]
    best_h = None
    best_c5 = float('inf')
    best_n = 0

                h = np.interp(x_target, x_orig, h)
            h = h * (res / 2) / np.sum(h)
            c5 = evaluate_c5(h, res)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = res
    else:
        # Initialize with random feasible solutions
        for res in resolutions:
            if time.time() - start_time > budget_s - 10:
                break
            pop_size = min(50, max(10, res // 2))
            population = []
            for _ in range(pop_size):
                h = np.random.rand(res)
                h = h * (res / 2) / np.sum(h)
                population.append(h)
            # Simple evolutionary search
            for gen in range(max(50, res // 2)):
                if time.time() - start_time > budget_s - 5:
                    break
                # Evaluate
                fitness = []
                for h in population:
                    c5 = evaluate_c5(h, res)
                    fitness.append(c5)
                # Selection
                sorted_idx = np.argsort(fitness)
                top = sorted_idx[:pop_size // 2]
                population = [population[i] for i in top]
                # Crossover and mutation
                new_pop = deepcopy(population)
                for _ in range(pop_size - len(population)):
                    i1, i2 = np.random.choice(len(population), 2, replace=False)
                    parent1, parent2 = population[i1], population[i2]
                    # SBX crossover
                    alpha = 0.5
                    beta = 1 - alpha
                    child1 = alpha * parent1 + beta * parent2
                    child2 = beta * parent1 + alpha * parent2
                    # Mutate with probability 0.1
                    if np.random.rand() < 0.1:
                        idx = np.random.randint(len(child1))
                        child1[idx] += np.random.normal(0, 0.01)
                    if np.random.rand() < 0.1:
                        idx = np.random.randint(len(child2))
                        child2[idx] += np.random.normal(0, 0.01)
                    # Repair
                    child1 = np.clip(child1, 0, 1)
                    child1 = child1 * (res / 2) / np.sum(child1)
                    child2 = np.clip(child2, 0, 1)
                    child2 = child2 * (res / 2) / np.sum(child2)
                    new_pop.extend([child1, child2])
                population = new_pop[:pop_size]
            # Check best from this resolution
            for h in population:
                c5 = evaluate_c5(h, res)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = res
    
    # If no good result from evolutionary search, try gradient descent refinement
    if best_h is None or time.time() - start_time > budget_s - 5:
        if 'initial_h_values' in globals():

            h = h / np.sum(h) * (len(h) / 2)
        else:
            h = np.random.rand(100)
            h = h * 50 / np.sum(h)

    # Gradient-based refinement on best so far
    if best_h is not None:
        h_refined = best_h.copy()
        min_c5 = best_c5
        lr = 0.01
        iterations = 0
        max_iter = 1000
        while iterations < max_iter and time.time() - start_time < budget_s - 2:
            c5 = evaluate_c5(h_refined, len(h_refined))

                min_c5 = c5

                best_c5 = min_c5
                best_n = len(h_refined)

            eps = 1e-4
            grad = np.zeros_like(h_refined)

                grad[i] = (c5_plus - c5) / eps

            h_refined = np.clip(h_refined, 0, 1)

            lr *= 0.999
    
    # Final validation
    if best_h is not None:
        c5_final = evaluate_c5(best_h, best_n)
        if c5_final < best_c5:
            best_c5 = c5_final
    
    return (best_h.tolist() if best_h is not None else np.zeros(100).tolist(), 
            best_c5, 
            best_n if best_n > 0 else 100)

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5
