def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points

            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)

    else:
        h = np.ones(n_points) * 0.5
        
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def compute_c5(h_vals):
        # Compute correlation with shifted version
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    # Phase 1: Convex relaxation to get a good starting point
    if 'initial_h_values' not in globals():
        # Use CVXPY for convex relaxation
        h_var = cp.Variable(n_points)
        # Objective: minimize sum of all overlaps, which approximates max overlap
        # But we'll use a different approach: approximate the max overlap with a convex relaxation
        # For small-scale problems, we can still use a relaxed version
        # Actually, we'll first use a grid search to get a reasonable starting point
        
        # Simple grid search for multiple good solutions
        best_x = h.copy()
        best_val = compute_c5(h)
        
        for _ in range(20):
            # Perturb
            h_trial = np.random.uniform(0, 1, n_points)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            val = compute_c5(h_trial)
            if val < best_val:
                best_val = val
                best_x = h_trial.copy()
                
        h = best_x
    
    # Phase 2: Local gradient descent refinement with projected updates
    def objective(h_vals):
        return compute_c5(h_vals)
    
    # Gradient computation via finite differences
    def compute_gradient(h_vals, eps=1e-5):
        grad = np.zeros_like(h_vals)
        base_val = objective(h_vals)
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            plus_val = objective(h_plus)
            grad[i] = (plus_val - base_val) / eps
        return grad

    max_iter = 500
    patience = 50
    no_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        grad = compute_gradient(h)
        lr = 0.01 / (1 + iteration * 1e-3)
        h_new = h - lr * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement >= patience:
            break
            
        h = h_new

    # Final validation
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (best_h.tolist(), best_c5, n_points)
