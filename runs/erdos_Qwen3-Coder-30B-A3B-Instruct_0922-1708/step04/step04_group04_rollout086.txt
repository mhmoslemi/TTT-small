import numpy as np
import time
import heapq
from scipy import signal

def overlap_computation(h, n_points):
    """Compute overlaps for all shifts using FFT convolution"""
    # h is already normalized to sum to n_points/2
    # 1-h is the complementary function
    # The overlap for shift k is sum_{i=0}^{n-|k|-1} h[i] * (1-h[i+|k|])
    # Using convolution with flipped kernel
    h_padded = np.pad(h, (0, n_points-1), mode='constant')
    one_minus_h_padded = np.pad(1-h, (0, n_points-1), mode='constant')
    conv = signal.convolve(h_padded, one_minus_h_padded, mode='valid')
    # Taking only the positive shifts (k = 1 to n-1)
    # Middle shift (k=0) is handled separately
    overlaps = conv[n_points-1:]
    return overlaps

def prioritize_shifts(overlaps):
    """Return priority queue of shifts sorted by overlap magnitude (max-heap)"""
    # max-heap using negative values
    pq = [(-abs(overlap), i) for i, overlap in enumerate(overlaps) if overlap != 0]
    heapq.heapify(pq)
    return pq

def apply_adjustments(h, k, window_width, n_points):
    """Apply local adjustments to h to reduce overlap for shift k"""
    # k is negative for right shifts, positive for left shifts
    # k in [1, n-1], represents shift right by k positions
    if k <= 0:
        return h.copy()
    
    # Find where the overlap region is: i from 0 to n-k
    # Need to consider h[i] * h[i+k] in overlapping area
    # Update the intervals affected
    # Bias toward reducing strengthening h where overlaps are triggered
    # Apply changes in a window around influential points
    
    # Correct interpretation: for minimum overlap, make h more sparse on positively overlapping regions
    # But we're minimizing max of sum over overlaps, so affect the most problematic shift
    # Let's find where the maximum overlap occurs and adjust that
    h_new = h.copy()
    # This is heuristic: if we know which shift is causing problems, 
    # and the overlap is at indices i, i+k, we can locally change h
    # The gradient aspects defy standard calculus; mimic gradient descent behavior 
    # by reducing overlap manually in certain regions
    
    window_start = max(0, k - window_width)
    window_end = min(n_points, k + window_width + 1)
    
    # Simple update policy: reduce h at affected indices
    for i in range(window_start, window_end):
        if i+k < n_points:
            # For shift k, elements i and i+k interact
            # Setting h[i] = max(h[i] - adjustment, 0)
            # But keep sum(h) = n/2
            
            # Make a small decrease where overlap was most substantial
            # But within boundaries to prevent outliers
            h_new[i] = np.clip(h_new[i] - 0.01, 0, 1)
            
    # Re-normalize to maintain constraints  
    h_new = h_new * (n_points / 2) / np.sum(h_new)
    return h_new

def random_perturb(h, window_width, scale):
    """Apply small random perturbations to h"""
    h_new = h.copy()
    n_points = len(h)
    # Select a few random indices
    num_perts = max(1, n_points // 20)
    indices = np.random.choice(n_points, num_perts, replace=False)
    
    for idx in indices:
        # Apply perturbation to local region
        w = min(window_width, n_points - idx)
        for i in range(max(0, idx - w), min(n_points, idx + w)):
            h_new[i] += np.random.normal(0, scale)
    
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    return h_new

def validate_h(h, n_points):
    """Ensure h ∈ [0,1] and sum(h) ≈ n/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, n_points):
    """Compute the maximum overlap for the current h"""
    overlaps = overlap_computation(h, n_points)
    # Size of overlaps is n-1, corresponding to shifts k = 1 to n-1
    max_overlap = np.max(overlaps)
    return max_overlap

        best_h = h.copy()
    
    # Optimization loop
    max_iterations = 10000
    iteration = 0
    min_improvement = 1e-6
    window_width = 5
    perturb_scale = 0.02

        iteration += 1

            # Perturb the optimization step
