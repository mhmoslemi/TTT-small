import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')

    def objective(h):
        # Compute correlation and return max
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def gradient(h):
        # Use finite differences for gradient estimation
        eps = 1e-4
        grad = np.zeros_like(h)
        base_obj = objective(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            grad[i] = (objective(h_plus) - base_obj) / eps
        return grad

    # Trust region optimization
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Trust region optimization using scipy minimize
        result = minimize(
            objective,
            h,
            method='TRF',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint_sum},
            jac=gradient,
            options={'maxiter': 50}
        )
        
        if result.success:
            h = result.x
            # Project to feasible region
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Check if better solution
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            break

    return (best_h.tolist(), best_c5, n_points)
