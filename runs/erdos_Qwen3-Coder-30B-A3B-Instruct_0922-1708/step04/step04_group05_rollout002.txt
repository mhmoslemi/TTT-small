import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h = h_init.copy()
    
    # PSO parameters
    n_particles = 20
    max_iter = 200
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    
    # Initialize swarm
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    for i in range(n_particles):
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    personal_best = particles.copy()
    personal_best_fitness = np.full(n_particles, np.inf)
    global_best_idx = 0
    global_best_fitness = np.inf
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            # Clip and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            fitness = np.max(corr) * dx
            
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best[i] = particles[i].copy()
                
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best_idx = i
                
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            particles[i] += velocities[i]
            
            # Boundary handling
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
        # Add noise occasionally
        if iteration % 20 == 0:
            for i in range(n_particles):
                noise = np.random.normal(0, 0.005, n_points)
                particles[i] += noise
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Refinement stage for top particles
    top_k = 5
    top_indices = np.argsort(personal_best_fitness)[:top_k]
    
    for idx in top_indices:
        h_candidate = personal_best[idx].copy()
        for _ in range(200):  # Small number of iterations for refinement
            if time.time() - start_time > budget_s - 5:
                break
                
            # Gradient approximation
            grad = np.zeros(n_points)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h_candidate.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5 = np.max(corr) * dx
                
                grad[j] = (c5_plus - c5) / eps
            
            lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-2)
            h_candidate -= lr * grad
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Early exit if improvement is small
            new_c5 = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
            if new_c5 >= c5:
                break
    
    # Final evaluation
    final_corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (h_candidate.tolist(), final_c5, n_points)
