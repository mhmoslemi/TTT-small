import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # LP relaxation approach
    # Variables
    h_vars = cp.Variable(n_points)
    C5_var = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2,
        C5_var >= 0
    ]
    
    # Add auxiliary variables and constraints for correlation terms
    # For each shift k, we have:
    # sum(h(x) * (1 - h(x + k))) * dx <= C5
    # This is equivalent to:
    # sum(h(x) - h(x) * h(x + k)) * dx <= C5
    
    # Since we can't directly handle the product in CVXPY, we'll use a more direct approach
    # using a gradient-based method that works well with the objective
    
    # So we'll fall back to the simpler optimization approach but make it more efficient
    best_h = h.copy()
    best_c5 = float('inf')
    
    # For a smaller, more efficient exploration, we'll use gradient descent directly
    max_iter = 1000
    lr = 0.01
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final refinement using scipy minimize
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx  # negative because we minimize
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Re-initialize with best solution
    h = best_h.copy()
    
    # More refined optimization using scipy
    constraints = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        result = minimize(objective, h, method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': 100})
        h = np.clip(result.x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    except:
        pass
    
    # Final evaluation
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    correlation_final = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(correlation_final) * dx
    
    return (h_final.tolist(), final_c5, n_points)
