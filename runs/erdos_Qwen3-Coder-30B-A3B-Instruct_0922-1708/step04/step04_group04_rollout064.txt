import numpy as np
import time
from scipy import signal
import heapq

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Dynamic shift prioritization with adaptive adjustments
    def compute_overlap(h):
        # Compute overlap for all shifts
        conv = signal.correlate(h, 1 - h, mode='full')
        shifts = np.arange(-len(h) + 1, len(h))
        overlaps = conv * dx
        return shifts, overlaps
    
    def prioritize_shifts(shifts, overlaps):
        # Create max heap with (negative_overlap, shift) for max heap behavior
        heap = [(-overlaps[i], shifts[i]) for i in range(len(shifts)) if shifts[i] != 0]
        heapq.heapify(heap)
        return heap
    
    def apply_adjustments(h, k, w=5):
        h_new = h.copy()
        if k > 0:  # h_i * (1 - h_{i+k})
            # Adjust h in positions where overlap occurs
            for i in range(min(w, len(h) - k)):
                # Modify positions that influence the overlap with shift k
                if i + k < len(h):
                    h_new[i] = max(0, min(1, h_new[i] - 0.01))
                    h_new[i + k] = max(0, min(1, h_new[i + k] + 0.01))
        elif k < 0:  # h_{i+k} * (1 - h_i)
            k_abs = -k
            for i in range(min(w, len(h) - k_abs)):
                if i + k_abs < len(h):
                    h_new[i] = max(0, min(1, h_new[i] + 0.01))
                    h_new[i + k_abs] = max(0, min(1, h_new[i + k_abs] - 0.01))
        else:
            pass  # No meaningful shift
            
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def random_perturb(h, w, scale=0.02):
        h_new = h.copy()
        start = np.random.randint(0, len(h) - w)
        for i in range(w):
            if start + i < len(h):
                delta = np.random.normal(0, scale)
                h_new[start + i] = max(0, min(1, h_new[start + i] + delta))
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Main loop
    convergence_stall = 0
    max_convergence_stall = 100
    window_width = 5
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute overlaps and current max overlap
        shifts, overlaps = compute_overlap(h)
        shift_heap = prioritize_shifts(shifts, overlaps)
        
        # Extract the shift with highest overlap (in absolute value)
        if len(shift_heap) == 0:
            break
        max_overlap, k = heapq.heappop(shift_heap)
        max_overlap = -max_overlap  # Convert back
        
        # Apply localized adjustments
        h_local = apply_adjustments(h, k, window_width)
        
        # Random perturbation to escape local minima
        h_perturbed = random_perturb(h_local, window_width, 0.02)
        
        # Compute overlap of new solution
        shifts_new, overlaps_new = compute_overlap(h_perturbed)
        new_c5 = np.max(np.abs(overlaps_new))
        
        # Accept improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_perturbed.copy()
            convergence_stall = 0
        else:
            convergence_stall += 1
            # Increase window width if convergence stall
            if convergence_stall > 10:
                window_width = min(window_width + 1, 20)
                convergence_stall = 0
        
        # Update h with good solution
        h = h_perturbed
        
        # Periodically refresh the priority queue
        if iteration % 50 == 0:
            shifts, overlaps = compute_overlap(h)
            shift_heap = prioritize_shifts(shifts, overlaps)
    
    # Final validation and return
    shifts_final, overlaps_final = compute_overlap(best_h)
    final_c5 = np.max(np.abs(overlaps_final))
    
    return (best_h.tolist(), final_c5, n_points)
