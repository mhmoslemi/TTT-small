import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init / np.sum(h_init) * 100  # Scale to sum to 100
    else:
        h_init = np.ones(100) * 0.5
    
    # Sparse representation with 10 segments
    n_segments = 10
    n_points = 100
    dx = 2.0 / n_points
    
    def sparse_to_dense(pos, h_vals):
        """Convert sparse representation to dense array."""
        h = np.zeros(n_points)
        for i in range(n_segments):
            start = int(pos[i] / 2.0 * n_points)
            end = int(pos[i+1] / 2.0 * n_points)
            if start < end:
                h[start:end] = h_vals[i]
        return h
    
    def dense_to_sparse(h_dense):
        """Convert dense array to sparse representation."""
        # Find segment boundaries
        pos = [0.0]
        heights = []
        prev_height = h_dense[0]
        for i in range(1, n_points):
            if h_dense[i] != prev_height:
                pos.append(i * dx)
                heights.append(prev_height)
                prev_height = h_dense[i]
        heights.append(prev_height)
        pos.append(2.0)
        return np.array(pos), np.array(heights)
    
    def compute_c5_sparse(pos, h_vals):
        """Compute C5 for sparse representation."""
        h = sparse_to_dense(pos, h_vals)
        # Convert to full correlation and return max
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def optimize_sparse():
        # Initialize sparse representation
        pos = np.linspace(0, 2, n_segments + 1)
        h_vals = np.ones(n_segments) * (1.0 / n_segments)
        
        # Define optimization variables
        var_list = []
        var_list.extend(pos[1:-1])  # Exclude endpoints
        var_list.extend(h_vals)
        
        def objective(var_list):
            pos_vals = np.array([0.0] + var_list[:n_segments-1] + [2.0])
            h_vals = np.array(var_list[n_segments-1:])
            
            # Ensure strict ordering of positions
            if not np.all(np.diff(pos_vals) > 1e-8):
                return 1e10
                
            # Ensure h_vals are within [0,1] 
            if np.any(h_vals < 0) or np.any(h_vals > 1):
                return 1e10
            
            # Ensure integral constraint
            integral = 0.0
            for i in range(n_segments):
                integral += (pos_vals[i+1] - pos_vals[i]) * h_vals[i]
            if abs(integral - 1.0) > 1e-3:
                return 1e10
                
            return compute_c5_sparse(pos_vals, h_vals)
        
        # Use a simple local search with adaptive perturbation
        best_obj = float('inf')
        best_pos = pos.copy()
        best_h = h_vals.copy()
        
        # Multi-start with different initializations
        for iter_num in range(20):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Random initialization
            pos_rand = np.random.rand(n_segments - 1) * 2
            pos_rand = np.sort(pos_rand)

            pos_rand = np.append(pos_rand, [2.0])
            h_rand = np.random.rand(n_segments)
            h_rand = h_rand / np.sum(h_rand)  # Normalize sum to 1
            h_rand = h_rand * (1.0 / n_segments)  # Scale to have average value

            current_pos = pos_rand.copy()
            current_h = h_rand.copy()
            
            # Gradient descent-style search
            for iter_inner in range(1000):
                if time.time() - start_time > budget_s - 5:
                    break

                grad_pos = np.zeros(n_segments - 1)
                grad_h = np.zeros(n_segments)

                    pos_pert = current_pos.copy()
                    pos_pert[i] += eps
                    if i > 0 and pos_pert[i] <= pos_pert[i-1]:
                        pos_pert[i] = pos_pert[i-1] + 1e-8
                    if i < n_segments - 2 and pos_pert[i] >= pos_pert[i+1]:
                        pos_pert[i] = pos_pert[i+1] - 1e-8
                        
                    obj_plus = objective(list(pos_pert) + list(current_h))
                    grad_pos[i] = (obj_plus - objective(list(current_pos) + list(current_h))) / eps
                
                for i in range(n_segments):
                    h_pert = current_h.copy()
                    h_pert[i] += eps
                    if h_pert[i] < 0 or h_pert[i] > 1:
                        continue
                        
                    obj_plus = objective(list(current_pos) + list(h_pert))
                    grad_h[i] = (obj_plus - objective(list(current_pos) + list(current_h))) / eps
                
                # Update variables with scaled gradient
                lr = 1e-3
                new_pos = current_pos - lr * grad_pos
                new_h = current_h - lr * grad_h
                
                # Project back to constraints
                new_pos = np.clip(new_pos, 0, 2)
                for i in range(1, len(new_pos)):
                    new_pos[i] = max(new_pos[i], new_pos[i-1] + 1e-8)
                new_pos[0] = 0.0
                new_pos[-1] = 2.0
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h / np.sum(new_h) * (n_segments / 2.0)
                
                current_pos = new_pos
                current_h = new_h
                
                # Evaluate with updated variables
                obj_val = objective(list(current_pos) + list(current_h))
                if obj_val < best_obj:
                    best_obj = obj_val
                    best_pos = current_pos.copy()
                    best_h = current_h.copy()
        
        return best_pos, best_h
    
    # Run the optimization
    try:
        pos_best, h_best = optimize_sparse()
        # Convert to dense representation
        h_dense = sparse_to_dense(pos_best, h_best)
        c5 = compute_c5_sparse(pos_best, h_best)

        h_dense = h_init.copy()

    # Ensure we have a valid result within budget
    if time.time() - start_time < budget_s - 5:
        # Try an additional refinement
        h_dense = np.clip(h_dense, 0, 1)
        h_dense = h_dense / np.sum(h_dense) * 100  # rescale to sum to 100
        corr = np.correlate(h_dense, 1 - h_dense, mode='full')
        c5 = np.max(corr) * dx
    
    return (h_dense.tolist(), c5, n_points)
