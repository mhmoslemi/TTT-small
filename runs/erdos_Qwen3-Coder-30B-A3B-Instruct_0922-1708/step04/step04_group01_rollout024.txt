import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or create starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # Initial uniform distribution
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Coarse grid search phase
    coarse_points = 20
    candidates = []
    
    # Generate coarse candidates with different patterns
    for _ in range(20):
        # Random distribution with sum constraint
        h_coarse = np.random.rand(coarse_points)
        h_coarse = h_coarse / np.sum(h_coarse) * (coarse_points / 2)
        
        # Interpolate to fine grid
        h_fine = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, coarse_points), 
                           h_coarse)
        
        # Compute C5 for this candidate
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full')
        c5 = np.max(correlation) * dx
        candidates.append((c5, h_fine.copy()))
    
    # Sort by C5 values
    candidates.sort(key=lambda x: x[0])
    top_candidates = candidates[:10]  # Top 10 candidates
    
    # Refinement phase
    for _, h_base in top_candidates:
        h_refined = h_base.copy()
        
        # Run several optimization steps with CVXPY for each candidate
        for _ in range(10):
            try:
                # Create variable
                h_vars = cp.Variable(n_points)
                
                # Constraints
                constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
                
                # Objective: min max_k integral
                # This problem is challenging, we approximate with empirical approach
                
                # Use a simplified convex relaxation with gradient-based method
                def objective_func(h_vals):
                    h_vals = np.clip(h_vals, 0, 1)
                    h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
                    correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
                    return np.max(correlation) * dx
                
                # Simple gradient descent approach
                lr = 0.01
                h_revised = h_refined.copy()
                for grad_iter in range(50):
                    if time.time() - start_time > budget_s - 5:
                        break
                        
                    grad = np.zeros_like(h_revised)
                    eps = 1e-3
                    for i in range(n_points):
                        h_plus = h_revised.copy()
                        h_plus[i] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[i] = (c5_plus - objective_func(h_revised)) / eps
                    
                    h_new = h_revised - lr * grad
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    h_revised = h_new
                    
                # Final evaluation
                c5_final = objective_func(h_revised)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_revised.copy()
                    
                # Check time again
                if time.time() - start_time > budget_s - 1:
                    break
                    
            except Exception as e:
                continue  # Skip problematic candidates
    
    # Final check of best solution
    if time.time() - start_time < budget_s - 1:
        # Try one final convex optimization
        try:
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            # Simplified objective that can be handled by convex solver
            # For a true optimization we solve the dual or use approximate methods
            # Optimization might be difficult due to nonconvexity
            # Just ensure it's feasible and above our current best
            
            c5_eval = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
            if c5_eval < best_c5:
                best_c5 = c5_eval
        except:
            pass
    
    # Return final results
    return (best_h.tolist(), best_c5, n_points)
