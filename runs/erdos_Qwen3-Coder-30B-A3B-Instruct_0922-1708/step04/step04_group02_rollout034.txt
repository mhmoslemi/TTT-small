import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS, CVXOPT, GLPK

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h based on provided parent if available
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a diverse set of initializations
        h_init_list = []
        # Random initialization
        h1 = np.random.uniform(0, 1, n_points)
        h1 = h1 / np.sum(h1) * (n_points / 2)
        h_init_list.append(h1)
        
        # Block initialization
        h2 = np.zeros(n_points)
        h2[:n_points//2] = 1.0
        h2 = h2 / np.sum(h2) * (n_points / 2)
        h_init_list.append(h2)
        
        # Sine-based initialization
        x = np.linspace(0, 2, n_points)
        h3 = 0.5 + 0.5 * np.sin(np.pi * x)
        h3 = h3 / np.sum(h3) * (n_points / 2)
        h_init_list.append(h3)
        
        # Select best among initializations
        best_h = None
        best_c5 = float('inf')
        for h_init in h_init_list:
            corr = np.correlate(h_init, 1 - h_init, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_init.copy()
        h = best_h
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Timing control
    start_time = time.time()
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Convex relaxation stage
        try:
            h_relaxed = cp.Variable(n_points)
            constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points / 2]
            
            # We'll approximate by using bounds on the overlap
            C = cp.Variable()
            lag_range = range(1, n_points)
            for k in lag_range:
                # Constrain the lag correlation sum
                corr_k = 0
                for i in range(n_points - k):
                    corr_k += h_relaxed[i] * (1 - h_relaxed[i + k])
                constraints.append(corr_k <= C / dx)
            
            prob = cp.Problem(cp.Minimize(C), constraints)
            prob.solve(solver=CVXOPT, verbose=False)
            z_relaxed = h_relaxed.value
            
            if z_relaxed is not None:
                h = np.clip(z_relaxed, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass
        
        # Integer rounding stage based on the relaxed solution
        # The approach is to compute a threshold and round the relaxed solution
        # Adjust to match the desired sum constraint
        
        if iter_num % 10 == 0:
            # Simple heuristic based on the median
            threshold = np.median(h)
            h_binary = (h > threshold).astype(float)
            sum_diff = np.sum(h_binary) - n_points / 2
            
            if abs(sum_diff) > 1e-6:
                # Adjust binary solution to satisfy sum constraint
                # Find indices of 1's and 0's to flip
                indices_1 = np.where(h_binary > 0.5)[0]
                indices_0 = np.where(h_binary < 0.5)[0]
                
                # Pick which indices to flip
                if sum_diff > 0:
                    # Need to flip 0's to 1's to decrease diff
                    target_flip = int(sum_diff)
                    indices = indices_0[:target_flip]
                    if len(indices) > 0:
                        h_binary[indices] = 1.0
                else:
                    # Need to flip 1's to 0's to decrease diff
                    target_flip = int(-sum_diff)
                    indices = indices_1[:target_flip]
                    if len(indices) > 0:
                        h_binary[indices] = 0.0
                        
            # Rescale to match sum constraint
            h = h_binary
            h = h * (n_points / 2) / np.sum(h)
            
        # Simple neighborhood search
        if iter_num % 5 == 0:
            h_best = h.copy()
            c5_best = c5
            
            # Try small perturbations
            epsilon = 0.01
            for _ in range(20):
                h_test = h.copy()
                # Perturb a few random elements
                indices = np.random.choice(n_points, size=min(10, n_points//10), replace=False)
                # Perturb each by either +epsilon or -epsilon
                for i in indices:
                    if np.random.rand() > 0.5:
                        h_test[i] = min(1.0, h_test[i] + epsilon)
                    else:
                        h_test[i] = max(0.0, h_test[i] - epsilon)
                
                # Rescale to meet integral constraint
                h_test = h_test * (n_points / 2) / np.sum(h_test)
                
                # Compute objective
                corr = np.correlate(h_test, 1 - h_test, mode='full')
                c5_test = np.max(corr) * dx
                
                if c5_test < c5_best:
                    c5_best = c5_test
                    h_best = h_test.copy()
                    
            h = h_best
            c5 = c5_best

    return (best_h.tolist(), best_c5, n_points)
