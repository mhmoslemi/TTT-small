import numpy as np

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population with binary vectors that sum to n_points/2"""
    population = []
    
    if initial_h is not None:
        # Convert to binary
        h = np.array(initial_h)
        h_binary = (h > 0.5).astype(int)
        # Adjust to have exactly n_points/2 ones
        current_sum = np.sum(h_binary)
        target = n_points // 2
        if current_sum < target:
            # Add ones
            zero_indices = np.where(h_binary == 0)[0]
            for i in range(target - current_sum):
                if i < len(zero_indices):
                    h_binary[zero_indices[i]] = 1
        elif current_sum > target:
            # Remove ones
            one_indices = np.where(h_binary == 1)[0]
            for i in range(current_sum - target):
                if i < len(one_indices):
                    h_binary[one_indices[i]] = 0
        population.append(h_binary)
    
    # Add random individuals
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h_binary = (h > 0.5).astype(int)
        # Adjust to sum to exactly n_points/2
        current_sum = np.sum(h_binary)
        target = n_points // 2
        if current_sum < target:
            zero_indices = np.where(h_binary == 0)[0]
            for i in range(target - current_sum):
                if i < len(zero_indices):
                    h_binary[zero_indices[i]] = 1
        elif current_sum > target:
            one_indices = np.where(h_binary == 1)[0]
            for i in range(current_sum - target):
                if i < len(one_indices):
                    h_binary[one_indices[i]] = 0
        population.append(h_binary)
    
    return np.array(population)

def mutate_individual(h, mutation_rate=0.05):
    """Mutate an individual by randomly flipping bits"""
    h_new = h.copy()
    for i in range(len(h_new)):
        if np.random.random() < mutation_rate:
            h_new[i] = 1 - h_new[i]
    
    # Ensure sum is maintained
    current_sum = np.sum(h_new)
    target = len(h_new) // 2
    if current_sum < target:
        zero_indices = np.where(h_new == 0)[0]
        for i in range(target - current_sum):
            if i < len(zero_indices):
                h_new[zero_indices[i]] = 1
    elif current_sum > target:
        one_indices = np.where(h_new == 1)[0]
        for i in range(current_sum - target):
            if i < len(one_indices):
                h_new[one_indices[i]] = 0
    
    return h_new

def crossover_parents(p1, p2):
    """Perform crossover between two parents"""
    n_points = len(p1)
    crossover_point = np.random.randint(1, n_points-1)
    
    h_new = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    
    # Adjust to sum to n_points//2
    current_sum = np.sum(h_new)
    target = n_points // 2
    if current_sum < target:
        zero_indices = np.where(h_new == 0)[0]
        for i in range(target - current_sum):
            if i < len(zero_indices):
                h_new[zero_indices[i]] = 1
    elif current_sum > target:
        one_indices = np.where(h_new == 1)[0]
        for i in range(current_sum - target):
            if i < len(one_indices):
                h_new[one_indices[i]] = 0
    
    return h_new

def refine_with_convex(h, n_points, dx):
    """Refine a binary solution by solving a convex relaxation"""
    # Use CVXPY to solve a convex relaxation
    h_vars = cp.Variable(n_points)
    obj = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (n_points, 1)))))
    cons = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
    
    try:
        prob = cp.Problem(obj, cons)
        prob.solve(solver=cp.CBC, verbose=False)
        if prob.status == cp.OPTIMAL and h_vars.value is not None:
            h_refined = np.clip(h_vars.value, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            return h_refined
    except:
        pass
    return h

    dx = 2.0 / n_points
    pop_size = 100
    generations = 100
    mutation_rate = 0.05
    initial_h = None

    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = population[sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Selection: keep top 10%
        elite_size = max(1, pop_size // 10)
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        # Crossover: generate 50% new offspring
        while len(new_population) < pop_size:
            parent1, parent2 = random.sample(elite, 2)
            child = crossover_parents(parent1, parent2)
            new_population.append(child)
            
        # Mutation: mutate 30% of new population
        mutation_count = max(1, int(0.3 * pop_size))
        for _ in range(mutation_count):
            idx = np.random.randint(len(new_population))
            new_population[idx] = mutate_individual(new_population[idx], mutation_rate)
            
        population = np.array(new_population)
    
    # Final refinement with convex optimization
    if best_h is not None:
        h_refined = refine_with_convex(best_h, n_points, dx)
        c5_refined = evaluate_h(h_refined, dx)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined
            
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to simple approach
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        h = h / np.sum(h) * (n_points / 2)
        c5 = evaluate_h(h, dx)
        return (h.tolist(), c5, n_points)
