import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Try multiple initialization strategies
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Strategy 1: Gradient-based local search with different initialization
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Try different initializations
    for trial in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        if trial == 0:
            h_init = h.copy()
        elif trial == 1:
            # Use a step function
            h_init = np.zeros(n_points)
            mid = n_points // 2
            h_init[:mid] = 1.0
            h_init = h_init * (n_points / 2) / np.sum(h_init)
        elif trial == 2:
            # Use a sinusoidal function
            x = np.linspace(0, 2, n_points)
            h_init = 0.5 + 0.5 * np.sin(np.pi * x)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
        elif trial == 3:
            # Use random with strong bias
            h_init = np.random.uniform(0.3, 0.7, n_points)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
        else:  # trial == 4
            # Use modified version of parent
            h_init = np.array(initial_h_values) * 0.95
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_new = np.linspace(0, 2, n_points)
                h_init = np.interp(x_new, x_orig, h_init)
            # Apply some smoothing
            h_init = np.convolve(h_init, np.ones(5)/5, mode='same')
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            
        # Local optimization with L-BFGS
        result = minimize(objective, h_init, method='L-BFGS-B', bounds=[(0, 1)] * n_points, 
                          options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            c5 = objective(result.x)
            if c5 < best_c5:
                best_h = result.x.copy()
                best_c5 = c5
    
    # Further improvement using simulated annealing 
    # Only do if we have time
    if time.time() - start_time < budget_s - 5:
        def sa_objective(h_vec):
            h = np.clip(h_vec, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlation = np.correlate(h, 1 - h, mode='full')
            return np.max(correlation) * dx
            
        # Simple simulated annealing
        current_h = best_h.copy()
        current_c5 = sa_objective(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        
        temp = 0.1
        decay = 0.999
        iterations = 0
        max_iter = 20000
        
        while iterations < max_iter and time.time() - start_time < budget_s - 5:
            # Perturb solution
            neighbor_h = current_h + np.random.normal(0, 0.01, n_points)
            # Enforce bounds and normalization
            neighbor_h = np.clip(neighbor_h, 0, 1)
            neighbor_h = neighbor_h * (n_points / 2) / np.sum(neighbor_h)
            
            neighbor_c5 = sa_objective(neighbor_h)
            
            # Accept or reject
            if neighbor_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - neighbor_c5) / temp):
                current_h = neighbor_h.copy()
                current_c5 = neighbor_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
            temp *= decay
            iterations += 1
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (final_h.tolist(), float(final_c5), n_points)
