import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h from parent or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Try random initializations
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Relaxation stage
    def solve_convex_relaxation(h_init):
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Objective variable
        C = cp.Variable()
        
        # Compute overlaps for all lags
        # Because h is cyclic we'll use index wraparound
        max_lag = n_points - 1
        for k in range(1, max_lag):
            # This is computationally expensive to do for all k so 
            # we'll do a sampled version with more focus on small lags
            if k > 50: continue  # Limit for efficiency
            # For k=1 to n_points-1 we compute: 
            # sum_i h[i] * (1 - h[i+k])
            ovs = 0
            # zip with delayed vector
            for i in range(n_points-k):
                ovs += h_vars[i] * (1 - h_vars[i+k])
            # wraparound
            for i in range(n_points-k, n_points):
                ovs += h_vars[i] * (1 - h_vars[i-(n_points-k)])
            constraints.append(ovs <= C * dx**-1)
        
        # No objective function; we will optimize for minimizing C
        prob = cp.Problem(cp.Minimize(C), constraints)
        try:
            # Use solver with tight tolerance
            prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-8, reltol=1e-8)
            if prob.status == 'optimal':
                return h_vars.value
        except Exception:
            pass
        return h_init
    
    # Integer rounding stage
    def binary_round(h_relaxed):
        # Use median thresholding
        threshold = np.median(h_relaxed)
        h_binary = (h_relaxed > threshold).astype(float)
        # Adjust to meet sum constraint
        actual_sum = np.sum(h_binary)
        target_sum = n_points / 2
        diff = int(actual_sum - target_sum)
        
        if diff > 0:
            # Need to reduce by diff
            # Find indices with high value that can be flipped
            mask = h_relaxed > threshold
            indices = np.where(mask)[0]
            to_flip = indices[:diff]
            for i in to_flip:
                h_binary[i] = 0
        elif diff < 0:
            # Need to increase by |diff|
            mask = h_relaxed <= threshold
            indices = np.where(mask)[0]
            to_flip = indices[:abs(diff)]
            for i in to_flip:
                h_binary[i] = 1
                
        # Final adjusted sum
        final_sum = np.sum(h_binary)
        if abs(final_sum - target_sum) > 1e-6:
            # Ensure correct sum by adjusting
            h_binary = h_binary * target_sum / final_sum
            
        return h_binary
    
    # Evaluate C5
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if iteration % 2 == 0:
            # Apply convex relaxation
            h_relaxed = solve_convex_relaxation(h)
            if h_relaxed is not None:
                h_binary = binary_round(h_relaxed)
                h = h_binary
        else:
            # Try neighborhood search: small random perturbations
            h_new = h.copy()
            # Perturb some indices
            indices = np.random.choice(n_points, size=min(n_points//3, 20), replace=False)
            for i in indices:
                pert = np.random.uniform(-0.05, 0.05)
                h_new[i] += pert
            # Clip and normalize
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
        if iteration > 1000: break  # avoid infinite loops

    # Final refinement using gradient estimation
    h_refined = best_h.copy()
    old_c5 = best_c5
    
    def evaluate_grad(h_vals, eps=1e-4):
        base_c5 = compute_c5(h_vals)
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            h_test = h_vals.copy()
            h_test[i] += eps
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            test_c5 = compute_c5(h_test)
            grad[i] = (test_c5 - base_c5) / eps
        return grad
    
    # Gradient descent steps on a few selected points
    min_grad_step = 1e-3
    for _ in range(20):
        if time.time() - start_time > budget_s - 0.1:
            break
        grad = evaluate_grad(h_refined)
        h_new = h_refined - min_grad_step * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        if compute_c5(h_new) < compute_c5(h_refined):
            h_refined = h_new

    final_c5 = compute_c5(h_refined)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_refined

    return (best_h.tolist(), best_c5, n_points)
