import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or generate random
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Generate initial candidates
        candidates = []
        for _ in range(3):
            h_temp = np.random.uniform(0, 1, n_points)
            h_temp = h_temp / np.sum(h_temp) * (n_points / 2)
            candidates.append(h_temp)
        
        # Select best among candidates
        best_c5 = float('inf')
        best_h = None
        for h_temp in candidates:
            correlation = np.correlate(h_temp, 1 - h_temp, mode='full')
            c5 = np.max(correlation) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_temp.copy()
        h = best_h

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Relaxation stage using CVXPY
    if time.time() - start_time < budget_s * 0.4:
        try:
            # Define variables
            h_vars = cp.Variable(n_points)
            C = cp.Variable()
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # For all possible lags
            for k in range(1, n_points):
                # Compute correlation
                corr = 0
                for i in range(n_points - k):
                    corr += h_vars[i] * (1 - h_vars[i + k])
                constraints.append(corr <= C * dx ** (-1))
                
            # Objective
            objective = cp.Minimize(C)
            prob = cp.Problem(objective, constraints)
            
            # Solve with CVXOPT
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed = h_vars.value
                h_relaxed = np.array(h_relaxed)
                h_relaxed = np.clip(h_relaxed, 0, 1)
                h_relaxed = h_relaxed * (n_points / 2) / np.sum(h_relaxed)
                
                # Binary rounding with thresholding
                threshold = np.median(h_relaxed)
                h_binary = (h_relaxed > threshold).astype(float)
                
                # Adjust sum to be n_points / 2
                current_sum = np.sum(h_binary)
                diff = current_sum - n_points / 2
                
                if diff > 0:  # Too many 1s
                    # Find 1s to flip to 0
                    idxs = np.where(h_binary == 1)[0]
                    flip_indices = np.random.choice(idxs, int(diff), replace=False)
                    h_binary[flip_indices] = 0
                elif diff < 0:  # Too many 0s
                    # Find 0s to flip to 1
                    idxs = np.where(h_binary == 0)[0]
                    flip_indices = np.random.choice(idxs, int(-diff), replace=False)
                    h_binary[flip_indices] = 1
                
                h = h_binary
        except:
            pass
    
    # Evaluate
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    best_c5 = c5
    best_h = h.copy()
    
    # Post-processing neighborhood search (find better local solutions)
    max_neighbors = min(50, n_points * 2)
    max_iterations = min(300, int((budget_s - (time.time() - start_time)) / 0.1))
    
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
        # Generate neighbor
        h_new = h.copy()
        center = np.random.randint(0, n_points)
        change = np.random.normal(0, 0.02)
        # Possibility of small changes
        for j in range(max(1, n_points // 5)):
            idx = (center + j) % n_points
            h_new[idx] = np.clip(h_new[idx] + change, 0, 1)
        # Rescale to sum = n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Final refinement
    refines = min(10, int((budget_s - (time.time() - start_time)) / 0.5))
    for i in range(refines):
        if time.time() - start_time > budget_s - 1:
            break
        h_test = best_h.copy()
        # Small adjustment
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.01, 0.01)
        h_test[idx] = np.clip(h_test[idx] + delta, 0, 1)
        # Renormalize
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        
        corr_test = np.correlate(h_test, 1 - h_test, mode='full')
        c5_test = np.max(corr_test) * dx
        
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
    
    return (best_h.tolist(), best_c5, n_points)
