import numpy as np
import time

def project_feasible(h, n_points):
    """Project h to satisfy 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def compute_grad(h, eps=1e-5):
    """Compute finite difference gradient of C5 with respect to h"""
    n_points = len(h)
    grad = np.zeros(n_points)
    base_c5 = compute_c5(h)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        plus_c5 = compute_c5(h_plus)
        grad[i] = (plus_c5 - base_c5) / eps
        
    return grad

def line_search(h, dir, max_iter=20):
    """Golden section line search for step size"""
    phi = (1 + np.sqrt(5)) / 2
    res = (phi - 1) / phi
    
    a, b = 0.0, 1.0
    v1 = a + res * (b - a)
    v2 = b - res * (b - a)
    
    for _ in range(max_iter):
        h1 = project_feasible(h + v1 * dir, len(h))
        h2 = project_feasible(h + v2 * dir, len(h))
        
        if compute_c5(h1) < compute_c5(h2):
            b = v2
            v2 = v1
            v1 = a + res * (b - a)
        else:
            a = v1
            v1 = v2
            v2 = b - res * (b - a)
    
    return (a + b) / 2.0

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    h = project_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    
    # Coarse grid search
    n_coarse = 100
    coarse_h_values = []
    
    for i in range(n_coarse):
        if time.time() - start_time > budget_s - 100:
            break
        h_coarse = np.random.rand(n_points)
        h_coarse = project_feasible(h_coarse, n_points)
        c5 = compute_c5(h_coarse)
        coarse_h_values.append((c5, h_coarse))
    
    coarse_h_values.sort(key=lambda x: x[0])
    top_candidates = [h for _, h in coarse_h_values[:n_coarse//10]]
    
    # Refine top candidates
    for i, h_refine in enumerate(top_candidates):
        if time.time() - start_time > budget_s - 50:
            break
            
        h = h_refine.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 2:
                break
            grad = compute_grad(h)
            lr = 0.001  # Fixed learning rate for simplicity
            dir = -grad
            step_size = line_search(h, dir)
            h = project_feasible(h + step_size * dir, n_points)
            
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            else:
                # Add small noise to escape local minima
                noise = np.random.normal(0, 0.001, n_points)
                h = project_feasible(h + noise, n_points)
    
    # Final refinement
    h = best_h.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s - 2:
            break
        grad = compute_grad(h)
        lr = 0.001
        dir = -grad
        step_size = line_search(h, dir)
        h = project_feasible(h + step_size * dir, n_points)
        
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Add small noise
            noise = np.random.normal(0, 0.001, n_points)
            h = project_feasible(h + noise, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
