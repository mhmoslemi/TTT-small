import numpy as np
import time
from scipy.signal import correlate
import heapq

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Rescale to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Validation function
    def validate_h(h_vec):
        return np.all(h_vec >= 0) and np.all(h_vec <= 1) and abs(np.sum(h_vec) - n_points / 2) < 1e-5
    
    # Overlap computation
    def compute_overlap(h_vec):
        # Compute all overlaps
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Priority queue for shifts by overlap (min-heap sim)
    def init_priority_queue(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        shifts = np.arange(-n_points + 1, n_points)
        overlaps = corr * dx
        # Build list of (overlap, shift)
        pairs = [(overlaps[i], shifts[i]) for i in range(len(shifts)) if shifts[i] != 0]
        # Create min heap by negating overlaps
        heap = [(-overlap, shift) for overlap, shift in pairs]
        heapq.heapify(heap)
        return heap
    
    # Fast adjust
    def adjust_local(h_vec, k, window=5):
        h_new = h_vec.copy()
        if k > 0:
            # h[i] * (1 - h[i+k])
            start = 0
            end = min(n_points - k, n_points)
            indices = np.arange(start, end)
            # Decrease h where overlaps are high
            for i in indices:
                h_new[i] = max(0, h_new[i] - 0.01 * (1 - 2 * h_new[i]))
            # Increase h where complement overlaps
            indices = np.arange(max(0, k), min(n_points, n_points + k))
            for i in indices:
                j = i - k
                if j >= 0:
                    h_new[j] = min(1, h_new[j] + 0.01 * (1 - 2 * h_new[j]))
        elif k < 0:
            # h[i] * (1 - h[i+k])
            k_abs = -k
            start = 0
            end = min(n_points - k_abs, n_points)
            indices = np.arange(start, end)
            for i in indices:
                h_new[i] = max(0, h_new[i] - 0.01 * (1 - 2 * h_new[i]))
            indices = np.arange(max(0, k_abs), min(n_points, n_points + k_abs))
            for i in indices:
                j = i - k_abs
                if j >= 0:
                    h_new[j] = min(1, h_new[j] + 0.01 * (1 - 2 * h_new[j]))
        else:
            return h_vec  # skip k=0

        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return np.clip(h_new, 0, 1)
    
    def random_perturb(h_vec, w=5, scale=0.02):
        h_new = h_vec.copy()
        # Randomly perturb around randomly chosen position
        pos = np.random.randint(0, n_points)
        start = max(0, pos - w)
        end = min(n_points, pos + w)
        for i in range(start, end):
            pert = np.random.normal(0, scale)
            h_new[i] += pert
        # Normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return np.clip(h_new, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_overlap(h)
    
    # Initialize priority queue
    pqueue = init_priority_queue(h)
    
    # For adaptive parameters
    adaptation_counter = 0
    last_improvement = 0
    window = 5
    scale = 0.02
    
    start_time = time.time()
    
    # Main optimization loop
    max_iter = 10000
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Check if we should recompute pq
        if it % 50 == 0:
            pqueue = init_priority_queue(h)
        
        # Get max overlap shift
        if pqueue:
            max_overlap_neg, shift = heapq.heappop(pqueue)
            max_overlap = -max_overlap_neg
            
            # Adjust
            h_adj = adjust_local(h, shift, window)
            c5_adj = compute_overlap(h_adj)
            
            # Check if better than current
            if c5_adj < best_c5:
                best_h = h_adj.copy()
                best_c5 = c5_adj
                last_improvement = it
            else:
                # Use perturbation path
                h_pert = random_perturb(h, window, scale)
                c5_pert = compute_overlap(h_pert)
                
                if c5_pert < best_c5:
                    best_h = h_pert.copy()
                    best_c5 = c5_pert
                    last_improvement = it
                else:
                    h_adj = h  # return to original
        else:
            # If no shift left, random mutation
            h = random_perturb(h, window, scale)
        
        # Update with some final adjustments or switching
        h = adjust_local(h, shift, window) if 'shift' in locals() else h
        
        # Re-evaluate and update
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Mistake fix: update C5 after adjustment
        c5_new = compute_overlap(h)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
            last_improvement = it
        
        # Adapt parameters if needed
        if it - last_improvement > 10:
            window = min(20, window + 1)
            scale *= 2.0
            last_improvement = it
            adaptation_counter += 1
            
        # Check convergence
        if it - last_improvement > 200:
            break
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
