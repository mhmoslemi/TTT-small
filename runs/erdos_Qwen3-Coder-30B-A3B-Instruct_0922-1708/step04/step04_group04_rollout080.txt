import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = float('inf')

    start_time = time.time()
    max_iter = 100
    tol = 1e-5
    step_size = 0.5
    backtracking_factor = 0.5
    solver_tolerance = 1e-4

    # Precompute indices
    indices = np.arange(n_points)
    # For each shift k, we need to compute correlations
    # But we work with a convex approximation of the non-convex problem

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Create convex approximation
        # Variables
        h_var = cp.Variable(n_points)
        C = cp.Variable()

        # Constraints
        constraints = [cp.sum(h_var) == n_points / 2]

        for k in range(1, n_points):
            # This handles the linearized version of h_i * (1 - h_{i+k})
            # We reconstruct by staying in the linearized form for each shift
            # For simplicity, we may define a full constraint per k
            # But we'll optimize per shift.
            # Replace h_{i+k} by h_cur_{i+k} in objective, and compute linearized terms
            h_k = h_var[k:] if k > 0 else h_var
            h_i = h_var[:n_points-k] if k > 0 else h_var
            linearized_sum = cp.sum(cp.multiply(h_i, 1 - h_var[k:]))
            # However, to make it linear in arguments, we use the old h value
            h_with_k_old = h[k:] if k > 0 else h
            h_with_i_old = h[:n_points-k] if k > 0 else h
            # For h_i * (1 - h_{i+k}) = h_i - h_i * h_{i+k}
            # Linearization of h_i * h_{i+k} with fixed h_{i+k}^old:
            # Bound = h_i * h_{i+k}^old
            # Hence: h_i * (1 - h_{i+k}) <= h_i - h_i * h_{i+k}^old
            h_linear = h_with_i_old * h_with_k_old
            constraint = cp.sum(cp.multiply(h_i, 1 - h_var[k:])) <= C + 1e-10
            constraints.append(constraint)

        # Objective
        objective = cp.Minimize(C)

        # Solve problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False, eps=solver_tolerance)
            new_h = h_var.value
        except Exception:
            # Fallback to gradient instead of convex optimization
            new_h = h - step_size * compute_gradient(h, dx)

        # Clip and rescale
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)

        # Compute line search
        t = 1.0
        best_new_h = new_h
        best_new_c5 = float('inf')
        for _ in range(10):  # only 10 line search steps max
            candidate_h = (1 - t) * h + t * new_h
            candidate_h = np.clip(candidate_h, 0, 1)
            candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)

            corr = np.correlate(candidate_h, 1 - candidate_h, mode='full')
            candidate_c5 = np.max(corr) * dx

            if candidate_c5 < best_new_c5:
                best_new_c5 = candidate_c5
                best_new_h = candidate_h
            t *= backtracking_factor
            if t < 1e-5:
                break

        h = best_new_h

    # Final check if we improved the best
    correlation = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(correlation) * dx

    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)

def compute_gradient(h, dx):
    n_points = len(h)
    # Use finite differences 
    grad = np.zeros_like(h)
    eps = 1e-4
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
        
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        
        grad[i] = (c5_plus - np.max(np.correlate(h, 1-h, mode='full')) * dx) / eps
    return grad
