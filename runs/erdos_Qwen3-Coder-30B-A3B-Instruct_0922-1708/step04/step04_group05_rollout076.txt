import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize integral constraint: sum(h) * dx = 1
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    n_particles = 20
    max_iter = 200
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    v_max = 0.1
    
    # Initialize PSO swarm
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    for i in range(n_particles):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    velocities = np.random.uniform(-v_max, v_max, (n_particles, n_points))
    personal_best = particles.copy()
    personal_best_fitness = np.full(n_particles, np.inf)
    global_best_idx = 0
    global_best_fitness = np.inf
    global_best = particles[0].copy()
    
    # PSO optimization loop
    start_time = time.time()
    iter_count = 0
    
    # If constraint violation is small, use line search improvement
    def optimize_with_line_search(h, max_iter_refine=50):
        def objective(v):
            h_v = np.clip(v, 0, 1)
            h_v = h_v * (n_points / 2) / np.sum(h_v)
            corr = np.correlate(h_v, 1 - h_v, mode='full')
            return np.max(corr) * dx
        
        def gradient(v):
            h_v = np.clip(v, 0, 1)
            h_v = h_v * (n_points / 2) / np.sum(h_v)
            corr = np.correlate(h_v, 1 - h_v, mode='full')
            max_k = np.argmax(corr)
            grad = np.zeros_like(v)
            eps = 1e-6
            for i in range(n_points):
                v_plus = v.copy()
                v_plus[i] += eps
                v_plus = np.clip(v_plus, 0, 1)
                v_plus = v_plus * (n_points / 2) / np.sum(v_plus)
                corr_plus = np.correlate(v_plus, 1 - v_plus, mode='full')
                grad[i] = (corr_plus[max_k] - corr[max_k]) / eps
            return grad
        
        if time.time() - start_time > budget_s - 10:
            return h
        
        try:
            for _ in range(max_iter_refine):
                if time.time() - start_time > budget_s - 10:
                    break
                grad = gradient(h)
                h -= 0.01 * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass
        return h
    
    # PSO iteration
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * iter / max_iter
        
        for i in range(n_particles):
            # Clip positions to bounds and enforce integral constraint
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            fitness = np.max(corr) * dx
            
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
            
            if fitness < global_best_fitness:
                global_best = particles[i].copy()
                global_best_fitness = fitness
                global_best_idx = i
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            particles[i] += velocities[i]
            
            # Clamp again to enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Final refinement with line search
    best_h = global_best.copy()
    for _ in range(3):
        if time.time() - start_time > budget_s - 5:
            break
        improved_h = optimize_with_line_search(best_h)
        corr = np.correlate(improved_h, 1 - improved_h, mode='full')
        improved_c5 = np.max(corr) * dx
        if improved_c5 < np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx:
            best_h = improved_h
        else:
            break
    
    # Final evaluation
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
