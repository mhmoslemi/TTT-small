import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Grid Search at Coarse Resolution
    coarse_points = 20
    coarse_dx = 2.0 / coarse_points
    coarse_h = np.zeros(coarse_points)
    coarse_h[:coarse_points//2] = 2.0
    coarse_h = coarse_h / np.sum(coarse_h) * (coarse_points / 2)
    
    # Sample some coarse candidates
    candidates = []
    
    # Use a few hand-crafted candidates as starting points
    if 'initial_h_values' in globals():
        # Convert initial_h_values to coarse resolution first
        h_orig = np.array(initial_h_values)
        coarse_h_from_parent = np.zeros(coarse_points)
        for i in range(coarse_points):
            start_idx = int(i * 2.0 / coarse_points * len(h_orig))
            end_idx = int((i + 1) * 2.0 / coarse_points * len(h_orig))
            coarse_h_from_parent[i] = np.mean(h_orig[start_idx:end_idx])
        candidates.append(coarse_h_from_parent)
        
        # Normalize the parent
        coarse_h_from_parent = coarse_h_from_parent / np.sum(coarse_h_from_parent) * (coarse_points / 2)
        candidates.append(coarse_h_from_parent)
    
    # Add a few random candidates
    for i in range(5):
        h_random = np.random.rand(coarse_points)
        h_random = h_random / np.sum(h_random) * (coarse_points / 2)
        candidates.append(h_random)
    
    # Evaluate candidates at coarse resolution
    best_coarse = float('inf')
    best_coarse_h = None
    coarse_results = []
    
    for h_c in candidates:
        h_c = np.clip(h_c, 0, 1)
        h_c = h_c / np.sum(h_c) * (coarse_points / 2)
        correlation = np.correlate(h_c, 1 - h_c, mode='full')
        c5 = np.max(correlation) * (2.0 / coarse_points)
        
        coarse_results.append((c5, h_c))
        if c5 < best_coarse:
            best_coarse = c5
            best_coarse_h = h_c
    
    # Sort coarse candidates by performance
    coarse_results.sort(key=lambda x: x[0])
    
    # Phase 2: Refine the best candidates at high resolution
    for i, (c5_coarse, h_coarse) in enumerate(coarse_results):
        if i > 5: # Process top few only
            break
        # Upsample to fine resolution
        h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, coarse_points), 
                               h_coarse)
        h_upsampled = h_upsampled / np.sum(h_upsampled) * (n_points / 2)
        
        # Refine using gradient descent and constraint handling
        h_refined = h_upsampled.copy()
        steps = 500
        lr = 0.01
        
        for step in range(steps):
            if time.time() - start_time > budget_s - 3:
                break
                
            # Compute correlation and gradient approximation
            correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5_refined = np.max(correlation) * dx
            
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
            
            # Finite differences for gradient estimation
            eps = 1e-4
            grad = np.zeros_like(h_refined)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - c5_refined) / eps
            
            # Update with gradient
            h_new = h_refined - lr * grad
            
            # Project back to constraint set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_refined = h_new
    
    # Final check using CVX if time permits
    if time.time() - start_time < budget_s - 3:
        try:
            h_vars = cp.Variable(n_points)
            constraints = [
                h_vars >= 0,
                h_vars <= 1,
                cp.sum(h_vars) == n_points / 2
            ]
            # Objective function (approximate as maximizing a norm or similar)
            # Here we will try to optimize using quadratic programming approach
            objective = cp.Minimize(cp.max(cp.conv(h_vars, 1 - h_vars)))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL and prob.value < best_c5:
                h_cvx = h_vars.value
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
        except:
            pass

    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
