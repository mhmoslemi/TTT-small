import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a simple block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    grid_resolution = 20
    grid_n_points = grid_resolution
    
    # Phase 1: Grid search with lower resolution
    if grid_n_points < n_points:
        grid_h = np.linspace(0, 1, grid_n_points)
        if grid_n_points > 100:
            grid_n_points = 100
        # Generate grid points for exploration
        grid_points = []
        # Create a simple structured grid
        for i in range(grid_n_points):
            point = np.zeros(grid_n_points)
            point[i] = 1.0
            grid_points.append(point)
            
        # Also try a uniform distribution
        uniform_point = np.ones(grid_n_points) * (0.5 * grid_n_points / grid_n_points)
        grid_points.append(uniform_point)
        
        # Explore some random points
        for _ in range(20):
            temp_h = np.random.rand(grid_n_points)
            temp_h = temp_h * (grid_n_points / 2) / np.sum(temp_h)
            grid_points.append(temp_h)
        
        # evaluate grid points
        for g_point in grid_points:
            if np.sum(g_point) < 1e-6:
                continue
            # rescale to correct integral
            g_point = g_point * (grid_n_points / 2) / np.sum(g_point)
            # upsample
            upsampled = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, grid_n_points), g_point)
            upsampled = np.clip(upsampled, 0, 1)
            upsampled = upsampled * (n_points / 2) / np.sum(upsampled)
            correlation = np.correlate(upsampled, 1 - upsampled, mode='full')
            c5 = np.max(correlation) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = upsampled.copy()
    
    # Phase 2: Refinement of top candidates with fine resolution
    
    # Use the best found so far and refine it
    h = best_h
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    h = h.copy()
    
    # Refinement iterations
    max_iterations = 1000
    lr = 0.01
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlation) * dx
        
        # Compute approximate gradient
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5_current) / eps
        
        # Update h
        h_new = h - lr * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update best if improvement
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final CVX optimization
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        # Direct/maximization formulation: minimize C5
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (n_points, 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
