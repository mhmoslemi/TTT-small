import numpy as np
import time

def project_feasible(h, n_points):
    """Project h to satisfy 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points) * 0.5
    return h

def compute_c5(h):
    """Compute C5 = max_k int h(x)(1 - h(x+k)) dx"""
    correlation_full = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation_full) * (2.0 / len(h))

def finite_diff_grad(h, eps=1e-4):
    """Compute gradient of C5 using finite differences"""
    n_points = len(h)
    grad = np.zeros(n_points)
    base_c5 = compute_c5(h)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        c5_plus = compute_c5(h_plus)
        grad[i] = (c5_plus - base_c5) / eps
    
    return grad

def golden_section_search(f, a, b, tol=1e-5, max_iter=50):
    """Golden section search for step size"""
    phi = (1 + np.sqrt(5)) / 2
    resphi = 2 - phi
    
    # Initialize
    x1 = a + resphi * (b - a)
    x2 = a + (1 - resphi) * (b - a)
    f1 = f(x1)
    f2 = f(x2)
    
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if f1 < f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = a + resphi * (b - a)
            f1 = f(x1)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + (1 - resphi) * (b - a)
            f2 = f(x2)
    
    return (a + b) / 2

def line_search(h, grad, n_points, max_step=0.01):
    """Line search along gradient direction"""
    def phi(alpha):
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n_points)
        return compute_c5(h_new)
    
    # Find bracketing interval
    a, b = 0.0, max_step
    fa, fb = phi(a), phi(b)
    
    if fa <= fb:
        # Move backward
        while fa > fb:
            a, b = max(0, b - 0.001), b
            fa, fb = phi(a), phi(b)
    else:
        # Move forward
        while fa > fb:
            a, b = b, 2 * b
            fa, fb = phi(a), phi(b)
    
    # Golden section search
    alpha = golden_section_search(phi, a, b)
    return alpha

            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    h_init = project_feasible(h_init, n_points)
    h_best = h_init.copy()
    c5_best = compute_c5(h_best)
    
    start_time = time.time()
    
    # Early exit if time is nearly up
    if time.time() - start_time > budget_s - 1:
        return (h_best.tolist(), c5_best, n_points)

    lr = 0.001
    eps = 1e-6
    
    for iter_num in range(10000):
        if time.time() - start_time > budget_s - 1:

        # Compute gradient
        grad = finite_diff_grad(h_best)
        
        # Line search step size
        alpha = line_search(h_best, grad, n_points, lr)
        
        # Update
        h_new = h_best - alpha * grad
        h_new = project_feasible(h_new, n_points)
        
        # Evaluate
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()

            lr = min(0.1, lr * 1.05)
