import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation
    # Minimize sum of all overlaps (convex relaxation)
    h_relaxed = h.copy()
    
    # Create variables for convex optimization
    h_vars = cp.Variable(n_points)
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2
    ]
    
    # Objective: minimize sum of overlaps
    # We'll compute this approximately for the convex solver
    # For now just use the current solution to initialize the relaxed one
    
    # Phase 2: Projected gradient descent
    def evaluate_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(h_vec, eps=1e-5):
        grad = np.zeros_like(h_vec)
        base_c5 = evaluate_c5(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Initialize with relaxed solution or existing one
    h = h.copy()
    correlation = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(correlation) * dx
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
        
    patience = 0
    max_patience = 100
    lr = 0.01
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update with projected gradient descent
        h_new = h - lr * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Adaptive learning rate
        if patience > 10:
            lr *= 0.9
            patience = 0
            
        if patience > max_patience:
            break
    
    # Final check
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
