import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
        dx = 2.0 / n_points
        h = initial_h / np.sum(initial_h) * (n_points / 2)
    else:
        n_points = 100
        dx = 2.0 / n_points
        h = np.ones(n_points) * 0.5  # default uniform distribution
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Grid Search with coarse resolution to identify promising areas
    if time.time() - start_time < budget_s * 0.2:
        coarse_points = 20
        coarse_dx = 2.0 / coarse_points
        best_candidates = []
        
        # Sample grid points
        for i in range(1000):  # limit iterations
            if time.time() - start_time > budget_s * 0.2:
                break
            h_coarse = np.random.rand(coarse_points)
            h_coarse = h_coarse / np.sum(h_coarse) * (coarse_points / 2)
            
            # upsample to full resolution
            h_full = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, coarse_points), h_coarse)
            h_full = np.clip(h_full, 0, 1)
            h_full = h_full / np.sum(h_full) * (n_points / 2)
            
            correlation = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_full.copy()
            
            best_candidates.append((c5, h_full.copy()))
            best_candidates.sort(key=lambda x: x[0])
            best_candidates = best_candidates[:10]  # keep top 10
            
        # refine best candidates from grid
        for c5, h_candidate in best_candidates:
            if time.time() - start_time > budget_s - 10:
                break
                
            h_refined = h_candidate.copy()
            for _ in range(500):
                if time.time() - start_time > budget_s - 10:
                    break
                grad = np.zeros_like(h_refined)
                eps = 1e-4
                
                # compute gradient
                for i in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[i] = (c5_plus - c5) / eps
                
                # update
                lr = 0.01
                h_new = h_refined - lr * grad
                
                # project to feasible set
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                h_refined = h_new
                
                # recompute objective
                correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
                c5 = np.max(correlation) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_refined.copy()
    
    # Phase 2: Refine using optimization
    if time.time() - start_time < budget_s - 10:
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(correlation) * dx
        
        # Use direct optimization to refine current best
        h_refined = best_h.copy()
        for _ in range(1000):
            if time.time() - start_time > budget_s - 10:
                break
                
            # compute gradient
            grad = np.zeros_like(h_refined)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - best_c5) / eps
            
            # gradient step
            lr = 0.01
            h_new = h_refined - lr * grad
            
            # projection
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_refined = h_new
            
            # recompute objective
            correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5 = np.max(correlation) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_refined.copy()

    # Final check with CVXPY if time allows
    if time.time() - start_time < budget_s - 5:
        try:
            h_vars = cp.Variable(n_points)
            constraints = [
                h_vars >= 0,
                h_vars <= 1,
                cp.sum(h_vars) == n_points / 2
            ]
            # Use a quadratic form optimization
            # This is a more complex but potentially better constraint
            corr_dots = []
            for k in range(n_points):
                # Manually compute overlap terms, approximate with subset of lags
                if k == 0:
                    continue
                corr_dots.append(-cp.sum(cp.multiply(
                    h_vars, cp.roll(1 - h_vars, k)
                )))
                
            # approximate optimization by maximizing any overlap
            objective = cp.Minimize(cp.maximum(*corr_dots))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_cvx = h_vars.value
                if h_cvx is not None:
                    h_cvx = np.clip(h_cvx, 0, 1)
                    h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                    
                    correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                    c5_cvx = np.max(correlation) * dx
                    
                    if c5_cvx < best_c5:
                        best_h = h_cvx
                        best_c5 = c5_cvx
        except:
            pass

    # Final verification
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
