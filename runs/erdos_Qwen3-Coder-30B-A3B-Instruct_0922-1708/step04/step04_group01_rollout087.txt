import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    # Convert to binary approximation
    h_binary = (h > 0.5).astype(float)
    h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
    
    best_h = h_binary.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Evolutionary algorithm
    population_size = 100
    generations = 100
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h_temp = np.random.rand(n_points)
        h_temp = h_temp / np.sum(h_temp) * (n_points / 2)
        h_binary = (h_temp > 0.5).astype(float)
        h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
        population.append(h_binary)
    
    # If we have initial_h_values, use them in population
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_orig)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        h_init_binary = (h_init > 0.5).astype(float)
        h_init_binary = h_init_binary * (n_points / 2) / np.sum(h_init_binary)
        
        # Replace first individual with initial construction
        population[0] = h_init_binary
        
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate population
        fitness = []
        for h_ind in population:
            corr = np.correlate(h_ind, 1 - h_ind, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Select top 10% as parents
        top_parents = population[:population_size//10]
        
        # Generate new population
        new_population = top_parents[:]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < 0.5 and len(top_parents) >= 2:
                parent1, parent2 = np.random.choice(top_parents, 2, replace=False)
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                
                # Adjust to maintain sum = n_points/2
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            
            # Mutation
            if np.random.rand() < 0.3 and len(population) > 0:
                ind = np.random.choice(population)
                mut_ind = np.random.randint(0, n_points)
                # Flip bit
                new_ind = ind.copy()
                new_ind[mut_ind] = 1 - new_ind[mut_ind]  # Toggle
                # Adjust to maintain sum = n_points/2
                new_ind = new_ind * (n_points / 2) / np.sum(new_ind)
                new_population.append(new_ind)
            
            # Random individuals
            if len(new_population) < population_size:
                h_temp = np.random.rand(n_points)
                h_temp = h_temp / np.sum(h_temp) * (n_points / 2)
                h_binary = (h_temp > 0.5).astype(float)
                h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
                new_population.append(h_binary)
            
            new_population = new_population[:population_size]
    
    # Final optimization using what we found
    try:
        # Convex optimization on best candidates
        top_candidates = population[:10]
        best_candidate = None
        best_c5_cvx = float('inf')
        
        for h_try in top_candidates:
            # Convex relaxation
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Approximate the overlap as max_k sum h[i] * (1 - h[(i+k)%n_points])
            # This is tricky in convex form; we will instead run local optimization
            
            # Direct optimization using scipy
            def eval_overlap(h_vec):
                return -np.max(np.correlate(h_vec, 1 - h_vec, mode='full')) * dx
            
            # Run a local optimizer to refine
            result = minimize(eval_overlap, h_try, bounds=[(0,1) for _ in range(n_points)], 
                              method='L-BFGS-B', 
                              options={'maxiter': 100})
            
            if result.success:
                h_refined = result.x
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                corr = np.correlate(h_refined, 1 - h_refined, mode='full')
                c5_refined = np.max(corr) * dx
                if c5_refined < best_c5_cvx:
                    best_c5_cvx = c5_refined
                    best_candidate = h_refined
                    
        if best_candidate is not None:
            # Verify feasibility and update best
            h_candidate = np.clip(best_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_final = np.max(corr) * dx
            if c5_final < best_c5:
                best_h = h_candidate
                best_c5 = c5_final
    except:
        pass

    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
