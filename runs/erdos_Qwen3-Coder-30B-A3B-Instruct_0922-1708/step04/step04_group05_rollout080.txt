import numpy as np
import time

def project_feasible(h, n_points):
    """Project h to satisfy bounds and integral constraint"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, n_points):
    """Compute C5 value for given h"""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a resolution that balances speed and quality
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or use uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    h = project_feasible(h_init, n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    
    # Grid search phase
    coarse_points = 50
    fine_points = 150
    
    # Generate coarse samples
    n_coarse = 100
    coarse_h_samples = []
    for _ in range(n_coarse):
        if time.time() - start_time > budget_s - 5:
            break
        sample = np.random.rand(coarse_points)
        sample = project_feasible(sample, coarse_points)
        coarse_h_samples.append(sample)
    
    # Evaluate coarse samples and refine top performers
    coarse_scores = []
    for h_sample in coarse_h_samples:
        if time.time() - start_time > budget_s - 5:
            break
        h_full = np.pad(h_sample, (0, n_points - coarse_points), 'constant')
        c5 = compute_c5(h_full[:n_points], n_points)
        coarse_scores.append(c5)
    
    # Sort by performance and refine top 10%
    sorted_indices = np.argsort(coarse_scores)[:max(1, n_coarse // 10)]
    
    for idx in sorted_indices:
        if time.time() - start_time > budget_s - 5:
            break
        h_refined = coarse_h_samples[idx].copy()
        # Interpolate to higher resolution
        x_orig = np.linspace(0, 2, coarse_points)
        x_new = np.linspace(0, 2, n_points)
        h_interpolated = np.interp(x_new, x_orig, h_refined)
        h = project_feasible(h_interpolated, n_points)
        
        # Local gradient descent
        iter_count = 0
        while time.time() - start_time < budget_s - 2 and iter_count < 100:
            iter_count += 1
            # Compute gradient
            grad = np.zeros(n_points)
            eps = 1e-4
            current_c5 = compute_c5(h, n_points)
            
            # Finite difference
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_feasible(h_plus, n_points)
                c5_plus = compute_c5(h_plus, n_points)
                grad[i] = (c5_plus - current_c5) / eps
            
            # Update
            lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n_points)
            
            # Check improvement
            new_c5 = compute_c5(h_new, n_points)
            if new_c5 < current_c5:
                h = h_new
            else:
                # Add small noise occasionally
                if iter_count % 20 == 0:
                    noise = np.random.normal(0, 0.001, n_points)
                    h = project_feasible(h + noise, n_points)
        
        # Update global best
        c5 = compute_c5(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
