import numpy as np
import time

def check_feasible(h, n_points):
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n_points/2) < 1e-10)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Adjust to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iterations = 0
    lr = 0.001
    stagnation_count = 0
    prev_c5 = float('inf')
    
    # FFT setup
    full_len = 2 * n_points - 1
    
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        
        # Compute convolution using FFT
        h1 = np.concatenate([h, np.zeros(n_points - 1)])
        h2 = np.concatenate([1 - h, np.zeros(n_points - 1)])
        conv_fft = np.fft.fft(h1) * np.fft.fft(h2)
        overlap = np.fft.ifft(conv_fft).real[:full_len]
        
        # Shift indices for overlap
        max_idx = np.argmax(overlap)
        max_shift = max_idx - (n_points - 1)
        
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h1p = np.concatenate([h_plus, np.zeros(n_points - 1)])
            h2p = np.concatenate([1 - h_plus, np.zeros(n_points - 1)])
            conv_plus = np.fft.fft(h1p) * np.fft.fft(h2p)
            overlap_plus = np.fft.ifft(conv_plus).real[:full_len]
            
            grad[i] = (np.max(overlap_plus) - np.max(overlap)) * dx / eps
        
        # Update
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update with momentum
        h = 0.9 * h + 0.1 * h_new
        
        # Evaluate
        h_curr = h.copy()
        h1_curr = np.concatenate([h_curr, np.zeros(n_points - 1)])
        h2_curr = np.concatenate([1 - h_curr, np.zeros(n_points - 1)])
        conv_curr = np.fft.fft(h1_curr) * np.fft.fft(h2_curr)
        overlap_curr = np.fft.ifft(conv_curr).real[:full_len]
        current_c5 = np.max(overlap_curr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1

        # Early stopping
        if stagnation_count > 100:
            break
        
        # Adapt learning rate
        if abs(current_c5 - prev_c5) < 1e-6:
            lr *= 0.9
        prev_c5 = current_c5

    # Final evaluation
    h_final = best_h.copy()
    h1_final = np.concatenate([h_final, np.zeros(n_points - 1)])
    h2_final = np.concatenate([1 - h_final, np.zeros(n_points - 1)])
    conv_final = np.fft.fft(h1_final) * np.fft.fft(h2_final)
    overlap_final = np.fft.ifft(conv_final).real[:full_len]
    final_c5 = np.max(overlap_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
