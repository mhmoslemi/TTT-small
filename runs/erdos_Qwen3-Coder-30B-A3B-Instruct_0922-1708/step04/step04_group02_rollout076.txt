import numpy as np
from scipy.optimize import minimize
import time
import random
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Resolution levels to try
    resolutions = [50, 100, 200]
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for n_points in resolutions:
        if time.time() - start_time > budget_s - 1:
            break
            
        dx = 2.0 / n_points
        
        # Initialize population
        population_size = 100
        population = []
        
        # Start with initial solution if available
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                h_orig = np.array(initial_h_values)
                x_orig = np.linspace(0, 2, len(h_orig))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h_orig)
            h = h / np.sum(h) * (n_points / 2)
            population.append(h.copy())
        else:
            # Generate simple block functions with varying heights
            for _ in range(5):
                h = np.zeros(n_points)
                split = random.randint(1, n_points-1)
                h[:split] = random.uniform(0.5, 2.0)
                h[split:] = random.uniform(0.5, 2.0)
                h = h * (n_points / 2) / np.sum(h)
                population.append(h.copy())
        
        # Generate rest of the population
        for _ in range(population_size - len(population)):
            h = np.random.random(n_points) * 2.0
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
            
        # Genetic algorithm
        for gen in range(50):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Evaluate fitness
            fitness = []
            for h in population:
                corr = np.correlate(h, 1-h, mode='full')
                c5 = np.max(corr) * dx
                fitness.append(c5)
                
            # Update best solution
            for i, c5 in enumerate(fitness):
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = population[i].copy()
            
            # Sort by fitness
            indexed_fitness = list(enumerate(fitness))
            indexed_fitness.sort(key=lambda x: x[1])
            
            # Select top 50%
            selected_indices = [i for i, _ in indexed_fitness[:population_size//2]]
            selected_population = [population[i] for i in selected_indices]
            
            # Generate offspring
            offspring = []
            while len(offspring) < population_size - len(selected_population):
                # Tournament selection
                parent1 = selected_population[random.randint(0, len(selected_population)-1)]
                parent2 = selected_population[random.randint(0, len(selected_population)-1)]
                
                # Crossover
                if random.random() < 0.6:
                    # Uniform crossover
                    child = np.zeros(n_points)
                    for i in range(n_points):
                        if random.random() < 0.5:
                            child[i] = parent1[i]
                        else:
                            child[i] = parent2[i]
                else:
                    # Use one parent
                    child = deepcopy(parent1)
                
                # Mutation
                if random.random() < 0.3:
                    idx = random.randint(0, n_points-1)
                    child[idx] = max(0, min(1, child[idx] + random.gauss(0, 0.1)))
                    
                # Repair
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
                offspring.append(child)
            
            # Combine selected and offspring
            population = selected_population + offspring
            
            # Add some diversity occasionally
            if gen % 10 == 0 and gen > 0:
                for _ in range(10):
                    h = np.random.random(n_points) * 2
                    h = h * (n_points / 2) / np.sum(h)
                    population.append(h)
    
    # Final gradient refinement on best solution
    if best_h is not None and time.time() - start_time < budget_s - 0.5:
        # Create a function for numerical gradient
        def objective(h_vec):
            corr = np.correlate(h_vec, 1-h_vec, mode='full')
            return np.max(corr) * (2.0 / len(h_vec))
            
        def gradient(h_vec):
            # Approximate gradient using finite differences
            eps = 1e-5
            grad = np.zeros_like(h_vec)
            orig_val = objective(h_vec)
            
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (len(h_vec) / 2) / np.sum(h_plus)
                plus_val = objective(h_plus)
                grad[i] = (plus_val - orig_val) / eps
            return grad
            
        # Use a simple gradient descent
        h_refined = best_h.copy()
        lr = 0.001
        for _ in range(100):
            if time.time() - start_time > budget_s - 0.1:
                break
            grad = gradient(h_refined)
            h_refined = h_refined - lr * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (len(h_refined) / 2) / np.sum(h_refined)
            
        # Final validation
        corr = np.correlate(h_refined, 1-h_refined, mode='full')
        final_c5 = np.max(corr) * (2.0 / len(h_refined))
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_refined.copy()
            
    if best_h is None:
        # Fallback to a simple solution
        n_points = 100
        best_h = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        corr = np.correlate(best_h, 1-best_h, mode='full')
        best_c5 = np.max(corr) * dx
        
    return (best_h.tolist(), best_c5, len(best_h))
