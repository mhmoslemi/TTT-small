import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    max_iter = 1000
    iter_count = 0
    
    # Coarse grid search
    coarse_size = 20
    coarse_h = np.linspace(0, 1, coarse_size)
    best_coarse = []
    
    # Generate coarse candidates
    for _ in range(10):
        candidate = np.random.choice(coarse_h, size=coarse_size)
        candidate = candidate * (coarse_size / 2) / np.sum(candidate)
        # Interpolate to fine grid
        fine_candidate = np.interp(np.linspace(0, 2, n_points), 
                                  np.linspace(0, 2, coarse_size), 
                                  candidate)
        fine_candidate = fine_candidate * (n_points / 2) / np.sum(fine_candidate)
        best_coarse.append(fine_candidate)
    
    # Pick top candidates
    coarse_candidates = []
    for candidate in best_coarse:
        corr = np.correlate(candidate, 1 - candidate, mode='full')
        c5_val = np.max(corr) * dx
        coarse_candidates.append((c5_val, candidate))
    
    coarse_candidates.sort(key=lambda x: x[0])
    top_candidates = [x[1] for x in coarse_candidates[:5]]
    
    for candidate in top_candidates:
        h = candidate.copy()
        for _ in range(50):
            iter_count += 1
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current value
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Compute gradient using finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - current_c5) / eps
            
            # Update h with gradient descent
            lr = 0.01
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
    
    # Final CVX optimization using CVXPY
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # This is non-convex; use it for refinement
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                corr = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(corr) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
                    
    except:
        pass

    # Final verification
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
