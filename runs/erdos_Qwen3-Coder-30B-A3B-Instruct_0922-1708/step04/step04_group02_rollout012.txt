import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def compute_c5(h, dx):
    """Compute C5 value from h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def repair_h(h, n_points):
    """Repair h to satisfy constraints: 0 <= h <= 1 and sum = n_points/2"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Rescale to correct sum
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scale = (n_points/2) / current_sum
        h = h * scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or create random
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # PSO Parameters
    swarm_size = 100
    max_iter = 1000
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    local_search_freq = 50
    rejuvenate_rate = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Add initial solution
    h_init = repair_h(h_init, n_points)
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    pbest.append(h_init.copy())
    pbest_vals.append(compute_c5(h_init, dx))
    
    # Fill rest randomly
    for _ in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = repair_h(h, n_points)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
        pbest_vals.append(compute_c5(h, dx))
    
    # Global best
    best_idx = np.argmin(pbest_vals)
    gbest = pbest[best_idx].copy()
    gbest_val = pbest_vals[best_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_val
    
    iter_count = 0
    
    while time.time() - start_time < budget_s - 10 and iter_count < max_iter:
        iter_count += 1
        # Update each particle
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (inertia * v + 
                 cognitive * r1 * (pbest[i] - h) + 
                 social * r2 * (gbest - h))
            
            # Update position
            h_new = h + v
            
            # Repair
            h_new = repair_h(h_new, n_points)
            
            # Compute new C5
            c5_new = compute_c5(h_new, dx)
            
            # Update personal best
            if c5_new < pbest_vals[i]:
                pbest[i] = h_new.copy()
                pbest_vals[i] = c5_new
                
                # Update global best
                if c5_new < gbest_val:
                    gbest = h_new.copy()
                    gbest_val = c5_new
                    if gbest_val < best_c5:
                        best_c5 = gbest_val
                        best_h = gbest.copy()
            
            particles[i] = h_new
            velocities[i] = v
            
        # Local search occasionally
        if iter_count % local_search_freq == 0:
            # Get top performers
            top_indices = np.argsort(pbest_vals)[:10]
            for i in top_indices:
                h = pbest[i].copy()
                c5 = pbest_vals[i]
                
                # Gradient-based local search
                for _ in range(10):
                    # Approximate gradient
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = repair_h(h_plus, n_points)
                        c5_plus = compute_c5(h_plus, dx)
                        grad[j] = (c5_plus - c5) / eps
                    
                    # Update step
                    h_new = h - 0.001 * grad
                    h_new = repair_h(h_new, n_points)
                    c5_new = compute_c5(h_new, dx)
                    
                    if c5_new < c5:
                        h = h_new
                        c5 = c5_new
                        if c5 < pbest_vals[i]:
                            pbest[i] = h.copy()
                            pbest_vals[i] = c5
                            if c5 < gbest_val:
                                gbest = h.copy()
                                gbest_val = c5
                                if gbest_val < best_c5:
                                    best_c5 = gbest_val
                                    best_h = gbest.copy()
                    else:
                        break
        
        # Rejuvenate worst particles
        if iter_count % 200 == 0 and iter_count > 0:
            worst_indices = np.argsort(pbest_vals)[-int(swarm_size * rejuvenate_rate):]
            for i in worst_indices:
                h = np.random.uniform(0, 1, n_points)
                h = repair_h(h, n_points)
                particles[i] = h
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                pbest[i] = h
                pbest_vals[i] = compute_c5(h, dx)
    
    # Finally, do some additional local refinement on the best
    h_final = best_h.copy()
    c5_final = best_c5
    for _ in range(20):  # Limited refinement 
        gradient = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = compute_c5(h_plus, dx)
            gradient[i] = (c5_plus - c5_final) / eps
        
        # Update and refine
        h_new = h_final - 0.001 * gradient
        h_new = repair_h(h_new, n_points)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5_final:
            h_final = h_new
            c5_final = c5_new
        else:
            break
    
    return (h_final.tolist(), c5_final, n_points)
