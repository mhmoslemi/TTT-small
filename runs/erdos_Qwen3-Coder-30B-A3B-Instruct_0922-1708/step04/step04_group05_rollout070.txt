import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Create diverse initial candidates
    candidates = []
    candidates.append(h_init.copy())
    
    # Add some variations of the initial solution
    for i in range(19):
        h_var = h_init + np.random.normal(0, 0.05, n_points)
        h_var = np.clip(h_var, 0, 1)
        h_var = h_var * (n_points / 2) / np.sum(h_var)
        candidates.append(h_var.copy())
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    iteration = 0
    stagnation_counter = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Pick a random candidate
        h = candidates[np.random.randint(len(candidates))].copy()
        
        # Local gradient-based refinement
        for step in range(50):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Compute correlation
            corr_full = np.correlate(h, 1 - h, mode='full')
            corr_max = np.max(corr_full)
            max_idx = np.argmax(corr_full)
            
            # Gradient computation
            grad = np.zeros(n_points)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                grad[i] = (np.max(corr_plus) - corr_max) / eps
            
            # Update step
            lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
            # Track best
            new_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stagnation_counter = 0
            else:
                stagnation_counter += 1
                
            if stagnation_counter > max_stagnation:
                break
                
        # Add new candidate from current h
        h_noisy = h + np.random.normal(0, 0.01, n_points)
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = h_noisy * (n_points / 2) / np.sum(h_noisy)
        candidates.append(h_noisy)
        
        # Remove oldest candidate periodically
        if len(candidates) > 20:
            candidates.pop(0)
            
    # Final verification
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
