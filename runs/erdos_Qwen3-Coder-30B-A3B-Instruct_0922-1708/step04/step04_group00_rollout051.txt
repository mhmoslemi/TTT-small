import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():

    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # PSO parameters
    n_particles = 30
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    vmax = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals() and np.random.rand() < 0.5:
            # Use parent for some particles
            particle_h = h.copy()
        else:
            particle_h = np.random.uniform(0, 1, n_points)
            particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
            particle_h = np.clip(particle_h, 0, 1)
        particles.append(particle_h)
        velocities.append(np.random.uniform(-vmax, vmax, n_points))
        pbest.append(particle_h.copy())
        pbest_c5.append(float('inf'))
    
    gbest = h.copy()
    gbest_c5 = float('inf')
    
    # PSO iterations
    max_iter = 200
    no_improvement = 0
    max_no_improvement = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Compute fitness
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particle.copy()
                pbest_c5[i] = c5
                
            # Update global best
            if c5 < gbest_c5:
                gbest = particle.copy()
                gbest_c5 = c5
                
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = particle.copy()
                
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            vel = (inertia * vel + 
                   c1 * r1 * (pbest[i] - particle) + 
                   c2 * r2 * (gbest - particle))
            
            # Clip velocity
            vel = np.clip(vel, -vmax, vmax)
            
            # Update position
            particle += vel
            
            # Enforce constraints
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            velocities[i] = vel
            particles[i] = particle
            
        # Early stopping
        if gbest_c5 < best_c5:
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > max_no_improvement:
                break

    h = best_h.copy()

    def objective(h):
        # Normalize h
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        correlation = np.correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx  # Negative for minimization

    # Define constraint for Nelder-Mead
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Create initial simplex using random perturbations
    simplex_size = 5
    simplex = [h]

        perturb = np.random.normal(0, 0.01, n_points)
        perturb = np.clip(perturb, -1, 1)
        h_new = h + perturb
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        simplex.append(h_new)
    
    # Nelder-Mead optimization parameters
    max_iter = 1000
    alpha = 1.0  # reflection coefficient
    gamma = 2.0  # expansion coefficient
    rho = 0.5    # contraction coefficient
    sigma = 0.5  # shrink coefficient
    
    # Normalize initial simplex
    for i in range(len(simplex)):
        simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
        simplex[i] = np.clip(simplex[i], 0, 1)
        
    # Create function for optimization
    def fun(x):
        x = x * (n_points / 2) / np.sum(x)
        x = np.clip(x, 0, 1)
        correlation = np.correlate(x, 1 - x, mode='full')
        return -np.max(correlation) * dx
    
    # Nelder-Mead steps
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort vertices of the simplex by function value
        values = [fun(s) for s in simplex]
        indices = np.argsort(values)
        simplex = [simplex[i] for i in indices]
        values = [values[i] for i in indices]
        
        # Calculate centroid (excluding the worst point)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        xr = centroid + alpha * (centroid - simplex[-1])
        xr = xr * (n_points / 2) / np.sum(xr)
        xr = np.clip(xr, 0, 1)
        fr = fun(xr)
        
        if values[0] <= fr < values[-2]:
            simplex[-1] = xr
        elif fr < values[0]:
            # Expansion
            xe = centroid + gamma * (xr - centroid)
            xe = xe * (n_points / 2) / np.sum(xe)
            xe = np.clip(xe, 0, 1)
            fe = fun(xe)
            if fe < fr:
                simplex[-1] = xe
            else:
                simplex[-1] = xr
        else:
            # Contraction
            if fr < values[-1]:
                xc1 = centroid + rho * (xr - centroid)
                xc1 = xc1 * (n_points / 2) / np.sum(xc1)
                xc1 = np.clip(xc1, 0, 1)
                fc1 = fun(xc1)
                if fc1 <= fr:
                    simplex[-1] = xc1
                else:
                    # Shrink
                    for i in range(1, len(simplex)):
                        simplex[i] = simplex[0] + sigma * (simplex[i] - simplex[0])
                        simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
                        simplex[i] = np.clip(simplex[i], 0, 1)
            else:
                # Outer contraction
                xc2 = centroid + rho * (simplex[-1] - centroid)
                xc2 = xc2 * (n_points / 2) / np.sum(xc2)
                xc2 = np.clip(xc2, 0, 1)
                fc2 = fun(xc2)
                if fc2 < values[-1]:
                    simplex[-1] = xc2
                else:
                    # Shrink
                    for i in range(1, len(simplex)):
                        simplex[i] = simplex[0] + sigma * (simplex[i] - simplex[0])
                        simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
                        simplex[i] = np.clip(simplex[i], 0, 1)
      
    # Final evaluation
    h = simplex[0]
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
