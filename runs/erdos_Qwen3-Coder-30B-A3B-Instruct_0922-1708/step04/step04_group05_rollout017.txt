import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Gradient descent in frequency domain
    gamma = 0.001
    max_iter = 10000
    patience = 100
    no_improve_count = 0
    last_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap using FFT convolution
        h_full = np.concatenate([h, np.zeros(n_points-1)])
        h_conj = np.concatenate([h, np.zeros(n_points-1)])
        h_conj = np.flip(h_conj)
        overlap = np.real(np.fft.ifft(np.fft.fft(h_full) * np.fft.fft(h_conj)))
        
        # Find maximum overlap
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count > patience:
            # Early stopping if no improvement
            break
            
        # Frequency domain optimization
        H = np.fft.fft(h)
        k = max_idx - (n_points - 1)  # convert to shift index
        
        # Create directional filter to reduce overlap
        omega = np.arange(n_points) * 2 * np.pi / n_points
        # Filter to reduce the specific overlap at shift k
        if k == 0:
            filter_freq = np.ones(n_points)
        else:
            filter_freq = 1 - np.exp(-1j * omega * k)
            # Avoid division by zero
            filter_freq = np.where(np.abs(filter_freq) < 1e-10, 1, filter_freq)
            filter_freq = filter_freq / np.abs(filter_freq)
        
        # Apply the filter
        H_filtered = H * filter_freq
        
        # Inverse transform
        h_new = np.real(np.fft.ifft(H_filtered))
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Gradient step
        h = (1 - gamma) * h + gamma * h_new
        
        # Project again to ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive step size
        if i % 100 == 0 and i > 0:
            gamma *= 0.9
    
    # Final evaluation
    h_full = np.concatenate([best_h, np.zeros(n_points-1)])
    h_conj = np.concatenate([best_h, np.zeros(n_points-1)])
    h_conj = np.flip(h_conj)
    overlap = np.real(np.fft.ifft(np.fft.fft(h_full) * np.fft.fft(h_conj)))
    final_c5 = np.max(overlap) * dx
    
    return (best_h.tolist(), final_c5, n_points)
