import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Grid search with coarse resolution for exploration
    coarse_points = 20
    coarse_dx = 2.0 / coarse_points
    subsample_factor = n_points // coarse_points
    
    # Generate some coarse candidates
    candidates = []
    num_candidates = 10
    
    for _ in range(50):  # Generate many candidates
        # Create random candidate in [0,1] space
        h_coarse = np.random.rand(coarse_points)
        h_coarse = h_coarse * (coarse_points / 2) / np.sum(h_coarse)
        
        # Upsample to full resolution
        h_full = np.repeat(h_coarse, subsample_factor)
        # Pad or truncate if needed
        if len(h_full) > n_points:
            h_full = h_full[:n_points]
        elif len(h_full) < n_points:
            h_full = np.pad(h_full, (0, n_points-len(h_full)), mode='constant', constant_values=0)
            
        # Normalize
        h_full = h_full * (n_points / 2) / np.sum(h_full)
        h_full = np.clip(h_full, 0, 1)
        
        # Evaluate
        correlation = np.correlate(h_full, 1 - h_full, mode='full')
        c5 = np.max(correlation) * dx
        
        candidates.append((h_full.copy(), c5))
        
    # Sort candidates
    candidates.sort(key=lambda x: x[1])
    
    # Keep top candidates
    candidates = candidates[:num_candidates]
    
    # Phase 2: Refinement using CVX and gradient descent
    # Refine each candidate from grid search
    refinement_steps = 500
    learning_rate = 0.01
    
    for h_start, _ in candidates:
        h_refine = h_start.copy()
        
        # Use gradient descent to refine
        for _ in range(refinement_steps):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute loss (C5)
            correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
            c5_current = np.max(correlation) * dx
            
            # Compute approximate gradient using finite differences
            grad = np.zeros_like(h_refine)
            eps = 1e-6
            
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5_current) / eps
            
            # Update h
            h_new = h_refine - learning_rate * grad
            
            # Projection
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_refine = h_new
            
            # Check improvement
            correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
            c5_new = np.max(correlation) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_refine.copy()
    
    # Final CVX optimization on best candidate
    try:
        # Define the optimization variables and constraints
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        #Objective: minimize the maximum overlap by creating an auxiliary variable
        # This matches the structure we can solve using CVXPY
        # We use a different approach to optimize with CVXPY
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass

    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
