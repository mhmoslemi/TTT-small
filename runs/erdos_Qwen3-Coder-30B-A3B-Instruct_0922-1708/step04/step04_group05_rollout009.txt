import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    correlation_full = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation_full) * dx
    return c5

def gradient_step(h, n_points, lr=0.001):
    eps = 1e-4
    grad = np.zeros(n_points)
    h_base = h.copy()
    
    for i in range(n_points):
        h_plus = h_base.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        c5_plus = compute_c5(h_plus, n_points)
        c5_base = compute_c5(h_base, n_points)
        grad[i] = (c5_plus - c5_base) / eps
    
    h_new = h_base - lr * grad
    h_new = project_feasible(h_new, n_points)
    return h_new

def adaptive_grid_search(n_points, budget_s):
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    # Initial coarse search
    N_coarse = 50
    for i in range(N_coarse):
        if time.time() - start_time > budget_s - 10:
            break
        # Random construction
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        c5 = compute_c5(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Refinement phase
    # Focus on top candidates (first 20%)
    candidates = []
    for i in range(20):
        if time.time() - start_time > budget_s - 10:
            break
        h_try = np.random.rand(n_points)
        h_try = project_feasible(h_try, n_points)
        c5_try = compute_c5(h_try, n_points)
        candidates.append((c5_try, h_try))
    
    candidates.sort(key=lambda x: x[0])
    candidates = candidates[:10]  # Top 10
    
    for _, h_seed in candidates:
        if time.time() - start_time > budget_s - 10:
            break
        h = h_seed.copy()
        for iter in range(50):
            if time.time() - start_time > budget_s - 10:
                break
            h = gradient_step(h, n_points, lr=0.001)
            c5 = compute_c5(h, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            if np.abs(c5 - compute_c5(h, n_points)) < 1e-6:
                break
    
    return best_h, best_c5

    else:

    # Normalize integral constraint

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n_points)
    
    # Adaptive grid search with refinement
    h_best, c5_best = adaptive_grid_search(n_points, budget_s)
