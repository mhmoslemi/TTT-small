import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert to binary
    h = (h_init >= 0.5).astype(int)
    
    # Ensure exactly n_points/2 ones
    ones_count = np.sum(h)
    if ones_count > n_points // 2:
        # Set some 1s to 0s
        zero_indices = np.where(h == 1)[0]
        np.random.shuffle(zero_indices)
        for i in range(ones_count - n_points // 2):
            h[zero_indices[i]] = 0
    elif ones_count < n_points // 2:
        # Set some 0s to 1s
        one_indices = np.where(h == 0)[0]
        np.random.shuffle(one_indices)
        for i in range(n_points // 2 - ones_count):
            h[one_indices[i]] = 1
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Simulated Annealing
    T = 1.0
    temp_decay = 0.999
    last_improvement = 0
    max_no_improvement = 1000
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Create a new candidate by flipping a random bit
        i = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[i] = 1 - h_new[i]
        
        # Ensure exactly n_points/2 ones
        ones_count = np.sum(h_new)
        if ones_count > n_points // 2:
            # Need to set some 1s to 0s
            zero_indices = np.where(h_new == 1)[0]
            np.random.shuffle(zero_indices)
            for i in range(ones_count - n_points // 2):
                h_new[zero_indices[i]] = 0
        elif ones_count < n_points // 2:
            # Need to set some 0s to 1s
            one_indices = np.where(h_new == 0)[0]
            np.random.shuffle(one_indices)
            for i in range(n_points // 2 - ones_count):
                h_new[one_indices[i]] = 1
        
        c5_new = compute_c5(h_new)
        c5_curr = compute_c5(h)
        
        # Accept or reject the new solution
        if c5_new < c5_curr:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                last_improvement = 0
        else:
            # Probabilistically accept worse solutions
            delta = c5_new - c5_curr
            if np.random.rand() < np.exp(-delta / T):
                h = h_new
                last_improvement = 0
            else:
                last_improvement += 1
        
        # Temperature update
        T *= temp_decay
        
        # Random perturbation every 1000 iterations
        if iteration % 1000 == 0:
            # Reset 5% of values
            num_to_reset = n_points // 20
            reset_indices = np.random.choice(n_points, num_to_reset, replace=False)
            for idx in reset_indices:
                h[idx] = np.random.randint(0, 2)
            
            # Ensure exact count
            ones_count = np.sum(h)
            if ones_count > n_points // 2:
                zero_indices = np.where(h == 1)[0]
                np.random.shuffle(zero_indices)
                for i in range(ones_count - n_points // 2):
                    h[zero_indices[i]] = 0
            elif ones_count < n_points // 2:
                one_indices = np.where(h == 0)[0]
                np.random.shuffle(one_indices)
                for i in range(n_points // 2 - ones_count):
                    h[one_indices[i]] = 1
        
        # Early termination if no improvement for many iterations
        if last_improvement > max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
