import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Try a convex optimization approach
    # Use CVXPY to solve the problem with constraints
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Create the constraints for all possible shifts
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Box constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # For each shift k, add constraint on overlap
    for k in range(1, n_points):
        # Compute overlap for shift k
        if k < n_points:
            # This is the correlation computation for shift k
            # Overlap = sum(h[i] * (1 - h[i+k])) i=0..n-k-1
            overlap = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
        else:
            overlap = 0
        constraints.append(overlap <= C)
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers_to_try = ['ECOS', 'SCS', 'OSQP']
    for solver in solvers_to_try:
        if time.time() - start_time > budget_s - 1:
            break
        try:
            prob.solve(solver=solver, verbose=False, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_sol = h_var.value
                # Adjust solution to satisfy sum constraint
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                # Compute C5
                correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                c5 = np.max(correlation) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_sol.copy()
        except:
            continue
    
    # If not good enough, do some local refinement
    if best_c5 > 0.381:
        # Try gradient-based refinement from the better solution
        h = best_h.copy()
        for iter in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Estimate gradient using finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Apply gradient step
            lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-4)
            h_new = h - lr * grad
            
            # Project back to feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Update best if better
            correlation = np.correlate(h, 1 - h, mode='full')
            c5_new = np.max(correlation) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
            # Check convergence
            if np.abs(c5_new - c5) < 1e-10:
                break
    
    return (best_h.tolist(), best_c5, n_points)
