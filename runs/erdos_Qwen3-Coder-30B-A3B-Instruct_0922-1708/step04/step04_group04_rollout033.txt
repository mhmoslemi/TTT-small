import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Iterative convex optimization with line search
    max_iter = 100
    step_size = 0.5
    tol = 1e-5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Formulate and solve convex subproblem
        h_old = h.copy()
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [cp.sum(h_var) == n_points / 2]
        constraints += [0 <= h_var, h_var <= 1]
        
        # Linearize cross-correlation terms using current values
        # For k in [0, 2*n_points-2], compute sum(h[i] * (1 - h[i+k])) <= C
        for k in range(2 * n_points - 1):
            # Determine valid indices
            if k < n_points:
                # First part of the convolution
                i_indices = np.arange(n_points - k)
                shift_indices = np.arange(k, n_points)
            else:
                # Second part of convolution
                i_indices = np.arange(k - n_points + 1, n_points)
                shift_indices = np.arange(n_points - (k - n_points + 1))
            
            if len(i_indices) > 0:
                # Use linearized constraint: h[i] (1 - h_old[i+k]) + h_old[i] (h[i+k] - h_old[i+k]) <= C
                # Simplified: h[i] - h[i] * h_old[i+k] + h_old[i] * h[i+k] - h_old[i] * h_old[i+k] <= C
                # Rearranged: h[i] + h_old[i] * h[i+k] - h_old[i] * h_old[i+k] - h[i] * h_old[i+k] <= C  
                # Which we do by linearizing h[i] * h_old[i+k] as h_old[i+k] * h[i] + h[i] * (h_old[i+k] - h_old[i+k])
                # Simplifies to h_old[i+k] * h[i] + h[i] * (h_old[i+k] - h_old[i+k]) = h_old[i+k] * h[i]
                # This is h[i] * h_old[i+k] = h_old[i+k] * h[i]
                # Need better approximation...
                # Instead, use h[i] * (1 - h_old[i+k])
                # So we add constraint (h[i] * (1 - h_old[i+k])) <= C - (h[i] (1 - h_old[i+k])) + h_old[i] * h[i+k]
                # But not that composite. Use h[i] - h[i] * h_old[i+k] <= C (also its direct linearization)
                # Better: We can solve for marginal contribution to each constraint
                # Compute indices that contribute to k-step correlation
                left_indices = i_indices
                right_indices = shift_indices
                
                # Refined feasibility: inner products over valid pairs
                # Add linearized constraint
                expr = 0
                for idx in range(len(left_indices)):
                    i_idx = left_indices[idx]
                    j_idx = right_indices[idx]
                    expr += h_var[i_idx] - h_var[i_idx] * h_old[j_idx]
                constraints.append(expr <= C)
                
        # This gets messy
        # Let's do a simpler way:
        # For each k, we want to minimize max_k sum(h[i] * (1 - h[i+k]))
        # So we add C >= sum(h[i] * (1 - h_old[i+k]))
        # But this is still unclear. Approach: compute correlation of h with 1-h
        # Linearization: h[i]*h[i+k] ≈ h[i]h_old[i+k] + h_old[i]h[i+k] - h_old[i]h_old[i+k]
        # We want sum(h[i] * (1 - h[i+k])) = sum(h[i] - h[i]*h[i+k])
        # So we add constraint sum(h[i] - h[i]*h_old[i+k] - h_old[i]*h[i+k] + h_old[i]*h_old[i+k]) <= C
        # новых US

        # Simple approach: linearize correlation as sum(h[i]*(1-h_old[i+k])) <= C
        for k in range(n_points):
            # sum over valid i
            i_range = np.arange(n_points - k)
            expr = 0
            for i in i_range:
                expr += h_var[i] - h_var[i] * h_old[i + k]
            constraints.append(expr <= C)
        
        # Objective: minimize C
        prob = cp.Problem(cp.Minimize(C), constraints)
        
        # Try with different solvers
        try:
            prob.solve(solver=cp.ECOS, abstol=1e-4, reltol=1e-4, feastol=1e-4)
        except:
            # Fallback if main solver fails
            try:
                prob.solve(solver=cp.SCS, abstol=1e-3, reltol=1e-3, feastol=1e-3)
            except:
                # Final fallback
                break
        
        # Update h using line search
        h_new = h_var.value
        if h_new is None:
            break
            
        h_new = np.maximum(0, np.minimum(1, h_new))
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Apply line search for step size
        s = step_size
        for _ in range(100):  # max backtracking steps
            h_trial = h + s * (h_new - h)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial) 
            
            corr = np.correlate(h_trial, 1 - h_trial, mode='full')
            c5_trial = np.max(corr) * dx
            
            # Accept if better
            if c5_trial < current_c5:
                h = h_trial
                break
            else:
                s *= 0.5
                
        # Convergence check
        if np.linalg.norm(h_new - h_old, ord=2) < tol:
            break
    
    # Final evaluation
    correlation = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(correlation) * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    # Ensure boundary conditions
    # Check if h sum is correct to within tolerance
    if abs(np.sum(best_h) - n_points / 2) > 1e-5:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    # Clamp to interval [0,1]
    best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
