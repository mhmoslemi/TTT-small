import numpy as np
from scipy.optimize import minimize
import time
import random
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Generate a simple initial solution
        h = np.zeros(n_points)
        # Two blocks: 1 on left half, 0 on right half, then adjust
        h[:n_points//2] = 2.0
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h_vector):
        # Compute correlation using numpy's correlate
        corr = np.correlate(h_vector, 1 - h_vector, mode='full')
        # We only care about valid overlaps (not padding)
        valid_corr = corr[n_points-1:2*n_points-1]
        return np.max(valid_corr) * dx
    
    def repair_h(h_vector):
        # Ensure h is in [0,1] and sum = n_points/2
        h_vector = np.clip(h_vector, 0, 1)
        scale = (n_points / 2) / np.sum(h_vector)
        h_vector = h_vector * scale
        return h_vector
    
    def mutation(h_vector, rate):
        h_new = h_vector.copy()
        for i in range(len(h_new)):
            if random.random() < rate:
                # Add small random change
                delta = random.gauss(0, 0.05)
                h_new[i] += delta
        return repair_h(h_new)
    
    def crossover(h1, h2):
        # Uniform crossover
        h_new = np.zeros_like(h1)
        for i in range(len(h1)):
            if random.random() < 0.5:
                h_new[i] = h1[i]
            else:
                h_new[i] = h2[i]
        return repair_h(h_new)
    
    # Initialize population with diverse solutions
    population_size = 50
    population = [repair_h(h + np.random.normal(0, 0.02, n_points)) for _ in range(population_size)]
    population[0] = h.copy()  # Ensure initial solution is in population
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Evaluate fitness
        fitnesses = [compute_c5(h) for h in population]
        
        # Update best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
        
        # Selection (tournament)
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament = [random.choice(population) for _ in range(tournament_size)]
            tournament_fitness = [compute_c5(h) for h in tournament]
            winner = tournament[np.argmin(tournament_fitness)]
            selected.append(winner.copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                # Crossover
                child1 = crossover(selected[i], selected[i+1])
                child2 = crossover(selected[i+1], selected[i])
                # Mutation
                child1 = mutation(child1, 0.1 + 0.05 * (1 - iteration/100))
                child2 = mutation(child2, 0.1 + 0.05 * (1 - iteration/100))
                new_population.extend([child1, child2])
            else:
                # Mutation only
                child = selected[i]
                child = mutation(child, 0.1 + 0.05 * (1 - iteration/100))
                new_population.append(child)
        
        population = new_population[:population_size]
        
        # Periodic gradient refinement for best solutions
        if iteration % 10 == 0:
            for _ in range(5):
                # Randomly pick candidates to refine
                idx = random.randint(0, len(population)-1)
                h_cur = population[idx]
                # Apply small gradient step to improve
                eps = 1e-4
                base_c5 = compute_c5(h_cur)
                for i in range(n_points):
                    h_plus = h_cur.copy()
                    h_plus[i] += eps
                    h_plus = repair_h(h_plus)
                    c5_plus = compute_c5(h_plus)
                    if c5_plus < base_c5:
                        h_cur[i] += eps * 0.1  # Small step
                h_cur = repair_h(h_cur)
                population[idx] = h_cur
    
    # Final refinement step
    h_final = best_h.copy()
    for _ in range(20):
        h_test = h_final.copy()
        # Perturb a few random positions
        j = random.randint(0, n_points-1)
        h_test[j] += random.gauss(0, 0.005)
        h_test = repair_h(h_test)
        c5_test = compute_c5(h_test)
        if c5_test < best_c5:
            best_h = h_test
            best_c5 = c5_test
    
    return (best_h.tolist(), best_c5, n_points)
