import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a uniform distribution
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Coarse grid search
    coarse_size = 20
    coarse_points = np.linspace(0, 1, coarse_size)
    
    # Generate coarse candidates
    candidates = []
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Generate some random candidates
    for _ in range(50):
        h_test = np.random.choice(coarse_points, size=coarse_size)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        if len(h_test) < n_points:
            h_test = np.pad(h_test, (0, n_points - len(h_test)), 'edge')
        elif len(h_test) > n_points:
            h_test = h_test[:n_points]
        c5 = compute_c5(h_test)
        candidates.append((h_test.copy(), c5))
    
    # Sort candidates
    candidates.sort(key=lambda x: x[1])
    
    # Refine the best candidates
    for h_start, _ in candidates[:10]:
        h_refined = h_start.copy()
        local_time = time.time()
        max_iter = 500
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Compute gradient
            grad = np.zeros_like(h_refined)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c5_plus = compute_c5(h_plus)
                c5_current = compute_c5(h_refined)
                grad[j] = (c5_plus - c5_current) / eps
            
            # Update step
            lr = 0.01
            h_new = h_refined - lr * grad
            
            # Clip and normalize
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_refined = h_new
            
            # Update best
            c5_refined = compute_c5(h_refined)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
    
    # Final optimization using CVXPY if time allows
    if time.time() - start_time < budget_s - 10:
        try:
            # Use CVXPy to solve a more precise version
            h_vars = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Use the best current h as starting point
            h_start = best_h
            
            # Modify the approach: directly optimize a composite expression
            # Since direct CVX optimization might be too slow, we try
            # a hybrid with gradient and explicit check
            
            # Check a set of possibilities with small variation around best solution
            for _ in range(50):
                h_perturbed = h_start + np.random.randn(n_points) * 0.01
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
                c5_pert = compute_c5(h_perturbed)
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h_perturbed.copy()
        except:
            pass
    
    # Final assessment
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
