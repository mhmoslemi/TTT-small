import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Create a more intelligent initialization
    # Try to make h concentrated in a way that minimizes overlap
    h_init = np.zeros(n_points)
    # Place mass more efficiently - concentrate around 1/3 and 2/3 regions
    center1 = n_points // 3
    center2 = 2 * n_points // 3
    spread = n_points // 8
    
    for i in range(max(0, center1 - spread), min(n_points, center1 + spread)):
        h_init[i] = 0.5 + 0.5 * np.sin(np.pi * (i - center1) / (2 * spread))
    
    for i in range(max(0, center2 - spread), min(n_points, center2 + spread)):
        h_init[i] = 0.5 + 0.5 * np.sin(np.pi * (i - center2) / (2 * spread))
    
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    h = h_init.copy()
    
    # Convert to CVXPY problem
    # Expand constraints to include the bisection method approach
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # LP approach: use cvxpy with auxiliary variables to relax the bilinear terms
    try:
        # Make a tighter linearized approach using CVXPY
        h_var = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # For each shift k, create auxiliary variable w[x,k] for h[x]*h[x+k]
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2, C5 >= 0]
        
        # Add quadratic constraints as linear approximations
        # This is a simplification of the actual approach but gets us close
        # Instead, we'll do combined optimization
        
        # Direct optimization approach with better starting point
        
        # Gradient descent refinement
        max_iter = 5000
        lr = 0.01
        patience = 0
        min_lr = 1e-6
        patience_threshold = 20
        
        h = h_init.copy()
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Finite differences for gradient
            eps = 1e-4
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_c5(h_plus)
                grad[j] = (c5_plus - current_c5) / eps
            
            # Update step with gradient descent
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update learning rate
            if current_c5 <= best_c5:
                patience = 0
            else:
                patience += 1
                if patience > patience_threshold:
                    lr = max(min_lr, lr * 0.5)
                    patience = 0
                    
            h = h_new
            
        # Final check
        final_c5 = compute_c5(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            
    except Exception as e:
        # Fallback to a simpler approach if cvxpy fails
        # Reuse the previous PSO-based code with modified parameters
        # Use a limited variation of the previous method with better search space
        try:
            # Run SCG gradient descent for better convergence
            def objective(h_vals):
                correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
                return np.max(correlation) * dx
                
            def gradient(h_vals):
                eps = 1e-4
                grad = np.zeros_like(h_vals)
                base_c5 = objective(h_vals)
                for i in range(n_points):
                    h_plus = h_vals.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    # Approximate gradient
                    grad[i] = (objective(h_plus) - base_c5) / eps
                return grad
            
            # Optimize with scipy optimize
            h_opt = h_init.copy()
            options = {'maxiter': 2000, 'disp': False}
            
            # Use L-BFGS-B as it handles bounds well
            result = minimize(
                objective, 
                h_opt,
                method='L-BFGS-B', 
                bounds=[(0, 1)] * n_points,
                jac=gradient,
                options=options,
                callback=lambda x: None  # no callback since functional approach used
            )
            
            if not result.success:
                result = minimize(objective, h_opt, method='Powell')
                
            h_final = result.x.copy()
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            
            final_c5 = objective(h_final)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_final.copy()
                
        except:
            pass

    # Ensure best_h is valid and compute final result
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
