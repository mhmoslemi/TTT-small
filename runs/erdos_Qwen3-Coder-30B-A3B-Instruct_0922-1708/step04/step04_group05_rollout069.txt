import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to meet integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Frequency domain approach
    max_iter = 1000
    gamma = 0.001
    patience = 0
    prev_c5 = float('inf')
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap using FFT for efficiency
        h_full = np.concatenate([h, np.zeros(n_points - 1)])
        h_conj = np.concatenate([h, np.zeros(n_points - 1)])
        overlap_fft = np.fft.fft(h_full) * np.fft.fft(1 - h_conj)
        overlap = np.fft.ifft(overlap_fft).real[:n_points]
        
        # Normalize and find maximum overlap
        max_overlap_idx = np.argmax(overlap)
        max_overlap = overlap[max_overlap_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Compute gradient in frequency domain
        H = np.fft.fft(h)
        G = np.zeros(n_points, dtype=complex)
        
        # Design frequency filter to reduce peak overlap
        k = max_overlap_idx
        if k <= n_points // 2:
            G = 1 - np.exp(-1j * 2 * np.pi * np.arange(n_points) * k / n_points)
        else:
            G = 1 - np.exp(1j * 2 * np.pi * np.arange(n_points) * (n_points - k) / n_points)
            
        # Apply filter
        G = G / (np.abs(G) + 1e-8)
        H_new = H * G
        
        # Inverse transform
        h_new = np.fft.ifft(H_new).real
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update with momentum
        h = (1 - gamma) * h + gamma * h_new
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping
        if abs(prev_c5 - max_overlap) < 1e-5:
            patience += 1
            if patience > 100:
                break
        else:
            patience = 0
        prev_c5 = max_overlap
        
        # Periodic random perturbation
        if iter_count % 50 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_overlap = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_overlap) * dx
    
    return (best_h.tolist(), final_c5, n_points)
