import numpy as np
from scipy.stats import qmc
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use Latin Hypercube Sampling for exploration
    sampler = qmc.LatinHypercube(d=n_points, seed=seed)
    lhs_samples = sampler.random(n=50)
    
    # Convert LHS samples to feasible solutions
    candidates = []
    for i in range(lhs_samples.shape[0]):
        sample = lhs_samples[i]
        # transform to [0,1]^n and rescale to sum to n_points/2
        h_candidate = sample * (n_points / 2) / np.sum(sample)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        candidates.append(h_candidate)
    
    # Add the initial solution if it's feasible
    if np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n_points/2) < 1e-5:
        candidates.append(h)
    
    # Gradient descent refinement
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx

    def gradient_step(h_vals, eps=1e-5):
        grad = np.zeros_like(h_vals)
        c5_baseline = compute_c5(h_vals)
        
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = h_vals.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * eps)
            
        return grad

    max_iterations = 1000
    lr = 0.01
    patience = 0
    best_iter = 0
    
    for i, h_candidate in enumerate(candidates):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_current = h_candidate.copy()
        prev_c5 = compute_c5(h_current)
        improved = True
        iter_count = 0
        
        # Local refinement with gradient descent
        while improved and iter_count < max_iterations:
            if time.time() - start_time > budget_s - 1:
                break
                
            grad = gradient_step(h_current)
            h_new = h_current - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)

            new_c5 = compute_c5(h_new)
            improved = new_c5 < prev_c5
            
            if improved:
                h_current = h_new
                prev_c5 = new_c5
                patience = 0
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_current.copy()
            else:
                patience += 1
                lr *= 0.5  # Reduce learning rate if no improvement
                if patience > 50 or lr < 1e-6:
                    break
                    
            iter_count += 1
    
    # Final refinement
    h = best_h.copy()
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_before = compute_c5(h)
        grad = gradient_step(h)
        h_new = h - 0.001 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_after = compute_c5(h_new)
        if c5_after < c5_before:
            h = h_new
            if c5_after < best_c5:
                best_c5 = c5_after
                best_h = h.copy()
        else:
            break
    
    return (best_h.tolist(), best_c5, n_points)
