import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using parent construction or block function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Phase 1: Grid search with coarse resolution
    coarse_grid_size = 20
    top_candidates = []
    n_grid_points = coarse_grid_size ** coarse_grid_size
    
    if n_grid_points <= 50000:  # Reasonable limit
        # Use a random sampling approach instead of full enumeration
        n_samples = min(1000, n_grid_points)
        for _ in range(n_samples):
            h_grid = np.random.rand(coarse_grid_size)
            h_grid = h_grid / np.sum(h_grid) * (coarse_grid_size / 2)
            # Upsample to fine resolution
            h_upsampled = np.interp(
                np.linspace(0, 2, n_points),
                np.linspace(0, 2, coarse_grid_size),
                h_grid
            )
            h_upsampled = np.clip(h_upsampled, 0, 1)
            h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
            corr = correlate(h_upsampled, 1 - h_upsampled, mode='full')
            c5 = np.max(corr) * dx
            top_candidates.append((c5, h_upsampled))
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:10]
    else:
        # If grid is too large, just sample the initial solution
        top_candidates = [(float('inf'), h)]
    
    # Phase 2: Refinement using the best candidates from grid search
    for _, h_seed in top_candidates:
        h_refine = h_seed.copy()
        
        # Gradient-based refinement
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Gradient estimate using finite differences
            grad = np.zeros_like(h_refine)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr_refine = correlate(h_refine, 1 - h_refine, mode='full')
                c5_refine = np.max(corr_refine) * dx
                
                grad[i] = (c5_plus - c5_refine) / eps
                
            # Update step
            lr = 0.01
            h_new = h_refine - lr * grad
            
            # Projection onto feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_refine = h_new
            
        # Compute final overlap
        corr = correlate(h_refine, 1 - h_refine, mode='full')
        c5_refine = np.max(corr) * dx
        
        if c5_refine < best_c5:
            best_c5 = c5_refine
            best_h = h_refine.copy()
    
    # Final CVX optimization for best solution
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Compute the objective as the max correlation
        # Since correlation is dense, we approximate or work directly with the loss
        
        # Reformulate the problem to work with cvxpy (this is a convex relaxation)
        # Not exact, but will give a good direction
        
        # We introduce an auxiliary variable for the maximum overlap
        obj = cp.Minimize(cp.max(cp.conv(h_vars, 1 - h_vars)))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.CBC, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                corr_cvx = correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(corr_cvx) * dx
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
    except:
        pass
    
    # Final validation
    correlation = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
