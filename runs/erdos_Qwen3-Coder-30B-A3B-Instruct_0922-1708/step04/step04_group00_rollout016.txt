import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Trust region interior-point approach
    
    # Trust region parameters
    max_iter = 500
    trust_radius = 0.1
    trust_radius_decay = 0.99
    min_trust_radius = 1e-6
    
    # Use a gradient-based trust-region method with finite differences
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        # Compute gradient in each direction
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update using trust region
        # Solve: min 0.5 * d^T * H * d + grad^T * d subject to ||d|| <= trust_radius
        # For simplicity, skip H and use projected gradient step
        
        # Projected gradient descent with trust region
        d = -grad
        d_norm = np.linalg.norm(d)
        if d_norm > 0:
            d = d * trust_radius / d_norm
        else:
            d = np.zeros_like(d)
            
        # Take the step, ensuring feasibility
        h_new = h + d
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update trust region radius
        trust_radius = max(min_trust_radius, trust_radius * trust_radius_decay)
        
        h = h_new
        
    # Final check
    correlation = np.correlate(h, 1 - h, mode='full')
    c5_final = np.max(correlation) * dx
    
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
