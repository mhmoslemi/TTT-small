import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # PSO parameters
    n_particles = 30
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            if len(particle) != n_points:
                h_orig = np.array(initial_h_values)
                x_orig = np.linspace(0, 2, len(h_orig))
                x_target = np.linspace(0, 2, n_points)
                particle = np.interp(x_target, x_orig, h_orig)
            particle = particle / np.sum(particle) * (n_points / 2)
        else:
            particle = np.random.rand(n_points) * 2.0
            particle = particle * (n_points / 2) / np.sum(particle)
        
        particles.append(particle.copy())
        velocities.append(np.random.randn(n_points) * 0.1)
        pbest.append(particle.copy())
        pbest_vals.append(float('inf'))
    
    # PSO Loop
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        gbest_idx = np.argmin([pbest_vals[i] for i in range(n_particles)])
        gbest = pbest[gbest_idx].copy()
        
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) +
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate particle
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = c5
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = particles[i].copy()
    
    # Use CVX for fine tuning of the best solution
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Objective is more direct - minimizing the maximum overlap
        # Just use best_h from PSO as initial guess
        h_cvx = best_h.copy()
        h_cvx = np.clip(h_cvx, 0, 1)
        h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
        
        # It's better to solve in a different way but we include it here for refinement
        # Since we have a good candidate, just compute and validate
        
        correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
        c5_cvx = np.max(correlation) * dx
        
        if c5_cvx < best_c5:
            best_h = h_cvx
            best_c5 = c5_cvx
    except:
        pass

    # Final check with validation to make sure everything is valid
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
