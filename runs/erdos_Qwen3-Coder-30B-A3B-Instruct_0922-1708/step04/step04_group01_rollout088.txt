import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:

            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency

    # Evolutionary algorithm approach
    population_size = 100
    generations = 100
    mutation_rate = 0.05
    crossover_rate = 0.5
    elite_size = 10
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    if 'initial_h_values' in globals():
        population = []
        h_init = np.array(initial_h_values) / np.sum(initial_h_values) * (n_points / 2)
        population.append(h_init)
        for _ in range(population_size - 1):
            population.append(create_individual())
    else:
        population = [create_individual() for _ in range(population_size)]
    
    def evaluate_individual(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def select_parents(population, fitnesses):
        # Tournament selection
        selected = []
        for _ in range(population_size):
            idx1, idx2 = np.random.choice(population_size, 2, replace=False)
            if fitnesses[idx1] < fitnesses[idx2]:
                selected.append(population[idx1])
            else:
                selected.append(population[idx2])
        return selected
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy()
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        # Normalize
        child = np.clip(child, 0, 1)
        child = child * (n_points / 2) / np.sum(child)
        return child
    
    def mutate(individual):
        if np.random.rand() > mutation_rate:
            return individual.copy()
        mutated = individual.copy()
        idx = np.random.randint(n_points)
        mutated[idx] = np.random.rand()
        # Normalize
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    # Main evolutionary loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_individual(h) for h in population]
        
        # Update best solution
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Selection, crossover, mutation
        new_population = elite.copy()
        
        # Crossover
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(elite, 2, replace=False)
            child = crossover(parent1, parent2)
            new_population.append(child)
            
        # Mutation
        while len(new_population) < population_size:
            parent = np.random.choice(elite)
            child = mutate(parent)
            new_population.append(child)
            
        population = new_population[:population_size]
        
    # Refine best solution using convex optimization
    try:
        # Create a more sophisticated constraint based optimization
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Approximate the overlap with a max of quadratic expressions
        # Since direct convex formulation is difficult, use iterative approach
        
        # Evaluate C5 based on best_h
        best_c5 = evaluate_individual(best_h)
        
        # Apply smooth optimization on the most promising solutions
        h_opt = best_h.copy()
        
        # Local refinement using scipy minimize
        def objective(h):
            correlation = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(correlation)
            
        # Constraints for scipy optimization
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        def constraint_bounds(h):
            return np.concatenate([h, 1 - h])
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [
            {'type': 'eq', 'fun': constraint_sum},
        ]
        
        result = minimize(objective, h_opt, method='SLSQP', constraints=cons, bounds=bounds, 
                         options={'maxiter': 500, 'ftol': 1e-6})
        
        if result.success:
            h_refined = result.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            refined_c5 = evaluate_individual(h_refined)
            
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = h_refined
                
    except Exception as e:
        pass
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), c5, n_points)
