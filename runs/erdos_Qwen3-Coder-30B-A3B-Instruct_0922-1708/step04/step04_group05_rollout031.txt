import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, dx, epsilon=1e-5):
    """Compute finite difference gradient of C5 w.r.t. h"""
    n_points = len(h)
    grad = np.zeros(n_points)
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project_feasible(h_plus, n_points)
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / epsilon
    return grad

def adaptive_grid_search(initial_h_values, n_points, dx, budget_s, seed=42):
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Coarse grid search
    N_coarse = 100
    n_low = 50
    low_dx = 2.0 / n_low
    coarse_h_list = []
    
    # Generate diverse initial solutions
    for _ in range(N_coarse):
        if np.random.random() < 0.3 and 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        else:
            # Random step function
            h = np.random.rand(n_points)
            h = project_feasible(h, n_points)
        coarse_h_list.append(h)
    
    # Evaluate coarse candidates and refine top performers
    top_candidates = []
    for h in coarse_h_list:
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        top_candidates.append((c5, h))
    
    top_candidates.sort(key=lambda x: x[0])
    
    # Refine top 10% performers
    n_refine = max(1, N_coarse // 10)
    for _, h in top_candidates[:n_refine]:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Local gradient descent
        max_iter = 100
        lr = 0.001
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            grad = compute_gradient(h, dx)
            h = h - lr * grad
            h = project_feasible(h, n_points)
            
            new_c5 = evaluate_c5(h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            
            # Adaptive learning rate
            if iteration > 0 and iteration % 10 == 0:
                lr *= 0.9
            
            # Small random perturbation occasionally to escape local optima
            if iteration % 50 == 0:
                noise = np.random.normal(0, 0.001, n_points)
                h = np.clip(h + noise, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    n_points = 150  # Increased resolution for better accuracy
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.full(n_points, 0.5)
    
    h = project_feasible(h, n_points)
    start_time = time.time()
    
    # First run adaptive grid search
    h, c5 = adaptive_grid_search(initial_h_values, n_points, dx, budget_s, seed)
    
    # Further local refinement if time permits
    while time.time() - start_time < budget_s - 2:
        grad = compute_gradient(h, dx)
        h = h - 0.001 * grad
        h = project_feasible(h, n_points)
        new_c5 = evaluate_c5(h, dx)
        if new_c5 < c5:
            c5 = new_c5
        else:
            break
    
    return (h.tolist(), c5, n_points)
