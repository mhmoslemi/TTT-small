import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 900
    inertia_weight = 0.7
    cognitive_param = 2.0
    social_param = 2.0
    local_search_freq = 50
    rejuvenate_freq = 200
    resolution_steps = [50, 100, 200]
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        swarm.append(h_init)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h_init.copy())
        pbest_fitness.append(float('inf'))
    else:
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
        swarm.append(h_init)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h_init.copy())
        pbest_fitness.append(float('inf'))
    
    # Fill the rest with random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
        pbest_fitness.append(float('inf'))
    
    # Initialize global best
    gbest_idx = 0
    gbest = swarm[0].copy()
    gbest_fitness = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Main PSO loop
    while iteration < max_iter and time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Evaluate all particles
        for i in range(swarm_size):
            h = swarm[i]
            
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(correlation) * dx
            
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest[i] = h.copy()
                
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest = h.copy()
                    gbest_idx = i
        
        # Update swarm
        for i in range(swarm_size):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (inertia_weight * v + 
                 cognitive_param * r1 * (pbest[i] - h) + 
                 social_param * r2 * (gbest - h))
            
            # Update position
            h_new = h + v
            
            # Repair constraints
            h_new = np.clip(h_new, 0, 1)
            total = np.sum(h_new)
            diff = total - n_points / 2
            
            if diff > 1e-8:
                # Excess: reduce largest elements
                sorted_indices = np.argsort(h_new)[::-1]
                for idx in sorted_indices:
                    if diff <= 0:
                        break
                    reduce_amount = min(diff, h_new[idx])
                    h_new[idx] -= reduce_amount
                    diff -= reduce_amount
            elif diff < -1e-8:
                # Deficit: increase smallest elements
                sorted_indices = np.argsort(h_new)
                for idx in sorted_indices:
                    if diff >= 0:
                        break
                    increase_amount = min(-diff, 1 - h_new[idx])
                    h_new[idx] += increase_amount
                    diff += increase_amount
            
            # Normalize to correct sum
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update
            swarm[i] = h_new
            velocities[i] = v
        
        # Local search
        if iteration % local_search_freq == 0:
            # Perform gradient-based refinement on top 10%
            top_indices = np.argsort([pbest_fitness[i] for i in range(swarm_size)])[:swarm_size//10]
            for i in top_indices:
                h = pbest[i]
                for _ in range(10):
                    grad = np.zeros_like(h)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - pbest_fitness[i]) / eps
                    
                    h_new = h - 0.001 * grad
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    
                    corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                    c5_new = np.max(corr_new) * dx
                    
                    if c5_new < pbest_fitness[i]:
                        pbest[i] = h_new
                        pbest_fitness[i] = c5_new
                        
                        if c5_new < gbest_fitness:
                            gbest_fitness = c5_new
                            gbest = h_new
                            gbest_idx = i
        
        # Rejuvenation
        if iteration % rejuvenate_freq == 0:
            # Replace worst 10% particles
            worst_indices = np.argsort([pbest_fitness[i] for i in range(swarm_size)])[-swarm_size//10:]
            for idx in worst_indices:
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                swarm[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                pbest[idx] = h.copy()
                pbest_fitness[idx] = float('inf')
    
    # Final refinement on best solution
    h_final = gbest.copy()
    best_c5 = gbest_fitness
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
            
        grad = np.zeros_like(h_final)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
            
        h_new = h_final - 0.001 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            h_final = h_new
            best_c5 = c5_new
        else:
            break
    
    return (h_final.tolist(), best_c5, n_points)
