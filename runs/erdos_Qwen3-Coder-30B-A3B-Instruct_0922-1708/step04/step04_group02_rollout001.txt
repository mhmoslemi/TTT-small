import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_evaluations = 600  # Roughly budget_s * 100 evaluations per second (approx)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # Default initialization
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Helper functions
    def compute_c5(h_vals):
        # Compute the overlap integral
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def ensure_feasible(h_vals):
        # Clip to [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to sum = n_points / 2
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return h_vals
    
    def mutate_individual(h_vals, mutation_rate):
        """Apply mutation to a single individual"""
        h_new = h_vals.copy()
        for i in range(len(h_vals)):
            if random.random() < mutation_rate:
                # Add small random change
                delta = random.gauss(0, 0.02)
                h_new[i] += delta
        return ensure_feasible(h_new)
    
    def crossover(h1, h2):
        """Create offspring from two parents via uniform crossover"""
        h_new = np.zeros_like(h1)
        for i in range(len(h1)):
            if random.random() < 0.5:
                h_new[i] = h1[i]
            else:
                h_new[i] = h2[i]
        return ensure_feasible(h_new)
    
    def evaluate_population(population):
        """Evaluate all individuals in the population"""
        fitnesses = []
        for indiv in population:
            c5 = compute_c5(indiv)
            fitnesses.append(c5)
        return fitnesses
    
    # Initialize population
    population_size = 50
    population = [ensure_feasible(h + np.random.normal(0, 0.01, n_points)) 
                  for _ in range(population_size)]
    
    # Add original solution to population if it's feasible
    if np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) * dx - 1.0) < 1e-10:
        population = [h] + population[1:]
    
    # Generic algorithm
    evaluations = 0
    epoch = 0
    stagnation_count = 0
    best_in_generation = float('inf')
    
    while time.time() - start_time < budget_s - 1 and evaluations < max_evaluations:
        epoch += 1
        
        # Compute fitness
        fitnesses = evaluate_population(population)
        
        # Find best in this generation
        min_fitness = min(fitnesses)
        if min_fitness < best_in_generation:
            best_in_generation = min_fitness
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Update global best
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
            
        # Selection - keep top 50%
        sorted_indices = np.argsort(fitnesses)
        selected_indices = sorted_indices[:len(sorted_indices)//2]
        selected_population = [population[i] for i in selected_indices]
        
        # Create new population
        new_population = selected_population[:]
        
        # Add mutations and crossovers
        while len(new_population) < population_size:
            if len(selected_population) < 2:
                # If not enough, mutate from existing
                parent = random.choice(selected_population)
                child = mutate_individual(parent, 0.1)
                new_population.append(child)
            else:
                # Crossover 2 random parents
                parent1, parent2 = random.sample(selected_population, 2)
                child = crossover(parent1, parent2)
                # Mutate child
                child = mutate_individual(child, 0.1 + stagnation_count * 0.001)
                new_population.append(child)
        
        population = new_population[:population_size]
        
        # Gradient-based refinement on top candidates occasionally
        if epoch % 10 == 0:
            # Use gradient-based update on best few
            top_k = min(5, len(population))
            top_indices = sorted(range(len(fitnesses)), key=lambda i: fitnesses[i])[:top_k]
            for i in range(top_k):
                idx = top_indices[i]
                h_refined = population[idx].copy()
                
                # Small adaptive gradient descent for 10 iterations
                step_size = 0.001
                for _ in range(10):
                    h_current = h_refined.copy()
                    c5_current = compute_c5(h_current)
                    
                    # Gradient estimation using finite differences
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    
                    for j in range(n_points):
                        h_plus = h_current.copy()
                        h_plus[j] += eps
                        h_plus = ensure_feasible(h_plus)
                        c5_plus = compute_c5(h_plus)
                        grad[j] = (c5_plus - c5_current) / eps
                    
                    # Update
                    h_new = h_current - step_size * grad
                    h_refined = ensure_feasible(h_new)
                    
                    # Accept if improvement
                    c5_new = compute_c5(h_refined)
                    if c5_new < c5_current:
                        h_refined = h_new.copy()
                    
                    h_refined = ensure_feasible(h_refined)
                population[idx] = h_refined
                
        evaluations += population_size
        
    # Final validation
    best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
