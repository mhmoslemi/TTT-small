import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Optimization with hybrid approach
    while time.time() - start_time < budget_s - 5:
        # Compute objective
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Try different optimization strategies
        # Strategy 1: Gradient descent with adaptive learning rate
        if np.random.random() < 0.3:
            h_new = h.copy()
            lr = 0.02
            
            # Simple finite difference gradient
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
        # Strategy 2: Random local search
        else:
            # Random small perturbations
            h_test = h.copy()
            num_perturb = min(5, n_points // 10)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            
            for i in indices:
                # Slightly change value with bounded change
                delta = np.random.uniform(-0.05, 0.05)
                h_test[i] += delta
                h_test[i] = np.clip(h_test[i], 0, 1)
            
            # Normalize
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            c5_test = np.max(corr_test) * dx
            
            if c5_test < c5:
                h = h_test
    
    # Final refinement
    try:
        # Try to improve through close local search
        h_final = best_h.copy()
        orig_c5 = best_c5
        
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            h_test = h_final.copy()
            
            # Small random changes
            idx = np.random.randint(0, n_points)
            h_test[idx] += np.random.uniform(-0.02, 0.02)
            h_test[idx] = np.clip(h_test[idx], 0, 1)
            
            # Ensure normalization
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            c5_test = np.max(corr_test) * dx
            
            if c5_test < orig_c5:
                h_final = h_test
                orig_c5 = c5_test
                
        best_c5 = orig_c5
        best_h = h_final
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
