import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_new = np.linspace(0, 2, n_points)
            h_base = np.interp(x_new, x_orig, h_base)
    else:
        h_base = np.ones(n_points) * 0.5

    # Ensure integral constraint
    h_base = h_base * (n_points / 2) / np.sum(h_base)
    best_h = h_base.copy()
    best_c5 = float('inf')

    def evaluate_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    def project_h(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return h_vals

    def gradient_step(h_vals, lr):
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = evaluate_c5(h_plus)
            c5 = evaluate_c5(h_vals)
            grad[i] = (c5_plus - c5) / eps
        h_new = h_vals - lr * grad
        return project_h(h_new)

    def generate_candidate(h_base):
        # Generate diverse candidates
        noise = np.random.normal(0, 0.005, n_points)
        h = h_base + noise
        h = project_h(h)
        return h

    candidates = [h_base]
    for _ in range(19):
        new_h = generate_candidate(h_base)
        candidates.append(new_h)

    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        current_best = None
        current_best_c5 = float('inf')
        current_candidates = []

        for h in candidates:
            c5 = evaluate_c5(h)
            if c5 < current_best_c5:
                current_best = h
                current_best_c5 = c5
            current_candidates.append(h)

        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best.copy()

        # Gradient-based refinement
        refined = []
        for h in current_candidates:
            lr = 0.01 / (1.0 + iter_count * 0.001)
            for _ in range(5):
                if time.time() - start_time > budget_s - 2:
                    break
                h = gradient_step(h, lr)
            refined.append(h)

        # Introduce new candidates occasionally
        if iter_count % 10 == 0 and len(candidates) < 50:
            for _ in range(3):
                new_h = generate_candidate(best_h)
                refined.append(new_h)

        candidates = refined

    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
