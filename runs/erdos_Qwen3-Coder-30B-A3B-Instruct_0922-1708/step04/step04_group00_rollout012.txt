import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Method 1: CVXPY LP relaxation
    # Formulate the LP relaxation:
    # Variables: h, w (auxiliary), C5
    # minimize C5
    # subject to:
    #   forall k, sum(h(x) - w(x,k)) * dx <= C5
    #   forall k, w(x,k) <= h(x)
    #   forall k, w(x,k) <= h(x+k)
    #   forall k, w(x,k) >= 0
    #   sum(h) * dx = 1
    #   h(x) in [0,1]
    
    # Try LP relaxation first
    try:
        # Create CVXPY variables
        h_vars = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # For each shift, we need auxiliary w variables
        # We'll create constraints for specific shifts to maintain tractability
        # Use only significant shifts (non-zero correlation)
        max_shift = n_points  # This might be overkill but safe
        constraints = []
        
        # Fix h to sum to 1
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Add bounds
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # The main constraints to be approximate
        # We'll compute maximum correlation of h with 1-h
        # and use it as a bound over a set of shifts
        
        # Create minute correlation approximation using looser bounds
        # The core idea: handle one key correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_corr_idx = np.argmax(corr)
        c5_approx = max_corr * dx
        
        # More rigorous: optimize the wire representation while keeping some 
        # constraints in approximate form
        try:
            # Launch a minimal LP relaxation
            prob = cp.Problem(cp.Minimize(C5_var), 
                            [cp.sum(h_vars) == n_points / 2,
                             h_vars >= 0, h_vars <= 1])
            
            # Set up and solve
            prob.solve(solver=GLPK, verbose=False, time_limit=300)
            
            if prob.status == cp.OPTIMAL:
                h_lp = h_vars.value
                # Scale
                h_lp = np.clip(h_lp, 0, 1)
                h_lp = h_lp * (n_points / 2) / np.sum(h_lp)
                h = h_lp.copy()
        except:
            pass
        
        # Second mechanism: iterative local search with smaller steps and annealing
        # Slightly different approach for better local search
        # Use gradient descent, but compute small perturbations
        
        # Model: Current best is already potentially good
        # Refine with Stochastic Gradient Descent
        
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Let's use a stabilized version
        max_iter = 1000
        lr = 0.005
        decay = 0.999
        patience = 0
        min_lr = 1e-6
        min_impr = 1e-6

        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current C5
            corr = np.correlate(h, 1 - h, mode='full')
            c5_old = np.max(corr) * dx
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - c5_old) / eps
            
            # Update h
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute new C5
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < c5_old - min_impr:
                h = h_new.copy()
                patience = 0
            else:
                patience += 1
                
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
            if patience > 100:
                if lr < min_lr:
                    break
                lr *= 0.5
                patience = 0
                
            lr *= decay
            
            if c5_new < 0.3807:
                break # Early exit when we're under the target
                
    except Exception as e:
        pass
    
    # Final refinement with simple LS approach
    try:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
        
        # Local search: 
        h = best_h.copy()
        old_c5 = float('inf')
        for i in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Small perturbations
            for j in range(1, n_points-1):
                h_save = h.copy()
                old_val = h[j]
                
                # Try a small change
                h[j] = min(1.0, max(0.0, old_val + (np.random.rand()-0.5)*0.05))
                h = h * (n_points / 2) / np.sum(h)
                
                corr_new = np.correlate(h, 1-h, mode='full')
                c5_new = np.max(corr_new) * dx
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                else:
                    h = h_save.copy()
                    
            if abs(old_c5 - best_c5) < 1e-6:
                break
            old_c5 = best_c5
            
    except Exception as e:
        pass
    
    # Final verification
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
