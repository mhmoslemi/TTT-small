import time

    c2 = 1.5
    v_max = 0.1
    noise_factor = 0.01
    
    # Initialize PSO swarm
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(n_particles):
        # Add small noise to initial solution
        pos = np.clip(h_init + np.random.normal(0, noise_factor, n_points), 0, 1)
        pos = pos * (n_points / 2) / np.sum(pos)
        particles.append(pos.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        pbest_positions.append(pos.copy())
        pbest_values.append(float('inf'))
    
    gbest_position = h_init.copy()
    gbest_value = float('inf')
    
    iter_count = 0
    no_improvement_count = 0
    
    def evaluate_c5(h_data):
        corr = np.correlate(h_data, 1 - h_data, mode='full')
        return np.max(corr) * dx
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > max_time:
            break
        
        # Evaluate all particles
        for i in range(n_particles):
            pos = particles[i]
            # Ensure constraint
            pos = pos * (n_points / 2) / np.sum(pos)
            particles[i] = pos
            
            # Evaluate fitness
            fitness = evaluate_c5(pos)
            
            # Update personal best
            if fitness < pbest_values[i]:
                pbest_values[i] = fitness
                pbest_positions[i] = pos.copy()
            
            # Update global best
            if fitness < gbest_value:
                gbest_value = fitness
                gbest_position = pos.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1

            velocities[i] = vel

            pos = pos + vel

        # Decay inertia weight

        # Early termination if no improvement for 50 iterations

                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                corr = np.correlate(h_refine, 1 - h_refine, mode='full')
                c5 = np.max(corr) * dx
                gradient[j] = (c5_plus - c5) / eps

            # Move in direction of negative gradient
            lr = 0.005 / (1 + np.linalg.norm(gradient) * 1e-2)

            new_h = h_refine + delta

                h_refine = new_h
                ref_c5 = new_c5
            else:
                break
        candidates.append(h_refine)
    
    # Select best from all candidates
    final_h = min(candidates, key=evaluate_c5)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
