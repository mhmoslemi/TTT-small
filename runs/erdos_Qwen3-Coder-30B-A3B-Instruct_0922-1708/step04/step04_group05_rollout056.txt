import numpy as np
import time
from scipy.optimize import minimize_scalar

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, n_points, eps=1e-4):
    grad = np.zeros_like(h)
    base_c5 = evaluate_c5(h, n_points)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        plus_c5 = evaluate_c5(h_plus, n_points)
        grad[i] = (plus_c5 - base_c5) / eps
    return grad

def line_search(h, grad, n_points, max_alpha=1.0):
    def phi(alpha):
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n_points)
        return evaluate_c5(h_new, n_points)
    
    # Golden section search
    golden_ratio = (np.sqrt(5) - 1) / 2
    a, b = 0.0, max_alpha
    resphi = 1.0 - golden_ratio
    
    x1 = a + resphi * (b - a)
    x2 = a + golden_ratio * (b - a)
    f1 = phi(x1)
    f2 = phi(x2)
    
    while abs(b - a) > 1e-6:
        if f1 < f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = a + resphi * (b - a)
            f1 = phi(x1)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + golden_ratio * (b - a)
            f2 = phi(x2)
    
    return (a + b) / 2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = project_feasible(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    
    # Adaptive sampling approach
    coarse_grid_iter = 0
    fine_refine_iter = 0
    max_coarse = 100
    max_fine = 100
    
    # Initial coarse sampling
    candidates = []
    for i in range(max_coarse):
        if time.time() - start_time > budget_s - 5:
            break
        h_temp = np.random.rand(n_points)
        h_temp = project_feasible(h_temp, n_points)
        c5 = evaluate_c5(h_temp, n_points)
        candidates.append((h_temp, c5))
        coarse_grid_iter += 1
    
    # Sort and refine top candidates
    candidates.sort(key=lambda x: x[1])
    top_candidates = candidates[:max(min(10, len(candidates)), 2)]
    
    for h_init, _ in top_candidates:
        if time.time() - start_time > budget_s - 5:
            break
        h = h_init.copy()
        for _ in range(50):
            if time.time() - start_time > budget_s - 2:
                break
            grad = compute_gradient(h, n_points)
            alpha = line_search(h, grad, n_points)
            h = h - alpha * grad
            h = project_feasible(h, n_points)
            
            c5 = evaluate_c5(h, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Random noise injection
            if np.random.rand() < 0.05:
                noise = np.random.normal(0, 0.001, n_points)
                h = h + noise
                h = project_feasible(h, n_points)
            
        fine_refine_iter += 1
    
    final_c5 = evaluate_c5(best_h, n_points)
    return (best_h.tolist(), final_c5, n_points)
