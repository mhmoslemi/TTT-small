import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation
    # Minimize sum of all overlaps (not max) to get a good starting point
    # This is a relaxation of the true problem
    try:
        # Use CVXPY for convex relaxation
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Construct matrix for overlap computation
        # We'll compute sum of all correlations
        # This is a linear approximation of the problem
        objective = cp.Minimize(cp.sum_squares(h_var) + 
                              cp.sum_squares(1 - h_var))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h = np.array(h_var.value).reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        # If convex optimization fails, continue with uniform
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Phase 2: Local refinement with projected gradient descent
    max_iter = int(1000)
    lr = 0.01
    patience = 50
    no_improvement_count = 0
    
    def compute_obj_and_grad(h):
        # Compute overlap integral efficiently
        corr = np.correlate(h, 1 - h, mode='full')
        obj = np.max(corr) * dx
        
        # Compute gradients via finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) * dx - obj) / eps
        return obj, grad
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        obj, grad = compute_obj_and_grad(h)
        
        if obj < best_c5:
            best_c5 = obj
            best_h = h.copy()
            
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if updated solution improves
        corr = np.correlate(h_new, 1 - h_new, mode='full')
        new_obj = np.max(corr) * dx
        
        if new_obj < obj:
            h = h_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Reduce learning rate
            lr *= 0.99
            
        if no_improvement_count > patience:
            break
            
    return (best_h.tolist(), best_c5, n_points)
