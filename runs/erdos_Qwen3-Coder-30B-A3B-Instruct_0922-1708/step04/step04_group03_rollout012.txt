import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Convex relaxation to get better initial guess
    # Relaxation: minimize sum of all overlaps, which is a convex problem
    # This is easier to optimize than the max function directly
    
    # Variables: h[i] for i = 0..n_points-1
    h_vars = cp.Variable(n_points)
    
    # Objective: sum of (h[i] * (1-h[(i+j) % n_points])) for j = 0..n_points-1
    # This is the total overlap over all k, so we minimize that
    # But we need to be careful: we want to minimize max_k integral
    # So we'll use a more direct approach using finite differences
    
    # Phase 1: More robust approach using projected gradient descent from either initial or random
    # Start with a refined initialization
    
    # Evaluate current solution
    def eval_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Gradient approximation with finite differences
    def compute_gradient(h_vec, eps=1e-5):
        grad = np.zeros_like(h_vec)
        c5_base = eval_c5(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            # Project back to feasible set
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = eval_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    # Main optimization loop
    max_iter = 2000
    no_improvement_limit = 50
    no_improvement_count = 0
    lr = 0.01
    
    h = h.copy()
    best_c5 = eval_c5(h)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        if iteration % 10 == 0:
            # Periodic evaluation
            current_c5 = eval_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            
            if no_improvement_count >= no_improvement_limit:
                break
            
            # Decay learning rate
            if iteration % 100 == 0:
                lr *= 0.95
        
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update step
        h_new = h - lr * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final check
    current_c5 = eval_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
