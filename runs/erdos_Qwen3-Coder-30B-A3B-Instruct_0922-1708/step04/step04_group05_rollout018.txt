import time

def check_feasible(h, n_points):
    """Check if h satisfies all constraints"""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n_points/2) < 1e-10)

def evaluate_c5(h, n_points):
    """Evaluate C5 using full convolution"""
    dx = 2.0 / n_points
    corr_full = np.correlate(h, 1 - h, mode='full')
    return np.max(corr_full) * dx
