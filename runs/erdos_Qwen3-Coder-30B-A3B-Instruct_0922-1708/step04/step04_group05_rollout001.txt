import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise use uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Ensure the integral constraint sum(h) * dx = 1
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h = h_init.copy()
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # PSO parameters
    n_particles = 20
    max_iter = 200
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    inertia_decay = (w_start - w_end) / max_iter
    
    # Initialize PSO particles
    particles = []
    velocities = []
    best_positions = []
    best_costs = []
    
    for _ in range(n_particles):
        pos = h_init + np.random.normal(0, 0.05, n_points)
        pos = np.clip(pos, 0, 1)
        pos = pos * (n_points / 2) / np.sum(pos)
        particles.append(pos.copy())
        velocities.append(np.random.normal(0, 0.01, n_points))
        best_positions.append(pos.copy())
        best_costs.append(float('inf'))
    
    # PSO loop
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Compute fitness and update personal bests
        current_swarm_best_cost = float('inf')
        current_swarm_best_pos = None
        
        for i in range(n_particles):
            pos = particles[i]
            # Ensure it's feasible
            pos = np.clip(pos, 0, 1)
            pos = pos * (n_points / 2) / np.sum(pos)
            particles[i] = pos
            
            # Evaluate
            corr = np.correlate(pos, 1 - pos, mode='full')
            cost = np.max(corr) * dx
            
            if cost < best_costs[i]:
                best_costs[i] = cost
                best_positions[i] = pos.copy()
                
            if cost < current_swarm_best_cost:
                current_swarm_best_cost = cost
                current_swarm_best_pos = pos.copy()
                
        # Update global best
        if current_swarm_best_cost < best_c5:
            best_c5 = current_swarm_best_cost
            best_h = current_swarm_best_pos.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adapt population if no improvement
        if no_improvement_count > 50 and n_particles > 5:
            n_particles -= 5
            particles = particles[:n_particles]
            velocities = velocities[:n_particles]
            best_positions = best_positions[:n_particles]
            best_costs = best_costs[:n_particles]
            
        # Update inertia weight
        w = w_start - iter_count * inertia_decay
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            v = w * velocities[i] + c1 * r1 * (best_positions[i] - particles[i]) + \
                c2 * r2 * (current_swarm_best_pos - particles[i])
            velocities[i] = v
            particles[i] = particles[i] + v
            
            # Add small noise periodically to prevent falling into local minima
            if iter_count % 20 == 0:
                noise = np.random.normal(0, 0.001, n_points)
                particles[i] = particles[i] + noise
                # Ensure feasibility
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Refinement with gradient ascent (line search)
    refined_h = best_h.copy()
    refined_c5 = best_c5
    
    # Compute initial cost and gradient approximations for refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        base_corr = np.correlate(refined_h, 1 - refined_h, mode='full')
        base_cost = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = refined_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            cost_plus = np.max(corr_plus) * dx
            grad[i] = (cost_plus - base_cost) / eps
            
        # Update with negative gradient
        step_size = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
        refined_h = refined_h - step_size * grad
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
        
        # Recalculate
        corr = np.correlate(refined_h, 1 - refined_h, mode='full')
        updated_c5 = np.max(corr) * dx
        if updated_c5 < refined_c5:
            refined_c5 = updated_c5
        else:
            break
            
    # Final evaluation
    corr = np.correlate(refined_h, 1 - refined_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    if final_c5 < best_c5:
        best_h = refined_h
        best_c5 = final_c5
        
    return (best_h.tolist(), best_c5, n_points)
