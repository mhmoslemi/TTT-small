import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Differential Evolution parameters
    pop_size = 50
    F = 0.8
    CR = 0.9
    max_no_improvement = 100
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            individual = h.copy()
        else:
            individual = np.random.uniform(0, 1, n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        correlation = np.correlate(individual, 1 - individual, mode='full')
        fit = np.max(correlation) * dx
        fitness.append(fit)
        if fit < best_c5:
            best_c5 = fit
            best_h = individual.copy()
    
    gbest_idx = np.argmin(fitness)
    gbest = population[gbest_idx].copy()
    gbest_val = fitness[gbest_idx]
    
    no_improvement_count = 0
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        improved = False
        
        # DE mutation and crossover
        new_population = []
        new_fitness = []
        
        for i in range(pop_size):
            # Mutation
            idxs = [idx for idx in range(pop_size) if idx != i]
            a, b, c = population[np.random.choice(idxs, 3, replace=False)]
            
            # Differential mutation
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (n_points / 2) / np.sum(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            j_rand = np.random.randint(0, n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
            
            # Evaluate trial
            correlation = np.correlate(trial, 1 - trial, mode='full')
            c5 = np.max(correlation) * dx
            
            # If trial is better, replace
            if c5 < fitness[i]:
                new_population.append(trial)
                new_fitness.append(c5)
                improved = True
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = trial.copy()
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
        
        population = new_population
        fitness = new_fitness
        
        # Local refinement on the best
        if iteration % 20 == 0:
            h = gbest.copy()
            for refine_iter in range(20):
                correlation = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(correlation) * dx
                
                # Simple gradient approximation
                grad = np.zeros_like(h)
                eps = 1e-5
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5) / eps
                
                # Gradient descent step
                lr = 0.02 / (1 + np.linalg.norm(grad) * 1e-2)
                h_new = h - lr * grad
                
                # Enforce constraints
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                h = h_new
                
                # Evaluate
                corr = np.correlate(h, 1 - h, mode='full')
                c5_refine = np.max(corr) * dx
                
                if c5_refine < best_c5:
                    best_c5 = c5_refine
                    best_h = h.copy()
            
            gbest = h.copy()
            gbest_val = best_c5
            
        # Update global best
        gbest_idx = np.argmin(fitness)
        if fitness[gbest_idx] < gbest_val:
            gbest = population[gbest_idx].copy()
            gbest_val = fitness[gbest_idx]
            improved = True
            
        # Early stopping
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
