import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Refine with convex optimization
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    max_iterations = 200
    
    for iter_num in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Build optimization model
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            C >= 0
        ]
        
        # Add overlap constraints for all shifts
        for k in range(1, n_points):
            # Compute correlation term for shift k
            # h[i] * (1 - h[i + k]) for valid i
            shift_idx = slice(k, None)
            valid_len = n_points - k
            if valid_len > 0:
                corr = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[shift_idx]))
                constraints.append(corr <= C)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, abstol=1e-5, reltol=1e-5)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                if h_sol is not None:
                    # Project to feasible region
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    
                    # Compute actual C5
                    correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                    c5 = np.max(correlation) * dx
                    
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_sol.copy()
        except Exception:
            pass
        
        # Use the current solution for next iteration
        h = best_h.copy()
    
    # Final refinement with gradient descent
    h = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Use finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-4)
        h_new = h - lr * grad
        
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Check if we improved
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
