import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Trust region optimization parameters
    trust_radius = 0.1
    max_iter = 500
    decay_rate = 0.99
    min_radius = 1e-6
    
    # Function to compute C5
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Gradient computation via finite differences
    def compute_gradient(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        c5_base = compute_c5(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    # Update step using trust region
    def update_trust_region(h_vec, grad, trust_radius):
        # Simple trust region update: constrain step to trust region
        step = -grad
        step_norm = np.linalg.norm(step)
        if step_norm > trust_radius:
            step = step * trust_radius / step_norm
        h_new = h_vec + step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5 and gradient
        c5_current = compute_c5(h)
        grad = compute_gradient(h)
        
        # Update best solution
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Update trust region
        trust_radius = max(min_radius, trust_radius * decay_rate)
        
        # Compute new solution
        h_new = update_trust_region(h, grad, trust_radius)
        
        # Accept step if it improves C5
        c5_new = compute_c5(h_new)
        if c5_new < c5_current:
            h = h_new
        else:
            # If no improvement, try a smaller trust radius
            trust_radius = trust_radius * 0.5
    
    # Final check
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
