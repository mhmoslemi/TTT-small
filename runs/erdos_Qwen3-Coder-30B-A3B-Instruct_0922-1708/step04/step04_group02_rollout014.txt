import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 900
    
    # Particle Swarm Optimization with local search acceleration
    
    def initialize_swarm():
        if 'initial_h_values' in globals():
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        else:
            h = np.random.rand(n_points)
        
        h = h * (n_points / 2) / np.sum(h)
        swarm = []
        velocities = []
        for _ in range(swarm_size):
            particle = h + np.random.normal(0, 0.01, n_points)
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            swarm.append(particle)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        return swarm, velocities
    
    def repair_particle(h):
        # Clip to bounds
        h = np.clip(h, 0, 1)
        # Adjust to sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2) > 1e-8:
            diff = current_sum - n_points/2
            if diff > 0:
                # Reduce from largest elements
                indices = np.argsort(h)[::-1]
                for i in indices:
                    if diff <= 0:
                        break
                    decrease = min(h[i], diff)
                    h[i] -= decrease
                    diff -= decrease
            else:
                # Increase in smallest elements
                indices = np.argsort(h)
                for i in indices:
                    if diff >= 0:
                        break
                    increase = min(1 - h[i], -diff)
                    h[i] += increase
                    diff += increase
        return h
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    # Initialize
    swarm, velocities = initialize_swarm()
    
    # Track best positions
    pbest = [swarm[i].copy() for i in range(swarm_size)]
    pbest_fitness = [compute_c5(swarm[i]) for i in range(swarm_size)]
    
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # PSO parameters
    w = 0.7  # Inertia weight
    c1, c2 = 2.0, 2.0  # Acceleration coefficients
    
    start_time = time.time()
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 5:
        iter_count += 1
        
        # Evaluate particles and update personal bests
        for i in range(swarm_size):
            fitness = compute_c5(swarm[i])
            if fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fitness
                
                if fitness < gbest_fitness:
                    gbest = swarm[i].copy()
                    gbest_fitness = fitness
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            swarm[i] += velocities[i]
            swarm[i] = repair_particle(swarm[i])
        
        # Local search every 50 iterations
        if iter_count % 50 == 0:
            # Gradient-based refinement on top 10% swarm
            top_k = int(swarm_size * 0.1)
            indices = np.argsort(pbest_fitness)[:top_k]
            
            for idx in indices:
                h_current = pbest[idx].copy()
                current_fitness = pbest_fitness[idx]
                
                lr = 0.001
                for _ in range(10):
                    # Approximate gradient by finite differences
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h_current.copy()
                        h_plus[j] += eps
                        h_plus = repair_particle(h_plus)
                        c_plus = compute_c5(h_plus)
                        grad[j] = (c_plus - current_fitness) / eps
                    
                    h_new = h_current - lr * grad
                    h_new = repair_particle(h_new)
                    
                    new_fitness = compute_c5(h_new)
                    if new_fitness < current_fitness:
                        h_current = h_new
                        current_fitness = new_fitness
                    else:
                        break
                
                # Update if improved
                if current_fitness < pbest_fitness[idx]:
                    pbest[idx] = h_current.copy()
                    pbest_fitness[idx] = current_fitness
                    
                    if current_fitness < gbest_fitness:
                        gbest = h_current.copy()
                        gbest_fitness = current_fitness
        
        # Rejuvenation every 200 iterations
        if iter_count % 200 == 0:
            # Replace worst 10% with random particles
            worst_indices = np.argsort(pbest_fitness)[-int(swarm_size * 0.1):]
            for idx in worst_indices:
                swarm[idx] = np.random.rand(n_points)
                swarm[idx] = swarm[idx] * (n_points / 2) / np.sum(swarm[idx])
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
    
    # Final refinement with gradient descent on gbest
    h_refined = gbest.copy()
    lr = 0.001
    for _ in range(100):
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_particle(h_plus)
            c_plus = compute_c5(h_plus)
            grad[j] = (c_plus - gbest_fitness) / eps
            
        h_new = h_refined - lr * grad
        h_new = repair_particle(h_new)
        new_fitness = compute_c5(h_new)
        if new_fitness < gbest_fitness:
            h_refined = h_new
            gbest_fitness = new_fitness
        else:
            break
            
    return (h_refined.tolist(), gbest_fitness, n_points)
