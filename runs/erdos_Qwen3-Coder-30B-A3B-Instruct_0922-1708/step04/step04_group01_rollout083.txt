import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Grid search parameters
    coarse_n = 20
    fine_n = 100
    n_top = 10
    max_iter = 500
    lr = 0.01
    
    start_time = time.time()
    
    # Generate coarse grid and evaluate
    best_c5 = float('inf')
    best_h = None
    
    # Start with initial solution if available
    if 'initial_h_values' in globals():
        coarse_h = np.array(initial_h_values)
        if len(coarse_h) != coarse_n:
            x_orig = np.linspace(0, 2, len(initial_h_values))
            x_target = np.linspace(0, 2, coarse_n)
            coarse_h = np.interp(x_target, x_orig, initial_h_values)
        coarse_h = coarse_h / np.sum(coarse_h) * (coarse_n / 2)
    else:
        coarse_h = np.ones(coarse_n) * 0.5
    
    # Store top candidates
    candidates = []
    
    # Grid search over specific patterns for coarse resolution
    grid_points = 100
    for i in range(grid_points):
        if time.time() - start_time > budget_s - 5:
            break
        # Random sample
        h_temp = np.random.rand(coarse_n)
        h_temp = h_temp / np.sum(h_temp) * (coarse_n / 2)
        
        # Compute overlap
        corr = np.correlate(h_temp, 1 - h_temp, mode='full')
        c5 = np.max(corr) * (2.0 / coarse_n)
        
        if not np.isnan(c5) and c5 < best_c5:
            best_c5 = c5
            best_h = h_temp.copy()
        
        # Store top candidates
        candidates.append((c5, h_temp.copy()))
        candidates.sort(key=lambda x: x[0])
        if len(candidates) > n_top:
            candidates.pop()
    
    # Refinement phase
    for _, h_seed in candidates:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Upsample
        h = np.interp(np.linspace(0, 2, fine_n), np.linspace(0, 2, coarse_n), h_seed)
        h = h * (fine_n / 2) / np.sum(h)
        
        # Gradient-based refinement
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(fine_n):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (fine_n / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * (2.0 / fine_n)
                
                corr = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(corr) * (2.0 / fine_n)
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (fine_n / 2) / np.sum(h_new)
            
            h = h_new
            
            # Evaluate
            corr = np.correlate(h, 1 - h, mode='full')
            c5_new = np.max(corr) * (2.0 / fine_n)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    # Final convex optimization if time allows
    if time.time() - start_time < budget_s - 2:
        try:
            h_vars = cp.Variable(fine_n)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == fine_n / 2]
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (fine_n, 1)))))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL and prob.value < best_c5:
                h_cvx = h_vars.value
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (fine_n / 2) / np.sum(h_cvx)
                
                corr = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(corr) * (2.0 / fine_n)
                
                if c5_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c5_cvx
        except:
            pass
    
    if best_h is None:
        # Fall back to coarse grid result
        best_h = h_seed.copy()
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * (2.0 / fine_n)
    
    return (best_h.tolist(), best_c5, fine_n)
