import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a default construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Random initialization approach
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Store best solution seen so far
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Relaxation stage with CVXPY
    if time.time() - start_time < budget_s * 0.4:
        try:
            # Relaxation optimization (simplified)
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            objective = cp.Minimize(cp.sum(h_vars))  # placeholder objective
            
            # Need to approximate max contribution, not easy to solve directly
                
            # Just try basic relaxation by sampling  
            h_relaxed = h.copy()
            # Start with smooth solution as basic heuristic 
        except:
            h_relaxed = h.copy()
    
    # Round to binary with thresholding
    h_candidate = np.zeros(n_points)
    t = np.median(h_relaxed)
    for i in range(n_points):
        h_candidate[i] = 1 if h_relaxed[i] > t else 0
    
    # Correct the sum to meet constraint
    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
    
    # Final optimization stage
    def objective(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    current_time = time.time()
    
    # Mechanical optimization approach
    trials = 0
    while current_time - start_time < budget_s - 1 and trials < 2000:
        trials += 1
        h_new = h_candidate.copy()
        
        # Small mutations
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.05, 0.05)
        h_new[idx] += delta
        
        # Enforce bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = objective(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        current_time = time.time()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    # After major optimization, standard fine-tune
    for _ in range(100):
        h_new = best_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.01, 0.01)
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = objective(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
