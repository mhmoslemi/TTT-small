import numpy as np
import time
from heapq import heappush, heappop
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters
    w = 5  # window width for localized adjustments
    pert_scale = 0.02
    min_improvement = 1e-6
    max_no_improvement = 20
    
    # Priority queue for shifts (max heap simulated with negative values)
    shift_queue = []
    
    def compute_overlap(h):
        # Compute all overlaps for all shifts
        overlaps = []
        # Use convolution with reflection
        for k in range(-(n_points-1), n_points):
            if k == 0:
                continue
            # Compute overlap for shift k
            if k > 0:
                # h[0:-k] * (1 - h[k:])
                overlap = np.sum(h[:-k] * (1 - h[k:])) * dx
            else:
                # h[-k:] * (1 - h[:k])
                overlap = np.sum(h[-k:] * (1 - h[:k])) * dx
            overlaps.append((overlap, k))
        return overlaps
    
    def update_shift_queue(h):
        overlaps = compute_overlap(h)
        # Reset queue
        shift_queue.clear()
        for overlap, k in overlaps:
            heappush(shift_queue, (-overlap, k))
        return overlaps
    
    def apply_adjustments(h, k, w):
        h_new = h.copy()
        if k > 0:
            # h[0:-k] * (1 - h[k:])
            # Adjust both sides
            start_i = max(0, w - k)
            end_i = min(n_points - k, n_points - w)
            if end_i > start_i:
                # Reduce h in problematic region
                for i in range(start_i, end_i):
                    h_new[i] = max(0, h_new[i] - 0.005 * (1 - 2 * h_new[i]))
                # Increase in corresponding area
                if w < n_points:
                    for i in range(max(0, w - k), min(n_points - k, w)):
                        h_new[i] = min(1.0, h_new[i] + 0.005)
        else:
            # k < 0
            # h[-k:] * (1 - h[:k])
            start_i = max(0, -k + w)
            end_i = min(n_points, n_points + w + k)
            if end_i > start_i:
                # Reduce h in problematic region
                for i in range(start_i, end_i):
                    h_new[i] = max(0, h_new[i] - 0.005 * (1 - 2 * h_new[i]))
                # Increase in corresponding area
                if w + k < n_points:
                    for i in range(max(0, w + k), min(n_points, w)):
                        h_new[i] = min(1.0, h_new[i] + 0.005)
        return h_new
    
    def random_perturb(h, w, scale):
        h_new = h.copy()
        # Randomly adjust a few points
        for _ in range(5):
            i = np.random.randint(0, n_points)
            # Apply small local adjustment
            h_new[i] = max(0, min(1, h_new[i] + np.random.normal(0, scale)))
        return h_new
    
    # Initialize the queue
    start_overlaps = compute_overlap(h)
    for overlap, k in start_overlaps:
        heappush(shift_queue, (-overlap, k))
    
    no_improvement_count = 0
    last_improvement = -1
    
    max_iter = 800
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update overlap computation
        cumulative = np.array([0.0] * (2 * n_points - 1))
        for k in range(-(n_points-1), n_points):
            if k == 0:
                continue
            c = correlate(h, 1 - h, mode='full')
            overlap = c[n_points-1+k] * dx
            cumulative[n_points-1+k] = overlap
        
        # Update global best
        max_c5 = np.max(cumulative)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
            no_improvement_count = 0
            last_improvement = iteration
        else:
            no_improvement_count += 1
            
        # Adaptive strategies
        if no_improvement_count > max_no_improvement:
            # Try perturbation if no progress
            if no_improvement_count % 5 == 0:
                h = random_perturb(h, w, pert_scale)
                # restore constraint
                h = h * (n_points / 2) / np.sum(h)
            
            if no_improvement_count % 10 == 0:
                # Increase window if no progress
                w = min(w + 1, 10)
                # Increase perturbation scale if needed
                pert_scale *= 1.2
                
            no_improvement_count = 0
        
        # Select worst shift and apply improvement
        worst_overlap = -heappop(shift_queue)[0]
        k = shift_queue[0][1] if shift_queue else 0  # synthetic worst
        
        # More efficient method for overlapping:
        # Get indices quickly for a given shift k
        if k > 0:
            # Left part of h multiplies with right part of (1-h)
            overlap_idx = k + n_points - 1
            if overlap_idx < len(cumulative): 
                # Apply adjustment directly to h to reduce this overlap
                start_i = max(0, w - k)
                end_i = min(n_points - k, n_points - w)
                for i in range(start_i, end_i):
                    h[i] = max(0, h[i] - 0.002 * (1 - 2 * h[i]))
                for i in range(max(0, w - k), min(n_points - k, w)):
                    h[i] = min(1.0, h[i] + 0.002)
            else:
                # Random perturb otherwise
                h = random_perturb(h, w, pert_scale)
        else:
            # for k < 0
            overlap_idx = n_points - 1 + k
            if overlap_idx >= 0 and overlap_idx < len(cumulative):
                # Apply update
                start_i = max(0, -k + w)
                end_i = min(n_points, n_points + w + k)
                for i in range(start_i, end_i):
                    h[i] = max(0, h[i] - 0.002 * (1 - 2 * h[i]))
                for i in range(max(0, w + k), min(n_points, w)):
                    h[i] = min(1.0, h[i] + 0.002)
            else:
                # Random perturb
                h = random_perturb(h, w, pert_scale)
                
        h = h * (n_points / 2) / np.sum(h)  # normalize
    return (best_h.tolist(), best_c5, n_points)
