import numpy as np
import time

def project_feasible(h, n_points):
    """Project h to satisfy bounds and integral constraint."""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_c5(h):
    """Compute C5 from h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def golden_section_search(f, a, b, tol=1e-5, max_iter=50):
    """Golden section search for step size."""
    gr = (1 + np.sqrt(5)) / 2
    res = (gr - 1) / gr
    c = a + res * (b - a)
    d = b - res * (b - a)
    for _ in range(max_iter):
        if abs(c - d) < tol:
            break
        if f(c) < f(d):
            b = d
            d = c
            c = a + res * (b - a)
        else:
            a = c
            c = d
            d = b - res * (b - a)
    return (a + b) / 2

def gradient_step(h, n_points, dx):
    """Compute gradient and perform step."""
    eps = 1e-4
    grad = np.zeros(n_points)
    base_c5 = compute_c5(h)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_feasible(h_plus, n_points)
        plus_c5 = compute_c5(h_plus)
        grad[i] = (plus_c5 - base_c5) / eps
    
    lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n_points)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize from parent if available, otherwise default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5

    h = project_feasible(h, n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Outer loop with grid search
    N_coarse = 100
    n_low = 50
    
    # Coarse grid
    for i in range(N_coarse):
        if time.time() - start_time > budget_s - 5:
            break
        h_test = np.random.rand(n_low)
        h_test = project_feasible(h_test, n_low)
        # Interpolate to full size
        h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_low), h_test)
        h_interp = project_feasible(h_interp, n_points)
        
        # Refine with gradient descent
        h_refined = h_interp.copy()
        for _ in range(50):  # Local iterations
            if time.time() - start_time > budget_s - 5:
                break
            h_refined = gradient_step(h_refined, n_points, dx)
            c5 = compute_c5(h_refined)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_refined.copy()
    
    # Final refinement
    h_final = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 2:
            break
        h_final = gradient_step(h_final, n_points, dx)
        c5 = compute_c5(h_final)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final.copy()
    
    return (best_h.tolist(), best_c5, n_points)
