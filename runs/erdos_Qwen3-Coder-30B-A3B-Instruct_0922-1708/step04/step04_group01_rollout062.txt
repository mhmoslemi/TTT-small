import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using the provided parent construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * target_sum
    else:
        # Start with a symmetric construction
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = float('inf')
    
    # Evolutionary Algorithm with binary representation
    population_size = 100
    generations = 100
    mutation_rate = 0.05
    crossover_rate = 0.5
    elite_size = 10
    start_time = time.time()
    
    # Create initial population with binary constraints (exact sum)
    def create_binary_individual():
        # Create binary array with exactly target_sum ones
        idx = np.random.choice(n_points, int(target_sum), replace=False)
        ind = np.zeros(n_points)
        ind[idx] = 1
        return ind
        
    def binary_to_feasible(h_binary):
        h_float = h_binary.astype(float)
        h_float = h_float * target_sum / np.sum(h_float)
        return np.clip(h_float, 0, 1)
        
    # Initialize population
    population = [create_binary_individual() for _ in range(population_size)]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness (C5) for all individuals
        fitness = []
        for ind in population:
            # Convert to binary if needed
            h = binary_to_feasible(ind)
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = binary_to_feasible(population[0])
            
        if gen % 10 == 0:
            print(f"Gen {gen}: Best C5 = {best_c5:.6f}")
            
        # Create new population
        new_population = []
        
        # Elite
        for i in range(elite_size):
            new_population.append(population[i].copy())
            
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate:
                # Perform crossover
                parent1, parent2 = np.random.choice(population[:elite_size], 2, replace=False)
                crossover_point = np.random.randint(1, n_points-1)
                child = np.zeros(n_points)
                child[:crossover_point] = parent1[:crossover_point]
                child[crossover_point:] = parent2[crossover_point:]
                # Fix to have exact sum
                new_population.append(child)
            else:
                # Mutation
                parent = population[np.random.randint(elite_size)]
                child = parent.copy()
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] = 1 - child[i]
                # Fix to have exact sum
                new_population.append(child)
                
        # Maintain exact constraint
        for i in range(len(new_population)):
            # Rescale to correct sum
            h_rescaled = binary_to_feasible(new_population[i])
            new_population[i] = h_rescaled
            
        population = new_population
        
    # Post-processing with convex optimization 
    def refine_solution(h_guess):
        try:
            h_vars = cp.Variable(n_points)
            constraints = [
                h_vars >= 0,
                h_vars <= 1,
                cp.sum(h_vars) == target_sum
            ]
            
            # Calculate overlap
            corr = cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))
            objective = cp.Minimize(cp.max(corr))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_refined = h_vars.value
                if h_refined is not None:
                    correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
                    c5_refined = np.max(correlation) * dx
                    if c5_refined < best_c5:
                        return h_refined, c5_refined
        except:
            pass
        return h_guess, best_c5
        
    # Try to refine best solution
    h_final, c5_final = refine_solution(best_h)
    
    if c5_final < best_c5:
        best_h = h_final
        best_c5 = c5_final
        
    # Final check using the full correlation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
