import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a simple piecewise constant construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Evolve a population using genetic algorithm
    pop_size = 100
    generations = 100
    mutation_rate = 0.05
    crossover_rate = 0.5
    selection_rate = 0.1
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            # Use initial construction as first individual
            ind = h.copy()
        else:
            # Random binary construction with sum = n_points/2
            ind = np.zeros(n_points)
            ones_indices = np.random.choice(n_points, int(n_points/2), replace=False)
            ind[ones_indices] = 1.0
            
        # Ensure sum constraint
        current_sum = np.sum(ind)
        if current_sum != n_points / 2:
            diff = int(n_points/2 - current_sum)
            if diff > 0:
                # Add more 1s
                zero_indices = np.where(ind == 0)[0]
                add_indices = np.random.choice(zero_indices, diff, replace=False)
                ind[add_indices] = 1.0
            else:
                # Remove some 1s
                one_indices = np.where(ind == 1)[0]
                remove_indices = np.random.choice(one_indices, -diff, replace=False)
                ind[remove_indices] = 0.0
                
        population.append(ind)
    
    # Genetic algorithm loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for ind in population:
            corr = np.correlate(ind, 1 - ind, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        # Selection
        num_selected = max(1, int(pop_size * selection_rate))
        selected = population[:num_selected]
        selected_fitness = fitness[:num_selected]
        
        # Create new population
        new_population = []
        
        # Keep best individuals
        new_population.extend(selected)
        
        # Crossover
        while len(new_population) < pop_size:
            if np.random.random() < crossover_rate and len(selected) >= 2:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Enforce sum constraint
                def enforce_sum(x):
                    current_sum = np.sum(x)
                    diff = int(n_points/2 - current_sum)
                    if diff > 0:
                        zero_indices = np.where(x == 0)[0]
                        add_indices = np.random.choice(zero_indices, diff, replace=False)
                        x[add_indices] = 1.0
                    elif diff < 0:
                        one_indices = np.where(x == 1)[0]
                        remove_indices = np.random.choice(one_indices, -diff, replace=False)
                        x[remove_indices] = 0.0
                    return x
                
                child1 = enforce_sum(child1)
                child2 = enforce_sum(child2)
                
                new_population.append(child1)
                if len(new_population) < pop_size:
                    new_population.append(child2)
            else:
                # Mutation
                if len(selected) > 0:
                    parent = np.random.choice(selected)
                    child = parent.copy()
                    # Flip some bits
                    for i in range(n_points):
                        if np.random.random() < mutation_rate:
                            child[i] = 1 - child[i]
                            
                    # Enforce sum constraint
                    child = enforce_sum(child)
                    new_population.append(child)
        
        # Ensure population size
        population = new_population[:pop_size]
    
    # Finish with convex optimization on the best candidates
    try:
        # Take top individuals from population for convex refinement
        top_candidates = population[:10]
        best_candidate = None
        best_refined_c5 = float('inf')
        
        # For each top candidate, refine with convex optimization
        for candidate in top_candidates:
            # Relax the binary constraint and solve
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Create objective using quadratic approximation of correlation  
            # This is a simplified approach - directly compute the maximum overlap
            corr = np.correlate(candidate, 1 - candidate, mode='full')
            max_corr = np.max(corr)
            
            # Try a more direct structured approach:
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
            prob = cp.Problem(objective, constraints)
            
            try:
                prob.solve(solver=cp.CBC, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_refined = h_vars.value
                    if h_refined is not None:
                        h_refined = np.clip(h_refined, 0, 1)
                        # Ensure sum is correct
                        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                        
                        corr_refined = np.correlate(h_refined, 1 - h_refined, mode='full')
                        c5_refined = np.max(corr_refined) * dx
                        
                        if c5_refined < best_refined_c5:
                            best_refined_c5 = c5_refined
                            best_candidate = h_refined
            except:
                pass
            
        if best_candidate is not None:
            best_h = best_candidate
            best_c5 = best_refined_c5
    except:
        pass
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
