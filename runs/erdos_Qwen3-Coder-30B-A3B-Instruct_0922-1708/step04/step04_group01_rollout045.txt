import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Initialize with the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a simple block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Coarse grid search
    n_coarse = 20
    coarse_candidates = []
    
    # Generate coarse candidates
    for _ in range(200):
        h_coarse = np.random.rand(n_coarse)
        h_coarse = h_coarse / np.sum(h_coarse) * (n_coarse / 2)
        coarse_candidates.append(h_coarse)
    
    # Evaluate coarse candidates
    coarse_scores = []
    for h_coarse in coarse_candidates:
        # Interpolate to fine grid
        h_fine = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, n_coarse), h_coarse)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full')
        c5 = np.max(correlation) * dx
        coarse_scores.append(c5)
    
    # Get top candidates
    top_indices = np.argsort(coarse_scores)[:10]
    top_candidates = [coarse_candidates[i] for i in top_indices]
    
    for i, h_coarse in enumerate(top_candidates):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Interpolate coarse candidate to fine grid
        x_coarse = np.linspace(0, 2, len(h_coarse))
        x_fine = np.linspace(0, 2, n_points)
        h_fine = np.interp(x_fine, x_coarse, h_coarse)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        # Local refinement with gradient-based method
        for _ in range(500):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Compute gradient
            eps = 1e-4
            grad = np.zeros(n_points)
            
            c5_current = np.max(np.correlate(h_fine, 1 - h_fine, mode='full')) * dx
            
            for j in range(n_points):
                h_plus = h_fine.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode='full')) * dx
                grad[j] = (c5_plus - c5_current) / eps
            
            # Update using gradient descent
            lr = 0.01
            h_new = h_fine - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_fine = h_new
            
        # Final evaluation
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fine.copy()
    
    # Final convex optimization refinement
    try:
        if best_h is not None and time.time() - start_time < budget_s - 5:
            # Construct constraints for CVXPY
            h_vars = cp.Variable(n_points)
            h_vars.value = best_h
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Define the objective: max_k ∫ h(x)(1 - h(x+k)) dx
            # Use a convex relaxation or direct minimization approach
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL and h_vars.value is not None:
                h_cvx = np.clip(h_vars.value, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                
                correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                c5_cvx = np.max(correlation) * dx
                
                if c5_cvx < best_c5:
                    best_c5 = c5_cvx
                    best_h = h_cvx
    except:
        pass
        
    # Final validation
    if best_h is not None:
        final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(final_correlation) * dx
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fall back to the initial solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return (h.tolist(), c5, n_points)
