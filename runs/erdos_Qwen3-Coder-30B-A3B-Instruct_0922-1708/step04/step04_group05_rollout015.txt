import time

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Scale to meet integral constraint
    h = h_init * (n_points / 2) / np.sum(h_init)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Iterative refinement with gradient steps
    max_iter = 2000
    lr = 0.001
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integral
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient approximation via finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            # Re-scale to maintain constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
        
        # Update h using gradient descent
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Adaptive learning rate
        if i % 100 == 0 and i > 0:
            lr *= 0.95
        
        h = h_new
        
        # Random perturbations to escape local optima
        if i % 50 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
