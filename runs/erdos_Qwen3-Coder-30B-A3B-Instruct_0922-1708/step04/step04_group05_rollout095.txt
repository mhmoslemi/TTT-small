import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise use structured waveform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        # Create a periodic waveform that (often) gives a better starting point
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 + 0.3 * np.sin(2*np.pi*x) + 0.1 * np.sin(4*np.pi*x)
        h_init = np.clip(h_init, 0, 1)
    
    # Normalize to satisfy integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate multiple diverse initial candidates
    candidates = []
    num_candidates = 20
    
    for i in range(num_candidates):
        # Original initialization
        if i == 0:
            h = h_init.copy()
        else:
            # Random perturbations
            h = h_init.copy()
            noise = np.random.normal(0, 0.03, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        candidates.append(h.copy())
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    # Gradient-based update with projection
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(h):
        grad = np.zeros_like(h)
        eps = 1e-5
        h_orig = h.copy()
        c5_base = compute_c5(h_orig)
        
        for i in range(n_points):
            h_plus = h_orig.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    # Optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Evaluate and keep best current solution
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
        
        # Randomly select a candidate to refine
        idx = np.random.randint(0, len(candidates))
        h = candidates[idx].copy()
        
        # Perform gradient descent for a fixed number of iterations
        for _ in range(10):
            if time.time() - start_time > budget_s - 2:
                break
            grad = compute_gradient(h)
            lr = 0.005
            h = h - lr * grad
            h = project_h(h)
            
            # Early stopping if no significant improvement
            new_c5 = compute_c5(h)
            if new_c5 > c5 * 0.99999:
                break
        
        # Store the refined candidate
        candidates[idx] = h
        
        # Occasionally add new candidates
        if iter_count % 10 == 0 and len(candidates) < 50:
            # Add a new noisy candidate
            h_new = h_init.copy()
            noise = np.random.normal(0, 0.07, n_points)
            h_new = h_new + noise
            h_new = project_h(h_new)
            candidates.append(h_new)
    
    # Final evaluation of the best solution
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
