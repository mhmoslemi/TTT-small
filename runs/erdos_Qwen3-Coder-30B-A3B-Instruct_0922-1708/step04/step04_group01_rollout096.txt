import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Default initialization
        h = np.ones(n_points) * 0.5  # uniform distribution
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Coarse optimization to find good starting points
    coarse_grid_size = 20
    top_candidates = 10
    coarse_h_list = []
    
    # Generate coarse grid candidates
    for _ in range(100):
        if time.time() - start_time > budget_s - 100:
            break
        h_coarse = np.random.rand(coarse_grid_size)
        h_coarse = h_coarse / np.sum(h_coarse) * (coarse_grid_size / 2)
        
        # Upsample to fine grid
        h_fine = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, coarse_grid_size), h_coarse)
        
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full')
        c5_val = np.max(correlation) * dx
        coarse_h_list.append((h_fine.copy(), c5_val))
    
    # Sort and keep top candidates
    coarse_h_list.sort(key=lambda x: x[1])
    top_coarse = coarse_h_list[:top_candidates]
    
    # Now try to refine these top coarse candidates
    for i, (h_init, _) in enumerate(top_coarse):
        if time.time() - start_time > budget_s - 50:
            break
            
        h_refined = h_init.copy()
        # Simple local optimization around the candidate
        for _ in range(500):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute gradient
            eps = 1e-4
            grad = np.zeros(n_points)
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - np.max(np.correlate(h_refined, 1 - h_refined, mode='full')) * dx) / eps
            
            # Gradient descent step
            lr = 0.01
            h_new = h_refined - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_refined = h_new
            
        # Check final value
        correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_refined.copy()
    
    # Try a convex optimization approach for final improvement
    try:
        if best_c5 > 0.35:
            h_vars = cp.Variable(n_points)
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            # Approximate the max overlap with a quadratic constraint
            # Use a proxy - minimize the sum of squared overlaps
            # This is a heuristic, but can help smooth out the search
            objective = cp.Minimize(cp.sum_squares(cp.conv(h_vars, cp.reshape(1 - h_vars, (n_points, 1)))))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CBC, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_cvx = h_vars.value
                if h_cvx is not None:
                    h_cvx = np.clip(h_cvx, 0, 1)
                    h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                    
                    correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                    c5_cvx = np.max(correlation) * dx
                    
                    if c5_cvx < best_c5:
                        best_h = h_cvx
                        best_c5 = c5_cvx
    except:
        pass

    # Final check on best solution
    if np.sum(best_h) != n_points / 2:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
