import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200
    dx = 2.0 / n_points
    max_iterations = 10000
    initial_temp = 1.0
    cooling_rate = 0.99
    perturb_fraction = 0.1
    perturb_magnitude = 0.1
    
    # Initialize
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Compute initial C5
    correlation = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(correlation) * dx
    
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing
    temperature = initial_temp
    iteration = 0
    accept_count = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Select random points to perturb
        n_perturb = max(1, int(n_points * perturb_fraction))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Create candidate solution
        h_candidate = h.copy()
        
        # Perturb selected points
        for i in indices:
            delta = random.uniform(-perturb_magnitude, perturb_magnitude)
            h_candidate[i] += delta
        
        # Clip to bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to maintain integral constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute new C5
        correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(correlation) * dx
        
        # Accept or reject
        delta = candidate_c5 - current_c5
        
        if delta < 0 or np.exp(-delta / temperature) > random.random():
            h = h_candidate
            current_c5 = candidate_c5
            accept_count += 1
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
        # Cool down
        if iteration % 100 == 0:
            temperature *= cooling_rate
    
    # Final refinement: local search around best solution
    h_refined = best_h.copy()
    refinement_iterations = 500
    
    for _ in range(refinement_iterations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Make small perturbations
        h_trial = h_refined.copy()
        idx = random.choice(range(n_points))
        delta = random.uniform(-0.02, 0.02)
        h_trial[idx] += delta
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Evaluate
        correlation = np.correlate(h_trial, 1 - h_trial, mode='full')
        trial_c5 = np.max(correlation) * dx
        
        if trial_c5 < best_c5:
            h_refined = h_trial
            best_c5 = trial_c5
    
    # Final computation
    final_correlation = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
