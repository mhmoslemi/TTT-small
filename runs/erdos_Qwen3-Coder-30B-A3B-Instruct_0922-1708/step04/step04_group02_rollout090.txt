import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or a simple construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
    else:
        # Simple step function: ramp up to 1
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0 / mid
        h[mid:] = 0.0
        
    # Recommended number of segments
    m = 10
    # Represent h as sparse segments
    # We will maintain 0 <= h[i] <= 1, and sum(h)*dx = 1
    
    # Position constraints
    positions = np.linspace(0.0, 2.0, m+1)
    # We optimize middle position deltas and heights
    # Mid positions: 0 < pos[1] < pos[2] < ... < pos[m-1] < 2
    # Heights: 0 <= h[i] <= 1
    
    # Use smooth function for initialization
    def get_h_from_params(pos_params, h_params):
        # pos_params is (m-1) positions, h_params is m heights
        pos = np.concatenate([[0], pos_params, [2]])
        heights = np.clip(h_params, 0, 1)
        # Total area
        area = np.sum((pos[1:] - pos[:-1]) * heights)
        # Normalize to 1
        heights = heights / area * (n_points / 2)
        h = np.zeros(n_points)
        for i in range(m):
            start_idx = int(pos[i] / 2 * n_points)
            end_idx = int(pos[i+1] / 2 * n_points)
            if end_idx > start_idx:
                h[start_idx:end_idx] = heights[i]
        return h
    
    # Optimization function
    def compute_c5(h):
        # Compute overlap integral C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Convert h to segments
    def h_to_segments(h):
        segments = []
        # Find step changes
        i = 0
        while i < n_points:
            val = h[i]
            start = i
            while i < n_points and h[i] == val:
                i += 1
            end = i
            segments.append((start, end, val))
        return segments
    
    def segments_to_h(segments):
        h = np.zeros(n_points)
        for start, end, val in segments:
            h[start:end] = val
        return h
    
    # Objective function for optimization (sparse segments)
    def objective(x):
        # x contains [pos[1], ..., pos[m-1], h[0], ..., h[m-1]]
        pos_params = x[:m-1]
        h_params = x[m-1:]
        
        # Enforce increasing positions
        # pos is [0, pos[0], ..., pos[m-1], 2]
        pos_params = np.clip(pos_params, 0.001, 1.999)
        pos = np.concatenate([[0], np.sort(pos_params), [2]])
        heights = np.clip(h_params, 0, 1)
        
        # Normalize heights to meet integral constraint
        area = np.sum((pos[1:] - pos[:-1]) * heights)
        if area > 0:
            heights = heights / area * (n_points / 2)
        
        # Construct h from segments
        h = np.zeros(n_points)
        for i in range(m):
            start = int(pos[i] / 2 * n_points)
            end = int(pos[i+1] / 2 * n_points)
            if end > start:
                h[start:end] = heights[i]
        
        return compute_c5(h)
    
    # Constraints
    def constraint_area(x):
        pos_params = x[:m-1]
        h_params = x[m-1:]
        pos = np.concatenate([[0], np.sort(pos_params), [2]])
        heights = np.clip(h_params, 0, 1)
        area = np.sum((pos[1:] - pos[:-1]) * heights)
        return area * dx - 1.0  # Should be zero
    
    def constraint_pos_order(x):
        pos_params = x[:m-1]
        return np.diff(pos_params).min()  # Should be positive
    
    def constraint_bounds(x):
        h_params = x[m-1:]
        return min(h_params)  # Should be non-negative
    
    constraints = [
        {'type': 'eq', 'fun': constraint_area},
        {'type': 'ineq', 'fun': lambda x: constraint_pos_order(x)},
        {'type': 'ineq', 'fun': lambda x: constraint_bounds(x)},
    ]
    
    # Start optimization
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Optimization loop with constraints
    for iteration in range(10):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Initialize parameters
        pos_guess = np.linspace(0.2, 1.8, m-1)
        h_guess = np.ones(m) * (1.0 / m)
        
        # Adjust range so that area equals 1
        h_guess = h_guess * (n_points / 2) / np.sum(h_guess)
        
        x0 = np.concatenate([pos_guess, h_guess])
        
        # Optimize
        try:
            result = minimize(objective, x0, method='SLSQP', 
                              constraints=constraints, 
                              bounds=[(0.001, 1.999)] * (m-1) + [(0, 1)] * m,
                              options={'maxiter': 100, 'ftol': 1e-8})
            
            if result.success:
                x_opt = result.x
                pos_params = x_opt[:m-1]
                h_params = x_opt[m-1:]
                pos = np.concatenate([[0], np.sort(pos_params), [2]])
                heights = np.clip(h_params, 0, 1)
                area = np.sum((pos[1:] - pos[:-1]) * heights)
                heights = heights / area * (n_points / 2)
                
                h = np.zeros(n_points)
                for i in range(m):
                    start = int(pos[i] / 2 * n_points)
                    end = int(pos[i+1] / 2 * n_points)
                    if end > start:
                        h[start:end] = heights[i]
                
                current_c5 = compute_c5(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    
                # Small gradient step to refine
                epsilon = 1e-4
                grad = np.zeros_like(x0)
                for i in range(len(x0)):
                    x_plus = x0.copy()
                    x_plus[i] += epsilon
                    grad[i] = (objective(x_plus) - objective(x0)) / epsilon
                
                x_new = x0 - 0.1 * grad
                x_new = np.clip(x_new, 0.001, 1.999)
                # Apply bounds and keep h parameters in [0,1]
                x_new[m-1:] = np.clip(x_new[m-1:], 0, 1)
                
                h_test = get_h_from_params(x_new[:m-1], x_new[m-1:])
                c5_test = compute_c5(h_test)
                if c5_test < best_c5:
                    best_h = h_test
                    best_c5 = c5_test
                    
        except Exception as e:
            pass
    
    # Final refinement
    # Do additional small local optimization to improve C5
    def compute_c5_local(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Focus on largest c5 entries
    h_local = best_h.copy()
    for i in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        # Try small random perturbations
        idx = np.random.randint(n_points)
        h_new = h_local.copy()
        # Modify both ends of a small interval around idx
        s = max(0, idx - 3)
        e = min(n_points, idx + 3)
        h_new[s:e] += (np.random.random(e-s) - 0.5) * 0.05
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = compute_c5_local(h_new)
        if c5_new < compute_c5_local(h_local):
            h_local = h_new
    
    # Final evaluation
    final_c5 = compute_c5_local(h_local)
    
    return (h_local.tolist(), final_c5, n_points)
