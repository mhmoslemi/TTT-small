import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            # Interpolate to target resolution
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Start with a block function
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
    
    # Ensure sum constraint
    h_init = h_init * target_sum / np.sum(h_init)
    
    # Evolutionary algorithm approach
    pop_size = 100
    generations = 100
    mutation_rate = 0.05
    crossover_rate = 0.5
    elite_size = 10
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_rand = h_init.copy()
        # Randomly perturb
        if 'initial_h_values' in globals():
            # Use a binary version of the parent as starting point
            h_bin = (h_init > 0.5).astype(float)
            # Adjust to exactly target_sum
            current_sum = np.sum(h_bin)
            if current_sum < target_sum:
                # Add ones
                indices = np.random.choice(n_points, int(target_sum - current_sum), replace=False)
                h_bin[indices] = 1.0
            elif current_sum > target_sum:
                # Remove ones
                indices = np.random.choice(np.where(h_bin > 0)[0], int(current_sum - target_sum), replace=False)
                h_bin[indices] = 0.0
            h_rand = h_bin
        else:
            # Make binary
            h_rand = (np.random.rand(n_points) > 0.5).astype(float)
            # Adjust to sum target_sum
            current_sum = np.sum(h_rand)
            if current_sum < target_sum:
                indices = np.random.choice(n_points, int(target_sum - current_sum), replace=False)
                h_rand[indices] = 1.0
            elif current_sum > target_sum:
                indices = np.random.choice(np.where(h_rand > 0)[0], int(current_sum - target_sum), replace=False)
                h_rand[indices] = 0.0
        population.append(h_rand)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness for each individual
        fitness = []
        for h in population:
            # Check feasibility
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Select elite
        elite = population[:elite_size]
        
        # Generate new population
        new_pop = elite[:]
        while len(new_pop) < pop_size:
            if np.random.rand() < crossover_rate and len(elite) >= 2:
                # Crossover
                parent1, parent2 = np.random.choice(elite, 2, replace=False)
                crossover_point = np.random.randint(1, n_points)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                # Normalize to sum=50
                child = child * target_sum / np.sum(child)
                new_pop.append(child)
            else:
                # Mutation
                parent = np.random.choice(elite)
                child = parent.copy()
                # Flip some bits
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] = 1 - child[i]
                # Adjust to maintain sum
                current_sum = np.sum(child)
                if current_sum < target_sum:
                    # Add some ones
                    indices = np.random.choice(np.where(child < 1)[0], int(target_sum - current_sum), replace=False)
                    child[indices] = 1.0
                elif current_sum > target_sum:
                    # Remove some ones
                    indices = np.random.choice(np.where(child > 0)[0], int(current_sum - target_sum), replace=False)
                    child[indices] = 0.0
                new_pop.append(child)
        
        population = new_pop[:pop_size]
    
    # Final refinement using convex optimization
    # Take top few candidates
    top_candidates = population[:10]
    
    for candidate in top_candidates:
        try:
            # Relax constraints and use CVX
            h_vars = cp.Variable(n_points)
            constraints = [
                h_vars >= 0,
                h_vars <= 1,
                cp.sum(h_vars) == target_sum
            ]
            
            # For this simplified approach, try line search with heuristics
            # But we'll just refine best found so far with local search
            
            # Instead use improved local search step for better results
            h_local = candidate.copy()
            for _ in range(100):
                # Evaluate gradient
                corr_ref = np.correlate(h_local, 1 - h_local, mode='full')
                c5_ref = np.max(corr_ref) * dx
                
                # Compute finite differences gradient
                eps = 1e-4
                grad = np.zeros(n_points)
                for i in range(n_points):
                    h_plus = h_local.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * target_sum / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5_ref) / eps
                
                # Update with gradient descent
                lr = 0.01
                h_new = h_local - lr * grad
                
                # Enforce constraints
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * target_sum / np.sum(h_new)
                
                # Accept better solution
                corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr_new) * dx
                
                if c5_new < c5_ref:
                    h_local = h_new
                else:
                    break
                    
            # Check final result
            corr_final = np.correlate(h_local, 1 - h_local, mode='full')
            c5_final = np.max(corr_final) * dx
            
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_local.copy()
                
        except Exception:
            pass
    
    # Final validation
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
