import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from global initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use CVXPY for optimization
    # Define the optimization variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1,
        C >= 0
    ]
    
    # Compute overlap for each shift and add constraints
    for k in range(1, n_points):
        # For positive shift k
        if k < n_points:
            overlap = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
            constraints.append(overlap <= C)
        # For negative shift -k
        if k < n_points:
            overlap = cp.sum(cp.multiply(h_var[k:], 1 - h_var[:n_points-k]))
            constraints.append(overlap <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Create and solve the problem
    prob = cp.Problem(objective, constraints)
    
    # Use a fast solver, ECOS or SCS
    # The problem is set up to be convex in nature
    solver = cp.ECOS
    
    try:
        prob.solve(solver=solver, verbose=False, max_iters=1000, feastol=1e-6, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_solution = h_var.value
            c5_solution = C.value
            if c5_solution < best_c5:
                best_h = h_solution.copy()
                best_c5 = c5_solution
    except:
        pass
    
    # If that didn't improve much, try fine-tuning with gradient descent
    h = best_h.copy()
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
        
    # Instead, use a simpler gradient-based method
    # Define objective function for a given h
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # Gradient-like descent approach
    for step in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        current_c5 = compute_c5(h)
        
        # Gradient estimation via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale to satisfy constraint
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update with gradient descent
        alpha = 1e-3
        h_new = h - alpha * grad
        
        # Project back onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update best if current is better
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
