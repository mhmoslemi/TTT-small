import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation
    # Try to solve a convex relaxation first
    h_relaxed = h.copy()
    
    # Use CVXPY for convex relaxation (minimize sum of overlaps)
    h_cvx = cp.Variable(n_points)
    constraints = [0 <= h_cvx, h_cvx <= 1, cp.sum(h_cvx) == n_points / 2]
    
    # For this convex relaxation we can approximate with a simple sum
    # Since we're solving for the min of total overlap, we try to spread h evenly
    obj = cp.Minimize(cp.sum(h_cvx))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h_relaxed = np.array(h_cvx.value).flatten()
    except:
        pass

    # Use the relaxed solution as a starting point
    h = h_relaxed.copy()
    
    # Local refinement using gradient descent
    iteration = 0
    max_no_improvement = 50
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update step
        lr = 0.01 / (1 + 0.01 * iteration)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Check for improvement
        corr = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(corr) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
