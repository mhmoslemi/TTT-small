import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # initialization
    n_points = 150
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # normalize to correct integral sum
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # objective function with gradient
    def objective_and_grad(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5, None
    
    # For gradient-based optimization, we use finite differences
    def grad_h(h, eps=1e-5):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5_base, _ = objective_and_grad(h)
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus, _ = objective_and_grad(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    # Project to feasible region
    def project_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # main search loop
    max_inner_iter = 2000
    
    # coarse grid search
    N_coarse = 100
    coarse_h_candidates = []
    for i in range(N_coarse):
        if time.time() - start_time > budget_s - 30:
            break
        # Generate random distribution with sum=1
        h_candidate = np.random.rand(n_points)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        h_candidate = np.clip(h_candidate, 0, 1)
        coarse_h_candidates.append(h_candidate)
    
    # evaluate coarse candidates
    c5_vals = []
    for h in coarse_h_candidates:
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        c5_vals.append(c5)
    
    # get top candidates
    top_idx = np.argsort(c5_vals)[:min(10, len(c5_vals))]
    top_candidates = [coarse_h_candidates[i] for i in top_idx]
    
    # refine using gradient descent
    refiner_start = time.time()
    for i, h in enumerate(top_candidates):
        if time.time() - start_time > budget_s - 30:
            break
        prev_c5 = float('inf')
        for iter in range(max_inner_iter):
            if time.time() - start_time > budget_s - 10:
                break
            h = project_feasible(h)
            grad = grad_h(h)
            lr = 0.0001 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            h_new = project_feasible(h_new)
            
            current_c5 = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_new.copy()
                
            if abs(prev_c5 - current_c5) < 1e-6:
                break
            h = h_new
            prev_c5 = current_c5

    # final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
