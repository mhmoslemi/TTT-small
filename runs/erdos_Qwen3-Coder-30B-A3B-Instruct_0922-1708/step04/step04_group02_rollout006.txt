import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a random distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Generate random initializations and take best
        best_h = None
        best_c5 = float('inf')
        
        for _ in range(3):
            h_init = np.random.rand(n_points)
            h_init = h_init / np.sum(h_init) * (n_points / 2)
            
            correlation = np.correlate(h_init, 1 - h_init, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_init.copy()
                
        h = best_h

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Relaxation stage
    if time.time() - start_time < budget_s * 0.4:
        try:
            h_relaxed = cp.Variable(n_points)
            constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points / 2]
            
            # Use CVXOPT for relaxation
            prob = cp.Problem(cp.Minimize(1), constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed_vals = h_relaxed.value
                h = h_relaxed_vals.copy()
        except:
            pass

    # Integer rounding stage
    if time.time() - start_time < budget_s * 0.6:
        # Simple quantization
        threshold = np.median(h)
        h_binary = (h > threshold).astype(float)
        
        # Adjust to match constraint
        current_sum = np.sum(h_binary)
        target_sum = n_points / 2
        
        if current_sum != target_sum:
            diff = int(target_sum - current_sum)
            if diff > 0:
                # Need to flip zeros to ones
                zero_indices = np.where(h_binary == 0)[0]
                if len(zero_indices) >= diff:
                    flip_indices = np.random.choice(zero_indices, size=diff, replace=False)
                    h_binary[flip_indices] = 1
            else:
                # Need to flip ones to zeros
                one_indices = np.where(h_binary == 1)[0]
                if len(one_indices) >= -diff:
                    flip_indices = np.random.choice(one_indices, size=-diff, replace=False)
                    h_binary[flip_indices] = 0

        h = h_binary * (n_points / 2) / np.sum(h_binary)

    # Fine-tuning with local search
    while time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try small perturbations to improve
        improved = False
        h_trial = h.copy()
        
        for i in range(n_points):
            # Try small change
            step = 0.01
            old_val = h_trial[i]
            
            # Ensure we are in bounds
            new_val = old_val + step
            if new_val > 1:
                new_val = 1
            h_trial[i] = new_val
            
            # Normalize
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            # Evaluate
            correlation_test = np.correlate(h_trial, 1 - h_trial, mode='full')
            c5_test = np.max(correlation_test) * dx
            
            if c5_test < c5:
                h = h_trial.copy()
                c5 = c5_test
                improved = True
                break
                
            # Reset
            h_trial[i] = old_val
            
            # Try other direction
            new_val = old_val - step
            if new_val < 0:
                new_val = 0
            h_trial[i] = new_val
            
            # Normalize
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            # Evaluate
            correlation_test = np.correlate(h_trial, 1 - h_trial, mode='full')
            c5_test = np.max(correlation_test) * dx
            
            if c5_test < c5:
                h = h_trial.copy()
                c5 = c5_test
                improved = True
                break
                
            # Reset
            h_trial[i] = old_val
            
        if not improved:
            break

    # Final evaluation
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
