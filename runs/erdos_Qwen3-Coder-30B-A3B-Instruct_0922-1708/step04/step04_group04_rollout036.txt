import numpy as np
import time
from heapq import heappush, heappop
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Optimization parameters
    w = 5  # window width for local adjustments
    perturb_scale = 0.02
    min_improvement = 1e-6
    max_no_improve = 10
    refresh_freq = 50
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute shifts
    shifts = list(range(-n_points+1, n_points))
    
    # Function to compute overlap for all shifts
    def compute_overlaps(h):
        overlaps = {}
        # Use correlate to compute each shift
        for k in shifts:
            if k == 0:
                continue
            corr = correlate(h, 1-h, mode='full')
            overlap = corr[n_points-1+k] * dx
            overlaps[k] = overlap
        return overlaps
    
    # Function to compute maximum overlap
    def compute_max_overlap(h):
        overlaps = compute_overlaps(h)
        return max(overlaps.values()) if overlaps else 0.0
    
    # Initialize priority queue
    def initialize_queue(h):
        overlaps = compute_overlaps(h)
        queue = []
        for k, overlap in overlaps.items():
            heappush(queue, (-overlap, k))  # max-heap with negative values
        return queue, overlaps
    
    queue, overlaps = initialize_queue(h)
    
    # Improve solution
    no_improve_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        if iteration % refresh_freq == 0:
            queue, overlaps = initialize_queue(h)
            
        if not queue:
            break
            
        # Get highest overlap shift
        neg_overlap, k = heappop(queue)
        current_overlap = -neg_overlap
        
        # Find indices that contribute to this overlap
        if k > 0:
            # h[i] * (1 - h[i+k])
            start_indices = range(n_points - k)
            end_indices = range(k, n_points)
        elif k < 0:
            # h[i+k] * (1 - h[i])
            start_indices = range(-k, n_points)
            end_indices = range(n_points + k)
        else:
            # Not applicable for k = 0
            continue
            
        # Local adjustment based on overlap significance
        adjustment = 0.02
        for i in range(n_points):
            if i in start_indices and (i + abs(k)) in end_indices:
                if k > 0:
                    # h[i] * (1 - h[i+k])
                    if h[i] > 0.5 and h[i+k] < 0.5:
                        h[i] -= adjustment
                        h[i+k] += adjustment
                    elif h[i] < 0.5 and h[i+k] > 0.5:
                        h[i] += adjustment
                        h[i+k] -= adjustment
                else:
                    # h[i+k] * (1 - h[i])
                    if h[i+k] > 0.5 and h[i] < 0.5:
                        h[i+k] -= adjustment
                        h[i] += adjustment
                    elif h[i+k] < 0.5 and h[i] > 0.5:
                        h[i+k] += adjustment
                        h[i] -= adjustment
                    
        # Add random perturbation
        for _ in range(5):
            idx = np.random.randint(0, n_points)
            perturb = np.random.normal(0, perturb_scale)
            h[idx] += perturb
            
        # Ensure h is in valid range
        h = np.clip(h, 0, 1)
        # Rescale to required integral
        h = h * (n_points / 2) / np.sum(h)
        
        # Update overlaps and queue
        new_overlap = compute_max_overlap(h)
        if new_overlap < best_c5 - min_improvement:
            best_c5 = new_overlap
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > max_no_improve:
            # Increase window or perturbation
            w = min(w + 1, 20)
            perturb_scale *= 1.2
            
        iteration += 1
        
        # Recompute overlaps and queue
        queue, overlaps = initialize_queue(h)
        
        # Exit if not much improvement or time is close
        if no_improve_count > 50 or (time.time() - start_time > budget_s - 5):
            break
    
    # Final check
    final_overlap = compute_max_overlap(best_h)
    return (best_h.tolist(), final_overlap, n_points)
