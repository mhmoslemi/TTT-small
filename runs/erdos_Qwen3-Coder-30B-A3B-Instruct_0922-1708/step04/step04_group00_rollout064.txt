import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # First, try solving via LP relaxation + gradient descent refinement
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Try solving using LP relaxation
    try:
        # Set up variables
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Add constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add auxiliary variables for overlap penalty
        # For each possible shift k, we'll make w[i,k] = h[i] * h[i+k]
        # This is overcomplicated for the actual problem but gets us a bound
        # Instead, just compute correlation directly with a convex relaxation
        
        # Relaxation approach: use that h(x) * (1 - h(x+k)) has bounded max
        # We're essentially interested in max of correlation of h with (1-h)
        
        # Let's instead try to find minimum C5 using a combination of
        # convex programming approach with linear constraints for max
        
        # Core optimization approach:
        # Minimize C5 subject to:
        # max_{k,x} h[x] * (1 - h[(x+k)%n]) <= C5 * dx
        # This is non-convex.
        
        # Approximate using piecewise linear constraints
        # This is tricky for the full problem, so instead just do direct BLAS
        
        # Use scip to relax the problem, but we solve 
        # LP relaxation over C5 + constraints that force:
        # for all x,k: h[x] * (1 - h[start]) <= C5 * dx
        # But that's still true only if we bound products
        
        # Simpler one: start MP with alternating algorithms
        # Start with GP or QP approach for product terms 
        # But these are difficult because h[i] * h[i+k] used for overlap
        
        # Use not-QP relaxation approach
        # h in [0,1], sum h = 1/dx ie n_points/2
        # minimize max h[i] * (1-h[i+k]) over all shifts k, i
        
        # Solution: try gradient descent from LP relaxation
        
        # Try suitable CVXPY approach  
        h_init = h.copy()
        h_cvx = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # The tricky part is convex relaxation of products. We don't have a perfect indefinite implementation
        # So instead we compute the correlation and see whether min(F) over all shifts
        # is a good approximation with our transformed convex problem
        
        constraints = [
            h_cvx >= 0,
            h_cvx <= 1, 
            cp.sum(h_cvx) == n_points / 2,
            # Just add naive relaxation for C5
            C5 >= 0
        ]
        
        # Solve this as a basic relaxation that incorporates the sum constraint
        # Without proper polynomial constraints for h(x)*h(x+k) 
        # Let's try a heuristic better than just gradient descent
        
        # Compute correlation for h and 1-h directly since these are linear operations
        # The key insight is that we're computing a correlation-type thing which has
        # special structure not very concave or convex...
        
        # Use solver for feasibility, but solve numerically rather than QSAT
        prob = cp.Problem(cp.Minimize(C5), constraints)
        
        # Solve with direct solver
        try:
            prob.solve(solver=GLPK, time_limit=200)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                # Extract the values
                h_solution = h_cvx.value
                if h_solution is not None:
                    h = np.array(h_solution)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
        except Exception:
            pass
            
    except Exception:
        pass  # Keep going with initial approach
    
    # Do a refined local search
    h = best_h.copy()
    
    # Use gradient descent with finite differences to refine
    iterations = 0
    max_iter = 8000
    lr = 0.005
    
    last_c5 = float('inf')
    patience = 0
    
    while iterations < max_iter and time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Decrease learning rate slowly
        lr = max(1e-6, lr * 0.9995)
        
        iterations += 1
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    # Ensure sum constraint
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
