import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Direct optimization approach with bounds
    def objective(x):
        h = np.clip(x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def constraint_func(x):
        h = np.clip(x, 0, 1)
        return np.sum(h) * dx - 1.0
    
    # Use scipy minimize
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Try multiple starting points to escape local minima
    best_result = None
    best_value = float('inf')
    
    for attempt in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random initialization with some pattern
        if attempt == 0:
            # Use existing solution if available
            x0 = h.copy()
        elif attempt == 1:
            # Alternative pattern
            x0 = np.random.uniform(0.2, 0.8, n_points)
            x0 = x0 * (n_points / 2) / np.sum(x0)
        else:
            # Random shuffled copy (with some randomness)
            x0 = np.random.uniform(0, 1, n_points)
            x0 = x0 * (n_points / 2) / np.sum(x0)
            
        try:
            result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                            constraints=constraints, options={'maxiter': 500})
            
            if result.success:
                current_objective = result.fun
                if current_objective < best_value:
                    best_value = current_objective
                    best_result = result.x.copy()
                    if best_value < best_c5:
                        best_c5 = best_value
                        best_h = best_result.copy()
                        
        except:
            continue
            
    # If we haven't found a good solution from above, run with PSO
    if best_c5 == float('inf'):
        # PSO approach
        n_particles = 20
        inertia = 0.7
        c1 = 2.0
        c2 = 2.0
        vmax = 0.1
        
        # Initialize particles
        particles = []
        velocities = []
        pbest = []
        pbest_c5 = []
        
        for _ in range(n_particles):
            if 'initial_h_values' in globals() and np.random.rand() < 0.5:
                particle_h = h.copy()
            else:
                particle_h = np.random.uniform(0, 1, n_points)
                particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
                particle_h = np.clip(particle_h, 0, 1)
            particles.append(particle_h)
            velocities.append(np.random.uniform(-vmax, vmax, n_points))
            pbest.append(particle_h.copy())
            pbest_c5.append(float('inf'))
        
        gbest = h.copy()
        gbest_c5 = float('inf')
        
        # PSO iterations
        max_iter = 100
        no_improvement = 0
        max_no_improvement = 30
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Update each particle
            for i in range(n_particles):
                particle = particles[i]
                vel = velocities[i]
                
                # Compute fitness
                correlation = np.correlate(particle, 1 - particle, mode='full')
                c5 = np.max(correlation) * dx
                
                # Update personal best
                if c5 < pbest_c5[i]:
                    pbest[i] = particle.copy()
                    pbest_c5[i] = c5
                    
                # Update global best
                if c5 < gbest_c5:
                    gbest = particle.copy()
                    gbest_c5 = c5
                    
                # Update best solution
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = particle.copy()
                    
                # Update velocity
                r1, r2 = np.random.rand(2, n_points)
                vel = (inertia * vel + 
                       c1 * r1 * (pbest[i] - particle) + 
                       c2 * r2 * (gbest - particle))
                
                # Clip velocity
                vel = np.clip(vel, -vmax, vmax)
                
                # Update position
                particle += vel
                
                # Enforce constraints
                particle = np.clip(particle, 0, 1)
                particle = particle * (n_points / 2) / np.sum(particle)
                
                velocities[i] = vel
                particles[i] = particle
                
            # Early stopping
            if gbest_c5 < best_c5:
                no_improvement = 0
            else:
                no_improvement += 1
                if no_improvement > max_no_improvement:
                    break

    # Final refinement
    h_refine = best_h.copy()
    
    # Add gradient-based refinement if not found with PSO
    if best_c5 == float('inf') or len(best_h) == 0:
        h_refine = h.copy()  # Just use initial guess
        
    correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Final constraint check
    h_refine = np.clip(h_refine, 0, 1)
    h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
    correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (h_refine.tolist(), final_c5, n_points)
