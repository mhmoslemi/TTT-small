import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Trust region parameters
    trust_radius = 0.1
    trust_radius_decay = 0.99
    max_iter = 500
    min_trust_radius = 1e-6
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    def project_and_scale(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return h_vals
    
    def compute_gradient(h_vals, eps=1e-4):
        grad = np.zeros_like(h_vals)
        base_c5 = compute_c5(h_vals)
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = project_and_scale(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Trust region optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update Hessian approximation (simplified)
        # For trust region subproblem:
        # min  g^T d + (1/2) d^T H d
        # s.t. ||d|| <= trust_radius
        # We use a spherical trust region
        # d = -grad if ||grad|| <= trust_radius
        # d = -trust_radius * grad / ||grad|| if ||grad|| > trust_radius
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            d = -trust_radius * grad / grad_norm
        else:
            d = np.zeros_like(grad)
            
        # Trust region step
        h_new = h + d
        
        # Project to feasible region
        h_new = project_and_scale(h_new)
        
        # Update trust radius
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
            trust_radius = min(0.1, trust_radius * 1.2)  # Expand trust radius
        else:
            trust_radius *= trust_radius_decay  # Shrink trust radius
            if trust_radius < min_trust_radius:
                trust_radius = min_trust_radius
                
        # Update best solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if trust_radius < min_trust_radius or current_c5 < 0.3807:
            break
    
    # Perform final refinement with gradient descent
    h_refined = best_h.copy()
    lr = 0.01
    max_refine_iter = 500
    for refine_iter in range(max_refine_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        grad = compute_gradient(h_refined)
        h_refined = h_refined - lr * grad
        h_refined = project_and_scale(h_refined)
        lr = max(1e-6, lr * 0.999)
        
        new_c5 = compute_c5(h_refined)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_refined.copy()
            
        if new_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
