```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def optimize_single_start(h_init, dx, max_iter=1000):
    """Optimize a single initialization using Nelder-Mead then gradient descent."""
    # Nelder-Mead optimization
    result = minimize(
        lambda x: -evaluate_h(x, dx),
        h_init,
        method='Nelder-Mead',
        options={'maxiter': max_iter, 'xatol': 1e-6, 'fatol': 1e-6}
    )
    h_nm = result.x
    
    # Refinement with gradient descent
    h_refined = h_nm.copy()
    learning_rate = 1e-5
    for i in range(1000):
        # Compute gradient via finite differences
        eps = 1e-6
        grad = np.zeros_like(h_refined)
        for j in range(len(h_refined)):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, len(h_refined))
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            h_minus = h_refined.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, len(h_refined))
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            grad[j] = (corr_plus[np.argmax(corr_plus)] - corr_minus[np.argmax(corr_minus)]) / (2 * eps)
        
        # Update step
        h_refined = h_refined - learning_rate * grad
        h_refined = repair_h(h_refined, len(h_refined))
        
        # Adaptive learning rate
        if i % 100 == 0:
            learning_rate *= 0.95
    
    return h_refined

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Sobol sequence initialization
    sampler = qmc.Sobol(d=n_points, scramble=True, seed=seed)
    initial_points = sampler.random(n=500)
    
    best_h = None
    best_c5 = float('inf')
    
    for i in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate initial point
        h = initial_points[i]
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        
        # Optimize
        h_opt = optimize_single_start(h, dx)
        c5 = evaluate_h(h_opt, dx)
        
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
    
    if best_h is None:
        # Fallback to random initialization
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = optimize_single_start(h_init, dx)
        best_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
```
