import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate a solution with adaptive noise and segment flips."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    return mutated

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # ES parameters
    population_size = 50
    elite_size = population_size // 5  # Keep top 20%
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        parent = population[0] if len(population) > 0 else h_init
        mutated = mutate(parent, 0, 0.1)
        population.append(mutated)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fitness.append(evaluate_h(h, dx))
    
    h_best = population[np.argmin(fitness)].copy()
    c5_best = min(fitness)
    
    # Evolution parameters
    mutation_scale = 0.1
    stagnation_count = 0
    max_stagnation = 100
    generation = 0
    
    while time.time() - start_time < budget_s - 5:
        generation += 1
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        elite_fitness = [fitness[i] for i in sorted_indices[:elite_size]]
        
        # Check for improvement
        if min(elite_fitness) < c5_best:
            c5_best = min(elite_fitness)
            h_best = elite[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adapt mutation scale
        if stagnation_count > max_stagnation:
            mutation_scale *= 0.9
            stagnation_count = 0
            # Inject random variation
            for i in range(min(elite_size, population_size // 10)):
                idx = np.random.randint(len(elite))
                new_h = mutate(elite[idx], generation, mutation_scale)
                elite.append(new_h)
        
        # Generate new population
        new_population = elite[:]
        while len(new_population) < population_size:
            # Select parents
            parent1 = elite[np.random.randint(len(elite))]
            parent2 = elite[np.random.randint(len(elite))]

            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutate
            child = mutate(child, generation, mutation_scale)
            new_population.append(child)
        
        # Evaluate new population
        population = new_population[:population_size]
        fitness = [evaluate_h(h, dx) for h in population]
    
    return (h_best.tolist(), c5_best, n_points)
