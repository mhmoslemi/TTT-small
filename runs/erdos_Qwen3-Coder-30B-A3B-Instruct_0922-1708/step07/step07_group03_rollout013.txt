import numpy as np
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a binary-like initialization with sum constraint
        h = np.zeros(n_points)
        half_points = n_points // 2
        h[:half_points] = 1.0
        # Randomize to avoid symmetry
        np.random.shuffle(h)
        # Rescale to meet constraint
        h = h * (n_points / 2) / np.sum(h)
    
    # For storing the best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Cooling parameters
    temperature = 0.1
    cooling_rate = 0.995
    min_temperature = 1e-8
    
    # Main optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate a perturbation
        h_new = h.copy()
        i, j = np.random.randint(0, n_points, size=2)
        
        # Determine step size
        delta = np.random.uniform(0.01, 0.05)
        
        # Adjust values
        if np.random.rand() > 0.5:
            delta = -delta
            
        delta_i = min(delta, 1.0 - h_new[i]) if delta > 0 else max(delta, -h_new[i])
        delta_j = -delta_i
        
        h_new[i] += delta_i
        h_new[j] += delta_j
        
        # Ensure values stay within bounds
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        
        # Re-scale to maintain sum constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-8:
            h_new = h_new * target_sum / current_sum
        
        # Compute C_5 for the new solution
        corr = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr) * dx
        
        # Accept or reject based on simulated annealing
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
        else:
            # Accept with probability based on temperature
            if np.random.rand() < math.exp(-(c5_new - best_c5) / temperature):
                h = h_new.copy()
        
        # Cool down temperature
        temperature = max(min_temperature, temperature * cooling_rate)
        
    # Final evaluation of the best solution
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
