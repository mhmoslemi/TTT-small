import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Evolution Strategy parameters
    pop_size = 200
    elites = 40  # 20% of population
    offsprings = 5
    mutation_std = 0.1
    patience = 30
    best_c5 = float('inf')
    best_h = h_init.copy()
    
    # Initialize population
    pop = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        pop.append(h)
    
    # Add initial solution to population
    pop.append(h_init)
    pop = pop[:pop_size]
    
    no_improvement = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 5:
        # Evaluate all individuals
        fitness = []
        for h in pop:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        pop = [pop[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = pop[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement >= patience:
            break
        
        # Select elites
        elite_pop = pop[:elites]
        
        # Generate new population
        new_pop = elite_pop.copy()
        for elite in elite_pop:
            for _ in range(offsprings):
                if time.time() - start_time > budget_s - 5:
                    break
                mutant = elite + np.random.randn(n_points) * mutation_std
                mutant = repair_h(mutant, n_points)
                new_pop.append(mutant)
        
        # Keep only the best individuals
        pop = new_pop[:pop_size]
        
        # Adaptive mutation
        if no_improvement > 10:
            mutation_std *= 0.95
    
    # Final evaluation of best solution
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
