import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidate(n_points, seed):
    np.random.seed(seed)
    # Generate 3-5 modes with random coefficients
    num_modes = np.random.randint(3, 6)
    h = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    
    for i in range(num_modes):
        k = np.random.uniform(1, 10)
        a = np.random.uniform(-1, 1)
        b = np.random.uniform(-1, 1)
        h += a * np.cos(k * x) + b * np.sin(k * x)
    
    # Normalize to (0,1) range and sum constraint
    h_min, h_max = np.min(h), np.max(h)
    if h_max > h_min:
        h = (h - h_min) / (h_max - h_min)
    h = h * 0.5 + 0.25  # move to (0.25, 0.75) range
    h = repair_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    start_time = time.time()
    
    # Collect initial candidates
    candidates = []
    
    # Use initial_h_values if available

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        candidates.append((h_init, evaluate_h(h_init, dx)))
    else:
        # Generate a candidate from structured construction 
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)
        candidates.append((h_init, evaluate_h(h_init, dx)))

    # Generate 50 diverse Fourier-based candidates
    for i in range(50):
        if time.time() - start_time > budget_s - 5:
            break
        h = generate_fourier_candidate(n_points, seed + i)
        c5 = evaluate_h(h, dx)
        candidates.append((h, c5))

    # Sort candidates by their C5 values
    candidates.sort(key=lambda x: x[1])
    top_candidates = candidates[:10]

    # Refinement using gradient descent
    best_h = None
    best_c5 = float('inf')
    
    for h_start, _ in top_candidates:
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current, dx)
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Gradient descent for refinement
        learning_rate = 0.01
        patience = 200
        last_improvement = 0
        
        for i in range(2000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient using finite differences
            epsilon = 1e-6
            grad = np.zeros(n_points)
            c5_base = evaluate_h(h_current, dx)
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[j] += epsilon
                h_minus[j] -= epsilon
                h_plus = repair_h(h_plus, n_points)
                h_minus = repair_h(h_minus, n_points)
                
                c5_plus = evaluate_h(h_plus, dx)
                c5_minus = evaluate_h(h_minus, dx)
                grad[j] = (c5_plus - c5_minus) / (2 * epsilon)

            # Update step
            h_new = h_current - learning_rate * grad
            h_new = repair_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new

                    best_c5 = c5_current
                    last_improvement = i
            else:
                if np.random.rand() < np.exp(-(c5_new - c5_current) / (c5_current * 0.01)):
                    h_current = h_new
                    c5_current = c5_new
                    
            if i - last_improvement > patience:
                learning_rate *= 0.95
                if learning_rate < 1e-6:
                    break

    return (best_h.tolist(), best_c5, n_points)
