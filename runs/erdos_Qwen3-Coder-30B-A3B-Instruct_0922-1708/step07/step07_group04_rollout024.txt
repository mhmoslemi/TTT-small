import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from existing construction or triangular
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Triangular initialization
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current

    # Gradient-based optimization using scipy
    def objective(h):
        h = repair_h(h, n_points)
        return evaluate_h(h, dx)
    
    def constraint(h):
        h = repair_h(h, n_points)
        return np.sum(h) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint}
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize using scipy minimize with SLSQP
    max_iter = 1000
    patience = 50
    last_improvement = 0
    
    # First optimize the continuous representation using scipy
    try:
        result = minimize(objective, h_current, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': max_iter, 'ftol': 1e-8, 'gtol': 1e-8})
        if result.success:
            opt_h = repair_h(result.x, n_points)
            c5_opt = evaluate_h(opt_h, dx)
            if c5_opt < c5_best:
                h_best = opt_h.copy()
                c5_best = c5_opt
    except:
        pass

    # Post-processing: coordinate descent with gradient updates
    eta = 0.005
    eps = 1e-6
    for i in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Coordinate-wise updates based on finite differences
        h_new = h_current.copy()
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)
            
            # Estimate gradient
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with gradient
            h_new[j] = h_current[j] - eta * grad_j
            
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
        # Adaptive learning rate
        if i - last_improvement > patience:
            eta *= 0.95
            last_improvement = i
            if eta < 1e-6:
                break

    # Final evaluation
    final_c5 = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), final_c5, n_points)
