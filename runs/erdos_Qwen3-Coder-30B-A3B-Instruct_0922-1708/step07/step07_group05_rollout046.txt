import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidate(n_points, seed=None):
    if seed is not None:
        np.random.seed(seed)
    
    # Generate a candidate using a sum of sine/cosine modes
    h = np.zeros(n_points)
    n_modes = 4
    for _ in range(n_modes):
        k = np.random.randint(1, n_points//4)
        a = np.random.uniform(-0.5, 0.5)
        b = np.random.uniform(-0.5, 0.5)
        phase = np.random.uniform(0, 2*np.pi)
        x = np.linspace(0, 2, n_points)
        h += a * np.cos(k * np.pi * x / 1.0) + b * np.sin(k * np.pi * x / 1.0) + 0.5
    h = repair_h(h, n_points)
    return h

        candidates.append(h_init)
    else:
        # Create some structured initial candidates
        candidates.append(np.ones(n_points) * 0.5)  # flat line
        
        # Create a piecewise linear function 
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:

            elif x < 1.5:

        if time.time() - start_time > budget_s - 5:
            break
        h = generate_fourier_candidate(n_points, seed=seed + i)
        candidates.append(h)
    
    # Evaluate initial candidates
    best_h = None
    best_c5 = float('inf')

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Refinement with gradient descent
    patience_cnt = 0

        # Select a random candidate for refinement
        h = best_h.copy()

        grad = np.zeros(n_points)
