import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Binary search for minimal C
    C_low = 0.0
    C_high = 0.5
    C_best = float('inf')
    h_best = h_init.copy()
    
    max_iterations = 50
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        C_mid = (C_low + C_high) / 2.0
        
        # Formulate and solve LP feasibility problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = []
        constraints.append(cp.sum(h_var) == n_points / 2)
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Overlap constraints - for each k, ∫ h(x) (1-h(x+k)) dx <= C*dx
        # In discrete form: sum h[i] (1 - h[i+k]) <= C * dx for all valid k
        for k in range(n_points):
            if k == 0:
                continue  # Skip k=0 as it's just integration constraint
            overlap_sum = 0
            for i in range(n_points - k):
                overlap_sum += h_var[i] * (1 - h_var[i + k])
            constraints.append(overlap_sum <= C_mid * dx)
        
        # LP problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_solution = h_var.value
                h_solution = project_h(h_solution, n_points)
                c5_sol = evaluate_h(h_solution, dx)
                if c5_sol < C_best:
                    C_best = c5_sol
                    h_best = h_solution.copy()
                C_high = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid
    
    # Final refinement
    h_final = h_best.copy()
    c5_final = C_best
    
    # Local search for better solution
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Perturb
        h_new = h_final.copy()
        idx = np.random.randint(0, n_points)
        h_new[idx] += np.random.normal(0, 0.01)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_final:
            h_final = h_new
            c5_final = c5_new
    
    return (h_final.tolist(), c5_final, n_points)
