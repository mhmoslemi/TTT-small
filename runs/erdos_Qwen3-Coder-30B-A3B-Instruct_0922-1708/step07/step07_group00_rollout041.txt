import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Evolution Strategy parameters
    pop_size = 200
    elites = 40  # 20% of pop_size
    offsprings = 5
    mutation_std = 0.1
    patience = 30
    min_std = 1e-6
    
    # Initialize population
    pop = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        pop.append(h)
    
    # Evaluation
    fitnesses = [evaluate_h(h, dx) for h in pop]
    best_idx = np.argmin(fitnesses)
    best_h = pop[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    no_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 5 and no_improvement < patience:
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        elites_indices = sorted_indices[:elites]
        elites_pop = [pop[i] for i in elites_indices]
        
        # Generate new population
        new_pop = elites_pop.copy()
        
        for elite in elites_pop:
            for _ in range(offsprings):
                mutant = elite + np.random.randn(n_points) * mutation_std
                mutant = repair_h(mutant, n_points)
                new_pop.append(mutant)
        
        # Keep only pop_size solutions
        new_pop = new_pop[:pop_size]
        
        # Evaluate new population
        fitnesses = [evaluate_h(h, dx) for h in new_pop]
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        
        if current_best_fitness < best_c5:
            best_h = new_pop[current_best_idx].copy()
            best_c5 = current_best_fitness
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Update population
        pop = new_pop
        
        # Adaptive mutation
        if no_improvement > 10:
            mutation_std = max(mutation_std * 0.95, min_std)
            
        generation += 1
    
    # Final refinement with gradient method for best solution
    h_refined = best_h.copy()
    learning_rate = 1e-5
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        # Compute approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        base_corr = correlate(h_refined, 1 - h_refined, mode='full')
        max_k_idx = np.argmax(base_corr)
        
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (corr_plus[max_k_idx] - base_corr[max_k_idx]) / eps
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad) + 1e-12
        grad = grad / grad_norm
        
        # Update step
        h_refined = h_refined - learning_rate * grad
        h_refined = repair_h(h_refined, n_points)
        
        # Evaluate new solution
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        c5_refined = np.max(corr) * dx
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
