import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # PSO parameters
    m = 100  # swarm size
    c1 = 2.0
    c2 = 2.0
    w_start = 0.9
    w_end = 0.4
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Create first particle from initial solution
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    pbest.append(h_init.copy())
    pbest_values.append(evaluate_h(h_init, dx))
    
    # Create remaining particles randomly
    for i in range(1, m):
        h_rand = np.random.rand(n_points)
        h_rand = repair_h(h_rand, n_points)
        particles.append(h_rand.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h_rand.copy())
        pbest_values.append(evaluate_h(h_rand, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO Iterations
    max_iter = 10000
    patience = 1000
    last_improvement = 0
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(m):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair to constraints
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            fitness = evaluate_h(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = fitness
                
                # Update global best
                if fitness < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = fitness
                    last_improvement = iter_num
        
        # Early termination
        if iter_num - last_improvement > patience:
            break
    
    # Gradient refinement
    h_current = gbest.copy()
    c5_current = gbest_value
    eta = 0.001
    eps = 1e-6
    max_iter_refine = 500
    
    for i in range(max_iter_refine):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_current.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_current[j] - eta * grad_j
        
        # Apply repair step to maintain feasibility
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if better
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
    
    return (h_current.tolist(), c5_current, n_points)
