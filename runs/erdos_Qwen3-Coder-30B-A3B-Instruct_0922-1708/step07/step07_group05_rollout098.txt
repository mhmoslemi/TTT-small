import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidate(n_points, seed=None):
    if seed is not None:
        np.random.seed(seed)
    
    # Generate a candidate using sum of sinusoids
    h = np.zeros(n_points)
    num_modes = 4
    for _ in range(num_modes):
        k = np.random.uniform(0.5, 5)
        a = np.random.uniform(-0.5, 0.5)
        b = np.random.uniform(-0.5, 0.5)
        x = np.linspace(0, 2, n_points)
        h += a * np.cos(k * x) + b * np.sin(k * x)
    
    # Ensure h is between 0 and 1
    h = np.clip(h, 0, 1)
    # Normalize sum to n_points/2
    h = repair_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize candidates
    candidates = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        candidates.append(h_init)
    else:
        # Generate a few candidates from Fourier basis
        for i in range(50):
            h = generate_fourier_candidate(n_points, seed=seed + i)
            candidates.append(h)
    
    # Evaluate all candidates
    candidate_scores = []
    for h in candidates:
        c5 = evaluate_h(h, dx)
        candidate_scores.append((c5, h))
    
    # Sort by score and keep top 10
    candidate_scores.sort(key=lambda x: x[0])
    top_candidates = candidate_scores[:10]
    
    h_best = top_candidates[0][1]
    c5_best = top_candidates[0][0]
    
    # Refinement with gradient descent
    patience = 200
    last_improvement = 0
    for i, (c5, h_current) in enumerate(top_candidates):
        h_refined = h_current.copy()
        c5_current = c5
        lr = 0.01
        
        for iter_num in range(2000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient via finite differences
            epsilon = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_minus = h_refined.copy()
                h_plus[j] += epsilon
                h_minus[j] -= epsilon
                h_plus = repair_h(h_plus, n_points)
                h_minus = repair_h(h_minus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                c5_minus = evaluate_h(h_minus, dx)
                grad[j] = (c5_plus - c5_minus) / (2 * epsilon)

            # Update
            h_new = h_refined - lr * grad
            h_new = repair_h(h_new, n_points)
            
            # Evaluate
            c5_new = evaluate_h(h_new, dx)
            
            # Accept
            if c5_new < c5_current:
                h_refined = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_refined.copy()
                    c5_best = c5_current
                    last_improvement = iter_num
            else:
                # Accept with some probability
                if np.random.rand() < np.exp(-(c5_new - c5_current) / (c5_current * 0.01)):
                    h_refined = h_new
                    c5_current = c5_new
            
            # Adapt learning rate
            if iter_num - last_improvement > patience // 2:
                lr *= 0.95
                if lr < 1e-6:
                    break
    
    return (h_best.tolist(), c5_best, n_points)
