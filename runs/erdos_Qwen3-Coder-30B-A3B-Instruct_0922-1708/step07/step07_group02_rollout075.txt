import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)

    # PSO parameters
    swarm_size = 50
    max_iter = 10000
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    stagnation_limit = 50
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution
    swarm.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Add random solutions with structured patterns
    for _ in range(swarm_size - 1):
        h_new = np.random.rand(n_points)
        h_new = project_h(h_new, n_points, dx)
        swarm.append(h_new)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Initialize personal bests
    pbest = [swarm[i].copy() for i in range(swarm_size)]
    pbest_fitness = [evaluate_h(swarm[i], dx) for i in range(swarm_size)]
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # Main PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - swarm[i]) + 
                            c2 * r2 * (gbest - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_h(swarm[i], n_points, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest = swarm[i].copy()
                    gbest_fitness = fitness
        
        # Early termination if improvement is minimal
        if iter_num > 100 and abs(pbest_fitness[gbest_idx] - gbest_fitness) < 1e-8:
            break
            
    return (gbest.tolist(), gbest_fitness, n_points)
