import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Binary search on the upper bound C
    c_low = 0.0
    c_high = 0.5
    tol = 1e-5
    max_iter = 50
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Formulate and solve LP feasibility problem
        h_vars = cp.Variable(n_points)
        
        # Objective: minimize C (we only check feasibility)
        # Constraints:
        # - For each k, sum h_i * (1 - h_{i+k}) <= c_mid * dx
        # - sum h = n_points / 2
        # - 0 <= h_i <= 1
        
        constraints = []
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # For overlap constraints, compute all valid indices for each k
        for k in range(n_points):
            # For index i in range n_points - k, we have h_i * (1 - h_{i+k})
            sum_term = 0
            for i in range(n_points - k):
                sum_term += h_vars[i] * (1 - h_vars[i+k])
            # Adding a small numerical margin to prevent constraints force to equality
            constraints.append(sum_term <= c_mid * dx + 1e-12)
            
        # Box constraints
        for j in range(n_points):
            constraints.append(0 <= h_vars[j])
            constraints.append(h_vars[j] <= 1)
            
        # Check feasibility
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE:
                # Feasible solution found
                h_feasible = h_vars.value
                if h_feasible is not None:
                    h_feasible = np.array(h_feasible).flatten()
                    h_feasible = project_h(h_feasible, n_points)
                    c5_feasible = evaluate_h(h_feasible, dx)
                    if c5_feasible < c5_best:
                        h_best = h_feasible.copy()
                        c5_best = c5_feasible
                    c_high = c_mid
                else:
                    c_low = c_mid
            else:
                c_low = c_mid 
        except Exception:
            c_low = c_mid
            
    # Perform final refinement with gradient-based local search on best solution
    h_current = h_best.copy()
    c5_current = c5_best
    
    # Use actual improvement from 100 descent steps
    for iter in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        # Approximate gradient computation
        epsilon = 1e-4
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += epsilon
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)

            h_minus = h_current.copy()
            h_minus[j] -= epsilon
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
            
        # Gradient descent step
        h_new = h_current - 0.005 * grad
        h_new = project_h(h_new, n_points)
        
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
    
    return (h_best.tolist(), c5_best, n_points)
