import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Sine-based initialization that's smooth and centered
        h_init = 0.5 * (1 + np.sin(np.pi * x - np.pi/2))
        h_init = project_h(h_init, n_points)
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Try several strategies with different randomness
    num_strategies = 5
    for strategy in range(num_strategies):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Strategy: random perturbations with simulated annealing-like approach
        h_current = h_init.copy()
        c5_current = c5_best
        temperature = 0.1
        
        # Local search with adaptive perturbation
        for iter_no in range(20000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Make a small random change
            h_new = h_current.copy()
            idx = np.random.randint(n_points)
            delta = np.random.normal(0, temperature)
            h_new[idx] += delta
            
            # Project to feasible set
            h_new = project_h(h_new, n_points)
            
            # Evaluate
            c5_new = evaluate_h(h_new, dx)
            
            # Accept based on Metropolis criterion
            if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / (temperature + 1e-8)):
                h_current = h_new
                c5_current = c5_new
                
            # Track best
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                
            # Cool down temperature
            if iter_no % 100 == 0:
                temperature *= 0.999
                
            # Occasionally restart from best
            if iter_no % 5000 == 0:
                h_current = h_best.copy()
                c5_current = c5_best
                
    return (h_best.tolist(), c5_best, n_points)
