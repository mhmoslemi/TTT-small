import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate a solution with adaptive mutation."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip with probability 0.3
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    mutated = np.clip(mutated, 0, 1)
    
    # Project to feasible region
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial_h_values if provided
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a simple piecewise constant solution
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 0.5
        h_init = project_h(h_init, n_points, dx)
    
    # Add initial solution to population
    population.append(h_init.copy())
    
    # Generate additional initial solutions
    for _ in range(population_size - 1):
        # Try a few different mutations
        h = h_init.copy()
        h = mutate(h, 0, 0.1)
        h = project_h(h, n_points, dx)
        population.append(h)
    
    # Best individual tracking
    best_h = None
    best_c5 = float('inf')
    
    # Evolution parameters
    mutation_scale = 0.1
    generation = 0
    stagnation_count = 0
    min_mutation_scale = 1e-6
    
    while time.time() - start_time < budget_s - 5 and generation < 1000:
        generation += 1
        
        # Evaluate population
        fitness_scores = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness_scores.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_h = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Selection: keep top 20%
        elite_size = max(1, population_size // 5)
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        while len(new_population) < population_size:
            # Select parents
            parent1 = elite[np.random.randint(len(elite))]
            parent2 = elite[np.random.randint(len(elite))]
            
            # Crossover: blending
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutate
            child = mutate(child, generation, mutation_scale)
            
            new_population.append(child)
        
        population = new_population
        
        # Adapt mutation scale
        if stagnation_count > 100:
            mutation_scale = max(min_mutation_scale, mutation_scale * 0.999)
            stagnation_count = 0
        
        # Occasionally inject random solutions
        if generation % 10 == 0 and len(new_population) < population_size:
            for _ in range(population_size - len(new_population)):
                h = np.random.rand(n_points)
                h = project_h(h, n_points, dx)
                new_population.append(h)
        
        # Update mutation scale
        mutation_scale = max(min_mutation_scale, mutation_scale * 0.999)
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
    else:
        # Fall back to the first individual if nothing found
        final_c5 = evaluate_h(population[0], dx)
        best_h = population[0]
    
    return (best_h.tolist(), final_c5, n_points)
