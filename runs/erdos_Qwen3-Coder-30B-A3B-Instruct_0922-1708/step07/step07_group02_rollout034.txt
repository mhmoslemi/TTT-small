import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate a solution with adaptive mutation parameters."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
        
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Population-based evolution strategy
    population_size = 50
    pop = []
    fitness = []
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
        pop.append(h_init)
    else:
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
        pop.append(h_init)

    # Add diverse initial solutions
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        pop.append(h)

    # Main evolution loop
    generation = 0
    mutation_scale = 0.1
    stagnation_count = 0
    best_fitness = float('inf')
    best_individual = pop[0].copy()
    
    while time.time() - start_time < budget_s - 5 and generation < 10000:
        # Evaluate all individuals
        fitness = []
        for i, h in enumerate(pop):
            fit = evaluate_h(h, dx)
            fitness.append(fit)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        pop = [pop[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_fitness:
            best_fitness = fitness[0]
            best_individual = pop[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Select top 20%
        top_count = max(2, population_size // 5)
        top_pop = pop[:top_count]
        
        # Generate new population
        new_pop = []
        
        # Keep top individuals
        new_pop.extend(top_pop)
        
        # Generate offspring through mutation and crossover
        while len(new_pop) < population_size:
            # Select two parents randomly from top 20%
            parent1 = top_pop[np.random.randint(0, top_count)]
            if np.random.rand() < 0.7:
                parent2 = top_pop[np.random.randint(0, top_count)]
            else:
                parent2 = pop[np.random.randint(0, population_size)]
                
            # Crossover: blend two parents
            alpha = np.random.rand()
            offspring = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutate
            offspring = mutate(offspring, generation, mutation_scale)
            new_pop.append(offspring)
            
        pop = new_pop
        
        # Adapt mutation scale
        mutation_scale *= 0.999
        
        # Inject randomness if stalled
        if stagnation_count > 100:
            for i in range(min(5, len(pop))):
                pop[i] = mutate(pop[i], generation, 0.05)
            stagnation_count = 0
            
        generation += 1
        
    # Final evaluation of best solution
    final_fitness = evaluate_h(best_individual, dx)
    
    return (best_individual.tolist(), final_fitness, n_points)
