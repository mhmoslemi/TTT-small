import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate h with adaptive noise and segment flips."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip (30% chance)
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Population-based evolution strategy
    population_size = 50
    population = [h_init.copy()]
    population_fitness = [evaluate_h(h_init, dx)]
    
    # Add diverse initial solutions
    for _ in range(population_size - 1):
        h = mutate(h_init, 0, 0.1)
        population.append(h)
        population_fitness.append(evaluate_h(h, dx))
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(h_init, dx)
    
    generation = 0
    mutation_scale = 0.1
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 5:
        generation += 1
        
        # Sort population by fitness (lower is better)
        sorted_indices = np.argsort(population_fitness)
        top_indices = sorted_indices[:population_size // 5]
        top_individuals = [population[i] for i in top_indices]
        top_fitness = [population_fitness[i] for i in top_indices]
        
        # Keep top performers
        population = [population[i] for i in sorted_indices[:population_size]]
        population_fitness = [population_fitness[i] for i in sorted_indices[:population_size]]
        
        # Update best solution
        current_best_fitness = population_fitness[0]
        if current_best_fitness < best_c5:
            best_c5 = current_best_fitness
            best_h = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Mutation scale adaptation
        if generation % 10 == 0:
            mutation_scale *= 0.999
        
        # Create new generation
        new_population = [population[0].copy()]  # Elite
        new_fitness = [population_fitness[0]]
        
        # Generate new candidates
        for _ in range(population_size - 1):
            if np.random.rand() < 0.5 and len(top_individuals) > 1:
                # Pairwise averaging
                idx1, idx2 = np.random.choice(len(top_individuals), 2, replace=False)
                parent1, parent2 = top_individuals[idx1], top_individuals[idx2]
                child = (parent1 + parent2) / 2
            else:
                # Mutation of elite
                child = mutate(population[0], generation, mutation_scale)
            
            new_population.append(child)
            new_fitness.append(evaluate_h(child, dx))
        
        # Add new population
        population = new_population
        population_fitness = new_fitness
        
        # Inject random perturbation if stuck
        if stagnation_count > max_stagnation:
            for i in range(population_size):

                    population[i] = project_h(np.random.rand(n_points) * 0.1, n_points, dx)
                    population_fitness[i] = evaluate_h(population[i], dx)
            stagnation_count = 0
    
    # Final validation of best solution
    best_h = project_h(best_h, n_points, dx)
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
