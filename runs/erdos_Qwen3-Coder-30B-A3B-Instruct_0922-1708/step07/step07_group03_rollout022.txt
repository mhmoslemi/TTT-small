import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Use CVXPY for optimization
    
    # Create variables
    h_var = cp.Variable(n_points)
    c5_var = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
    
    # For all possible lags, add constraint that correlation is <= c5_var
    # Total length of full correlation is 2*n_points - 1
    for k in range(2 * n_points - 1):
        # Extract the appropriate slice for lag k
        if k < n_points:
            # When lag is positive: correlation = sum(h[i] * (1-h[i+k]))
            lag_indices = np.arange(k, n_points)
            h_indices = np.arange(0, n_points - k)
        else:
            # When lag is negative: correlation = sum(h[i+k] * (1-h[i]))
            lag_indices = np.arange(0, n_points - k)
            h_indices = np.arange(k - n_points + 1, n_points)
            
        if len(h_indices) > 0:
            # Build the constraint using sum
            const_lag = cp.sum(cp.multiply(h_var[h_indices], 1 - h_var[lag_indices]))
            constraints.append(const_lag <= c5_var)
    
    # Objective
    objective = cp.Minimize(c5_var)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    solver = cp.MOSEK  # Try MOSEK first, fallback to Gurobi, then ECOS
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        try:
            solver = cp.GUROBI
            prob.solve(solver=solver, verbose=False)
        except:
            solver = cp.ECOS
            prob.solve(solver=solver, verbose=False)
    
    # Extract solution
    if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
        h_opt = np.array(h_var.value).flatten()
        c5_opt = float(c5_var.value)
    else:
        # Fallback to gradient descent if optimization fails
        h = h_init.copy()
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Simple gradient descent with projection
        learning_rate = 1e-3
        max_iter = 50000
        best_c5 = float('inf')
        best_h = h.copy()
        
        for iteration in range(max_iter):
            if (time.time() - start_time) > budget_s - 5:
                break

            # Compute correlation
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Estimate gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-6
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                if h_plus[i] > 1:
                    h_plus[i] = 1
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps

            # Update step
            h = h - learning_rate * grad
            
            # Projection to [0,1]
            h = np.clip(h, 0, 1)
            
            # Rescale to meet constraint
            h_sum = np.sum(h)
            target_sum = n_points / 2
            if abs(h_sum - target_sum) > 1e-8:
                h = h * target_sum / h_sum
                
        return (best_h.tolist(), best_c5, n_points)
    
    # Final checks
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    
    # Re-evaluate final c5
    corr = correlate(h_opt, 1 - h_opt, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_opt.tolist(), final_c5, n_points)
