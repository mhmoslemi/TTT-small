import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Binary search on the bound C
    c_low = 0.0
    c_high = 0.5
    tolerance = 1e-5
    max_iter = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Try to find a feasible h for this C
        # Use a feasibility LP approach
        h_vars = cp.Variable(n_points)
        constraints = []
        
        # Add the constraint: sum h_i = n_points/2
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Add constraints: h_i in [0, 1]
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Add overlap constraints
        # For all k, sum_{i=0}^{n-k} h_i * (1 - h_{i+k}) <= C * dx
        with np.nditer(np.arange(n_points), flags=['refs_ok']) as it:
            for k in range(1, n_points):
                if time.time() - start_time > budget_s - 5:
                    break
                # Compute indices for convolution pattern
                if k < n_points:
                    indices_i = np.arange(n_points - k)
                    indices_j = np.arange(k, n_points)
                    # Create the constraint
                    expr = cp.sum(cp.multiply(h_vars[indices_i], 1 - h_vars[indices_j]))
                    constraints.append(expr <= c_mid * dx)
        
        # Create and solve the optimization problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_vars.value
                h_feasible = project_h(h_feasible, n_points)
                c5_feasible = evaluate_h(h_feasible, dx)
                if c5_feasible < c5_best:
                    h_best = h_feasible.copy()
                    c5_best = c5_feasible
                c_high = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid

        if c_high - c_low < tolerance:
            break

    # Final refinement to find a slightly better solution
    h_refine = h_best.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = h_refine.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.003)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
            
    return (h_best.tolist(), c5_best, n_points)
