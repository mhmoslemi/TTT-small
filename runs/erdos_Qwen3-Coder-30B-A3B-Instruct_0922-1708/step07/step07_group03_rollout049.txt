import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    # Use CVXPY for optimization
    h_var = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1]
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Generate all correlations and create quadratic constraints
    # For each lag k, we want integral h(x)(1-h(x+k)) <= C5
    # This is equivalent to sum(h[i] * (1 - h[i+k])) * dx <= C5
    # We need to handle the circular nature in convolution
    
    # Since we're dealing with finite sequences, we use full convolution
    # and handle the periodicity via padding or clipping appropriately
    full_corr = np.zeros(2 * n_points - 1)
    full_corr_full = correlate(h, 1 - h, mode='full')
    
    # This is too slow - compute constraints more efficiently
    # Instead, use a direct gradient approach with more sophisticated optimization
    
    # Simple gradient descent with better convergence
    def compute_c5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def project(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        return h_vals * (n_points / 2) / np.sum(h_vals)
    
    # Gradient estimation using finite differences
    def estimate_grad(h_vals, eps=1e-6):
        grad = np.zeros_like(h_vals)
        c5_base = compute_c5(h_vals)
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    # Parameters for optimization
    learning_rate = 1e-3
    max_iter = 50000
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current C5
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Estimate gradient
        grad = estimate_grad(h)
        
        # Update
        h = h - learning_rate * grad
        
        # Project
        h = project(h)
    
    # Final check of best solution
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Ensure constraints are satisfied exactly
    best_h = project(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
