import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Multiple random initializations
    initializations = []
    initializations.append(h_init.copy())
    
    # Add a few variations
    for _ in range(3):
        h_var = h_init + np.random.normal(0, 0.05, n_points)
        initializations.append(project_h(h_var, n_points))
    
    # Add a second structured initialization
    h_var2 = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    h_var2 = np.sin(np.pi * x / 2) * (1 - np.sin(np.pi * x / 2))  # Beta-like shape
    initializations.append(project_h(h_var2, n_points))
    
    # Add another random perturbed version
    h_var3 = h_init + np.random.normal(0, 0.1, n_points)
    initializations.append(project_h(h_var3, n_points))
    
    h_best = None
    c5_best = float('inf')
    
    for h_start in initializations:
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current, dx)
        
        # Optimization loop with adaptive learning rate
        learning_rate = 0.01
        max_iter = int(1e6)
        patience = 50
        last_improvement = 0
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute the maximum overlap shift
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr)
            max_shift = max_shift_idx - (len(h_current) - 1)  # Convert to shift offset
            
            # Compute gradient for the worst-case shift
            # Gradient = 1 - h_shifted_for_max_shift
            h_shifted = np.roll(h_current, max_shift)
            grad = 1 - h_shifted
            
            # Update with gradient
            h_new = h_current - learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept improvement
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                last_improvement = i
            else:
                # Occasionally accept worse moves
                if np.random.rand() < 0.01:
                    h_current = h_new.copy()
                    c5_current = c5_new
                    last_improvement = i
            
            # Adaptive learning rate
            if i - last_improvement > patience:
                learning_rate *= 0.95
                last_improvement = i
                
            # Perturbation to escape local minima
            if i % 100 == 0 and i > 0:
                noise = np.random.normal(0, 0.005, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
            
            # Track best
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
        
        # Second stage: local refinement
        for _ in range(200):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_new = h_best.copy()
            # Perturb a random subset of coordinates
            indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
            for idx in indices:
                if np.random.rand() < 0.5:
                    h_new[idx] += np.random.normal(0, 0.005)
                else:
                    h_new[idx] += np.random.uniform(-0.005, 0.005)
            
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
