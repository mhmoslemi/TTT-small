import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_max_shift_gradient(h, dx):
    n_points = len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_k = np.argmax(corr)
    
    # Compute gradient only for the max shift
    grad = np.zeros(n_points)
    for i in range(n_points):
        # Gradient of the overlap with respect to h[i] for shift k
        if max_k < n_points:
            # h[i] contributes to overlap at position k
            grad[i] = 1 - h[(i + max_k) % n_points]
        else:
            # h[i] contributes to overlap at position (n_points - (max_k - n_points))
            # This is a bit tricky for the full correlation case
            actual_shift = max_k - n_points + 1
            grad[i] = 1 - h[(i + actual_shift) % n_points] if (i + actual_shift) < n_points else 0

    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular pulse
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)

    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    patience = 200
    last_improvement = 0
    
    for i in range(50000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradients
        grad = compute_max_shift_gradient(h_current, dx)
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h_new = h_current + velocity
        h_new = repair_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            h_current = h_new
            c5_current = c5_new
        
        # Adapt learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                break
    
    return (h_best.tolist(), c5_best, n_points)
