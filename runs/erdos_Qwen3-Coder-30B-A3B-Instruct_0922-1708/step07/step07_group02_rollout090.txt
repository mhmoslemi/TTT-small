import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Start with a symmetric step function
        h_init = np.zeros(n_points)
        midpoint = n_points // 2
        h_init[:midpoint] = 1.0
        h_init[midpoint:] = 0.0
        h_init = project_h(h_init, n_points, dx)

    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current

    # SOCP approach using CVXPY
    # We'll perform a few iterations of convex relaxation
    max_iter = 5000
    solver = 'ECOS'  # Fast and suitable for small to medium problems

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        try:
            # Create a relaxed version via SOCP
            h_var = cp.Variable(n_points)
            
            # Constraints
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            
            # The relaxation approach:
            # Instead of minimizing max_corr directly,
            # we add a convex constraint that bounds the correlations.
            # This is just a baseline convext relaxation to jump-start.
            # Actual maximum is still computed via correlate.

            # Minimize a relaxed version of max correlation
            # We constrain: ||h||_1 * 0.5 <= C5 (very rough)
            objective = cp.Minimize(cp.sum(h_var))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=solver, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.array(h_new)
                    h_new = project_h(h_new, n_points, dx)
                    c5_new = evaluate_h(h_new, dx)
                    
                    if c5_new < c5_current:
                        h_current = h_new
                        c5_current = c5_new
                        
                        if c5_current < c5_best:
                            h_best = h_current.copy()
                            c5_best = c5_current
            else:
                # fallback to gradient descent like approach if we cannot solve SOCP
                h_new = h_current.copy()
                indices = list(range(n_points))
                np.random.shuffle(indices)
                
                for j in indices:
                    eps = 1e-6
                    h_plus = h_new.copy()
                    h_minus = h_new.copy()
                    h_plus[j] += eps
                    h_minus[j] -= eps
                    h_plus = project_h(h_plus, n_points, dx)
                    h_minus = project_h(h_minus, n_points, dx)
                    c5_plus = evaluate_h(h_plus, dx)
                    c5_minus = evaluate_h(h_minus, dx)
                    grad_j = (c5_plus - c5_minus) / (2 * eps)
                    h_new[j] -= 0.01 * grad_j
                    h_new[j] = np.clip(h_new[j], 0, 1)
                
                h_new = project_h(h_new, n_points, dx)
                c5_new = evaluate_h(h_new, dx)
                
                if c5_new < c5_current:
                    h_current = h_new
                    c5_current = c5_new
                    
                    if c5_current < c5_best:
                        h_best = h_current.copy()
                        c5_best = c5_current
                        
        except Exception as e:
            # Fallback to gradient-like descent if SOCP fails
            h_new = h_current.copy()
            indices = list(range(n_points))
            np.random.shuffle(indices)
            
            for j in indices:
                eps = 1e-6
                h_plus = h_new.copy()
                h_minus = h_new.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                h_plus = project_h(h_plus, n_points, dx)
                h_minus = project_h(h_minus, n_points, dx)
                c5_plus = evaluate_h(h_plus, dx)
                c5_minus = evaluate_h(h_minus, dx)
                grad_j = (c5_plus - c5_minus) / (2 * eps)
                h_new[j] -= 0.01 * grad_j
                h_new[j] = np.clip(h_new[j], 0, 1)
            
            h_new = project_h(h_new, n_points, dx)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current

    return (h_best.tolist(), c5_best, n_points)
