import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize swarm
    S = 50  # swarm size
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    T = 1.0  # initial temperature
    cooling_rate = 0.999  # cooling rate for SA
    
    # Initialize particles
    particles = np.random.rand(S, n_points)
    for i in range(S):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        particles[0] = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize velocities
    velocities = np.zeros((S, n_points))
    
    # Personal best and global best
    personal_best = particles.copy()
    personal_best_fitness = np.full(S, float('inf'))
    
    global_best_idx = 0
    global_best_fitness = float('inf')
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate all particles
        for i in range(S):
            h = particles[i]
            
            # Compute correlation and C5
            corr = correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best[i] = h.copy()
                
            # Update global best
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best_idx = i
        
        # Update velocities and positions
        for i in range(S):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            
            # Velocity update
            v = (w * v + 
                 c1 * r1 * (personal_best[i] - h) + 
                 c2 * r2 * (personal_best[global_best_idx] - h))
            
            # Apply simulated annealing for diversity
            if np.random.rand() < T:
                v += np.random.uniform(-0.1, 0.1, n_points)
            
            # Update position
            h_new = h + v
            
            # Repair constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update particle
            particles[i] = h_new
            velocities[i] = v
            
        # Cooling schedule
        T = max(T * cooling_rate, 0.01)
    
    # Return best solution
    best_h = personal_best[global_best_idx]
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
