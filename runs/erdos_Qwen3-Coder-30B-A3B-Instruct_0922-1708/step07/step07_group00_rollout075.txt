import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate multiple initializations with Sobol sequences
    sampler = qmc.Sobol(n=n_points, seed=seed)
    initial_points = sampler.random(n=500)  # 500 different initializations
    initial_points = np.clip(initial_points, 0, 1)  # Ensure in [0,1]
    
    # Warm-up with the initial h if available
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Optimization loop
    for i, point in enumerate(initial_points):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create initial h from Sobol point
        h = point.copy()
        h = h * (n_points / 2) / np.sum(h)  # Ensure sum constraint
        
        # Nelder-Mead optimization
        def objective(h_vec):
            h_repaired = repair_h(h_vec, n_points)
            return evaluate_h(h_repaired, dx)
            
        try:
            res = minimize(objective, h, method='Nelder-Mead', options={
                'maxiter': 1000, 'xatol': 1e-6, 'fatol': 1e-6
            })
            if res.success:
                h_opt = repair_h(res.x, n_points)
                c5_opt = evaluate_h(h_opt, dx)
                
                # Gradient refinement
                h_refined = h_opt.copy()
                lr = 1e-5
                for _ in range(1000):
                    if time.time() - start_time > budget_s - 5:
                        break
                    # Compute gradient using finite differences
                    eps = 1e-6
                    grad = np.zeros(n_points)
                    for j in range(n_points):
                        h_plus = h_refined.copy()
                        h_plus[j] += eps
                        h_plus = repair_h(h_plus, n_points)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                        max_k = np.argmax(corr_plus)
                        corr_plus_max = corr_plus[max_k]
                        
                        h_minus = h_refined.copy()
                        h_minus[j] -= eps
                        h_minus = repair_h(h_minus, n_points)
                        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                        max_k = np.argmax(corr_minus)
                        corr_minus_max = corr_minus[max_k]
                        
                        grad[j] = (corr_plus_max - corr_minus_max) / (2 * eps)
                    
                    # Update step
                    h_refined = h_refined - lr * grad
                    h_refined = repair_h(h_refined, n_points)
                    corr = correlate(h_refined, 1 - h_refined, mode='full')
                    c5_refined = np.max(corr) * dx
                    
                    if c5_refined < c5_opt:
                        h_opt = h_refined.copy()
                        c5_opt = c5_refined
                        
                    lr *= 0.995  # Adaptive decay
                    if lr < 1e-8:
                        lr = 1e-8
                        
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except Exception:
            continue
    
    # Final verification
    best_h = repair_h(best_h, n_points)
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
