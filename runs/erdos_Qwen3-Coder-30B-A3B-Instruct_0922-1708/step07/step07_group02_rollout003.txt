import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize swarm
    swarm_size = 50
    particles = np.zeros((swarm_size, n_points))
    velocities = np.zeros((swarm_size, n_points))
    pbest = np.zeros((swarm_size, n_points))
    pbest_scores = np.full(swarm_size, np.inf)
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 0.5
        h_init = project_h(h_init, n_points, dx)
    
    # First particle from initial solution
    particles[0] = h_init
    pbest[0] = h_init
    pbest_scores[0] = evaluate_h(h_init, dx)
    
    # Other particles with structured starts
    for i in range(1, swarm_size):
        # Mix of different initializations
        if i % 3 == 0:
            # Sawtooth shape
            h = np.linspace(0, 1, n_points//2)
            h = np.concatenate([h, np.zeros(n_points - len(h))])
        elif i % 3 == 1:
            # Step functions
            h = np.zeros(n_points)
            h[:n_points//3] = 0.7
            h[n_points//3:2*n_points//3] = 0.3
        else:
            # Random with constraint satisfaction
            h = np.random.rand(n_points)
            h = project_h(h, n_points, dx)
        particles[i] = h
        pbest[i] = h
        pbest_scores[i] = evaluate_h(h, dx)
    
    # Global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    # PSO parameters
    w_start, w_end = 0.9, 0.4
    c1 = c2 = 1.5
    vMax = 0.1
    stagnation_count = 0
    max_stagnation = 200
    
    # PSO optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        w = w_start - (w_start - w_end) * iter_count / 1000

        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -vMax, vMax)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_h(particles[i], n_points, dx)
            
            # Evaluate fitness
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                
                # Update global best
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        
        # Perturbation if stagnation
        if stagnation_count > max_stagnation:
            for i in range(swarm_size):
                rand_idx = np.random.randint(0, n_points)
                particles[i][rand_idx] += np.random.uniform(-0.1, 0.1)
                particles[i][rand_idx] = np.clip(particles[i][rand_idx], 0, 1)
                # Re-project
                particles[i] = project_h(particles[i], n_points, dx)
                score = evaluate_h(particles[i], dx)
                if score < pbest_scores[i]:
                    pbest_scores[i] = score
                    pbest[i] = particles[i].copy()
                    if score < gbest_score:
                        gbest = particles[i].copy()
                        gbest_score = score
            stagnation_count = 0
    
    return (gbest.tolist(), gbest_score, n_points)
