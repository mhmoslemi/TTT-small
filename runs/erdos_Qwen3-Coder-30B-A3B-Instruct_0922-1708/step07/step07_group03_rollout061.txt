import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Better initialization with approximate solution
        h = np.ones(n_points) * 0.5
        # Make it slightly more structured
        mid = n_points // 2
        h[:mid] = 0.5
        h[mid:] = 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Annealing parameters
    temp_start = 0.1
    cooling_rate = 0.995
    min_temp = 1e-6
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Helper functions
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-10:
            h = h * (n_points / 2) / h_sum
        return h
    
    def evaluate_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def perturb_h(h, delta=0.05):
        h_new = h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        # Adjust two points
        delta_h = np.random.uniform(-delta, delta)
        h_new[i] += delta_h
        h_new[j] -= delta_h
        h_new = project_h(h_new)
        return h_new
    
    # Main loop
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h)
    temp = temp_start
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate new candidate
        candidate_h = perturb_h(current_h, delta=0.02 + temp)
        candidate_c5 = evaluate_c5(candidate_h)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            current_h = candidate_h
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = candidate_h.copy()
        else:
            # Accept with probability
            if temp > min_temp:
                prob = np.exp(-(candidate_c5 - current_c5) / temp)
                if np.random.rand() < prob:
                    current_h = candidate_h
                    current_c5 = candidate_c5
        
        # Update temperature
        temp *= cooling_rate
        
        # Occasionally reinitialize with a better solution if found
        if iteration % 1000 == 0 and iteration > 0:
            temp = max(temp, temp_start * 0.5)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
