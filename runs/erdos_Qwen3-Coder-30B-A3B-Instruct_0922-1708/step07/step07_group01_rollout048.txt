import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)

    # Initialize
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Binary search for minimal C
    c_low = 0.0
    c_high = 0.5
    tolerance = 1e-5
    max_iterations = 50
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        c = (c_low + c_high) / 2.0
        
        # Create and solve the feasibility LP
        h = cp.Variable(n_points)
        constraints = []
        
        # Constraint: h in [0, 1]
        constraints += [h >= 0]
        constraints += [h <= 1]
        
        # Constraint: integral = 1
        constraints += [cp.sum(h) == n_points / 2]
        
        # Constraints: overlap inequalities
        for k in range(1, n_points):
            # This is the k-th overlap integral
            # We do not use cvxpy to compute correlations directly 
            # but instead do feasibility check based on all k values
            # Note: we allow numerical errors in this implementation
            # but it's a good enough heuristic for this problem
            
        # We'll optimize using direct projection and gradient descent on 
        # heuristic overlap calculation that captures main characteristics
        
        # Use the existing gradient-based approach
        h_current = h_best.copy()
        
        # Try to improve with multiple restarts from different initializations
        num_restarts = 5
        for restart in range(num_restarts):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Random small perturbation
            h_rand = h_current.copy()
            noise = np.random.normal(0, 0.01, n_points)
            h_rand = project_h(h_rand + noise, n_points)
            
            # Local optimization for this restart
            for step in range(200):
                if time.time() - start_time > budget_s - 5:
                    break
                    
                # Compute finite difference approx gradient
                grad = np.zeros(n_points)
                eps = 1e-5
                
                for j in range(n_points):
                    h_plus = h_rand.copy()
                    h_minus = h_rand.copy()
                    
                    h_plus[j] += eps
                    h_minus[j] -= eps
                    
                    h_plus = project_h(h_plus, n_points)
                    h_minus = project_h(h_minus, n_points)
                    
                    c_plus = evaluate_h(h_plus, dx)
                    c_minus = evaluate_h(h_minus, n_points)
                    
                    grad[j] = (c_plus - c_minus) / (2 * eps)
                
                # Update step
                h_new = h_rand - 0.02 * grad
                h_new = project_h(h_new, n_points)
                
                c_new = evaluate_h(h_new, dx)
                
                if c_new < c_rand:
                    h_rand = h_new.copy()
                    c_rand = c_new
                else:
                    # Decrease the learning rate periodically
                    h_rand = project_h(h_rand + np.random.normal(0, 0.001, n_points), n_points)
                    c_rand = evaluate_h(h_rand, dx)
            
            # Update best if we improved
            c_rand = evaluate_h(h_rand, dx)
            if c_rand < c5_best:
                c5_best = c_rand
                h_best = h_rand.copy()
        
        # Update binary search bounds
        if c5_best <= c:
            c_high = c
        else:
            c_low = c
            
        if c_high - c_low < tolerance:
            break
    
    # Final local search
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 10), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
