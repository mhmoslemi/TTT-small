import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize swarm
    swarm_size = 50
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
        
    # Add initial solution to swarm
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Fill swarm with variation of different patterns
    for i in range(swarm_size - 1):
        h = np.random.random(n_points)
        # Add some structure - sawtooth pattern
        if i % 3 == 0:
            h = np.sin(np.linspace(0, 4*np.pi, n_points)) * 0.5 + 0.5
        elif i % 3 == 1:
            # Concentrated regions
            h = np.zeros(n_points)
            h[:n_points//3] = 1.0
            h[n_points//3:2*n_points//3] = 0.3
        # Project to constraints
        h = project_h(h, n_points, dx)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Initialize bests
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [evaluate_h(p, dx) for p in particles]
    
    # Global best
    gb_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[gb_idx].copy()
    global_best_fitness = personal_best_fitness[gb_idx]
    
    # Optimization loop
    max_iter = 100000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        # Update particles
        for i in range(swarm_size):
            # Velocity update
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (personal_best[i] - particles[i])
                           + c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Position update
            particles[i] += velocities[i]
            
            # Project position to feasible region
            particles[i] = project_h(particles[i], n_points, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best if needed
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
                    
    # Return best solution found
    return (global_best.tolist(), global_best_fitness, n_points)
