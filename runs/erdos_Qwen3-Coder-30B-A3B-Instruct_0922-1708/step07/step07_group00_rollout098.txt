import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(params, n_points):
    """Convert parameters to h(x) function."""
    K = len(params) // 2
    heights = params[:K]
    boundaries = np.concatenate([[0], params[K:]])
    boundaries = np.clip(boundaries, 0, 2)
    boundaries = np.sort(boundaries)
    boundaries = np.concatenate([boundaries, [2]])
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(K):
        start_idx = int(boundaries[i] / dx)
        end_idx = int(boundaries[i+1] / dx)
        if end_idx > start_idx:
            h[start_idx:end_idx] = heights[i]
    
    return repair_h(h, n_points)

def compute_gradient_h(params, n_points, dx, k_max):
    """Compute gradient of C5 with respect to params."""
    h = parameterize_h(params, n_points)
    n = len(h)
    
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n - 1)
    
    eps = 1e-6
    grad = np.zeros_like(params)
    for i in range(len(params)):
        params_plus = params.copy()
        params_plus[i] += eps
        h_plus = parameterize_h(params_plus, n_points)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
        
    return grad
