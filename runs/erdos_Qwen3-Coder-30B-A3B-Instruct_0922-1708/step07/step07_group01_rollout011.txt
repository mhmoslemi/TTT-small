import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Multiple initializations
    initializations = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        initializations.append(project_h(h_init, n_points))
    else:
        # Sine wave
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.sin(np.pi * x))
        initializations.append(project_h(h_init, n_points))
        
        # Trapezoidal
        h_init = 0.5 * (1 + np.cos(np.pi * x))
        initializations.append(project_h(h_init, n_points))
    
    # Random initialization
    h_init = np.random.rand(n_points)
    initializations.append(project_h(h_init, n_points))
    
    # Uniform initial
    h_init = np.ones(n_points) * 0.5
    initializations.append(project_h(h_init, n_points))
    
    h_best = None
    c5_best = float('inf')
    
    # For each initialization, optimize with gradient descent
    for i, h_start in enumerate(initializations):
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
        
        # Gradient descent
        learning_rate = 0.01
        max_iter = int(50000)
        patience = 200
        last_improvement = 0
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Identify the maximum overlap shift
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr)
            max_shift = max_shift_idx - (n_points - 1)
            
            # Compute gradient w.r.t. max overlap only
            # For shift k_max, gradient is (1 - h_{k_max})
            h_shifted = np.roll(h_current, -max_shift)
            grad = 1 - h_shifted
            
            # Update with gradient
            h_new = h_current + learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate
            c5_new = evaluate_h(h_new, dx)
            
            # Accept or reject
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                last_improvement = iteration
            else:
                # Occasionally accept worse
                if np.random.rand() < 0.005:
                    h_current = h_new.copy()
                    c5_current = c5_new
            
            # Adaptive rate
            if iteration - last_improvement > patience:
                learning_rate *= 0.95
                if learning_rate < 1e-6:
                    break
                last_improvement = iteration
            
            # Occasionally jitter
            if iteration % 200 == 0 and iteration > 0:
                noise = np.random.normal(0, 0.003, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
    
    # Final local refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = h_best.copy()
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.003)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
            
    return (h_best.tolist(), c5_best, n_points)
