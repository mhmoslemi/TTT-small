import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Initialize PSO
    m = 100  # swarm size
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Initialize particles
    for i in range(m):
        if i == 0 and 'initial_h_values' in globals():
            h = h_init.copy()
        else:
            h = np.random.rand(n_points)
            h = repair_h(h, n_points)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_h(h, dx))
    
    # Swarm best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    max_iter = 10000
    patience = 1000
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * i / max_iter
        
        for p in range(m):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[p] = (w * velocities[p] + 
                            c1 * r1 * (pbest[p] - particles[p]) + 
                            c2 * r2 * (gbest - particles[p]))
            
            # Clip velocity
            velocities[p] = np.clip(velocities[p], -0.2, 0.2)
            
            # Update position
            particles[p] = particles[p] + velocities[p]
            
            # Repair position
            particles[p] = repair_h(particles[p], n_points)
            
            # Evaluate
            current_val = evaluate_h(particles[p], dx)
            
            # Update personal best
            if current_val < pbest_vals[p]:
                pbest[p] = particles[p].copy()
                pbest_vals[p] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = particles[p].copy()
                    gbest_val = current_val
                    last_improvement = i
                    
        # Check for early termination
        if i - last_improvement > patience:
            break
    
    # Gradient refinement for the global best
    h_current = gbest.copy()
    c5_current = gbest_val
    eta = 0.001
    eps = 1e-6
    max_iter_refine = 500
    
    for i in range(max_iter_refine):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Compute gradient
            grad[j] = (c5_plus - c5_current) / eps
        
        # Update
        h_current = h_current - eta * grad
        h_current = repair_h(h_current, n_points)
        
        # Evaluate new solution
        c5_current = evaluate_h(h_current, dx)
    
    return (h_current.tolist(), c5_current, n_points)
