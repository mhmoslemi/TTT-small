import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradients(h, dx, k_max):
    """Compute exact gradient for the maximum overlap shift k_max."""
    n = len(h)
    grad = np.zeros(n)
    # Compute gradient only for the k_max shift that produces maximum overlap
    # For shift k_max, overlap is sum over i of h[i] * (1 - h[i+k_max])
    # So gradient is -1 for i, +1 for i-k_max
    for i in range(n):
        if i - k_max >= 0:
            grad[i] -= 1
        if i + k_max < n:
            grad[i] += 1
    return grad * dx

def find_max_shift(h):
    """Find the shift k that maximizes the overlap."""
    corr = correlate(h, 1 - h, mode='full')
    k_max = np.argmax(corr)
    return k_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = int(1e6)
    epsilon = 1e-6
    patience = 100
    last_improvement = 0
    
    # Try multiple random restarts
    for restart in range(5):
        # Start fresh for each restart
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        local_patience = 0
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break()
                
            # Find the shift k_max that produces maximum overlap
            k_max = find_max_shift(h_current)
            
            # Compute exact gradient for that shift
            grad = compute_gradients(h_current, dx, k_max)
            
            # Update with gradient
            h_new = h_current - learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept move if improvement
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                local_patience = 0
            else:
                local_patience += 1
                
            # Update best
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
            # Adaptive learning rate
            if local_patience > patience:
                learning_rate *= 0.95
                local_patience = 0
                
            # Occasionally jitter to escape local minima
            if i % 500 == 0 and i > 0:
                noise = np.random.normal(0, 0.005, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
            
            # Occasionally restore best solution to avoid getting trapped
            if i % 2000 == 0:
                h_current = h_best.copy()
                c5_current = c5_best
        
        # Perturb initial for next restart
        h_init = h_best + np.random.normal(0, 0.001, n_points)
        h_init = project_h(h_init, n_points)
    
    return (h_best.tolist(), c5_best, n_points)
