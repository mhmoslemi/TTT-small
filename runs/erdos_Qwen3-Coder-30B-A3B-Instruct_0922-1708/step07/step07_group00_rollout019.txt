import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Evolution Strategy parameters
    pop_size = 200
    elites_frac = 0.2
    offsprings_per_elite = 5
    initial_mutation_std = 0.1
    patience = 30
    
    # Initialize population
    pop = [h_init.copy()]
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        pop.append(h_rand)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 5:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in pop:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        elites = [pop[i] for i in sorted_indices[:int(elites_frac * pop_size)]]
        best_in_gen = fitness[sorted_indices[0]]
        
        # Update best solution
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = elites[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= patience:
            break
            
        # Generate new population
        new_pop = elites.copy()
        mutation_std = initial_mutation_std * (0.95 ** generation)
        
        for elite in elites:
            for _ in range(offsprings_per_elite):
                mutant = elite + np.random.randn(n_points) * mutation_std
                mutant = repair_h(mutant, n_points)
                new_pop.append(mutant)
        
        # Keep best pop_size individuals
        pop = new_pop[:pop_size]
    
    return (best_h.tolist(), best_c5, n_points)
