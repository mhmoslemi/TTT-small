import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    # Precompute all correlations for efficiency
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # CVXPY optimization
    h_var = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1]
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Add quadratic constraints for each lag
    max_lag = 2 * n_points - 1
    full_corr = correlate(h_var, 1 - h_var, mode='full')
    
    # For a convex relaxation approach, we will add constraints for a few key lags
    # instead of all, to reduce computational complexity
    all_lags = np.arange(max_lag)
    sub_lags = np.unique(np.linspace(0, max_lag-1, min(50, max_lag), dtype=int))
    
    for k in sub_lags:
        if k < len(full_corr):
            # We need to bound the correlation entry by C5
            constraint = full_corr[k] <= C5
            constraints.append(constraint)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers_to_try = ['MOSEK', 'GUROBI', 'ECOS']
    solver = None
    for s in solvers_to_try:
        try:
            if s == 'MOSEK' and cp.MOSEK in cp.installed_solvers():
                solver = s
                break
            elif s == 'GUROBI' and cp.GUROBI in cp.installed_solvers():
                solver = s
                break
            elif s == 'ECOS' and cp.ECOS in cp.installed_solvers():
                solver = s
                break
        except:
            continue
    
    if solver is None:
        solver = 'ECOS'
    
    # Optimization loop with early stopping
    best_c5 = float('inf')
    best_h = h.copy()
    max_iter = 1000
    iter_count = 0
    
    current_h = h.copy()
    
    # Use gradient based optimization to guide local search
    learning_rate = 1e-3
    eps = 1e-5
    
    while (time.time() - start_time) < budget_s - 5 and iter_count < max_iter:
        iter_count += 1
        if (time.time() - start_time) > budget_s - 10:
            break
            
        # First try the continuous relaxation
        try:
            h_var.value = current_h
            prob.solve(solver=solver, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_new = h_var.value
                c5_new = C5.value
                if c5_new is not None and c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except Exception as e:
            pass
            
        # Gradient estimation and update (for local refinement)
        c5_current = compute_c5(current_h)
        grad = np.zeros_like(current_h)
        
        for i in range(n_points):
            h_plus = current_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_current) / eps
            
        # Gradient step
        current_h = current_h - learning_rate * grad
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # Final check
        c5_final = compute_c5(current_h)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = current_h.copy()
            
    # Use final computed best_h
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
