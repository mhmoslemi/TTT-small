import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(population, parents, idx1, idx2, n_points):
    """Create offspring via segment-wise blending."""
    parent1 = population[parents[idx1]]
    parent2 = population[parents[idx2]]
    
    # Blend crossover
    offspring = np.zeros(n_points)
    segment_len = n_points // 4
    for i in range(4):
        start = i * segment_len
        end = start + segment_len if i < 3 else n_points
        if i % 2 == 0:
            offspring[start:end] = parent1[start:end]
        else:
            offspring[start:end] = parent2[start:end]
    return offspring

def mutate(individual, n_points, mutation_scale):
    """Apply mutation to an individual."""
    mutated = individual.copy()
    # Random segment flips
    if np.random.rand() < 0.3:
        start = np.random.randint(0, n_points)
        end = min(start + np.random.randint(1, n_points//10), n_points)
        mutated[start:end] = 1 - mutated[start:end]
    # Add Gaussian noise
    noise = np.random.normal(0, mutation_scale, n_points)
    mutated += noise
    return project_h(mutated, n_points, 2.0/n_points)

def local_refine(h, n_points, dx, max_iter=50):
    """Refine h using fast coordinate descent."""
    h_refined = h.copy()
    eta = 0.01
    eps = 1e-6
    
    for _ in range(max_iter):
        h_new = h_refined.copy()
        improved = False
        
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
        
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        c5_old = evaluate_h(h_refined, dx)
        
        if c5_new < c5_old:
            h_refined = h_new
            improved = True
            
        if not improved:
            eta *= 0.9995
        else:
            eta = min(0.01, eta * 1.001)
            
    return h_refined

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    # Generate initial population
    for _ in range(population_size):
        if np.random.rand() < 0.7:
            h = h_init.copy()
            # Add some noise to diversify
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = project_h(h, n_points, dx)
        else:
            h = np.random.rand(n_points)
            h = project_h(h, n_points, dx)
        population.append(h)
    
    # Best solution tracking
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    mutation_scale = 0.1
    generation = 0
    
    while time.time() - start_time < budget_s - 5:
        generation += 1
        
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Selection - top 20%
        sorted_indices = np.argsort(fitness)
        parents = sorted_indices[:population_size//5]
        
        # Create new population
        new_population = []
        
        # Keep best individuals
        for i in range(population_size//5):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent_idx1 = np.random.choice(parents)
            parent_idx2 = np.random.choice(parents)
            
            offspring = crossover(population, [parent_idx1, parent_idx2], 0, 1, n_points)
            
            # Add mutation
            offspring = mutate(offspring, n_points, mutation_scale)
            
            # Local refinement
            offspring = local_refine(offspring, n_points, dx, 50)
            
            new_population.append(offspring)
        
        # Update population
        population = new_population
        
        # Update best solution
        for h in population:
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Adaptive mutation
        mutation_scale *= 0.999
        
        # Periodic injection of new diversity
        if generation % 20 == 0:
            for i in range(population_size // 10):
                new_individual = np.random.rand(n_points)
                new_individual = project_h(new_individual, n_points, dx)
                population[np.random.randint(0, population_size)] = new_individual
    
    return (best_h.tolist(), best_c5, n_points)
