import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx):
    """Compute gradient of the maximum overlap with respect to h."""
    n = len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n - 1)
    
    # Analytical gradient approximation
    eps = 1e-6
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    # Normalize
    grad = grad / (np.linalg.norm(grad) + 1e-12)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    n_particles = 100
    w_max, w_min = 0.9, 0.1
    c1, c2 = 2.0, 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_scores = []
    
    for _ in range(n_particles):
        h = h_init + np.random.normal(0, 0.01, n_points)
        h = repair_h(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        p_best.append(h.copy())
        p_best_scores.append(evaluate_h(h, dx))
    
    # Global best tracking
    g_best_idx = np.argmin(p_best_scores)
    g_best = p_best[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    # PSO iterations
    max_iter = 5000
    stagnation_count = 0
    stagnation_limit = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_min + (w_max - w_min) * (1 - iteration / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair and re-evaluate
            particles[i] = repair_h(particles[i], n_points)
            current_score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_score < p_best_scores[i]:
                p_best[i] = particles[i].copy()
                p_best_scores[i] = current_score
                
                # Update global best
                if current_score < g_best_score:
                    g_best = particles[i].copy()
                    g_best_score = current_score
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
                
        # Stagnation check
        if stagnation_count >= stagnation_limit:
            # Reinitialize some particles
            for i in range(n_particles // 10):
                idx = np.random.randint(0, n_particles)
                particles[idx] = h_init + np.random.normal(0, 0.02, n_points)
                particles[idx] = repair_h(particles[idx], n_points)
                p_best[idx] = particles[idx].copy()
                p_best_scores[idx] = evaluate_h(particles[idx], dx)
            stagnation_count = 0
            
    # Local refinement of the global best
    h_refined = g_best.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        grad = compute_gradient(h_refined, dx)
        h_refined = h_refined - 1e-5 * grad
        h_refined = repair_h(h_refined, n_points)
        current_c5 = evaluate_h(h_refined, dx)
        if current_c5 < g_best_score:
            g_best = h_refined.copy()
            g_best_score = current_c5
    
    return (g_best.tolist(), g_best_score, n_points)
