import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidate(n_points, seed=None):
    if seed is not None:
        np.random.seed(seed)
    # Generate a candidate using sum of cos/sin modes
    h = np.zeros(n_points)
    # Use 3-5 modes
    n_modes = np.random.randint(3, 6)
    for _ in range(n_modes):
        a = np.random.uniform(-1, 1)
        b = np.random.uniform(-1, 1)
        k = np.random.uniform(1, 10)
        x = np.linspace(0, 2, n_points)
        h += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
    
    # Ensure h is in [0,1] and sum to n_points/2
    h = h - h.min()
    h = h / (h.max() - h.min()) if h.max() > h.min() else h
    h = repair_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Create a composite initial solution with diverse profiles
        h_init = np.zeros(n_points)
        # Two step functions with overlapping regions
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.0:
                h_init[i] = 1.0
            elif x < 1.5:
                h_init[i] = 1.0 - (x - 1.0) * 2
            else:
                h_init[i] = 0.0
        h_init = repair_h(h_init, n_points)
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Generate 50 diverse candidates using Fourier components
    candidates = []
    for i in range(50):
        if time.time() - start_time > budget_s - 5:
            break
        h_candidate = generate_fourier_candidate(n_points, seed=seed + i)
        c5_candidate = evaluate_h(h_candidate, dx)
        candidates.append((h_candidate, c5_candidate))
    
    # Sort and take top 10
    candidates.sort(key=lambda x: x[1])
    top_candidates = candidates[:10]
    
    # Refinement phase
    max_iter = 2000
    patience = 200
    for h_cur, c5_cur in top_candidates:
        if time.time() - start_time > budget_s - 5:
            break
        h_current = h_cur.copy()
        c5_current = c5_cur
        lr = 0.01
        last_improvement = 0
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            # Simple gradient approximation
            grad = np.zeros(n_points)
            eps = 1e-5
            for j in range(n_points):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                h_plus = repair_h(h_plus, n_points)
                h_minus = repair_h(h_minus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                c5_minus = evaluate_h(h_minus, dx)
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            h_new = h_current - lr * grad
            h_new = repair_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
            elif np.random.rand() < np.exp(-(c5_new - c5_current) / (c5_current * 0.01)):
                h_current = h_new
                c5_current = c5_new
            
            if i - last_improvement > patience:
                lr *= 0.95
                if lr < 1e-6:
                    break
    
    return (h_best.tolist(), c5_best, n_points)
