import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Create a structured start from different patterns
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Cosine taper as a starting point
        h_init = 0.5 * (1 + np.cos(np.pi * x))
        h_init = project_h(h_init, n_points)
    
    # Multiple random starts
    num_starts = 10
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    for start in range(num_starts):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Initialize with current best or random perturbation
        if start == 0:
            h_current = h_init.copy()
        else:
            h_current = h_init.copy()
            # Add small random noise
            noise = np.random.normal(0, 0.01, n_points)
            h_current += noise
            h_current = project_h(h_current, n_points)
            
        c5_current = evaluate_h(h_current, dx)
        
        # Optimizer parameters
        learning_rate = 0.05
        max_iter = 5000
        patience = 100
        last_improvement = 0
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Find current maximum overlap shift
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr)
            max_shift = max_shift_idx - (n_points - 1)  # Convert to actual shift
            
            # Compute gradient at max shift
            h_shifted = np.roll(1 - h_current, max_shift)
            grad = -h_shifted  # Gradient of the overlap at max shift
            
            # Update with gradient
            h_new = h_current - learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            # Accept move
            h_current = h_new
            c5_current = c5_new
            
            # Update best
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                # Reduce learning rate for fine-tuning
                if learning_rate > 0.005:
                    learning_rate *= 0.95
            
            # Early stopping
            if i - last_improvement > patience:
                break
                
            # Periodic perturbation
            if i % 200 == 0 and i > 0:
                noise = np.random.normal(0, 0.005, n_points)
                h_current += noise
                h_current = project_h(h_current, n_points)
                c5_current = evaluate_h(h_current, dx)
        
        # Final refinement with local search
        for _ in range(500):
            if time.time() - start_time > budget_s - 5:
                break
            h_new = h_best.copy()
            # Randomly perturb a few points
            indices = np.random.choice(n_points, max(2, n_points // 20), replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.005)
            
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
