import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a trapezoidal waveform
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient-based search with local refinement
    step_size = 0.02
    max_iter = int(1e6)
    epsilon = 1e-6
    patience = 200
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradients using central differences
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            
            h_plus = project_h(h_plus, n_points)
            h_minus = project_h(h_minus, n_points)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update with gradient
        h_new = h_current - step_size * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement or with small probability
        if c5_new < c5_current or np.random.rand() < 0.05:
            h_current = h_new
            c5_current = c5_new
        
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            
        # Adaptive step size
        if i % 500 == 0 and i > 0:
            if c5_current > c5_best + 1e-5:
                step_size *= 0.95
            else:
                step_size *= 1.05
                
        # Occasionally jitter to escape local minima
        if i % 1000 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Final refinement using local perturbations
    for _ in range(2000):
        if time.time() - start_time > budget_s - 10:
            break
            
        h_new = h_best.copy()
        perturb_indices = np.random.choice(n_points, max(1, n_points // 30), replace=False)

            h_new[idx] += np.random.normal(0, 0.01)

        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    # Final repair
    h_final = project_h(h_best, n_points)
    c5_final = evaluate_h(h_final, dx)
    
    return (h_final.tolist(), c5_final, n_points)
