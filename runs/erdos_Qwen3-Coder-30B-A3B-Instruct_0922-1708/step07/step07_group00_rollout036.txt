import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Fourier-based optimization
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update via FFT-based approach
        H = fft(current_h)
        H_magnitude = np.abs(H)
        H_phase = np.angle(H)
        
        # Optimize magnitude by solving a related convex problem
        # We'll do a simplified optimization to reduce magnitude of large correlations
        # This is an approximate method to encourage the correlation to be smaller
        H_new_magnitude = H_magnitude.copy()
        
        # Heuristic: reduce magnitude in areas that contribute to high correlation
        # We look at frequency domain properties to encourage smoother h
        H_new_magnitude = H_new_magnitude ** 0.8  # Slight smoothing of spectrum
        
        # Reconstruct with new magnitude but same phase
        H_new = H_new_magnitude * np.exp(1j * H_phase)
        h_new = np.real(ifft(H_new))
        
        # Repair to feasible solution
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Adaptive step for the current iteration
        current_h = h_new.copy()
    
    # Final refinement using local search
    h_refined = best_h.copy()
    refine_steps = 1000
    for i in range(refine_steps):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Small random perturbations
        perturbation = np.random.normal(0, 1e-5, n_points)
        h_trial = h_refined + perturbation
        h_trial = repair_h(h_trial, n_points)
        c5_trial = evaluate_h(h_trial, dx)
        
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            h_refined = h_trial.copy()
    
    return (best_h.tolist(), best_c5, n_points)
