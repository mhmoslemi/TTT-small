import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Create swarm with multiple initializations
    swarm_size = 50
    particles = []
    velocities = []
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    
    # Fill the rest with structured patterns
    for _ in range(swarm_size - 1):
        h = np.zeros(n_points)
        # Mix structured with randomness
        if np.random.random() < 0.5:
            # Sawtooth pattern
            h = np.linspace(0, 1, n_points//2) 
            h = np.concatenate([h, np.zeros(n_points - n_points//2)])
        else:
            # Two segments
            mid = n_points // 2
            h[:mid] = 0.7
            h[mid:] = 0.3
        h = project_h(h, n_points, dx)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive = 1.5
    social = 1.5
    vel_max = 0.1
    stagnation_threshold = 500
    
    # personal best tracking
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [evaluate_h(p, dx) for p in particles]
    
    # swarm best tracking
    best_idx = np.argmin(personal_best_fitness)
    swarm_best = personal_best[best_idx].copy()
    swarm_best_fitness = personal_best_fitness[best_idx]
    
    # Main PSO loop
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 5 and iteration < 100000:
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * iteration / 10000
        
        for i in range(swarm_size):
            # Update velocity and position
            r1, r2 = np.random.random(2)
            vel = (inertia * velocities[i] + 
                   cognitive * r1 * (personal_best[i] - particles[i]) + 
                   social * r2 * (swarm_best - particles[i]))
            
            # Clip velocity
            vel = np.clip(vel, -vel_max, vel_max)
            velocities[i] = vel
            
            # Update particle position
            new_pos = particles[i] + vel
            
            # Project to feasible region
            new_pos = project_h(new_pos, n_points, dx)
            
            # Evaluate
            fit = evaluate_h(new_pos, dx)
            
            # Update personal best
            if fit < personal_best_fitness[i]:
                personal_best[i] = new_pos.copy()
                personal_best_fitness[i] = fit
                
                # Update swarm best
                if fit < swarm_best_fitness:
                    swarm_best = new_pos.copy()
                    swarm_best_fitness = fit
                    last_improvement = iteration
                    
            particles[i] = new_pos
        
        # Stagnation detection and perturbation
        if iteration - last_improvement > stagnation_threshold:
            for i in range(swarm_size):
                # Add perturbation to particles around personal best
                perturbation = np.random.normal(0, 0.01, n_points)
                particles[i] = project_h(particles[i] + perturbation, n_points, dx)
                fit = evaluate_h(particles[i], dx)
                if fit < personal_best_fitness[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_fitness[i] = fit
                    if fit < swarm_best_fitness:
                        swarm_best = particles[i].copy()
                        swarm_best_fitness = fit
                        last_improvement = iteration
        
        iteration += 1
    
    return (swarm_best.tolist(), swarm_best_fitness, n_points)
