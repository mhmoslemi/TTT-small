import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter_barrier = 1000
    max_iter_gd = 10000
    learning_rate = 1e-3
    mu_initial = 1e3
    barrier_decrease_factor = 0.99
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    # Compute initial C5
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    # Barrier method for constraints enforcement
    mu = mu_initial
    for barrier_iter in range(max_iter_barrier):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Find maximum correlation and at which lag
        corr = correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr = corr[max_corr_idx] * dx
        max_lag = max_corr_idx - (len(h) - 1)  # compute actual lag
        
        # Compute gradient via adjoint method
        grad = np.zeros_like(h)
        # This computes gradient for lag that achieves maximum overlap
        shifted = np.zeros_like(h)
        shifted[max_lag:] = h[:-max_lag] if max_lag > 0 else h[-max_lag:]
        grad = 1 - shifted  # Gradient computation for the particular lag
        
        # Update step
        h = h - learning_rate * grad
        
        # Project to [0,1] 
        h = np.clip(h, 0, 1)
        
        # Rescale to maintain constraint sum(h) = n_points / 2
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
            
        # Update barrier parameter
        mu *= barrier_decrease_factor
        
        # Track best solution
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Fine tune with standard gradient descent
    for gd_iter in range(max_iter_gd):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Find maximum correlation and at which lag
        corr = correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr = corr[max_corr_idx] * dx
        max_lag = max_corr_idx - (len(h) - 1) 
        
        # Compute gradient via adjoint
        grad = np.zeros_like(h)
        # This computes gradient for lag that achieves maximum overlap
        shifted = np.zeros_like(h)
        shifted[max_lag:] = h[:-max_lag] if max_lag > 0 else h[-max_lag:]
        grad = 1 - shifted
        
        # Update step
        h = h - learning_rate * grad
        
        # Project to [0,1] 
        h = np.clip(h, 0, 1)
        
        # Rescale to maintain constraint sum(h) = n_points / 2
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
            
        # Track best solution
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final check
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
