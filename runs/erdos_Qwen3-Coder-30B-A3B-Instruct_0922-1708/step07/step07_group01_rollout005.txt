import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Use a piecewise constant construction for good baseline
        h_init[:n_points//4] = 0.5
        h_init[n_points//4:3*n_points//4] = 1.0
        h_init[3*n_points//4:] = 0.0
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Simulated Annealing with local search
    T = 1.0
    Tmin = 1e-6
    alpha = 0.995
    max_iter = int(1e6)
    patience = 1000
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate neighbor by perturbing one element
        idx = np.random.randint(0, n_points)
        h_new = h_current.copy()
        delta = np.random.normal(0, 0.02)
        h_new[idx] += delta
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept with probability based on Boltzmann distribution
        if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / T):
            h_current = h_new.copy()
            c5_current = c5_new
        
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Cooling schedule
        if i - last_improvement > patience:
            T = max(Tmin, T * alpha)
            last_improvement = i
            
        # Periodic refinement with local search
        if i % 100 == 0 and i > 0:
            h_local = h_best.copy()
            local_improvements = 0
            for _ in range(500):
                if time.time() - start_time > budget_s - 5:
                    break
                # Perturb a few elements
                indices = np.random.choice(n_points, max(2, n_points // 20), replace=False)
                for idx in indices:
                    h_local[idx] += np.random.normal(0, 0.002)
                h_local = project_h(h_local, n_points)
                
                c5_local = evaluate_h(h_local, dx)
                if c5_local < c5_best:
                    h_best = h_local.copy()
                    c5_best = c5_local
                    local_improvements += 1

                        c5_best = c5_local

        indices = np.random.choice(n_points, max(2, n_points // 20), replace=False)

            h_final[idx] += np.random.normal(0, 0.001)

        c5_final = evaluate_h(h_final, dx)
        if c5_final < c5_best:
            h_best = h_final.copy()
            c5_best = c5_final
