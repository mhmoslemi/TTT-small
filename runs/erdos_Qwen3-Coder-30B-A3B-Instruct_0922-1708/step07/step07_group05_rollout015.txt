import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a structured construction
        h_init = np.zeros(n_points)
        # Create a piecewise linear function that is 0 on [0,0.5) and 1 on (1.5,2]
        # and slopes linearly in between
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0

        # Repair to sum to 1
        h_init = repair_h(h_init, n_points)

    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    patience = 200
    last_improvement = 0
    
    for i in range(50000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute current C5 and the corresponding shift
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        
        # Compute gradient for this specific shift only
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = (j - max_k) % n_points
            grad[j] = (1 - h_current[k]) if (k < n_points) else (1 - h_current[k - n_points])
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h_new = h_current + velocity
        h_new = repair_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject with momentum
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            # Accept with probability based on simulated annealing
            if np.random.rand() < np.exp(-(c5_new - c5_current) / (c5_current * 0.01)):
                h_current = h_new
                c5_current = c5_new
        
        # Adapt learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                break
    
    return (h_best.tolist(), c5_best, n_points)
