import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Phase 1: CCCP-like iterations to get close to a local optimum
    learning_rate = 0.01
    max_iter_1 = 5000
    patience = 500
    last_improvement = 0
    
    for i in range(max_iter_1):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Identify the shift where overlap is maximum
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient analytically for the max shift
        # Gradient w.r.t. h[i] is (1 - h[i-k_max]) for k_max shift
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = k_max - j
            if 0 <= k < n_points:
                grad[j] = 1 - h_current[k]
        
        # Update with gradient
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution  
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse moves (simulated annealing)
            if np.random.rand() < 0.01:
                h_current = h_new.copy()
                c5_current = c5_new
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
        
        # Adaptive learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = i
            
        # Occasionally jitter to escape local minima
        if i % 500 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Phase 2: Coordinate descent with exact gradients
    max_iter_2 = 10000
    learning_rate = 0.01
    
    for i in range(max_iter_2):
        if time.time() - start_time > budget_s - 5:
            break
            
        # This maintains cyclic order, but does compute exact gradient at each time
        for j in range(n_points):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Identify shift with max correlation
            corr = correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr)
            
            # Gradient component w.r.t. h[j]
            k = k_max - j
            if 0 <= k < n_points:
                grad_j = 1 - h_current[k]
            else:
                grad_j = 0.0
            
            # Update h[j]
            h_new = h_current.copy()
            h_new[j] -= learning_rate * grad_j
            h_new = project_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                
                # Need to recompute correlation and k_max for subsequent coordinates
                corr = correlate(h_current, 1 - h_current, mode='full')
                k_max = np.argmax(corr)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            
        # Occasionally perturb to escape local optima
        if i % 1000 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    return (h_best.tolist(), c5_best, n_points)
