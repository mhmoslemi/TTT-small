import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, mutation_rate=0.1, delta=0.1):
    """Apply mutation to h"""
    h_new = h.copy()
    mask = np.random.rand(len(h_new)) < mutation_rate
    h_new[mask] += np.random.uniform(-delta, delta, np.sum(mask))
    return repair_h(h_new, len(h_new))

def crossover_h(h1, h2):
    """Perform crossover between two h vectors"""
    n = len(h1)
    cut = np.random.randint(1, n-1)
    h_new = np.concatenate([h1[:cut], h2[cut:]])
    return repair_h(h_new, n)

def selection_tournament(population, fitnesses, k=3):
    """Select individuals using tournament selection"""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), k, replace=False)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Initialize population
    population_size = 100
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = mutate_h(h_init.copy())
        population.append(h)
        
    # Evaluate initial population
    fitnesses = [evaluate_h(h, dx) for h in population]
    h_best = population[np.argmin(fitnesses)]
    c5_best = min(fitnesses)
    
    # Evolution parameters
    n_generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    patience = 10
    last_improvement = 0
    
    # Genetic algorithm loop
    for gen in range(n_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Selection
        parents = selection_tournament(population, fitnesses)
        
        # Create new generation
        new_population = []
        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_rate and i+1 < population_size:
                child1 = crossover_h(parents[i], parents[i+1])
                child2 = crossover_h(parents[i+1], parents[i])
            else:
                child1 = mutate_h(parents[i], mutation_rate)
                child2 = mutate_h(parents[i+1], mutation_rate)
            new_population.extend([child1, child2])
            
        # Trim if necessary
        if len(new_population) > population_size:
            new_population = new_population[:population_size]
            
        # Evaluate new population
        new_fitnesses = [evaluate_h(h, dx) for h in new_population]
        
        # Replace worst individuals
        combined_population = population + new_population
        combined_fitnesses = fitnesses + new_fitnesses
        sorted_indices = np.argsort(combined_fitnesses)
        population = [combined_population[i] for i in sorted_indices[:population_size]]
        fitnesses = [combined_fitnesses[i] for i in sorted_indices[:population_size]]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < c5_best:
            c5_best = fitnesses[current_best_idx]
            h_best = population[current_best_idx].copy()
            last_improvement = gen
            
        # Early stopping
        if gen - last_improvement > patience:
            break
            
    # Gradient refinement on best individual
    h_current = h_best.copy()
    c5_current = c5_best
    eta = 0.001
    eps = 1e-6
    max_iter = 20000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_current.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_current[j] - eta * grad_j
            
        # Apply repair step
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            
    return (h_current.tolist(), c5_current, n_points)
