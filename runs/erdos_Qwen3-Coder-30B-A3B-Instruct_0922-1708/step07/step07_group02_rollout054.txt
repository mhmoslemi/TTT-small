import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # PSO parameters
    swarm_size = 50
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive = 1.5
    social = 1.5
    velocity_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution to swarm
    swarm.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Fill swarm with diverse initializations
    for _ in range(swarm_size - 1):
        # Mix structured and random start
        if np.random.random() < 0.5:
            # Structured: tent pattern
            h = np.zeros(n_points)
            mid = n_points // 2
            for i in range(n_points):
                if i <= mid:
                    h[i] = i / mid * 0.5
                else:
                    h[i] = (n_points - i) / (n_points - mid) * 0.5
        else:
            # Random: mostly zeros with some spikes
            h = np.random.random(n_points) * 0.3
            # Add some clumps
            for _ in range(5):
                start = np.random.randint(0, n_points - 10)
                for i in range(start, min(start + 10, n_points)):
                    h[i] = min(1.0, h[i] + 0.4)
        
        h = project_h(h, n_points, dx)
        swarm.append(h)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
    
    # Track personal best and global best
    personal_best = [h.copy() for h in swarm]
    personal_best_fitness = [evaluate_h(h, dx) for h in swarm]
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO main loop
    max_iter = 100000
    stagnation_count = 0
    stagnation_limit = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        inertia = inertia_start - (inertia_start - inertia_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive_component = cognitive * r1 * (personal_best[i] - swarm[i])
            social_component = social * r2 * (global_best - swarm[i])
            velocities[i] = inertia * velocities[i] + cognitive_component + social_component
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_h(swarm[i], n_points, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = swarm[i].copy()
                    global_best_fitness = fitness
                    stagnation_count = 0
        
        stagnation_count += 1
        if stagnation_count > stagnation_limit:
            # Perturb best solution and restart
            for i in range(swarm_size):
                if i == global_best_idx:
                    continue
                # Add small random perturbation
                perturbation = np.random.normal(0, 0.01, n_points)
                swarm[i] += perturbation
                swarm[i] = project_h(swarm[i], n_points, dx)
                # Re-evaluate
                fitness = evaluate_h(swarm[i], dx)
                if fitness < personal_best_fitness[i]:
                    personal_best[i] = swarm[i].copy()
                    personal_best_fitness[i] = fitness
            stagnation_count = 0
    
    return (global_best.tolist(), global_best_fitness, n_points)
