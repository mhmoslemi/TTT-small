import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a cosine taper centered in the interval
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # CCCP Phase: 5000 iterations maximum
    learning_rate = 0.01
    max_cccp_iter = 5000
    patience = 500
    last_improvement = 0
    
    # Phase 1: CCCP to get a good local minimum
    for i in range(max_cccp_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute the maximum overlap shift k_max
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        
        # Construct gradient w.r.t. h for this shift
        grad = np.zeros(n_points)
        shifted_h = np.roll(h_current, k_max - n_points + 1)  # Shift the h to get the convolution term
        if k_max >= n_points - 1:
            # Positive shift
            shift_len = n_points - (k_max - n_points + 1) - 1
            grad[:shift_len] = 1 - shifted_h[shift_len:]
        else:
            # Negative shift
            shift_len = k_max - n_points + 1
            grad[shift_len:] = 1 - shifted_h[:n_points - shift_len]
            
        # Update h using gradient descent
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept the update
        h_current = h_new
        c5_current = c5_new
        
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = i
            
        # Occasionally apply perturbation to escape local minima
        if i % 500 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Phase 2: Cyclic coordinate descent
    max_cd_iter = 10000
    cd_learning_rate = 0.05
    for i in range(max_cd_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cycle through coordinates
        for j in range(n_points):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient for this coordinate
            corr = correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr)
            
            # Create shifted versions to compute needed terms
            shifted_h = np.roll(h_current, k_max - n_points + 1)
            offset = k_max - n_points + 1  # How the shift is applied
            
            # Gradient component at position j
            grad_j = 0
            if offset <= 0 and j < len(shifted_h) + offset:
                # This case is a bit complex to compute:
                if j >= offset:
                    grad_j = 1 - shifted_h[j - offset]
                else:
                    # No contribution to gradient at j since shift puts 
                    # values outside of valid range
                    grad_j = 0
            elif offset > 0:
                # When offset > 0, it's a positive shift
                if j >= offset and j < len(shifted_h):
                    grad_j = 1 - shifted_h[j - offset]              
                elif j < offset:
                    # shifted_h does not extend to the left, so 1 - h contributes 1 
                    grad_j = 1
                else:
                    grad_j = 0
                    
            # Apply gradient update
            h_new = h_current.copy()
            h_new[j] = h_current[j] - cd_learning_rate * grad_j
            
            # Clip
            h_new[j] = np.clip(h_new[j], 0, 1)
            
            # Renormalize
            total_sum = np.sum(h_new)
            h_new = h_new * (n_points / 2) / total_sum
            
            # Evaluate
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                
                # Update best if improvement
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
                    
        # Adaptive learning rate for CD phase
        if i - last_improvement > patience / 2:
            cd_learning_rate *= 0.98
            last_improvement = i
            
        # Possibly perturb to escape local minima
        if i % 1000 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Final refinement 
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
