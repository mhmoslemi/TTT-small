import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def triangular_pulse(n_points):
    h = np.zeros(n_points)
    midpoint = n_points // 2
    for i in range(n_points):
        if i < midpoint:
            h[i] = i / midpoint
        else:
            h[i] = (n_points - i) / (n_points - midpoint)
    return h
