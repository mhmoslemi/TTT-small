import numpy as np
import time
import math
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize from parent construction or create binary
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a simple binary starting point
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Temperature for simulated annealing
    temp = 0.1
    cooling_rate = 0.995
    min_temp = 1e-8
    
    # Keep track of best solution found
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 2:
            break
            
        # Generate neighbor solution
        h_new = h.copy()
        
        # Choose two random indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Perturb
        delta = np.random.normal(0, 0.05)
        delta = np.clip(delta, -0.1, 0.1)
        
        # Apply perturbation with bounds
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain constraint
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            scale_factor = target_sum / h_sum
            h_new = h_new * scale_factor
        
        # Evaluate new solution
        corr = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr) * dx
        
        # Accept or reject based on simulated annealing
        if c5_new < best_c5:
            # Always accept if better
            h = h_new
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            # Accept worse solutions with probability
            delta_e = c5_new - best_c5
            if delta_e > 0:
                prob = math.exp(-delta_e / temp)
                if np.random.rand() < prob:
                    h = h_new
        
        # Cool down
        temp = max(temp * cooling_rate, min_temp)
        
        # Occasionally print progress
        if iteration % 5000 == 0:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 10:
                break
    
    # Final evaluation with error handling
    if np.sum(best_h) < 1e-10:
        # Fallback if constraints are violated
        h = np.ones(n_points) / 2
        corr = correlate(h, 1 - h, mode='full')
        final_c5 = np.max(corr) * dx
        return (h.tolist(), final_c5, n_points)
    
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Ensure final returned solution meets constraints
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    
    # Rescale if needed
    h_sum = np.sum(final_h)
    if abs(h_sum - n_points/2) > 1e-8:
        final_h = final_h * (n_points / 2) / h_sum
    
    corr_final = correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (final_h.tolist(), final_c5, n_points)
