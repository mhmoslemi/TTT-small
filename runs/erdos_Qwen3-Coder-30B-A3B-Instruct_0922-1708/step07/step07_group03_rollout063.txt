import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    initial_temp = 0.1
    cooling_rate = 0.995
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create binary initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Local search with simulated annealing
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Compute initial C5
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    temp = initial_temp
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate a candidate solution by perturbing two indices
        h_candidate = h.copy()
        
        # Select two random indices
        i, j = np.random.randint(0, n_points, 2)
        
        # Perturb by small amount
        delta = np.random.uniform(-0.05, 0.05)
        
        # Apply perturbation respecting constraints
        h_candidate[i] += delta
        h_candidate[j] -= delta
        
        # Clamp to [0, 1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to maintain sum constraint
        h_sum = np.sum(h_candidate)
        if abs(h_sum) > 1e-10:
            h_candidate = h_candidate * (n_points / 2) / h_sum
            
        # Compute C5 for candidate
        corr = correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(corr) * dx
        
        # Accept or reject based on Metropolis criterion
        if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / temp):
            h = h_candidate
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Decrease temperature
        temp *= cooling_rate
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
