import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # PSO parameters
    swarm_size = 50
    inertia_weight_start = 0.9
    inertia_weight_end = 0.4
    cognitive_coeff = 1.5
    social_coeff = 1.5
    velocity_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Create a structured initial solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Create first particle from initial solution
    p = project_h(h_init.copy(), n_points, dx)
    particles.append(p)
    velocities.append(np.zeros(n_points))
    
    # Fill the rest with different structured patterns
    for i in range(1, swarm_size):
        p = np.zeros(n_points)
        # Use different patterns for variety
        if i % 4 == 0:
            # Sawtooth
            p = np.linspace(0, 1, n_points)
        elif i % 4 == 1:
            # Step
            p[:n_points//2] = 1.0
        elif i % 4 == 2:
            # Sine-like
            x = np.linspace(0, 2, n_points)
            p = 0.5 + 0.5 * np.sin(np.pi * x)
        else:
            # Random
            p = np.random.rand(n_points)
        # Ensure feasibility
        p = project_h(p, n_points, dx)
        particles.append(p)
        velocities.append(np.zeros(n_points))

    # PSO main loop
    pbest = [p.copy() for p in particles]
    pbest_fitness = [evaluate_h(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    iteration = 0
    stagnation_counter = 0
    
    while time.time() - start_time < budget_s - 5 and stagnation_counter < 500:
        inertia_weight = inertia_weight_start - (inertia_weight_start - inertia_weight_end) * (iteration / 1000)
        iteration += 1
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            vel_update = (
                inertia_weight * velocities[i] +
                cognitive_coeff * r1 * (pbest[i] - particles[i]) +
                social_coeff * r2 * (gbest - particles[i])
            )
            # Limit velocity
            vel_update = np.clip(vel_update, -velocity_max, velocity_max)
            velocities[i] = vel_update
            
            # Update position
            new_pos = particles[i] + velocities[i]
            # Project to feasible region
            particles[i] = project_h(new_pos, n_points, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = fitness
                    stagnation_counter = 0
                    
        stagnation_counter += 1
        
        # Check for stagnation and perturb
        if stagnation_counter > 100:
            # Perturb the best particle
            for i in range(swarm_size):
                if i != gbest_idx:
                    perturb_idx = np.random.randint(0, n_points)
                    particles[i][perturb_idx] += np.random.uniform(-0.05, 0.05)
                    particles[i] = project_h(particles[i], n_points, dx)
            
            # Reset stagnation counter
            stagnation_counter = 0
        
        # Occasionally re-evaluate best solutions to ensure no better ones were missed
        if iteration % 100 == 0:
            for i in range(swarm_size):
                fitness = evaluate_h(particles[i], dx)
                if fitness < pbest_fitness[i]:
                    pbest[i] = particles[i].copy()
                    pbest_fitness[i] = fitness
                    if fitness < gbest_fitness:
                        gbest = particles[i].copy()
                        gbest_fitness = fitness
    
    return (gbest.tolist(), gbest_fitness, n_points)
