import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    learning_rate = 1e-3
    max_iter = 10000
    best_c5 = float('inf')
    best_h = h.copy()
    barrier_param = 1e3
    barrier_decay = 0.99
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate current C5
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient using adjoint method
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h) - 1)
        
        # Compute gradient directly from the correlation structure
        grad = np.zeros_like(h)
        if max_lag >= 0:
            # 1-h(x+max_lag) contributes to h(x)
            for i in range(len(h) - max_lag):
                grad[i] += (1 - h[i + max_lag])
        else:
            # 1-h(x+max_lag) contributes to h(x-max_lag)
            for i in range(-max_lag, len(h)):
                grad[i] += (1 - h[i + max_lag])
        
        # Update with gradient
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Rescale to satisfy integral constraint
        h_sum = np.sum(h) * dx
        if abs(h_sum - 1.0) > 1e-8:
            h = h * (1.0 / h_sum)
            
        # Apply barrier method
        barrier_param *= barrier_decay
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
