import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2, n_points):
    # Segment-wise blending crossover
    segment_length = max(1, n_points // 4)
    start = np.random.randint(0, n_points - segment_length)
    offspring = parent1.copy()
    offspring[start:start+segment_length] = parent2[start:start+segment_length]
    return offspring

def mutate(individual, mutation_scale, n_points):
    # Flip segments with probability 0.3, add Gaussian noise with probability 0.7
    mutated = individual.copy()
    if np.random.rand() < 0.3:
        # Segment flip
        segment_length = max(1, n_points // 4)
        start = np.random.randint(0, n_points - segment_length)
        new_value = np.random.rand()
        mutated[start:start+segment_length] = new_value
    else:
        # Gaussian noise
        noise = np.random.normal(0, mutation_scale, n_points)
        mutated += noise
    return mutated

def local_refinement(h, n_points, dx, max_iter=50):
    # Simple gradient-based local search
    h_refined = h.copy()
    eta = 0.01
    eps = 1e-6
    for _ in range(max_iter):
        h_new = h_refined.copy()
        improved = False
        indices = list(range(n_points))
        np.random.shuffle(indices)
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < evaluate_h(h_refined, dx):
            h_refined = h_new
            improved = True
        if not improved:
            eta *= 0.995
        else:
            eta = min(0.01, eta * 1.001)
    return h_refined

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.75
        h_init[mid:] = 0.25
        h_init = project_h(h_init, n_points, dx)
    
    # Memetic algorithm parameters
    population_size = 50
    num_parents = max(1, population_size // 5)
    max_generations = 1500
    mutation_scale = 0.1
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = project_h(h_rand, n_points, dx)
        population.append(h_rand)
    
    # Best solution tracking
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Eval and track improvements
    for i, ind in enumerate(population):
        fitness = evaluate_h(ind, dx)
        if fitness < c5_best:
            c5_best = fitness
            h_best = ind.copy()
    
    # Main memetic loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all candidates
        population_fitness = [evaluate_h(ind, dx) for ind in population]
        
        # Select top individuals
        sorted_indices = np.argsort(population_fitness)
        parents = [population[i] for i in sorted_indices[:num_parents]]
        parents_fitness = [population_fitness[i] for i in sorted_indices[:num_parents]]

        offspring = []

            child = crossover(parent1, parent2, n_points)
            child = mutate(child, mutation_scale, n_points)
            child = project_h(child, n_points, dx)
            # Local refinement
            child_refined = local_refinement(child, n_points, dx)
            offspring.append(child_refined)
            
        # Add parent population to offspring
        offspring.extend(parents)
        
        # Replace population
        population = offspring
    
        # Update best solution
        for ind in population:
            fitness = evaluate_h(ind, dx)
            if fitness < c5_best:
                c5_best = fitness

        # Prevent stagnation
        if gen % 100 == 0 and gen > 0:
            if gen % 300 == 0:
                # Random injection
                for i in range(population_size // 10):
                    h_inject = np.random.rand(n_points)
                    h_inject = project_h(h_inject, n_points, dx)
                    population[i] = h_inject
        
    return (h_best.tolist(), c5_best, n_points)
