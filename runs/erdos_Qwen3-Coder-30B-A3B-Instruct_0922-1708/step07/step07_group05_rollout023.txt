import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_basis(n_points, k_min=1, k_max=5):
    # Generate diverse Fourier-like candidates
    candidates = []
    for _ in range(50):
        # Sample random coefficients
        A = np.random.uniform(-0.5, 0.5)
        B = np.random.uniform(0.2, 0.8)
        # Create a simple combination of cos and sin
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            h[i] = A * np.cos(2 * np.pi * k_min * x) + B  # simplified spectrum
        # Ensure bounds and normalization
        h = repair_h(h, n_points)
        candidates.append(h)
    return candidates
