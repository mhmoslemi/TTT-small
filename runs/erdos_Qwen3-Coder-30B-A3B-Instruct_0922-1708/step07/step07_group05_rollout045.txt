import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidates(n_points, num_candidates=50):
    candidates = []
    for _ in range(num_candidates):
        # Generate random Fourier-like basis
        h = np.zeros(n_points)
        # Use 3-5 basis functions
        num_bases = np.random.randint(3, 6)
        for _ in range(num_bases):
            k = np.random.randint(1, n_points // 4)
            a = np.random.uniform(-1, 1)
            b = np.random.uniform(-1, 1)
            phase = np.random.uniform(0, 2 * np.pi)
            h += a * np.cos(k * np.linspace(0, 2 * np.pi, n_points) + phase) + \
                 b * np.sin(k * np.linspace(0, 2 * np.pi, n_points) + phase)
        
        # Normalize and clip
        h = np.clip(h, 0, 1)
        h = repair_h(h, n_points)
        candidates.append(h)
    return candidates

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        h_init = np.zeros(n_points)
        # Create a piecewise linear function that is 0 on [0,0.5) and 1 on (1.5,2]
        # and slopes linearly in between
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        # Repair to sum to 1
        h_init = repair_h(h_init, n_points)

    # Phase 1: Generate diverse candidates
    candidates = generate_fourier_candidates(n_points, 50)
    candidates.append(h_init)  # Include the initial solution
    
    # Evaluate candidates
    candidate_scores = []
    for candidate in candidates:
        score = evaluate_h(candidate, dx)
        candidate_scores.append((score, candidate))
    
    # Get top candidates
    candidate_scores.sort(key=lambda x: x[0])
    top_candidates = [h for _, h in candidate_scores[:10]]
    
    # Phase 2: Refinement with gradient descent
    h_best = top_candidates[0].copy()
    c5_best = evaluate_h(h_best, dx)
    
    patience_limit = 200
    max_refinement_iter = 2000
    learning_rate_init = 0.01
    
    for h_candidate in top_candidates:
        h_current = h_candidate.copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            
        patience = 0
        learning_rate = learning_rate_init
        
        for iter_refine in range(max_refinement_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient via finite differences
            grad = np.zeros(n_points)
            epsilon = 1e-6
            c5_base = evaluate_h(h_current, dx)
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[i] += epsilon
                h_minus[i] -= epsilon
                h_plus = repair_h(h_plus, n_points)
                h_minus = repair_h(h_minus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                c5_minus = evaluate_h(h_minus, dx)
                grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update with gradient step
            h_new = h_current - learning_rate * grad
            h_new = repair_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept or reject
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    patience = 0
                else:
                    patience += 1
            else:
                patience += 1
            
            # Adaptive learning rate
            if patience > 100:
                learning_rate *= 0.95
                patience = 0
                
            if patience > patience_limit:
                break
                
    return (h_best.tolist(), c5_best, n_points)
