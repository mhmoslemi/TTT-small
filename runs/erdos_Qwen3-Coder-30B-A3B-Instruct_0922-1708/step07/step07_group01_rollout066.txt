import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Phase 1: CCCP-like approach with convex approximations
    max_iter1 = 5000
    lr1 = 0.01
    patience1 = 500
    last_improvement1 = 0
    
    for i in range(max_iter1):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find k_max where max overlap occurs
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient
        h_shifted = np.roll(1 - h_current, k_max)
        grad = h_shifted
        
        # Update step
        h_new = h_current - lr1 * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if improvement
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse moves
            if np.random.rand() < 0.01:
                h_current = h_new.copy()
                c5_current = c5_new
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement1 = i
            
        # Adjust learning rate
        if i - last_improvement1 > patience1:
            lr1 *= 0.95
            last_improvement1 = i
            
        # Perturb occasionally
        if i % 500 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Phase 2: Coordinate descent with exact gradients
    max_iter2 = 10000
    lr2 = 0.01
    patience2 = 1000
    last_improvement2 = 0
    perturbation_sigma = 0.005
    
    for i in range(max_iter2):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic coordinate descent
        indices = np.random.permutation(n_points)
        for idx in indices:
            # Compute current k_max and gradient component
            corr = correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr)
            h_shifted = np.roll(1 - h_current, k_max)
            grad_component = h_shifted[idx]
            
            # Update
            h_current[idx] = h_current[idx] - lr2 * grad_component
            h_current = project_h(h_current, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_current, dx)
        
        # Update best
        if c5_new < c5_best:
            h_best = h_current.copy()
            c5_best = c5_new
            last_improvement2 = i
            
        # Adjust learning rate
        if i - last_improvement2 > patience2:
            lr2 *= 0.95
            last_improvement2 = i
            
        # Perturb occasionally
        if i % 1000 == 0 and i > 0:
            noise = np.random.normal(0, perturbation_sigma, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
            perturbation_sigma *= 1.1  # Increase perturbation
    
    return (h_best.tolist(), c5_best, n_points)
