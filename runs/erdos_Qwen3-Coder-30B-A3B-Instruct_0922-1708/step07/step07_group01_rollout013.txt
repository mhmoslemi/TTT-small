import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Create a sophisticated initial construction
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Use a combination of waveforms to create diverse initial point
        h_init = 0.5 * (1 + np.cos(np.pi * x)) + 0.3 * np.sin(2 * np.pi * x)
        h_init = project_h(h_init, n_points)
    
    # Multiple random initializations to avoid local minima
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Optimization parameters
    max_iter = 10000
    patience = 200
    learning_rate = 0.02
    decay_rate = 0.98
    
    # Generate diverse starting points
    initializations = []
    initializations.append(h_init)
    
    # Add random perturbations
    for i in range(10):
        h = h_init + np.random.normal(0, 0.01, n_points)
        h = project_h(h, n_points)
        initializations.append(h)
        
    # Add structured waveforms
    x = np.linspace(0, 2, n_points)
    for i in range(5):
        h = np.zeros(n_points)
        # Square wave train
        period = 0.4
        h = np.floor((x / period) % 2) * 0.8 + 0.2 * (1 - np.floor((x / period) % 2))
        h = project_h(h, n_points)
        initializations.append(h)

    for i, h_start in enumerate(initializations):
        if time.time() - start_time > budget_s - 5:
            break
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current, dx)
        
        last_improvement = 0
        lr = learning_rate
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute correlation for all shifts and find the maximizer
            corr_all = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr_all)
            max_shift = max_shift_idx - (n_points - 1)
            
            # Compute gradient w.r.t. the maximizing shift only
            h_shifted = np.roll(h_current, -max_shift)
            grad = 1 - h_shifted
            
            # Update with gradient: this assumes the maximum overlaps
            h_new = h_current - lr * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Adaptive learning rate
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                last_improvement = iteration
            else:
                if iteration - last_improvement > patience:
                    # Reduce learning rate if no progress
                    lr *= decay_rate
                    last_improvement = iteration
                # Occasionally accept worse moves for diversity
                elif np.random.rand() < 0.01:
                    h_current = h_new.copy()
                    c5_current = c5_new
            
            # Update best solution
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = iteration
            
            # Random perturbation every few iterations
            if iteration % 300 == 0 and iteration > 0:
                noise = np.random.normal(0, 0.005, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
    
    # Final refinement using local search
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    # Final check
    c5_best = evaluate_h(h_best, dx)
    return (h_best.tolist(), c5_best, n_points)
