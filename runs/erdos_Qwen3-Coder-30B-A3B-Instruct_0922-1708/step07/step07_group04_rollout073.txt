import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 128  # Use a bit more points for better resolution
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Alternative: use a sine-based construction
    # This may escape local optima better
    if n_points >= 128:
        # Generate a cosine-based refiner
        x = np.linspace(0, 2, n_points)
        # Mix of cosine terms to create a smoother shape
        h_cos = 0.5 + 0.3 * np.cos(np.pi * x) + 0.1 * np.cos(2 * np.pi * x)
        h_cos = repair_h(h_cos, n_points)
        
        # Run a more refined optimization from both starting points
        c5_cos = evaluate_h(h_cos, dx)
        c5_init = evaluate_h(h_init, dx)
        
        h_start = h_cos if c5_cos < c5_init else h_init
    else:
        h_start = h_init
    
    h_current = h_start.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Enhanced gradient descent
    def grad_descent_step(h, dx):
        h_new = h.copy()
        eps = 1e-6
        eta = 0.01
        for j in range(n_points):
            # Forward difference
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h[j] - eta * grad_j
            
        return repair_h(h_new, n_points)
    
    # Simulated annealing inspired variation
    max_iter = 20000
    patience = 200
    last_improvement = 0
    temp = 1.0
    decay = 0.999
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = grad_descent_step(h_current, dx)
        c5_new = evaluate_h(h_new, dx)

        if c5_new < c5_current or np.random.rand() < np.exp((c5_current - c5_new) / temp):

            c5_current = c5_new
            
            # Update best
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
        temp *= decay
        
        if i - last_improvement > patience:
            # Perturb to escape local minima
            h_perturb = h_current.copy()
            for j in range(n_points):
                if np.random.rand() < 0.05:
                    h_perturb[j] += (np.random.rand() - 0.5) * 0.02
            h_perturb = repair_h(h_perturb, n_points)
            c5_perturb = evaluate_h(h_perturb, dx)
            if c5_perturb < c5_current:
                h_current = h_perturb
                c5_current = c5_perturb
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current

            best_c5 = c5_best
            # Try several nearby values
            for delta in [-0.02, -0.01, 0, 0.01, 0.02]:
                test_h = h_best.copy()
                test_h[j] += delta
                test_h = repair_h(test_h, n_points)
                test_c5 = evaluate_h(test_h, dx)
                if test_c5 < best_c5:
                    best_c5 = test_c5
                    best_val = test_h[j]
            h_new[j] = best_val
        h_new = repair_h(h_new, n_points)
        new_c5 = evaluate_h(h_new, dx)
        if new_c5 < c5_best:
            h_best = h_new
            c5_best = new_c5
    
    return (h_best.tolist(), c5_best, n_points)
