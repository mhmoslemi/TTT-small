import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize swarm
    swarm_size = 50
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        # Create initial solutions with different patterns
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    
    # Fill up the swarm with diverse starting positions
    for i in range(1, swarm_size):
        # Create a sawtooth pattern
        h = np.zeros(n_points)
        h[::2] = 0.7
        h[1::2] = 0.3
        h = project_h(h, n_points, dx)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 1.5, 1.5
    v_max = 0.1
    
    # Personal bests
    p_best = [p.copy() for p in particles]
    p_best_fitness = [evaluate_h(p, dx) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(p_best_fitness)
    g_best = p_best[global_best_idx].copy()
    g_best_fitness = p_best_fitness[global_best_idx]
    
    # PSO iteration
    max_iter = 100000
    stagnation_counter = 0
    stagnation_threshold = 1000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * (i / max_iter)
        
        for j in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[j] = (w * velocities[j] + 
                           c1 * r1 * (p_best[j] - particles[j]) + 
                           c2 * r2 * (g_best - particles[j]))
            
            # Apply velocity limits
            velocities[j] = np.clip(velocities[j], -v_max, v_max)
            
            # Update position
            particles[j] += velocities[j]
            
            # Project to feasible region
            particles[j] = project_h(particles[j], n_points, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(particles[j], dx)
            
            # Update personal best
            if fitness < p_best_fitness[j]:
                p_best[j] = particles[j].copy()
                p_best_fitness[j] = fitness
                
                # Update global best
                if fitness < g_best_fitness:
                    g_best = particles[j].copy()
                    g_best_fitness = fitness
                    stagnation_counter = 0
                else:
                    stagnation_counter += 1
            else:
                stagnation_counter += 1
        
        # Perturbation if stagnating
        if stagnation_counter > stagnation_threshold:
            for j in range(swarm_size):
                if np.random.rand() < 0.1:  # 10% chance
                    perturb = np.random.normal(0, 0.05, n_points)
                    particles[j] += perturb
                    particles[j] = project_h(particles[j], n_points, dx)
                    fitness = evaluate_h(particles[j], dx)
                    if fitness < p_best_fitness[j]:
                        p_best[j] = particles[j].copy()
                        p_best_fitness[j] = fitness
                        
                        if fitness < g_best_fitness:
                            g_best = particles[j].copy()
                            g_best_fitness = fitness
            stagnation_counter = 0
    
    # Return best found
    return (g_best.tolist(), g_best_fitness, n_points)
