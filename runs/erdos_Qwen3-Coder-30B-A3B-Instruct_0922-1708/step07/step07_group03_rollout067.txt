import numpy as np
import time
import random
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize particles
    S = 50  # swarm size
    particles = []
    velocities = []
    
    # Create initial swarm
    for i in range(S):
        if 'initial_h_values' in globals() and i == 0:
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h)
        else:
            h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # SA parameters
    T = 1.0
    cooling_rate = 0.999
    
    # Global best
    g_best = None
    g_best_c5 = float('inf')
    
    # Personal bests
    p_best = [p.copy() for p in particles]
    p_best_c5 = [float('inf')] * S
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate all particles
        for i in range(S):
            h = particles[i]
            # Compute correlation
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < p_best_c5[i]:
                p_best_c5[i] = c5
                p_best[i] = h.copy()
                
            # Update global best
            if c5 < g_best_c5:
                g_best_c5 = c5
                g_best = h.copy()
        
        # Update velocities and positions
        for i in range(S):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            v = (w * v + 
                 c1 * r1 * (p_best[i] - h) + 
                 c2 * r2 * (g_best - h))
            
            # Simulated Annealing perturbation
            if np.random.rand() < T:
                v += np.random.uniform(-0.1, 0.1, n_points)
            
            # Update position
            h = h + v
            
            # Repair constraints
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-8:
                h = h * (n_points / 2) / h_sum
                
            particles[i] = h
            velocities[i] = v
            
        # Cool down
        T = max(T * cooling_rate, 0.01)
    
    # Final evaluation of best solution
    if g_best is not None:
        corr = correlate(g_best, 1 - g_best, mode='full')
        final_c5 = np.max(corr) * dx
        return (g_best.tolist(), final_c5, n_points)
    else:
        # Fallback to initial if available
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h)
            h = h * (n_points / 2) / np.sum(h)
            corr = correlate(h, 1 - h, mode='full')
            final_c5 = np.max(corr) * dx
            return (h.tolist(), final_c5, n_points)
        else:
            # Random fallback
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            corr = correlate(h, 1 - h, mode='full')
            final_c5 = np.max(corr) * dx
            return (h.tolist(), final_c5, n_points)
