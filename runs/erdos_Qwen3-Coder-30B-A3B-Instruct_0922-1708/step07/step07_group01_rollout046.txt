import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Binary search for optimal C
    c_low = 0.0
    c_high = 0.5
    c_best = 1.0
    h_best = h_init.copy()
    
    max_iterations = 50
    tolerance = 1e-5
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Formulate feasibility LP
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.constraints.NonNegativity(h_vars),
            cp.constraints.LowerBounded(h_vars, 0),
            cp.constraints.UpperBounded(h_vars, 1),
        ]
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Overlap constraints
        for k in range(n_points):
            if k == 0:
                continue
            start_idx = 0
            end_idx = n_points - k
            overlap_expr = cp.sum(cp.multiply(h_vars[start_idx:end_idx], 
                                               1 - h_vars[start_idx+k:end_idx+k]))
            constraints.append(overlap_expr <= c_mid * dx)
            
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_sol = h_vars.value
                h_sol = project_h(h_sol, n_points)
                c_val = evaluate_h(h_sol, dx)
                if c_val <= c_mid:
                    c_best = c_mid
                    h_best = h_sol.copy()
                    c_high = c_mid
                else:
                    c_low = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
    
    # Final refinement
    h_final = h_best.copy()
    c_final = c_best
    
    # Occasionally improve through noise injection
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        noise = np.random.normal(0, 0.001, n_points)
        h_candidate = project_h(h_final + noise, n_points)
        c_candidate = evaluate_h(h_candidate, dx)
        if c_candidate < c_final:
            h_final = h_candidate
            c_final = c_candidate
    
    return (h_final.tolist(), c_final, n_points)
