import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    backtracking_factor = 0.5
    initial_step_size = 0.5
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def solve_linear_subproblem(gradient):
        # Solve: argmin_s <gradient, s> subject to 0 <= s <= 1, sum(s) = n_points/2
        # This is a simple optimization: set s_i = 1 if gradient_i is minimum, 0 otherwise
        # but also adjust to meet the sum constraint
        sorted_idx = np.argsort(gradient)
        s = np.zeros_like(gradient)
        
        # Initialize with 0 or 1 based on sign of gradient with some tolerance
        target_sum = n_points / 2
        
        # Assign to current best indices first (most negative gradient)
        current_sum = 0
        i = 0
        while i < n_points and current_sum < target_sum:
            if gradient[sorted_idx[i]] < 0:
                s[sorted_idx[i]] = 1
                current_sum += 1
            i += 1
            
        # Adjust for exact sum constraint
        if current_sum > target_sum:
            # Adjust some 1s to 0s
            indices_1 = np.where(s == 1)[0]  # positions where s=1
            diff = current_sum - target_sum
            for idx in indices_1[:diff]:
                s[idx] = 0
        elif current_sum < target_sum:
            # Adjust some 0s to 1s
            indices_0 = np.where(s == 0)[0]  # positions where s=0
            diff = target_sum - current_sum
            for idx in indices_0[:diff]:
                s[idx] = 1
                
        # Rescale to exact sum
        if np.sum(s) > 0:
            s = s * target_sum / np.sum(s)
        else:
            # Fallback to uniform distribution
            s = np.full(n_points, target_sum / n_points)
            
        return s
    
    def line_search(h, s, c5, gradient):
        # Try to find step size that improves objective
        gamma = initial_step_size
        
        max_iterations = 20
        for _ in range(max_iterations):
            h_new = (1 - gamma) * h + gamma * s
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = compute_c5(h_new)
            
            if new_c5 < c5:
                return gamma, h_new
            else:
                gamma *= backtracking_factor
                
        return gamma, h
               
    # Main optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Compute gradient using finite difference
        eps = 1e-6
        gradient = np.zeros_like(h) 

        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1.0
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            gradient[i] = (c5_plus - c5) / eps

        # Solve linear subproblem
        s = solve_linear_subproblem(gradient)
        
        # Line search
        gamma, h_new = line_search(h, s, c5, gradient)
        
        # Update
        h = h_new
    
    # Final check
    c5_final = compute_c5(best_h)
    
    return (best_h.tolist(), c5_final, n_points)
