import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # PSO parameters
    swarm_size = 50
    max_iter = 100000
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive = 1.5
    social = 1.5
    velocity_limit = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Create a basic structured initial solution
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    # Initialize first particle with good solution
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Initialize remaining particles with different patterns
    for _ in range(swarm_size - 1):
        h_rand = np.random.random(n_points)
        h_rand = project_h(h_rand, n_points, dx)
        particles.append(h_rand)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Track personal bests
    personal_bests = [evaluate_h(p, dx) for p in particles]
    personal_best_positions = [p.copy() for p in particles]
    
    # Global best
    global_best_idx = np.argmin(personal_bests)
    global_best = personal_bests[global_best_idx]
    global_best_position = personal_best_positions[global_best_idx].copy()
    
    # Optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight linearly
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_num / max_iter)
        
        # Update each particle
        for i in range(swarm_size):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity using PSO formulas
            r1, r2 = np.random.random(2)
            cognitive_component = cognitive * r1 * (personal_best_positions[i] - particle)
            social_component = social * r2 * (global_best_position - particle)
            velocity = inertia * velocity + cognitive_component + social_component
            
            # Clip velocity
            velocity = np.clip(velocity, -velocity_limit, velocity_limit)
            velocities[i] = velocity
            
            # Update position
            new_position = particle + velocity
            
            # Project new position to feasible region
            new_position = project_h(new_position, n_points, dx)
            particles[i] = new_position
            
            # Evaluate fitness
            fitness = evaluate_h(new_position, dx)
            
            # Update personal best
            if fitness < personal_bests[i]:
                personal_bests[i] = fitness
                personal_best_positions[i] = new_position.copy()
                
                # Update global best if needed
                if fitness < global_best:
                    global_best = fitness
                    global_best_position = new_position.copy()
    
    return (global_best_position.tolist(), global_best, n_points)
