import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Multiple initializations
    initializations = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
        initializations.append(h_init.copy())
    
    # Add structured initializations
    x = np.linspace(0, 2, n_points)
    
    # Sine-based initialization
    h_sine = 0.5 * (1 + np.sin(np.pi * x - np.pi/2))
    h_sine = project_h(h_sine, n_points)
    initializations.append(h_sine.copy())
    
    # Trapezoidal
    h_trap = np.zeros(n_points)
    mid = n_points // 2
    h_trap[:mid] = np.linspace(0, 1, mid)
    h_trap[mid:] = np.linspace(1, 0, n_points - mid)
    h_trap = project_h(h_trap, n_points)
    initializations.append(h_trap.copy())
    
    # Random with constraints
    h_rand = np.random.rand(n_points)
    h_rand = project_h(h_rand, n_points)
    initializations.append(h_rand.copy())
    
    # Bimodal
    h_bimodal = np.zeros(n_points)
    h_bimodal[:n_points//4] = 1
    h_bimodal[3*n_points//4:] = 1
    h_bimodal = project_h(h_bimodal, n_points)
    initializations.append(h_bimodal.copy())
    
    # Better initialization to keep multiple starting points
    h_best = None
    c5_best = float('inf')
    
    # Run optimizations from each initialization
    for h_init in initializations:
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
        
        # Gradient descent with multiple restarts
        for restart in range(5):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_current = h_init.copy()
            c5_current = evaluate_h(h_current, dx) 
            lr = 0.05
            patience = 0
            last_improvement = 0
            
            for iteration in range(2000):
                if time.time() - start_time > budget_s - 5:
                    break
                    
                # Find shift that maximizes the correlation
                corr = correlate(h_current, 1 - h_current, mode='full')
                k_max = np.argmax(corr)
                max_corr = corr[k_max]
                
                # Compute gradient via finite differences on that shift only
                grad = np.zeros(n_points)
                epsilon = 1e-5
                
                for j in range(n_points):
                    h_plus = h_current.copy()
                    h_minus = h_current.copy()
                    
                    h_plus[j] += epsilon
                    h_minus[j] -= epsilon
                    
                    h_plus = project_h(h_plus, n_points)
                    h_minus = project_h(h_minus, n_points)
                    
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                    
                    grad[j] = (corr_plus[k_max] - corr_minus[k_max]) / (2 * epsilon)
                
                # Update
                h_new = h_current - lr * grad
                
                # Project and evaluate
                h_new = project_h(h_new, n_points)
                c5_new = evaluate_h(h_new, dx)
                
                # Accept or reject
                if c5_new < c5_current:
                    h_current = h_new
                    c5_current = c5_new
                    patience = 0
                else:
                    patience += 1
                
                # Update best
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = iteration
                
                # Learning rate scheduling
                if patience > 50:
                    lr *= 0.8
                    patience = 0
                    
                # Perturb occasionally
                if iteration % 200 == 0 and iteration > 0:
                    noise = np.random.randn(n_points) * 0.01
                    h_current = project_h(h_current + noise, n_points)
                    c5_current = evaluate_h(h_current, dx)
            
            # Final improvement step
            for _ in range(100):
                if time.time() - start_time > budget_s - 5:
                    break
                    
                h_new = h_current.copy()
                indices = np.random.choice(n_points, max(1, n_points // 20), replace=False)
                for idx in indices:
                    h_new[idx] += np.random.normal(0, 0.005)
                h_new = project_h(h_new, n_points)
                c5_new = evaluate_h(h_new, dx)
                
                if c5_new < c5_current:
                    h_current = h_new
                    c5_current = c5_new
                    
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
    
    # Final improvement with simulated annealing
    if h_best is not None:
        h_current = h_best.copy()
        c5_current = c5_best
        
        for iter in range(500):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Local search
            h_new = h_current.copy()
            idx = np.random.randint(n_points)
            h_new[idx] += np.random.normal(0, 0.01)
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            # Metropolis criterion
            if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / (0.001 + np.abs(c5_current))):
                h_current = h_new
                c5_current = c5_new
                
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
    
    return (h_best.tolist(), c5_best, n_points)
