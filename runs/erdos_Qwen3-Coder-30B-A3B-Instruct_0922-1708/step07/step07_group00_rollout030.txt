import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of the maximum overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Identifying the k that gives maximum overlap
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n - 1)
    
    # Computing gradient for this k (finite difference)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    # Normalize the gradient
    grad = grad / (np.linalg.norm(grad) + 1e-12)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    S = 100  # swarm size
    max_iter = 10000
    w_start = 0.9
    w_end = 0.1
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    particles = [h_init.copy() for _ in range(S)]
    velocities = [np.random.uniform(-v_max, v_max, n_points) for _ in range(S)]
    personal_best_positions = [p.copy() for p in particles]
    personal_best_scores = [evaluate_h(p, dx) for p in particles]
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia
        w = w_start - (w_start - w_end) * (iteration / max_iter)
        
        for i in range(S):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best_positions[i] - particles[i]) +
                c2 * r2 * (global_best_position - particles[i])
            )
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair position
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = particles[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = particles[i].copy()
        
        # Local search refinement (every 50 iterations)
        if iteration % 50 == 0 and iteration > 0:
            refined = global_best_position.copy()
            for _ in range(100):
                if time.time() - start_time > budget_s - 5:
                    break
                grad = compute_gradient(refined, dx, n_points)
                refined = refined - 1e-5 * grad
                refined = repair_h(refined, n_points)
                score = evaluate_h(refined, dx)
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = refined.copy()
    
    return (global_best_position.tolist(), global_best_score, n_points)
