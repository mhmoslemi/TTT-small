import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def find_max_shift(h, dx):
    """Find the shift k that maximizes overlap"""
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    return max_idx - (len(h) - 1)

def compute_gradient(h, k_max, dx):
    """Compute exact gradient for the max overlap shift k_max"""
    n = len(h)
    grad = np.zeros(n)
    if k_max >= 0:
        # h[i] contributes to (1-h[i+k_max]) when i+k_max < n
        for i in range(n):
            if i + k_max < n:
                grad[i] = -1
    else:
        # h[i] contributes to (1-h[i+k_max]) when i+k_max >= 0
        for i in range(n):
            if i + k_max >= 0:
                grad[i] = -1
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # CCCP Phase: 5000 iterations
    learning_rate = 0.01
    patience = 500
    last_improvement = 0
    
    for i in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find current max overlap shift
        k_max = find_max_shift(h_current, dx)
        
        # Compute gradient for this shift
        grad = compute_gradient(h_current, k_max, dx)
        
        # Update with gradient
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept move if improvement
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse moves (annealing)
            if np.random.rand() < 0.01:
                h_current = h_new.copy()
                c5_current = c5_new
                
        # Apply perturbation every 500 iterations
        if i % 500 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
        
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = i
    
    # Coordinate Descent Phase: 10000 iterations
    learning_rate = 0.01
    patience = 1000
    last_improvement = 0
    perturb_strength = 0.005
    
    for i in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update each parameter cyclically
        for j in range(n_points):
            # Compute gradient for current position
            k_max = find_max_shift(h_current, dx)
            # Compute gradient component for h[j]
            grad_j = 0
            if k_max >= 0:
                if j + k_max < n_points:
                    grad_j = -1
            else:
                if j + k_max >= 0:
                    grad_j = -1
            
            # Update h[j]
            h_current[j] = h_current[j] - learning_rate * grad_j * dx
            # Clip to bound
            h_current[j] = np.clip(h_current[j], 0, 1)
            # Renormalize to maintain integral constraint
            h_current = project_h(h_current, n_points)
            
        # Evaluate current solution
        c5_current = evaluate_h(h_current, dx)
        
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Apply perturbation every 1000 iterations
        if i % 1000 == 0 and i > 0:
            noise = np.random.normal(0, perturb_strength, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
            perturb_strength *= 1.1  # Increase perturbation
        
        # Adaptive learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = i
    
    # Final optimization through local search
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
