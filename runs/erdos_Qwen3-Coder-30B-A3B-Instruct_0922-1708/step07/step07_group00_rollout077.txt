import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.stats import qmc

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Generate low-discrepancy sequences using Sobol
    sampler = qmc.Sobol(d=n_points, seed=seed)
    n_initials = 500
    initial_points = sampler.random(n=n_initials)
    
    best_h = None
    best_c5 = float('inf')
    
    # Try each initialization
    for i in range(n_initials):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_init = initial_points[i, :]
        h_init = repair_h(h_init, n_points)
        
        # Nelder-Mead optimization
        result_nm = minimize(
            lambda x: evaluate_h(repair_h(x, n_points), dx),
            h_init,
            method='Nelder-Mead',
            options={
                'maxiter': 1000,
                'xatol': 1e-6,
                'fatol': 1e-6
            }
        )
        
        h_opt = result_nm.x
        
        # Refinement with gradient descent
        h_refined = repair_h(h_opt, n_points)
        max_refinements = 1000
        learning_rate = 1e-5
        for _ in range(max_refinements):
            if time.time() - start_time > budget_s - 5:
                break
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            max_k_idx = np.argmax(corr)
            max_k = max_k_idx - (n_points - 1)
            
            # Approximate gradient via finite differences
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                grad[j] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
            
            # Update
            h_refined = h_refined - learning_rate * grad
            h_refined = repair_h(h_refined, n_points)
            current_c5 = evaluate_h(h_refined, dx)
            
            if current_c5 < best_c5:
                best_h = h_refined.copy()
                best_c5 = current_c5
                
            # Decay learning rate
            learning_rate *= 0.999
            
            if learning_rate < 1e-8:
                break
    
    # Final check
    if best_h is None:
        if 'initial_h_values' in globals():
            h_start = np.array(initial_h_values)
            if len(h_start) != n_points:
                x_orig = np.linspace(0, 2, len(h_start))
                x_target = np.linspace(0, 2, n_points)
                h_start = np.interp(x_target, x_orig, h_start)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
        else:
            h_start = np.random.rand(n_points)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
        best_h = h_start
        best_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
