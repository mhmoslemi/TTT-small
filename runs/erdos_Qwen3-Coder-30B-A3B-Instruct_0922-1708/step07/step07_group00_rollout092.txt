import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Parameters for Evolution Strategy
    pop_size = 200
    elites = 40  # 20% of pop_size
    offsprings = 5
    mutation_std = 0.1
    patience = 30
    min_improvement = 1e-6

    # Initialize population
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = h_init.copy()
    no_improvement_count = 0
    
    # ES loop
    generation = 0
    while time.time() - start_time < budget_s - 5 and no_improvement_count < patience:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Select elites
        elites_list = population[:elites]
        
        # Generate new population
        new_population = elites_list.copy()
        
        for elite in elites_list:
            for _ in range(offsprings):
                # Mutation
                mutant = elite + np.random.randn(n_points) * mutation_std
                mutant = repair_h(mutant, n_points)
                new_population.append(mutant)
        
        # Keep best pop_size individuals
        population = new_population[:pop_size]
        generation += 1
    
    # Final refinement with local search
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Pick the best from our local search
        test_h = best_h.copy() + np.random.randn(n_points) * 0.01
        test_h = repair_h(test_h, n_points)
        test_c5 = evaluate_h(test_h, dx)
        if test_c5 < best_c5:
            best_h = test_h.copy()
            best_c5 = test_c5
    
    return (best_h.tolist(), best_c5, n_points)
