import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size):
    population = []
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    else:
        # Create a simple triangular pulse for diversity
        h_init = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            if i < mid:
                h_init[i] = 2 * i / (mid - 1) if mid > 1 else 0
            else:
                h_init[i] = 2 * (n_points - i) / (n_points - mid) if n_points > mid else 0
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Fill the rest with random step functions
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    return population

def selection(population, fitnesses, keep_top=0.5):
    sorted_indices = np.argsort(fitnesses)
    top_indices = sorted_indices[:int(len(population)*keep_top)]
    return [population[i] for i in top_indices]

def crossover(parent1, parent2):
    n = len(parent1)
    crossover_point = np.random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n = len(h)
    mask = np.random.rand(n) < mutation_rate
    if np.any(mask):
        noise = np.random.normal(0, 0.05, n)
        h = h + noise * mask
        h = np.clip(h, 0, 1)
        
        # Repair to maintain sum constraint
        current_sum = np.sum(h)
        target_sum = len(h) / 2
        if abs(current_sum - target_sum) > 1e-8:
            h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    pop_size = 50
    max_generations = 1000
    patience = 200
    last_improvement = 0
    
    population = initialize_population(n_points, pop_size)
    
    fitnesses = [evaluate_h(h, dx) for h in population]
    
    best_h = population[np.argmin(fitnesses)].copy()
    best_c5 = min(fitnesses)
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection
        selected = selection(population, fitnesses)
        
        # Elitism: keep the best
        new_population = [best_h.copy()]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            if np.random.rand() < 0.5:
                child1, child2 = crossover(parent1, parent2)
                new_population.append(mutate(child1))
                if len(new_population) < pop_size:
                    new_population.append(mutate(child2))
            else:
                new_population.append(mutate(parent1))
                if len(new_population) < pop_size:
                    new_population.append(mutate(parent2))
                    
        population = new_population[:pop_size]
        
        # Evaluate population
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            last_improvement = gen
            
        # Early stopping
        if gen - last_improvement > patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
