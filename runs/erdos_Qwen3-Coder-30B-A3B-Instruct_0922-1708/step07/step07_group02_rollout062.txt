import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def segment_blend(parent1, parent2, n_points):
    """Blend two parents by taking segments from each."""
    # Choose a random number of segments to take from parent1
    n_segments = np.random.randint(1, n_points // 2)
    segments = np.random.choice(n_points, n_segments, replace=False)
    offspring = parent1.copy()
    for i in segments:
        offspring[i] = parent2[i]
    return offspring

def mutate_h(h, n_points, mutation_scale):
    """Mutate h with Gaussian noise and segment flips."""
    h = h.copy()
    # Gaussian noise
    noise = np.random.normal(0, mutation_scale, n_points)
    h += noise
    # Segment flips
    if np.random.random() < 0.3:
        flip_count = max(1, int(0.1 * n_points))
        flip_indices = np.random.choice(n_points, flip_count, replace=False)
        for i in flip_indices:
            h[i] = 1 - h[i]
    return h

def local_refine(h, n_points, dx, max_iter=50):
    """Run a fast local refinement."""
    eta = 0.01
    eps = 1e-6
    for _ in range(max_iter):
        h_new = h.copy()
        indices = list(range(n_points))
        np.random.shuffle(indices)
        improved = False
        
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
        
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < evaluate_h(h, dx):
            h = h_new
            improved = True
        if not improved:
            eta *= 0.99
    return h

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    # Memetic algorithm parameters
    pop_size = 50
    elite_size = 10
    mutation_scale = 0.1
    max_generations = 1500
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        h = mutate_h(h_init, n_points, mutation_scale)
        h = project_h(h, n_points, dx)
        population.append(h)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(h_init, dx)
    
    generation = 0
    last_improvement = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 5 and generation < max_generations:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:elite_size]
        population = [population[i] for i in sorted_indices]
        elite_fitness = [fitness[i] for i in elite_indices]
        
        if min(elite_fitness) < best_c5:
            best_h = population[0].copy()
            best_c5 = min(elite_fitness)
            last_improvement = generation
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adapt mutation scale
        if stagnation_count > 20:
            mutation_scale *= 0.95
            stagnation_count = 0
        
        # Generate new population
        new_population = []
        # Elitism
        new_population.extend([population[i].copy() for i in elite_indices])
        
        # Create offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.choice(elite_size)
            parent2_idx = np.random.choice(elite_size)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            offspring = segment_blend(parent1, parent2, n_points)
            
            # Mutation
            offspring = mutate_h(offspring, n_points, mutation_scale)
            
            # Project to feasible region
            offspring = project_h(offspring, n_points, dx)
            
            # Local refinement
            offspring = local_refine(offspring, n_points, dx, 50)
            
            new_population.append(offspring)
        
        population = new_population[:pop_size]
    
    return (best_h.tolist(), best_c5, n_points)
