import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, k_max, n_points):
    """Compute gradient for the maximum overlap at shift k_max"""
    # Create the shifted version of h
    h_shifted = np.zeros(n_points)
    if k_max >= 0:
        h_shifted[k_max:] = h[:-k_max]
    else:
        h_shifted[:k_max] = h[-k_max:]
    
    # Gradient is 1 - h_shifted for each entry
    return 1 - h_shifted

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a trapezoidal wave
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))
        h_init = project_h(h_init, n_points)
    
    # Current best solution
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Optimization parameters
    max_iter = int(1e6)
    patience = 50
    learning_rate = 0.05
    last_improvement = 0
    
    # Multiple initializations to avoid local optima
    for trial in range(5):
        np.random.seed(seed + trial)
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Find shift k that gives maximum correlation
            corr = correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr) - (n_points - 1)
            
            # Compute exact gradient
            grad = compute_gradient(h_current, k_max, n_points)
            
            # Update using gradient
            h_new = h_current - learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept improvement or with some probability
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
            elif np.random.rand() < 0.005:  # Allow occasional worse moves
                h_current = h_new
                c5_current = c5_new
            
            # Update best
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
            
            # Adapt learning rate
            if i - last_improvement > patience:
                learning_rate *= 0.95
                last_improvement = i
            
            # Periodic perturbation
            if i % 100 == 0 and i > 0:
                noise = np.random.normal(0, 0.005, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
    
    return (h_best.tolist(), c5_best, n_points)
