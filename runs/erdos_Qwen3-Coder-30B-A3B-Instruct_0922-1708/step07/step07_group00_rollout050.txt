import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from sobol_sequence import sobol_sequence

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx):
    """Compute gradient of the maximum overlap with respect to h."""
    n = len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n - 1)
    
    # Compute finite difference gradient
    eps = 1e-6
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    # Normalize
    grad = grad / (np.linalg.norm(grad) + 1e-12)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = None
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate diverse initial points
    n_initials = 300
    initial_points = []
    
    # Generate Sobol sequence
    sobol_points = sobol_sequence(n_initials, n_points)
    for i in range(n_initials):
        h = np.clip(sobol_points[i], 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        initial_points.append(h)
    
    # Add the best known solution as an initialization
    initial_points.append(h_init)
    
    # Try multiple initial points
    for i, h in enumerate(initial_points[:n_initials]):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Nelder-Mead optimization
        def obj_func(h_vec):
            h_repaired = repair_h(h_vec, n_points)
            return evaluate_h(h_repaired, dx)
        
        # Run Nelder-Mead with different starting points
        res = minimize(obj_func, h, method='Nelder-Mead',
                       options={'maxiter': 1000, 'xatol': 1e-6, 'fatol': 1e-6})
        
        if not res.success:
            continue
            
        h_opt = repair_h(res.x, n_points)
        c5_opt = evaluate_h(h_opt, dx)
        
        # Refinement with gradient descent
        h_refined = h_opt.copy()
        lr = 1e-5
        for _ in range(1000):
            if time.time() - start_time > budget_s - 10:
                break
            grad = compute_gradient(h_refined, dx)
            h_refined = h_refined - lr * grad
            h_refined = repair_h(h_refined, n_points)
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            c5_refined = np.max(corr) * dx
            if c5_refined < c5_opt:
                c5_opt = c5_refined
                h_opt = h_refined.copy()
            lr *= 0.999
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    if best_h is None:
        best_h = h_init.copy()
        best_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
