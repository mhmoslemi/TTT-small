import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize multiple starting points
    initial_hs = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
        initial_hs.append(h_init)
    
    # Additional starting points
    # Sine wave pattern
    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 * (1 + np.sin(np.pi * x)) 
    h_sine = project_h(h_sine, n_points)
    initial_hs.append(h_sine)
    
    # Trapezoidal pattern
    h_trap = np.zeros(n_points)
    mid = n_points // 2
    h_trap[:mid] = np.linspace(0, 1, mid)
    h_trap[mid:] = np.linspace(1, 0, n_points - mid)
    h_trap = project_h(h_trap, n_points)
    initial_hs.append(h_trap)
    
    # Random starting point
    h_rand = np.random.rand(n_points)
    h_rand = project_h(h_rand, n_points)
    initial_hs.append(h_rand)
    
    # Initialize optimization variables
    h_best = None
    c5_best = float('inf')
    
    # Allow for multiple runs, each with different starting points
    for i, h_init in enumerate(initial_hs):
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
        
        # Run optimization for this initialization
        # Use a more sophisticated optimization approach with step size control
        learning_rate = 0.05
        max_iter = 2000
        patience = 100
        last_improvement = 0
        
        for j in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute max correlation shift
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_shift = np.argmax(corr)
            max_overlap = corr[max_shift] * dx
            
            # Compute gradient for the maximum shift
            d_h = np.zeros(n_points)
            if max_shift < n_points:
                d_h[max_shift:] = -h_current[:n_points - max_shift]
            else:
                d_h[:n_points - max_shift] = -h_current[max_shift:]
            
            # Gradient update
            h_new = h_current + learning_rate * d_h
            h_new = project_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                last_improvement = j
            else:
                if np.random.rand() < 0.02:  # Occasionally accept worse steps
                    h_current = h_new
                    c5_current = c5_new
                    last_improvement = j
            
            # Reduce learning rate if stagnating
            if j - last_improvement > patience:
                learning_rate *= 0.95
                last_improvement = j
                
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                
            # Occasionally perturb to escape local minima
            if j % 50 == 0 and j > 0:
                noise = np.random.normal(0, 0.003, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
                
            # Occasionally restore best solution
            if j % 100 == 0:
                # Restore best if not current
                if evaluate_h(h_current, dx) > c5_best + 1e-8:
                    h_current = h_best.copy()
                    c5_current = c5_best
        
        # Final local search around the best from this initialization
        for _ in range(500):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_new = h_current.copy()
            indices = np.random.choice(n_points, max(3, n_points // 30), replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.003)
                
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
    
    return (h_best.tolist(), c5_best, n_points)
