import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Simulated Annealing approach with local search
    time_left = lambda: budget_s - (time.time() - start_time)
    
    # Parameters
    temp = 1.0
    cooling_rate = 0.9995
    min_temp = 1e-6
    max_iter = int(1e6)
    patience = 1000
    last_improvement = 0
    
    # Start with binary version for exploration
    h_binary = np.round(h_current).astype(int)
    h_current = project_h(h_binary.astype(float), n_points)
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    for i in range(max_iter):
        if time_left() < 5:
            break
            
        # Create neighbor by perturbing a few points
        h_new = h_current.copy()
        
        # Perturb at most 5 random indices
        perturb_indices = np.random.choice(n_points, np.random.randint(1, 6), replace=False)
        for idx in perturb_indices:
            # Try a small random change in value
            change = np.random.normal(0, 0.05)
            h_new[idx] += change
            # Ensure we stay feasible
            h_new[idx] = np.clip(h_new[idx], 0, 1) 
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Accept with probability based on temperature
        delta = c5_new - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h_current = h_new.copy()
            c5_current = c5_new
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
        # Schedule cooling
        if temp > min_temp:
            temp *= cooling_rate
            
        # Early stopping if no improvement
        if i - last_improvement > patience:
            # Perturb to escape if needed
            h_current = project_h(h_current + np.random.normal(0, 0.01, n_points), n_points)
            c5_current = evaluate_h(h_current, dx)
            last_improvement = i
    
    # Final local search refinement
    for _ in range(1000):
        if time_left() < 5:
            break
            
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
