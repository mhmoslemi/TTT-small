import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, rate=0.1, delta=0.1):
    """Mutate h by perturbing a fraction of entries."""
    h_new = h.copy()
    mask = np.random.rand(len(h_new)) < rate
    h_new[mask] += np.random.uniform(-delta, delta, np.sum(mask))
    return repair_h(h_new, len(h))

def crossover_h(h1, h2):
    """Crossover two h vectors by segment shuffling."""
    n = len(h1)
    split = np.random.randint(1, n)
    h_new = np.concatenate([h1[:split], h2[split:]])
    return repair_h(h_new, n)

def evaluate_population(population, dx):
    """Evaluate all individuals in the population."""
    return np.array([evaluate_h(h, dx) for h in population])

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Initialize population
    population_size = 100
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        population.append(mutate_h(h_init.copy(), rate=0.1, delta=0.1))
    
    # Evaluate initial population
    fitness = evaluate_population(population, dx)
    best_idx = np.argmin(fitness)
    h_best = population[best_idx].copy()
    c5_best = fitness[best_idx]
    
    # Evolution loop
    generations = 100
    patience = 20
    last_improvement = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection: keep top 50% 
        sorted_indices = np.argsort(fitness)
        top_half = sorted_indices[:population_size//2]
        parents = [population[i] for i in top_half]
        
        # Generate new population
        new_population = parents.copy()
        while len(new_population) < population_size:
            # Crossover
            p1, p2 = np.random.choice(parents, 2, replace=False)
            offspring = crossover_h(p1, p2)
            
            # Mutation
            mutated = mutate_h(offspring, rate=0.1, delta=0.1)
            
            new_population.append(mutated)
        
        # Evaluate new population
        population = new_population[:population_size]
        fitness = evaluate_population(population, dx)
        
        # Update best solution
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < c5_best:
            h_best = population[min_idx].copy()
            c5_best = fitness[min_idx]
            last_improvement = gen
        
        # Early stopping
        if gen - last_improvement > patience:
            break
    
    # Final gradient refinement on the best solution
    h_current = h_best.copy()
    c5_current = c5_best
    eta = 0.001
    eps = 1e-6
    max_iter = 5000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_current.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_current[j] - eta * grad_j
        
        # Apply repair step to maintain feasibility
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if better
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            
            # Update best if improved
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
    
    return (h_best.tolist(), c5_best, n_points)
