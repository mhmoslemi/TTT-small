import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Apply mutation to a candidate solution."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Population-based evolution strategy
    population_size = 50
    mutation_scale = 0.1
    population = []
    fitness = []
    
    # Initial solutions
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    # Generate initial population
    population = [h_init]
    for i in range(population_size - 1):
        # Mix initial solution with random mutations
        h_new = mutate(h_init, 0, 0.1)
        population.append(h_new)
    
    # Main evolution loop
    generation = 0
    stagnant = 0
    best_c5 = float('inf')
    best_h = h_init.copy()
    
    while time.time() - start_time < budget_s - 5 and generation < 1000:
        # Evaluate all population members
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        best_idx = sorted_indices[0]
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Select top 20% as parents
        num_parents = max(1, population_size // 5)
        parents = [population[i] for i in sorted_indices[:num_parents]]
        
        # Generate new population
        new_population = [parents[0].copy()]  # Keep best
        for i in range(population_size - 1):
            # Select two random parents and average
            parent1 = parents[np.random.randint(len(parents))]
            parent2 = parents[np.random.randint(len(parents))]
            child = (parent1 + parent2) / 2.0
            
            # Mutate
            child = mutate(child, generation, mutation_scale)
            new_population.append(child)
        
        population = new_population
        
        # Adapt mutation scale
        if stagnant > 10:
            mutation_scale *= 0.95
        
        generation += 1
    
    return (best_h.tolist(), best_c5, n_points)
