import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of the maximum overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Identifying the k that gives maximum overlap
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n - 1)
    
    # Computing gradient for this k
    # For small perturbations, we approximate the gradient
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    # Normalize the gradient
    grad = grad / (np.linalg.norm(grad) + 1e-12)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    swarm_size = 100
    max_iter = 10000
    w_start = 0.9
    w_end = 0.1
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    
    for _ in range(swarm_size):
        h = h_init.copy() + np.random.normal(0, 0.01, n_points)
        h = repair_h(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        pbest_positions.append(h.copy())
        pbest_scores.append(evaluate_h(h, dx))
    
    # Global best
    best_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    # PSO main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (gbest_position - particles[i]))
            
            # Apply velocity limits
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair position
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            current_score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest_positions[i] = particles[i].copy()
                pbest_scores[i] = current_score
            
            # Update global best
            if current_score < gbest_score:
                gbest_position = particles[i].copy()
                gbest_score = current_score
        
        # Occasionally refine global best with gradient descent
        if iter_num % 50 == 0 and iter_num > 0:
            # Perform gradient descent on best solution
            h_refined = gbest_position.copy()
            lr = 1e-5
            for _ in range(100):
                if time.time() - start_time > budget_s - 5:
                    break
                grad = compute_gradient(h_refined, dx, n_points)
                h_refined = h_refined - lr * grad
                h_refined = repair_h(h_refined, n_points)
                new_score = evaluate_h(h_refined, dx)
                if new_score < gbest_score:
                    gbest_position = h_refined.copy()
                    gbest_score = new_score
                else:
                    break
    
    # Final refinement
    h_refined = gbest_position.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = compute_gradient(h_refined, dx, n_points)
        h_refined = h_refined - 1e-5 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        c5_refined = np.max(corr) * dx
        if c5_refined < gbest_score:
            gbest_position = h_refined.copy()
            gbest_score = c5_refined
    
    return (gbest_position.tolist(), gbest_score, n_points)
