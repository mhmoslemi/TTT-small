import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Try a cosine-based taper
        h_init = 0.5 * (1 + np.cos(np.pi * x))
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Phase 1: CCCP-like approach with convex approximations
    learning_rate = 0.01
    max_iter_1 = 5000
    patience = 500
    last_improvement = 0
    for i in range(max_iter_1):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Find the maximizing shift
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - (len(h_current) - 1)
        
        # Compute gradient for current best shift
        h_shifted = np.roll(h_current, max_shift)
        grad = 1 - h_shifted
        
        # Update with gradient
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse moves (simulated annealing)
            if np.random.rand() < 0.05:
                h_current = h_new.copy()
                c5_current = c5_new
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = i
            
        # Occasionally jitter to escape local minima
        if i % 500 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Phase 2: Cyclic coordinate descent
    max_iter_2 = 10000
    patience = 1000
    last_improvement = 0
    learning_rate_cd = 0.1
    
    for i in range(max_iter_2):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cycle through all coordinates
        indices = np.random.permutation(n_points)
        improved = False
        for idx in indices:
            # Evaluate current best shift (this can be expensive; better to do once)
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr)
            max_shift = max_shift_idx - (len(h_current) - 1)
            
            # Compute gradient component
            h_shifted = np.roll(h_current, max_shift)
            grad_component = 1 - h_shifted[idx]
            
            # Update
            h_new = h_current.copy()
            h_new[idx] = h_current[idx] - learning_rate_cd * grad_component
            
            # Clip and renormalize
            h_new = np.clip(h_new, 0, 1)
            current_sum = np.sum(h_new)
            if abs(current_sum) > 1e-12:
                h_new = h_new * (n_points / 2) / current_sum
                
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                improved = True
        if not improved:
            # Occasionally accept worse moves to escape local minima
            if np.random.rand() < 0.02:
                idx = np.random.randint(0, n_points)
                noise = np.random.normal(0, 0.005)
                h_current[idx] += noise
                h_current = project_h(h_current, n_points)
                c5_current = evaluate_h(h_current, dx)
                
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            learning_rate_cd *= 0.95
            last_improvement = i
            
        # Occasionally perturb the entire solution
        if i % 1000 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)

    return (h_best.tolist(), c5_best, n_points)
