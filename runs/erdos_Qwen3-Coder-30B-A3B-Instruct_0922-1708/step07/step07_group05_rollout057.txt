import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size):
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Fill remaining population with triangular pulses
    for i in range(pop_size - len(population)):
        h = np.zeros(n_points)
        center = np.random.uniform(0.5, 1.5)
        width = np.random.uniform(0.2, 0.8)
        for j in range(n_points):
            x = j * 2.0 / n_points
            if abs(x - center) < width / 2:
                h[j] = 1.0 - 2.0 * abs(x - center) / width
        h = repair_h(h, n_points)
        population.append(h)
    
    return population

def crossover(parent1, parent2):
    n = len(parent1)
    crossover_point = np.random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n = len(h)
    num_mutations = int(n * mutation_rate)
    indices = np.random.choice(n, num_mutations, replace=False)
    for idx in indices:
        h[idx] += np.random.normal(0, 0.05)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # GA parameters
    pop_size = 50
    generations = 1000
    elitism_rate = 0.2
    mutation_rate = 0.2
    crossover_rate = 0.5
    patience = 200
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_h(h, dx)
        fitness.append(fit)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    last_improvement = 0
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_fitness:
            best_fitness = fitness[0]
            best_h = population[0].copy()
            last_improvement = generation
        
        # Elitism
        num_elite = int(elitism_rate * pop_size)
        elite = population[:num_elite]
        
        # Create new population through selection, crossover, and mutation
        new_population = elite[:]
        
        # Selection: tournament selection
        def tournament_selection(pop, fit, k=3):
            idxs = np.random.choice(len(pop), k)
            best_idx = idxs[np.argmin([fit[i] for i in idxs])]
            return pop[best_idx]
            
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitness)
            if np.random.rand() < crossover_rate:
                parent2 = tournament_selection(population, fitness)
                child1, child2 = crossover(parent1, parent2)
                child1 = repair_h(mutate(child1, mutation_rate), n_points)
                child2 = repair_h(mutate(child2, mutation_rate), n_points)
                new_population.extend([child1, child2])
            else:
                child = repair_h(mutate(parent1, mutation_rate), n_points)
                new_population.append(child)
        
        # Trim to exact population size
        population = new_population[:pop_size]
        
        # Re-evaluate fitness
        fitness = [evaluate_h(h, dx) for h in population]
        
        # Early termination
        if generation - last_improvement > patience:
            break
    
    return (best_h.tolist(), best_fitness, n_points)
