import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with the provided initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Fallback initialization if no initial_h_values
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Define optimization variables
    h = cp.Variable(n_points)
    
    # Objective is a non-convex function depending on cross-correlation
    # For this approach, we'll use projected gradient descent with adjustments
    # Set up the SOCP relaxation
    # Add a small epsilon to avoid singularities
    epsilon = 1e-8
    
    # Try an SOCP relaxation approach
    # For the current greedy improvement, we still use coordinate descent with better heuristics
    
    max_iter = 50000
    eta = 0.02
    patience = 100
    last_improvement = 0
    
    # Create a version of h with a small noise for potentially better escape
    def noisy_h(h_ref):
        h_noisy = h_ref + np.random.normal(0, 0.001, n_points)
        return np.clip(h_noisy, 0, 1)
    
    # Coordinate descent with gradient approximation enhanced
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_current.copy()
        improved = False
        
        # Batch updates
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices:
            eps = 1e-6
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
            
        # Apply final projection
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            improved = True
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
        # Adjust learning rate
        if not improved:
            eta *= 0.999
        else:
            eta = min(0.05, eta * 1.001)
            
        # Re-randomize if no improvement
        if i - last_improvement > patience:
            h_current = noisy_h(h_current)
            c5_current = evaluate_h(h_current, dx)
            last_improvement = i
            
    return (h_best.tolist(), c5_best, n_points)
