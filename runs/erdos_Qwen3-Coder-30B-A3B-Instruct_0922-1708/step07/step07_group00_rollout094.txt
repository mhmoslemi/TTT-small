import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Evolution Strategy parameters
    pop_size = 200
    elites_fraction = 0.2
    offsprings_per_elite = 5
    initial_mutation_std = 0.1
    patience = 30
    
    # Initialize population
    pop = [h_init.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        pop.append(h)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    patience_counter = 0
    
    generation = 0
    while time.time() - start_time < budget_s - 5 and patience_counter < patience:
        # Evaluate population
        fitness = []
        for h in pop:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort population by fitness
        indices = np.argsort(fitness)
        pop = [pop[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = pop[0].copy()
            best_c5 = fitness[0]
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Select elites
        elites_count = max(1, int(pop_size * elites_fraction))
        elites = pop[:elites_count]
        
        # Generate new population
        new_pop = elites.copy()
        mutation_std = initial_mutation_std * (0.95 ** generation)
        
        for elite in elites:
            for _ in range(offsprings_per_elite):
                mutant = elite + np.random.randn(n_points) * mutation_std
                mutant = repair_h(mutant, n_points)
                new_pop.append(mutant)
        
        # Keep only the best solutions
        pop = new_pop[:pop_size]
        generation += 1
    
    return (best_h.tolist(), best_c5, n_points)
