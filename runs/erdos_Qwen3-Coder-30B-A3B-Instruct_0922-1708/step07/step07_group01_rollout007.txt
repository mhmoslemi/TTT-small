import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a step function
        h_init = np.zeros(n_points)
        midpoint = n_points // 2
        h_init[:midpoint] = 1.0
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Enhanced optimization using local search
    max_iter = int(1e6)
    patience = 500
    last_improvement = 0
    learning_rate = 0.02
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Local search with perturbation
        h_new = h_current.copy()
        # Select a random subset of indices to modify
        indices = np.random.choice(n_points, max(5, n_points // 20), replace=False)
        for idx in indices:
            # Small random perturbation
            perturbation = np.random.normal(0, 0.01)
            h_new[idx] += perturbation
            
        # Project back to feasible region
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if it improves
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
            last_improvement = i
        else:
            # Occasionally accept worse moves (simulated annealing)
            if np.random.rand() < 0.03:
                h_current = h_new.copy()
                c5_current = c5_new
                last_improvement = i
                
        # Track best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            if learning_rate < 1e-5:
                learning_rate = 1e-5
            
        # Periodic restarts with best solution
        if i % 1000 == 0 and i > 0:
            h_current = h_best.copy()
            c5_current = c5_best

    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        # Perturb a few coordinates
        indices = np.random.choice(n_points, max(3, n_points // 30), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new

    return (h_best.tolist(), c5_best, n_points)
