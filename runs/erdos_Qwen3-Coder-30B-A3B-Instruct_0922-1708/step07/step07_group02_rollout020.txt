import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Use coordinate descent with analytical gradients for the dominant term
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    eta = 0.01
    patience = 100
    last_improvement = 0
    max_iter = 100000
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find the maximizing lag term
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient more efficiently using the dominant term
        h_new = h_current.copy()
        grad = np.zeros(n_points)
        
        # Construct gradient based on the k_max term
        # d/dh[i] (h[i] * (1-h[i+k])) = 1 - 2*h[i] if k = k_max, 0 elsewhere
        # But we need to be careful about how to compute i+k
        # For the k'th shift in cross-correlation: h[i] * (1-h[i+k])
        # We make the gradient d/dh[i] = -(1 - h[i+k]) if k = k_max, 0 else
        # Actually, for the full integral, it's more subtle
        
        # Let's reconstruct with more care:
        # For indexing into the convolution (full):
        # corr[k] corresponds to sum_{i} h[i] * (1-h[i+k-offset]) where offset = n_points - 1
        # so for k = k_max, k_offset = k_max - (n_points - 1) = k_max - n_points + 1
        # For i in range(n_points), we want to see which term in h contributes in the k_max shift
        # This means we find the gradient wrt h[i] when k_max = i + k_offset --> means k_offset = i - k_max + n_points - 1
        offset = n_points - 1
        k_offset = k_max - offset  # this is the actual shift in h[i+k_max]
        # shift for indexing back to expansion
        # Need to run corr computation for actual pattern

        # if k_max = i + offset - k, i.e. i = k_max - offset + k, then the term
        # (1 - h[i+k]) = (1 - h[k_max - offset + k + k])

        h_new = h_current - eta * grad

                h_current = h_perturb
                c5_current = c5_perturb
                last_improvement = i
                
    return (h_best.tolist(), c5_best, n_points)
