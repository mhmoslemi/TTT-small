import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from sobol import sobol_sequence  # Assuming sobol_sequence is available

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Generate multiple initial points using Sobol sequences
    n_initials = 500
    sobol_points = sobol_sequence(n_initials, n_points)
    
    for i in range(n_initials):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate initial h from Sobol point
        h = sobol_points[i] * 2  # Scale to [0,2]
        h = repair_h(h, n_points)
        
        # Nelder-Mead optimization
        try:
            res = minimize(
                lambda x: evaluate_h(repair_h(x, n_points), dx),
                h,
                method='Nelder-Mead',
                options={'maxiter': 1000, 'xatol': 1e-6, 'fatol': 1e-6}
            )
            h_opt = res.x
        except:
            continue
            
        # Refinement with gradient descent
        h_refined = repair_h(h_opt, n_points)
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient via finite differences
            grad = np.zeros(n_points)
            eps = 1e-6
            base_val = evaluate_h(h_refined, dx)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                grad[j] = (evaluate_h(h_plus, dx) - base_val) / eps
            
            # Update with gradient descent
            h_refined = h_refined - 1e-5 * grad
            h_refined = repair_h(h_refined, n_points)
            new_c5 = evaluate_h(h_refined, dx)
            
            if new_c5 < best_c5:
                best_h = h_refined.copy()
                best_c5 = new_c5
                
        if evaluate_h(h_refined, dx) < best_c5:
            best_h = h_refined.copy()
            best_c5 = evaluate_h(h_refined, dx)
    
    return (best_h.tolist(), best_c5, n_points)
