import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    # CVXPY approach for convex relaxation
    h = cp.Variable(n_points, name="h")
    
    # Set up constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]
    
    # Use a simple upper bound approximation via SOCP (relaxation)
    # Since the problem is non-convex, we use a heuristic approach to approximate
    # the correlation term
    
    # Define a hypothetical bound variable
    C = cp.Variable(nonneg=True, name="C")
    
    # Simplified relaxation approach:
    # Formulate a convex optimization problem with a relaxation
    # We treat it as convex by using `h` in a quadratic sense
    # This is a heuristic; intended to avoid local minima
    
    # Create a basic simplest-use relaxation:
    # Minimize quadratic norm of h*1 which serves as proxy for minimum overlap
    objective = cp.Minimize(cp.sum_squares(h))
    
    # Handle boundary effects in a basic way
    # A more sophisticated form would involve SOCP etc., but for
    # speed and resources, we use trial-and-error with a good initial
    
    prob = cp.Problem(objective, constraints)
    
    # Try solving the convex problem with warm start
    try:
        prob.solve(solver=cp.MOSEK, warm_start=True)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = h.value
        else:
            h_sol = h_init
    except:
        h_sol = h_init
    
    h_sol = np.clip(h_sol, 0, 1)
    h_sol = project_h(h_sol, n_points, dx)
    
    h_best = h_sol.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Try further minimizing using a local search method
    # within a simplified trust-region like approach:
    # 1. Use initial h as base
    # 2. Perform small coordinate-wise changes
    # 3. Use history and prior good points to guide further search
    
    # Perform randomized coordinate descent with gradient estimation
    # using finite differences (reduced variant)
    h_current = h_best.copy()
    c5_current = c5_best
    
    # Perturbations and gradient estimation
    max_iter = 50000
    eta = 0.02
    eps = 1e-6
    patience = 50
    
    last_improvement = 0
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_current.copy()
        improved = False
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices[:20]:  # Limit to some indices to reduce compute
            # Gradient estimation using finite differences
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] -= eta * grad_j
            h_new[j] = np.clip(h_new[j], 0, 1)
            
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            improved = True
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
        # Adaptive learning rate
        if not improved:
            eta *= 0.995
        else:
            eta = min(0.05, eta * 1.01)
            
        # Escape local minima
        if i - last_improvement > patience:
            h_perturb = h_current.copy()
            num_perturb = max(1, int(0.02 * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                delta = np.random.uniform(-0.02, 0.02)
                h_perturb[idx] += delta
            h_perturb = project_h(h_perturb, n_points, dx)
            c5_perturb = evaluate_h(h_perturb, dx)
            if c5_perturb < c5_current:
                h_current = h_perturb
                c5_current = c5_perturb
                last_improvement = i
    
    return (h_best.tolist(), c5_best, n_points)
