import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Binary initialization with correct sum
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)

    # Annealing parameters
    initial_temp = 0.1
    cooling_rate = 0.995
    min_temp = 1e-6
    
    best_c5 = float('inf')
    best_h = h.copy()
    current_h = h.copy()
    
    # Helper to enforce constraints
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-10:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Compute C5
    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Local perturbation: move mass between two points
    def perturb(h, delta=0.05):
        new_h = h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        move = np.random.uniform(-delta, delta)
        new_h[i] += move
        new_h[j] -= move
        return project_h(new_h)
    
    current_c5 = compute_c5(current_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        temp = initial_temp * (cooling_rate ** iteration)
        if temp < min_temp:
            temp = min_temp
            
        # Generate candidate
        new_h = perturb(current_h)
        new_c5 = compute_c5(new_h)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
                current_h = new_h
                current_c5 = new_c5
                
        # Keep best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    
    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
