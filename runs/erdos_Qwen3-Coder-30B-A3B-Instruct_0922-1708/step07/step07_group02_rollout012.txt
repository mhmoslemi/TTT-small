import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def segment_blend(p1, p2, n_points):
    """Blend two parents by segment swapping."""
    # Determine number of segments
    n_segments = 4
    segment_size = n_points // n_segments
    segments1 = [p1[i*segment_size:(i+1)*segment_size] for i in range(n_segments)]
    segments2 = [p2[i*segment_size:(i+1)*segment_size] for i in range(n_segments)]
    
    # Randomly choose segments from each parent
    offspring = []
    for i in range(n_segments):
        if np.random.random() < 0.5:
            offspring.extend(segments1[i])
        else:
            offspring.extend(segments2[i])
            
    return np.array(offspring)

def memetic_optimization(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Memetic algorithm parameters
    population_size = 50
    num_parents = 10
    max_generations = 1500
    mutation_scale = 0.1
    learning_rate = 0.01
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = h_init.copy()
        # Add small random perturbations
        perturb_indices = np.random.choice(n_points, size=int(0.1*n_points), replace=False)
        for idx in perturb_indices:
            h[idx] += np.random.normal(0, 0.02)
        h = project_h(h, n_points, dx)
        population.append(h)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluation
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            
        # Selection (tournament)
        selected = []
        for _ in range(num_parents):
            tournament = np.random.choice(len(population), size=3, replace=False)
            best_idx = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[best_idx])
            
        # Crossover and mutation
        new_population = [best_h]  # Elitism
        while len(new_population) < population_size:
            # Crossover
            p1, p2 = np.random.choice(selected, size=2, replace=False)
            offspring = segment_blend(p1, p2, n_points)
            
            # Mutation: add Gaussian noise
            if np.random.random() < 0.8:  # 80% chance of mutation
                noise_indices = np.random.choice(n_points, size=int(0.05*n_points), replace=False)
                for idx in noise_indices:
                    offspring[idx] += np.random.normal(0, mutation_scale)
            
            # Project to feasibility
            offspring = project_h(offspring, n_points, dx)
            
            # Local refinement using coordinate descent
            h_refined = offspring.copy()
            c5_refined = evaluate_h(h_refined, dx)
            
            for _ in range(50):  # Limited refinement steps
                h_new = h_refined.copy()
                improved = False
                indices = list(range(n_points))
                np.random.shuffle(indices)
                
                for j in indices:
                    eps = 1e-6
                    h_plus = h_new.copy()
                    h_minus = h_new.copy()
                    h_plus[j] += eps
                    h_minus[j] -= eps
                    
                    h_plus = project_h(h_plus, n_points, dx)
                    h_minus = project_h(h_minus, n_points, dx)
                    
                    c5_plus = evaluate_h(h_plus, dx)
                    c5_minus = evaluate_h(h_minus, dx)
                    
                    grad_j = (c5_plus - c5_minus) / (2 * eps)
                    h_new[j] -= learning_rate * grad_j
                    h_new[j] = max(0, min(1, h_new[j]))
                    
                h_new = project_h(h_new, n_points, dx)
                c5_new = evaluate_h(h_new, dx)
                
                if c5_new < c5_refined:
                    h_refined = h_new
                    c5_refined = c5_new
                    improved = True
                
                if not improved:
                    break
                    
            new_population.append(h_refined)
            
            # Update best solution
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
                
        population = new_population
        mutation_scale *= 0.99
        learning_rate *= 0.999
        
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return memetic_optimization(seed, budget_s)
