import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        h_init = np.zeros(n_points)
        # Create a piecewise linear function that's 0 on [0,0.5) and 1 on (1.5,2]
        # and slopes linearly in between
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)

    # Particle swarm optimization
    n_particles = 100
    velocities = np.zeros((n_particles, n_points))
    particles = np.tile(h_init, (n_particles, 1)) + np.random.normal(0, 0.01, (n_particles, n_points))
    for p in range(n_particles):
        particles[p] = repair_h(particles[p], n_points)
    
    pbest = particles.copy()
    pbest_vals = np.array([evaluate_h(p, dx) for p in particles])
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO parameters
    w = 0.9
    c1 = 2.0
    c2 = 2.0
    inertia_decay = 0.995
    max_iter = 400
    patience = 100
    last_improvement = 0
    
    # Mutation parameters
    mutation_rate = 0.1
    mutation_sigma = 0.1
    mutation_size = int(0.2 * n_points)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        if i > 100:
            w = max(0.4, w * inertia_decay)
            
        # Evaluate all particles
        current_vals = np.array([evaluate_h(p, dx) for p in particles])
        
        # Update personal best
        mask = current_vals < pbest_vals
        pbest[mask] = particles[mask]
        pbest_vals[mask] = current_vals[mask]
        
        # Update global best
        new_gbest = np.argmin(pbest_vals)
        if pbest_vals[new_gbest] < gbest_val:
            gbest = pbest[new_gbest].copy()
            gbest_val = pbest_vals[new_gbest]
            last_improvement = i
            
        # Update velocities and positions
        r1, r2 = np.random.rand(n_particles, 1), np.random.rand(n_particles, 1)
        for p in range(n_particles):
            velocities[p] = (w * velocities[p] + 
                           c1 * r1[p] * (pbest[p] - particles[p]) + 
                           c2 * r2[p] * (gbest - particles[p]))
            
            # Add mutation noise occasionally
            if i % 25 == 0 and np.random.rand() < mutation_rate:
                indices = np.random.choice(n_points, size=mutation_size, replace=False)
                velocities[p][indices] += np.random.normal(0, mutation_sigma, mutation_size)
            
            # Update position
            particles[p] += velocities[p]
            particles[p] = repair_h(particles[p], n_points)
            
        # Early termination if no improvement
        if i - last_improvement > patience:
            break

    final_c5 = evaluate_h(gbest, dx)
    return (gbest.tolist(), final_c5, n_points)
