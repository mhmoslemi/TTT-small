import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter_barrier = 1000
    max_iter_gd = 10000
    learning_rate = 1e-3
    barrier_decrease_factor = 0.99
    mu_initial = 1e3
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Barrier method
    mu = mu_initial
    for iter_barrier in range(max_iter_barrier):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find index of maximum
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(h) - 1)
        
        # Compute exact gradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_shifted = np.roll(h, max_lag)
            if max_lag < 0:
                h_shifted[:max_lag] = 0
            else:
                h_shifted[max_lag:] = 0
            grad[i] = h_shifted[i] * dx
        
        # Update
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adapt barrier parameter
        mu *= barrier_decrease_factor
    
    # Gradient descent phase
    for iter_gd in range(max_iter_gd):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find index of maximum
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(h) - 1)
        
        # Compute gradient via adjoint
        grad = np.zeros_like(h)
        
        # For the current maximum, we analyze the derivative
        # This is essentially computing the derivative of correlation
        # Functionally it's a convolution of 1-h with h shifted forward by lag
        # But we're interested in the shift which maximizes the correlation
        
        # Separately, we can consider gradients analytically:
        # Let s = max_k ∫ h(x) (1 - h(x + k)) dx
        # ∂s/∂h_i = sum over k of (1 - h_i + k) * indicator that this contributes to max_k
        # We simplify by noting max_k is known at this point
        
        # More precisely: 
        # if correlation at lag k is maximum
        # then at each position i:
        # ∂/∂h_i [ ∫ h(x) (1 − h(x+k)) dx ] = (1 - h[i+k]) for i such that (i+k) ≤ len(h)
        # But we want ∫ h(x) (1 - h(x+k)) dx which is over i, but we're differentiating w.r.t h_i
        
        # We compute direction of descent
        # There are several approximations possible; simplest for now:
        # best_lag = argmax correlation over all lags
        # Then for each index i: derivative is negative of h[i+best_lag] if that exists
        
        if abs(max_lag) < len(h):
            for i in range(len(h)):
                j = i + max_lag
                if 0 <= j < len(h):
                    grad[i] += (1 - h[j]) * dx
        else:
            grad[:] = 0.0
            
        # Update step
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
