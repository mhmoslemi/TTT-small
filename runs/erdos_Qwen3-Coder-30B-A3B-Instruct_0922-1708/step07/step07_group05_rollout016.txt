import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_max_shift_gradient(h, dx, epsilon=1e-6):
    n_points = len(h)
    
    # Find the shift that gives the maximum overlap
    corr = correlate(h, 1 - h, mode='full')
    max_shift = np.argmax(corr)
    
    # Compute gradient for this particular shift
    grad = np.zeros(n_points)
    for i in range(n_points):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += epsilon
        h_minus[i] -= epsilon
        h_plus = repair_h(h_plus, n_points)
        h_minus = repair_h(h_minus, n_points)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        
        grad[i] = (corr_plus[max_shift] - corr_minus[max_shift]) / (2 * epsilon)
    
    return grad

    else:

        # Simple staircase-like function that satisfies constraints
        mid = n_points // 2

    learning_rate = 0.01
    momentum = 0.9

    patience = 200
    last_improvement = 0

        if time.time() - start_time > budget_s - 5:

        # Compute gradient for the max shift
        grad = compute_max_shift_gradient(h_current, dx)
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h_new = h_current + velocity
        h_new = repair_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        if c5_new < c5_current:

        if i - last_improvement > patience:
