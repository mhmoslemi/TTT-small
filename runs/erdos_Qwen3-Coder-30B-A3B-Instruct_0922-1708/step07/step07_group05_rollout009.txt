import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, population_size):
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Fill remaining slots with random triangular pulses
    for i in range(population_size - len(population)):
        h = np.zeros(n_points)
        # Place a triangular pulse somewhere in the interval
        start = np.random.randint(0, n_points // 2)
        end = np.random.randint(n_points // 2, n_points)
        peak_pos = np.random.randint(start, end)
        peak_height = np.random.uniform(0.5, 1.0)
        
        for j in range(start, peak_pos):
            h[j] = peak_height * (j - start) / (peak_pos - start)
        for j in range(peak_pos, end):
            h[j] = peak_height * (end - j) / (end - peak_pos)
        
        h = repair_h(h, n_points)
        population.append(h)
    
    return population

def crossover(parent1, parent2):
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n_points = len(h)
    num_mutations = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, num_mutations, replace=False)
    for idx in indices:
        h[idx] += np.random.normal(0, 0.05)
    return h

def genetic_algorithm(seed, budget_s, n_points=100):
    np.random.seed(seed)
    start_time = time.time()
    
    # GA parameters
    population_size = 50
    generations = 1000
    elite_size = 1
    mutation_rate = 0.2
    crossover_rate = 0.5
    
    # Initialize population
    population = initialize_population(n_points, population_size)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_h(h, 2.0 / n_points)
        fitness.append(fit)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    # Tracking for early stopping
    last_improvement = 0
    patience = 200
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort population by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism: keep best individuals
        new_population = sorted_population[:elite_size]
        new_fitness = sorted_fitness[:elite_size]
        
        # Select parents (tournament selection)
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 3
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [sorted_fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            if np.random.rand() < crossover_rate and len(new_population) + 1 < population_size:
                # Select second parent
                while True:
                    second_idx = np.random.choice(population_size)
                    if second_idx != winner_idx:
                        break
                parent1 = sorted_population[winner_idx]
                parent2 = sorted_population[second_idx]
                child1, child2 = crossover(parent1, parent2)
                child1 = repair_h(mutate(child1, mutation_rate), n_points)
                child2 = repair_h(mutate(child2, mutation_rate), n_points)
                new_population.extend([child1, child2])
                new_fitness.extend([evaluate_h(child1, 2.0 / n_points),
                                    evaluate_h(child2, 2.0 / n_points)])
            else:
                # Just mutate and repair
                parent = sorted_population[winner_idx]
                child = repair_h(mutate(parent.copy(), mutation_rate), n_points)
                new_population.append(child)
                new_fitness.append(evaluate_h(child, 2.0 / n_points))
        
        population = new_population[:population_size]
        fitness = new_fitness[:population_size]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_fitness:
            best_h = population[min_fitness_idx].copy()
            best_fitness = fitness[min_fitness_idx]
            last_improvement = gen
        else:
            if gen - last_improvement > patience:
                break
    
    return best_h, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    h_best, c5_best = genetic_algorithm(seed, budget_s)
    n_points = len(h_best)
    return (h_best.tolist(), c5_best, n_points)
