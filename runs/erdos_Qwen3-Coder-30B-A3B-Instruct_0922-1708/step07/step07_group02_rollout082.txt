import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate a solution with adaptive noise and flips."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Population-based evolution strategy for better exploration
    population_size = 50
    mutation_scale = 0.1
    max_generations = 1000
    
    # Initialize the population
    population = []
    fitnesses = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
        population.append(h_init)
        fitnesses.append(evaluate_h(h_init, dx))
    else:
        # Create structured initial population
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
        population.append(h_init)
        fitnesses.append(evaluate_h(h_init, dx))
    
    # Add more diverse initial solutions
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        population.append(h)
        fitnesses.append(evaluate_h(h, dx))
    
    # Best individual tracking
    best_fitness = min(fitnesses)
    best_individual = population[fitnesses.index(best_fitness)].copy()
    
    # Evolution loop
    generation = 0
    stagnation_count = 0
    last_improvement = 0
    
    while generation < max_generations and (time.time() - start_time) < budget_s - 5:
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        top_indices = sorted_indices[:population_size // 5]
        
        # Selection
        selected = [population[i] for i in top_indices]
        selected_fitness = [fitnesses[i] for i in top_indices]
        
        # Generate new population
        new_population = selected[:]
        new_fitnesses = selected_fitness[:]
        
        # Mutate and create children
        while len(new_population) < population_size:
            parent = selected[np.random.randint(len(selected))]
            child = mutate(parent, generation, mutation_scale)
            c5_child = evaluate_h(child, dx)
            new_population.append(child)
            new_fitnesses.append(c5_child)
        
        # Update population
        population = new_population
        fitnesses = new_fitnesses
        
        # Update best
        current_best_fitness = min(fitnesses)
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[fitnesses.index(best_fitness)].copy()
            last_improvement = generation
            
        # Mutation adaptation
        if generation - last_improvement > 100:
            mutation_scale *= 0.9
            if mutation_scale < 1e-6:
                # Inject random solution to break stagnation
                h = np.random.rand(n_points)
                h = project_h(h, n_points, dx)
                population[-1] = h
                fitnesses[-1] = evaluate_h(h, dx)
                last_improvement = generation
        
        generation += 1
        # Adaptive mutation decay
        mutation_scale *= 0.999
    
    return (best_individual.tolist(), best_fitness, n_points)
