import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(theta, n_points):
    """Parameterize h from theta"""
    K = len(theta) // 2
    heights = theta[:K]
    boundaries = np.concatenate([[0], theta[K:]])
    boundaries = np.sort(np.clip(boundaries, 0, 2))
    boundaries = np.concatenate([boundaries, [2]])
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(K):
        start = int(boundaries[i] / dx)
        end = int(boundaries[i+1] / dx)
        h[start:end] = heights[i]
    
    return h

def unparameterize_h(h_vec, n_points):
    """Convert h back to parameterized form"""
    K = 5  # Fixed number of segments
    dx = 2.0 / n_points
    heights = []
    boundaries = []
    
    # Simple uniform partition
    segment_width = 2.0 / K
    for i in range(K):
        start = int(i * segment_width / dx)
        end = int((i+1) * segment_width / dx)
        avg_h = np.mean(h_vec[start:end])
        heights.append(avg_h)
        boundaries.append((i+1) * segment_width)
    
    return np.array(heights + boundaries)

def compute_gradient_parametrized(theta, n_points, dx):
    """Compute gradient with respect to parameters"""
    h = parameterize_h(theta, n_points)
    h = repair_h(h, n_points)
    
    # Identify max k
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n_points - 1)
    
    # Compute gradient using finite differences
    eps = 1e-6
    grad = np.zeros_like(theta)
    for i in range(len(theta)):
        theta_plus = theta.copy()
        theta_plus[i] += eps
        h_plus = parameterize_h(theta_plus, n_points)
        h_plus = repair_h(h_plus, n_points)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    # Normalize gradient
    norm = np.linalg.norm(grad) + 1e-12
    grad = grad / norm
    
    return grad

    dx = 2.0 / n_points
    start_time = time.time()

        if len(initial_h) != n_points:

            initial_h = np.interp(x_target, x_orig, initial_h)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_theta = unparameterize_h(initial_h, n_points)
    else:
        # Start with a uniform distribution
        initial_theta = np.random.rand(10)  # 5 heights + 5 boundaries
        # Ensure boundaries are within [0,2] and in order
        initial_theta[5:] = np.sort(np.clip(initial_theta[5:], 0, 2))
    
    # Standard parameterization with 5 segments
    K = 5
    current_theta = initial_theta.copy()
    best_theta = current_theta.copy()
    
    # Optimization parameters
    max_iter = 5000
    lr = 1e-3
    decay_rate = 0.99
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Compute the current h from parameters
        h_current = parameterize_h(current_theta, n_points)
        h_current = repair_h(h_current, n_points)
        current_c5 = evaluate_h(h_current, dx)
        
        # Update best solution
        if i == 0 or current_c5 < evaluate_h(parameterize_h(best_theta, n_points), dx):
            best_theta = current_theta.copy()
        
        # Compute gradient
        try:
            grad = compute_gradient_parametrized(current_theta, n_points, dx)
        except:
            # fallback if gradient fails
            grad = np.random.randn(len(current_theta)) * 1e-5
        
        # Update parameters
        current_theta = current_theta - lr * grad
        # Repair parameters to maintain constraints
        heights = current_theta[:K]
        boundaries = current_theta[K:]
        heights = np.clip(heights, 0, 1)
        boundaries = np.sort(np.clip(boundaries, 0, 2))
        current_theta = np.concatenate([heights, boundaries])
        
        # Adaptive learning rate
        if i % 100 == 0:
            lr *= decay_rate
            
    # Final refinement with gradient descent
    final_h = parameterize_h(best_theta, n_points)
    final_h = repair_h(final_h, n_points)
    final_c5 = evaluate_h(final_h, dx)
    
    # Run final optimization with low learning rate
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = compute_gradient_parametrized(unparameterize_h(final_h, n_points), n_points, dx)
        new_theta = unparameterize_h(final_h, n_points) - 1e-5 * grad

        new_h = parameterize_h(new_theta, n_points)
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        if new_c5 < final_c5:
            final_c5 = new_c5
            final_h = new_h
            
    return (final_h.tolist(), final_c5, n_points)
