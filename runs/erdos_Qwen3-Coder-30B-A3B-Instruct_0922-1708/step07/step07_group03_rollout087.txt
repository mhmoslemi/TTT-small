import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Binary initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        h = h * (n_points / 2) / np.sum(h)

    # Initialize best solution
    best_h = h.copy()
    corr = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Annealing parameters
    temp = 0.1
    cooling_rate = 0.9995
    min_temp = 1e-6
    
    # Main optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by perturbing two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.05, 0.05)
        
        # Apply perturbation
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clamp values to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet constraint
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h_new = h_new * target_sum / h_sum
        
        # Compute new C5
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept or reject based on Metropolis criterion
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Decrease temperature
        temp = max(temp * cooling_rate, min_temp)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
