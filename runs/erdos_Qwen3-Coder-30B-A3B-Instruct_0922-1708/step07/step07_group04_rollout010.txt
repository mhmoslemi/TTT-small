import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutation(h, rate=0.1, strength=0.1):
    """Apply mutation to a solution."""
    h_new = h.copy()
    n_points = len(h)
    mask = np.random.rand(n_points) < rate
    h_new[mask] += np.random.normal(0, strength, np.sum(mask))
    return repair_h(h_new, n_points)

def crossover(parent1, parent2):
    """Perform crossover between two parents."""
    n_points = len(parent1)
    cut_point = np.random.randint(1, n_points)
    child = np.concatenate([parent1[:cut_point], parent2[cut_point:]])
    return child

def tournament_selection(population, fitnesses, tourn_size=3):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), tourn_size)
    best_idx = indices[np.argmax([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # GA settings
    pop_size = 100
    generations = 100
    elite_size = 10
    mutation_rate = 0.1
    mutation_strength = 0.1
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        population.append(mutation(h_init, mutation_rate, mutation_strength))
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate population
        fitnesses = [-evaluate_h(ind, dx) for ind in population]
        
        # Track best
        best_idx = np.argmin(fitnesses)
        if -fitnesses[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = -fitnesses[best_idx]
        
        # Selection
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutation(child, mutation_rate, mutation_strength)
            new_population.append(child)
            
        population = new_population
    
    # Final gradient refinement of best solution
    h_refined = best_h.copy()
    eta = 0.001
    eps = 1e-6
    max_iter = 10000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_refined.copy()
        for j in range(n_points):
            # Finite difference approximation
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_refined.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            h_new[j] = h_refined[j] - eta * grad_j
            
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < evaluate_h(h_refined, dx):
            h_refined = h_new
            best_c5 = c5_new
        else:
            eta *= 0.95
            if eta < 1e-6:
                break
    
    return (h_refined.tolist(), best_c5, n_points)
