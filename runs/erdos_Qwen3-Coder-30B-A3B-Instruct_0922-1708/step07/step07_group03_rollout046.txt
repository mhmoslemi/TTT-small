import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimizer using convex relaxation
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use a more sophisticated approach with CVXPY
    # Set up the optimization
    h_var = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints: h in [0, 1]
    constraints = [h_var >= 0, h_var <= 1]
    
    # Constraint: sum(h) = n_points/2
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Add quadratic constraints for each lag
    max_lag = n_points - 1
    for lag in range(max_lag):
        # We compute the correlation for this specific lag
        # The correlation value is integral of h(x) * (1 - h(x + lag))
        # For discrete case: sum(h[i] * (1 - h[i + lag])) for valid indices
        # This is quadratic constraint in h_var
        corr_expr = 0
        for i in range(n_points - lag):
            corr_expr += h_var[i] * (1 - h_var[i + lag])
        corr_expr *= dx
        
        # Add constraint that this is <= C5
        constraints.append(corr_expr <= C5)
    
    # Objective: minimize C5
    objective = cp.Minimize(C5)
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers_to_try = [cp.MOSEK, cp.GUROBI, cp.ECOS]
    solver_used = None
    for solver in solvers_to_try:
        try:
            if time.time() - start_time > budget_s - 10:
                break
            prob.solve(solver=solver, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                solver_used = solver
                break
        except:
            continue
    
    # If successful, get the solution
    if solver_used and prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = h_var.value
        c5_sol = C5.value
        
        # Ensure bounded and rescale
        h_sol = np.clip(h_sol, 0, 1)
        h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
        
        # Final check
        corr = correlate(h_sol, 1 - h_sol, mode='full')
        final_c5 = np.max(corr) * dx
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_sol.copy()
    else:
        # Use gradient descent if CVXPY fails - fallback
        # Parameters
        learning_rate = 1e-3
        max_iter = 30000
        best_c5 = float('inf')
        best_h = h.copy()
        
        for iteration in range(max_iter):
            if (time.time() - start_time) > budget_s - 5:
                break

            # Compute correlation
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Estimate gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-6
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                if h_plus[i] > 1:
                    h_plus[i] = 1
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps

            # Update step
            h = h - learning_rate * grad
            
            # Projection to [0,1]
            h = np.clip(h, 0, 1)
            
            # Rescale to meet constraint
            h_sum = np.sum(h)
            target_sum = n_points / 2
            if abs(h_sum - target_sum) > 1e-8:
                h = h * target_sum / h_sum
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
