import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate a solution with adaptive strategy."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip with 30% probability
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    return mutated

    dx = 2.0 / n_points
    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Evolution Strategy parameters
    population_size = 50
    elite_size = max(1, population_size // 5)  # Top 20% are elite
    
    # Initialize population
    population = [h_init.copy()]
    fitness = [evaluate_h(h_init, dx)]
    
    # Add more diverse initial solutions
    for i in range(population_size - 1):
        h = h_init.copy()
        # Add structured mutations
        if i < 10:
            # Random segments
            segments = np.random.randint(1, 5)
            for _ in range(segments):
                start = np.random.randint(n_points)
                end = min(start + np.random.randint(5, 20), n_points)
                h[start:end] = 1 - h[start:end]
        else:
            # Triangle pattern
            mid = n_points // 2
            h[:mid] = np.linspace(0, 1, mid)
            h[mid:] = np.linspace(1, 0, n_points - mid)
            
        h = project_h(h, n_points, dx)
        population.append(h)
        fitness.append(evaluate_h(h, dx))
    
    # Best solution tracking
    best_idx = np.argmin(fitness)
    h_best = population[best_idx].copy()
    c5_best = fitness[best_idx]
    
    # Evolution loop
    max_generations = 1000
    mutation_scale = 0.1
    stagnation_counter = 0
    last_improvement = 0
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all population
        current_fitness = []
        for h in population:
            f = evaluate_h(h, dx)
            current_fitness.append(f)

        elite_indices = np.argsort(current_fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        elite_fitness = [current_fitness[i] for i in elite_indices]

        new_best_idx = np.argmin(current_fitness)

            stagnation_counter = 0

            stagnation_counter = 0

                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                parents.append(elite[winner_idx])
            
            # Crossover: averaging
            if len(parents) >= 2:
                child = (parents[0] + parents[1]) / 2
            else:
                child = parents[0].copy()
            
            # Mutation
            child = mutate(child, gen, mutation_scale)
            
            new_population.append(child)
        
        population = new_population[:population_size]
        fitness = [evaluate_h(h, dx) for h in population]
    
    return (h_best.tolist(), c5_best, n_points)
