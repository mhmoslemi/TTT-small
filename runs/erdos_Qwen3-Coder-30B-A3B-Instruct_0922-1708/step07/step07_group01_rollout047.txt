import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Binary search parameters
    c_low = 0.0
    c_high = 0.5
    tolerance = 1e-5
    max_iterations = 50
    
    # Adaptive initialization
    h_current = h_init.copy()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Create and solve optimization problem
        h = cp.Variable(n_points)
        constraints = []
        
        # Feasibility constraints for this C
        for k in range(n_points):
            # Calculate the convolution sum for overlap constraint
            left_bound = max(0, k)
            right_bound = min(n_points, n_points - k)
            if right_bound > left_bound:
                # Sum over valid indices
                overlap_sum = cp.sum(cp.multiply(h[left_bound:right_bound], 
                                                1 - h[left_bound+k:right_bound+k]))
                constraints.append(overlap_sum <= c_mid * dx * (right_bound - left_bound))
        
        # Constraint for integral = 1
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Bounds
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        
        # Formulate and solve the problem
        objective = cp.Minimize(0)  # Dummy objective for feasibility
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h.value
                h_solution = project_h(h_solution, n_points)
                c5_test = evaluate_h(h_solution, dx)
                if c5_test < c5_best:
                    h_best = h_solution.copy()
                    c5_best = c5_test
                c_high = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid

    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        
        h_new = h_best.copy()
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new

    return (h_best.tolist(), c5_best, n_points)
