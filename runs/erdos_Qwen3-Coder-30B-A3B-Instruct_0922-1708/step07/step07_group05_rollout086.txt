import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Create a structured initial construction: triangular pulse
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)

    # PSO parameters
    n_particles = 100
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    max_iter = 400
    patience = 100
    
    # Initialize particles and velocities
    particles = np.array([h_init + np.random.normal(0, 0.01, n_points) for _ in range(n_particles)])
    particles = np.array([repair_h(p, n_points) for p in particles])
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    # Track personal bests
    pbest = particles.copy()
    pbest_values = np.array([evaluate_h(p, dx) for p in particles])
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            current_value = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
                    last_improvement = iteration
        
        # Mutation step
        if iteration % 25 == 0 and iteration > 0:
            for i in range(int(n_particles * mutation_rate)):
                idx = np.random.randint(n_particles)
                mutate_indices = np.random.choice(n_points, int(n_points * 0.2), replace=False)
                particles[idx][mutate_indices] += np.random.normal(0, mutation_sigma, len(mutate_indices))
                particles[idx] = repair_h(particles[idx], n_points)
                
                # Re-evaluate if necessary
                current_value = evaluate_h(particles[idx], dx)
                if current_value < pbest_values[idx]:
                    pbest[idx] = particles[idx].copy()
                    pbest_values[idx] = current_value
                    if current_value < gbest_value:
                        gbest = particles[idx].copy()
                        gbest_value = current_value
                        last_improvement = iteration
                        
        # Early termination if no improvement
        if iteration - last_improvement > patience:
            break
    
    # Final evaluation of best solution
    final_c5 = evaluate_h(gbest, dx)
    
    return (gbest.tolist(), final_c5, n_points)
