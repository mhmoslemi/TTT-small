import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # PSO parameters
    m = 100  # number of particles
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    v_max = 0.2
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # One particle from initial construction
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    pbest.append(h_init.copy())
    pbest_vals.append(evaluate_h(h_init, dx))
    
    # Other particles randomly initialized
    for _ in range(m - 1):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_h(h, dx))
    
    # Global best
    best_idx = np.argmin(pbest_vals)
    gbest = pbest[best_idx].copy()
    gbest_val = pbest_vals[best_idx]
    
    # PSO loop
    max_iter = 10000
    patience = 1000
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * i / max_iter
        
        for p in range(m):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[p] = (w * velocities[p] 
                           + c1 * r1 * (pbest[p] - particles[p])
                           + c2 * r2 * (gbest - particles[p]))
            
            # Clip velocity
            velocities[p] = np.clip(velocities[p], -v_max, v_max)
            
            # Update position
            particles[p] += velocities[p]
            
            # Repair position
            particles[p] = repair_h(particles[p], n_points)
            
            # Evaluate
            current_val = evaluate_h(particles[p], dx)
            
            # Update particle best
            if current_val < pbest_vals[p]:
                pbest[p] = particles[p].copy()
                pbest_vals[p] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = particles[p].copy()
                    gbest_val = current_val
                    last_improvement = i
                    
        if i - last_improvement > patience:
            break
    
    # Gradient descent refinement on gbest
    h_current = gbest.copy()
    c5_current = gbest_val
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient descent parameters
    eta = 0.001
    max_iter = 500
    eps = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_current.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_current[j] - eta * grad_j
            
        # Apply repair step
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if better
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            
            # Update best if improved
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
    
    return (h_best.tolist(), c5_best, n_points)
