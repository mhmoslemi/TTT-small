import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    # Define variables
    h_var = cp.Variable(n_points)
    C5_var = cp.Variable()

    # Constraints
    constraints = []
    
    # Box constraints
    constraints += [h_var >= 0, h_var <= 1]
    
    # Sum constraint
    constraints += [cp.sum(h_var) == n_points / 2]
    
    # Overlap constraints
    h_full = cp.hstack([h_var, np.zeros(n_points - 1)])
    h_flip_full = cp.hstack([np.zeros(n_points - 1), 1 - h_var])
    
    # Compute correlations with proper indexing
    for k in range(2 * n_points - 1):
        if k < n_points:
            corr_val = cp.sum(cp.multiply(h_full[k:k+n_points], h_flip_full[0:n_points]))
        else:
            corr_val = cp.sum(cp.multiply(h_full[k:n_points*2-1], h_flip_full[0:n_points*2-1-k]))
        constraints += [corr_val <= C5_var]
    
    # Objective
    objective = cp.Minimize(C5_var)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with different solvers
    solvers = ['MOSEK', 'GUROBI', 'ECOS', 'SCIP']
    solution_found = False
    
    for solver in solvers:
        if (time.time() - start_time) > budget_s - 10:
            break
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                solution_found = True
                break
        except Exception:
            continue
    
    # If solved successfully, extract solution
    if solution_found:
        h_result = np.array(h_var.value).flatten()
        best_c5 = C5_var.value
    else:
        # Fallback to gradient descent approach
        # Simple gradient descent with projection
        learning_rate = 1e-4
        max_iter = 100000
        best_c5 = float('inf')
        h_current = h.copy()
        
        for iteration in range(max_iter):
            if (time.time() - start_time) > budget_s - 5:
                break

            # Compute correlation
            corr = correlate(h_current, 1 - h_current, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_current.copy()
            
            # Estimate gradient using finite differences
            grad = np.zeros_like(h_current)
            eps = 1e-6
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                if h_plus[i] > 1:
                    h_plus[i] = 1
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps

            # Update step
            h_current = h_current - learning_rate * grad
            
            # Projection to [0,1]
            h_current = np.clip(h_current, 0, 1)
            
            # Rescale to meet constraint
            h_sum = np.sum(h_current)
            target_sum = n_points / 2
            if abs(h_sum - target_sum) > 1e-8:
                h_current = h_current * target_sum / h_sum
                
        best_h = best_h if 'best_h' in locals() else h_current
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
