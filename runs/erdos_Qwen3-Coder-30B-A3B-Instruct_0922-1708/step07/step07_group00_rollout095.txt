import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    # Use a reduced parameterization approach
    # Represent h as piecewise constant with fewer segments: 5 segments
    n_segments = 5
    # Initial parameters: positions and heights
    # Positions are between 0 and 2
    # Heights are between 0 and 1
    params = np.random.rand(n_segments * 2)  # 5 positions + 5 heights
    params[:n_segments] = np.sort(params[:n_segments])  # ensure increasing order
    params[n_segments:] = np.clip(params[n_segments:], 0, 1)  # ensure height constraint
    
    # Map parameters to h(x)
    def params_to_h(params):
        positions = np.concatenate([[0], params[:n_segments], [2]])
        heights = params[n_segments:]
        h = np.zeros(n_points)
        x_vals = np.linspace(0, 2, n_points)
        for i in range(n_segments):
            mask = (x_vals >= positions[i]) & (x_vals < positions[i+1])
            h[mask] = heights[i]
        return h
    
    # Optimizer

        grad = np.zeros_like(current_params)
        for j in range(len(current_params)):
            params_plus = current_params.copy()
            params_plus[j] += eps
            h_plus = params_to_h(params_plus)
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            
            params_minus = current_params.copy()
            params_minus[j] -= eps
            h_minus = params_to_h(params_minus)
            h_minus = repair_h(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)

            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
