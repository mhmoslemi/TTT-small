import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Multiple initializations
    initializations = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        initializations.append(project_h(h_init, n_points))
    else:
        # Initialize with structured waveforms
        x = np.linspace(0, 2, n_points)
        # Cosine taper
        h = 0.5 * (1 + np.cos(np.pi * x))
        initializations.append(project_h(h, n_points))
        
        # Sine wave
        h = 0.5 * (1 + np.sin(np.pi * x))
        initializations.append(project_h(h, n_points))
        
        # Step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        initializations.append(project_h(h, n_points))
        
        # Random perturbation of uniform distribution
        h = np.random.rand(n_points) * 0.5
        initializations.append(project_h(h, n_points))

    h_best = None
    c5_best = float('inf')
    
    # For each initialization, run optimization
    for i, h_start in enumerate(initializations):
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            
        # Run optimization for this initialization
        max_iter = 10000
        patience = 500
        patience_counter = 0
        learning_rate = 0.01
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Find which shift gives the maximum overlap
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr)
            max_shift = max_shift_idx - (n_points - 1)
            
            # Compute gradient with respect to that shift
            # Gradient is (1 - h_shifted_max_shift) 
            h_shifted = np.roll(h_current, max_shift)
            gradient = 1 - h_shifted
            
            # Update h
            h_new = h_current - learning_rate * gradient
            h_new = project_h(h_new, n_points)
            
            # Evaluate
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                patience_counter = 0
            else:
                patience_counter += 1
                
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
            
            if patience_counter > patience:
                learning_rate *= 0.9
                patience_counter = 0
                # Add some noise to escape local minima
                noise = np.random.normal(0, 0.001, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
    
    # Final refinement with local search
    if h_best is not None:
        h_final = h_best.copy()
        c5_final = evaluate_h(h_final, dx)
        
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_new = h_final.copy()
            # Perturb small portion of coordinates
            indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.005)
            
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_final:
                h_final = h_new
                c5_final = c5_new
                
        h_best = h_final
        c5_best = c5_final
    
    return (h_best.tolist(), c5_best, n_points)
