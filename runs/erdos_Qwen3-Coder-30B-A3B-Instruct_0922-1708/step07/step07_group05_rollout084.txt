import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size):
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init.copy())
    
    # Fill the rest with triangular pulses
    for i in range(pop_size - len(population)):
        h = np.zeros(n_points)
        peak_pos = np.random.uniform(0.5, 1.5)
        width = np.random.uniform(0.1, 0.8)
        for j in range(n_points):
            x = j * 2.0 / n_points
            if abs(x - peak_pos) < width / 2:
                h[j] = 1 - 2 * abs(x - peak_pos) / width
        h = repair_h(h, n_points)
        population.append(h.copy())
    
    return population

def crossover(parent1, parent2):
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n_points = len(h)
    num_mutations = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, num_mutations, replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.05)
    return h

def genetic_algorithm(n_points, pop_size, max_generations, mutation_rate, budget_s):
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    fitness_scores = [evaluate_h(h, 2.0/n_points) for h in population]
    
    best_h = population[np.argmin(fitness_scores)].copy()
    best_fitness = min(fitness_scores)
    
    generation = 0
    no_improvement_count = 0
    best_fitness_history = [best_fitness]
    
    while generation < max_generations and time.time() - start_time < budget_s - 5:
        # Evaluate current population
        fitness_scores = [evaluate_h(h, 2.0/n_points) for h in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitness_scores)
        if fitness_scores[current_best_idx] < best_fitness:
            best_fitness = fitness_scores[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 200:
            break
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        elite = population[sorted_indices[0]].copy()
        selected = [population[i] for i in sorted_indices[:pop_size//2]]
        
        # Create new population through crossover and mutation
        new_population = [elite]  # Elitism
        
        # Crossover
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                child1, child2 = crossover(selected[i], selected[i+1])
                new_population.extend([child1, child2])
            else:
                new_population.append(selected[i])
        
        # Mutation
        for i in range(1, len(new_population)):
            new_population[i] = mutate(new_population[i], mutation_rate)
            new_population[i] = repair_h(new_population[i], n_points)
        
        # Trim to population size
        population = new_population[:pop_size]
        
        generation += 1
        best_fitness_history.append(best_fitness)
    
    return best_h, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Parameters for genetic algorithm
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.2
    
    best_h, best_fitness = genetic_algorithm(n_points, pop_size, max_generations, mutation_rate, budget_s)
    
    return (best_h.tolist(), best_fitness, n_points)
