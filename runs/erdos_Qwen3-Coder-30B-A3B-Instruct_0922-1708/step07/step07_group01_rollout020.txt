import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize search bounds
    C_low = 0.0
    C_high = 0.5
    best_C = 1.0
    best_h = h_init.copy()
    
    # Binary search for optimal C
    for iter_no in range(50):
        if time.time() - start_time > budget_s - 5:
            break
            
        C_mid = (C_low + C_high) / 2.0
        
        # Formulate the feasibility problem
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # This computes the sum ∑ h_i * (1 - h_{i+k}) for the current k
            # We need to sum over valid indices for the overlap
            overlap_sum = 0
            for i in range(n_points - k):
                overlap_sum += h_vars[i] * (1 - h_vars[i + k])
            constraints.append(overlap_sum <= C_mid * dx)
            
        # Objective (not used, just to create problem)
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h_vars.value
                h_solution = project_h(h_solution, n_points)
                current_C = evaluate_h(h_solution, dx)
                if current_C < best_C:
                    best_C = current_C
                    best_h = h_solution.copy()
                C_high = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid
    
    return (best_h.tolist(), best_C, n_points)
