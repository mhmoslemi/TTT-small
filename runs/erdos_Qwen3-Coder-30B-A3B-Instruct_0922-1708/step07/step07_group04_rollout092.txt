import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # PSO parameters
    m = 100  # swarm size
    c1 = 2.0
    c2 = 2.0
    w_init = 0.9
    w_min = 0.4
    max_iter = 10000
    patience = 1000
    
    # Initialize PSO swarm
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Initialize particles
    particles = np.random.rand(m, n_points) * 0.5  # Start with small values
    particles[0] = h_init  # Use initial solution if available
    
    # Repair particles to meet sum constraint
    for i in range(m):
        particles[i] = repair_h(particles[i], n_points)
    
    velocities = (np.random.rand(m, n_points) - 0.5) * 0.2
    pbest = particles.copy()
    pbest_values = np.full(m, np.inf)
    gbest_idx = 0
    gbest = particles[gbest_idx].copy()
    gbest_value = np.inf
    
    # PSO main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Evaluate all particles
        for i in range(m):
            h = particles[i]
            current_value = evaluate_h(h, dx)
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest[i] = h.copy()
            
            if current_value < gbest_value:
                gbest_value = current_value
                gbest = h.copy()
                gbest_idx = i
        
        # Update inertia weight
        w = w_init - (w_init - w_min) * (iter_num / max_iter)
        
        # Update velocities and positions
        for i in range(m):
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (gbest - particles[i])
            )
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair position
            particles[i] = repair_h(particles[i], n_points)
            
        if iter_num % 100 == 0:
            if iter_num > 0 and gbest_value == np.inf:
                continue
    
    # Gradient refinement on gbest
    h_refined = gbest.copy()
    c5_refined = gbest_value
    
    # 500 gradient steps
    eta = 0.001
    eps = 1e-6
    
    for step in range(500):
        if time.time() - start_time > budget_s - 2:
            break
            
        h_new = h_refined.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_refined.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_refined[j] - eta * grad_j
            
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_refined:
            h_refined = h_new
            c5_refined = c5_new
            
    return (h_refined.tolist(), c5_refined, n_points)
