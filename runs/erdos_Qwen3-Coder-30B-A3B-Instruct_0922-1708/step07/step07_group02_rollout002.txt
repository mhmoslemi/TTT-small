import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    # PSO parameters
    swarm_size = 50
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_coeff = 1.5
    social_coeff = 1.5
    velocity_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Initialize remaining particles
    for _ in range(swarm_size - 1):
        # Create structured particle
        h = np.zeros(n_points)
        # Random segments
        n_segments = np.random.randint(2, 6)
        segments = np.random.permutation(n_points)[:n_segments]
        segments = np.sort(segments)
        segments = np.append(segments, n_points)
        
        start_idx = 0
        for i, end_idx in enumerate(segments):
            if i % 2 == 0:
                h[start_idx:end_idx] = np.random.uniform(0.3, 0.7)
            else:
                h[start_idx:end_idx] = np.random.uniform(0, 0.3)
            start_idx = end_idx
        
        h = project_h(h, n_points, dx)
        particles.append(h)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
    
    # Personal bests
    pbest = [p.copy() for p in particles]
    pbest_vals = [evaluate_h(p, dx) for p in particles]
    
    # Global best
    best_idx = np.argmin(pbest_vals)
    gbest = pbest[best_idx].copy()
    gbest_val = pbest_vals[best_idx]
    
    # PSO main loop
    max_iter = int(1e6)
    iter_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        inertia = inertia_start - (inertia_start - inertia_end) * (i / max_iter)
        
        # Update each particle
        for j in range(swarm_size):
            particle = particles[j]
            velocity = velocities[j]
            pbest_particle = pbest[j]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocity = (inertia * velocity + 
                       cognitive_coeff * r1 * (pbest_particle - particle) + 
                       social_coeff * r2 * (gbest - particle))
            
            # Clamp velocity
            velocity = np.clip(velocity, -velocity_max, velocity_max)
            
            # Update position
            particle += velocity
            
            # Project to feasible region
            particle = project_h(particle, n_points, dx)
            
            # Evaluate
            fitness = evaluate_h(particle, dx)
            
            # Update personal best
            if fitness < pbest_vals[j]:
                pbest[j] = particle.copy()
                pbest_vals[j] = fitness
                
                # Update global best if needed
                if fitness < gbest_val:
                    gbest = particle.copy()
                    gbest_val = fitness
        
        iter_count += 1
    
    return (gbest.tolist(), gbest_val, n_points)
