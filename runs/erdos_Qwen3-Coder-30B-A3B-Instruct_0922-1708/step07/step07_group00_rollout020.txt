import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(params, n_points):
    """Convert parameters to h(x) with piecewise constant representation."""
    K = len(params) // 2  # Number of segments
    a = params[:K]
    x = np.concatenate([[0], params[K:K+K-1], [2]])
    
    # Ensure x is in increasing order
    x = np.sort(x)
    
    # Create h as piecewise constant
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(K):
        start = int(np.round(x[i] / dx))
        end = int(np.round(x[i+1] / dx))
        if start < end:
            h[start:end] = a[i]
    
    # Ensure sum matches constraint
    h = repair_h(h, n_points)
    return h

def compute_gradient(params, dx, n_points, epsilon=1e-6):
    """Compute gradient using finite differences."""
    grad = np.zeros_like(params)
    current_h = parameterize_h(params, n_points)
    current_c5 = evaluate_h(current_h, dx)
    
    for i in range(len(params)):
        params_plus = params.copy()
        params_plus[i] += epsilon
        h_plus = parameterize_h(params_plus, n_points)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - current_c5) / epsilon
    
    return grad

    # Initial parameterization (piecewise with 5 segments)

    # Optimize with projected gradient descent
    current_params = params_init.copy()

    # Initialize
    current_h = parameterize_h(current_params, n_points)
    current_c5 = evaluate_h(current_h, dx)

    # Optimization parameters
    learning_rate = 1e-3
    max_iter = 50000
    step_count = 0
