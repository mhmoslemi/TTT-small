import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular pulse
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)

    # Initialize PSO parameters
    n_particles = 100
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    patience = 100
    
    # Initialize particles and velocities
    particles = np.array([h_init + np.random.normal(0, 0.01, n_points) for _ in range(n_particles)])
    particles = np.array([repair_h(p, n_points) for p in particles])
    
    velocities = np.zeros_like(particles)
    
    # Initialize personal bests
    pbest = particles.copy()
    pbest_scores = np.array([evaluate_h(p, dx) for p in pbest])
    
    # Find global best
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    last_improvement = 0
    best_h = gbest.copy()
    best_c5 = gbest_score
    
    for iter_num in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
        
        w = w_start - (w_start - w_end) * iter_num / 1000
        
        # Update each particle
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
            
            # Update global best
            if score < gbest_score:
                gbest = particles[i].copy()
                gbest_score = score
                last_improvement = iter_num
                if gbest_score < best_c5:
                    best_h = gbest.copy()
                    best_c5 = gbest_score
        
        # Apply mutation every 25 iterations
        if iter_num % 25 == 0 and iter_num > 0:
            for i in range(n_particles):
                if np.random.rand() < mutation_rate:
                    indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
                    particles[i][indices] += np.random.normal(0, mutation_sigma, len(indices))
                    particles[i] = repair_h(particles[i], n_points)
        
        # Early stopping
        if iter_num - last_improvement > patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
