import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        h_init = 0.5 * (1 + np.cos(np.pi * x))
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Try different strategies
    strategies = [
        lambda h, i: h.copy(),  # do nothing
        lambda h, i: np.roll(h, 1),  # shift
        lambda h, i: np.roll(h, -1),  # shift anti
        lambda h, i: np.flip(h),  # flip
    ]
    
    # Use a stochastic approach with different perturbations
    for iter_num in range(50000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Randomly perturb
        h_new = h_current.copy()
        # Randomly choose a number of positions to perturb
        n_perturb = np.random.randint(1, max(1, n_points // 10))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for idx in indices:
            # Small random change
            h_new[idx] += np.random.normal(0, 0.01)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        elif np.random.rand() < 0.01:  # Sometimes accept worse
            h_current = h_new.copy()
            c5_current = c5_new
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
    
    # Final local optimization using coordinate descent
    for _ in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        for i in range(n_points):
            best_h_i = h_new[i]
            best_c5 = c5_best
            
            # Try different small changes
            for delta in [-0.02, -0.01, 0, 0.01, 0.02]:
                h_test = h_new.copy()
                h_test[i] += delta
                h_test = project_h(h_test, n_points)
                c5_test = evaluate_h(h_test, dx)
                
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h_i = h_test[i]
            
            h_new[i] = best_h_i
            if best_c5 < c5_best:
                c5_best = best_c5
        
        h_best = h_new.copy()
    
    return (h_best.tolist(), c5_best, n_points)
