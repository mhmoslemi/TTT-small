import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with either initial_h_values or structured waveform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Use structured waveform as starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Binary search on C
    C_low = 0.0
    C_high = 0.5
    best_C = C_high
    best_h = h_init.copy()
    
    # Track time and max iterations
    max_iterations = 50
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        C_mid = (C_low + C_high) / 2.0
        
        # Formulate and solve feasibility problem
        h_vars = cp.Variable(n_points)
        constraints = []
        
        # Bound constraints
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Constraint: sum h = 1
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Overlap constraints: ∫ h(x) (1 - h(x+k)) dx <= C * dx for all k
        # We will compute full convolution
        # Note: correlation of h with 1-h
        for k in range(n_points):
            # Compute the constraint for this k
            # This is the integral of h(x)(1 - h(x+k)) for all x
            # We compute using the fact that we want max of correlations
            # In the full convolution, we have size 2*n_points - 1 terms
            # But since we want to avoid directly computing the full correlation,
            # let's just do constrained optimization via CVXPY with approximate bounds
            
        # Approximate feasibility check: 
        h_test = h_init.copy()
        corr = correlate(h_test, 1 - h_test, mode='full')
        max_corr = np.max(corr)
        if max_corr * dx <= C_mid:
            # Feasible
            C_high = C_mid
            best_C = C_mid
            best_h = h_test.copy()
        else:
            # Infeasible
            C_low = C_mid
            
    # Postprocess and refine optimal solution
    refined_h = best_h.copy()
    current_C = evaluate_h(refined_h, dx)
    
    # Local search to improve result
    for iter_count in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
        new_h = refined_h.copy()
        idx = np.random.randint(0, n_points)
        # Small random perturbation
        pert = np.random.normal(0, 0.01)
        new_h[idx] += pert
        new_h = project_h(new_h, n_points)
        new_C = evaluate_h(new_h, dx)
        if new_C < current_C:
            refined_h = new_h
            current_C = new_C
    
    # Final refinement
    for i in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        new_h = refined_h.copy()
        idx = np.random.randint(0, n_points)
        new_h[idx] += np.random.normal(0, 0.005)
        new_h = project_h(new_h, n_points)
        new_C = evaluate_h(new_h, dx)
        if new_C < current_C:
            refined_h = new_h
            current_C = new_C
    
    return (refined_h.tolist(), current_C, n_points)
