import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize particles
    n_particles = 100
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular construction
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)

    # Create initial particles
    for _ in range(n_particles):
        perturbed = h_init + np.random.normal(0, 0.02, n_points)
        particles.append(repair_h(perturbed, n_points))
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))

    # Initialize best
    best_particle = h_init.copy()
    best_c5 = evaluate_h(best_particle, dx)
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    patience = 100
    last_improvement = 0
    
    for iter_count in range(400):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Adjust inertia weight
        w = w_start - (w_start - w_end) * iter_count / 400
        
        # Evaluate all particles
        particle_scores = []
        for p in particles:
            score = evaluate_h(p, dx)
            particle_scores.append(score)
            
        # Update global best
        min_idx = np.argmin(particle_scores)
        if particle_scores[min_idx] < best_c5:
            best_c5 = particle_scores[min_idx]
            best_particle = particles[min_idx].copy()
            last_improvement = iter_count
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (particles[i] - particles[i])
            social = c2 * r2 * (best_particle - particles[i])
            noise = np.random.normal(0, 0.01, n_points)
            velocities[i] = w * velocities[i] + cognitive + social + noise
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair to constraints
            particles[i] = repair_h(particles[i], n_points)
            
        # Apply mutation every 25 iterations
        if iter_count % 25 == 0:
            for i in range(int(n_particles * mutation_rate)):
                idx = np.random.randint(0, n_particles)
                # Mutate 20% of the values
                n_mutate = int(0.2 * n_points)
                mutate_indices = np.random.choice(n_points, n_mutate, replace=False)
                for j in mutate_indices:
                    particles[idx][j] += np.random.normal(0, mutation_sigma)
                particles[idx] = repair_h(particles[idx], n_points)
                
        # Check for stagnation
        if iter_count - last_improvement > patience:
            break

    return (best_particle.tolist(), best_c5, n_points)
