import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Simulated Annealing with adaptive cooling
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Parameters
    temp = 0.1
    cooling_rate = 0.95
    min_temp = 1e-6
    max_iter = 100000
    patience = 500
    last_improvement = 0
    
    # Track best solution every 60 seconds
    checkpoint_interval = 60
    last_checkpoint = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        if time.time() - last_checkpoint > checkpoint_interval:
            last_checkpoint = time.time()
            # Save best solution so far
            h_best = h_current.copy()
            c5_best = c5_current
        
        # Generate a neighbor by perturbing one element
        h_new = h_current.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.05, 0.05)
        h_new[idx] += delta
        
        # Repair the perturbed solution
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-(c5_new - c5_current) / temp):
                h_current = h_new
                c5_current = c5_new
        
        # Update temperature
        if temp > min_temp:
            temp *= cooling_rate
        
        # Early stopping
        if i - last_improvement > patience:
            break
    
    # Final gradient-based refinement
    h_refined = h_best.copy()
    c5_refined = c5_best
    
    # Try 1000 gradient steps with adaptive learning rate
    eta = 0.01
    for _ in range(1000):

            break

            grad_j = (c5_plus - c5_minus) / (2e-6)
            
            # Update
            h_new[j] = h_refined[j] - eta * grad_j
            
        # Apply repair step to maintain feasibility
        h_new = repair_h(h_new, n_points)

        # Early stopping
        if abs(c5_refined - c5_best) < 1e-8:
            break
            
    return (h_refined.tolist(), c5_refined, n_points)
