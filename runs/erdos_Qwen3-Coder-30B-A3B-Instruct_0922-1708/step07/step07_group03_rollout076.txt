import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    learning_rate = 1e-3
    max_iter = 100000
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap integral
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient using adjoint method
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h) - 1)
        
        # Gradient is part of correlation with (1 - h)
        grad = np.zeros_like(h)
        if max_lag >= 0:
            # h[i] appears in correlation with (1 - h[i + max_lag])
            for i in range(len(h) - max_lag):
                grad[i] += (1 - h[i + max_lag])
        else:
            # max_lag < 0, so h[i + max_lag] appears in correlation with (1 - h[i])
            for i in range(-max_lag, len(h)):
                grad[i] += (1 - h[i + max_lag])
        
        # Update with gradient descent
        h = h - learning_rate * grad
        
        # Project h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to satisfy constraint ∫ h(x) dx = 1
        h_sum = np.sum(h) * dx
        if abs(h_sum) > 1e-10:
            h = h * (n_points / 2) / h_sum
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
