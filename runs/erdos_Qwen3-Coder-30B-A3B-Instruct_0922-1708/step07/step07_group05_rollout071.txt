import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidate(n_points, seed=None):
    if seed is not None:
        np.random.seed(seed)
    # Generate candidate using Fourier modes
    h = np.zeros(n_points)
    # Use 3-5 Fourier modes
    num_modes = np.random.randint(3, 6)
    for i in range(num_modes):
        k = np.random.randint(1, n_points//4)
        a = np.random.uniform(-0.5, 0.5)
        b = np.random.uniform(-0.5, 0.5)
        phase = np.random.uniform(0, 2*np.pi)
        h += a * np.cos(k * np.linspace(0, 2*np.pi, n_points) + phase) + \
             b * np.sin(k * np.linspace(0, 2*np.pi, n_points) + phase)
    
    # Ensure it's in [0,1] and sum = n_points/2
    h = h - h.min()
    h = h / (h.max() - h.min()) if h.max() > h.min() else h
    h = repair_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Generate diverse candidates using Fourier modes
    candidates = []
    num_candidates = 50
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        candidates.append((h_init, evaluate_h(h_init, dx)))
    else:
        # Add a simple piecewise linear construction
        h_simple = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_simple[i] = 0.0
            elif x < 1.5:
                h_simple[i] = (x - 0.5) / 1.0
            else:
                h_simple[i] = 1.0
        h_simple = repair_h(h_simple, n_points)
        candidates.append((h_simple, evaluate_h(h_simple, dx)))
    
    # Generate Fourier candidates
    for i in range(num_candidates):
        candidate = generate_fourier_candidate(n_points, seed=seed+i)
        c5_val = evaluate_h(candidate, dx)
        candidates.append((candidate, c5_val))
    
    # Sort by C5 value
    candidates.sort(key=lambda x: x[1])
    
    # Keep top candidates to refine
    top_candidates = candidates[:10]
    h_best = top_candidates[0][0].copy()
    c5_best = top_candidates[0][1]
    
    # Refinement phase
    patience = 200
    max_refine_iter = 2000
    for h_init, _ in top_candidates:
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        learning_rate = 0.01
        last_improvement = 0
        
        for i in range(max_refine_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradients via finite differences
            grad = np.zeros(n_points)
            epsilon = 1e-6
            for j in range(n_points):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[j] += epsilon
                h_minus[j] -= epsilon
                h_plus = repair_h(h_plus, n_points)
                h_minus = repair_h(h_minus, n_points)
                
                c5_plus = evaluate_h(h_plus, dx)
                c5_minus = evaluate_h(h_minus, dx)
                grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update with gradient descent
            h_new = h_current - learning_rate * grad
            h_new = repair_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
            else:
                if np.random.rand() < np.exp(-(c5_new - c5_current) / (c5_current * 0.01)):
                    h_current = h_new
                    c5_current = c5_new
            
            if i - last_improvement > patience:
                learning_rate *= 0.95
                if learning_rate < 1e-6:
                    break
                
            if time.time() - start_time > budget_s - 5:
                break
    
    return (h_best.tolist(), c5_best, n_points)
