import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, population_size, initial_h=None):
    population = []
    if initial_h is not None:
        population.append(initial_h.copy())
        population[0] = repair_h(population[0], n_points)
    
    # Generate diverse initial candidates
    for i in range(population_size - len(population)):
        # Create triangular pulse
        h = np.zeros(n_points)
        center = np.random.uniform(0.2, 1.8)
        width = np.random.uniform(0.2, 0.6)
        for j in range(n_points):
            x = j * (2.0 / n_points)
            dist = abs(x - center)
            if dist <= width / 2:
                h[j] = 1 - 2 * dist / width
        h = repair_h(h, n_points)
        population.append(h)
    return population

def selection(population, fitness):
    # Tournament selection
    selected = []
    for _ in range(len(population)):
        idx1, idx2 = np.random.choice(len(population), 2, replace=False)
        if fitness[idx1] < fitness[idx2]:
            selected.append(population[idx1].copy())
        else:
            selected.append(population[idx2].copy())
    return selected

def crossover(parent1, parent2):
    # Single-point crossover
    n_points = len(parent1)
    point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n_points = len(h)
    mask = np.random.rand(n_points) < mutation_rate
    noise = np.random.normal(0, 0.1, n_points)
    h = h + noise * mask
    return h

def genetic_algorithm(n_points, population_size=50, generations=1000, 
                      crossover_rate=0.5, mutation_rate=0.2, 
                      initial_h=None, budget_s=600):
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, population_size, initial_h)
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_h(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    last_improvement = 0
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection
        selected = selection(population, fitness)
        
        # Create new population
        new_population = [best_h.copy()]  # Elitism
        
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < crossover_rate and len(new_population) + 1 < population_size:
                idx1, idx2 = np.random.choice(len(selected), 2, replace=False)
                child1, child2 = crossover(selected[idx1], selected[idx2])
                new_population.append(child1)
                new_population.append(child2)
            else:
                # Mutation
                idx = np.random.randint(len(selected))
                mutated = mutate(selected[idx], mutation_rate)
                mutated = repair_h(mutated, n_points)
                new_population.append(mutated)
                
        # Trim to population size
        population = new_population[:population_size]
        
        # Evaluate new population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = gen
                
        # Early termination if no improvement
        if gen - last_improvement > 200:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            x_orig = np.linspace(0, 2, len(initial_h))
            x_target = np.linspace(0, 2, n_points)
            initial_h = np.interp(x_target, x_orig, initial_h)
        initial_h = repair_h(initial_h, n_points)
    else:
        initial_h = None
    
    # Run GA
    h_best, c5_best = genetic_algorithm(n_points, population_size=50, 
                                       generations=1000, initial_h=initial_h,
                                       budget_s=budget_s)
    
    return (h_best.tolist(), c5_best, n_points)
