import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)

    # Global best
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Try a simple gradient descent with better initialization and heuristics
    h_current = h_init.copy()
    c5_current = c5_best

    # Hyperparameters for optimization
    lr = 0.005
    max_iter = 20000
    patience = 1000
    last_improvement = 0
    
    # Optimizer
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient approximation using finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-5
        
        for j in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            # Ensure feasibility
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update step
        h_new = h_current - lr * grad
        h_new = project_h(h_new, n_points, dx)
        
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            lr *= 0.9995
            
        # Early stopping if no improvement
        if i - last_improvement > patience:
            break
            
    return (h_best.tolist(), c5_best, n_points)
