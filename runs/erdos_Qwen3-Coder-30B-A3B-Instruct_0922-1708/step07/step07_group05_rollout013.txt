import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Create a triangular pulse
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)

    # PSO parameters
    n_particles = 100
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    max_iter = 400
    patience = 100
    last_improvement = 0
    
    # Initialize particles
    particles = np.array([h_init + np.random.normal(0, 0.01, n_points) for _ in range(n_particles)])
    particles = np.array([repair_h(p, n_points) for p in particles])
    
    velocities = np.zeros((n_particles, n_points))
    
    # Track best
    pbest = particles.copy()
    pbest_scores = np.array([evaluate_h(p, dx) for p in pbest])
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # Main loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * i / max_iter
        
        # Evaluate all particles
        current_scores = np.array([evaluate_h(p, dx) for p in particles])
        
        # Update personal bests
        better_mask = current_scores < pbest_scores
        pbest[better_mask] = particles[better_mask]
        pbest_scores[better_mask] = current_scores[better_mask]
        
        # Update global best
        new_gbest_idx = np.argmin(pbest_scores)
        if pbest_scores[new_gbest_idx] < gbest_score:
            gbest = pbest[new_gbest_idx].copy()
            gbest_score = pbest_scores[new_gbest_idx]
            last_improvement = i
        
        # Update velocities and positions
        for j in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[j] = (w * velocities[j] + 
                           c1 * r1 * (pbest[j] - particles[j]) + 
                           c2 * r2 * (gbest - particles[j]))
            
            # Apply mutation randomly
            if np.random.rand() < mutation_rate:
                # Mutate 20% of elements
                mutate_idx = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
                velocities[j][mutate_idx] += np.random.normal(0, mutation_sigma, len(mutate_idx))
            
            particles[j] = particles[j] + velocities[j]
            particles[j] = repair_h(particles[j], n_points)
        
        # Check for early termination
        if i - last_improvement > patience:
            break
    
    return (gbest.tolist(), gbest_score, n_points)
