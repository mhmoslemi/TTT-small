import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple starting point
        h = np.ones(n_points) * 0.5
    
    # Frank-Wolfe optimization
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current C5
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient (approximated by finite differences)
        epsilon = 1e-6
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / epsilon
        
        # Find descent direction by solving linear subproblem
        # minimize <grad, s> subject to 0 <= s <= 1, sum(s) = n_points/2
        s = np.zeros(n_points)
        
        # Sort indices by negative gradient
        idx = np.argsort(-grad)
        remaining_weight = n_points / 2
        s_sum = 0
        
        for i in idx:
            if remaining_weight <= 0:
                break
            if s_sum + 1 <= remaining_weight:
                s[i] = 1
                s_sum += 1
            else:
                s[i] = remaining_weight - s_sum
                s_sum = remaining_weight
                break
        
        # Scale s to target sum
        if s_sum > 0:
            s = s * (n_points / 2) / s_sum
        
        # Line search (simple backtracking)
        gamma = 0.5
        max_backtrack = 10
        for _ in range(max_backtrack):
            h_new = (1 - gamma) * h + gamma * s
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if abs(h_new_sum - n_points / 2) < 1e-10:
                break
            gamma *= 0.5
            
        h = h_new.copy()
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
