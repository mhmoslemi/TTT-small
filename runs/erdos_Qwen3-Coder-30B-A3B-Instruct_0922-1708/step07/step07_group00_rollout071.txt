import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(theta, n_points):
    """Convert parameterized representation to h."""
    # theta = [a1, a2, ..., aK, x2, x3, ..., xK]
    K = (len(theta) + 1) // 2
    a = theta[:K]
    x = np.concatenate([[0], theta[K:], [2]])
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(K):
        start_idx = int(x[i] / dx)
        end_idx = int(x[i+1] / dx)
        h[start_idx:end_idx] = a[i]
    
    return h

def repair_parameters(theta):
    """Repair parameters to satisfy constraints."""
    K = (len(theta) + 1) // 2
    a = np.clip(theta[:K], 0, 1)
    x = np.concatenate([[0], np.sort(theta[K:]), [2]])
    
    # Rescale a to meet integral constraint
    dx = 2.0 / 100
    integral = np.sum(a * np.diff(x))
    if abs(integral) > 1e-10:
        a = a / integral * 1.0
    
    return np.concatenate([a, x[1:-1]])

def compute_gradient(theta, n_points, dx):
    """Compute gradient of maximum overlap with respect to parameters."""
    eps = 1e-6
    grad = np.zeros_like(theta)
    h_current = parameterize_h(theta, n_points)
    h_current = repair_h(h_current, n_points)
    corr_current = correlate(h_current, 1 - h_current, mode='full')
    max_k_idx = np.argmax(corr_current)
    
    for i in range(len(theta)):
        theta_plus = theta.copy()
        theta_plus[i] += eps
        theta_plus = repair_parameters(theta_plus)
        h_plus = parameterize_h(theta_plus, n_points)
        h_plus = repair_h(h_plus, n_points)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr_current[max_k_idx]) / eps
    
    # Normalize
    norm = np.linalg.norm(grad) + 1e-12
    return grad / norm

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Use 5 segments to represent h
    K = 5
    # Initialize theta as [a1, ..., aK, x2, ..., xK]
    h_sample = h_init[:K+1]
    x_init = np.linspace(0, 2, K+1)
    a_init = np.array([np.mean(h_sample[i:i+2]) for i in range(K)])
    a_init = np.clip(a_init, 0, 1)
    x_init = x_init[1:-1]
    theta = np.concatenate([a_init, x_init])
    theta = repair_parameters(theta)
    
    # Initialize
    current_theta = theta.copy()
    current_h = parameterize_h(current_theta, n_points)
    current_h = repair_h(current_h, n_points)
    current_c5 = evaluate_h(current_h, dx)
    best_theta = current_theta.copy()
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimization parameters
    max_iter = 5000
    learning_rate = 1e-3
    decay_rate = 0.99
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        grad = compute_gradient(current_theta, n_points, dx)
        
        # Update step
        current_theta = current_theta - learning_rate * grad
        current_theta = repair_parameters(current_theta)
        current_h = parameterize_h(current_theta, n_points)
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_theta = current_theta.copy()
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Adaptive learning rate
        if i % 100 == 0:
            learning_rate *= decay_rate
    
    # Final refinement
    h_refined = best_h.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        grad = compute_gradient(best_theta, n_points, dx)
        best_theta = best_theta - 1e-5 * grad
        best_theta = repair_parameters(best_theta)
        h_refined = parameterize_h(best_theta, n_points)
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        c5_refined = np.max(corr) * dx
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
