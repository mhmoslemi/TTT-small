import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
        
    # Start with initial solution
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Parameters
    max_iter = 50000
    eta = 0.01
    eps = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Random coordinate descent
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        improved = False
        for j in indices:
            # Compute gradient using finite differences
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            # Ensure feasibility after perturbation
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update using gradient
            h_current[j] -= eta * grad_j
            h_current[j] = max(0, min(1, h_current[j]))
            
        # Apply final projection and re-evaluate
        h_current = project_h(h_current, n_points, dx)
        c5_new = evaluate_h(h_current, dx)
        
        if c5_new < c5_current:
            c5_current = c5_new
            improved = True
            
            # Update best if improved
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                
        # Adaptive learning rate
        if not improved:
            eta *= 0.999
        else:
            eta = min(0.01, eta * 1.001)
            
    # Additional local search using gradient descent with different initializations
    for trial in range(5):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Random perturbation
        h_trial = h_best.copy()
        perturb_indices = np.random.choice(n_points, size=max(1, n_points // 30), replace=False)
        for idx in perturb_indices:
            h_trial[idx] += np.random.normal(0, 0.02)
            
        h_trial = project_h(h_trial, n_points, dx)
        c5_trial = evaluate_h(h_trial, dx)
        
        if c5_trial < c5_best:
            h_best = h_trial
            c5_best = c5_trial

    return (h_best.tolist(), c5_best, n_points)
