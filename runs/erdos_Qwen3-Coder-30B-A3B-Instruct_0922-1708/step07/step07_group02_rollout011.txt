import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2, n_points):
    # Segment-wise blending
    idx = np.random.randint(1, n_points)
    child = np.concatenate([parent1[:idx], parent2[idx:]])
    return child

def mutate(individual, n_points, mutation_scale):
    # Random segment flip and Gaussian noise
    child = individual.copy()
    if np.random.rand() < 0.3:
        # Flip a random segment
        start = np.random.randint(0, n_points)
        end = np.random.randint(start + 1, n_points + 1)
        child[start:end] = 1 - child[start:end]
    # Add Gaussian noise
    noise = np.random.normal(0, mutation_scale, n_points)
    child += noise
    # Project to feasible region
    child = project_h(child, n_points, 2.0 / n_points)
    return child

def local_refinement(h, n_points, dx, max_iter=50):
    # Coordinate descent with gradient approximation
    h_current = h.copy()
    c5_current = evaluate_h(h_current, dx)
    
    eta = 0.01
    eps = 1e-6
    
    for _ in range(max_iter):
        h_new = h_current.copy()
        improved = False
        
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
        
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            improved = True
            eta *= 1.01
        else:
            eta *= 0.99
        
        if not improved:
            break
    
    return h_current, c5_current

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    # Memetic algorithm parameters
    population_size = 50
    num_parents = max(5, population_size // 4)
    mutation_scale = 0.1
    max_generations = 1500
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = h_init.copy()
        h = mutate(h, n_points, mutation_scale)
        population.append(h)
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < c5_best:
            h_best = population[0].copy()
            c5_best = fitness[0]
        
        # Select top parents
        parents = population[:num_parents]
        
        # Generate offspring
        offspring = []
        while len(offspring) < population_size - num_parents:
            # Crossover
            p1, p2 = np.random.choice(parents, size=2, replace=False)
            child = crossover(p1, p2, n_points)
            # Mutation
            child = mutate(child, n_points, mutation_scale)
            # Local refinement
            child, _ = local_refinement(child, n_points, dx, 50)
            offspring.append(child)
        
        # Form new population
        population = parents + offspring
        mutation_scale *= 0.999
    
    # Final evaluation of best
    c5_best = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), c5_best, n_points)
