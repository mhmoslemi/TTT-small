import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # CCCP Phase
    for iter_ccp in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find k_max corresponding to maximum overlap
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        
        # Calculate gradient (approximated by indexing)
        # The gradient for the max overlap at shift k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            shift = k_max - (n_points - 1)  # Correct for correlation indexing
            j = i - shift
            if 0 <= j < n_points:
                grad[i] = (1 - h_current[j])
        
        # Update using gradient descent
        learning_rate = 0.01 * (0.95 ** (iter_ccp // 500))
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
        else:
            # Occasionally accept worse moves (simulated annealing)
            if np.random.rand() < 0.01:
                h_current = h_new
                c5_current = c5_new
                
        # Track best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            
        # Periodic perturbations
        if iter_ccp % 500 == 0 and iter_ccp > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Coordinate Descent Phase
    for iter_cd in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic coordinate descent over all indices
        indices = np.arange(n_points)
        np.random.shuffle(indices)
        
        for i in indices:
            # Best shift for current h
            corr = correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr)
            shift = k_max - (n_points - 1)
            
            # Update gradient for this coordinate
            j = i - shift
            if 0 <= j < n_points:
                grad_i = (1 - h_current[j])
            else:
                grad_i = 0.0
            
            # Update h[i] and maintain feasibility
            h_current[i] = h_current[i] - 0.01 * grad_i
            h_current[i] = np.clip(h_current[i], 0, 1)
            
            # Renormalize
            h_current = project_h(h_current, n_points)
            
        # Check if C5 decreased
        c5_new = evaluate_h(h_current, dx)
        if c5_new < c5_best:
            h_best = h_current.copy()
            c5_best = c5_new
            
        # Periodic perturbations
        if iter_cd % 1000 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    return (h_best.tolist(), c5_best, n_points)
