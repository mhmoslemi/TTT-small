import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate a solution."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Population-based evolution
    population_size = 50
    mutation_scale = 0.1
    generation = 0
    stagnation_counter = 0
    best_c5 = float('inf')
    best_h = None
    
    # Initialize population
    population = []
    fitness = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Create a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Start with the initial solution
    population.append(h_init)
    
    # Add random structured mutations
    for i in range(population_size - 1):
        h = h_init.copy()
        # Apply some random variation
        h = mutate(h, 0, 0.1)
        population.append(h)
        
    # Evaluate initial population
    for h in population:
        c5 = evaluate_h(h, dx)
        fitness.append(c5)
    
    # Main evolution loop
    while time.time() - start_time < budget_s - 5:
        generation += 1
        if generation % 50 == 0:
            mutation_scale *= 0.999
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size // 5]  # Top 20%
        
        # Track best
        current_best_idx = sorted_indices[0]
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Evolve new population
        new_population = []
        new_fitness = []
        
        # Add top performers
        for idx in top_indices:
            new_population.append(population[idx])
            new_fitness.append(fitness[idx])
        
        # Generate new candidates
        while len(new_population) < population_size:
            # Select parents
            parent1 = population[np.random.choice(top_indices)]
            if np.random.rand() < 0.7:
                parent2 = population[np.random.choice(top_indices)]
                # Blend
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutate
            child = mutate(child, generation, mutation_scale)
            
            # Evaluate
            c5 = evaluate_h(child, dx)
            
            new_population.append(child)
            new_fitness.append(c5)
        
        # Update population
        population = new_population
        fitness = new_fitness
        
        # Inject diversity if stalled
        if stagnation_counter > 100:
            for i in range(min(5, len(population))):
                if np.random.rand() < 0.3:
                    population[i] = mutate(population[i], generation, 0.05)
                    fitness[i] = evaluate_h(population[i], dx)
            stagnation_counter = 0
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to the best from initial population
        final_idx = np.argmin(fitness)
        final_h = population[final_idx]
        final_c5 = fitness[final_idx]
        return (final_h.tolist(), final_c5, n_points)
