import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Phase 1: Convex optimization with CVXPY
    h = cp.Variable(n_points)
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points / 2
    ]
    
    # Simplified objective: minimize squared correlation
    h_vec = h_init.copy()
    corr = correlate(h_vec, 1 - h_vec, mode='full')
    max_val = np.max(corr)
    # Create a quadratic approximation
    objective = cp.Minimize(cp.sum_squares(h) + 0.01 * cp.sum_squares(h - h_vec))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=5000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
        else:
            h_opt = h_init
    except:
        h_opt = h_init
    
    h_opt = repair_h(h_opt, n_points)
    c5_opt = evaluate_h(h_opt, dx)
    h_best = h_opt.copy()
    c5_best = c5_opt
    
    # Phase 2: Proximal gradient descent refinement
    eta = 0.01
    eps = 1e-6
    max_iter = 2000
    patience = 100
    last_improvement = 0
    
    h_current = h_opt.copy()
    c5_current = c5_opt
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient approximation via finite differences
        h_new = h_current.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with learning rate
            h_new[j] = h_current[j] - eta * grad_j
            
        # Repair to maintain constraints
        h_new = repair_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept update
        h_current = h_new
        c5_current = c5_new
        
        # Store best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Reduce learning rate with patience
        if i - last_improvement > patience:
            eta *= 0.95
            last_improvement = i
    
    return (h_best.tolist(), c5_best, n_points)
