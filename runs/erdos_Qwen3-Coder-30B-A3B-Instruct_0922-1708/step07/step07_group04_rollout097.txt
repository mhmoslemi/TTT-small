import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_basis(x, k):
    """Generate Fourier basis functions."""
    return np.concatenate([np.ones_like(x), np.cos(np.pi * x * np.arange(1, k+1)), np.sin(np.pi * x * np.arange(1, k+1))])

def fourier_to_h(coeffs, x, k):
    """Convert Fourier coefficients to step function."""
    basis = fourier_basis(x, k)
    # Apply sigmoid and ensure [0,1] range
    linear_comb = np.dot(basis, coeffs)
    h = 1.0 / (1.0 + np.exp(-linear_comb))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Initialize with Fourier series approach
    x = np.linspace(0, 2, n_points)
    k = 20  # number of harmonics
    
    # Create Fourier basis matrix
    F = fourier_basis(x, k)
    
    # Convert h_init to coefficients
    h_init_shifted = h_init - 0.5  # Shift for easier fitting
    coeffs_init = np.linalg.lstsq(F, h_init_shifted, rcond=None)[0]
    
    # Define objective in terms of coefficients
    def objective(coeffs):
        h = fourier_to_h(coeffs, x, k)
        # Repair to ensure constraint and bound
        h = repair_h(h, n_points)
        # Ensure norm constraint on integral
        h = h * (n_points/2) / np.sum(h)
        return evaluate_h(h, dx)
    
    # Define constraint for integral
    def integral_constraint(coeffs):
        h = fourier_to_h(coeffs, x, k)
        h = h * (n_points/2) / np.sum(h)
        return np.sum(h) - n_points/2
    
    # Optimize using scipy minimize
    constraints = {'type': 'eq', 'fun': integral_constraint}
    bounds = [(-10, 10)] * len(coeffs_init)  # Reasonable bounds for coefficients
    
    try:
        result = minimize(objective, coeffs_init, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 1000, 'ftol': 1e-6})
        coeffs_opt = result.x
        h_opt = fourier_to_h(coeffs_opt, x, k)
        h_opt = repair_h(h_opt, n_points)
        h_opt = h_opt * (n_points/2) / np.sum(h_opt)
        c5_opt = evaluate_h(h_opt, dx)
    except:
        # Fallback to coordinate descent if optimization fails
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        h_best = h_current.copy()
        c5_best = c5_current

        # Parameters
        eta = 0.01
        eps = 1e-6
        max_iter = 50000
        patience = 100
        last_improvement = 0
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Update each coordinate using gradient approximation
            h_new = h_current.copy()
            for j in range(n_points):
                # Forward difference
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                # Backward difference
                h_minus = h_current.copy()
                h_minus[j] -= eps
                h_minus = repair_h(h_minus, n_points)
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                # Gradient estimate
                grad_j = (c5_plus - c5_minus) / (2 * eps)
                
                # Update
                h_new[j] = h_current[j] - eta * grad_j
                
            # Apply repair step to maintain feasibility
            h_new = repair_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept if better or with some probability
            if c5_new < c5_current or np.random.rand() < 0.01:
                h_current = h_new
                c5_current = c5_new
                
                # Update best if improved
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
            
            # Adjust learning rate based on progress
            if i - last_improvement > patience:
                eta *= 0.95
                last_improvement = i
        
        h_opt = h_best.copy()
        c5_opt = c5_best
    
    return (h_opt.tolist(), c5_opt, n_points)
