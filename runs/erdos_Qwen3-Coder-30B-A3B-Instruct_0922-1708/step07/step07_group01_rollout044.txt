```python
import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def find_max_shift(h):
    """Find the shift k that maximizes overlap."""
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    return max_idx - (len(h) - 1)

def compute_grad(h, k_max, n_points):
    """Compute gradient for current shift k_max."""
    grad = np.zeros(n_points)
    if k_max >= 0:
        # h[i] contributes to (1 - h[i+k_max]) 
        for i in range(len(h)):
            if i + k_max < len(h):
                grad[i] = -1.0
    else:
        # h[i+k_max] contributes to (1 - h[i])
        for i in range(len(h)):
            if i + k_max >= 0:
                grad[i] = -1.0
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        h_init = 0.5 * (1 + np.cos(np.pi * x))
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # CCCP Phase (5k iterations max)
    learning_rate = 0.01
    for i in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find the shift that maximizes overlap
        k_max = find_max_shift(h_current)
        
        # Compute gradient for current maximum shift
        grad = compute_grad(h_current, k_max, n_points)
        
        # Update with gradient
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept move
        h_current = h_new.copy()
        c5_current = c5_new
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
        
        # Adaptive learning rate
        if i > 0 and i % 500 == 0:
            learning_rate *= 0.95
            
        # Occasionally jitter  
        if i % 500 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Cyclic coordinate descent phase (10k iterations max)
    for i in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        
        # Cyclic update of each component
        for j in range(n_points):
            # Find current max shift
            k_max = find_max_shift(h_new)
            
            # Compute gradient for specific component
            grad = np.zeros(n_points)
            if k_max >= 0:
                if j + k_max < len(h_new):
                    grad[j] = -1.0
            else:
                if j + k_max >= 0:
                    grad[j] = -1.0
            
            # Update h[j]
            h_new[j] = h_new[j] - 0.01 * grad[j]
            
            # Clip and renormalize
            h_new = project_h(h_new, n_points)
            
        # Compute new objective
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if improvement
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            
        # Perturb occasionally
        if i % 1000 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_new = project_h(h_new + noise, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
```
