import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Heuristic 1: SQP with trust region for better convergence
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    def compute_grad(h, eps=1e-6):
        grad = np.zeros_like(h)
        h_base = h.copy()
        for i in range(len(h)):
            h_plus = h_base.copy()
            h_minus = h_base.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    # Optimization loop
    max_iter = 10000
    eta = 0.01
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        grad = compute_grad(h_current)
        h_new = h_current - eta * grad
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
        else:
            eta *= 0.999  # Adaptive learning rate
            
        # Perturbation to escape local minima
        if i % 100 == 0:
            h_perturb = h_current.copy()
            num_perturb = max(1, int(0.02 * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                delta = np.random.uniform(-0.05, 0.05)
                h_perturb[idx] += delta
            h_perturb = project_h(h_perturb, n_points, dx)
            c5_perturb = evaluate_h(h_perturb, dx)
            if c5_perturb < c5_current:
                h_current = h_perturb
                c5_current = c5_perturb
    
    # Final projection to ensure size and bounds
    h_best = project_h(h_best, n_points, dx)
    c5_best = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), c5_best, n_points)
