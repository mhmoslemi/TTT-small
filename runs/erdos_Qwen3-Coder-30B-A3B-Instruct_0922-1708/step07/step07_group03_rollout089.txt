import numpy as np
import time
import math
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Binary initialization with correct sum
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Simulated Annealing parameters
    temperature = 0.1
    cooling_rate = 0.995
    min_temperature = 1e-6
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Evaluate initial solution
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor solution
        h_new = h.copy()
        
        # Select two random indices to perturb
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Small random perturbation
        delta = np.random.uniform(-0.05, 0.05)
        
        # Apply perturbation maintaining sum constraint
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain sum constraint
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h_new = h_new * target_sum / h_sum
        
        # Evaluate new solution
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Metropolis criterion
        if new_c5 < current_c5 or np.random.rand() < math.exp(-(new_c5 - current_c5) / temperature):
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Cooling schedule
        temperature = max(min_temperature, temperature * cooling_rate)
    
    # Final evaluation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
