import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(params, n_points):
    """Convert parameters to h(x), assuming piecewise constant function with 5 segments"""
    K = 5
    a = params[:K]
    x = np.concatenate([[0], params[K:K+K-1], [2]])
    
    # Ensure x is sorted
    x = np.sort(x)
    
    # Clip a values
    a = np.clip(a, 0, 1)
    
    # Re-normalize a values
    widths = np.diff(x)
    integral = np.sum(a * widths)
    if integral > 0:
        a = a * (n_points / 2) / integral
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(K):
        start_idx = int(x[i] / dx)
        end_idx = int(x[i+1] / dx)
        if end_idx > start_idx:
            h[start_idx:end_idx] = a[i]
    
    return h

def compute_gradient(params, n_points, dx, k_max):

    eps = 1e-6
    grad = np.zeros_like(params)

    current_c5 = evaluate_h(current_h, dx)
    
    for i in range(len(params)):
        params_plus = params.copy()
        params_plus[i] += eps
        h_plus = parameterize_h(params_plus, n_points)
        h_plus = repair_h(h_plus, n_points)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - current_c5) / eps
    
    return grad
