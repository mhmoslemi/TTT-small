import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    # Try to solve with CVXPY
    try:
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            C5_var >= 0
        ]
        
        # Compute correlations for all lags
        # The full correlation has length 2*n_points - 1
        corr_length = 2 * n_points - 1
        for k in range(corr_length):
            # Compute the convolution term h(x)(1-h(x+k))
            # We compute this manually as a quadratic constraint
            # For index k in full result, we sum over i: h[i] * (1 - h[i-k]) 
            # Padded to handle boundaries
            lag_term = 0
            for i in range(n_points):
                if i-k >= 0 and i-k < n_points:
                    lag_term += h_var[i] * (1 - h_var[i-k])
            
            # Add the constraint
            constraints.append(lag_term <= C5_var)
        
        # Objective
        objective = cp.Minimize(C5_var)
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        # Try different solvers with timeout
        solvers_to_try = ['MOSEK', 'GUROBI', 'ECOS', 'SCIP']
        solution = None
        for solver in solvers_to_try:
            if (time.time() - start_time) > budget_s - 10:
                break
            try:
                prob.solve(solver=solver, verbose=False, max_iters=1000)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    solution = h_var.value
                    break
            except:
                continue
        
        if solution is not None:
            h = np.maximum(0, np.minimum(1, solution))
            h_sum = np.sum(h)
            if abs(h_sum - n_points/2) > 1e-8:
                h = h * (n_points / 2) / h_sum
                
    except Exception as e:
        pass  # Fallback to gradient descent
    
    # Final gradient descent refinement
    if (time.time() - start_time) < budget_s - 5:
        # Simple gradient descent with projection
        learning_rate = 1e-3
        max_iter = 50000
        best_c5 = float('inf')
        best_h = h.copy()
        
        for iteration in range(max_iter):
            if (time.time() - start_time) > budget_s - 5:
                break

            # Compute correlation
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Estimate gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-6
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                if h_plus[i] > 1:
                    h_plus[i] = 1
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps

            # Update step
            h = h - learning_rate * grad
            
            # Projection to [0,1]
            h = np.clip(h, 0, 1)
            
            # Rescale to meet constraint
            h_sum = np.sum(h)
            target_sum = n_points / 2
            if abs(h_sum - target_sum) > 1e-8:
                h = h * target_sum / h_sum
            
        h = best_h
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
    
    return (h.tolist(), c5, n_points)
