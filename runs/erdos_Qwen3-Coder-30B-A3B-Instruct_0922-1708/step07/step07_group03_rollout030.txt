import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    backtracking_factor = 0.5
    initial_gamma = 0.5
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe iterations
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current overlap integral
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find descent direction via linear subproblem
        # Gradient is essentially the correlation with (1-h)
        grad = correlate(h, 1 - h, mode='full')
        # We need to compute the gradient for each point
        grad_values = np.zeros(n_points)
        for i in range(n_points):
            # Compute partial derivative of max_k integral w.r.t h_i
            grad_values[i] = 0
            # Sum up contributions for all k where h_i contributes
            for k in range(n_points):
                if i+k < n_points:
                    grad_values[i] += (1 - h[i+k]) * dx
                if i-k >= 0:
                    grad_values[i] += h[i-k] * dx
        
        # Linear subproblem: minimize ∇f(h) · s subject to constraints
        # Find s_i that minimizes Σ_i grad_i * s_i, s_i in [0,1], sum(s) = 1
        # This is solved by setting s_i = 0 if grad_i > 0, or 1 if grad_i < 0
        s = np.zeros(n_points)
        neg_grad_idx = np.where(grad_values < 0)[0]
        pos_grad_idx = np.where(grad_values > 0)[0]
        if len(neg_grad_idx) > 0:
            s[neg_grad_idx] = 1
        else:
            s[pos_grad_idx] = 1
        
        # Fix the sum
        s_sum = np.sum(s)
        if abs(s_sum) < 1e-10:
            # Avoid division by zero
            s = np.ones(n_points) / n_points
        else:
            s = s * (n_points / 2) / s_sum
            
        # Line search for optimal step size gamma
        gamma = initial_gamma
        while gamma > 1e-10:
            h_new = (1 - gamma) * h + gamma * s
            # Ensure bounds and sum
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if abs(h_new_sum - n_points / 2) < 1e-6:
                corr_new = correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr_new) * dx
                if c5_new < c5:
                    h = h_new
                    break
            gamma *= backtracking_factor
            
    # Final evaluation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
