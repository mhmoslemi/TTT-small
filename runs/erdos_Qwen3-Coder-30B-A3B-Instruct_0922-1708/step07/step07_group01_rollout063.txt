import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Try a combination of cosine taper and sawtooth wave
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Multi-start approach with different initialization strategies
    initializations = []
    
    # 1. Original initialization
    initializations.append(h_init.copy())
    
    # 2. Random initialization
    h_random = np.random.rand(n_points)
    h_random = project_h(h_random, n_points)
    initializations.append(h_random)
    
    # 3. Sine wave initialization
    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 * (1 + np.sin(np.pi * x))
    h_sine = project_h(h_sine, n_points)
    initializations.append(h_sine)
    
    # 4. Step function with central peak
    h_step = np.zeros(n_points)
    center = n_points // 2
    width = n_points // 4
    h_step[center-width//2:center+width//2] = 1.0
    h_step = project_h(h_step, n_points)
    initializations.append(h_step)
    
    # 5. Flat with small bump
    h_bump = np.full(n_points, 0.5)
    h_bump[n_points//2] = 1.0
    h_bump = project_h(h_bump, n_points)
    initializations.append(h_bump)
    
    # 6. Exponential decay
    x = np.linspace(0, 2, n_points)
    h_exp = np.exp(-x) * np.exp(2.0)
    h_exp = project_h(h_exp, n_points)
    initializations.append(h_exp)
    
    # Multi-start optimization with different learning rates and strategies
    best_solution = h_best.copy()
    best_c5 = c5_best
    
    num_starts = len(initializations)
    
    for start_idx in range(num_starts):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_current = initializations[start_idx].copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_solution = h_current.copy()
        
        # Gradient descent with adaptive learning rate
        lr = 0.01
        patience_count = 0
        last_improvement = 0
        max_iter = int(2e5)
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Find the index of maximum overlap shift
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_k = np.argmax(corr)
            max_overlap = corr[max_k]
            
            # Gradient is (1 - h_shifted) at k_max
            # We manually compute the derivative
            grad = np.zeros(n_points)
            epsilon = 1e-5
            
            # Compute finite difference gradient
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += epsilon
                h_plus = project_h(h_plus, n_points)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                max_corr_plus = np.max(corr_plus)
                
                grad[j] = (max_corr_plus - max_overlap) / epsilon
            
            # Update with gradient
            h_new = h_current - lr * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept the move if it improves
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                patience_count = 0
                last_improvement = i
            else:
                patience_count += 1
                # Adaptive learning rate
                if patience_count > 50:
                    lr *= 0.95
                    patience_count = 0
            
            # Periodic perturbation to escape local minima
            if i % 100 == 0 and i > 0:
                noise = np.random.normal(0, 0.005, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
            
            # Update global best
            if c5_current < best_c5:
                best_c5 = c5_current
                best_solution = h_current.copy()
                last_improvement = i
                lr = 0.01  # Reset learning rate
        
        # Final local refinement
        for _ in range(200):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_new = best_solution.copy()
            # Perturb small portion of coordinates
            indices = np.random.choice(n_points, max(1, n_points // 30), replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.001)
            
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < best_c5:
                best_solution = h_new
                best_c5 = c5_new
    
    return (best_solution.tolist(), best_c5, n_points)
