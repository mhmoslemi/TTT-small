import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, k_max, n_points, dx):
    """Compute gradient of C5 with respect to h for the dominant term."""
    grad = np.zeros_like(h)
    if k_max >= 0:
        # The gradient is 1 at positions where 1-h[i+k] contributes to the max
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] = 1.0 * dx
    else:
        # k_max < 0, so we shift the gradient
        for i in range(n_points):
            j = i - k_max
            if 0 <= j < n_points:
                grad[i] = 1.0 * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Coordinate descent with gradient approximation
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    eta = 0.01
    eps = 1e-6
    patience = 100
    last_improvement = 0
    max_iter = 100000
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 5:
            break

        # Compute correlation to get maximum term
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        c5_max = corr[k_max] * dx

        # If current is already better than best, update best
        if c5_max < c5_best:
            h_best = h_current.copy()
            c5_best = c5_max

        # Gradient-based update
        h_new = h_current.copy()
        
        # Compute gradient for k_max
        grad = compute_gradient(h_current, k_max - (n_points - 1), n_points, dx)
        
        # Perform gradient descent step
        h_new -= eta * grad
        
        # Project h to [0,1] and re-scale
        h_new = project_h(h_new, n_points, dx)
        
        # Evaluate new value
        c5_new = evaluate_h(h_new, dx)
        
        # If improvement, accept
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            last_improvement = iter_count
        else:
            # Reduce learning rate
            eta *= 0.999
        
        # Add small perturbation if no improvement for a while
        if iter_count - last_improvement > patience:
            h_perturb = h_current.copy()
            # Make a small Gaussian perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            h_perturb += perturbation
            h_perturb = project_h(h_perturb, n_points, dx)
            c5_perturb = evaluate_h(h_perturb, dx)
            if c5_perturb < c5_current:
                h_current = h_perturb
                c5_current = c5_perturb
                last_improvement = iter_count
                
        iter_count += 1

    return (h_best.tolist(), c5_best, n_points)
