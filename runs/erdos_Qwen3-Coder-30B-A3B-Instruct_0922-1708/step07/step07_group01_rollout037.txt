import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Multiple initializations to escape local optima
    initializations = []
    initializations.append(h_init.copy())
    
    # Add some random perturbations
    for _ in range(3):
        h_pert = h_init + np.random.normal(0, 0.05, n_points)
        initializations.append(project_h(h_pert, n_points))
    
    # Add a few more structured waveforms
    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 * (1 + np.sin(np.pi * x))  # Sine ramp
    h_sine = project_h(h_sine, n_points)
    initializations.append(h_sine)
    
    h_step = np.zeros(n_points)
    h_step[n_points//4 : 3*n_points//4] = 1.0
    h_step = project_h(h_step, n_points)
    initializations.append(h_step)
    
    # Optimizations
    best_h = None
    best_c5 = float('inf')
    
    # Try multiple initializations
    for i, h_start in enumerate(initializations):
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
        
        # Local optimization for this initialization
        max_iter = int(1e5)
        learning_rate = 0.02
        patience = 200
        last_improvement = 0
        epsilon = 1e-6
        
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute the maximum overlap shift
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr)
            max_shift = max_shift_idx - (n_points - 1)
            # Compute gradient for the shift that gives max overlap
            h_shifted = np.roll(h_current, max_shift)
            grad = -(1 - h_shifted)
            
            # Update with gradient
            h_new = h_current - learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                last_improvement = iter_count
            else:
                # Sometimes accept worse moves to escape local minima
                if np.random.rand() < 0.02:
                    h_current = h_new.copy()
                    c5_current = c5_new
                    last_improvement = iter_count
            
            # Early stopping
            if iter_count - last_improvement > patience:
                break
            
            # Perturb periodically
            if iter_count % 1000 == 0 and iter_count > 0:
                noise = np.random.normal(0, 0.01, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                
    return (best_h.tolist(), best_c5, n_points)
