import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles for PSO
    n_particles = 50
    max_iter = 5000
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        # Random initialization for first particle
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    # Initialize remaining particles
    for _ in range(n_particles - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    # Personal bests and global best
    personal_bests = [float('inf')] * n_particles
    global_best_idx = 0
    global_best_value = float('inf')
    global_best_position = None
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Compute correlation
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < personal_bests[i]:
                personal_bests[i] = c5
            
            # Update global best
            if c5 < global_best_value:
                global_best_value = c5
                global_best_position = h.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            pbest = particles[np.argmin(personal_bests)]
            
            v = (w * v + 
                 c1 * r1 * (pbest - h) + 
                 c2 * r2 * (global_best_position - h))
            
            # Apply simulated annealing
            if np.random.rand() < T:
                v += np.random.uniform(-0.1, 0.1, n_points)
            
            # Update position
            h_new = h + v
            
            # Repair constraints
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if abs(h_new_sum - (n_points / 2)) > 1e-8:
                h_new = h_new * (n_points / 2) / h_new_sum
            
            particles[i] = h_new
            velocities[i] = v
            
        # Cooling schedule
        T = max(T * cooling_rate, 0.01)
    
    # Final evaluation
    corr = correlate(global_best_position, 1 - global_best_position, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (global_best_position.tolist(), final_c5, n_points)
