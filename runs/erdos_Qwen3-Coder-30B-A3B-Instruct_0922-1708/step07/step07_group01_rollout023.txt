import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Binary search
    c_low = 0.0
    c_high = 0.5
    tolerance = 1e-5
    max_iter = 50

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        c_mid = (c_low + c_high) / 2.0
        
        # Try to find a feasible solution for the given C
        # Formulate as a linear program
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = []
        constraints.append(cp.sum(h) == n_points / 2)
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        
        # Overlap constraints
        for k in range(n_points):
            # Only consider valid indices
            if 2*k < n_points:
                # Convolution-like constraint: sum of h[i] * (1 - h[i+k]) <= c_mid * dx
                overlap = 0
                for i in range(n_points - k):
                    overlap += h[i] * (1 - h[i + k])
                constraints.append(overlap <= c_mid * dx)
        
        # Objective (minimize C, but we're just checking feasibility)
        # We don't need an objective since we're doing feasibility checking
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_solution = h.value
                h_solution = project_h(h_solution, n_points)
                c5_new = evaluate_h(h_solution, dx)
                if c5_new < c5_best:
                    h_best = h_solution
                    c5_best = c5_new
                c_high = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
                
        if c_high - c_low < tolerance:
            break
    
    return (h_best.tolist(), c5_best, n_points)
