import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    swarm_size = 100
    inertia_start = 0.9
    inertia_end = 0.1
    c1 = 2.0
    c2 = 2.0
    max_iter = 5000
    local_search_freq = 100
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    
    for _ in range(swarm_size):
        h = h_init + np.random.normal(0, 0.01, n_points)
        h = repair_h(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_h(h, dx)
        pbest_positions.append(h.copy())
        pbest_scores.append(score)
    
    # Global best tracking
    global_best_idx = np.argmin(pbest_scores)
    global_best_position = pbest_positions[global_best_idx].copy()
    global_best_score = pbest_scores[global_best_idx]
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # PSO update
        inertia = inertia_start - (inertia_start - inertia_end) * iteration / max_iter
        
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (global_best_position - particles[i]))
            
            particles[i] = particles[i] + velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate new particle
            score = evaluate_h(particles[i], dx)
            
            # Update pbest
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = particles[i].copy()
                
                # Update global best if needed
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = particles[i].copy()
        
        # Local refinement
        if iteration % local_search_freq == 0 and iteration > 0:
            # Perform some gradient descent on the global best
            h_refined = global_best_position.copy()
            for _ in range(100):
                if time.time() - start_time > budget_s - 5:
                    break
                grad = np.zeros(n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                max_k_idx = np.argmax(corr)
                eps = 1e-6
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    grad[j] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
                h_refined = h_refined - 1e-5 * grad
                h_refined = repair_h(h_refined, n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                c5_refined = np.max(corr) * dx
                if c5_refined < global_best_score:
                    global_best_score = c5_refined
                    global_best_position = h_refined.copy()
    
    return (global_best_position.tolist(), global_best_score, n_points)
