import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def get_k_max(h, n_points, dx):
    """Return the shift index where max overlap occurs"""
    corr = correlate(h, 1 - h, mode='full')
    k_max = np.argmax(corr)
    # k ranges from 0 to 2*n_points-2, center is n_points-1
    return k_max - (n_points - 1)

def update_h_gradient(h, k_max, n_points, dx, alpha=0.005):
    """Update h using gradient w.r.t. k_max shift"""
    h_shifted = np.roll(h, k_max)
    grad = 1 - h_shifted
    h_new = h - alpha * grad
    return project_h(h_new, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    learning_rate = 0.01
        
    # Phase 1: CCCP-like - iterative convex approximation
    for i in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Get the shift index where maximum overlap occurs
        k_max = get_k_max(h_current, n_points, dx)
        
        # Update using gradient w.r.t. that shift
        h_new = update_h_gradient(h_current, k_max, n_points, dx, learning_rate)
        
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
        else:
            # Occasionally accept worse moves (simulated annealing)
            if np.random.rand() < 0.01:
                h_current = h_new.copy()
                c5_current = c5_new
                
        # Adaptive learning rate
        if i % 500 == 0 and i > 0:
            learning_rate *= 0.95
            
        # Perturb occasionally to escape local minima
        if i % 500 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Phase 2: Coordinate descent with exact gradient updates
    for i in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic coordinate descent
        for j in range(n_points):
            h_temp = h_current.copy()
            
            # Get k_max for current h
            k_max = get_k_max(h_current, n_points, dx)
            
            # Compute exact gradient for this component
            h_shifted = np.roll(h_current, k_max)
            grad_j = 1 - h_shifted[j]
            
            # Update this component
            h_temp[j] = h_current[j] - 0.005 * grad_j
            
            # Project and renormalize
            h_temp = project_h(h_temp, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_temp, dx)
            
            if c5_new < c5_current:
                h_current = h_temp.copy()
                c5_current = c5_new
                
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    
        # Periodic perturbations
        if i % 1000 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
            
    return (h_best.tolist(), c5_best, n_points)
