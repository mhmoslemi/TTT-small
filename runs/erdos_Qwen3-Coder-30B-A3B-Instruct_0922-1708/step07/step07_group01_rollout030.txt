import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with simple construction
        h_init = np.ones(n_points) * 0.5
        h_init = project_h(h_init, n_points)
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Binary search + gradient descent hybrid
    max_iter = int(1e6)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Dynamic programming step: try to improve each position
        h_trial = h_best.copy()
        improved = False
        
        # Try a random subset of positions
        indices = np.random.choice(n_points, max(10, n_points // 10), replace=False)
        
        for idx in indices:
            if improved:
                break
                
            # Try small modifications
            old_val = h_trial[idx]
            h_trial[idx] = np.clip(old_val + np.random.normal(0, 0.02), 0, 1)
            h_trial = project_h(h_trial, n_points)
            c5_trial = evaluate_h(h_trial, dx)
            
            if c5_trial < c5_best:
                h_best = h_trial.copy()
                c5_best = c5_trial
                improved = True
            else:
                h_trial[idx] = old_val
        
        # Occasionally restart with better structure
        if i % 100 == 0:
            # Try a sinusoidal pattern
            x = np.linspace(0, 2, n_points)
            h_new = np.sin(np.pi * x) * 0.5 + 0.5
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
                
            # Also try binary pattern
            h_binary = np.zeros(n_points)
            for j in range(n_points):
                if (j / n_points) < 0.2 or (j / n_points) > 0.8:
                    h_binary[j] = 1.0
            h_binary = project_h(h_binary, n_points)
            c5_new = evaluate_h(h_binary, dx)
            
            if c5_new < c5_best:
                h_best = h_binary
                c5_best = c5_new
                
    # Final refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_trial = h_best.copy()
        # Perturb a few points
        indices = np.random.choice(n_points, max(3, n_points // 20), replace=False)
        for idx in indices:
            h_trial[idx] = np.clip(h_trial[idx] + np.random.normal(0, 0.01), 0, 1)
        h_trial = project_h(h_trial, n_points)
        c5_trial = evaluate_h(h_trial, dx)
        
        if c5_trial < c5_best:
            h_best = h_trial
            c5_best = c5_trial
    
    return (h_best.tolist(), c5_best, n_points)
