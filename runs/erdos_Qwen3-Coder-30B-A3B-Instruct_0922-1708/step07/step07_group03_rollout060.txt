import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    temp = 0.1
    cooling_rate = 0.995
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Binary initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Perturb h
        h_new = h.copy()
        
        # Select two random indices
        i, j = np.random.randint(0, n_points, 2)
        
        # Perturb values
        delta = np.random.uniform(-0.05, 0.05)
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clamp to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy constraint
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        h_new = h_new * target_sum / h_sum
        
        # Compute C5
        corr = correlate(h_new, 1 - h_new, mode='full')
        c5 = np.max(corr) * dx
        
        # Accept or reject
        if c5 < best_c5:
            h = h_new
            best_c5 = c5
            best_h = h.copy()
        else:
            # Accept with probability
            if np.random.rand() < np.exp(-(c5 - best_c5) / temp):
                h = h_new
                
        # Cool down
        temp *= cooling_rate
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Ensure correct sum
    assert abs(np.sum(best_h) - n_points/2) < 1e-8
    
    return (best_h.tolist(), final_c5, n_points)
