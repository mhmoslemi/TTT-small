import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solutions to try
    initial_solutions = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
        initial_solutions.append(h_init)
    
    # Add some structured solutions
    # Cosine taper
    x = np.linspace(0, 2, n_points)
    h_cos = 0.5 * (1 + np.cos(np.pi * x))
    h_cos = project_h(h_cos, n_points)
    initial_solutions.append(h_cos)
    
    # Sine wave with amplitude modulation
    h_sin = 0.5 * (1 + np.sin(np.pi * x)) * np.exp(-0.5 * (x - 1)**2)
    h_sin = project_h(h_sin, n_points)
    initial_solutions.append(h_sin)
    
    # Random with structured pattern
    h_random = np.random.rand(n_points) * 0.5
    h_random = project_h(h_random, n_points)
    initial_solutions.append(h_random)
    
    # Simple rectangular pulse
    h_rect = np.zeros(n_points)
    h_rect[n_points//4:3*n_points//4] = 1.0
    h_rect = project_h(h_rect, n_points)
    initial_solutions.append(h_rect)
    
    # Optimization settings
    max_iter = 5000
    patience = 500
    learning_rates = [0.02, 0.01, 0.005]
    
    h_best = None
    c5_best = float('inf')
    
    for i_init, h_init in enumerate(initial_solutions):
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
        
        # Try with different learning rates
        for lr in learning_rates:
            if time.time() - start_time > budget_s - 5:
                break
            
            h_local = h_init.copy()
            c5_local = c5_current
            last_improvement = 0
            
            for i in range(max_iter):
                if time.time() - start_time > budget_s - 5:
                    break
                    
                # Get the k that currently maximizes overlap
                corr = correlate(h_local, 1 - h_local, mode='full')
                max_k = np.argmax(corr)
                max_val = corr[max_k]
                
                # Gradient w.r.t. that shift
                # We compute the gradient of C₅ with respect to h
                # The gradient at each point can be derived from the derivative of the maximal overlap
                # ∇C₅ = (1 - h_shifted_k) where k is the maximizer
                h_shifted = np.zeros_like(h_local)
                if max_k < n_points:
                    h_shifted[max_k:] = h_local[:-max_k]
                else:
                    h_shifted[:n_points - max_k] = h_local[max_k:]
                
                gradient = 1.0 - h_shifted
                
                # Update with gradient
                h_new = h_local - lr * gradient
                h_new = project_h(h_new, n_points)
                
                # Evaluate new value
                c5_new = evaluate_h(h_new, dx)
                
                if c5_new < c5_local:
                    h_local = h_new.copy()
                    c5_local = c5_new
                    last_improvement = i
                else:
                    if i - last_improvement > patience:
                        break
                
                if c5_local < c5_best:
                    h_best = h_local.copy()
                    c5_best = c5_local
                    last_improvement = i
                    if i % 100 == 0:
                        # Occasionally add a small random perturbation to escape local minima
                        noise = np.random.normal(0, 0.001, n_points)
                        h_local = project_h(h_local + noise, n_points)
                        c5_local = evaluate_h(h_local, dx)
    
    # Final refinement
    if h_best is not None and time.time() - start_time < budget_s - 2:
        for _ in range(1000):
            if time.time() - start_time > budget_s - 2:
                break
            
            h_new = h_best.copy()
            # Perturb some coordinates
            indices = np.random.choice(n_points, max(2, n_points // 20), replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.002)
            
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
    
    if h_best is None:
        # Return the best initialization if nothing else was found
        h_best = initial_solutions[0]
        c5_best = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), c5_best, n_points)
