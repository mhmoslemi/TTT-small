import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    S = 100  # swarm size
    max_iter = 5000
    w_min, w_max = 0.1, 0.9
    c1 = c2 = 2.0
    vel_range = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    
    for _ in range(S):
        if 'initial_h_values' in globals():
            h = h_init + np.random.normal(0, 0.01, n_points)
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-vel_range, vel_range, n_points))
        pbest_positions.append(h.copy())
        pbest_scores.append(evaluate_h(h, dx))
    
    # Initialize global best
    global_best_idx = np.argmin(pbest_scores)
    global_best_position = pbest_positions[global_best_idx].copy()
    global_best_score = pbest_scores[global_best_idx]
    
    # PSO optimization
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Update inertia weight (linear decay)
        w = w_max - (w_max - w_min) * iter_count / max_iter
        
        for i in range(S):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) +
                           c2 * r2 * (global_best_position - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair position
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = particles[i].copy()
        
        # Occasionally apply local refinement
        if iter_count % 50 == 0 and iter_count > 0:
            # Gradient descent on global best
            h_refined = global_best_position.copy()
            for _ in range(200):

                corr = correlate(h_refined, 1 - h_refined, mode='full')
                max_k_idx = np.argmax(corr)
                max_k = max_k_idx - (n_points - 1)

                grad = np.zeros(n_points)
                for j in range(n_points):

                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    grad[j] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
                
                # Normalize gradient
                norm = np.linalg.norm(grad) + 1e-12
                grad = grad / norm
                
                h_refined = h_refined - 1e-5 * grad
                h_refined = repair_h(h_refined, n_points)

                    global_best_score = score
                    global_best_position = h_refined.copy()
    
    return (global_best_position.tolist(), global_best_score, n_points)
