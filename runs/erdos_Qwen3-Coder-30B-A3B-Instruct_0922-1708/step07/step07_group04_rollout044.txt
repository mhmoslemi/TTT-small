import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize swarm
    m = 100  # number of particles
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        swarm.append(h_init)
    else:
        # Start with triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
        swarm.append(h_init)
    
    # Initialize other particles randomly with repair
    for _ in range(m - 1):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        swarm.append(h)
    
    # Initialize velocities
    for i in range(m):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    c1, c2 = 2.0, 2.0
    w_start, w_end = 0.9, 0.4
    max_iter = 10000
    patience = 1000
    
    # Keep track of personal and global best
    pbest = [swarm[i].copy() for i in range(m)]
    pbest_fitness = [evaluate_h(swarm[i], dx) for i in range(m)]
    gbest = pbest[np.argmin(pbest_fitness)].copy()
    gbest_fitness = min(pbest_fitness)
    
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * i / max_iter
        
        for j in range(m):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[j] = (
                w * velocities[j] +
                c1 * r1 * (pbest[j] - swarm[j]) +
                c2 * r2 * (gbest - swarm[j])
            )
            
            # Clip velocity
            velocities[j] = np.clip(velocities[j], -0.2, 0.2)
            
            # Update position
            swarm[j] = swarm[j] + velocities[j]
            
            # Repair solution
            swarm[j] = repair_h(swarm[j], n_points)
            
            # Evaluate fitness
            fit = evaluate_h(swarm[j], dx)
            
            # Update personal best
            if fit < pbest_fitness[j]:
                pbest[j] = swarm[j].copy()
                pbest_fitness[j] = fit
                
                # Update global best
                if fit < gbest_fitness:
                    gbest = swarm[j].copy()
                    gbest_fitness = fit
                    last_improvement = i
        
        if i - last_improvement > patience:
            break
    
    # Final gradient descent refinement on gbest
    h_current = gbest.copy()
    c5_current = g5_fitness = gbest_fitness
    h_best = h_current.copy()
    c5_best = c5_current
    
    # local optimization with gradient steps
    eta = 0.001
    eps = 1e-6
    max_local_iter = 500
    
    for local_iter in range(max_local_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient approximation using finite differences
        h_new = h_current.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_current[j] - eta * grad_j
    
        # Apply repair step
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
    
    return (h_best.tolist(), c5_best, n_points)
