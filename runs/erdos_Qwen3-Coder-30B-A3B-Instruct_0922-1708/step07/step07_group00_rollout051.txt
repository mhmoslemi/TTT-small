import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate multiple initial points using Sobol sequences
    sampler = qmc.Sobol(d=n_points, seed=seed)
    n_initials = 500
    initial_points = sampler.random(n=n_initials)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Optimization loop
    for i in range(n_initials):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Transform Sobol samples to [0,1]^n_points
        h = initial_points[i]
        h = h * (n_points / 2) / np.sum(h)
        h = repair_h(h, n_points)
        
        # Nelder-Mead optimization
        try:
            result = minimize(evaluate_h, h, args=(dx,), method='Nelder-Mead', 
                              options={'maxiter': 1000, 'xatol': 1e-6, 'fatol': 1e-6})
            h_opt = result.x
            h_opt = repair_h(h_opt, n_points)
        except:
            h_opt = h
            
        # Gradient-based refinement
        max_refine_steps = 1000
        learning_rate = 1e-5
        for step in range(max_refine_steps):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute finite differences gradient
            eps = 1e-6
            grad = np.zeros(n_points)
            corr_ref = correlate(h_opt, 1 - h_opt, mode='full')
            max_k_idx = np.argmax(corr_ref)
            
            for j in range(n_points):
                h_plus = h_opt.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                grad[j] = (corr_plus[max_k_idx] - corr_ref[max_k_idx]) / eps
            
            # Update step
            h_opt = h_opt - learning_rate * grad
            h_opt = repair_h(h_opt, n_points)
            
            # Check for convergence
            if np.linalg.norm(grad) < 1e-8:
                break
                
        # Evaluate final solution
        c5 = evaluate_h(h_opt, dx)
        if c5 < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
    
    return (best_h.tolist(), best_c5, n_points)
