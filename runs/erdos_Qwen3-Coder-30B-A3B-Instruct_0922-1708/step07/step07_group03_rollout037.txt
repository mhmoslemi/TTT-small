import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random binary initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing parameters
    max_iter = 100000
    temperature = 0.1
    cooling_rate = 0.995
    min_temperature = 1e-6
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Local search loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Current C5 value
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Generate perturbation
        h_new = h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Perturb two points
        delta = np.random.uniform(-0.05, 0.05)
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clamp to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet constraint
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        h_new = h_new * target_sum / h_sum
        
        # Compute new C5
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject based on Metropolis criterion
        if new_c5 < current_c5:
            h = h_new
        else:
            # Accept worse solution with probability
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / temperature):
                h = h_new
                
        # Cool down temperature
        temperature = max(min_temperature, temperature * cooling_rate)
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Ensure final solution meets constraints
    assert abs(np.sum(best_h) - n_points / 2) < 1e-8, "Sum constraint violated"
    assert np.all((best_h >= 0) & (best_h <= 1)), "Bounds constraint violated"
    
    return (best_h.tolist(), final_c5, n_points)
