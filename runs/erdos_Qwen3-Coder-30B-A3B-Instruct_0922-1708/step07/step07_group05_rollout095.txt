import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidate(n_points, k_max=5):
    h = np.zeros(n_points)
    for _ in range(10):  # try 10 times to get a valid candidate
        coeffs = np.random.randn(k_max * 2)
        for i in range(n_points):
            x = i * 2.0 / n_points
            h[i] = np.sum([coeffs[2*j] * np.cos((j+1) * np.pi * x) + 
                          coeffs[2*j+1] * np.sin((j+1) * np.pi * x) 
                          for j in range(k_max)])
        h = (h - h.min()) / (h.max() - h.min()) if h.max() > h.min() else np.ones(n_points) * 0.5
        h = repair_h(h, n_points)
        if np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n_points/2) < 1e-6:
            return h
    return np.ones(n_points) * 0.5

    candidate_scores = []

        candidate_scores.append((c5, i))

    best_candidates = [candidates[i] for _, i in candidate_scores[:10]]
    
    # Phase 2: Gradient descent refinement
    h_best = best_candidates[0].copy()
    c5_best = evaluate_h(h_best, dx)
    
    patience = 200
    last_improvement = 0
    
    for i, h_refine in enumerate(best_candidates):
        h_current = h_refine.copy()
        c5_current = evaluate_h(h_current, dx)
        
        # Optimization parameters
        learning_rate = 0.01
        max_iter = 2000
        
        for iter_id in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradients with finite differences
            epsilon = 1e-6
            grad = np.zeros(n_points)

                h_plus = h_current.copy()

                c5_minus = evaluate_h(h_minus, dx)
