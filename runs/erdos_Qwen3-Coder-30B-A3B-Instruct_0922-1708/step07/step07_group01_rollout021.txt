import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  
        h_init = project_h(h_init, n_points)
    
    # Binary search for optimal C
    c_low = 0.0
    c_high = 0.5
    tolerance = 1e-5
    best_h = h_init.copy()
    best_c = evaluate_h(h_init, dx)
    
    # Precompute overlap indices
    overlap_indices = []
    for k in range(n_points):
        overlap_indices.append((k, list(range(n_points - k))))
        
    max_iter = 50
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Solve feasibility problem using CVXPY
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints
        for k in range(1, n_points):
            # Sum h[i] * (1 - h[i+k]) for valid i
            if k < n_points:
                expr = 0
                for i in range(n_points - k):
                    expr += h_vars[i] * (1 - h_vars[i+k])
                constraints.append(expr <= c_mid * dx)
        
        objective = cp.Minimize(0)  # Dummy objective
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h_vars.value
                if h_solution is not None:
                    h_solution = np.array(h_solution)
                    h_solution = project_h(h_solution, n_points)
                    c_val = evaluate_h(h_solution, dx)
                    if c_val < best_c:
                        best_h = h_solution.copy()
                        best_c = c_val
                    c_high = c_mid
                else:
                    c_low = c_mid
            else:
                c_low = c_mid
        except Exception:
            c_low = c_mid
            
        if abs(c_high - c_low) < tolerance:
            break
            
    # Final refinement
    h_current = best_h.copy()
    c_current = best_c
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Local search
        h_new = h_current.copy()
        # Perturb multiple points
        indices = np.random.choice(n_points, max(5, n_points // 20), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        h_new = project_h(h_new, n_points)
        c_new = evaluate_h(h_new, dx)
        if c_new < c_current:
            h_current = h_new
            c_current = c_new
    
    return (h_current.tolist(), c_current, n_points)
