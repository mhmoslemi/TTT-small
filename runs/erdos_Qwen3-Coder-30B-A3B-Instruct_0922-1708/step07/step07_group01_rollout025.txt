import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = int(1e6)
    epsilon = 1e-6
    patience = 100
    last_improvement = 0
    
    # Run multiple random restarts to avoid local minima
    num_restarts = 5
    for restart in range(num_restarts):
        h_current = h_init.copy()
        if restart > 0:
            # Perturb slightly
            noise = np.random.normal(0, 0.02, n_points)
            h_current = project_h(h_current + noise, n_points)
        
        c5_current = evaluate_h(h_current, dx)
        h_best_restart = h_current.copy()
        c5_best_restart = c5_current
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute all correlations
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr)
            max_shift = max_shift_idx - (len(h_current) - 1)
            
            # Compute gradient for the maximum overlap shift
            grad = np.zeros(n_points)
            
            # For shift k, gradient is (1 - h_{i-k}) where h_shifted is 1-h
            # This means we want to compute the correlation of 1-h with h shifted
            # But we want gradient of max shift overlap w.r.t. h values
            
            # Here we manually construct the gradient for max_shift
            # If we shift h by k (k=0 to 2*n_points-2):
            #  overlap = sum h[i] * (1 - h[i+k]) = sum h[i] - sum h[i]*h[i+k]
            #  so d/dh[i] (overlap) = 1 - h[i+k] for i+k < n
            #  若 h[i+k] 不在范围内，则对应为0
            
            if max_shift >= 0:
                # h_shifted = h_current[max_shift:]
                for j in range(len(h_current)):
                    if j + max_shift < len(h_current):
                        grad[j] = -h_current[j + max_shift]
                    else:
                        grad[j] = 0
                grad = grad + 1.0
            else:
                # h_shifted = h_current[:max_shift]
                # Actually the effect is reverse due to Python negative indexing issues
                # But shifting by -k means h[i+k] for k=-max_shift
                # which is h[i-k] for positive k  
                # We go from left to right, shift +k means get h[i-k]: i >= k
                for j in range(len(h_current)):
                    if j + abs(max_shift) < len(h_current):
                        grad[j] = -h_current[j + abs(max_shift)]
                    else:
                        grad[j] = 0
                grad = grad + 1.0
            
            # Update with gradient
            h_new = h_current - learning_rate * grad
            
            # Clamp and project
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept move if improvement or with some probability
            if c5_new < c5_current or np.random.rand() < 0.01:
                h_current = h_new.copy()
                c5_current = c5_new
            else:
                # Occasionally accept worse moves (simulated annealing)
                if np.random.rand() < 0.01:
                    h_current = h_new.copy()
                    c5_current = c5_new
            
            # Update best
            if c5_current < c5_best_restart:
                h_best_restart = h_current.copy()
                c5_best_restart = c5_current
                last_improvement = i
                
            # Adaptive learning rate
            if i - last_improvement > patience:
                learning_rate *= 0.95
                last_improvement = i
                
            # Occasionally jitter to escape local minima
            if i % 500 == 0 and i > 0:
                noise = np.random.normal(0, 0.005, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
            
            # Occasionally restore best solution to avoid getting trapped
            if i % 1000 == 0 and i > 0:
                h_current = h_best_restart.copy()
                c5_current = c5_best_restart
        
        # Keep the best restart
        if c5_best_restart < c5_best:
            h_best = h_best_restart.copy()
            c5_best = c5_best_restart
    
    # Final refinement through local search
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
