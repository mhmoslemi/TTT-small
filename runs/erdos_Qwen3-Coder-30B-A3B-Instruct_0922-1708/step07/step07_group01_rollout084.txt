import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a bell-shaped curve
        h_init = np.exp(-16 * (x - 1)**2)
        h_init = project_h(h_init, n_points)
    
    # Start with the initial solution
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Try different optimization strategies
    strategies = [
        "gradient_descent", 
        "local_search",
        "simulated_annealing",
        "random_perturbations"
    ]
    
    # Phase 1: Gradient-based optimization
    def gradient_descent_step(h, dx, learning_rate=0.02, epsilon=1e-6):
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            h_plus = project_h(h_plus, n_points)
            h_minus = project_h(h_minus, n_points)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
        
        h_new = h - learning_rate * grad
        return project_h(h_new, n_points)
    
    # Phase 2: Fine-tune with local search
    def local_optimization(h, dx, iterations=100):
        current = h.copy()
        c5_current = evaluate_h(current, dx)
        best = current.copy()
        c5_best = c5_current
        
        for _ in range(iterations):
            if time.time() - start_time > budget_s - 5:
                break
            # Make small random perturbations
            h_new = current.copy()
            indices = np.random.choice(n_points, max(1, n_points // 20), replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.005)
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            if c5_new < c5_current:
                current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    best = current.copy()
                    c5_best = c5_current
        return best, c5_best

    iterations = 0
    while time.time() - start_time < budget_s - 20:
        iterations += 1
        
        # Try gradient descent step
        try:
            h_new = gradient_descent_step(h_current, dx, 0.02)
            c5_new = evaluate_h(h_new, dx)
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
        except:
            pass
            
        # Occasionally apply local search 
        if iterations % 50 == 0:
            try:
                h_search, c5_search = local_optimization(h_current, dx, 100)
                if c5_search < c5_best:
                    h_best = h_search
                    c5_best = c5_search
                h_current = h_best
                c5_current = c5_best
            except:
                pass
                
        # Occasionally perturb more aggressively
        if iterations % 100 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Final refinement
    h_final, c5_final = local_optimization(h_best, dx, 1000)
    if c5_final < c5_best:
        h_best = h_final
        c5_best = c5_final
    
    return (h_best.tolist(), c5_best, n_points)
