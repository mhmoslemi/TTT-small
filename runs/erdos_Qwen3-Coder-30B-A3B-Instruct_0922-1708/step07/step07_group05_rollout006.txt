import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size):
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Add random triangular pulses
    for _ in range(pop_size - len(population)):
        h = np.zeros(n_points)
        peak_loc = np.random.uniform(0.5, 1.5)
        width = np.random.uniform(0.2, 0.8)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if abs(x - peak_loc) < width / 2:
                h[i] = 1 - 2 * abs(x - peak_loc) / width
        h = repair_h(h, n_points)
        population.append(h)
    
    return population

def selection(population, fitness, num_selected):
    # Tournament selection
    selected = []
    for _ in range(num_selected):
        idx1, idx2 = np.random.choice(len(population), 2, replace=False)
        if fitness[idx1] < fitness[idx2]:
            selected.append(population[idx1])
        else:
            selected.append(population[idx2])
    return selected

def crossover(parent1, parent2):
    # Single-point crossover
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n_points = len(h)
    indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.05)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # GA parameters
    pop_size = 50
    num_generations = 1000
    num_selected = pop_size // 2
    mutation_rate = 0.2
    patience = 200
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    
    best_h = None
    best_c5 = float('inf')
    last_improvement = 0
    
    for generation in range(num_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = generation
        
        # Check for early stopping
        if generation - last_improvement > patience:
            break
            
        # Selection
        selected = selection(population, fitness, num_selected)
        
        # Crossover and mutation
        new_population = [h.copy() for h in selected]  # Elitism
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = repair_h(mutate(child1, mutation_rate), n_points)
            child2 = repair_h(mutate(child2, mutation_rate), n_points)
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
    
    return (best_h.tolist(), best_c5, n_points)
