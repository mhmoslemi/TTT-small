import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate a candidate solution."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    return mutated

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Create a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    # Population based evolution strategy
    population_size = 50
    population = []
    fitness = []
    
    # Initialize population
    population.append(h_init)
    fitness.append(evaluate_h(h_init, dx))
    
    # Add diverse initial solutions
    for _ in range(population_size - 1):
        h = project_h(np.random.rand(n_points), n_points, dx)
        # Add some structure
        if np.random.rand() < 0.5:
            h = mutate(h_init.copy(), 0, 0.1)
        population.append(h)
        fitness.append(evaluate_h(h, dx))
    
    # Best solution tracking
    best_idx = np.argmin(fitness)
    h_best = population[best_idx].copy()
    c5_best = fitness[best_idx]
    
    # Evolution parameters
    mutation_scale = 0.1
    stagnation_counter = 0
    max_stagnation = 100
    
    # Main evolution loop
    generation = 0
    while time.time() - start_time < budget_s - 5:
        generation += 1
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size // 5]
        
        # Selection
        selected = [population[i] for i in top_indices]
        selected_fitness = [fitness[i] for i in top_indices]
        
        # Generate new population
        new_population = []
        new_fitness = []
        
        # Keep top performers
        for i in range(len(selected)):
            new_population.append(selected[i])
            new_fitness.append(selected_fitness[i])
        
        # Generate offspring through mutation and crossover
        while len(new_population) < population_size:
            # Selection for parents
            parent1_idx = np.random.choice(len(selected))
            parent2_idx = np.random.choice(len(selected))
            
            # Crossover (averaging)
            alpha = np.random.rand()
            child = alpha * selected[parent1_idx] + (1 - alpha) * selected[parent2_idx]
            
            # Mutation
            child = mutate(child, generation, mutation_scale)
            
            # Evaluation
            child_fitness = evaluate_h(child, dx)
            
            # Add to new population
            new_population.append(child)
            new_fitness.append(child_fitness)
        
        # Update population
        population = new_population
        fitness = new_fitness
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < c5_best:
            c5_best = fitness[best_idx]
            h_best = population[best_idx].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Adjust mutation scale
        if stagnation_counter > 50:
            mutation_scale *= 0.95
            stagnation_counter = 0
            
        # Occasionally inject new randomness
        if np.random.rand() < 0.01 and len(population) < population_size:
            new_h = project_h(np.random.rand(n_points), n_points, dx)
            population.append(new_h)
            fitness.append(evaluate_h(new_h, dx))
            
    return (h_best.tolist(), c5_best, n_points)
