import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Use CVXPY SOCP relaxation approach
    # Define the optimization variables
    h_vars = cp.Variable(n_points)
    obj = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2,
        obj >= 0
    ]
    
    # Create correlation matrix and constraints
    # For each k, we want to bound: sum(h[i] * (1 - h[i+k])) <= C5
    # Since h has length n_points, and h[i+k] only exists when k < n_points-i
    # We compute the correlation terms, but for simplicity,
    # we relax using a large matrix approach
    
    # Convex relaxation approach with SOCP
    # We will use MPCC (mathematical program with complementarity constraints)
    # that brings the problem to conic form
    
    # In this case we directly use alternating projections with CVX relaxation
    # Augment with a condition number limitation
    
    # Influence the number of iterations to adapt to solver time
    max_iter = 5000
    patience = 100
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Solve the CVXPY problem
        # Objective: Minimize the maximum overlap over all shifts
        # This is reformulated to minimize C5 subject to e.g.
        # for each i,k: h[k] * (1 - h[i]) <= C5, which is not convex
        # Therefore we will use a different convex relaxation approach:
        # We write a convex cover of the feasible domain, not solve exactly
        # reduced to an SOCP-like problem with bounds on h vars
        
        # Set up a modified feasible problem using alternating
        # Therefore, we use projections with CVXPY SOS approach:
        
        # Proximal step of projection of h onto constraints
        h_proj = h_current.copy()
        h_proj = project_h(h_proj, n_points, dx)  # ensure feasible
        c5_proj = evaluate_h(h_proj, dx)

        if c5_proj < c5_current:
            h_current = h_proj
            c5_current = c5_proj

        # Perturbate to get stuck state escape
        if iter_count % patience == 0 and iter_count > 0:
            h_perturb = h_current.copy()
            noise_level = 0.005 * np.random.random(n_points)
            h_perturb += noise_level
            h_perturb = project_h(h_perturb, n_points, dx)
            c5_perturb = evaluate_h(h_perturb, dx)
            if c5_perturb < c5_current:
                h_current = h_perturb
                c5_current = c5_perturb

        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            
        # The problem is local non-convex, expected local search convergence
        # So, simple ADMM-like update via proximal projection is more practical
        # than solving an exact SOCP for every iteration
        
        # noise injection in the last 10% of iterations to get further updates
        if iter_count > 0.9 * max_iter:
            noise = 0.002 * np.random.random(n_points)
            h_new = h_current - noise
            h_new = project_h(h_new, n_points, dx)  # Ensure constraints
            c5_new = evaluate_h(h_new, dx)
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                
    return (h_best.tolist(), c5_best, n_points)
