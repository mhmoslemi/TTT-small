import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate a solution with adaptive noise and random flips."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Population-based evolution strategy
    population_size = 50
    mutation_scale = 0.1
    initial_mutation_scale = 0.1
    stagnation_counter = 0
    stagnation_threshold = 100
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
        population.append(h_init)
    else:
        # Create a structured initial solution
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
        population.append(h_init)
    
    # Add diverse initial solutions
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    while time.time() - start_time < budget_s - 5:
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        top_indices = sorted_indices[:population_size // 5]
        
        # Selection: keep top performers
        new_population = [population[i] for i in top_indices]
        
        # Generate new candidates via mutation
        for i in range(population_size - len(new_population)):
            # Select two parents from top performers
            parent1 = new_population[np.random.randint(len(new_population))]
            parent2 = new_population[np.random.randint(len(new_population))]
            
            # Blend
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutate
            child = mutate(child, generation, mutation_scale)
            new_population.append(child)
        
        population = new_population
        generation += 1
        
        # Adapt mutation scale
        if generation % 10 == 0:
            mutation_scale *= 0.999
        
        # Stagnation detection
        if generation % stagnation_threshold == 0:
            stagnation_counter += 1
            if stagnation_counter > 2:
                # Inject random solutions
                for i in range(5):
                    h = np.random.rand(n_points)
                    h = project_h(h, n_points, dx)
                    population.append(h)
                stagnation_counter = 0
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    
    # Fallback to first population member
    h_best = population[0]
    c5_best = evaluate_h(h_best, dx)
    return (h_best.tolist(), c5_best, n_points)
