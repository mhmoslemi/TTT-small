import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.spatial import distance

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate multiple initial points with Sobol sequences
    try:
        from sobol_seq import i4_sobol_generate
        sobol_points = i4_sobol_generate(n_points, 500)
        sobol_points = sobol_points.T
    except ImportError:
        # Fallback: use random points if sobol_seq is not available
        sobol_points = np.random.rand(500, n_points)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Run optimization for each initialization
    for i in range(min(500, sobol_points.shape[0])):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_current = sobol_points[i]
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Nelder-Mead optimization
        try:
            res = minimize(
                lambda x: evaluate_h(repair_h(x, n_points), dx),
                h_current,
                method='Nelder-Mead',
                options={'maxiter': 1000, 'xatol': 1e-6, 'fatol': 1e-6}
            )
            if res.success:
                h_opt = repair_h(res.x, n_points)
                c5_opt = evaluate_h(h_opt, dx)
                
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass
    
    # Final refinement with gradient descent
    current_h = best_h.copy()
    current_c5 = best_c5
    max_refinement_steps = 1000
    learning_rate = 1e-5
    decay_rate = 0.99
    step_count = 0
    
    for i in range(max_refinement_steps):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient with finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        base_val = evaluate_h(current_h, dx)
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            plus_val = evaluate_h(h_plus, dx)
            grad[j] = (plus_val - base_val) / eps
        
        # Update step with learning rate
        current_h = current_h - learning_rate * grad
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        
        # Adaptive learning rate decay
        step_count += 1
        if step_count % 100 == 0:
            learning_rate *= decay_rate
    
    return (best_h.tolist(), best_c5, n_points)
