import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Fourier-based optimization
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Fourier transform of current solution
        H = np.fft.fft(current_h)
        H_abs = np.abs(H)
        H_phase = np.angle(H)
        
        # Minimize max correlation in frequency domain
        # We want to minimize the maximum of correlation
        # Let's try a projection approach with constraints
        n = len(current_h)
        
        # Create a new search direction by small perturbations in phase
        phase_perturbation = np.random.normal(0, 0.1, n)
        H_new = H_abs * np.exp(1j * (H_phase + phase_perturbation))
        h_new = np.real(np.fft.ifft(H_new))
        
        # Repair to feasible space
        h_new = repair_h(h_new, n_points)
        
        # Evaluate the new solution
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Update current solution
        current_h = h_new.copy()
        
    # Final fine-tuning with local search
    h_final = best_h.copy()
    fine_iter = 1000
    for i in range(fine_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Small gradient-like update with finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        base_corr = evaluate_h(h_final, dx)
        
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = evaluate_h(h_plus, dx)
            grad[j] = (corr_plus - base_corr) / eps
            
        # Update with negative gradient
        h_final = h_final - 1e-6 * grad
        h_final = repair_h(h_final, n_points)
        new_c5 = evaluate_h(h_final, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_final.copy()
            
    return (best_h.tolist(), best_c5, n_points)
