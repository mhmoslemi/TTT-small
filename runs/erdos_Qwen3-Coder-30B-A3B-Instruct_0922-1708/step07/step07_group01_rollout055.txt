import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Dynamic Programming approach
    # Try discrete values for h[i]
    h_values = np.linspace(0, 1, 11)  # 0, 0.1, ..., 1.0
    
    # Helper to clamp a value to [0, 1]
    clamp = lambda x: np.clip(x, 0, 1)
    
    # Main optimization loop
    iterations = 0
    max_iterations = 10000
    patience = 100
    last_improvement = 0
    
    while iterations < max_iterations and time.time() - start_time < budget_s - 5:
        # DP-style approach: for each position try different values to minimize overlap
        improved = False
        
        # Try to improve by one element at a time
        for i in np.random.permutation(n_points):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_copy = h_current.copy()
            
            best_h_i = h_current[i]
            best_c5 = c5_current
            
            # Try several discrete values for h[i]
            for h_val in h_values:
                h_copy[i] = h_val
                h_copy = project_h(h_copy, n_points)
                c5_trial = evaluate_h(h_copy, dx)
                
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h_i = h_val
            
            if best_c5 < c5_current:
                h_current[i] = best_h_i
                h_current = project_h(h_current, n_points)
                c5_current = best_c5
                improved = True
        
        # Occasionally perturb the whole function to escape local minimum
        if iterations % 50 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
        
        # Update the best solution so far
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = iterations
        
        if not improved:
            if iterations - last_improvement > patience:
                # Restart with a better solution if no progress
                h_current = h_best.copy()
                c5_current = c5_best
                last_improvement = iterations
        
        iterations += 1
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = h_best.copy()
        indices = np.random.choice(n_points, max(2, n_points // 10), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
