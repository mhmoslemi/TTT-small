import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # PSO parameters
    swarm_size = 50
    max_iter = 50000
    inertia_start = 0.9
    inertia_end = 0.4
    cognition = 1.5
    social = 1.5
    velocity_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # One particle from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        # Create a structured feasible solution
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    
    # Fill rest with random and structured particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        particles.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Initialize personal bests
    personal_bests = [p.copy() for p in particles]
    personal_best_scores = [evaluate_h(p, dx) for p in particles]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Adaptive inertia weight
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_num / max_iter)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognition * r1 * (personal_bests[i] - particles[i])
            social_term = social * r2 * (global_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive + social_term
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_h(particles[i], n_points, dx)
            
            # Evaluate fitness
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_bests[i] = particles[i].copy()
                personal_best_scores[i] = score
                
                # Update global best if needed
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
        
        # Perturb particles to avoid stagnation
        if iter_num % 1000 == 0 and iter_num > 0:
            improvement = False
            for i in range(1, swarm_size):  # Skip first particle (initial)
                if np.random.rand() < 0.3:
                    # Small random perturbation
                    perturbation = np.random.normal(0, 0.005, n_points)
                    particles[i] = particles[i] + perturbation
                    particles[i] = project_h(particles[i], n_points, dx)
                    score = evaluate_h(particles[i], dx)
                    if score < personal_best_scores[i]:
                        personal_bests[i] = particles[i].copy()
                        personal_best_scores[i] = score
                        if score < global_best_score:
                            global_best = particles[i].copy()
                            global_best_score = score
                            improvement = True
                            
            if not improvement and iter_num > 10000:
                # Random restart for some particles
                for i in range(1, swarm_size):
                    if np.random.rand() < 0.1:
                        h = np.random.rand(n_points)
                        h = project_h(h, n_points, dx)
                        particles[i] = h
                        score = evaluate_h(particles[i], dx)
                        personal_bests[i] = h.copy()
                        personal_best_scores[i] = score
                        if score < global_best_score:
                            global_best = h.copy()
                            global_best_score = score
    
    return (global_best.tolist(), global_best_score, n_points)
