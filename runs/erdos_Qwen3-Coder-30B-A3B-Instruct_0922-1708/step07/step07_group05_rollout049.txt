import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidate(n_points, seed=None):
    if seed is not None:
        np.random.seed(seed)
    # Generate candidate using Fourier basis
    h = np.zeros(n_points)
    # Create 4 harmonics
    for k in range(1, 5):
        a = np.random.rand() * 2 - 1  # -1 to 1
        b = np.random.rand() * 2 - 1  # -1 to 1
        for i in range(n_points):
            x = i * 2.0 / n_points
            h[i] += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
    
    # Shift and scale to [0,1]
    h = h - np.min(h)
    h_max = np.max(h)
    if h_max > 0:
        h = h / h_max
    h = np.clip(h, 0, 1)
    
    # Repair to satisfy sum constraint
    h = repair_h(h, n_points)
    return h

    dx = 2.0 / n_points
    start_time = time.time()
    
    # Collect diverse candidates using Fourier modes
    candidates = []
    num_candidates = 50
    
    # Use initial_h_values if it exists
    if 'initial_h_values' in globals():

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        candidates.append((h_init, evaluate_h(h_init, dx)))
    else:
        # Generate initial candidate with piecewise linear construction if necessary
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)
        candidates.append((h_init, evaluate_h(h_init, dx)))

    # Generate diverse Fourier candidates
    for i in range(num_candidates):
        h = generate_fourier_candidate(n_points, seed=seed+i)
        c5 = evaluate_h(h, dx)
        candidates.append((h, c5))
        
    # Sort by C5 value and keep best 10
    candidates.sort(key=lambda x: x[1])
    top_candidates = candidates[:10]
    
    # Refinement phase with gradient descent
    best_h = None
    best_c5 = float('inf')
    patience = 200
    last_improvement = 0
    refinement_iters = 2000
    
    for i, (h_refine, c5_refine) in enumerate(top_candidates):
        h_current = h_refine.copy()
        c5_current = c5_refine
        h_best = h_current.copy()
        c5_best = c5_current
        learning_rate = 0.01
        
        for iter in range(refinement_iters):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Gradient approximation using finite differences
            grad = np.zeros(n_points)
            eps = 1e-6
            for j in range(n_points):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                h_plus = repair_h(h_plus, n_points)
                h_minus = repair_h(h_minus, n_points)

                c5_minus = evaluate_h(h_minus, dx)
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update step
            h_new = h_current - learning_rate * grad
            h_new = repair_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)

                    c5_best = c5_current

            else:

                    c5_current = c5_new
            
            # Adaptive learning rate

        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best

                h[i] = 0.0

                h[i] = (x - 0.5) / 1.0
            else:
                h[i] = 1.0
        h = repair_h(h, n_points)
        best_c5 = evaluate_h(h, dx)
        best_h = h
    
    return (best_h.tolist(), best_c5, n_points)
