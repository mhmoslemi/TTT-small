import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a smoother waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a smoothly varying function
        h_init = 0.3 + 0.4 * np.sin(np.pi * x) + 0.2 * np.cos(2 * np.pi * x)
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # CCCP Phase
    learning_rate = 0.01
    max_iter_ccp = 5000
    patience = 500
    last_improvement = 0
    
    for it in range(max_iter_ccp):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find k_max where the maximum overlap occurs
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        # The maximum correlation value
        max_corr = corr[k_max]
        
        # Approximate the gradient with current h and the shift k_max
        # For given x, the contribution to C5 is h(x)(1 - h(x+k_max))
        # Gradient w.r.t. h[i] is (1 - h[i+k_max]) where we consider periodic boundaries
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = 1 - h_current[j]
        
        # Update h using gradient descent with learning rate
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse moves
            if np.random.rand() < 0.01:
                h_current = h_new.copy()
                c5_current = c5_new
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = it
            
        # Adapt learning rate
        if it - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = it
            
        # Perturb to escape local minima
        if it % 500 == 0 and it > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Coordinate Descent Phase
    learning_rate_cd = 0.02
    max_iter_cd = 10000
    patience_cd = 1000
    last_improvement_cd = 0
    
    for it in range(max_iter_cd):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Loop through each h[i] cyclically
        for i in range(n_points):
            # Recompute indices for the shift k_max
            corr = correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr)
            
            # Compute the gradient component for h[i]
            j = (i + k_max) % n_points
            grad_i = 1 - h_current[j]
            
            # Update h[i]
            new_h_i = h_current[i] - learning_rate_cd * grad_i
            h_current[i] = new_h_i
            
            # Clipping and renormalization
            h_current[i] = np.clip(h_current[i], 0, 1)
            h_current = project_h(h_current, n_points)
            
        # Evaluate after each full cycle
        c5_new = evaluate_h(h_current, dx)
        
        if c5_new < c5_current:
            h_current = h_current.copy()
            c5_current = c5_new
        else:
            if np.random.rand() < 0.005:
                h_current = h_current.copy()
                c5_current = c5_new
                
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement_cd = it
            
        # Adapt learning rate
        if it - last_improvement_cd > patience_cd:
            learning_rate_cd *= 0.97
            last_improvement_cd = it
            
        # Perturbation
        if it % 1000 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    return (h_best.tolist(), c5_best, n_points)
