import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size):
    population = []
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Fill up the rest with triangular pulses
    for i in range(pop_size - len(population)):
        h = np.zeros(n_points)
        # Create triangular pulse centered at 1.0
        for j in range(n_points):
            x = j * 2.0 / n_points
            if abs(x - 1.0) < 0.5:
                h[j] = 1.0 - 2.0 * abs(x - 1.0)
        h = repair_h(h, n_points)
        population.append(h)
        
    return population

def selection(population, fitnesses):
    # Tournament selection
    selected = []
    for _ in range(len(population)):
        idx1, idx2 = np.random.choice(len(population), 2, replace=False)
        if fitnesses[idx1] < fitnesses[idx2]:
            selected.append(population[idx1].copy())
        else:
            selected.append(population[idx2].copy())
    return selected

def crossover(parent1, parent2):
    # Single-point crossover
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n_points = len(h)
    mutation_indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
    for i in mutation_indices:
        h[i] += np.random.normal(0, 0.02)
    h = repair_h(h, len(h))
    return h

def genetic_algorithm(n_points, pop_size, max_generations, mutation_rate, budget_s):
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    
    best_h = None
    best_fitness = float('inf')
    no_improvement_count = 0
    patience = 200
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_h(h, 2.0/n_points) for h in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= patience:
            break
            
        # Selection
        selected = selection(population, fitnesses)
        
        # Generate new population
        new_population = [best_h.copy()]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            if np.random.rand() < 0.5:  # Crossover
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                new_population.extend([child1, child2])
            else:
                child = mutate(parent1, mutation_rate)
                new_population.append(child)
                
        population = new_population[:pop_size]
        
    return best_h, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Use genetic algorithm with smaller population and fewer generations
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.2
    
    h_best, c5_best = genetic_algorithm(n_points, pop_size, max_generations, mutation_rate, budget_s)
    
    if h_best is None:
        # Fallback to simple initialization
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)
        h_best = h_init
        c5_best = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), c5_best, n_points)
