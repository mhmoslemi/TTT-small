import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 5000
    backtracking_factor = 0.5
    initial_step_size = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure constraints initially
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum) > 1e-10:
        h = h * (n_points / 2) / h_sum
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iterations):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current C5
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient approximation
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus_sum = np.sum(h_plus)
            h_plus = h_plus * (n_points / 2) / h_plus_sum
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Compute descent direction via Frank-Wolfe direction
        # This approximates argmin_s <grad, s> subject to constraints
        # We'll solve this approximately by taking the negative of gradient and projecting
        # This is approximate and somewhat heuristic but practical and fast
        
        s = -grad
        s = np.clip(s, 0, 1)
        
        # Rescale s to sum to n_points/2
        s_sum = np.sum(s)
        if abs(s_sum) > 1e-10:
            s = s * (n_points / 2) / s_sum
        
        # Line search
        gamma = initial_step_size
        while gamma > 1e-10:
            h_new = (1 - gamma) * h + gamma * s
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if abs(h_new_sum) > 1e-10:
                h_new = h_new * (n_points / 2) / h_new_sum
            
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            new_c5 = np.max(corr_new) * dx
            
            if new_c5 <= current_c5:
                h = h_new
                break
            gamma *= backtracking_factor
        
        if gamma <= 1e-10:
            # If we can't make progress, try small random changes
            h = h + 0.001 * (np.random.rand(n_points) - 0.5)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum) > 1e-10:
                h = h * (n_points / 2) / h_sum
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
