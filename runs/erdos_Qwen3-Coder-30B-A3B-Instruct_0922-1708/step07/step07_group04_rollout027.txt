import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Simulated Annealing with global search
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Annealing parameters
    temp = 0.1
    cooling_rate = 0.95
    max_iter = 50000
    patience = 500
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate neighbor solution via random perturbation
        h_new = h_current.copy()
        # Perturb a random set of points
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for idx in indices:
            # Add small random change
            delta = np.random.normal(0, 0.05)
            h_new[idx] += delta
        h_new = repair_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / temp):
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                temp = max(0.001, temp * cooling_rate)
        
        # Decrease temperature
        if i - last_improvement > patience:
            temp = max(0.001, temp * cooling_rate)
            last_improvement = i
    
    # Refine with gradient-based method (lower learning rate)
    eta = 0.005
    eps = 1e-6
    max_iter_refine = 10000
    
    for i in range(max_iter_refine):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = h_current.copy()
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            h_new[j] = h_current[j] - eta * grad_j
            
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
    
    return (h_best.tolist(), c5_best, n_points)
