import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter_barrier = 1000
    max_iter_gd = 10000
    learning_rate = 1e-3
    barrier_decrease_factor = 0.99
    μ_initial = 1e3
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Barrier method
    μ = μ_initial
    for iter_barrier in range(max_iter_barrier):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        c5 = corr[max_idx] * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute exact gradient via adjoint method
        max_lag = max_idx - (len(h) - 1)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i - max_lag >= 0 and i - max_lag < len(h):
                grad[i] += 1 - h[i - max_lag]
        
        # Update with gradient and barrier penalty
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
            
        μ *= barrier_decrease_factor
    
    # Gradient descent phase
    for iter_gd in range(max_iter_gd):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        c5 = corr[max_idx] * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Approximate gradient with finite differences
        grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum - target_sum) > 1e-8:
                h_plus = h_plus * target_sum / h_plus_sum
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
            
        # Update
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
