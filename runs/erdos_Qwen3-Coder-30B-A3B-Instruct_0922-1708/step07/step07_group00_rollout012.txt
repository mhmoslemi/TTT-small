import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Use scipy.optimize.minimize with L-BFGS-B for better convergence
    def objective(h):
        h = repair_h(h, n_points)
        return evaluate_h(h, dx)
    
    # Constraint function for sum constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    max_iter = 5000
    step_count = 0
    
    # As we're using scipy.optimize, we'll recalculate and refine from a good point
    h_current = current_h.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Use L-BFGS-B to optimize
        res = minimize(objective, h_current, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                       options={'maxiter': 10, 'ftol': 1e-12, 'gtol': 1e-12})
        
        if res.success:
            h_current = res.x
            h_current = repair_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                
            # Small random perturbation to escape local minima
            if i % 50 == 0 and i > 0:
                noise = np.random.normal(0, 1e-5, n_points)
                h_current = h_current + noise
                h_current = repair_h(h_current, n_points)
        
        step_count += 1
        
        # Early stopping for stability
        if step_count % 200 == 0 and step_count > 0:
            if abs(best_c5 - current_c5) < 1e-8:
                break
            current_c5 = best_c5

    # Final check
    h_final = repair_h(best_h, n_points)
    final_c5 = evaluate_h(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)
