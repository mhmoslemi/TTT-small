import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize particles
    m = 100  # number of particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        particles.append(h_init)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
        particles.append(h_init)
    
    # Fill remaining particles with random samples
    for _ in range(m - 1):
        h = np.random.random(n_points)
        h = repair_h(h, n_points)
        particles.append(h)
    
    # Initialize velocities
    for _ in range(m):
        vel = np.random.uniform(-0.1, 0.1, n_points)
        velocities.append(vel)
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    v_max = 0.2
    
    # Track best positions
    pbest = [h.copy() for h in particles]
    pbest_scores = [evaluate_h(h, dx) for h in particles]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO iterations
    max_iter = 10000
    patience = 1000
    last_improvement = 0
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(m):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair to constraints
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            score = evaluate_h(particles[i], dx)
            
            # Update pbest
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    last_improvement = iter_num
        
        if iter_num - last_improvement > patience:
            break
    
    # Gradient refinement
    h_current = gbest.copy()
    c5_current = g5_score = gbest_score
    eta = 0.001
    eps = 1e-6
    max_iter = 500
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_current.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_current[j] - eta * grad_j
            
        # Apply repair step
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept step if better
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < gbest_score:
                gbest = h_current.copy()
                gbest_score = c5_current
    
    return (gbest.tolist(), gbest_score, n_points)
