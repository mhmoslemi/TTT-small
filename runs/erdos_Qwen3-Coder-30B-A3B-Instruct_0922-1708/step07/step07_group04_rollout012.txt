import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, mutation_rate=0.1, perturbation=0.1):
    """Mutate h by perturbing a fraction of entries."""
    h_new = h.copy()
    n_points = len(h)
    n_mutate = max(1, int(n_points * mutation_rate))
    indices = np.random.choice(n_points, n_mutate, replace=False)
    for i in indices:
        h_new[i] += np.random.uniform(-perturbation, perturbation)
    return repair_h(h_new, n_points)

def crossover_h(h1, h2):
    """Crossover two h vectors by shuffling segments."""
    n_points = len(h1)
    cut_point = np.random.randint(1, n_points)
    h_new = np.concatenate([h1[:cut_point], h2[cut_point:]])
    return repair_h(h_new, n_points)

def genetic_algorithm(seed, budget_s, n_points=100):
    np.random.seed(seed)
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Initialize population
    population_size = 100
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h_mut = mutate_h(h_init)
        population.append(h_mut)
    
    # Evolution loop
    generation = 0
    patience = 0
    best_fitness = float('inf')
    best_individual = h_init.copy()
    
    while time.time() - start_time < budget_s - 5 and patience < 50:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fitness = evaluate_h(h, dx)
            fitnesses.append(fitness)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if fitnesses[0] < best_fitness:
            best_fitness = fitnesses[0]
            best_individual = population[0].copy()
            patience = 0
        else:
            patience += 1
        
        # Selection and reproduction
        selected = population[:population_size//2]
        new_population = selected.copy()
        
        # Create offspring
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            h_child = crossover_h(parent1, parent2)
            h_child = mutate_h(h_child)
            new_population.append(h_child)
        
        population = new_population
        generation += 1
    
    return best_individual, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    # Use genetic algorithm to find good solution
    h_best, c5_best = genetic_algorithm(seed, budget_s)
    
    # Refine with gradient steps if time permits
    n_points = len(h_best)
    dx = 2.0 / n_points
    start_time = time.time()
    eta = 0.001
    eps = 1e-6
    
    # Gradient refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
            
        h_refined = h_best.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_best.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            
            # Backward difference
            h_minus = h_best.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_refined[j] = h_best[j] - eta * grad_j
        
        h_refined = repair_h(h_refined, n_points)
        c5_refined = evaluate_h(h_refined, dx)
        
        if c5_refined < c5_best:
            h_best = h_refined
            c5_best = c5_refined
        else:
            # Reduce learning rate if no improvement
            eta *= 0.995
    
    return (h_best.tolist(), c5_best, n_points)
