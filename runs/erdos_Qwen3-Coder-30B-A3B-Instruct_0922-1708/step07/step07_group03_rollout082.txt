import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a different initial construction
        h = np.ones(n_points) * 0.5
        h[20:30] = 1.0  # Spike in middle
        h = h * (n_points / 2) / np.sum(h)
    
    # Frank-Wolfe algorithm
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute approximate gradient - using finite difference
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum) > 1e-10:
                h_plus = h_plus * (n_points / 2) / h_plus_sum
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Solve linear subproblem for direction s (satisfying constraints)
        # We solve argmin_s <grad, s> subject to 0 <= s <= 1, sum(s) = 1/2 * n_points
        # This can be approximated as: take the indices where grad is most negative
        # Construct s as having maximum negative gradient values
        s = np.zeros(n_points)
        sorted_indices = np.argsort(grad)
        # Set the indices with most negative gradients to 1
        num_on = n_points // 2
        s[sorted_indices[:num_on]] = 1.0
        # Normalize to sum to n_points / 2
        s_sum = np.sum(s)
        if abs(s_sum) > 1e-10:
            s = s * (n_points / 2) / s_sum
        
        # Line search for step size gamma
        gamma = 0.5
        c5_new = c5 + 1
        max_backtrack = 10
        backtrack_count = 0
        
        while c5_new >= c5 and backtrack_count < max_backtrack:
            new_h = (1 - gamma) * h + gamma * s
            new_h = np.clip(new_h, 0, 1)
            new_h_sum = np.sum(new_h)
            if abs(new_h_sum) > 1e-10:
                new_h = new_h * (n_points / 2) / new_h_sum
            corr_new = correlate(new_h, 1 - new_h, mode='full')
            c5_new = np.max(corr_new) * dx
            gamma *= 0.5
            backtrack_count += 1
            
        h = (1 - gamma) * h + gamma * s
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
