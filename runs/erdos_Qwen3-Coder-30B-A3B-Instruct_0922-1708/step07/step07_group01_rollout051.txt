import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, k_max, n_points, dx):
    """Compute exact gradient for the maximum overlap term."""
    # Create the shifted version (1-h) for the k_max shift
    h_shifted = np.zeros(n_points)
    if k_max >= 0:
        h_shifted[k_max:] = 1 - h[:-k_max]
    else:
        h_shifted[:k_max] = 1 - h[-k_max:]
    
    # The gradient is (1 - h_k_max) for the correct indices
    gradient = -h_shifted
    # Normalization is not needed for gradient, but ensure feasibility
    return gradient

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

        h_init = project_h(h_init, n_points)
    else:

        h_init = np.zeros(n_points)

        # Create a piecewise constant function that is mostly 0, with a pulse in the middle
        pulse_start = n_points // 3
        pulse_end = 2 * n_points // 3
        h_init[pulse_start:pulse_end] = 1.0
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = int(1e6)
    patience = 100
    last_improvement = 0
    
    # Identify best k for gradient calculation once per iteration
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break

        # Compute full correlation to find maximum overlap index
        corr_full = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr_full) - (n_points - 1)
        max_val = corr_full[np.argmax(corr_full)]
        
        # Compute gradient using only the critical k_max
        grad = compute_gradient(h_current, k_max, n_points, dx)
        
        # Update with gradient where there's gradient information
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept move if improvement
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse moves (simulated annealing)
            if np.random.rand() < 0.01:
                h_current = h_new.copy()
                c5_current = c5_new
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:

        if i % 500 == 0 and i > 0:
