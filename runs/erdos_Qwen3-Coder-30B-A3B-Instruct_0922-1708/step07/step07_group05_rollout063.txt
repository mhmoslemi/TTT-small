import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular pulse
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)

    # PSO parameters
    n_particles = 100
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for i in range(n_particles):
        if i == 0:
            h = h_init.copy()
        else:
            # Perturb initial solution
            h = h_init + np.random.normal(0, 0.05, n_points)
        h = repair_h(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate_h(h, dx))
    
    # Global best
    best_idx = np.argmin(pbest_values)
    gbest = pbest[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    # PSO iteration
    max_iter = 400
    last_improvement = 0
    patience = 100
    
    for iter_ in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * (iter_ / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (pbest[i] - particles[i])
            social = c2 * r2 * (gbest - particles[i])
            noise = np.random.normal(0, mutation_sigma, n_points)
            velocities[i] = w * velocities[i] + cognitive + social + noise
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            current_value = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
                    last_improvement = iter_
        
        # Mutation (every 25 iterations)
        if iter_ % 25 == 0 and iter_ > 0:
            for i in range(int(n_particles * mutation_rate)):
                particle_idx = np.random.randint(0, n_particles)
                # Mutate 20% of values
                indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
                particles[particle_idx][indices] += np.random.normal(0, mutation_sigma, len(indices))
                particles[particle_idx] = repair_h(particles[particle_idx], n_points)
                # Update personal best if needed
                current_value = evaluate_h(particles[particle_idx], dx)
                if current_value < pbest_values[particle_idx]:
                    pbest[particle_idx] = particles[particle_idx].copy()
                    pbest_values[particle_idx] = current_value
                    if current_value < gbest_value:
                        gbest = particles[particle_idx].copy()
                        gbest_value = current_value
                        last_improvement = iter_
        
        if iter_ - last_improvement > patience:
            break
    
    return (gbest.tolist(), gbest_value, n_points)
