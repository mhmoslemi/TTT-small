import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(params, n_points):
    # params = [a1, a2, ..., aK, x2, x3, ..., xK]
    # where x1=0, xK+1=2
    K = (len(params) + 1) // 2
    a = params[:K]
    x = np.concatenate([[0], params[K:], [2]])
    x = np.sort(x)
    x = np.clip(x, 0, 2)
    x[0] = 0
    x[-1] = 2
    
    h = np.zeros(n_points)
    dx_points = 2.0 / n_points
    for i in range(K):
        start_idx = int(x[i] / dx_points)
        end_idx = int(x[i+1] / dx_points)
        if start_idx < n_points:
            h[start_idx:end_idx] = a[i]
    return h

def compute_gradient(params, dx, n_points, k_max=None):
    eps = 1e-5
    grad = np.zeros(len(params))
    h_base = parameterize_h(params, n_points)
    base_c5 = evaluate_h(h_base, dx)
    
    for i in range(len(params)):
        params_plus = params.copy()
        params_plus[i] += eps
        h_plus = parameterize_h(params_plus, n_points)
        h_plus = repair_h(h_plus, n_points)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    return grad

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Parameterize: use 5 segments for low dimensional search
    K = 5
    h_avg = np.mean(h_init)
    x_init = np.linspace(0, 2, K+1)
    x_init[0] = 0
    x_init[-1] = 2
    a_init = np.full(K, h_avg)
    
    # Convert to initial parameters
    initial_params = np.concatenate([a_init, x_init[1:-1]])
    
    # Optimization parameters
    max_iter = 5000
    learning_rate = 1e-3
    decay_rate = 0.995
    
    current_params = initial_params.copy()
    current_h = parameterize_h(current_params, n_points)
    current_h = repair_h(current_h, n_points)
    current_c5 = evaluate_h(current_h, dx)
    best_params = current_params.copy()
    best_c5 = current_c5
    best_h = current_h.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        grad = compute_gradient(current_params, dx, n_points)
        
        # Update step
        current_params = current_params - learning_rate * grad
        current_h = parameterize_h(current_params, n_points)
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_params = current_params.copy()
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Adaptive learning rate
        if i % 100 == 0:
            learning_rate *= decay_rate
    
    # Final refinement using more precise gradient
    final_params = best_params.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        grad = compute_gradient(final_params, dx, n_points)
        final_params = final_params - 1e-4 * grad
        h_refined = parameterize_h(final_params, n_points)
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        c5_refined = np.max(corr) * dx
        
        if c5_refined < best_c5:
            best_params = final_params.copy()
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    return (best_h.tolist(), best_c5, n_points)
