import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimization parameters
    max_iter = 5000
    learning_rate = 1e-3
    decay_rate = 0.999
    
    # Gradient computation with finite differences
    eps = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        base_corr = correlate(current_h, 1 - current_h, mode='full')
        base_max_k = np.argmax(base_corr)
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[j] = (corr_plus[base_max_k] - base_corr[base_max_k]) / eps
            
        # Update step with gradient and learning rate
        current_h = current_h - learning_rate * grad
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Adaptive learning rate
        if i % 100 == 0:
            learning_rate *= decay_rate
    
    # Final refinement with lower learning rate
    h_refined = best_h.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        grad = np.zeros(n_points)
        base_corr = correlate(h_refined, 1 - h_refined, mode='full')
        base_max_k = np.argmax(base_corr)
        
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[j] = (corr_plus[base_max_k] - base_corr[base_max_k]) / eps
            
        h_refined = h_refined - 1e-5 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        c5_refined = np.max(corr) * dx
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
