import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def segment_blending(parent1, parent2, n_points):
    """Create offspring via segment-wise blending."""
    offspring = np.zeros(n_points)
    segments = np.random.randint(1, n_points//4 + 1)
    start_points = np.sort(np.random.choice(n_points - 1, segments, replace=False))
    start_points = np.append(start_points, n_points)
    
    current_segment = 0
    current_parent = np.random.choice([0, 1])
    for i in range(n_points):
        if i >= start_points[current_segment]:
            current_parent = 1 - current_parent
            current_segment += 1
        if current_parent == 0:
            offspring[i] = parent1[i]
        else:
            offspring[i] = parent2[i]
    return offspring

def gaussian_mutation(h, scale):
    """Add Gaussian noise to the solution."""
    noise = np.random.normal(0, scale, len(h))
    h_mutation = h + noise
    return h_mutation

def hill_climb(h_init, n_points, dx, max_iter=50):
    """Fast local hill climb to refine candidates."""
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    eta = 0.01
    eps = 1e-6
    
    for _ in range(max_iter):
        h_new = h_current.copy()
        improved = False
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
        
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            improved = True
            eta = min(0.01, eta * 1.01)
        else:
            eta *= 0.999
            
        if not improved:
            break
    
    return h_current, c5_current

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Memetic algorithm setup
    population_size = 50
    elite_size = 10
    mutation_scale = 0.1
    population = []
    
    # Initialize population
    for _ in range(population_size):
        h = h_init.copy()
        # Add small random perturbation
        if np.random.rand() > 0.5:
            h = gaussian_mutation(h, 0.05)
        h = project_h(h, n_points, dx)
        population.append(h)
    
    # Best solution tracking
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Memetic loop
    gen = 0
    while time.time() - start_time < budget_s - 5:
        gen += 1
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append((c5, h))
        
        # Sort by fitness
        fitness.sort(key=lambda x: x[0])
        
        # Update best
        best_c5, best_h = fitness[0]
        if best_c5 < c5_best:
            h_best = best_h.copy()
            c5_best = best_c5
        
        # Selection - elite
        elite = [h for _, h in fitness[:elite_size]]
        
        # Generate offspring
        offspring = []
        while len(offspring) < population_size - elite_size:
            if np.random.rand() < 0.7:  # Crossover
                parent1, parent2 = np.random.choice(elite, 2, replace=False)
                child = segment_blending(parent1, parent2, n_points)
            else:  # Mutation
                parent = np.random.choice(elite)
                child = gaussian_mutation(parent, mutation_scale)
                child = project_h(child, n_points, dx)
                
            # Local refinement
            child, _ = hill_climb(child, n_points, dx, max_iter=50)
            offspring.append(child)
        
        # Replace worst in population with new offspring
        new_population = elite + offspring
        population = new_population
        
        # Update mutation scale
        if gen % 10 == 0:
            mutation_scale *= 0.99
    
    return (h_best.tolist(), c5_best, n_points)
