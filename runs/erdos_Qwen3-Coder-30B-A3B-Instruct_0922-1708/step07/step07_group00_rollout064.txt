import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Generate multiple random starting points
    starts = [h_init.copy()]

        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        starts.append(h)
        
    # Try all starting points
    for h_start in starts:
        current_h = h_start.copy()
        current_c5 = evaluate_h(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Use scipy.optimize for fine-tuning
        def objective(h_vec):
            h = repair_h(h_vec, n_points)
            return evaluate_h(h, dx)
        
        def constraint(h_vec):
            return np.sum(repair_h(h_vec, n_points)) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        bounds = [(0, 1) for _ in range(n_points)]

        for _ in range(500):

                break
                
            try:

                if res.success:
                    current_h = res.x
                    current_h = repair_h(current_h, n_points)
                    current_c5 = evaluate_h(current_h, dx)
                    
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
            except:
                pass
            
            # Random perturbation
            pert = np.random.normal(0, 1e-4, n_points)
            current_h = current_h + pert
            current_h = repair_h(current_h, n_points)
            current_c5 = evaluate_h(current_h, dx)
