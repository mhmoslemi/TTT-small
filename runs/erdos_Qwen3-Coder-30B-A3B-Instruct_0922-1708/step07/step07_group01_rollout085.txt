import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Generate multiple diverse initializations
    initializations = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
        initializations.append(h_init.copy())
    
    # Add structured initializations
    x = np.linspace(0, 2, n_points)
    
    # Cosine taper
    h_cos = 0.5 * (1 + np.cos(np.pi * x))
    h_cos = project_h(h_cos, n_points)
    initializations.append(h_cos.copy())
    
    # Trapezoidal
    h_trap = np.zeros(n_points)
    mid = n_points // 2
    h_trap[:mid] = np.linspace(0, 1, mid)
    h_trap[mid:] = np.linspace(1, 0, n_points - mid)
    h_trap = project_h(h_trap, n_points)
    initializations.append(h_trap.copy())
    
    # Random with constraints
    h_rand = np.random.rand(n_points)
    h_rand = project_h(h_rand, n_points)
    initializations.append(h_rand.copy())
    
    # Sine wave
    h_sin = 0.5 * (1 + np.sin(np.pi * x))
    h_sin = project_h(h_sin, n_points)
    initializations.append(h_sin.copy())
    
    # Initialize tracking
    h_best = None
    c5_best = float('inf')
    
    # Multi-start optimization
    for init_idx, h_init in enumerate(initializations):
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        # Gradient descent with adaptive learning rate
        learning_rate = 0.01
        max_iter = int(1e5)
        patience = 500
        last_improvement = 0
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Identify the shift k that maximizes overlap
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_k = np.argmax(corr)
            max_overlap = corr[max_k]
            
            # Compute gradient for the maximizer shift
            # Gradient of C5 with respect to h for the k that maximizes it
            grad = np.zeros(n_points)
            # For the maximum shift k, we compute ∂/∂h_i of h_i * (1 - h_{i+k})
            # This is complicated, so we compute it via finite difference
            epsilon = 1e-6
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            
            for j in range(n_points):
                h_plus[j] += epsilon
                h_minus[j] -= epsilon
                
                # Ensure feasibility
                h_plus = project_h(h_plus, n_points)
                h_minus = project_h(h_minus, n_points)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
                
                # Reset for next participant
                h_plus[j] = h_current[j]
                h_minus[j] = h_current[j]
            
            # Update with gradient
            h_new = h_current - learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept or reject
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
            else:
                # Occasionally accept worse moves (simulated annealing)
                if np.random.rand() < 0.005:
                    h_current = h_new.copy()
                    c5_current = c5_new
            
            # Update best
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                # Reset learning rate when we improve
                learning_rate = 0.01
            
            # Adapt learning rate
            if i - last_improvement > patience:
                learning_rate *= 0.95
                last_improvement = i
                
                # Occasionally restore to better solution
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    
        # Periodic perturbations to escape local minima
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_pert = h_best.copy()
            # Add small random noise in 10% of coordinates
            indices = np.random.choice(n_points, max(1, n_points // 10), replace=False)
            for idx in indices:
                h_pert[idx] += np.random.normal(0, 0.002)
            
            h_pert = project_h(h_pert, n_points)
            c5_pert = evaluate_h(h_pert, dx)
            
            if c5_pert < c5_best:
                h_best = h_pert.copy()
                c5_best = c5_pert
    
    # Final refinement with local search
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        indices = np.random.choice(n_points, max(1, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.002)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
