import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 0.5
        h_init = project_h(h_init, n_points, dx)
    
    # PSO parameters
    swarm_size = 50
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_coeff = 1.5
    social_coeff = 1.5
    velocity_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add the initial solution
    swarm.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Add random solutions
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        swarm.append(h)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
    
    # Personal best tracking
    personal_best = [h.copy() for h in swarm]
    personal_best_fitness = [evaluate_h(h, dx) for h in swarm]
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO iteration
    max_iter = 100000
    stagnation_count = 0
    stagnation_threshold = 1000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * i / max_iter
        
        # Update each particle
        for p in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_coeff * r1 * (personal_best[p] - swarm[p])
            social = social_coeff * r2 * (global_best - swarm[p])
            velocities[p] = inertia * velocities[p] + cognitive + social
            velocities[p] = np.clip(velocities[p], -velocity_max, velocity_max)
            
            # Update position
            swarm[p] = swarm[p] + velocities[p]
            
            # Project to feasible region
            swarm[p] = project_h(swarm[p], n_points, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(swarm[p], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[p]:
                personal_best[p] = swarm[p].copy()
                personal_best_fitness[p] = fitness
            
            # Update global best
            if fitness < global_best_fitness:
                global_best = swarm[p].copy()
                global_best_fitness = fitness
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Check for stagnation
        if stagnation_count > stagnation_threshold:
            # Perturb the global best
            perturbed = global_best + np.random.normal(0, 0.01, n_points)
            perturbed = project_h(perturbed, n_points, dx)
            fitness = evaluate_h(perturbed, dx)
            
            if fitness < global_best_fitness:
                global_best = perturbed
                global_best_fitness = fitness
                stagnation_count = 0

    return (global_best.tolist(), global_best_fitness, n_points)
