import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    backtracking_factor = 0.5
    initial_step_size = 0.5
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default initialization
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe with line search
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient using finite differences with eps
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Solve linear subproblem to find descent direction s
        # minimize <grad, s> subject to 0 <= s <= 1, sum(s) = n_points/2
        s = np.zeros(n_points)
        indices = np.argsort(-grad)  # descending order
        remaining_sum = n_points / 2.0
        for i in indices:
            if remaining_sum <= 0:
                break
            if remaining_sum >= 1:
                s[i] = 1
                remaining_sum -= 1
            else:
                s[i] = remaining_sum
                remaining_sum = 0
                
        # Ensure sum is correct
        s = s * (n_points / 2) / np.sum(s)
        
        # Line search
        gamma = initial_step_size
        while gamma > 1e-10:
            h_new = (1 - gamma) * h + gamma * s
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            new_c5 = np.max(corr_new) * dx
            
            if new_c5 < current_c5:
                h = h_new
                break
            else:
                gamma *= backtracking_factor
                
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
