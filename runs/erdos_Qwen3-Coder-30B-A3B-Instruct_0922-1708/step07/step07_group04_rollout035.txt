import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, rate=0.1, delta=0.1):
    """Mutate h by perturbing some entries."""
    h_new = h.copy()
    indices = np.random.choice(len(h), size=int(len(h) * rate), replace=False)
    for i in indices:
        h_new[i] += np.random.uniform(-delta, delta)
    return repair_h(h_new, len(h))

def crossover_h(h1, h2):
    """Crossover two h vectors by shuffling segments."""
    n = len(h1)
    split = np.random.randint(1, n-1)
    h_new = np.concatenate([h1[:split], h2[split:]])
    return repair_h(h_new, n)

def tournament_selection(pop, fitness, k=3):
    """Select an individual via tournament selection."""
    idx = np.random.choice(len(pop), size=k)
    best_idx = idx[np.argmin([fitness[i] for i in idx])]
    return pop[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 100
    mutation_rate = 0.1
    mutation_delta = 0.1
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h_mut = mutate_h(h_init, mutation_rate, mutation_delta)
        population.append(h_mut)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fitness.append(evaluate_h(h, dx))
    
    # Evolution loop
    best_fitness = min(fitness)
    best_individual = population[np.argmin(fitness)].copy()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50% as parents
        parents = population[:population_size//2]
        
        # Create new population through crossover and mutation
        new_population = parents.copy()
        while len(new_population) < population_size:
            parent1 = tournament_selection(parents, fitness)
            parent2 = tournament_selection(parents, fitness)
            child = crossover_h(parent1, parent2)
            child = mutate_h(child, mutation_rate, mutation_delta)
            new_population.append(child)
        
        population = new_population
        
        # Evaluate new population
        fitness = []
        for h in population:
            fitness.append(evaluate_h(h, dx))
            
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_fitness:
            best_fitness = fitness[current_best_idx]
            best_individual = population[current_best_idx].copy()
    
    # Final gradient refinement of best individual
    h_refined = best_individual.copy()
    c5_refined = best_fitness
    
    # Coordinate-wise gradient descent
    eta = 0.001
    eps = 1e-6
    max_iter = 10000
    patience = 50
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite differences
        h_new = h_refined.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_refined.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_refined[j] - eta * grad_j
        
        # Repair and re-evaluate
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_refined:
            h_refined = h_new
            c5_refined = c5_new
            last_improvement = i
        else:
            if i - last_improvement > patience:
                eta *= 0.95
                if eta < 1e-6:
                    break
                last_improvement = i
    
    return (h_refined.tolist(), c5_refined, n_points)
