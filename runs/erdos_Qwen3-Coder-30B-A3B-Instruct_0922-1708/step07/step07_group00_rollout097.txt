import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(params, n_points):
    """Convert parameters to h(x)."""
    K = len(params) // 2
    a = params[:K]
    x = np.concatenate([[0], params[K:]])

    # Ensure boundaries are in proper order
    x = np.sort(np.clip(x, 0, 2))
    x = np.concatenate([x, [2]])
    
    # Compute h values
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(K):
        start_idx = int(np.round(x[i] / dx))
        end_idx = int(np.round(x[i+1] / dx))
        if start_idx < end_idx:
            h[start_idx:end_idx] = a[i]
            
    return h

def compute_gradient(params, dx, n_points):
    """Compute gradient of max overlap w.r.t. parameters."""
    n = len(params)
    h = parameterize_h(params, n_points)
    h = repair_h(h, n_points)
    
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    
    eps = 1e-6
    grad = np.zeros(n)
    
    for i in range(n):
        params_plus = params.copy()
        params_plus[i] += eps
        h_plus = parameterize_h(params_plus, n_points)
        h_plus = repair_h(h_plus, n_points)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    # Normalize
    norm = np.linalg.norm(grad) + 1e-12
    grad = grad / norm
    
    return grad
