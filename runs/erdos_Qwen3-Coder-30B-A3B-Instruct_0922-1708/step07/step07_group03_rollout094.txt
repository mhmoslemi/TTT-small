import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    # Add more random particles
    for _ in range(n_particles - len(particles)):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Simulated Annealing parameters
    T = 1.0
    cooling_rate = 0.999
    
    # Initialize personal and global bests
    personal_bests = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * len(particles)
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate all particles
        for i, particle in enumerate(particles):
            # Compute correlation
            corr = correlate(particle, 1 - particle, mode='full')
            score = np.max(corr) * dx
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_bests[i] = particle.copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = particle.copy()
        
        # Update velocities and positions
        for i in range(len(particles)):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            if i < len(particles) and i < len(personal_bests) and i < len(global_best):
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (personal_bests[i] - particles[i]) + 
                                c2 * r2 * (global_best - particles[i]))
            
            # Apply SA: perturb velocity with probability T
            if np.random.rand() < T:
                velocities[i] += np.random.uniform(-0.1, 0.1, n_points)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair constraints
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to maintain sum constraint
            h_sum = np.sum(particles[i])
            if abs(h_sum - n_points / 2) > 1e-8:
                particles[i] = particles[i] * (n_points / 2) / h_sum
                
        # Decrease temperature
        T = max(T * cooling_rate, 0.01)
    
    # Final evaluation
    corr = correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (global_best.tolist(), final_c5, n_points)
