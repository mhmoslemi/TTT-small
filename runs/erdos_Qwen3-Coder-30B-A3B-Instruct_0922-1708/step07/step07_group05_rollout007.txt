import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size):
    population = []
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Fill the rest with triangular constructions
    for i in range(pop_size - len(population)):
        h = np.zeros(n_points)
        # Create a triangular pulse with peak at x=1
        for j in range(n_points):
            x = j * (2.0 / n_points)
            if 0.5 <= x <= 1.5:
                h[j] = 1.0 - abs(x - 1.0)
            else:
                h[j] = 0.0
        h = repair_h(h, n_points)
        population.append(h)
    
    return population

def selection(population, fitnesses):
    # Tournament selection
    selected = []
    n_pop = len(population)
    for _ in range(n_pop // 2):
        # Tournament size = 3
        indices = np.random.choice(n_pop, 3, replace=False)
        tournament_fitness = [fitnesses[i] for i in indices]
        winner_idx = indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx])
    return selected

def crossover(parent1, parent2):
    # Single point crossover
    n_points = len(parent1)
    point = np.random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    h = h.copy()
    n_points = len(h)
    n_mutate = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, n_mutate, replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.05)
    return h

def genetic_algorithm(n_points, pop_size, generations, mutation_rate, seed=42):
    np.random.seed(seed)
    dx = 2.0 / n_points
    
    # Initialize
    population = initialize_population(n_points, pop_size)
    
    best_fitness = float('inf')
    best_individual = None
    
    for gen in range(generations):
        # Evaluate fitness
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Track best solution
        min_fitness = min(fitnesses)
        if min_fitness < best_fitness:
            best_fitness = min_fitness
            best_individual = population[np.argmin(fitnesses)].copy()
        
        # Selection
        selected = selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = []
        # Keep best individual (elitism)
        new_population.append(best_individual.copy())
        
        for i in range(0, len(selected) - 1, 2):
            p1, p2 = selected[i], selected[i+1]
            c1, c2 = crossover(p1, p2)
            c1 = repair_h(mutate(c1, mutation_rate), n_points)
            c2 = repair_h(mutate(c2, mutation_rate), n_points)
            new_population.extend([c1, c2])
        
        # Ensure population size
        while len(new_population) < pop_size:
            h = np.random.rand(n_points)
            h = repair_h(h, n_points)
            new_population.append(h)
        population = new_population[:pop_size]
    
    return best_individual, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    
    # If we have limited time, adjust parameters
    remaining_time = budget_s - 10  # Leave buffer
    max_generations = min(500, max(20, int(remaining_time * 10)))  # 500 max, at least 20
    
    # Run GA
    best_h, best_fitness = genetic_algorithm(n_points, pop_size=50, generations=max_generations, 
                                           mutation_rate=0.2, seed=seed)
    
    # Final check and return
    best_h = repair_h(best_h, n_points)
    final_fitness = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_fitness, n_points)
