import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a structured construction
        h_init = np.zeros(n_points)
        # Create a piecewise linear function that is 0 on [0,0.5) and 1 on (1.5,2]
        # and slopes linearly in between
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0

        # Repair to sum to 1
        h_init = repair_h(h_init, n_points)

    # PSO parameters
    n_particles = 100
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    max_iter = 400
    patience = 100
    last_improvement = 0
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest_idx = 0
    gbest_val = float('inf')
    
    for i in range(n_particles):
        if i == 0:
            h = h_init.copy()
        else:
            # Add small random perturbation
            h = h_init + np.random.normal(0, 0.01, n_points)
        h = repair_h(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        
        c5 = evaluate_h(h, dx)
        pbest.append(h.copy())
        pbest_vals.append(c5)
        
        if c5 < gbest_val:
            gbest_val = c5
            gbest_idx = i
    
    # PSO main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_count / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (pbest[gbest_idx] - particles[i])
            )
            
            # Add mutation noise
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_sigma, n_points)
                velocities[i] += noise
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            c5 = evaluate_h(particles[i], dx)
            
            # Update personal best
            if c5 < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = c5
                
                # Update global best
                if c5 < gbest_val:
                    gbest_val = c5
                    gbest_idx = i
                    last_improvement = iter_count
        
        # Early termination if no improvement
        if iter_count - last_improvement > patience:
            break
    
    return (particles[gbest_idx].tolist(), gbest_val, n_points)
