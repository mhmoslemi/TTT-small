import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe algorithm with line search
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current C5
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient (approximate by finite differences)
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Solve linear subproblem to find descent direction s
        s = np.zeros(n_points)
        
        # Direction is chosen to minimize gradient * s subject to constraints
        # Modify signs of gradient to find direction
        # Simple greedy - assign regions with most negative gradient to be 1
        # This is a heuristic to avoid complex LP solver
        sorted_indices = np.argsort(-grad)  # descending order
        s_cumsum = 0
        for idx in sorted_indices:
            if s_cumsum < n_points / 2:
                s[idx] = 1
                s_cumsum += 1
            else:
                break
        
        if np.sum(s) == 0:
            s = np.ones(n_points)
            s = s * (n_points / 2) / np.sum(s)
        else:
            s = s * (n_points / 2) / np.sum(s)
            
        # Perform line search
        gamma = 0.5
        max_backtrack = 10
        for _ in range(max_backtrack):
            h_new = (1 - gamma) * h + gamma * s
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if abs(h_new_sum - n_points / 2) > 1e-8:
                h_new = h_new * (n_points / 2) / h_new_sum
            
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            new_c5 = np.max(corr_new) * dx
            
            if new_c5 < current_c5:
                h = h_new
                break
            else:
                gamma *= 0.5
                
    # Final evaluation
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
