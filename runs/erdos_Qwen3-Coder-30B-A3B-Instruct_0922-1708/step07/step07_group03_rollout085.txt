import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create a binary initialization with correct sum
        h = np.ones(n_points // 2)
        h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=0)
        # Randomly shuffle
        np.random.shuffle(h)
        # Scale to meet constraint
        h = h * (n_points / 2) / np.sum(h)
        
    # Current best
    best_h = h.copy()
    corr = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Simulated Annealing parameters
    temperature = 0.1
    cooling_rate = 0.995
    min_temperature = 1e-8
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate a neighbor solution
        h_new = h.copy()
        
        # Choose two indices to perturb
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Small random perturbation
        delta = np.random.uniform(-0.05, 0.05)
        
        # Apply perturbation maintaining sum constraint
        if i != j:
            # Adjust h[i] and h[j] symmetrically to maintain sum
            h_new[i] += delta
            h_new[j] -= delta
        
        # Project back to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet constraint sum(h) = n_points / 2
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h_new = h_new * target_sum / h_sum
        
        # Compute C5 for new solution
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Metropolis acceptance criterion
        if c5_new < best_c5 or np.random.random() < np.exp(-(c5_new - best_c5) / temperature):
            h = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Cool down temperature
        temperature = max(min_temperature, temperature * cooling_rate)
    
    # Final evaluation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
