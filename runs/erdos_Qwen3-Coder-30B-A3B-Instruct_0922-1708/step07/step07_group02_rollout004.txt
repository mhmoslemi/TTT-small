import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # PSO parameters
    swarm_size = 50
    max_iter = 20000
    inertia_w = 0.9
    inertia_decay = (0.9 - 0.4) / max_iter
    cognitive_c = 1.5
    social_c = 1.5
    velocity_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Add one particle from initial solution
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    pbest_positions.append(h_init.copy())
    pbest_scores.append(evaluate_h(h_init, dx))
    
    # Add remaining particles
    for _ in range(swarm_size - 1):
        # Semi-structured initialization to diversify
        h_random = np.random.rand(n_points)
        h_random = project_h(h_random, n_points, dx)
        particles.append(h_random)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
        pbest_positions.append(h_random.copy())
        pbest_scores.append(evaluate_h(h_random, dx))
    
    # Global best tracking
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        inertia_w -= inertia_decay
        
        for i in range(swarm_size):
            # Update particle using PSO equations
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_w * velocities[i] + 
                            cognitive_c * r1 * (pbest_positions[i] - particles[i]) + 
                            social_c * r2 * (gbest_position - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project position to feasibility
            particles[i] = project_h(particles[i], n_points, dx)
            
            # Evaluate fitness
            score = evaluate_h(particles[i], dx)
            
            # Update personal best if improved
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = particles[i].copy()
                
                # Update global best if improved
                if score < gbest_score:
                    gbest_score = score
                    gbest_position = particles[i].copy()
                    gbest_idx = i
        
        # Optional: local search around best solution occasionally
        if iteration % 500 == 0 and iteration > 0:
            h_local = gbest_position.copy()
            for _ in range(100):
                idx = np.random.randint(n_points)
                delta = np.random.uniform(-0.05, 0.05)
                h_new = h_local.copy()
                h_new[idx] += delta
                h_new = project_h(h_new, n_points, dx)
                c5_new = evaluate_h(h_new, dx)
                if c5_new < gbest_score:
                    h_local = h_new
                    gbest_position = h_local.copy()
                    gbest_score = c5_new
        
        # Early stopping if improvement is minimal and time is running out
        if iteration > 1000 and iteration % 1000 == 0:
            if gbest_score < 0.3808:
                break
    
    # Return final best solution
    return (gbest_position.tolist(), gbest_score, n_points)
