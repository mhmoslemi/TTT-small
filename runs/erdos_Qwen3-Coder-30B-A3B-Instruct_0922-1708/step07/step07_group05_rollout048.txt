import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidates(n_points, n_candidates=50):
    candidates = []
    for _ in range(n_candidates):
        # Generate random fourier basis with 3-5 modes
        n_modes = np.random.randint(3, 6)
        modes = []
        for _ in range(n_modes):
            k = np.random.randint(1, 10)
            a = np.random.uniform(-1, 1)
            b = np.random.uniform(-1, 1)
            modes.append((k, a, b))
        
        # Create candidate
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for k, a, b in modes:
            h += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
        
        # Scale and shift to [0,1]
        h_min, h_max = np.min(h), np.max(h)
        if h_max > h_min:
            h = (h - h_min) / (h_max - h_min)
        h = np.clip(h, 0, 1)
        
        # Repair to sum constraint
        h = repair_h(h, n_points)
        candidates.append(h)
    
    return candidates

def gradient_step(h, dx, epsilon=1e-6):
    n_points = len(h)
    grad = np.zeros(n_points)
    c5_base = evaluate_h(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += epsilon
        h_minus[i] -= epsilon
        h_plus = repair_h(h_plus, n_points)
        h_minus = repair_h(h_minus, n_points)
        
        c5_plus = evaluate_h(h_plus, dx)
        c5_minus = evaluate_h(h_minus, dx)
        grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        h_init = np.zeros(n_points)
        # Create a piecewise linear function
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)

    # Phase 1: Generate diverse candidates
    candidates = generate_fourier_candidates(n_points, 50)
    candidates.append(h_init)
    
    # Evaluate all candidates
    candidate_scores = []
    for i, h in enumerate(candidates):
        c5 = evaluate_h(h, dx)
        candidate_scores.append((c5, h, i))
    
    # Sort by score and keep top 10
    candidate_scores.sort()
    top_candidates = candidate_scores[:10]
    
    # Phase 2: Refine each top candidate with gradient descent
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    patience = 200
    for c5_score, h_current, idx in top_candidates:
        if time.time() - start_time > budget_s - 5:
            break
            
        h_best = h_current.copy()
        c5_best = c5_score
        
        # Refinement loop
        learning_rate = 0.01
        last_improvement = 0
        
        for i in range(2000):
            if time.time() - start_time > budget_s - 5:
                break
                
            grad = gradient_step(h_current, dx)
            
            h_new = h_current - learning_rate * grad
            h_new = repair_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
            else:
                # Simulated annealing acceptance
                if np.random.rand() < np.exp(-(c5_new - c5_current) / (c5_current * 0.01)):
                    h_current = h_new
                    c5_current = c5_new
            
            if i - last_improvement > patience:
                learning_rate *= 0.95
                if learning_rate < 1e-6:
                    break
        
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
    
    return (best_h.tolist(), best_c5, n_points)
