import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, population_size):
    population = []
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        if len(h_start) != n_points:
            x_orig = np.linspace(0, 2, len(h_start))
            x_target = np.linspace(0, 2, n_points)
            h_start = np.interp(x_target, x_orig, h_start)
        h_start = repair_h(h_start, n_points)
        population.append(h_start)
    
    # Fill rest of population with triangular pulses
    for _ in range(population_size - len(population)):
        h = np.zeros(n_points)
        peak_pos = np.random.uniform(0.5, 1.5)
        for i in range(n_points):
            x = i * 2.0 / n_points
            if abs(x - peak_pos) < 0.5:
                h[i] = 1.0 - 2.0 * abs(x - peak_pos)
        h = repair_h(h, n_points)
        population.append(h)
    
    return population

def selection(population, fitness):
    # Tournament selection
    selected = []
    for _ in range(len(population) // 2):
        i1, i2 = np.random.choice(len(population), size=2, replace=False)
        if fitness[i1] < fitness[i2]:
            selected.append(population[i1])
        else:
            selected.append(population[i2])
    return selected

def crossover(parent1, parent2):
    # Uniform crossover
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(h, mutation_rate=0.2):
    # Add small noise to elements
    mutated = h.copy()
    n_mutate = int(len(h) * mutation_rate)
    indices = np.random.choice(len(h), size=n_mutate, replace=False)
    for i in indices:
        mutated[i] += np.random.normal(0, 0.05)
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Parameters
    population_size = 50
    generations = 1000
    mutation_rate = 0.2
    elite_size = 1
    
    # Initialize population
    population = initialize_population(n_points, population_size)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_h(h, dx)
        fitness.append(fit)
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection
        selected = selection(population, fitness)
        
        # Create new population
        new_population = [best_h]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if len(selected) >= 2:
                p1, p2 = np.random.choice(selected, size=2, replace=False)
                c1, c2 = crossover(p1, p2)
                c1 = mutate(c1, mutation_rate)
                c2 = mutate(c2, mutation_rate)
                c1 = repair_h(c1, n_points)
                c2 = repair_h(c2, n_points)
                new_population.extend([c1, c2])
            else:
                # If no selection, create new random individuals
                h1 = np.zeros(n_points)
                peak_pos = np.random.uniform(0.5, 1.5)
                for i in range(n_points):
                    x = i * 2.0 / n_points
                    if abs(x - peak_pos) < 0.5:
                        h1[i] = 1.0 - 2.0 * abs(x - peak_pos)
                h1 = repair_h(h1, n_points)
                new_population.append(h1)
        
        # Trim to exact size
        new_population = new_population[:population_size]
        population = new_population
        
        # Evaluate and update best
        fitness = []
        for h in population:
            fit = evaluate_h(h, dx)
            fitness.append(fit)
            
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    return (best_h.tolist(), best_c5, n_points)
