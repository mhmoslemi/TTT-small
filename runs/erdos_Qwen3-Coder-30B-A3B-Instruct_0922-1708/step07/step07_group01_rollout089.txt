import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Multiple initializations
    initializations = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
        initializations.append(h_init)
    
    # Add structured initializations
    # Cosine taper
    x = np.linspace(0, 2, n_points)
    h_cos = 0.5 * (1 + np.cos(np.pi * x))
    h_cos = project_h(h_cos, n_points)
    initializations.append(h_cos)
    
    # Step function
    h_step = np.zeros(n_points)
    mid = n_points // 2
    h_step[:mid] = 1
    h_step = project_h(h_step, n_points)
    initializations.append(h_step)
    
    # Random initialization
    h_rand = np.random.rand(n_points)
    h_rand = project_h(h_rand, n_points)
    initializations.append(h_rand)
    
    # Flip peaks
    h_flip = np.zeros(n_points)
    h_flip[:n_points//4] = 1
    h_flip[n_points//4:3*n_points//4] = 0.5
    h_flip[3*n_points//4:] = 1
    h_flip = project_h(h_flip, n_points)
    initializations.append(h_flip)
    
    # Initialize tracking
    h_best = None
    c5_best = float('inf')
    
    # Optimization with multiple starting points
    for h_init in initializations:
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
        
        # Gradient descent with adaptive learning rate
        learning_rate = 0.01
        max_iter = int(5e5)
        patience = 200
        last_improvement = 0
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute exact gradient for current maximum overlap
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            # The shift that gives maximum overlap
            shift = max_idx - (n_points - 1)
            # Gradient for the shift that achieves maximum
            h_shifted = np.roll(h_current, shift)
            grad = -(1 - h_shifted)
            
            # Update using gradient
            h_new = h_current + learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept move if improvement or occasionally accept worse moves
            if c5_new < c5_current or np.random.rand() < 0.05:
                h_current = h_new.copy()
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
            else:
                # Occasionally accept worse moves to escape local minima
                if np.random.rand() < 0.01:
                    h_current = h_new.copy()
                    c5_current = c5_new
            
            # Adaptive learning rate update
            if i - last_improvement > patience:
                learning_rate *= 0.98
                last_improvement = i
                
            # Perturb occasionally to aid exploration
            if i % 200 == 0 and i > 0:
                noise = np.random.normal(0, 0.005, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
    
    # Final refinement
    for _ in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.003)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
        
    return (h_best.tolist(), c5_best, n_points)
