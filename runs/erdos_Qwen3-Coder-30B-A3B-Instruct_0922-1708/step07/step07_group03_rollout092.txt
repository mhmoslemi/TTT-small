import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize particles
    S = 50  # swarm size
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    T = 1.0  # initial temperature
    cooling_rate = 0.999  # cooling rate for simulated annealing
    
    # Initialize swarm
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
    
    # Initialize remaining particles randomly
    for _ in range(S - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
    
    # Initialize personal bests and global best
    personal_bests = [None] * S
    personal_best_values = [float('inf')] * S
    global_best = None
    global_best_value = float('inf')
    
    # PSO-SA optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate each particle
        for i in range(S):
            h = particles[i]
            
            # Compute correlation
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < personal_best_values[i] or personal_best_values[i] is None:
                personal_bests[i] = h.copy()
                personal_best_values[i] = c5
                
                # Update global best
                if c5 < global_best_value:
                    global_best = h.copy()
                    global_best_value = c5
        
        # Update velocities and positions
        for i in range(S):
            if personal_bests[i] is None:
                continue
                
            # PSO velocity update
            r1, r2 = np.random.rand(2)
            v = w * velocities[i] + c1 * r1 * (personal_bests[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            
            # Apply simulated annealing
            if np.random.rand() < T:
                v += np.random.uniform(-0.1, 0.1, n_points)
            
            velocities[i] = v
            new_pos = particles[i] + v
            
            # Repair constraints
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * (n_points / 2) / np.sum(new_pos)
            
            particles[i] = new_pos
        
        # Update temperature
        T = max(T * cooling_rate, 0.01)
    
    # Final evaluation of the best solution
    if global_best is not None:
        corr = correlate(global_best, 1 - global_best, mode='full')
        final_c5 = np.max(corr) * dx
        return (global_best.tolist(), final_c5, n_points)
    else:
        # Fallback to initial construction if available
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_target = np.linspace(0, 2, n_points)
                h_init = np.interp(x_target, x_orig, h_init)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            corr = correlate(h_init, 1 - h_init, mode='full')
            final_c5 = np.max(corr) * dx
            return (h_init.tolist(), final_c5, n_points)
        else:
            # Fallback to a random construction
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            corr = correlate(h, 1 - h, mode='full')
            final_c5 = np.max(corr) * dx
            return (h.tolist(), final_c5, n_points)
