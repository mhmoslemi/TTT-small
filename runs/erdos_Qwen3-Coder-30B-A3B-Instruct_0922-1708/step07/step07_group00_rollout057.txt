import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    swarm_size = 100
    max_iter = 2000
    w_start = 0.9
    w_end = 0.1
    c1 = 2.0
    c2 = 2.0
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    gbest_idx = 0
    gbest_score = float('inf')
    
    for _ in range(swarm_size):
        # Start from a perturbed version of the initial solution
        h = h_init.copy() + np.random.normal(0, 0.01, n_points)
        h = repair_h(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_h(h, dx)
        pbest.append(h.copy())
        pbest_scores.append(score)
        if score < gbest_score:
            gbest_score = score
            gbest_idx = len(particles) - 1
    
    best_h = particles[gbest_idx].copy()
    best_c5 = gbest_score
    
    # PSO main loop
    for iter_idx in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia
        w = w_start - (w_start - w_end) * iter_idx / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (particles[gbest_idx] - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest_score = score
                    gbest_idx = i
                    best_h = particles[i].copy()
                    best_c5 = score
        
        # Local refinement every 100 iterations
        if iter_idx % 100 == 0 and iter_idx > 0:
            # Gradient-based refinement of gbest
            h_refined = best_h.copy()
            for _ in range(50):
                if time.time() - start_time > budget_s - 5:
                    break
                # Compute approximate gradient using finite differences
                eps = 1e-5
                grad = np.zeros(n_points)
                base_score = evaluate_h(h_refined, dx)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    score_plus = evaluate_h(h_plus, dx)
                    grad[j] = (score_plus - base_score) / eps
                h_refined = h_refined - 1e-4 * grad
                h_refined = repair_h(h_refined, n_points)
                new_score = evaluate_h(h_refined, dx)
                if new_score < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = new_score
    
    # Final polishing
    h_final = best_h.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-5
        grad = np.zeros(n_points)
        base_score = evaluate_h(h_final, dx)
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            score_plus = evaluate_h(h_plus, dx)
            grad[i] = (score_plus - base_score) / eps
        h_final = h_final - 1e-4 * grad
        h_final = repair_h(h_final, n_points)
        
        new_score = evaluate_h(h_final, dx)
        if new_score < best_c5:
            best_h = h_final.copy()
            best_c5 = new_score
    
    return (best_h.tolist(), best_c5, n_points)
