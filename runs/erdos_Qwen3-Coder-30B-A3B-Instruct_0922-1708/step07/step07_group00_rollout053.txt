import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Generate multiple initial points using Sobol sampling
    sampler = qmc.Sobol(d=n_points, seed=seed)
    initial_points = sampler.random(n=500)
    
    # Optimization parameters
    max_iter_nm = 1000
    max_iter_refine = 1000
    learning_rate = 1e-5
    decay_rate = 0.99
    
    for i in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Get the next initialization
        h_init = initial_points[i]
        # Scale to [0,1] and then rescale to required sum
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Apply Nelder-Mead optimization
        result = minimize(evaluate_h, h_init, args=(dx,), method='Nelder-Mead', 
                          options={'maxiter': max_iter_nm, 'xatol': 1e-6, 'fatol': 1e-6})
        h_opt = result.x.copy()
        
        # Refine with gradient descent
        h_refined = h_opt.copy()
        step_count = 0
        for _ in range(max_iter_refine):
            if time.time() - start_time > budget_s - 10:
                break
            
            # Simple finite difference gradient computation
            eps = 1e-6
            grad = np.zeros(n_points)
            corr_base = correlate(h_refined, 1 - h_refined, mode='full')
            max_k_idx = np.argmax(corr_base)
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                grad[j] = (corr_plus[max_k_idx] - corr_base[max_k_idx]) / eps
            
            h_refined = h_refined - learning_rate * grad
            h_refined = repair_h(h_refined, n_points)
            current_c5 = evaluate_h(h_refined, dx)
            
            if current_c5 < best_c5:
                best_h = h_refined.copy()
                best_c5 = current_c5
            
            # Adaptive learning rate
            step_count += 1
            if step_count % 100 == 0:
                learning_rate *= decay_rate
    
    return (best_h.tolist(), best_c5, n_points)
