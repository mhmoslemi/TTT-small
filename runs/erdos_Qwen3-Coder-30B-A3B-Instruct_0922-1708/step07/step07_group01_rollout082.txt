import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current

    # Use a hybrid approach: local search with systematic perturbations
    max_iter = int(1e6)
    patience = 200
    last_improvement = 0
    learning_rate = 0.05
    epsilon = 1e-6

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Try gradient-based update
        h_plus = h_current.copy()
        h_minus = h_current.copy()
        grad = np.zeros(n_points)
        
        # Compute finite difference gradients
        for j in range(n_points):
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            h_plus = project_h(h_plus, n_points)
            h_minus = project_h(h_minus, n_points)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Reset for next iteration
            h_plus[j] = h_current[j]
            h_minus[j] = h_current[j]
        
        # Apply gradient update
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if improvement
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse move to escape local minima
            if np.random.rand() < 0.02:
                h_current = h_new.copy()
                c5_current = c5_new
        
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = i
            
        # Periodic perturbation to avoid local minima
        if i % 100 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
            
        # Occasionally restore best
        if i % 500 == 0:
            h_current = h_best.copy()
            c5_current = c5_best

    # Final local refinement
    for _ in range(5000):  # More local searches
        if time.time() - start_time > budget_s - 5:
            break
            
        # Take a small random perturbation to explore
        h_new = h_best.copy()
        indices = np.random.choice(n_points, max(1, n_points // 30), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.001)
            
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new

    # Final evaluation
    final_c5 = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), final_c5, n_points)
