import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    backtracking_factor = 0.5
    initial_gamma = 0.5
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)

    # Frank-Wolfe algorithm
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current C5
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient (approximately via finite differences for direction)
        # Gradient direction
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Compute descent direction s
        # Solve linear subproblem: minimize <grad, s>
        # subject to: 0 <= s <= 1, sum(s) = n_points / 2
        # This is a projection onto the constraint set
        # Simplified: set s = argmin <grad, s> subject to box constraints
        
        # For simplicity, let's try to find s that minimizes gradient dot product
        # This is done by setting s_i = 0 if grad_i > 0, 1 if grad_i < 0, or adjust to meet sum constraint
        s = np.zeros(n_points)
        sorted_indices = np.argsort(grad)
        
        # Build s such that higher negative gradients get higher values
        # First, set all s_i = 1 when grad_i < 0
        for i in range(n_points):
            if grad[sorted_indices[i]] < 0:
                s[sorted_indices[i]] = 1.0
            else:
                break
                
        # Then adjust to enforce sum constraint
        s_sum = np.sum(s)
        if abs(s_sum - n_points / 2) > 1e-6:
            # Need to set some to values in between 0 and 1
            # Set s[i] = 0 when grad[i] >= 0 and s_sum > n_points/2
            # Set s[i] = 1 when grad[i] <= 0 and s_sum < n_points/2
            # We'll treat this by setting some in between to get exact sum
            diff = n_points / 2 - s_sum
            # Adjust the smallest grad we have that still keeps s[i] in [0,1]  
            adjustment = max(0, min(1, diff))
            # Find indices to adjust
            for i in range(n_points):
                if grad[i] >= 0 and s[i] < 1:
                    if diff > 0:
                        s[i] = min(1.0, s[i] + diff)
                        diff -= (s[i] - (s[i]-diff if i < len(s) else 0))
                    elif diff < 0:
                        s[i] = max(0.0, s[i] + diff)
                        diff -= (s[i] - (s[i]-diff if i < len(s) else 0))
                    if abs(diff) < 1e-8:
                        break
                
            # Now also make sure sum is correct  
            s = s * (n_points / 2) / np.sum(s)
            
        # Line search for gamma
        gamma = initial_gamma
        h_trial = (1 - gamma) * h + gamma * s
        # Project onto feasible set
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        corr_trial = correlate(h_trial, 1 - h_trial, mode='full')
        trial_c5 = np.max(corr_trial) * dx
        
        # Backtracking line search
        while trial_c5 >= current_c5 and gamma > 1e-8:
            gamma *= backtracking_factor
            h_trial = (1 - gamma) * h + gamma * s
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            corr_trial = correlate(h_trial, 1 - h_trial, mode='full')
            trial_c5 = np.max(corr_trial) * dx
                
        # Update
        h = h_trial.copy()
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx

    return (best_h.tolist(), final_c5, n_points)
