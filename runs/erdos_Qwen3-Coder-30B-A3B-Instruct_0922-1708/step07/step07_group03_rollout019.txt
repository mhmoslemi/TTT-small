import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.random.rand(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    n_particles = 50
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # SA parameters
    T = 1.0
    cooling_rate = 0.999
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Add initial solution as one particle
    particles.append(h.copy())
    velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
    pbest.append(h.copy())
    corr = correlate(h, 1 - h, mode='full')
    pbest_values.append(np.max(corr) * dx)
    
    # Initialize rest randomly
    for i in range(n_particles - 1):
        h_rand = np.random.rand(n_points) * 0.5
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        particles.append(h_rand)
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
        pbest.append(h_rand.copy())
        corr = correlate(h_rand, 1 - h_rand, mode='full')
        pbest_values.append(np.max(corr) * dx)
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Apply SA perturbation
            if np.random.rand() < T:
                velocities[i] += np.random.rand(n_points) * 0.1 - 0.05
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair constraints
            particles[i] = np.clip(particles[i], 0, 1)
            h_sum = np.sum(particles[i])
            if abs(h_sum - (n_points / 2)) > 1e-8:
                particles[i] = particles[i] * (n_points / 2) / h_sum
            
            # Evaluate
            corr = correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = c5
                
                # Update global best
                if c5 < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = c5
        
        #降温
        T = max(T * cooling_rate, 0.01)
    
    # Final evaluation
    corr = correlate(gbest, 1 - gbest, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (gbest.tolist(), final_c5, n_points)
