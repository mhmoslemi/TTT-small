import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Coordinate-wise optimization with gradient-based updates
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Parameters
    eta = 0.01
    eps = 1e-6
    max_iter = 50000
    patience = 100
    last_improvement = 0
    
    # Use small subset to fit a smoother
    # Sample evenly from large indices to get a better smooth shape
    n_subset = min(40, n_points // 2)
    subset_indices = np.linspace(0, n_points-1, n_subset, dtype=int)
    
    h_current = h_init.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Use a greedy subset approach with rectilinear coordinate descent
        for j in subset_indices:
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_current[j] = h_current[j] - eta * grad_j
            
        # Apply repair step to maintain feasibility
        h_current = repair_h(h_current, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_current, dx)
        
        # Accept if better or with some probability
        if c5_new < c5_current or np.random.rand() < 0.01:
            h_current = h_current.copy()
            c5_current = c5_new
            
            # Update best if improved
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        
        # Adjust learning rate based on progress
        if i - last_improvement > patience:
            eta *= 0.95
            last_improvement = i
    
    return (h_best.tolist(), c5_best, n_points)
