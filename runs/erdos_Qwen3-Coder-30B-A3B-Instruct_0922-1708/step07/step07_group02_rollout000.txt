import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 10000
    inertia_w = 0.9
    inertia_decay = (0.9 - 0.4) / max_iter
    c1, c2 = 1.5, 1.5
    velocity_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    swarm_best = None
    swarm_best_value = float('inf')
    
    # Start with initial solution
    particles.append(h_init.copy())
    pbest.append(h_init.copy())
    pbest_values.append(evaluate_h(h_init, dx))
    if pbest_values[0] < swarm_best_value:
        swarm_best = h_init.copy()
        swarm_best_value = pbest_values[0]
    
    # Initialize remaining particles
    for i in range(1, n_particles):
        # Use structured and random patterns
        particle = np.random.uniform(0, 1, n_points)
        particle = project_h(particle, n_points, dx)
        particles.append(particle)
        pbest.append(particle.copy())
        pbest_values.append(evaluate_h(particle, dx))
        if pbest_values[i] < swarm_best_value:
            swarm_best = particle.copy()
            swarm_best_value = pbest_values[i]
    
    velocities = [np.random.uniform(-velocity_max, velocity_max, n_points) for _ in range(n_particles)]
    
    # PSO loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(n_points), np.random.random(n_points)
            velocities[i] = (inertia_w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (swarm_best - particles[i]))
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            new_pos = particles[i] + velocities[i]
            
            # Apply constraints
            new_pos = project_h(new_pos, n_points, dx)
            
            # Evaluate
            new_value = evaluate_h(new_pos, dx)
            
            # Update personal best
            if new_value < pbest_values[i]:
                pbest[i] = new_pos.copy()
                pbest_values[i] = new_value
                if new_value < swarm_best_value:
                    swarm_best = new_pos.copy()
                    swarm_best_value = new_value
            
            # Update particle
            particles[i] = new_pos
            
        # Update inertia weight
        inertia_w -= inertia_decay
        
    return (swarm_best.tolist(), swarm_best_value, n_points)
