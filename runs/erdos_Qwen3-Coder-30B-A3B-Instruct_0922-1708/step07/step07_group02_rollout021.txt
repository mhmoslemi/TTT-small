import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, k_max, n_points, dx):
    """Compute the gradient of the maximum overlap term."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = i + k_max
        if j < n_points:
            grad[i] = -dx
    return grad

    dx = 2.0 / n_points
    start_time = time.time()

        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with periodic pattern
        h_init = np.zeros(n_points)
        period = n_points // 4
        for i in range(n_points):
            if i % period < period // 2:
                h_init[i] = 0.8
            else:
                h_init[i] = 0.2
        h_init = project_h(h_init, n_points, dx)
    
    # Optimization parameters
    eta = 0.01
    epsilon = 1e-6
    patience = 100
    last_improvement = 0
    
    # Start optimization
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        if iteration > 100000:
            break
            
        # Evaluate current maximum correlation and corresponding lag
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        max_corr = corr[k_max]

        grad = compute_gradient(h_current, k_max, n_points, dx)
        
        # Update
        h_new = h_current - eta * grad
        
        # Project back to feasible region
        h_new = project_h(h_new, n_points, dx)
        
        # Evaluate new value
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject update
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = iteration
        else:
            eta *= 0.999
        
        # Apply perturbation if no improvement in a while
        if iteration - last_improvement > patience:
            h_perturb = h_current.copy()

            h_perturb = project_h(h_perturb, n_points, dx)
            c5_perturb = evaluate_h(h_perturb, dx)
            if c5_perturb < c5_current:
                h_current = h_perturb
                c5_current = c5_perturb
