import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Try different optimization approaches
    for attempt in range(5):  # Multiple random restarts
        if time.time() - start_time > budget_s - 10:
            break
            
        h_attempt = h_init.copy()
        if attempt > 0:
            # Perturb the initial solution
            h_attempt += np.random.normal(0, 0.01, n_points)
            h_attempt = repair_h(h_attempt, n_points)
        
        # Generate initial guess for the optimizer
        # Use a different strategy: small intervals with 1s and 0s
        if attempt > 2:
            h_attempt = np.zeros(n_points)
            # Place 1s in two symmetric intervals
            width = n_points // 5
            h_attempt[n_points//2 - width//2:n_points//2 + width//2] = 1
            h_attempt = repair_h(h_attempt, n_points)
        
        # Fine tune with scipy.optimize
        def objective(h_vec):
            return evaluate_h(h_vec, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        constr = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = minimize(objective, h_attempt, method='SLSQP', 
                            bounds=bounds, constraints=constr, 
                            options={'maxiter': 2000, 'ftol': 1e-8})
            if result.success:
                h_opt = result.x
                h_opt = repair_h(h_opt, n_points)
                c5_opt = evaluate_h(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass
        
        # Simple gradient descent for additional refinement
        h_refined = h_attempt.copy()
        learning_rate = 1e-4
        for iter in range(500):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Gradient approximation
            grad = np.zeros(n_points)
            eps = 1e-6
            corr_base = correlate(h_refined, 1 - h_refined, mode='full')
            max_k_idx = np.argmax(corr_base)
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                grad[i] = (corr_plus[max_k_idx] - corr_base[max_k_idx]) / eps

                best_h = h_refined.copy()

        # Additional local search
        if attempt == 0:  # First attempt, try a more structured approach
            # Try a step function approach
            h_step = np.zeros(n_points)
            center = n_points // 2
            width = n_points // 4
            h_step[center - width//2:center + width//2] = 1
            h_step = repair_h(h_step, n_points)
            c5_step = evaluate_h(h_step, dx)
            if c5_step < best_c5:
                best_h = h_step.copy()
                best_c5 = c5_step

        h_refined = best_h.copy()
        for i in range(n_points):

            h_test[i] = np.clip(h_test[i] + 0.01, 0, 1)
            h_test = repair_h(h_test, n_points)
            c5_test = evaluate_h(h_test, dx)
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test
            else:
                h_test[i] = np.clip(h_test[i] - 0.02, 0, 1)
                h_test = repair_h(h_test, n_points)
                c5_test = evaluate_h(h_test, dx)
                if c5_test < best_c5:
                    best_h = h_test.copy()
                    best_c5 = c5_test
                    
    return (best_h.tolist(), best_c5, n_points)
