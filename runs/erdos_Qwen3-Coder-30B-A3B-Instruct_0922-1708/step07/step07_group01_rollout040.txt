import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Phase 1: CCCP-inspired optimization with alternating updates
    max_iter_c ccp = 5000
    lr_c ccp = 0.01
    patience_c ccp = 500
    
    for i in range(max_iter_c ccp):
        if time.time() - start_time > budget_s - 5:
            break

        # Identify the shift that maximizes overlap
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        
        # Form gradient at k_max (simplified for faster gradient estimation)
        h_shifted = np.roll(h_current, k_max - len(h_current) + 1)
        grad = 1 - h_shifted
        
        # Apply gradient update
        h_new = h_current - lr_c ccp * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Update current solution
        h_current = h_new
        c5_current = c5_new
        
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
        
        # Adaptive learning rate
        if i % patience_c ccp == 0:
            lr_c ccp *= 0.95
            
        # Perturb occasionally to escape local minima
        if i % 500 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Phase 2: Cyclic coordinate descent with exact gradient
    max_iter_cd = 10000
    lr_cd = 0.1
    patience_cd = 1000
    
    for i in range(max_iter_cd):
        if time.time() - start_time > budget_s - 5:
            break

        # Cyclic coordinate descent
        for j in range(n_points):
            # Compute gradient for current variable h[j]
            h_temp = h_current.copy()
            h_temp[j] += 1e-6
            h_temp = project_h(h_temp, n_points)
            
            corr_plus = correlate(h_temp, 1 - h_temp, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            h_temp = h_current.copy()
            h_temp[j] -= 1e-6
            h_temp = project_h(h_temp, n_points)
            
            corr_minus = correlate(h_temp, 1 - h_temp, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad_j = (c5_plus - c5_minus) / (2e-6)
            
            # Update h[j] with exact gradient
            h_current[j] = h_current[j] - lr_cd * grad_j
            
            # Clip h[j] to [0,1]
            h_current[j] = np.clip(h_current[j], 0, 1)
            
            # Maintain the sum constraint
            h_current = project_h(h_current, n_points)
            
            # Check for improvement
            if i % patience_cd == 0:
                c5_current = evaluate_h(h_current, dx)
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
        
        # Periodic perturbation
        if i % 1000 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
            
        # Adaptive learning rate for CD
        if i % patience_cd == 0:
            lr_cd *= 0.98
    
    # Final refinement through local search
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
