import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size, initial_h=None):
    population = []
    if initial_h is not None:
        population.append(repair_h(initial_h.copy(), n_points))
    
    # Initialize with triangular pulses
    for _ in range(pop_size - len(population)):
        h = np.zeros(n_points)
        # Create a triangular pulse
        peak_idx = n_points // 2
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if x < 0.5:
                h[i] = 0.0
            elif x < 1.5:
                h[i] = (x - 0.5) / 1.0
            else:
                h[i] = 1.0
        h = repair_h(h, n_points)
        population.append(h)
    return population

def selection(population, fitnesses):
    idx = np.argsort(fitnesses)
    selected = [population[i] for i in idx[:len(population)//2]]
    return selected

def crossover(parent1, parent2):
    n_points = len(parent1)
    if n_points != len(parent2):
        return parent1.copy(), parent2.copy()
    
    # Single point crossover
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n_points = len(h)
    indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.05)
    return h

def genetic_algorithm(seed, n_points, pop_size, initial_h, budget_s):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h)
    
    best_fitness = float('inf')
    best_individual = None
    
    # Evolution parameters
    max_generations = 1000
    patience = 200
    last_improvement = 0
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_h(h, 2.0 / n_points) for h in population]
        
        # Track best
        min_fitness = min(fitnesses)
        if min_fitness < best_fitness:
            best_fitness = min_fitness
            best_individual = population[np.argmin(fitnesses)].copy()
            last_improvement = generation
            
        # Selection
        selected = selection(population, fitnesses)
        
        # Create new population
        new_population = [best_individual.copy()]
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            child1 = repair_h(child1, n_points)
            child2 = repair_h(child2, n_points)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
        
        # Check for early stopping
        if generation - last_improvement > patience:
            break
    
    return best_individual, best_fitness

    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial_h_values if available

            x_target = np.linspace(0, 2, n_points)
            initial_h = np.interp(x_target, x_orig, initial_h)
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(seed, n_points, 50, initial_h, budget_s)
    
    return (best_h.tolist(), best_c5, n_points)
