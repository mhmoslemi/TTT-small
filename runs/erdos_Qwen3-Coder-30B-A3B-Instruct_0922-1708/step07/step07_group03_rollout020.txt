import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Use CVXPY for convex optimization
    h_var = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Set constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2,
        C5 >= 0
    ]
    
    # Compute all correlation terms
    # For each lag, we compute the integral
    for k in range(1, 2 * n_points - 1):
        corr_k = 0
        for i in range(n_points):
            if i + k < n_points:
                corr_k += h_var[i] * (1 - h_var[i + k])
        constraints.append(corr_k <= C5)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    solver = None
    try:
        import cvxpy as cp
        if cp.installed_solvers():
            if 'MOSEK' in cp.installed_solvers():
                solver = cp.MOSEK
            elif 'GUROBI' in cp.installed_solvers():
                solver = cp.GUROBI
            elif 'ECOS' in cp.installed_solvers():
                solver = cp.ECOS
    except:
        pass
    
    try:
        if solver:
            prob.solve(solver=solver, verbose=False, time_limit=budget_s - 10)
        else:
            prob.solve(verbose=False, time_limit=budget_s - 10)
            
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            # Ensure that h satisfies the constraints
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        else:
            h_opt = h.copy()
    except:
        h_opt = h.copy()
    
    # Final evaluation
    corr = correlate(h_opt, 1 - h_opt, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_opt.tolist(), final_c5, n_points)
