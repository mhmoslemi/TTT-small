    ]

        # Approximate quadratic constraints for each lag

    # Let's simplify and skip the full quadratic constraints approach

    def compute_c5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    best_c5 = float('inf')

    # Gradient descent with adaptive learning rate
    lr = 1e-3

        eps = 1e-5

        for i in range(n_points):
            h_plus = h.copy()
