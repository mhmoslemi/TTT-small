import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initial guess
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        
    # Frank-Wolfe algorithm
    best_c5 = float('inf')
    best_h = h.copy()
    
    # For line search
    backtracking_factor = 0.5
    min_step_size = 1e-6
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current overlap integral
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient direction
        # Approximate using finite differences on the correlation output
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
            
        # Frank-Wolfe linear subproblem: minimize grad * s subject to 0 <= s <= 1, sum(s) = n_points/2
        # The optimal s will be 0 or 1 in directions of negative gradient
        s = np.zeros(n_points)
        sorted_indices = np.argsort(-grad)  # descending order
        
        # Construct s by setting components to 1 in negative gradient directions
        remaining_sum = n_points / 2
        for idx in sorted_indices:
            if remaining_sum > 0:
                s[idx] = min(1, remaining_sum)
                remaining_sum -= s[idx]
            else:
                break
                
        # Rescale to meet sum constraint
        if abs(np.sum(s) - n_points / 2) > 1e-8:
            s = s * (n_points / 2) / np.sum(s)
            
        # Line search to find optimal step size
        gamma = 0.5
        while gamma > min_step_size:
            new_h = (1 - gamma) * h + gamma * s
            # Ensure new_h stays feasible
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            corr_new = correlate(new_h, 1 - new_h, mode='full')            
            c5_new = np.max(corr_new) * dx
            
            if c5_new <= c5:
                h = new_h.copy()
                break
            else:
                gamma *= backtracking_factor
                
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
