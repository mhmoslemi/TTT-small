import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Evolution Strategy parameters
    pop_size = 200
    elites_fraction = 0.2
    offsprings_per_elite = 5
    mutation_std = 0.1
    patience = 30
    
    # Initialize population
    pop = [h_init.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        pop.append(h)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    last_improvement = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 5:
        generation += 1
        # Evaluate population
        fitness = [evaluate_h(h, dx) for h in pop]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elites = [pop[i] for i in sorted_indices[:int(elites_fraction * pop_size)]]
        
        # Update best
        current_best_fitness = fitness[sorted_indices[0]]
        if current_best_fitness < best_c5:
            best_c5 = current_best_fitness
            best_h = pop[sorted_indices[0]].copy()
            last_improvement = generation
        
        # Generate new candidates
        new_pop = elites.copy()
        for elite in elites:
            for _ in range(offsprings_per_elite):
                mutant = elite + np.random.randn(n_points) * mutation_std
                mutant = repair_h(mutant, n_points)
                new_pop.append(mutant)
        
        # Keep best pop_size candidates
        combined_pop = new_pop[:pop_size]
        pop = combined_pop
    
    return (best_h.tolist(), best_c5, n_points)
