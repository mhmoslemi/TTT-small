import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def create_triangular_pulse(n_points):
    h = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    # Create triangular pulse
    for i in range(n_points):
        if x[i] < 1.0:
            h[i] = x[i] / 1.0
        else:
            h[i] = (2.0 - x[i]) / 1.0
    return h

def mutate_h(h, mutation_rate=0.2):
    h = h.copy()
    n_points = len(h)
    mask = np.random.rand(n_points) < mutation_rate
    h[mask] += np.random.normal(0, 0.05, np.sum(mask))
    return repair_h(h, n_points)

def crossover_h(h1, h2):
    n_points = len(h1)
    crossover_point = np.random.randint(1, n_points)
    h_child = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    return h_child

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    else:
        # Create a triangular pulse as initial solution
        h_init = create_triangular_pulse(n_points)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Fill population with diverse random solutions
    for i in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Evolutionary loop
    max_generations = 1000
    patience = 200
    no_improvement_count = 0
    generation = 0
    
    while generation < max_generations and (time.time() - start_time) < budget_s - 5:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for stagnation
        if generation > 0 and fitness[0] >= best_c5:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
        
        if no_improvement_count >= patience:
            break
            
        # Keep the best individual (elitism)
        new_population = [population[0].copy()]
        
        # Select top 50% for breeding
        top_count = population_size // 2
        selected = population[:top_count]
        
        # Generate offspring through crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = crossover_h(parent1, parent2)
            child = mutate_h(child)
            new_population.append(child)
        
        population = new_population
        generation += 1
    
    # Return best solution
    if best_h is None:
        # If no improvement, return the initial solution
        h_best = population[0] if 'population' in locals() and len(population) > 0 else np.zeros(n_points)
        if 'initial_h_values' in globals():
            h_best = np.array(initial_h_values)
            if len(h_best) != n_points:
                x_orig = np.linspace(0, 2, len(h_best))
                x_target = np.linspace(0, 2, n_points)
                h_best = np.interp(x_target, x_orig, h_best)
            h_best = repair_h(h_best, n_points)
        c5_best = evaluate_h(h_best, dx)
    else:
        h_best = best_h
        c5_best = best_c5
    
    return (h_best.tolist(), c5_best, n_points)
