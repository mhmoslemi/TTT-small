import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size, initial_h_values=None):
    population = []
    if initial_h_values is not None:
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Create other random triangular pulses
    for _ in range(pop_size - len(population)):
        h = np.zeros(n_points)
        peak_pos = np.random.uniform(0.5, 1.5)
        width = np.random.uniform(0.2, 0.8)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if abs(x - peak_pos) < width / 2:
                h[i] = 1 - 2 * abs(x - peak_pos) / width
        h = repair_h(h, n_points)
        population.append(h)
    
    return population

def genetic_algorithm(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    pop_size = 50
    generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.5
    elite_size = 1
    patience = 200
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = initialize_population(n_points, pop_size, initial_h_values)
    else:
        population = initialize_population(n_points, pop_size)
    
    best_fitness = float('inf')
    best_individual = None
    
    last_improvement = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in population:
            fit = evaluate_h(individual, dx)
            fitness.append(fit)
        
        # Sort population
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_fitness:
            best_fitness = fitness[0]
            best_individual = population[0].copy()
            last_improvement = gen
        
        # Elitism
        new_population = population[:elite_size]
        
        # Selection (tournament)
        def tournament_selection():
            k = 3
            indices = np.random.choice(len(population), k)
            best_idx = indices[np.argmin([fitness[i] for i in indices])]
            return population[best_idx]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate:
                parent1 = tournament_selection()
                parent2 = tournament_selection()
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = tournament_selection()
                child2 = tournament_selection()
            
            # Mutation
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, 0.05)
                
                child = repair_h(child, n_points)
                new_population.append(child)
                
                if len(new_population) >= pop_size:
                    break
        
        population = new_population
        
        if gen - last_improvement > patience:
            break
    
    return (best_individual.tolist(), best_fitness, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
