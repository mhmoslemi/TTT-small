import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Iterative optimization in frequency domain
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Frequency domain optimization
        H = fft(current_h)
        H_abs = np.abs(H)
        H_phase = np.angle(H)
        
        # Objective function in frequency domain
        def obj_fn(H_abs_new):
            H_new = H_abs_new * np.exp(1j * H_phase)
            h_new = np.real(ifft(H_new))
            h_new = repair_h(h_new, n_points)
            return evaluate_h(h_new, dx)
        
        # Simple optimization of magnitude (variational approach)
        H_abs_new = H_abs.copy()
        step_size = 0.01
        for _ in range(20):
            grad = np.zeros_like(H_abs)
            eps = 1e-6
            for j in range(len(H_abs)):
                H_plus = H_abs.copy()
                H_plus[j] += eps
                obj_plus = obj_fn(H_plus)
                grad[j] = (obj_plus - obj_fn(H_abs)) / eps
            H_abs_new = H_abs_new - step_size * grad
            H_abs_new = np.maximum(H_abs_new, 0)
        
        # Update H with new magnitude
        H_new = H_abs_new * np.exp(1j * H_phase)
        h_new = np.real(ifft(H_new))
        h_new = repair_h(h_new, n_points)
        
        # Update best solution
        c5_new = evaluate_h(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        current_h = h_new
    
    return (best_h.tolist(), best_c5, n_points)
