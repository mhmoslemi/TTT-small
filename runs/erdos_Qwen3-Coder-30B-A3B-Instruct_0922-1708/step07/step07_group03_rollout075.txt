import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter_barrier = 1000
    max_iter_gd = 10000
    learning_rate = 1e-3
    mu_initial = 1e3
    barrier_decrease_factor = 0.99
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Helper function to compute correlation and gradient
    def compute_corr_and_grad(h):
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_corr = corr[max_lag_idx]
        c5 = max_corr * dx
        
        # Compute gradient with respect to h
        grad = np.zeros_like(h)
        
        # For maximum lag, compute gradient contributions from both sides
        # This is more efficient than finite differences but requires knowing the active lag
        max_lag = max_lag_idx - (len(h) - 1)  # lag in range [-n_points+1, n_points-1]
        
        # Forward contribution
        if max_lag >= 0:
            for i in range(len(h) - max_lag):
                grad[i] += (1 - h[i + max_lag])
        else:
            for i in range(-max_lag, len(h)):
                grad[i] += (1 - h[i + max_lag])
                
        return c5, grad
    
    # Barrier phase
    mu = mu_initial
    best_c5 = float('inf')
    best_h = h.copy()
    
    for barrier_iter in range(max_iter_barrier):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current correlation and gradient
        c5, grad = compute_corr_and_grad(h)
    
        # Update h
        h = h - learning_rate * grad
        
        # Projection to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to meet constraint
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
            
        # Update barrier parameter
        mu *= barrier_decrease_factor
        
        # Track best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Gradient descent phase with tighter controls
    for gd_iter in range(max_iter_gd):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current correlation and gradient
        c5, grad = compute_corr_and_grad(h)
        
        # Update h
        h = h - learning_rate * grad
        
        # Projection to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to meet constraint
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
            
        # Track best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
