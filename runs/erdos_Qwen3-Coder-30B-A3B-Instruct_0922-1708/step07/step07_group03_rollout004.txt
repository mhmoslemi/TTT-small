import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter_barrier = 1000
    max_iter_gd = 10000
    learning_rate = 1e-3
    mu_initial = 1e3
    barrier_decrease_factor = 0.99
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Barrier method
    mu = mu_initial
    for barrier_iter in range(max_iter_barrier):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient via adjoint
        # We compute the gradient for the maximum correlation only
        gradient = np.zeros_like(h)
        
        # The gradient is derived from the maximum lag k
        # ∂C5/∂h_i = Σ_k (1 - h[i+k]) * indicator_i  
        # This matches the correlation derivative
        max_lag = max_lag_idx - (n_points - 1)  # Convert full correlation index to lag
    
        # Build the gradient from the maximum lag
        for i in range(n_points):
            if 0 <= i + max_lag < n_points:
                gradient[i] += (1 - h[i + max_lag])
                
        # Update with learning rate and project
        h = h - learning_rate * gradient
        
        # Project h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to meet constraint ∫h(x)dx = 1
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
            
        # Update barrier parameter
        mu *= barrier_decrease_factor
    
    # Gradient descent phase
    for gd_iter in range(max_iter_gd):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient via adjoint
        gradient = np.zeros_like(h)
        max_lag = max_lag_idx - (n_points - 1)
        
        for i in range(n_points):
            if 0 <= i + max_lag < n_points:
                gradient[i] += (1 - h[i + max_lag])
                
        # Update with learning rate
        h = h - learning_rate * gradient
        
        # Project h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to meet constraint ∫h(x)dx = 1
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
