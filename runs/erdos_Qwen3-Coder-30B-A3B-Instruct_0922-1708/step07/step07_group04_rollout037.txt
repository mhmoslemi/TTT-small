import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, rate=0.1, delta=0.1):
    """Apply random mutation to h."""
    h_new = h.copy()
    mask = np.random.rand(len(h_new)) < rate
    h_new[mask] += np.random.uniform(-delta, delta, np.sum(mask))
    return repair_h(h_new, len(h))

def crossover_h(parent1, parent2, rate=0.5):
    """Perform crossover between two parents."""
    cut_point = np.random.randint(1, len(parent1)-1)
    child = np.concatenate([parent1[:cut_point], parent2[cut_point:]])
    return child

def tournament_selection(population, fitnesses, k=3):
    """Tournament selection for genetic algorithm."""
    selected_indices = []
    for _ in range(len(population)):
        idxs = np.random.choice(len(population), k, replace=False)
        selected_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        selected_indices.append(selected_idx)
    return [population[i] for i in selected_indices]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # GA parameters
    pop_size = 100
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.5
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        h_mut = mutate_h(h_init.copy(), mutation_rate, 0.1)
        population.append(h_mut)
    
    # Main GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness (negative overlap)
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Select parents
        parents = tournament_selection(population, fitnesses)
        
        # Create new population
        new_population = []
        
        # Keep best individual
        best_idx = np.argmin(fitnesses)
        new_population.append(population[best_idx].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Crossover
            if np.random.rand() < crossover_rate and len(parents) >= 2:
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                child = crossover_h(parent1, parent2)
            else:
                # Clone
                child = np.random.choice(parents).copy()
            
            # Mutate
            child = mutate_h(child, mutation_rate, 0.1)
            
            new_population.append(child)
        
        population = new_population[:pop_size]
    
    # Find best individual
    fitnesses = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx]
    best_c5 = fitnesses[best_idx]
    
    # Final gradient refinement
    h_current = best_h.copy()
    c5_current = best_c5
    eta = 0.001
    eps = 1e-6
    max_iter = 10000
    patience = 100
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update each coordinate
        h_new = h_current.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_current[j] - eta * grad_j
            
        # Apply repair step
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if better or with small probability
        if c5_new < c5_current or np.random.rand() < 0.01:
            h_current = h_new
            c5_current = c5_new
            
        # Adjust learning rate
        if i - last_improvement > patience:
            eta *= 0.95
            last_improvement = i
    
    return (h_current.tolist(), c5_current, n_points)
