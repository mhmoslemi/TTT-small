import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, mutation_rate=0.1, perturbation=0.1):
    """Apply mutation to h."""
    h_mut = h.copy()
    n_points = len(h)
    n_mutate = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, n_mutate, replace=False)
    for i in indices:
        h_mut[i] += np.random.uniform(-perturbation, perturbation)
    return repair_h(h_mut, n_points)

def crossover_h(h1, h2):
    """Perform crossover between two h vectors."""
    n_points = len(h1)
    crossover_point = np.random.randint(1, n_points)
    h_child = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    return repair_h(h_child, n_points)

def tournament_selection(population, fitnesses, k=3):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Genetic algorithm parameters
    population_size = 100
    generations = 100
    mutation_rate = 0.1
    perturbation = 0.1
    elite_size = 10
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = mutate_h(h_init, mutation_rate, perturbation)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_h(h, dx) for h in population]
    
    # Best solution tracking
    best_idx = np.argmin(fitnesses)
    h_best = population[best_idx].copy()
    c5_best = fitnesses[best_idx]
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        elites = [population[i] for i in sorted_indices[:elite_size]]

        # Create new population

            child = mutate_h(child, mutation_rate, perturbation)

        fitnesses = [evaluate_h(h, dx) for h in population]

        best_idx = np.argmin(fitnesses)

            c5_best = fitnesses[best_idx]
