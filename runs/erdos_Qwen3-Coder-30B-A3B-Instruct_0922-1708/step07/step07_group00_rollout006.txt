import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    swarm_size = 100
    max_iter = 2000
    w_start = 0.9
    w_end = 0.1
    c1, c2 = 2.0, 2.0
    max_vel = 0.1
    
    # Initialize swarm
    particles = np.array([h_init.copy() for _ in range(swarm_size)])
    velocities = np.random.uniform(-max_vel, max_vel, (swarm_size, n_points))
    particle_best_positions = particles.copy()
    particle_best_scores = np.array([evaluate_h(p, dx) for p in particles])
    global_best_idx = np.argmin(particle_best_scores)
    global_best_score = particle_best_scores[global_best_idx]
    global_best_position = particles[global_best_idx].copy()
    
    # PSO main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * (iter_count / max_iter)
        
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (particle_best_positions[i] - particles[i]) + 
                            c2 * r2 * (global_best_position - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < particle_best_scores[i]:
                particle_best_scores[i] = score
                particle_best_positions[i] = particles[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = particles[i].copy()
    
    # Local refinement
    refined_h = global_best_position.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient with finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        base_corr = correlate(refined_h, 1 - refined_h, mode='full')
        max_k_idx = np.argmax(base_corr)
        
        for i in range(n_points):
            h_plus = refined_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (corr_plus[max_k_idx] - base_corr[max_k_idx]) / eps
            
        # Apply gradient step
        refined_h = refined_h - 1e-4 * grad
        refined_h = repair_h(refined_h, n_points)
        curr_score = evaluate_h(refined_h, dx)
        if curr_score < global_best_score:
            global_best_score = curr_score
            global_best_position = refined_h.copy()
    
    return (global_best_position.tolist(), global_best_score, n_points)
