import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2, n_points):
    """Create offspring from two parents using segment blending."""
    offspring = parent1.copy()
    # Randomly choose crossover segments
    n_segments = np.random.randint(2, n_points // 4)
    segments = np.random.choice(n_points, size=n_segments, replace=False)
    segments = np.sort(segments)
    
    # Alternate segments from parents
    start_idx = 0
    for i, seg_start in enumerate(segments):
        if i % 2 == 1:
            offspring[start_idx:seg_start] = parent2[start_idx:seg_start]
        start_idx = seg_start
    if start_idx < n_points:
        if len(segments) % 2 == 1:
            offspring[start_idx:] = parent2[start_idx:]
    
    return offspring

def mutate(h, n_points, mutation_rate=0.3, noise_scale=0.1):
    """Apply mutation to a candidate."""
    h_mut = h.copy()
    
    # Apply segment flips
    if np.random.rand() < mutation_rate:
        n_flips = max(1, int(0.1 * n_points))
        flip_indices = np.random.choice(n_points, size=n_flips, replace=False)
        for idx in flip_indices:
            h_mut[idx] = 1 - h_mut[idx]
    
    # Apply Gaussian noise
    if np.random.rand() < mutation_rate:
        noise = np.random.normal(0, noise_scale, n_points)
        h_mut += noise
        h_mut = np.clip(h_mut, 0, 1)
    
    return h_mut

def local_refinement(h, n_points, dx, max_iter=50):
    """Fast local hill-climb refinement."""
    h_refined = h.copy()
    c5_current = evaluate_h(h_refined, dx)
    
    for _ in range(max_iter):
        improved = False
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices:
            eps = 1e-6
            h_plus = h_refined.copy()
            h_minus = h_refined.copy()
            
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update using gradient
            h_refined[j] -= 0.01 * grad_j
            h_refined[j] = max(0, min(1, h_refined[j]))
            
        h_refined = project_h(h_refined, n_points, dx)
        c5_new = evaluate_h(h_refined, dx)
        
        if c5_new < c5_current:
            h_refined = h_refined.copy()
            c5_current = c5_new
            improved = True
        else:
            break
            
    return h_refined

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Population-based memetic algorithm
    P = 50  # Population size
    M = max(1, P // 5)  # Number of parents to select
    mutation_scale = 0.1
    population = []
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)

    # Generate diverse initial population
    for _ in range(P):
        candidate = h_init.copy()
        if np.random.rand() < 0.5:
            indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            for idx in indices:
                candidate[idx] = 1 - candidate[idx]
        camera = np.random.normal(0, 0.05, n_points)
        candidate += camera
        candidate = project_h(candidate, n_points, dx)
        population.append(candidate)
    
    # Best solution tracking
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    generation = 0
    while time.time() - start_time < budget_s - 5:
        generation += 1
        if generation % 10 == 0:
            mutation_scale *= 0.99
        
        # Evaluate population
        fitness = []
        for h in population:
            fitness.append(evaluate_h(h, dx))
        
        # Sort and select parents
        sorted_indices = np.argsort(fitness)
        parents = [population[i] for i in sorted_indices[:M]]
        parent_fitness = [fitness[i] for i in sorted_indices[:M]]
        
        # Update best
        min_fitness = min(parent_fitness)
        if min_fitness < c5_best:
            best_idx = np.argmin(parent_fitness)
            h_best = parents[best_idx].copy()
            c5_best = min_fitness
        
        # Create offspring
        offspring = []
        for _ in range(P - M):
            # Select parents for crossover
            p1_idx, p2_idx = np.random.choice(M, size=2, replace=False)
            p1, p2 = parents[p1_idx], parents[p2_idx]
            
            # Crossover
            child = crossover(p1, p2, n_points)
            
            # Mutations
            child = mutate(child, n_points, noise_scale=mutation_scale)
            child = project_h(child, n_points, dx)
            
            # Local refinement
            child_refined = local_refinement(child, n_points, dx)
            
            offspring.append(child_refined)
        
        # Replace worst in population with offspring
        worst_indices = np.argsort(fitness)[-(P-M):]
        for i, idx in enumerate(worst_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
    
    return (h_best.tolist(), c5_best, n_points)
