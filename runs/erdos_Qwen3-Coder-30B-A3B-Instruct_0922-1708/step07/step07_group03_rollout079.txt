import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = 100
    dx = 2.0 / n_points
    max_iter_barrier = 1000
    max_iter_gd = 10000
    learning_rate = 1e-3
    mu_initial = 1e3
    barrier_decrease_factor = 0.99

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Barrier method phase
    mu = mu_initial
    for iter_barrier in range(max_iter_barrier):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation and find maximum
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_lag = np.argmax(corr)
        
        # Compute exact gradient via adjoint method
        grad = np.zeros_like(h)
        # The gradient is the correlation of h with (1-h) shifted back
        # But since we're maximizing one correlation over multiple lags,
        # we take the gradient at the lag that gives maximum correlation
        corr_grad = np.zeros_like(h)
        # Create correlation kernel with 1 at appropriate lag
        lag = max_lag - (len(h) - 1)  # shift to match correlation indexing
        correlation_kernel = np.zeros_like(h)
        if lag >= 0:
            correlation_kernel[lag:] = 1
        else:
            correlation_kernel[:lag] = 1
        corr_grad = correlate(correlation_kernel, 1 - h, mode='full')[:len(h)]
        corr_grad = corr_grad * dx
        
        # Update using gradient and barrier term
        h -= learning_rate * corr_grad
        # Project to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Rescale to maintain sum = n_points/2 constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
            
        # Update best solution
        if max_corr * dx < best_c5:
            best_c5 = max_corr * dx
            best_h = h.copy()

    # Gradient descent phase
    for iter_gd in range(max_iter_gd):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current maximum correlation
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_lag = np.argmax(corr)
        
        # Compute detailed gradient via adjoint method
        grad = np.zeros_like(h)
        # Compute the effect of small changes in h on the maximum overlap
        lag = max_lag - (len(h) - 1)
        
        # Compute adjoint gradient for the specific maximum lag
        # This simulates the chain rule for gradient of the max function
        # For each k in valid range:
        # f_k = integral h(x)(1 - h(x+k)) dx 
        # df_k/dh = (1 - h(x+k)) at the right place
        
        # Manual gradient computation 
        grad = np.zeros_like(h)
        indices = np.arange(len(h))
        # For specific lag(k) that gives max:
        if lag >= 0:
            grad[lag:] += (1 - h[:-lag]) * dx
        else:
            grad[:lag] += (1 - h[-lag:]) * dx
            
        # Apply gradient step
        h -= learning_rate * grad
        
        # Project to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Rescale to maintain sum = n_points/2 constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
            
        # Update best solution
        if max_corr * dx < best_c5:
            best_c5 = max_corr * dx
            best_h = h.copy()
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
