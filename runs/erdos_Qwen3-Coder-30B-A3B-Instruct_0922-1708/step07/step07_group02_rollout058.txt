import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate a solution with adaptive noise and segment flips."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip (30% chance)
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    return mutated

        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h[:mid] = 0.5
        h[mid:] = 0.0

    # Add diverse initial solutions

        population.append(h_var)

        gen += 1
        # Evaluate population
        fitness_scores = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness_scores.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_h = population[0].copy()

            offspring = mutate(offspring, gen, mutation_scale)
