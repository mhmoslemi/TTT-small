import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(params, n_points):

    K = len(params) // 2
    heights = params[:K]
    boundaries = np.concatenate([[0], params[K:]])
    boundaries = np.clip(boundaries, 0, 2)
    boundaries = np.sort(boundaries)
    boundaries = np.concatenate([boundaries, [2]])
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(K):
        start_idx = int(boundaries[i] / dx)
        end_idx = int(boundaries[i+1] / dx)
        if end_idx > start_idx:
            h[start_idx:end_idx] = heights[i]
    
    # Rescale to meet constraint
    integral = np.sum(h) * dx
    if integral > 0:
        h = h * (n_points / 2) / integral
    
    return h

def compute_gradient(params, n_points, dx, eps=1e-6):
    """Compute gradient of C5 with respect to params."""
    # Use finite differences to estimate gradient
    original_h = parameterize_h(params, n_points)
    original_c5 = evaluate_h(original_h, dx)
    
    grad = np.zeros_like(params)
    for i in range(len(params)):
        params_plus = params.copy()
        params_plus[i] += eps
        h_plus = parameterize_h(params_plus, n_points)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - original_c5) / eps
    
    return grad
