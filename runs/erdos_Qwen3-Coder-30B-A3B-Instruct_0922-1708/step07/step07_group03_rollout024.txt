        h = h_init * (n_points / 2) / np.sum(h_init)

        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    h_var = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Create constraints for all lags
    constraints = []
    
    # Add the sum constraint
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Add bounds constraints
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Add the overlap constraints for all lags
    for k in range(2 * n_points - 1):
        # Compute the correlation for this lag
        # This is a simplified approach - we'll directly compute the sum constraints
        # As we cannot easily express the full convolution in CVXPY,
        # we use a different approach:
        # We compute all pairwise products and add constraints
        
        # Build tridiagonal part or simplify - let's consider maximization problem differently:
        # Instead of handling full correlation, we use a different approach:
        # We can express the problem as min ||C5|| s.t. [constraints]
        # Let's do a simple approach: optimize with gradient-based method using scipy

    # Simple gradient descent with projection
    learning_rate = 1e-4
    max_iter = 30000  # Reduced to save time
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Estimate gradient using finite differences (simplified version)
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx

        target_sum = n_points / 2

    # Final evaluation
