import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def gradient_h(h, dx):
    """Compute the gradient for the maximum overlap term."""
    corr = correlate(h, 1 - h, mode='full')
    max_k = np.argmax(corr)
    n = len(h)
    # Create the gradient for the specific k that maximizes overlap
    grad = np.zeros(n)
    if max_k < n:
        # For k < n, we have h[i] * (1 - h[i+k]) in the overlap
        for i in range(n):
            if i + max_k < n:
                grad[i] = 1 - h[i + max_k]
    else:
        # For k >= n, we have h[i+k-n] * (1 - h[i]) in the overlap
        k_shift = max_k - n + 1
        for i in range(n):
            if i >= k_shift:
                grad[i] = -h[i - k_shift]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = int(1e6)
    patience = 100
    last_improvement = 0
    # Multiple restarts with slightly different initialization
    num_restarts = 5
    restart_interval = max_iter // num_restarts
    
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 5:
            break
        h_current = h_init.copy()
        # Add small random perturbation to avoid local minima
        if restart > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_current = project_h(h_current + noise, n_points)
        c5_current = evaluate_h(h_current, dx)
        
        for i in range(restart_interval):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute exact gradient using the maximum overlap term
            grad = gradient_h(h_current, dx)
            
            # Update with gradient descent
            h_new = h_current - learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept move if improvement
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
            else:
                # Occasionally accept slightly worse moves to escape local minima
                if np.random.rand() < 0.005:
                    h_current = h_new.copy()
                    c5_current = c5_new
            
            # Update best
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
            
            # Adapt learning rate
            if i - last_improvement > patience:
                learning_rate *= 0.95
                last_improvement = i
            
            # Periodic global perturbation to explore more
            if i % 300 == 0 and i > 0:
                noise = np.random.normal(0, 0.002, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
    
    # Final refinement through local search
    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.003)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
