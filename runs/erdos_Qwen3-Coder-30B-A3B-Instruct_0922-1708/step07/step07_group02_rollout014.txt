import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def generate_candidate(n_points, dx, initial_h=None):
    """Generate a feasible candidate solution."""
    if initial_h is not None:
        h = initial_h.copy()
    else:
        # Start with a simple piecewise constant
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 0.5
        h[mid:] = 0.0
    return project_h(h, n_points, dx)

def segment_crossover(parent1, parent2, n_points):
    """Create offspring via segment-wise blending."""
    offspring = parent1.copy()
    # Randomly decide which segments to take from parent2
    n_segments = 5
    segment_size = n_points // n_segments
    for i in range(n_segments):
        start_idx = i * segment_size
        end_idx = min((i + 1) * segment_size, n_points)
        if np.random.rand() < 0.5:
            offspring[start_idx:end_idx] = parent2[start_idx:end_idx]
    return offspring

def mutate_individual(h, n_points, mutation_scale):
    """Apply mutation to an individual."""
    h_mut = h.copy()
    # Random segment flip
    if np.random.rand() < 0.3:
        idx1, idx2 = np.random.randint(0, n_points, 2)
        if idx1 > idx2:
            idx1, idx2 = idx2, idx1
        h_mut[idx1:idx2] = 1 - h_mut[idx1:idx2]
    # Add Gaussian noise
    noise = np.random.normal(0, mutation_scale, n_points)
    h_mut += noise
    return h_mut

def hill_climb(h, n_points, dx, max_iter=50):
    """Fast local refinement via gradient descent."""
    h_current = h.copy()
    c5_current = evaluate_h(h_current, dx)
    eta = 0.01
    eps = 1e-6
    
    for _ in range(max_iter):
        h_new = h_current.copy()
        improved = False
        
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
            
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            improved = True
        else:
            eta *= 0.995
            
        if not improved:
            break
            
    return h_current, c5_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = generate_candidate(n_points, dx)
        
    # Memetic algorithm parameters
    population_size = 50
    elite_size = 10
    mutation_scale = 0.1
    max_generations = 1500
    n_generations = 0
    
    # Create initial population
    population = [generate_candidate(n_points, dx, h_init) for _ in range(population_size)]
    population_scores = [evaluate_h(h, dx) for h in population]
    
    # Track best solution
    best_idx = np.argmin(population_scores)
    h_best = population[best_idx].copy()
    c5_best = population_scores[best_idx]
    
    # Main loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        n_generations += 1
        
        # Evaluate current population
        population_scores = [evaluate_h(h, dx) for h in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(population_scores)
        population = [population[i] for i in sorted_indices]
        population_scores = [population_scores[i] for i in sorted_indices]
        
        # Update best if improved
        if population_scores[0] < c5_best:
            h_best = population[0].copy()
            c5_best = population_scores[0]
            
        # Selection: keep top elite_size
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite.copy()
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = elite[np.random.randint(0, elite_size)]
            parent2 = elite[np.random.randint(0, elite_size)]
            
            # Crossover
            offspring = segment_crossover(parent1, parent2, n_points)
            
            # Mutation
            offspring = mutate_individual(offspring, n_points, mutation_scale)
            
            # Local refinement
            offspring, _ = hill_climb(offspring, n_points, dx, 50)
            
            # Ensure feasibility
            offspring = project_h(offspring, n_points, dx)
            
            new_population.append(offspring)
            
        # Update population
        population = new_population
        
        # Reduce mutation scale
        mutation_scale *= 0.99
        
    return (h_best.tolist(), c5_best, n_points)
