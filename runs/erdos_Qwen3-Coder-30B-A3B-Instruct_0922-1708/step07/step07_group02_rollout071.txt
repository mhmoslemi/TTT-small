import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a heuristic feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Start with the initial solution
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Track best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    eta = 0.01
    max_iter = 100000
    patience = 50
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Identify which lag gives the maximum correlation
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient analytically for this specific k_max
        # For the k-th lag, d/dh[i] (h[i]* (1-h[i+k])) = (1-h[i+k]) for i+k < n_points
        # We are computing gradient for the maximum overlap term
        
        grad = np.zeros_like(h_current)
        if k_max < n_points:  # k = 0 to n_points-1
            # For lag k, we add gradient contributions from h[i] terms that contribute to h[i+k]
            # which is the same as the contribution to 1-h[i+k] 
            i_indices = np.arange(n_points - k_max)
            j_indices = i_indices + k_max
            # Each h[i] contributes to the sum with a weight (1-h[i+k])
            grad[i_indices] += (1 - h_current[j_indices])
        else:
            # The 'full' correlation goes to 2*n_points - 2; the k_max > n_points corresponds
            # to negative lags from the negative side
            k_real = k_max - n_points + 1  # Convert to actual negative lag
            j_indices = np.arange(n_points + k_real) 
            i_indices = j_indices - k_real
            # i must be valid
            valid_indices = i_indices >= 0  # Ensure i >= 0
            i_indices = i_indices[valid_indices]
            j_indices = j_indices[valid_indices]
            grad[i_indices] += (1 - h_current[j_indices])
        
        # Apply gradient descent with projected step
        h_new = h_current - eta * grad
        h_new = project_h(h_new, n_points, dx)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Update if improved
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            # Adaptive learning rate
            eta *= 0.999
        
        # Periodic perturbation to avoid local minima  
        if i - last_improvement > patience:
            h_perturb = h_current + np.random.normal(0, 0.01, n_points)
            h_perturb = project_h(h_perturb, n_points, dx)
            c5_perturb = evaluate_h(h_perturb, dx)
            if c5_perturb < c5_current:
                h_current = h_perturb
                c5_current = c5_perturb
                last_improvement = i
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
    
    return (h_best.tolist(), c5_best, n_points)
