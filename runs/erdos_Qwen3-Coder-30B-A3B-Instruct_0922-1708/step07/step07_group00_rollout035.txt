import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def update_magnitude(H, n_points):
    """Update magnitude to minimize maximum correlation."""
    # Get power spectral density
    P = np.abs(H)**2
    
    # We are looking to minimize maximum of inverse fft of P
    # This is a convex problem with constraints.
    # Use a quadratic approximation of the problem
    
    # In practice, for FFT-based optimization, this can be very complex
    # Simpler approximation: uniform distribution of power minimizes max
    P_new = np.ones_like(P)
    P_new = P_new * np.sum(P) / len(P)  # Uniform power
    return np.sqrt(P_new) * np.exp(1j * np.angle(H))

def optimize_h(h, n_points, dx, max_iter=500):
    """Optimize h using Fourier domain approach."""
    for iteration in range(max_iter):
        # Compute Fourier transform
        H = fft(h)
        
        # Update magnitude to minimize the main lobe
        # For simplicity, we give a uniform distribution to the magnitude
        H_new = update_magnitude(H, n_points)
        
        # Update phase to put more energy in low frequencies to minimize peak
        # Simple heuristic: randomize phase but keep magnitude
        H_new = np.abs(H) * np.exp(1j * np.random.random(n_points) * 2 * np.pi)
        
        # Inverse transform
        h_new = np.real(ifft(H_new))
        
        # Repair to satisfy constraints
        h_new = repair_h(h_new, n_points)
        
        # Evaluate improvement
        c5_new = evaluate_h(h_new, dx)
        
        # If improved, accept
        if c5_new < evaluate_h(h, dx):
            h = h_new.copy()
    
    return h
