import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a structured construction
        h_init = np.zeros(n_points)
        # Create a piecewise linear function that is 0 on [0,0.5) and 1 on (1.5,2]
        # and slopes linearly in between
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        # Repair to sum to 1
        h_init = repair_h(h_init, n_points)

    # PSO parameters
    n_particles = 100
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    patience = 100
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        # Start near h_init with small random perturbations
        h = h_init + np.random.normal(0, 0.01, n_points)
        h = repair_h(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_h(h, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iteration
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 5 and iter_count < 400:
        w = w_start - (w_start - w_end) * (iter_count / 400)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Add mutation noise
            if np.random.rand() < mutation_rate:
                velocities[i] += np.random.normal(0, mutation_sigma, n_points)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate fitness
            current_val = evaluate_h(particles[i], dx)
            
            # Update pbest
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update gbest
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
                    last_improvement = iter_count
        
        # Mutation - reset some particles
        if iter_count % 25 == 0:
            for i in range(n_particles):
                if np.random.rand() < mutation_rate:
                    # Perturb 20% of values
                    indices = np.random.choice(n_points, int(0.2 * n_points), replace=False)
                    particles[i][indices] += np.random.normal(0, mutation_sigma, len(indices))
                    particles[i] = repair_h(particles[i], n_points)
                    new_val = evaluate_h(particles[i], dx)
                    if new_val < pbest_vals[i]:
                        pbest[i] = particles[i].copy()
                        pbest_vals[i] = new_val
                        
                        if new_val < gbest_val:
                            gbest = particles[i].copy()
                            gbest_val = new_val
                            last_improvement = iter_count
        
        iter_count += 1
        
        # Early stopping if no improvement
        if iter_count - last_improvement > patience:
            break
    
    return (gbest.tolist(), gbest_val, n_points)
