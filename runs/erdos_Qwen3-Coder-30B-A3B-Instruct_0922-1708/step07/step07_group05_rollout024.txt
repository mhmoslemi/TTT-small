import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidate(n_points, seed=0):
    np.random.seed(seed)
    # Sample a few Fourier modes
    n_modes = 4
    coeffs = np.random.randn(n_modes)
    base = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    
    for i in range(n_modes):
        k = 2 * np.pi * (i + 1) / 2
        base += coeffs[i] * np.cos(k * x)
    
    # Normalize to [0,1]
    base = (base - base.min()) / (base.max() - base.min()) if base.max() > base.min() else np.ones(n_points) * 0.5
    base = np.clip(base, 0, 1)
    
    # Repair to sum constraint
    return repair_h(base, n_points)

def gradient_step(h, dx, epsilon=1e-6):

    grad = np.zeros(n_points)
    c5_base = evaluate_h(h, dx)

        h_minus[i] -= epsilon
        h_plus = repair_h(h_plus, n_points)
        h_minus = repair_h(h_minus, n_points)
        
        c5_plus = evaluate_h(h_plus, dx)
        c5_minus = evaluate_h(h_minus, dx)
        grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
    
    return grad

    top_candidates = [h for _, h in evals[:10]]

    # Refine top candidates using gradient descent
    max_iter = 2000
    patience = 200
    for h_init in top_candidates:
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)

        learning_rate = 0.01
