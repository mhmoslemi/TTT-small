import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Binary search parameters
    c_low = 0.0
    c_high = 0.5
    tolerance = 1e-5
    max_iterations = 50
    
    best_h = h_init.copy()
    best_c = evaluate_h(best_h, dx)
    
    # Binary search for optimal C5
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        c = (c_low + c_high) / 2.0
        
        # Create the LP problem
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = []
        
        # Integration constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Bound constraints
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Overlap constraints
        # For each k from 0 to n_points-1 (minimum overlap is at k=0)
        for k in range(n_points):
            # Construct the overlap constraint
            # Sum h[i] * (1 - h[i+k]) for i = 0 to n_points-k-1
            overlap_sum = 0
            for i in range(n_points - k):
                overlap_sum += h_vars[i] * (1 - h_vars[i + k])
            
            constraints.append(overlap_sum <= c * dx)
        
        # Create and solve the problem
        objective = cp.Minimize(0)  # We just want to check feasibility
        prob = cp.Problem(objective, constraints)
        
        try:
            # Try to solve with a fast solver
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=1000)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Found feasible solution
                h_val = h_vars.value
                if h_val is not None:
                    h_solution = np.array(h_val)
                    current_c = evaluate_h(h_solution, dx)
                    if current_c < best_c:
                        best_h = h_solution.copy()
                        best_c = current_c
                    c_high = c
                else:
                    c_low = c
            else:
                c_low = c
                
        except Exception:
            c_low = c
        
        # Early termination if close enough
        if c_high - c_low < tolerance:
            break
    
    # Final refinement
    h_final = best_h.copy()
    c_final = best_c
    
    # Allow some local search refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_final.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 20), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.002)
            
        h_new = project_h(h_new, n_points)
        c_new = evaluate_h(h_new, dx)
        
        if c_new < c_final:
            h_final = h_new
            c_final = c_new
    
    return (h_final.tolist(), c_final, n_points)
