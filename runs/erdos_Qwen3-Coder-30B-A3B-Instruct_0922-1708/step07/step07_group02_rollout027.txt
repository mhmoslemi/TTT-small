import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)

    # PSO parameters
    swarm_size = 50
    max_iter = 100000
    inertia_weight_start = 0.9
    inertia_weight_end = 0.4
    cognitive_coeff = 1.5
    social_coeff = 1.5
    velocity_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution as first particle
    swarm.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        swarm.append(h)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
    
    # Track personal bests and global best
    personal_bests = [evaluate_h(p, dx) for p in swarm]
    personal_best_positions = [p.copy() for p in swarm]
    
    global_best_idx = np.argmin(personal_bests)
    global_best = personal_bests[global_best_idx]
    global_best_position = swarm[global_best_idx].copy()
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        inertia_weight = inertia_weight_start - (inertia_weight_start - inertia_weight_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_coeff * r1 * (personal_best_positions[i] - swarm[i])
            social = social_coeff * r2 * (global_best_position - swarm[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive + social
            
            # Apply velocity limit
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_h(swarm[i], n_points, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if fitness < personal_bests[i]:
                personal_bests[i] = fitness
                personal_best_positions[i] = swarm[i].copy()
            
            # Update global best
            if fitness < global_best:
                global_best = fitness
                global_best_position = swarm[i].copy()
    
    return (global_best_position.tolist(), global_best, n_points)
