import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    swarm_size = 100
    max_iter = 5000
    w_start = 0.9
    w_end = 0.1
    c1, c2 = 2.0, 2.0
    inertia_weight = w_start
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    
    for _ in range(swarm_size):
        h = h_init + np.random.normal(0, 0.01, n_points) if 'initial_h_values' in globals() else np.random.rand(n_points)
        h = repair_h(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_h(h, dx)
        pbest_positions.append(h.copy())
        pbest_scores.append(score)
    
    # Initialize global best
    global_best_idx = np.argmin(pbest_scores)
    global_best_position = pbest_positions[global_best_idx].copy()
    global_best_score = pbest_scores[global_best_idx]
    
    last_improvement_iter = 0
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        # Update inertia weight
        inertia_weight = w_start - (w_start - w_end) * (iteration / max_iter)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (inertia_weight * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (global_best_position - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair position
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate new position
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = particles[i].copy()
                    last_improvement_iter = iteration
        
        # Local search refinement every few iterations
        if iteration % 20 == 0 and iteration > 0:
            # Gradient-based refinement of global best
            h_refined = global_best_position.copy()
            learning_rate = 1e-3
            for _ in range(200):
                if time.time() - start_time > budget_s - 5:
                    break
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                max_k_idx = np.argmax(corr)
                # Simple finite difference gradient approximation
                eps = 1e-6
                grad_approx = np.zeros_like(h_refined)
                for j in range(len(h_refined)):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    grad_approx[j] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
                grad_approx = grad_approx / (np.linalg.norm(grad_approx) + 1e-12)
                h_refined = h_refined - learning_rate * grad_approx
                h_refined = repair_h(h_refined, n_points)
                new_c5 = evaluate_h(h_refined, dx)
                if new_c5 < global_best_score:
                    global_best_score = new_c5
                    global_best_position = h_refined.copy()
        
        iteration += 1
        # Early termination if no improvement
        if iteration - last_improvement_iter > 100:
            break
    
    # Final refinement of the best solution
    h_final = global_best_position.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        corr = correlate(h_final, 1 - h_final, mode='full')
        max_k_idx = np.argmax(corr)
        eps = 1e-6
        grad_approx = np.zeros_like(h_final)
        for j in range(len(h_final)):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad_approx[j] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
        grad_approx = grad_approx / (np.linalg.norm(grad_approx) + 1e-12)
        h_final = h_final - 1e-5 * grad_approx
        h_final = repair_h(h_final, n_points)
        new_c5 = evaluate_h(h_final, dx)
        if new_c5 < global_best_score:
            global_best_score = new_c5
            global_best_position = h_final.copy()
    
    return (global_best_position.tolist(), global_best_score, n_points)
