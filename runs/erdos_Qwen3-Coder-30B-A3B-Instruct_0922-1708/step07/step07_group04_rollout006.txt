import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Phase 1: Convex optimization with CVXPy for initial refinement
    # Formulate as convex QP
    h = cp.Variable(n_points)
    constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
    # Approximate the max correlation with squared sum
    # This is a heuristic for convex relaxation
    corr = correlate(h.value, 1 - h.value, mode='full')
    objective = cp.Minimize(cp.sum_squares(corr))
    problem = cp.Problem(objective, constraints)
    
    # Solve with warm start
    try:
        problem.solve(solver=cp.CVXOPT, verbose=False, max_iters=5000)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_cvx = h.value
            h_cvx = repair_h(h_cvx, n_points)
            c5_cvx = evaluate_h(h_cvx, dx)
            if c5_cvx < c5_best:
                h_best = h_cvx
                c5_best = c5_cvx
                h_current = h_cvx
    except:
        pass
    
    # Phase 2: Coordinate-wise proximal gradient
    max_iter = 10000
    eta = 0.01
    eps = 1e-6
    patience = 100
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Gradient via finite differences
        h_new = h_current.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with proximal gradient
            h_new[j] = h_current[j] - eta * grad_j
            h_new[j] = np.clip(h_new[j], 0, 1)

        # Repair step to maintain feasibility and integral constraint
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current or np.random.rand() < 0.01:
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        
        # Adaptive learning rate decay
        if i - last_improvement > patience:
            eta *= 0.95
            last_improvement = i
    
    return (h_best.tolist(), c5_best, n_points)
