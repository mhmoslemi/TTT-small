import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip with 30% chance
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
        
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    population = []
    best_h = None
    best_c5 = float('inf')
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
        population.append(h_init)
        c5 = evaluate_h(h_init, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_init.copy()
    else:
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
        population.append(h_init)
        c5 = evaluate_h(h_init, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_init.copy()
    
    # Generate additional initial solutions with various patterns
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        population.append(h)
    
    # Evolve population
    mutation_scale = 0.1
    stagnation_counter = 0
    stagnation_threshold = 100
    
    for generation in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all candidates
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
        
        # Selection: keep top 20%
        sorted_indices = np.argsort(fitnesses)
        top_indices = sorted_indices[:population_size // 5]
        top_candidates = [population[i] for i in top_indices]
        
        # Update best
        best_idx = sorted_indices[0]
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Generate new population
        new_population = top_candidates.copy()
        
        # Add mutated versions of top performers
        while len(new_population) < population_size:
            # Select two parents for crossover
            parent1 = top_candidates[np.random.randint(len(top_candidates))]
            parent2 = top_candidates[np.random.randint(len(top_candidates))]
            
            # Simple averaging crossover
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutate
            child = mutate(child, generation, mutation_scale)
            new_population.append(child)
        
        # Truncate to population size
        population = new_population[:population_size]
        
        # Adaptive mutation scaling
        if stagnation_counter > stagnation_threshold:
            mutation_scale *= 0.5
            stagnation_counter = 0
        
        # Decrease mutation scale gradually
        mutation_scale *= 0.999
    
    # Final verification and evaluation
    best_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), best_c5, n_points)
