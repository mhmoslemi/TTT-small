import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize swarm
    n_particles = 50
    w = 0.7
    c1, c2 = 2.0, 2.0
    T = 1.0
    cooling_rate = 0.999
    
    # Position and velocity
    positions = np.random.rand(n_particles, n_points)
    velocities = np.random.rand(n_particles, n_points) - 0.5
    personal_best = positions.copy()
    personal_best_fitness = np.full(n_particles, float('inf'))
    
    # Global best
    global_best_idx = 0
    global_best_fitness = float('inf')
    global_best = None
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        # Use the initial solution for one particle
        positions[0] = h_init * (n_points / 2) / np.sum(h_init)
    
    # Normalize all particles to satisfy sum constraint
    for i in range(n_particles):
        positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
    
    # Initialize personal bests
    for i in range(n_particles):
        h = positions[i]
        corr = correlate(h, 1 - h, mode='full')
        fitness = np.max(corr) * dx
        personal_best_fitness[i] = fitness
        if fitness < personal_best_fitness[i]:
            personal_best[i] = h.copy()
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best = h.copy()
    
    # PSO optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Update particles
        for i in range(n_particles):
            h = positions[i]
            
            # Compute current fitness
            corr = correlate(h, 1 - h, mode='full')
            current_fitness = np.max(corr) * dx
            
            # Update personal best
            if current_fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = current_fitness
                personal_best[i] = h.copy()
                
                # Update global best
                if current_fitness < global_best_fitness:
                    global_best_fitness = current_fitness
                    global_best = h.copy()
            
            # Velocity update
            r1, r2 = np.random.rand(2)
            v = (w * velocities[i] + 
                 c1 * r1 * (personal_best[i] - h) + 
                 c2 * r2 * (global_best - h))
            
            # Apply simulated annealing step
            if np.random.rand() < T:
                v += np.random.uniform(-0.1, 0.1, n_points)
            
            velocities[i] = v
            
            # Update position
            positions[i] = positions[i] + velocities[i]
            
            # Repair constraints
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = positions[i] * (n_points / 2) / np.sum(positions[i])
        
        # Update temperature
        T = max(T * cooling_rate, 0.01)
    
    # Final evaluation
    corr = correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(corr) * dx
    return (global_best.tolist(), final_c5, n_points)
