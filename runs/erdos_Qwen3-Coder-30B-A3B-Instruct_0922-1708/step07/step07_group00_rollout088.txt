import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Iterative optimization with multiple strategies
    max_iter = 10000
    step_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Strategy 1: Gradient descent with adaptive step size
    learning_rate = 1e-4
    decay_rate = 0.995
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        corr_base = correlate(current_h, 1 - current_h, mode='full')
        max_k_idx = np.argmax(corr_base)
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[j] = (corr_plus[max_k_idx] - corr_base[max_k_idx]) / eps
            
        # Update step
        current_h = current_h - learning_rate * grad
        current_h = repair_h(current_h, n_points)
        current_c5 = evaluate_h(current_h, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate
        step_count += 1
        if step_count % 50 == 0:
            learning_rate *= decay_rate
            
        # Early stopping
        if no_improvement_count >= max_no_improvement:
            break
    
    # Strategy 2: Local search around best solution
    h_refined = best_h.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Random perturbation and local optimization
        perturbation = np.random.normal(0, 0.01, n_points)
        h_trial = h_refined + perturbation
        h_trial = repair_h(h_trial, n_points)
        
        trial_c5 = evaluate_h(h_trial, dx)
        if trial_c5 < best_c5:
            h_refined = h_trial.copy()
            best_c5 = trial_c5
            # Early termination if improvement is small
            if abs(best_c5 - current_c5) < 1e-6:
                break
    
    # Final verification
    final_h = repair_h(h_refined, n_points)
    final_c5 = evaluate_h(final_h, dx)
    
    return (final_h.tolist(), float(final_c5), n_points)
