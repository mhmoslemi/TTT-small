import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, k_max, n_points):
    """Compute exact gradient for the maximum overlap shift k_max"""
    # Create shifted version of h for the k_max shift
    h_shifted = np.zeros(n_points)
    if k_max >= 0:
        h_shifted[k_max:] = h[:-k_max]
    else:
        h_shifted[:k_max] = h[-k_max:]
    
    # Gradient is (1 - h_shifted) for each position
    grad = 1 - h_shifted
    return grad

def find_max_overlap_shift(h):
    """Find the shift k that maximizes overlap"""
    corr = correlate(h, 1 - h, mode='full')
    max_k = np.argmax(corr)
    # Adjust k to be in range [-n_points+1, n_points-1]
    max_k = max_k - (len(h) - 1)
    return max_k

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient descent parameters
    learning_rate = 0.05
    max_iter = int(1e6)
    patience = 100
    last_improvement = 0
    adaptive_lr = True
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find the shift that maximizes overlap
        k_max = find_max_overlap_shift(h_current)
        
        # Compute the exact gradient for this shift
        grad = compute_gradient(h_current, k_max, n_points)
        
        # Update with gradient
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept move if improvement
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse moves with low probability
            if np.random.rand() < 0.005:
                h_current = h_new.copy()
                c5_current = c5_new
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if adaptive_lr and i - last_improvement > patience:
            learning_rate *= 0.95
            if learning_rate < 1e-5:
                learning_rate = 1e-5
            last_improvement = i
            
        # Perturb periodically to escape local minima
        if i % 500 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Final local refinement
    best_h = h_best.copy()
    best_c5 = c5_best
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = best_h.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return (best_h.tolist(), best_c5, n_points)
