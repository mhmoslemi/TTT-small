import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(params, n_points):
    # params: [a1, a2, ..., aK, x2, x3, ..., xK]
    # x1 = 0, xK+1 = 2
    K = (len(params) - 1) // 2
    heights = params[:K]
    boundaries = np.concatenate([[0], params[K:]])
    boundaries = np.append(boundaries, [2])
    
    # Normalize boundaries
    boundaries = np.clip(boundaries, 0, 2)
    boundaries = np.sort(boundaries)
    
    # Reconstruct h
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(K):
        start_idx = int(boundaries[i] / dx)
        end_idx = int(boundaries[i+1] / dx)
        if end_idx > start_idx:
            h[start_idx:end_idx] = heights[i]
    
    return h

def repair_params(params, K):
    # Repair parameters
    heights = np.clip(params[:K], 0, 1)
    boundaries = np.concatenate([[0], params[K:]])
    boundaries = np.sort(np.clip(boundaries, 0, 2))
    boundaries = np.append(boundaries, [2])
    
    # Rescale heights to match constraint
    total_area = 0
    dx = 2.0 / 100
    for i in range(K):
        start_idx = int(boundaries[i] / dx)
        end_idx = int(boundaries[i+1] / dx)
        area = heights[i] * (boundaries[i+1] - boundaries[i])
        total_area += area
    
    if total_area > 0:
        heights = heights * (50 / total_area)
    
    return np.concatenate([np.clip(heights, 0, 1), boundaries[1:-1]])

def compute_gradient(params, n_points, dx, epsilon=1e-6):
    K = (len(params) - 1) // 2
    current_h = parameterize_h(params, n_points)
    current_c5 = evaluate_h(current_h, dx)
    
    grad = np.zeros_like(params)
    for i in range(len(params)):
        params_plus = params.copy()
        params_plus[i] += epsilon
        h_plus = parameterize_h(params_plus, n_points)
        h_plus = repair_h(h_plus, n_points)
        c5_plus = evaluate_h(h_plus, dx)
        grad[i] = (c5_plus - current_c5) / epsilon
    
    # Normalize gradient
    norm = np.linalg.norm(grad) + 1e-12
    return grad / norm

        heights = []

        segments = np.array_split(h_init, K)
        for i in range(K):
