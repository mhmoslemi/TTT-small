```python
import numpy as np
import time
from scipy.signal import correlate
from cvxpy import *

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Binary search for the minimal C
    C_low = 0.0
    C_high = 0.5
    C_best = 0.5
    h_best = h_init.copy()
    
    # Very conservative max iterations to avoid exceeding budget
    max_iter = 50
    tolerance = 1e-5
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        C = (C_low + C_high) / 2.0
        
        # Try to find a feasible h with this C
        h_vars = Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1]
        
        # Add the constraint for all lags k
        for k in range(n_points):
            if k == 0:
                continue  # Skip for zero lag
            # Sum of h[i] * (1 - h[i+k]) for valid indices
            expr = 0
            for i in range(n_points - k):
                expr += h_vars[i] * (1 - h_vars[i + k])
            constraints.append(expr <= C * dx)
        
        # Add the integral constraint
        constraints.append(sum(h_vars) == n_points / 2)
        
        # Minimize a dummy objective (feasibility check)
        prob = Problem(Minimize(0), constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_feasible = h_vars.value
                h_feasible = project_h(h_feasible, n_points)
                c5_feasible = evaluate_h(h_feasible, dx)
                if c5_feasible < C_best:
                    h_best = h_feasible
                    C_best = c5_feasible
                C_high = C
            else:
                C_low = C
        except:
            C_low = C
    
    return (h_best.tolist(), C_best, n_points)
```
