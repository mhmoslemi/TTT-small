import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def find_max_shift(h):
    """Find the shift k that maximizes the overlap."""
    corr = correlate(h, 1 - h, mode='full')
    k_max = np.argmax(corr)
    return k_max

def compute_gradient(h, k_max, n_points):
    """Compute gradient with respect to h at k_max."""
    # The gradient w.r.t. h[i] is (1 - h[i-k_max]) for valid indices.
    # When k_max > i, we look at the shifted positions properly.
    grad = np.zeros(n_points)
    # Reconstruct h shifted by k_max (full correlation indexing)
    shift = k_max - (n_points - 1)  # offset from mode='full'
    for i in range(n_points):
        j = i - shift
        if 0 <= j < n_points:
            grad[i] = 1 - h[j]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)

    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current

    # Phase 1: CCCP-type optimization with convex approximation
    learning_rate = 0.01
    for i in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find best shift
        k_max = find_max_shift(h_current)
        
        # Compute gradient
        grad = compute_gradient(h_current, k_max, n_points)
        
        # Update with gradient descent
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject based on improvement
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse moves (simulated annealing)
            if np.random.rand() < 0.01:
                h_current = h_new.copy()
                c5_current = c5_new
                
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current

        # Adaptive learning rate
        if i % 500 == 0 and i > 0:
            learning_rate *= 0.95

        # Perturbations
        if i % 500 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)

    # Phase 2: Cyclic coordinate descent
    for i in range(10000):
        if time.time() - start_time > budget_s - 5:
            break

        # Cycle through all coordinates
        for j in range(n_points):
            # Compute the shift which maximizes overlap with current h
            k_max = find_max_shift(h_best)
            
            # Gradient at current shift
            # We approximate the actual gradient accordingly
            shift = k_max - (n_points - 1)  # offset from mode='full'
            grad_current = 0
            k = j - shift
            if 0 <= k < n_points:
                grad_current = 1 - h_best[k]

            # Update
            h_trial = h_best.copy()
            h_trial[j] -= learning_rate * grad_current
            h_trial = project_h(h_trial, n_points)
            c5_trial = evaluate_h(h_trial, dx)

            # Accept improvement
            if c5_trial < c5_best:
                h_best = h_trial.copy()
                c5_best = c5_trial
            else:
                # Occasionally accept slight worsening
                if np.random.rand() < 0.01:
                    h_best = h_trial.copy()
                    c5_best = c5_trial

        # Slight perturbation to escape local minima
        if i % 1000 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_best = project_h(h_best + noise, n_points)
            c5_best = evaluate_h(h_best, dx)

    return (h_best.tolist(), c5_best, n_points)
