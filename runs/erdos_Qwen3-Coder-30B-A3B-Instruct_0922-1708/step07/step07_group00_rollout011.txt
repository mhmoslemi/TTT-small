import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Try different optimization strategies
    strategies = [
        # Strategy 1: Gradient descent with restarts
        lambda h, dx: optimize_gradient_descent(h, dx, start_time, budget_s),
        # Strategy 2: Simple Nelder-Mead
        lambda h, dx: optimize_nelder_mead(h, dx, start_time, budget_s),
    ]
    
    for strategy in strategies:
        if time.time() - start_time > budget_s - 5:
            break
            
        try:
            h_strategy, c5_strategy = strategy(current_h.copy(), dx)
            if c5_strategy < best_c5:
                best_h = h_strategy
                best_c5 = c5_strategy
        except Exception:
            continue
    
    # Final refinement with gradient descent
    if time.time() - start_time < budget_s - 5:
        final_h, final_c5 = optimize_gradient_descent(best_h, dx, start_time, budget_s)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            
    return (best_h.tolist(), best_c5, n_points)

def optimize_gradient_descent(h0, dx, start_time, budget_s):
    h = h0.copy()
    current_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Compute gradient using finite differences
    def compute_gradient(h, dx):
        n = len(h)
        grad = np.zeros(n)
        eps = 1e-6
        
        # For small perturbations, we approximate the gradient
        corr_base = correlate(h, 1 - h, mode='full')
        base_max_k_idx = np.argmax(corr_base)
        base_max_k = base_max_k_idx - (n - 1)
        
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (corr_plus[base_max_k_idx] - corr_base[base_max_k_idx]) / eps
        
        # Normalize the gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-12:
            grad = grad / grad_norm
            
        return grad
    
    max_iter = 5000
    learning_rate = 5e-4
    decay_rate = 0.997
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        grad = compute_gradient(h, dx)
        h = h - learning_rate * grad
        h = repair_h(h, len(h))
        c5 = evaluate_h(h, dx)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        learning_rate *= decay_rate
        
    return best_h, best_c5

def optimize_nelder_mead(h0, dx, start_time, budget_s):
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - 100.0  # n_points / 2 = 100.0
    
    n_points = len(h0)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use Nelder-Mead with constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimizer parameters
    options = {'maxiter': 2000, 'disp': False, 'maxfev': 3000}
    
    # Run Nelder-Mead
    result = minimize(
        objective, 
        h0, 
        method='Nelder-Mead', 
        bounds=bounds, 
        constraints=constraints,
        options=options,
        tol=1e-6
    )
    
    if result.success:
        h_opt = result.x.copy()
        h_opt = repair_h(h_opt, n_points)
        return h_opt, evaluate_h(h_opt, dx)
    else:
        return h0, evaluate_h(h0, dx)
