import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def create_initial_population(n_points, n_pop=50, seed=42):
    """Create initial population with diversity."""
    np.random.seed(seed)
    population = []
    
    # Base solution from initial_h_values if available
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values)
        if len(base) != n_points:
            x_orig = np.linspace(0, 2, len(base))
            x_target = np.linspace(0, 2, n_points)
            base = np.interp(x_target, x_orig, base)
        base = project_h(base, n_points, 2.0/n_points)
    else:
        # Structured solution
        base = np.zeros(n_points)
        mid = n_points // 2
        base[:mid] = 0.5
        base[mid:] = 0.0
        base = project_h(base, n_points, 2.0/n_points)
    
    for _ in range(n_pop):
        # Add variation to base solution
        candidate = base.copy()
        # Add small random noise
        noise = np.random.normal(0, 0.05, n_points)
        candidate += noise
        candidate = project_h(candidate, n_points, 2.0/n_points)
        population.append(candidate)
    
    return population

def crossover(parent1, parent2, n_points):
    """Perform segment-wise crossover."""
    # Simple segment-wise blend
    crossover_point = np.random.randint(1, n_points-1)
    child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    return child

def mutate(individual, n_points, mutation_scale=0.1):
    """Apply mutation with segment flipping and noise."""
    # Segment flipping
    if np.random.random() < 0.3:
        start = np.random.randint(0, n_points)
        length = np.random.randint(1, n_points//4)
        end = min(start + length, n_points)
        individual[start:end] = 1 - individual[start:end]
    
    # Add Gaussian noise
    noise = np.random.normal(0, mutation_scale, n_points)
    individual += noise
    individual = project_h(individual, n_points, 2.0/n_points)
    return individual

def local_refine(h, n_points, dx, max_iter=50):
    """Fast local refinement using gradient approximation."""
    h_new = h.copy()
    eta = 0.01
    eps = 1e-6
    
    for _ in range(max_iter):
        h_old = h_new.copy()
        
        # Coordinate-wise update
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices:
            # Finite differences gradient approximation
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
        
        h_new = project_h(h_new, n_points, dx)
        
        # Early stopping if improvement is small
        if np.allclose(h_old, h_new, atol=1e-6):
            break
            
        eta = max(0.001, eta * 0.999)
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Create population
    population = create_initial_population(n_points, seed=seed)
    n_pop = len(population)
    
    # Fitness evaluation
    fitness = [evaluate_h(ind, dx) for ind in population]
    
    # Best solution tracking
    best_idx = np.argmin(fitness)
    h_best = population[best_idx].copy()
    c5_best = fitness[best_idx]
    
    # Main memetic loop
    generation = 0
    patience = 0
    max_patience = 50
    mutation_scale = 0.1
    
    while time.time() - start_time < budget_s - 5 and patience < max_patience:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:n_pop//2]
        
        # Create offspring
        offspring = []
        for _ in range(n_pop):
            # Select parents
            p1_idx = np.random.choice(elite_indices)
            p2_idx = np.random.choice(elite_indices)
            
            child = crossover(population[p1_idx], population[p2_idx], n_points)
            child = mutate(child, n_points, mutation_scale)
            
            # Local refinement
            child = local_refine(child, n_points, dx, max_iter=50)
            offspring.append(child)
        
        # Evaluate offspring
        offspring_fitness = [evaluate_h(ind, dx) for ind in offspring]
        
        # Replace worst population members
        worst_indices = sorted_indices[-n_pop//2:]
        for i, idx in enumerate(worst_indices):
            if offspring_fitness[i] < fitness[idx]:
                population[idx] = offspring[i]
                fitness[idx] = offspring_fitness[i]
        
        # Update best solution
        new_best_idx = np.argmin(fitness)
        if fitness[new_best_idx] < c5_best:
            h_best = population[new_best_idx].copy()
            c5_best = fitness[new_best_idx]
            patience = 0
        else:
            patience += 1
        
        # Update mutation scale
        mutation_scale *= 0.99
        
        # Occasionally inject random individuals if stagnation
        if patience > 10 and np.random.random() < 0.1:
            for i in range(5):  # Inject 5 random individuals
                random_ind = np.random.rand(n_points)
                random_ind = project_h(random_ind, n_points, dx)
                random_fitness = evaluate_h(random_ind, dx)
                worst_idx = np.argmax(fitness)
                if random_fitness < fitness[worst_idx]:
                    population[worst_idx] = random_ind
                    fitness[worst_idx] = random_fitness
    
    return (h_best.tolist(), c5_best, n_points)
