import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Try convex relaxation for better solution
    try:
        # Define the optimization variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)

        # Objective: approximate the correlation-maximization
        # We instead seek min max_k int h(x) (1-h(x+k)) dx via convex relaxation
        # This is very hard to directly convex, so we attempt a heuristic
        # Let's do a safeguarded gradient descent variant that is more effective
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        h_best = h_current.copy()
        c5_best = c5_current

        # Gradient descent with coordinate wise updates, different stepsize strategy
        max_iter = 10000
        eta = 0.01

        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Simple coordinate descent for small number of updates per iteration
            utils_indices = list(range(n_points))
            np.random.shuffle(utils_indices)
            
            for j in utils_indices[:10]:  # Small subset for update
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                
                # Small finite differences
                perturb = 1e-4
                h_plus[j] = min(1.0, h_plus[j] + perturb)
                h_minus[j] = max(0.0, h_minus[j] - perturb)
                
                c5_plus = evaluate_h(h_plus, dx)
                c5_minus = evaluate_h(h_minus, dx)
                
                grad_j = (c5_plus - c5_minus) / (2 * perturb)
                
                # Update with adaptive step size
                h_current[j] -= eta * grad_j
                h_current[j] = max(0, min(1, h_current[j]))

            # Re-project just to be sure
            h_current = project_h(h_current, n_points, dx)
            c5_new = evaluate_h(h_current, dx)
            
            # Accept move
            if c5_new < c5_current:
                h_current = h_current.copy()
                c5_current = c5_new
                if c5_current < c5_best:
                    c5_best = c5_current
                    h_best = h_current.copy()
            else:
                # Reduce learning rate if no improvement
                eta *= 0.999
                
        if c5_best < evaluate_h(h_init, dx):
            return (h_best.tolist(), c5_best, n_points)
        else:
            # Return the non-improved version to be safe
            return (h_init.tolist(), c5_current, n_points)
    except:
        # Fallback to standard gradient descent if solver fails
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        h_best = h_current.copy()
        c5_best = c5_current

        # Simple adaptive gradient descent
        max_iter = 10000
        eta = 0.01
        patience = 30
        last_improvement = 0

        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Gradient approximation by finite differences
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            
            j = np.random.randint(0, n_points)
            perturb = 1e-4
            
            h_plus[j] = min(1.0, h_plus[j] + perturb)
            h_minus[j] = max(0.0, h_minus[j] - perturb)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * perturb)
            
            # Update
            h_current[j] -= eta * grad_j
            h_current[j] = max(0, min(1, h_current[j]))
            
            # Project and re-evaluate
            h_current = project_h(h_current, n_points, dx)
            c5_new = evaluate_h(h_current, dx)
            
            if c5_new < c5_current:
                h_current = h_current.copy()
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
            else:
                if i - last_improvement > patience:
                    h_current = h_init.copy()  # restart
                    last_improvement = i
                else:
                    # Decrement step size
                    eta *= 0.999
        
        return (h_best.tolist(), c5_best, n_points)
