import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
from sobol_seq import i4_sobol_generate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Generate multiple Sobol sequences for better coverage
    n_initials = 500
    sobol_points = i4_sobol_generate(n_points, n_initials)
    
    for i in range(n_initials):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate initial h from Sobol sequence
        h = sobol_points[i, :]
        h = repair_h(h, n_points)
        
        # Nelder-Mead optimization
        def objective(h_vec):
            h_fixed = repair_h(h_vec, n_points)
            return evaluate_h(h_fixed, dx)
        
        try:
            res = minimize(objective, h, method='Nelder-Mead', 
                          options={'maxiter': 1000, 'xtol': 1e-6, 'ftol': 1e-6})
            if res.success:
                h_opt = res.x
                h_opt = repair_h(h_opt, n_points)
                c5_opt = evaluate_h(h_opt, dx)
                
                # Refinement with gradient descent
                h_refined = h_opt.copy()
                lr = 1e-5
                for _ in range(1000):
                    if time.time() - start_time > budget_s - 5:
                        break
                    corr = correlate(h_refined, 1 - h_refined, mode='full')
                    max_k_idx = np.argmax(corr)
                    # Finite difference gradient
                    grad = np.zeros(n_points)
                    eps = 1e-6
                    for j in range(n_points):
                        h_plus = h_refined.copy()
                        h_plus[j] += eps
                        h_plus = repair_h(h_plus, n_points)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                        grad[j] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
                    
                    h_refined = h_refined - lr * grad
                    h_refined = repair_h(h_refined, n_points)
                    c5_refined = evaluate_h(h_refined, dx)
                    
                    if c5_refined < c5_opt:
                        h_opt = h_refined.copy()
                        c5_opt = c5_refined
                    else:
                        lr *= 0.999  # Adaptive decay
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
