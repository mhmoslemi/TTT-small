import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidate(n_points, seed=None):
    if seed is not None:
        np.random.seed(seed)
    # Generate a candidate using cosine and sine combinations
    k_vals = np.random.randint(1, 5, 3)  # 3 random frequencies
    a_vals = np.random.uniform(-0.5, 0.5, 3)  # Random amplitudes
    b_val = np.random.uniform(0.2, 0.8)  # DC component
    
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for k, a in zip(k_vals, a_vals):
        h += a * np.cos(k * np.pi * x) + a * np.sin(k * np.pi * x)
    h += b_val
    h = repair_h(h, n_points)
    return h

    start_time = time.time()

        h_init = repair_h(h_init, n_points)
        candidates.append(h_init)
    
    # Generate diverse candidates using Fourier basis
    for i in range(50):
        if time.time() - start_time > budget_s - 5:
            break
        candidate = generate_fourier_candidate(n_points, seed=seed + i)
        candidates.append(candidate)
    
    # Evaluate all candidates
    candidate_scores = []
    for h in candidates:
        score = evaluate_h(h, dx)
        candidate_scores.append((score, h))
    
    # Sort by score and select top candidates
    candidate_scores.sort(key=lambda x: x[0])
    top_candidates = [h for score, h in candidate_scores[:10]]
    
    h_best = top_candidates[0]
    c5_best = candidate_scores[0][0]
    
    # Refine top candidates
    for h in top_candidates:
        if time.time() - start_time > budget_s - 5:
            break
        h_current = h.copy()
        c5_current = evaluate_h(h_current, dx)
        
        # Gradient descent
        lr = 0.01
        patience = 200
        last_improvement = 0
        
        for iter_count in range(2000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Finite differences gradient
            grad = np.zeros(n_points)
            epsilon = 1e-6
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[i] += epsilon
                h_minus[i] -= epsilon
                h_plus = repair_h(h_plus, n_points)
                h_minus = repair_h(h_minus, n_points)
                
                c5_plus = evaluate_h(h_plus, dx)
                c5_minus = evaluate_h(h_minus, dx)
                grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update
            h_new = h_current - lr * grad
            h_new = repair_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:

                    c5_best = c5_current

                lr *= 0.95
                if lr < 1e-6:
