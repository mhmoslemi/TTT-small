import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize particles
    swarm_size = 50
    particles = []
    velocities = []
    
    # Create one particle from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Structured initial solution: two-step function
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    # Add structured particles
    particles = [h_init.copy()]
    velocities = [np.zeros(n_points)]
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_h(h, n_points, dx)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Update personal bests and global best
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [evaluate_h(p, dx) for p in particles]
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    max_vel = 0.1
    
    # PSO iteration
    max_iter = 50000
    stagnation_count = 0
    stagnation_limit = 1000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * i / max_iter
        
        # Update particles
        for j in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_update = (c1 * r1 * (personal_best[j] - particles[j]) +
                          c2 * r2 * (global_best - particles[j]))
            velocities[j] = w * velocities[j] + vel_update
            
            # Limit velocity
            velocities[j] = np.clip(velocities[j], -max_vel, max_vel)
            
            # Update position
            particles[j] = particles[j] + velocities[j]
            
            # Project to feasible region
            particles[j] = project_h(particles[j], n_points, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(particles[j], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[j]:
                personal_best[j] = particles[j].copy()
                personal_best_fitness[j] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = particles[j].copy()
                    global_best_fitness = fitness
                    stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Perturb stagnated particles
        if stagnation_count > stagnation_limit:
            for j in range(swarm_size):
                if np.random.rand() < 0.3:
                    perturb = np.random.normal(0, 0.05, n_points)
                    particles[j] = particles[j] + perturb
                    particles[j] = project_h(particles[j], n_points, dx)
            
            stagnation_count = 0
            
    return (global_best.tolist(), global_best_fitness, n_points)
