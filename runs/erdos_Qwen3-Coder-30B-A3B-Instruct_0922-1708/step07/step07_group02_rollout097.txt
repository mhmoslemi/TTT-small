import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    eta = 0.01
    max_iter = 100000
    patience = 100
    last_improvement = 0
    k_max = 0
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Get correlation values for all lags
        corr = correlate(h_current, 1 - h_current, mode='full')
        # Find the maximum lag k
        k_max = np.argmax(corr)
        c5_current = corr[k_max] * dx
        
        # Compute gradient analytically
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Gradient contribution from correlation at k_max
            if j + k_max < n_points:
                grad[j] = -1 * dx  # since 1-h[j+k_max] contributes to the maximum
            
        # Update with gradient descent
        h_new = h_current - eta * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Project to feasible region
        h_new = project_h(h_new, n_points, dx)
        
        # Evaluate the new solution
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            # Update best if improved
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            # Update learning rate (adaptive)
            eta = max(1e-6, eta * 0.999)
            
        # Add perturbations to avoid getting stuck
        if i - last_improvement > patience:
            h_perturb = h_current.copy()
            num_perturb = max(1, int(0.05 * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                delta = np.random.normal(0, 0.02)
                h_perturb[idx] += delta
            h_perturb = project_h(h_perturb, n_points, dx)
            c5_perturb = evaluate_h(h_perturb, dx)
            if c5_perturb < c5_current:
                h_current = h_perturb
                c5_current = c5_perturb
                last_improvement = i
                
    return (h_best.tolist(), c5_best, n_points)
