import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of the maximum overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Identifying the k that gives maximum overlap
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n - 1)
    
    # Computing gradient for this k
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    # Normalize the gradient
    grad = grad / (np.linalg.norm(grad) + 1e-12)
    
    return grad

    else:

    # PSO parameters
    S = 100  # swarm size
    max_iter = 10000
    c1, c2 = 2.0, 2.0
    w_start, w_end = 0.9, 0.1
    inertia = w_start
    
    # Initialize particles and velocities
    particles = np.random.rand(S, n_points) * 0.5
    for i in range(S):
        particles[i] = repair_h(particles[i], n_points)
    
    # Initialize personal bests
    pbest = particles.copy()
    pbest_scores = np.array([evaluate_h(p, dx) for p in particles])
    
    # Find global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    # Initialize velocities
    velocities = (np.random.rand(S, n_points) - 0.5) * 0.1
    
    # PSO optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia
        inertia = w_start - (w_start - w_end) * i / max_iter
        
        # Update each particle
        for j in range(S):
            r1, r2 = np.random.rand(2)
            velocities[j] = (
                inertia * velocities[j] +
                c1 * r1 * (pbest[j] - particles[j]) +
                c2 * r2 * (gbest - particles[j])
            )

            particles[j] = repair_h(particles[j], n_points)

            new_score = evaluate_h(particles[j], dx)
            
            # Update personal best
            if new_score < pbest_scores[j]:
                pbest[j] = particles[j].copy()
                pbest_scores[j] = new_score
                
                # Update global best
                if new_score < gbest_score:
                    gbest = particles[j].copy()
                    gbest_score = new_score
        
        # Occasionally apply local refinement to global best
        if i % 100 == 0:
            h_refined = gbest.copy()
