import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def create_initial_population(n_points, n_pop=50):
    """Create diverse initial population."""
    population = []
    for _ in range(n_pop):
        # Start with structured construction
        h = np.zeros(n_points)
        # Random piecewise constant function
        segments = np.random.randint(2, min(8, n_points//2))
        start = 0
        for i in range(segments):
            end = min(start + np.random.randint(1, n_points//segments), n_points)
            value = np.random.uniform(0, 1)
            h[start:end] = value
            start = end
        if start < n_points:
            h[start:] = np.random.uniform(0, 1)
        h = project_h(h, n_points, 2.0/n_points)
        population.append(h)
    return population

def crossover(parent1, parent2, n_points):
    """Blend crossover of two parents."""
    # Segment-wise blend
    h = np.zeros(n_points)
    segments = np.random.randint(2, min(10, n_points//4))
    start = 0
    for i in range(segments):
        end = min(start + np.random.randint(1, n_points//segments), n_points)
        if np.random.rand() < 0.5:
            h[start:end] = parent1[start:end]
        else:
            h[start:end] = parent2[start:end]
        start = end
    if start < n_points:
        h[start:] = parent1[start:] if np.random.rand() < 0.5 else parent2[start:]
    return h

def mutate(h, n_points, mutation_rate=0.3, noise_scale=0.1):
    """Apply mutation to h."""
    h_new = h.copy()
    # Random segment flips - 30% of segments
    segments = np.random.randint(1, min(20, n_points//2))
    for _ in range(segments):
        start = np.random.randint(0, n_points)
        end = min(start + np.random.randint(1, n_points//10), n_points)
        if np.random.rand() < mutation_rate:
            h_new[start:end] = np.random.uniform(0, 1)
        else:
            h_new[start:end] = h_new[start:end] + np.random.normal(0, noise_scale, end-start)
    return h_new

def local_refine(h, n_points, dx, max_iter=50):
    """Fast local refinement using coordinate descent."""
    h = h.copy()
    for _ in range(max_iter):
        h_new = h.copy()
        improved = False
        indices = list(range(n_points))
        np.random.shuffle(indices)
        eps = 1e-6
        eta = 0.01
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < evaluate_h(h, dx):
            h = h_new
            improved = True
        if not improved:
            eta *= 0.99
        else:
            eta = min(0.01, eta * 1.001)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    pop_size = 50
    population = create_initial_population(n_points, pop_size)
    
    # Best solution tracking
    h_best = None
    c5_best = float('inf')
    
    # Memetic algorithm parameters
    selection_ratio = 0.2
    mutation_scale = 0.1
    max_generations = 1500
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < c5_best:
                h_best = h.copy()
                c5_best = c5
        
        # Selection (top 20%)
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:int(pop_size * selection_ratio)]
        top_population = [population[i] for i in top_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size):
            parent1 = np.random.choice(top_population)
            parent2 = np.random.choice(top_population)
            child = crossover(parent1, parent2, n_points)
            child = mutate(child, n_points, mutation_scale, mutation_scale)
            child = project_h(child, n_points, dx)
            # Local refinement
            child = local_refine(child, n_points, dx, 50)
            offspring.append(child)
        
        # Replace worst in population
        worst_indices = sorted_indices[int(pop_size * selection_ratio):]
        for i, idx in enumerate(worst_indices):
            population[idx] = offspring[i]
            
        # Adapt mutation scale
        mutation_scale *= 0.99
        
        # Occasionally inject new diversity
        if gen % 20 == 0 and gen > 0:
            for _ in range(5):
                h_new = np.random.rand(n_points)
                h_new = project_h(h_new, n_points, dx)
                h_new = local_refine(h_new, n_points, dx, 20)
                # Replace worst with new random solution
                worst_fitness = max(fitness)
                worst_idx = fitness.index(worst_fitness) if worst_fitness in fitness else 0
                population[worst_idx] = h_new
        
    return (h_best.tolist(), c5_best, n_points)
