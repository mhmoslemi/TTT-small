import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(theta, n_points):
    """Convert parameters to h(x) function."""
    K = len(theta) // 2  # number of segments
    heights = theta[:K]
    boundaries = np.concatenate([[0], theta[K:]])
    
    # Ensure boundaries are in order and within [0, 2]
    boundaries = np.sort(np.clip(boundaries, 0, 2))
    boundaries = np.concatenate([boundaries, [2]])
    
    # Compute segment widths
    widths = boundaries[1:] - boundaries[:-1]
    
    # Compute h values
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(K):
        # Find indices in [boundaries[i], boundaries[i+1])
        start_idx = int(boundaries[i] / dx)
        end_idx = int(boundaries[i+1] / dx)
        if end_idx > start_idx:
            h[start_idx:end_idx] = heights[i]
            
    # Ensure the integral constraint
    h = repair_h(h, n_points)
    return h

def compute_gradient_param(h, dx, theta, K):
    """Compute gradient with respect to parameters."""
    n = len(h)
    eps = 1e-6
    grad = np.zeros_like(theta)
    
    # Find which k gives max overlap
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n - 1)
    
    # For each parameter
    for i in range(len(theta)):
        theta_plus = theta.copy()
        theta_plus[i] += eps
        h_plus = parameterize_h(theta_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    return grad

    else:

        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Parameterize h_init to get parameters
    # Use 5 segments for better control
    K = 5
    # Initialize parameters: first K are heights, next K-1 are boundary positions
    # Divide [0,2] into K segments, so we have K-1 boundary points between
    boundaries_init = np.linspace(0, 2, K+1)[1:-1]  # Exclude 0 and 2
    heights_init = np.ones(K) * 0.5
    theta_init = np.concatenate([heights_init, boundaries_init])
    
    # Optimization parameters
    max_iter = 5000
    learning_rate = 1e-3
    decay_rate = 0.995
    step_count = 0
    
    # Initialize optimization
    current_theta = theta_init.copy()
    h = parameterize_h(current_theta, n_points)

    best_h = h.copy()

    best_theta = current_theta.copy()
