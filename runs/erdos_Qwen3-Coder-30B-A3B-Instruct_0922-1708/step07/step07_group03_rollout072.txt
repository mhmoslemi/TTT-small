import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    # Define CVXPY variables
    h_var = cp.Variable(n_points)
    c5_var = cp.Variable(1, name='c5')
    
    # Set up constraints
    constraints = []
    
    # Constraint: h_var in [0,1]
    constraints.extend([h_var >= 0, h_var <= 1])
    
    # Constraint: sum(h_var) = n_points / 2
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Constraint: all lags <= c5_var
    # For each lag k, compute integral of h(x)(1-h(x+k))
    for k in range(1, 2 * n_points - 1):
        # Compute the correlation at lag k
        # 1-h_var is the reflected version but NumPy's correlate
        # handles this for us.
        # We compute correlation with the reflected h
        # This is fixed for each k ahead of time
        if k < n_points:
            # For negative lags
            corr_segment = h_var[k:] * (1 - h_var[:n_points-k])
            corr_val = cp.sum(corr_segment)
        else:
            # For positive lags
            corr_segment = h_var[:n_points-k] * (1 - h_var[k:])
            corr_val = cp.sum(corr_segment)
        
        constraints.append(corr_val <= c5_var)
        
    # Use a more efficient approach by avoiding explicit correlation
    # Set up the objective
    objective = cp.Minimize(c5_var)
    
    # Create the problem
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers = ['MOSEK', 'GUROBI', 'ECOS']
    solver_used = 'ECOS'  # default, fallback
    
    for solver in solvers:
        try:
            if solver == 'MOSEK' and cp.MOSEK in cp.installed_solvers():
                solver_used = solver
                break
            elif solver == 'GUROBI' and cp.GUROBI in cp.installed_solvers():
                solver_used = solver
                break
            elif solver == 'ECOS' and cp.ECOS in cp.installed_solvers():
                solver_used = solver
                break
        except:
            continue
    
    # Solve
    try:
        if solver_used == 'MOSEK':
            prob.solve(solver=cp.MOSEK, verbose=False, max_iters=10000)
        elif solver_used == 'GUROBI':
            prob.solve(solver=cp.GUROBI, verbose=False)
        elif solver_used == 'ECOS':
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=10000)
        else:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=10000)
    except Exception as e:
        pass  # fallback to gradient descent

    best_h = h
    best_c5 = float('inf')
    
    # Check if we got a solution
    if prob.status == 'optimal':
        h_solution = h_var.value
        if h_solution is not None:
            h_solution = np.clip(h_solution, 0, 1)
            h_sum = np.sum(h_solution)
            target_sum = n_points / 2
            if abs(h_sum - target_sum) > 1e-8:
                h_solution = h_solution * target_sum / h_sum
            corr = correlate(h_solution, 1 - h_solution, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_solution.copy()
        else:
            # fallback to gradient descent with patience
            h = best_h.copy()
    else:
        # fallback on gradient descent with patience
        h = best_h.copy()
        
    # Gradient descent fallback
    learning_rate = 1e-3
    max_iter = 50000
    
    # Final optimization with gradient descent
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break

        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Estimate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update step
        h = h - learning_rate * grad
        
        # Projection to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to meet constraint
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
