import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a binary-like initialization
        h = np.zeros(n_points)
        # Set roughly half the points to 1
        indices = np.random.choice(n_points, n_points//2, replace=False)
        h[indices] = 1.0
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cooling parameters
    initial_temp = 0.1
    cooling_rate = 0.995
    
    # Main optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        temp = initial_temp * (cooling_rate ** iteration)
        
        # Make perturbation
        h_new = h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.1, 0.1)
        
        # Adjust values
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy constraint
        h_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_sum
        
        # Compute C5
        corr = correlate(h_new, 1 - h_new, mode='full')
        c5 = np.max(corr) * dx
        
        # Accept or reject
        if c5 < best_c5:
            h = h_new.copy()
            best_c5 = c5
            best_h = h.copy()
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-(c5 - best_c5) / temp):
                h = h_new.copy()
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
