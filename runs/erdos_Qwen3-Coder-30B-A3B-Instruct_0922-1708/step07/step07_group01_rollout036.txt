import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with sine wave, trapezoidal, and random for multi-start
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Multi-start with different initialization strategies
    initializations = []
    
    # 1. Cosine taper
    x = np.linspace(0, 2, n_points)
    h1 = 0.5 * (1 + np.cos(np.pi * x))
    h1 = project_h(h1, n_points)
    initializations.append(h1)
    
    # 2. Piecewise constant
    h2 = np.zeros(n_points)
    mid = n_points // 2
    h2[:mid] = 0.5
    h2[mid:] = 1.5
    h2 = project_h(h2, n_points)
    initializations.append(h2)
    
    # 3. Random perturbation of initial solution
    if 'initial_h_values' in globals():
        h3 = np.array(initial_h_values) + np.random.normal(0, 0.05, n_points)
        h3 = project_h(h3, n_points)
        initializations.append(h3)
        
    # 4. Gaussian-shaped
    mu, sigma = 1.0, 0.2
    x = np.linspace(0, 2, n_points)
    h4 = np.exp(-0.5 * ((x - mu) / sigma)**2)
    h4 = project_h(h4, n_points)
    initializations.append(h4)
    
    # 5. Sine wave
    h5 = 0.5 + 0.5 * np.sin(2 * np.pi * x)
    h5 = project_h(h5, n_points)
    initializations.append(h5)
    
    # Optimize from each initialization
    for h_start in initializations:
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current, dx)
        best_from_start = h_current.copy()
        best_c5_from_start = c5_current
        
        # Local optimization with momentum
        learning_rate = 0.02
        momentum = 0.9
        velocity = np.zeros(n_points)
        patience = 100
        last_improvement = 0
        
        for iteration in range(10000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Exact gradient computation for max correlation
            corr_full = correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr_full)
            k_max = max_idx - (n_points - 1)  # shift value
            # Derivative for shift k_max: (1-h_{k_max}) for k_max shift
            # but we only care about derivative for objective
            # we're just approximating the shift that maximizes the overlap
            
            # Compute the derivative using the shift that maximizes the correlation:
            shifted = np.roll(1 - h_current, -k_max) if k_max > 0 else np.roll(1 - h_current, k_max)
            # Approximate gradient from derivative
            grad = shifted
            grad = np.clip(grad, 0, 1)
            
            # Update with momentum
            velocity = momentum * velocity - learning_rate * grad
            h_new = h_current + velocity
            
            # Project to constraints
            h_new = project_h(h_new, n_points)
            
            # Evaluate
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                if c5_current < best_c5_from_start:
                    best_c5_from_start = c5_current
                    best_from_start = h_current.copy()
                    last_improvement = iteration
            else:
                # Occasionally accept worse moves
                if np.random.rand() < 0.01:
                    h_current = h_new.copy()
                    c5_current = c5_new
            
            if iteration - last_improvement > patience:
                if learning_rate > 1e-5:
                    learning_rate *= 0.95
                # Random perturbation
                random_perturbation = np.random.normal(0, 0.005, n_points)
                h_current = project_h(h_current + random_perturbation, n_points)
                c5_current = evaluate_h(h_current, dx)
                last_improvement = iteration
            
            # Restore best from this initialization periodically
            if iteration % 500 == 0 and iteration > 0:
                h_current = best_from_start.copy()
                c5_current = best_c5_from_start
        
        # Update global best
        if best_c5_from_start < c5_best:
            h_best = best_from_start.copy()
            c5_best = best_c5_from_start
    
    # Final optimization of the best solution
    h_final = h_best.copy()
    c5_final = c5_best
    
    # Gradient descent refinement
    learning_rate = 0.01
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        corr_full = correlate(h_final, 1 - h_final, mode='full')
        max_idx = np.argmax(corr_full)
        k_max = max_idx - (n_points - 1)
        
        shifted = np.roll(1 - h_final, -k_max) if k_max > 0 else np.roll(1 - h_final, k_max)
        grad = shifted
        
        h_new = h_final - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_final:
            h_final = h_new
            c5_final = c5_new
        else:
            # Adaptive learning rate
            learning_rate *= 0.99
        
        if learning_rate < 1e-6:
            break
    
    return (h_final.tolist(), c5_final, n_points)
