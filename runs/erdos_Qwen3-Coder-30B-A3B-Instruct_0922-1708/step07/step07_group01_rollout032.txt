import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Dynamic Programming approach - construct h incrementally
    dp_iter = 0
    max_dp_iter = 5000
    
    for dp_iter in range(max_dp_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Try to improve by changing one element at a time
        best_delta = float('inf')
        best_idx = -1
        best_val = -1
        
        # Fine-tune one element at a time
        for i in range(n_points):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Try different values for h[i] 
            candidate_vals = [0.0, 0.25, 0.5, 0.75, 1.0]
            h_test = h_current.copy()
            
            for val in candidate_vals:
                h_test[i] = val
                h_test = project_h(h_test, n_points)
                c5_test = evaluate_h(h_test, dx)
                
                if c5_test < c5_current - 1e-10:  # We have an improvement
                    delta = c5_current - c5_test
                    if delta < best_delta:
                        best_delta = delta
                        best_idx = i
                        best_val = val
        
        # Apply the best improvement if one is found
        if best_idx != -1:
            h_current[best_idx] = best_val
            h_current = project_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            
            # Update best solution
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
        else:
            # No improvement found, try some random perturbations
            idx = np.random.randint(0, n_points)
            h_current[idx] = np.random.uniform(0, 1)
            h_current = project_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
    
    # Final refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 2:
            break
            
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 20), replace=False)
        for idx in indices:
            h_new[idx] = np.clip(h_new[idx] + np.random.normal(0, 0.01), 0, 1)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    # Final repair to exactly meet constraints
    h_best = repair_h(h_best, n_points)
    c5_best = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), c5_best, n_points)
