import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Start with a simpler heuristic: piecewise constant with 1s and 0s
        h = np.zeros(n_points)
        # Place 50 ones in the first 50 positions to satisfy ∫h = 1
        h[:50] = 1.0
    
    # Ensure constraints
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-8:
        h = h * (n_points / 2) / h_sum
    
    # Local search with perturbation
    max_iter = 100000
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated annealing parameters
    temp = 0.1
    cooling_rate = 0.9995
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate a new candidate by perturbing h
        h_new = h.copy()
        
        # Select two random indices to adjust
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Small random change
        delta = np.random.uniform(-0.02, 0.02)
        
        # Adjust values
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clamp to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint (e.g., redistribute excess/deficit)
        current_sum = np.sum(h_new)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-8:
            # Simple proportional adjustment
            scale_factor = target_sum / current_sum
            h_new *= scale_factor
        
        # Compute C5
        corr = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr) * dx
        
        # Accept or reject
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-(c5_new - best_c5) / (temp + 1e-10)):
                h = h_new.copy()
        
        # Cool down temperature
        temp *= cooling_rate
        
        # Occasionally reset to current best to avoid getting stuck
        if iteration % 1000 == 0 and iteration > 0:
            h = best_h.copy()
    
    # Final correction and evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
