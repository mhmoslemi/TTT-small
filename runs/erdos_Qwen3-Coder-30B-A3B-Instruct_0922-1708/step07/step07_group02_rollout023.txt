import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Gradient-based optimization
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    eta = 0.01
    eps = 1e-6
    patience = 100
    last_improvement = 0
    max_iter = 100000
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all correlations to find max
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute analytical gradient for the dominant term
        grad = np.zeros(n_points)
        if k_max < n_points:
            # Gradient for correlation at k_max
            for j in range(n_points):
                if j + k_max < n_points:
                    grad[j] += (1 - h_current[j + k_max]) * dx
        else:
            # Cross-correlation wraps around, handle differently
            for j in range(n_points):
                # Find the corresponding position in wrapped array
                idx = j + k_max - n_points
                if idx >= 0:
                    grad[j] += (1 - h_current[idx]) * dx
        
        # Update using gradient descent
        h_new = h_current - eta * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        
        # Project to maintain sum constraint
        h_new = project_h(h_new, n_points, dx)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            # Update best if improved
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            # Adaptive learning rate
            eta *= 0.999
            
        # Perturbation to escape local minima
        if i - last_improvement > patience:
            h_perturb = h_current.copy()
            num_perturb = max(1, int(0.01 * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                delta = np.random.normal(0, 0.02)
                h_perturb[idx] += delta
            h_perturb = project_h(h_perturb, n_points, dx)
            c5_perturb = evaluate_h(h_perturb, dx)
            if c5_perturb < c5_current:
                h_current = h_perturb
                c5_current = c5_perturb
                last_improvement = i
                
    return (h_best.tolist(), c5_best, n_points)
