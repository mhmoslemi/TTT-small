import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    initial_temp = 0.1
    cooling_rate = 0.995
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create binary initialization with correct sum
        h = np.zeros(n_points)
        ones_count = n_points // 2
        h[:ones_count] = 1.0
        np.random.shuffle(h)
    
    # Local search with simulated annealing
    best_h = h.copy()
    best_c5 = float('inf')
    
    current_h = h.copy()
    current_c5 = np.max(correlate(current_h, 1 - current_h, mode='full')) * dx
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    temp = initial_temp
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor solution
        h_new = current_h.copy()
        
        # Random perturbation
        i, j = np.random.randint(0, n_points, 2)
        delta = np.random.uniform(-0.05, 0.05)
        
        # Ensure feasibility
        if h_new[i] + delta >= 0 and h_new[i] + delta <= 1:
            h_new[i] += delta
            if h_new[j] - delta >= 0 and h_new[j] - delta <= 1:
                h_new[j] -= delta
                # Rescale to meet sum constraint
                h_sum = np.sum(h_new)
                target_sum = n_points / 2
                if abs(h_sum - target_sum) > 1e-8:
                    h_new = h_new * target_sum / h_sum
                
                # Evaluate
                c5_new = np.max(correlate(h_new, 1 - h_new, mode='full')) * dx
                
                # Accept or reject based on simulated annealing
                if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / temp):
                    current_h = h_new
                    current_c5 = c5_new
                    
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
        
        # Cool down temperature
        temp *= cooling_rate
    
    # Final check
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
