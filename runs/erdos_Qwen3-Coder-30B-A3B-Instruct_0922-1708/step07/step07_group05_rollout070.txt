import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidate(n_points, seed=None):
    if seed is not None:
        np.random.seed(seed)
    # Generate a candidate using Fourier-like modes
    h = np.zeros(n_points)
    # Use 4 basis functions to generate diverse patterns
    for i in range(n_points):
        x = i * 2.0 / n_points
        # Sum of cos and sin with random coefficients
        h[i] = (0.5 + 0.3 * np.cos(2 * np.pi * x * 2 + np.random.rand() * 2 * np.pi) +
                0.2 * np.sin(2 * np.pi * x * 3 + np.random.rand() * 2 * np.pi) +
                0.1 * np.cos(2 * np.pi * x * 5 + np.random.rand() * 2 * np.pi))
    h = repair_h(h, n_points)
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

        h_init = repair_h(h_init, n_points)
    else:
        # Start with a simple smooth construction
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Piecewise linear function
        for i in range(n_points):
            if x[i] < 0.5:
                h_init[i] = 0
            elif x[i] <= 1.5:
                h_init[i] = (x[i] - 0.5) / 1.0
            else:
                h_init[i] = 1
        h_init = repair_h(h_init, n_points)
        
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Phase 1: Generate diverse candidates using Fourier modes
    num_candidates = 50
    candidates = []
    for i in range(num_candidates):
        h_cand = generate_fourier_candidate(n_points, seed=seed + i)
        c5_cand = evaluate_h(h_cand, dx)
        candidates.append((h_cand, c5_cand))
        if c5_cand < c5_best:
            h_best = h_cand.copy()
            c5_best = c5_cand

    # Sort candidates by performance
    candidates.sort(key=lambda x: x[1])
    top_candidates = candidates[:10]  # Top 10 candidates to refine

    # Phase 2: Refinement with gradient descent
    max_refine_iter = 2000
    patience = 200
    for h_start, _ in top_candidates:
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current, dx)
        last_improvement = 0
        learning_rate = 0.01
        
        for i in range(max_refine_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient via finite differences
            epsilon = 1e-6
            grad = np.zeros(n_points)
            c5_base = evaluate_h(h_current, dx)
            
            for k in range(n_points):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[k] += epsilon
                h_minus[k] -= epsilon
                h_plus = repair_h(h_plus, n_points)
                h_minus = repair_h(h_minus, n_points)
                
                c5_plus = evaluate_h(h_plus, dx)
                c5_minus = evaluate_h(h_minus, dx)
                grad[k] = (c5_plus - c5_minus) / (2 * epsilon)
                
            # Update
            h_new = h_current - learning_rate * grad
            h_new = repair_h(h_new, n_points)
            
            # Evaluate
            c5_new = evaluate_h(h_new, dx)

            if c5_new < c5_current:

                if np.random.rand() < np.exp(-(c5_new - c5_current) / (c5_current * 0.01)):
                    h_current = h_new
                    c5_current = c5_new
                    
            # Adapt learning rate
            if i - last_improvement > patience:
                learning_rate *= 0.95
                if learning_rate < 1e-6:
                    break
    
    return (h_best.tolist(), c5_best, n_points)
