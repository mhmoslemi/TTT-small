import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidates(n_points, num_candidates=50):
    candidates = []
    for _ in range(num_candidates):
        # Generate a random linear combination of cosine/sine modes
        k_vals = np.random.randint(1, 6, size=3)
        a_vals = np.random.uniform(-0.5, 0.5, size=3)
        b_vals = np.random.uniform(-0.5, 0.5, size=3)
        
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        
        for k, a, b in zip(k_vals, a_vals, b_vals):
            h += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
        
        # Add a constant
        h += 0.5
        h = repair_h(h, n_points)
        candidates.append(h)
    
    return candidates

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from the parent
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        h_init = np.ones(n_points) * 0.5
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Phase 1: Generate diverse candidates
    candidates = generate_fourier_candidates(n_points, 50)
    candidates.append(h_init)
    
    # Evaluate all candidates
    eval_results = []
    for i, h in enumerate(candidates):
        c5 = evaluate_h(h, dx)
        eval_results.append((h, c5, i))
    
    # Sort candidates by C5 value
    eval_results.sort(key=lambda x: x[1])
    
    # Phase 2: Refine the best candidates using gradient descent
    num_refine = min(10, len(eval_results))
    patience = 200
    
    for h_start, c5_start, orig_idx in eval_results[:num_refine]:
        h_current = h_start.copy()
        c5_current = c5_start
        
        # Gradient descent refinement
        learning_rate = 0.01
        last_improvement = 0
        
        for i in range(2000):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Compute gradient using central differences
            grad = np.zeros(n_points)
            epsilon = 1e-6
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[j] += epsilon
                h_minus[j] -= epsilon
                h_plus = repair_h(h_plus, n_points)
                h_minus = repair_h(h_minus, n_points)
                
                c5_plus = evaluate_h(h_plus, dx)
                c5_minus = evaluate_h(h_minus, dx)
                grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update
            h_new = h_current - learning_rate * grad
            h_new = repair_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
            else:
                # Accept with probability decaying with time
                if np.random.rand() < np.exp(-(c5_new - c5_current) / (c5_current * 0.01)):
                    h_current = h_new
                    c5_current = c5_new
            
            if i - last_improvement > patience:
                learning_rate *= 0.95
                if learning_rate < 1e-6:
                    break
    
    return (h_best.tolist(), c5_best, n_points)
