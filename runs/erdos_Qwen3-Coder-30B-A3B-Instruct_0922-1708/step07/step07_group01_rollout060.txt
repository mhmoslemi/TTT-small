import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Multiple initializations
    initializations = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
        initializations.append(h_init.copy())
    else:
        # Add a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
        initializations.append(h_init.copy())
    
    # Add some random initializations
    for _ in range(3):
        h_rand = np.random.rand(n_points)
        h_rand = project_h(h_rand, n_points)
        initializations.append(h_rand.copy())
    
    # Add a trapezoidal waveform
    h_trap = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    # Create trapezoid shape
    half = n_points // 2
    h_trap[:half] = np.linspace(0, 1, half)
    h_trap[half:] = np.linspace(1, 0, n_points - half)
    h_trap = project_h(h_trap, n_points)
    initializations.append(h_trap.copy())
    
    # Add a second cosine with different phase
    h_cos2 = 0.5 * (1 + np.cos(np.pi * x + np.pi/4))
    h_cos2 = project_h(h_cos2, n_points)
    initializations.append(h_cos2.copy())
    
    # Optimization parameters
    max_iter = int(1e5)
    patience = 500
    learning_rate = 0.05
    epsilon = 1e-6
    
    h_best = None
    c5_best = float('inf')
    
    for h_init in initializations:
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
        
        last_improvement = 0
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros(n_points)
            
            # Get current maximum shift index
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr)
            
            # Compute gradient for max shift
            # Replicate the trick to compute exact gradient:
            # For shift k_max, the gradient of overlap with respect to h[i] 
            # is (1 - h[i+k_max]) (as h[i+k_max] is what gets multiplied with h[i])
            # However, it's easier to compute this by simulating finite difference
            # of the full correlation as a function of h.
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            
            for j in range(n_points):
                h_plus[j] += epsilon
                h_minus[j] -= epsilon
                h_plus = project_h(h_plus, n_points)
                h_minus = project_h(h_minus, n_points)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                
                grad[j] = (corr_plus[max_shift_idx] - corr_minus[max_shift_idx]) / (2 * epsilon)
                
                # Reset
                h_plus[j] = h_current[j]
                h_minus[j] = h_current[j]
            
            # Update
            h_new = h_current - learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept improvement
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                last_improvement = i
            else:
                # Occasionally accept worse moves
                if np.random.rand() < 0.02:
                    h_current = h_new.copy()
                    c5_current = c5_new
                    last_improvement = i
            
            # Update best
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
            
            # Adaptive learning rate
            if i - last_improvement > patience:
                learning_rate *= 0.95
                if learning_rate < 1e-6:
                    learning_rate = 1e-6
                last_improvement = i
            
            # Perturb periodically
            if i % 100 == 0 and i > 0:
                noise = np.random.normal(0, 0.005, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
                
            # Occasionally restore best solution
            if i % 500 == 0:
                h_current = h_best.copy()
                c5_current = c5_best
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
