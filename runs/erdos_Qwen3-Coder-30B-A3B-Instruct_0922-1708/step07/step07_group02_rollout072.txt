import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Gradient descent with analytical gradients
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    eta = 0.01
    patience = 50
    last_improvement = 0
    max_iter = 100000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all correlations to find max
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        max_corr = corr[k_max]
        
        # Compute gradient analytically for the maximum correlation term
        grad = np.zeros_like(h_current)
        if k_max < n_points:
            # Shift the gradient to match the contribution
            grad[k_max:] = -1 * (1 - h_current[:-k_max] if k_max > 0 else 1 - h_current)
        else:
            # Handle the case where k_max >= n_points 
            # (this would happen for the full shifted correlation)
            # In this case we're computing contributions from later parts
            # Add contribution only if it's valid
            offset = k_max - n_points + 1  # offset from right side of h
            if offset < n_points:
                # Forward shift indices
                grad[:n_points-offset] = -(1 - h_current[offset:])

        # Gradient step
        h_new = h_current - eta * grad
        
        # Ensure h remains in range [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum constraint
        h_new = project_h(h_new, n_points, dx)
        
        # Compute new C5 bound
        c5_new = evaluate_h(h_new, dx)
        
        # Update if improved
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
            # Increase learning rate
            eta = min(0.01, eta * 1.005)
        else:
            # Decrease learning rate
            eta *= 0.999
            
        # Check for stagnation
        if i - last_improvement > patience:
            # Add small noise to escape local minima
            noise = np.random.normal(0, 0.01, n_points)
            h_new = h_current + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = project_h(h_new, n_points, dx)
            c5_perturb = evaluate_h(h_new, dx)
            if c5_perturb < c5_current:
                h_current = h_new
                c5_current = c5_perturb
                last_improvement = i
    
    return (h_best.tolist(), c5_best, n_points)
