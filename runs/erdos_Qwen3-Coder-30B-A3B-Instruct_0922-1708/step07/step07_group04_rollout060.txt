import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate(h, mutation_rate=0.1, perturbation=0.1):
    """Apply mutation to h."""
    h_mut = h.copy()
    indices = np.random.choice(len(h), size=int(len(h) * mutation_rate), replace=False)
    for i in indices:
        h_mut[i] += np.random.normal(0, perturbation)
    return repair_h(h_mut, len(h))

def crossover(h1, h2):
    """Perform crossover between two solutions."""
    n = len(h1)
    point = np.random.randint(1, n-1)
    h_new = np.concatenate([h1[:point], h2[point:]])
    return repair_h(h_new, n)

def tournament_selection(population, fitness, k=3):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Initialize population
    population_size = 100
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h_random = np.random.rand(n_points)
        h_random = repair_h(h_random, n_points)
        population.append(h_random)
        
    # Evaluate initial population
    fitness = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    h_best = population[best_idx].copy()
    c5_best = fitness[best_idx]
    
    # Genetic algorithm parameters
    generations = 100
    mutation_rate = 0.1
    perturbation = 0.1
    crossover_rate = 0.8
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate current population
        fitness = [evaluate_h(h, dx) for h in population]
        
        # Create new population
        new_population = []
        
        # Keep best 50%
        sorted_indices = np.argsort(fitness)
        for i in range(population_size // 2):
            new_population.append(population[sorted_indices[i]])
        
        # Fill remaining slots
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            child = mutate(child, mutation_rate, perturbation)
            new_population.append(child)
        
        population = new_population
        
        # Update best solution
        current_fitness = [evaluate_h(h, dx) for h in population]
        current_best_idx = np.argmin(current_fitness)
        if current_fitness[current_best_idx] < c5_best:
            h_best = population[current_best_idx].copy()
            c5_best = current_fitness[current_best_idx]
    
    # Gradient refinement phase
    h_refined = h_best.copy()
    eta = 0.001
    max_refine_iter = 5000
    eps = 1e-6
    
    for i in range(max_refine_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_refined.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_refined.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_refined[j] - eta * grad_j
            
        h_refined = repair_h(h_new, n_points)
        c5_refined = evaluate_h(h_refined, dx)
        
        if c5_refined < c5_best:
            h_best = h_refined.copy()
            c5_best = c5_refined
        else:
            eta *= 0.995  # Gradually reduce learning rate
        
    return (h_best.tolist(), c5_best, n_points)
