import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of the maximum overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Identifying the k that gives maximum overlap
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n - 1)
    
    # Computing gradient for this k
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    # Normalize the gradient
    grad = grad / (np.linalg.norm(grad) + 1e-12)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # PSO parameters
    swarm_size = 100
    max_iter = 5000
    w_start = 0.9
    w_end = 0.1
    c1 = 2.0
    c2 = 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    p_best = []
    p_best_scores = []
    
    for _ in range(swarm_size):
        h = repair_h(h_init + np.random.normal(0, 0.01, n_points), n_points)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        p_best.append(h.copy())
        p_best_scores.append(evaluate_h(h, dx))
    
    # Global best
    g_best_idx = np.argmin(p_best_scores)
    g_best = p_best[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - swarm[i]) + 
                            c2 * r2 * (g_best - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Repair and evaluate
            swarm[i] = repair_h(swarm[i], n_points)
            
            # Evaluate
            score = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best[i] = swarm[i].copy()
                p_best_scores[i] = score
                
                # Update global best
                if score < g_best_score:
                    g_best = swarm[i].copy()
                    g_best_score = score
    
    # Local refinement of the best particle
    h_refined = g_best.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
            
        grad = compute_gradient(h_refined, dx, n_points)
        h_refined = h_refined - 1e-5 * grad
        h_refined = repair_h(h_refined, n_points)
        c5_refined = evaluate_h(h_refined, dx)
        
        if c5_refined < g_best_score:
            g_best_score = c5_refined
            g_best = h_refined.copy()
    
    # Final check
    if g_best_score < best_c5:
        best_h = g_best
        best_c5 = g_best_score
    
    return (best_h.tolist(), best_c5, n_points)
