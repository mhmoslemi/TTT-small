import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    swarm_size = 100
    inertia_w = 0.9
    c1 = 2.0
    c2 = 2.0
    max_iter = 10000
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    
    for _ in range(swarm_size):
        h = h_init.copy()
        # Small random perturbation
        h += np.random.normal(0, 0.01, n_points)
        h = repair_h(h, n_points)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest_positions.append(h.copy())
        pbest_scores.append(evaluate_h(h, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # Optimization
    iter_no_improve = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia
        inertia_w = 0.9 - (0.8 * iteration / max_iter)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_w * velocities[i] + 
                             c1 * r1 * (pbest_positions[i] - particles[i]) +
                             c2 * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            current_score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_score < gbest_score:
                    gbest_score = current_score
                    gbest_position = particles[i].copy()
                    iter_no_improve = 0
                else:
                    iter_no_improve += 1
                    
        # Early termination
        if iter_no_improve > 100:
            break
    
    # Refinement step with gradient
    h_best = gbest_position.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Simple finite difference gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        base_score = evaluate_h(h_best, dx)
        
        for i in range(n_points):
            h_test = h_best.copy()
            h_test[i] += eps
            h_test = repair_h(h_test, n_points)
            test_score = evaluate_h(h_test, dx)
            grad[i] = (test_score - base_score) / eps
            
        # Update
        h_best = h_best - 1e-5 * grad
        h_best = repair_h(h_best, n_points)
        new_score = evaluate_h(h_best, dx)
        
        if new_score < gbest_score:
            gbest_score = new_score
            gbest_position = h_best.copy()
    
    return (gbest_position.tolist(), gbest_score, n_points)
