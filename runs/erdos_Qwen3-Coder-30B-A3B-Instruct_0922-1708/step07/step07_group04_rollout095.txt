import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from seed - use available if present
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Start with a triangular approximation
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimizer wrapper for scipy.optimize
    def objective(h_vec):
        h = h_vec.copy()
        h = repair_h(h, n_points)
        return evaluate_h(h, dx)
    
    # Gradient based on finite differences for each point
    def grad_objective(h_vec):
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        base_val = objective(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    # Actual optimization with bounded trust-constr
    opt_result = minimize(
        objective,
        h_current,
        method='trust-constr',
        jac=grad_objective,
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 5000}
    )
    
    # Extract results
    h_final = repair_h(opt_result.x, n_points)
    c5_final = evaluate_h(h_final, dx)
    
    # Refine with local search to improve further
    if c5_final < c5_best:
        h_best = h_final
        c5_best = c5_final
        
    # Iterative local search with small perturbations
    for i in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        idx = np.random.randint(0, n_points)
        eps = 0.01 * (np.random.rand() - 0.5)
        h_new = h_best.copy()
        h_new[idx] += eps
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
