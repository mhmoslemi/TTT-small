import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Local search with simulated annealing
    max_iter = int(1e6)
    temp = 0.1
    decay = 0.9999
    patience = 500
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Random perturbation
        h_new = h_current.copy()
        idx = np.random.randint(0, n_points)
        h_new[idx] += np.random.normal(0, 0.02)
        
        # Project to feasible set
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject with simulated annealing
        if c5_new < c5_current or np.random.rand() < np.exp((c5_current - c5_new) / temp):
            h_current = h_new.copy()
            c5_current = c5_new
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
        temp *= decay
        
        # Occasionally jitter to escape local minima
        if i % 100 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
            
        # Occasionally restore best solution to avoid getting trapped
        if i % 1000 == 0 and i > 0:
            h_current = h_best.copy()
            c5_current = c5_best
            
        # Adaptive termination
        if i - last_improvement > patience:
            break
    
    # Final refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = h_best.copy()
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
