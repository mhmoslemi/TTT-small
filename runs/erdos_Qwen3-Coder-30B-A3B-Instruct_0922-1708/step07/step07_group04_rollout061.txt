import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, mutation_rate=0.1, delta=0.1):
    """Mutate h by perturbing a fraction of entries."""
    h_new = h.copy()
    mask = np.random.rand(len(h_new)) < mutation_rate
    h_new[mask] += np.random.uniform(-delta, delta, np.sum(mask))
    return repair_h(h_new, len(h))

def crossover_h(parent1, parent2, crossover_rate=0.5):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy()
    
    point = np.random.randint(1, len(parent1))
    child = np.concatenate([parent1[:point], parent2[point:]])
    return repair_h(child, len(parent1))

def tournament_selection(population, fitnesses, k=3):
    """Select an individual using tournament selection."""
    idxs = np.random.choice(len(population), size=k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    population_size = 100
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.5
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = mutate_h(h_init, mutation_rate)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    h_best = population[best_idx].copy()
    c5_best = fitnesses[best_idx]
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep top half
        new_population = population[:population_size//2]
        
        # Generate new individuals
        while len(new_population) < population_size:
            p1 = tournament_selection(population, fitnesses)
            p2 = tournament_selection(population, fitnesses)
            child = crossover_h(p1, p2, crossover_rate)
            child = mutate_h(child, mutation_rate)
            new_population.append(child)
        
        # Evaluate new population
        fitnesses = [evaluate_h(h, dx) for h in new_population]
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < c5_best:
            h_best = new_population[best_idx].copy()
            c5_best = fitnesses[best_idx]
        
        population = new_population
    
    # Final gradient refinement on the best solution
    h_current = h_best.copy()
    c5_current = c5_best
    eta = 0.001
    eps = 1e-6
    max_iter = 10000
    patience = 50
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_current.copy()
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            h_new[j] = h_current[j] - eta * grad_j
        
        h_new = repair_h(h_new, n_points)
        
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            if np.random.rand() < 0.01:
                h_current = h_new
                c5_current = c5_new
        
        if i - last_improvement > patience:
            eta *= 0.95
            last_improvement = i
    
    return (h_best.tolist(), c5_best, n_points)
