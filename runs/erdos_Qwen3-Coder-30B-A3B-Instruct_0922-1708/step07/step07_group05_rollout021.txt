import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidates(n_points, n_candidates=50):
    candidates = []
    for _ in range(n_candidates):
        # Generate random Fourier modes
        n_modes = np.random.randint(3, 6)
        modes = []
        for _ in range(n_modes):
            k = np.random.randint(1, 10)
            a = np.random.uniform(-1, 1)
            b = np.random.uniform(-1, 1)
            modes.append((k, a, b))
        
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for k, a, b in modes:
            h += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
        
        # Normalize to [0,1] and sum constraint
        h_min, h_max = np.min(h), np.max(h)
        if h_max > h_min:
            h = (h - h_min) / (h_max - h_min)
        h = h * 0.8 + 0.1  # Scale to [0.1, 0.9]
        h = repair_h(h, n_points)
        candidates.append(h)
    
    return candidates

    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize candidates

    # Use initial_h_values if available

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        candidates.append(h_init)
    
    # Generate diverse Fourier candidates
    candidates.extend(generate_fourier_candidates(n_points, 50))
    
    # Evaluate all candidates
    scores = []
    for h in candidates:
        c5 = evaluate_h(h, dx)
        scores.append((c5, h))
    
    # Sort candidates by score
    scores.sort(key=lambda x: x[0])
    top_candidates = scores[:10]  # Take top 10
    
    h_best = top_candidates[0][1]
    c5_best = top_candidates[0][0]
    
    # Refine top candidates using gradient descent
    for h_init, _ in top_candidates:
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        # Gradient descent parameters
        learning_rate = 0.01
        max_iter = 2000
        patience = 200
        last_improvement = 0
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros(n_points)
            epsilon = 1e-6
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[j] += epsilon
                h_minus[j] -= epsilon
                h_plus = repair_h(h_plus, n_points)
                h_minus = repair_h(h_minus, n_points)
                
                c5_plus = evaluate_h(h_plus, dx)
                c5_minus = evaluate_h(h_minus, dx)
                grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update
            h_new = h_current - learning_rate * grad
            h_new = repair_h(h_new, n_points)
            
            # Evaluate
            c5_new = evaluate_h(h_new, dx)
            
            # Accept improvement
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
            else:
                # Accept with probability
                if np.random.rand() < np.exp(-(c5_new - c5_current) / (c5_current * 0.01)):
                    h_current = h_new
                    c5_current = c5_new
            
            # Adapt learning rate
            if i - last_improvement > patience:
                learning_rate *= 0.95
                if learning_rate < 1e-6:
                    break
    
    return (h_best.tolist(), c5_best, n_points)
