import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2, n_points):
    # Segment-based crossover
    crossover_point = np.random.randint(1, n_points)
    child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    return child

def mutate(individual, n_points, mutation_scale):
    # Mutation with segment flips and Gaussian noise
    child = individual.copy()
    # Segment flips
    if np.random.random() < 0.3:
        flip_start = np.random.randint(0, n_points)
        flip_end = np.random.randint(flip_start + 1, n_points + 1)
        child[flip_start:flip_end] = 1 - child[flip_start:flip_end]
    
    # Gaussian noise
    noise = np.random.normal(0, mutation_scale, n_points)
    child += noise
    child = project_h(child, n_points, 2.0/n_points)
    return child

def hill_climb(h, n_points, dx, max_iter=50):
    # Fast local coordinate descent
    h_current = h.copy()
    c5_current = evaluate_h(h_current, dx)
    
    eta = 0.01
    eps = 1e-6
    
    for _ in range(max_iter):
        h_new = h_current.copy()
        improved = False
        
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
            
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            improved = True
        else:
            eta *= 0.995
        
        if not improved:
            break
            
    return h_current, c5_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 0.5
        h_init = project_h(h_init, n_points, dx)
    
    P = 50  # population size
    M = max(1, P // 5)  # number of parents
    population = []
    population_fitness = []
    
    # Initialize population
    for _ in range(P):
        if np.random.random() < 0.5:
            h = h_init.copy()
            h = mutate(h, n_points, 0.1)
        else:
            h = np.random.random(n_points)
            h = project_h(h, n_points, dx)
        population.append(h)
        population_fitness.append(evaluate_h(h, dx))
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Main memetic algorithm loop
    gen = 0
    mutation_scale = 0.1
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 5:
        gen += 1
        # Evaluate population
        for i in range(P):
            population_fitness[i] = evaluate_h(population[i], dx)
        
        # Sort by fitness
        sorted_indices = np.argsort(population_fitness)
        population = [population[i] for i in sorted_indices]
        population_fitness = [population_fitness[i] for i in sorted_indices]
        
        # Update best solution
        if population_fitness[0] < c5_best:
            h_best = population[0].copy()
            c5_best = population_fitness[0]
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 50:
                mutation_scale *= 0.95
                no_improvement = 0
                
        # Selection
        parents = population[:M]
        
        # Generate offspring
        new_population = parents.copy()
        
        while len(new_population) < P:
            # Crossover
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child = crossover(parent1, parent2, n_points)
            # Mutation
            child = mutate(child, n_points, mutation_scale)
            # Local refinement
            child, fitness = hill_climb(child, n_points, dx, max_iter=50)
            new_population.append(child)
            if len(new_population) >= P:
                break
        
        # Replace worst individuals
        population = new_population[:P]
        
        # Occasionally inject random individual to prevent stagnation
        if gen % 10 == 0:
            for i in range(P // 10):
                if np.random.random() < 0.5:
                    idx = np.random.randint(P // 2, P)
                    h = np.random.random(n_points)
                    h = project_h(h, n_points, dx)
                    h, fitness = hill_climb(h, n_points, dx, max_iter=10)
                    population[idx] = h
                    population_fitness[idx] = fitness
    
    return (h_best.tolist(), c5_best, n_points)
