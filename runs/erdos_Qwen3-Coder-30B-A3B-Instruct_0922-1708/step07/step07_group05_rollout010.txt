import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a structured construction
        h_init = np.zeros(n_points)
        # Create a piecewise linear function that is 0 on [0,0.5) and 1 on (1.5,2]
        # and slopes linearly in between
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        # Repair to sum to 1
        h_init = repair_h(h_init, n_points)

    # PSO parameters
    n_particles = 100
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    max_iter = 400
    patience = 100
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    for i in range(n_particles):
        if i == 0:
            h = h_init.copy()
        else:
            h = h_init + np.random.normal(0, 0.05, n_points)
        h = repair_h(h, n_points)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        score = evaluate_h(h, dx)
        pbest.append(h.copy())
        pbest_scores.append(score)
    
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    last_improvement = 0
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    last_improvement = iter_num
        
        # Mutation
        if iter_num % 25 == 0 and iter_num > 0:
            n_mutate = int(n_particles * mutation_rate)
            mutate_indices = np.random.choice(n_particles, n_mutate, replace=False)
            for idx in mutate_indices:
                # Mutate 20% of values
                n_mutate_vals = max(1, n_points // 5)
                mutate_positions = np.random.choice(n_points, n_mutate_vals, replace=False)
                for pos in mutate_positions:
                    particles[idx][pos] += np.random.normal(0, mutation_sigma)
                particles[idx] = repair_h(particles[idx], n_points)
    
    return (gbest.tolist(), gbest_score, n_points)
