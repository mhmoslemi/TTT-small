import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter_barrier = 1000
    max_iter_gd = 10000
    learning_rate = 1e-3
    mu_initial = 1e3
    barrier_decrease = 0.99
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    # Scale to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Barrier method phase
    mu = mu_initial
    for iter_barrier in range(max_iter_barrier):
        if (time.time() - start_time) > budget_s - 10:
            break
            
        # Compute maximum correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(corr) - 1) // 2
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using adjoint method
        grad = np.zeros_like(h)
        # For the maximum lag, we can compute the gradient directly
        for i in range(len(h)):
            if max_lag >= 0:
                if i + max_lag < len(h):
                    grad[i] += (1 - h[i + max_lag])
            else:
                if i - max_lag < len(h):
                    grad[i] += (1 - h[i - max_lag])
        
        # Update
        h = h - learning_rate * grad
        
        # Project and rescale
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / h_sum
            
        # Decrease barrier parameter
        mu *= barrier_decrease
    
    # Gradient descent phase
    for iter_gd in range(max_iter_gd):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute maximum correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(corr) - 1) // 2
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient with multiple lags (in case of multiple maxima)
        grad = np.zeros_like(h)
        # Collect indices of maximum correlation
        max_corr = np.max(corr)
        max_indices = np.where(np.abs(corr - max_corr) < 1e-10)[0]
        # Compute gradient from each maximum lag
        for idx in max_indices:
            lag = idx - (len(corr) - 1) // 2
            for i in range(len(h)):
                if lag >= 0:
                    if i + lag < len(h):
                        grad[i] += (1 - h[i + lag])
                else:
                    if i - lag < len(h):
                        grad[i] += (1 - h[i - lag])
        
        # Normalize gradient
        grad = grad / len(max_indices)
        
        # Update
        h = h - learning_rate * grad
        
        # Project and rescale
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / h_sum
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
