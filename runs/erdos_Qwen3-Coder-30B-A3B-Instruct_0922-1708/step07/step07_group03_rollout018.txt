import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize swarm
    S = 50  # number of particles
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    T = 1.0  # initial temperature
    cooling_rate = 0.999
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        # Scale to correct sum
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
    else:
        # Random initialization
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.rand(n_points) * 0.1)
    
    # Fill remaining particles
    for i in range(S - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.rand(n_points) * 0.1)
    
    # Personal best and global best
    p_best = [p.copy() for p in particles]
    p_best_scores = [float('inf')] * S
    g_best = particles[0].copy()
    g_best_score = float('inf')
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate particles
        for i in range(S):
            h = particles[i]
            corr = correlate(h, 1 - h, mode='full')
            score = np.max(corr) * dx
            
            if score < p_best_scores[i]:
                p_best_scores[i] = score
                p_best[i] = h.copy()
                
            if score < g_best_score:
                g_best_score = score
                g_best = h.copy()
        
        # Update velocities and positions
        for i in range(S):
            r1, r2 = np.random.rand(2)
            v = w * velocities[i] + \
                c1 * r1 * (p_best[i] - particles[i]) + \
                c2 * r2 * (g_best - particles[i])
            
            # Apply simulated annealing
            if np.random.rand() < T:
                v += np.random.uniform(-0.1, 0.1, n_points)
            
            particles[i] += v
            velocities[i] = v
            
            # Repair constraints
            particles[i] = np.clip(particles[i], 0, 1)
            h_sum = np.sum(particles[i])
            if abs(h_sum - n_points / 2) > 1e-8:
                particles[i] = particles[i] * (n_points / 2) / h_sum
        
        # Cooling
        T = max(T * cooling_rate, 0.01)
    
    # Final evaluation
    corr = correlate(g_best, 1 - g_best, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (g_best.tolist(), final_c5, n_points)
