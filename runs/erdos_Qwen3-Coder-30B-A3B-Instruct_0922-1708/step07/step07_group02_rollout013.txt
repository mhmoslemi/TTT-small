import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def blend_segments(h1, h2, blend_ratio=0.5):
    """Create a new h by blending two parent segments."""
    n = len(h1)
    # Create a random cut point
    cut = np.random.randint(1, n)
    # Blend segments
    h_new = h1.copy()
    h_new[cut:] = h2[cut:]
    return h_new

def mutate_h(h, mutation_rate=0.3, noise_scale=0.1):
    """Apply mutation to h."""
    h_new = h.copy()
    n = len(h)
    # Segment flip mutation
    if np.random.random() < mutation_rate:
        cut = np.random.randint(1, n)
        h_new[cut:] = 1 - h_new[cut:]
    # Gaussian noise mutation
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, noise_scale, n)
        h_new += noise
        h_new = np.clip(h_new, 0, 1)
    return h_new

def local_refinement(h, n_points, dx, max_iter=50):
    """Perform fast local refinement on h."""
    h_current = h.copy()
    c5_current = evaluate_h(h_current, dx)
    eta = 0.01
    eps = 1e-6

    for i in range(max_iter):
        h_new = h_current.copy()
        indices = list(range(n_points))
        np.random.shuffle(indices)
        improved = False
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            improved = True
        if not improved:
            eta *= 0.99
        else:
            eta = min(0.01, eta * 1.001)
    return h_current, c5_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)

    # Memetic algorithm
    population_size = 50
    selection_ratio = 0.2
    mutation_scale = 0.1
    max_generations = 1500

    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h_temp = mutate_h(h_init, mutation_rate=0.3, noise_scale=0.1)
        h_temp = project_h(h_temp, n_points, dx)
        population.append(h_temp)

    # Evaluate initial population
    fitness = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    h_best = population[best_idx].copy()
    c5_best = fitness[best_idx]

    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break

        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:int(selection_ratio * population_size)]

        # Create offspring
        offspring = []
        for _ in range(population_size):
            # Select two parents
            p1_idx, p2_idx = np.random.choice(top_indices, 2, replace=False)
            p1, p2 = population[p1_idx], population[p2_idx]
            # Crossover
            h_crossover = blend_segments(p1, p2)
            # Mutate
            h_mutated = mutate_h(h_crossover, noise_scale=mutation_scale)
            h_mutated = project_h(h_mutated, n_points, dx)
            offspring.append(h_mutated)

        # Local refinement
        refined_offspring = []
        for h in offspring:
            h_refined, c5_refined = local_refinement(h, n_points, dx, max_iter=50)
            refined_offspring.append((h_refined, c5_refined))

        # Replace worst individuals
        worst_indices = sorted_indices[-int(selection_ratio * population_size):]
        for i, (h_refined, c5_refined) in enumerate(refined_offspring):
            if c5_refined < fitness[worst_indices[i]]:
                population[worst_indices[i]] = h_refined
                fitness[worst_indices[i]] = c5_refined
                if c5_refined < c5_best:
                    h_best = h_refined.copy()
                    c5_best = c5_refined

        # Update mutation scale
        if gen % 10 == 0:
            mutation_scale *= 0.99

    return (h_best.tolist(), c5_best, n_points)
