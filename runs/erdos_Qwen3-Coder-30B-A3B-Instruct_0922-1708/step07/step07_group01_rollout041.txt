import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Phase 1: CCCP approach
    max_iter_cccp = 5000
    lr_cccp = 0.01
    patience_cccp = 500
    last_improvement_cccp = 0
    
    for i in range(max_iter_cccp):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Identify k_max where maximum overlap occurs
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        k_max = max_idx - (len(h_current) - 1)
        
        # Compute gradient for the max overlap
        # Gradient is (1 - h_shifted_k_max)[i] for each h[i]
        h_shifted = np.roll(h_current, k_max)
        grad = (1 - h_shifted)
        
        # Update h using gradient descent
        h_new = h_current - lr_cccp * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept move if improvement
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse moves
            if np.random.rand() < 0.01:
                h_current = h_new.copy()
                c5_current = c5_new
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement_cccp = i
        
        # Adaptive learning rate
        if i - last_improvement_cccp > patience_cccp:
            lr_cccp *= 0.95
            last_improvement_cccp = i
            
        # Occasionally perturb to escape local minima
        if i % 500 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Phase 2: Coordinate descent with exact gradient computation
    max_iter_cd = 10000
    lr_cd = 0.01
    patience_cd = 1000
    last_improvement_cd = 0
    perturb_strength = 0.005
    perturb_interval = 1000
    
    for i in range(max_iter_cd):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic coordinate descent
        for j in range(n_points):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Identify k_max for current h configuration
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            k_max = max_idx - (len(h_current) - 1)
            
            # Compute gradient for the max overlap shift
            h_shifted = np.roll(h_current, k_max)
            grad_component = (1 - h_shifted)[j]
            
            # Gradient step
            h_new = h_current.copy()
            h_new[j] = h_current[j] - lr_cd * grad_component
            
            # Clip and renormalize
            h_new[j] = np.clip(h_new[j], 0, 1)
            h_new = project_h(h_new, n_points)
            
            # Evaluate
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement_cd = i
            # Reset learning rate if improvement
            lr_cd = 0.01
            
        # Adaptive learning rate
        if i - last_improvement_cd > patience_cd:
            lr_cd *= 0.95
            last_improvement_cd = i
            
        # Occasionally perturb
        if i % perturb_interval == 0 and i > 0:
            noise = np.random.normal(0, perturb_strength, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
            
            # Increase perturbation strength if no improvement
            if c5_current > c5_best:
                perturb_strength *= 1.1
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
