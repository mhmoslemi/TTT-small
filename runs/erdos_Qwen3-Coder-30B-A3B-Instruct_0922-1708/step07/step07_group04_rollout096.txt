import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    n_points = 100

    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Start with initial solution
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Use optimization with bounds and constraints
    def objective(h_vec):
        h = h_vec.copy()
        h = repair_h(h, n_points)
        return evaluate_h(h, dx)
    
    def constraint(h_vec):
        h = h_vec.copy()
        return np.sum(h) - n_points / 2
    
    # Solve using scipy minimize
    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(n_points)]
    
    options = {'maxiter': 50000, 'disp': False}

    for attempt in range(3):
        if time.time() - start_time > budget_s - 5:

        # Start from best solution found

    max_iter = 10000
    patience = 50
    last_improvement = 0

        for j in range(n_points):
            # Forward difference approximation
            h_plus = h_current.copy()
            h_plus[j] += 1e-6
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)

    h_best = repair_h(h_best, n_points)
    c5_best = evaluate_h(h_best, dx)
