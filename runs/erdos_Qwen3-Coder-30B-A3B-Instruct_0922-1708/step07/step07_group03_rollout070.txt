import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Adjusted optimization approach with CVXPY
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Setup for optimization
    h_var = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Create constraints for all lags
    constraints = []
    
    # Add integral constraints
    for k in range(n_points*2 - 1):
        # Since h is a step function, we compute the correlation manually
        # but need to avoid the full correlation directly in constraints.
        # Instead we'll do this iteratively in a heuristic approach
        pass
    
    # Add general constraints
    constraints.append(cp.sum(h_var) == n_points / 2)
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Add the maximum overlap constraint
    # This is the slowest part - we have to be careful about solving
    # For large scale, we use a hybrid approach
    # First compute an initial value, then use gradient based refinement
    
    # Evaluate initial solution
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
        
    # Do iterative refinement with a simple gradient descent
    # But try to limit overfitting and avoid stuck points
    
    # Gradient descent with adaptive learning rate
    max_iter = 50000
    learning_rate = 1e-3
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap integral
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Estimate gradient by finite differences
        grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            # Rescale to sum constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update step
        h = h - learning_rate * grad
        
        # Project h to valid range
        h = np.clip(h, 0, 1)
        
        # Maintain integral constraint
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
