import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Use scipy's minimize for a more robust optimization
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Parameters
    max_iter = 5000
    patience = 500
    last_improvement = 0
    eta = 0.01
    
    def objective(h):
        # Add regularization to be more aggressive in constraint satisfaction
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2
        penalty = 1000 * (current_sum - target_sum)**2
        h = h * target_sum / current_sum if abs(current_sum - target_sum) > 1e-8 else h
        
        corr = correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx + penalty
    
    def callback(h):
        nonlocal h_current, c5_current, h_best, c5_best, last_improvement
        if time.time() - start_time > budget_s - 5:
            return True
        h_repaired = repair_h(h, n_points)

            h_best = h_repaired.copy()
            c5_best = c5_new
            last_improvement = len(h_current)
        return False

    try:
        # Optimize using L-BFGS-B, which handles bounds well
        result = minimize(objective, h_current, method='L-BFGS-B',
                          bounds=[(0, 1) for _ in range(n_points)],
                          options={'maxiter': max_iter, 'ftol': 1e-12},
                          callback=callback)
        if result.success:
            h_final = repair_h(result.x, n_points)
            c5_final = evaluate_h(h_final, dx)
            if c5_final < c5_best:
                h_best = h_final
                c5_best = c5_final
    except Exception:
        pass

    # Final gradient-based refinement if we're still in time
    if time.time() - start_time < budget_s - 10:
        # Coordinate-wise optimization
        h_current = h_best.copy()
        for i in range(10000):

                    c5_best = c5_current

                eta *= 0.995

    return (h_best.tolist(), c5_best, n_points)
