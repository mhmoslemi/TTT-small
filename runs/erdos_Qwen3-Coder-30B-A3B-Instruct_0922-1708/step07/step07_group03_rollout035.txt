import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create binary initialization with correct sum
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)

    # Local search with perturbations
    best_c5 = float('inf')
    best_h = h.copy()
    temperature = 0.1
    cooling_rate = 0.995
    delta = 0.05
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate candidate by perturbing two points
        h_candidate = h.copy()
        i, j = np.random.randint(0, n_points, 2)
        
        # Adjust values
        change = np.random.uniform(-delta, delta)
        h_candidate[i] += change
        h_candidate[j] -= change
        
        # Ensure within bounds [0,1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Rescale to meet sum constraint
        h_sum = np.sum(h_candidate)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h_candidate = h_candidate * target_sum / h_sum
            
        # Compute C5 for candidate
        corr = correlate(h_candidate, 1 - h_candidate, mode='full')
        c5 = np.max(corr) * dx
        
        # Accept or reject based on temperature
        if c5 < best_c5:
            h = h_candidate
            best_c5 = c5
            best_h = h.copy()
        else:
            # Accept worse solution with some probability
            if np.random.rand() < np.exp(-(c5 - best_c5) / temperature):
                h = h_candidate
                
        # Cool down temperature
        temperature *= cooling_rate
        
        # Periodic check for early termination
        if iteration % 1000 == 0 and best_c5 < 0.38085:
            break
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
