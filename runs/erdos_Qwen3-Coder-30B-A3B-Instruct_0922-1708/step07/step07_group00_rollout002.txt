import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate diverse initial points using Sobol sequences
    sampler = qmc.Sobol(d=n_points, scramble=True, seed=seed)
    n_initials = 500
    initial_points = sampler.random(n=n_initials)
    initial_points = initial_points * 2 - 1  # Transform to [-1,1]
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Optimization loop
    for i in range(n_initials):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create initial h from Sobol point
        h_candidate = initial_points[i] * 0.5 + 0.5  # Map to [0,1]
        h_candidate = repair_h(h_candidate, n_points)
        
        # Nelder-Mead optimization
        try:
            result = minimize(
                lambda x: evaluate_h(repair_h(x, n_points), dx),
                h_candidate,
                method='Nelder-Mead',
                options={'maxiter': 1000, 'xatol': 1e-6, 'fatol': 1e-6}
            )
            if result.success:
                h_candidate = result.x
        except:
            pass
            
        # Refinement with gradient descent
        h_refined = repair_h(h_candidate, n_points)
        current_c5 = evaluate_h(h_refined, dx)
        
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Gradient computation (finite differences)
            eps = 1e-6
            grad = np.zeros(n_points)
            corr_base = correlate(h_refined, 1 - h_refined, mode='full')
            max_k_idx = np.argmax(corr_base)
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                grad[j] = (corr_plus[max_k_idx] - corr_base[max_k_idx]) / eps
            
            # Update
            h_refined = h_refined - 1e-5 * grad
            h_refined = repair_h(h_refined, n_points)
            current_c5 = evaluate_h(h_refined, dx)
            
            if current_c5 < best_c5:
                best_h = h_refined.copy()
                best_c5 = current_c5
                
    return (best_h.tolist(), best_c5, n_points)
