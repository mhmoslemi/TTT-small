import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, dx):
    """Compute gradient for the maximum overlap term."""
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    # The full correlation has length 2*n_points - 1
    # The shift k is max_idx - (n_points - 1)
    k = max_idx - (len(h) - 1)
    n = len(h)
    
    # Derivative is (1 - h_{i+k}) for shifted positions only  
    grad = np.zeros(n)
    # How the overlap changes with small change in h[i]
    # If h[i] increases, we lose overlap with positions that are k away
    # -h[i+k] contribution if i+k < n, and +h[i-k] if i-k >= 0
    if k >= 0:
        # h[i] contributes to the overlap with h[i-k] (shifted backward)
        if k < n:
            for i in range(k, n):
                grad[i] = -1.0
    else:
        # Negative k means shift forward
        k_abs = -k
        if k_abs < n:
            for i in range(0, n - k_abs):
                grad[i] = -1.0
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Try a piecewise constant function spaced to create complex overlaps
        # Create two triangular peaks to encourage overlap minimization
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient descent parameters
    learning_rate = 0.05
    max_iter = int(1e6)
    patience = 1000  # More patience for gradient descent
    last_improvement = 0
    noise_intensity = 0.01
    
    # Try multiple random perturbations
    num_trials = 5
    best_trial_c5 = float('inf')
    best_trial_h = None
    
    for trial in range(num_trials):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Start fresh for this trial
        h_current = h_init.copy()
        h_current += np.random.normal(0, 0.001, n_points)
        h_current = project_h(h_current, n_points)
        c5_current = evaluate_h(h_current, dx)
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute exact gradient
            grad = compute_gradient(h_current, dx)

            # Update with gradient
            h_new = h_current - learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept move if improvement
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
            else:
                # Occasionally accept worse moves (simulated annealing)
                if np.random.rand() < 0.005:
                    h_current = h_new.copy()
                    c5_current = c5_new
                    
            # Update best
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
            # Adaptive learning rate on plateau
            if i - last_improvement > patience:
                learning_rate *= 0.95
                last_improvement = i
                
            # Periodic noise to escape local minima
            if i % 500 == 0 and i > 0:
                noise = np.random.normal(0, noise_intensity, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
                noise_intensity *= 0.99
                
        # Store best trial
        if c5_current < best_trial_c5:
            best_trial_c5 = c5_current
            best_trial_h = h_current.copy()
            
    # Return best over all trials
    if best_trial_h is not None:
        h_best = best_trial_h
        c5_best = best_trial_c5
    
    return (h_best.tolist(), c5_best, n_points)
