import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a trapezoidal wave that peaks at center
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Dynamic Programming approach with local search
    max_iter = int(1e6)
    dp_window = 10  # Look ahead window size
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Consider small modifications to the current solution
        h_new = h_current.copy()
        
        # Randomly pick a small set of indices to perturb
        indices = np.random.choice(n_points, max(1, n_points // 20), replace=False)
        for idx in indices:
            # Perturb gently
            h_new[idx] += np.random.normal(0, 0.01)
            h_new[idx] = max(0, min(1, h_new[idx]))
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
        else:
            # Occasionally accept worse moves
            if np.random.rand() < 0.05:
                h_current = h_new
                c5_current = c5_new
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
        
        # Occasionally jitter to escape local minima
        if i % 200 == 0 and i > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
            
        # Occasionally restore best solution
        if i % 1000 == 0:
            h_current = h_best.copy()
            c5_current = c5_best
    
    # Final refinement through local search
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
            h_new[idx] = max(0, min(1, h_new[idx]))
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
