import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # PSO parameters
    n_particles = 50
    w_init = 0.9
    w_min = 0.4
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Insert initial solution as first particle
    particles.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    pbest.append(h_init.copy())
    pbest_vals.append(evaluate_h(h_init, dx))
    
    # Generate remaining particles
    for _ in range(n_particles - 1):
        h = np.random.random(n_points)
        h = project_h(h, n_points, dx)
        particles.append(h)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_h(h, dx))
    
    # Initialize swarm best
    swarm_best_idx = np.argmin(pbest_vals)
    swarm_best = pbest[swarm_best_idx].copy()
    swarm_best_val = pbest_vals[swarm_best_idx]
    
    # PSO iteration
    max_iter = 100000
    stagnation_count = 0
    stagnation_limit = 500
    inertia_weight = w_init
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        inertia_weight = w_min + (w_init - w_min) * (1 - i / max_iter)
        for p in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[p] = (inertia_weight * velocities[p] 
                           + c1 * r1 * (pbest[p] - particles[p]) 
                           + c2 * r2 * (swarm_best - particles[p]))
            
            # Limit velocity
            velocities[p] = np.clip(velocities[p], -v_max, v_max)
            
            # Update position
            particles[p] += velocities[p]
            
            # Project to feasible region
            particles[p] = project_h(particles[p], n_points, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(particles[p], dx)
            
            # Update personal best
            if fitness < pbest_vals[p]:
                pbest[p] = particles[p].copy()
                pbest_vals[p] = fitness
                
                # Update swarm best
                if fitness < swarm_best_val:
                    swarm_best = particles[p].copy()
                    swarm_best_val = fitness
                    stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Reset one particle if no progress
        if stagnation_count > stagnation_limit:
            particle_idx = np.random.randint(0, n_particles)
            particles[particle_idx] = np.random.random(n_points)
            particles[particle_idx] = project_h(particles[particle_idx], n_points, dx)
            pbest[particle_idx] = particles[particle_idx].copy()
            pbest_vals[particle_idx] = evaluate_h(particles[particle_idx], dx)
            stagnation_count = 0
            
    # Final check
    best_particle = swarm_best
    best_val = swarm_best_val
    
    return (best_particle.tolist(), best_val, n_points)
