import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create initial binary-like solution with correct sum
        h = np.zeros(n_points)
        ones_count = n_points // 2
        h[:ones_count] = 1.0
        # Randomize the placement to avoid symmetry
        np.random.shuffle(h)
        # Rescale to meet constraint
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing approach
    current_h = h.copy()
    current_c5 = np.max(correlate(current_h, 1 - current_h, mode='full')) * dx
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    temp = 0.1
    cooling_rate = 0.995
    min_temp = 1e-6
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by swapping two values
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.05, 0.05)
        
        # Apply perturbation
        h_new = current_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            # Adjust all values proportionally
            h_new = h_new * target_sum / h_sum
        
        # Evaluate new solution
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject based on simulated annealing
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
            current_h = h_new
            current_c5 = new_c5
            
            # Update best solution if better
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Cool down the temperature
        temp = max(temp * cooling_rate, min_temp)
    
    # Final evaluation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
