import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient for the specific k_max that gives max overlap."""
    n = len(h)
    # Compute shifted version of h for the k_max shift
    if k_max >= 0:
        h_shifted = np.concatenate([h[k_max:], np.zeros(k_max)])
    else:
        h_shifted = np.concatenate([np.zeros(-k_max), h[:k_max]])
    
    # Gradient is (1 - h_shifted)
    grad = 1 - h_shifted[:n]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Cosine taper
        h_init = 0.5 * (1 + np.cos(np.pi * x))
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Parameters
    learning_rate = 0.05
    max_iter = int(1e6)
    patience = 50
    last_improvement = 0
    noise_scale = 0.01
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        c5_max = corr[k_max] * dx
        
        # Compute gradient for the specific k_max
        grad = compute_gradient(h_current, dx, k_max - (n_points - 1))
        
        # Update with gradient
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept move
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse moves
            if np.random.rand() < 0.05:
                h_current = h_new.copy()
                c5_current = c5_new
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                learning_rate = 1e-6
            last_improvement = i
            
        # Periodic perturbations to escape local minima
        if i % 200 == 0 and i > 0:
            noise = np.random.normal(0, noise_scale, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
            
        # Occasionally restore best solution
        if i % 1000 == 0:
            h_current = h_best.copy()
            c5_current = c5_best
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Single perturbation around best solution
        h_new = h_best.copy()
        idx = np.random.randint(n_points)
        h_new[idx] += np.random.normal(0, 0.005)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
