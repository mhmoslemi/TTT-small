import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a structured construction
        h_init = np.zeros(n_points)
        # Create a piecewise linear function that is 0 on [0,0.5) and 1 on (1.5,2]
        # and slopes linearly in between
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)

    # PSO parameters
    n_particles = 100
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    max_iter = 400
    patience = 100
    last_improvement = 0

    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for i in range(n_particles):
        if i == 0:
            h = h_init.copy()
        else:
            h = h_init + np.random.normal(0, 0.01, n_points)
        h = repair_h(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest_positions.append(h.copy())
        pbest_values.append(evaluate_h(h, dx))

    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]

    # PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break

        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_num / max_iter

        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - particles[i]) + 
                            c2 * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            current_value = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest_positions[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest_position = particles[i].copy()
                    gbest_value = current_value
                    last_improvement = iter_num

        # Mutation step
        if iter_num % 25 == 0:
            mutated_count = int(n_particles * mutation_rate)
            indices = np.random.choice(n_particles, mutated_count, replace=False)
            for idx in indices:
                # Mutate 20% of the values in the particle
                mutate_indices = np.random.choice(n_points, int(0.2 * n_points), replace=False)
                for j in mutate_indices:
                    particles[idx][j] += np.random.normal(0, mutation_sigma)
                particles[idx] = repair_h(particles[idx], n_points)
                new_value = evaluate_h(particles[idx], dx)
                if new_value < pbest_values[idx]:
                    pbest_positions[idx] = particles[idx].copy()
                    pbest_values[idx] = new_value
                    if new_value < gbest_value:
                        gbest_position = particles[idx].copy()
                        gbest_value = new_value
                        last_improvement = iter_num

        # Early stopping check
        if iter_num - last_improvement > patience:
            break

    return (gbest_position.tolist(), gbest_value, n_points)
