import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, mutation_rate=0.1, perturbation=0.1):
    """Mutate h by perturbing some entries."""
    h_new = h.copy()
    n_points = len(h)
    indices = np.random.choice(n_points, size=int(n_points * mutation_rate), replace=False)
    for i in indices:
        h_new[i] += np.random.uniform(-perturbation, perturbation)
    return repair_h(h_new, n_points)

def crossover_h(h1, h2, segment_length=10):
    """Crossover two h vectors by swapping segments."""
    h_new = h1.copy()
    n_points = len(h1)
    start = np.random.randint(0, n_points - segment_length + 1)
    end = start + segment_length
    h_new[start:end] = h2[start:end]
    return repair_h(h_new, n_points)

    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 100
    elite_size = 10
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h_mut = mutate_h(h_init, mutation_rate=0.2, perturbation=0.2)
        population.append(h_mut)
    
    # Evaluation
    fitnesses = []
    for h in population:
        fitnesses.append(-evaluate_h(h, dx))
    
    # Evolution loop
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Main evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if -sorted_fitnesses[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = -sorted_fitnesses[0]
        
        # Select elite
        elite = sorted_population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = sorted_population[np.random.randint(0, population_size//2)]
            parent2 = sorted_population[np.random.randint(0, population_size//2)]
            child = crossover_h(parent1, parent2)
            child = mutate_h(child, mutation_rate=0.1, perturbation=0.1)
            new_population.append(child)
        
        population = new_population
        
        # Evaluate new population
        fitnesses = []
        for h in population:
            fitnesses.append(-evaluate_h(h, dx))
    
    # Gradient refinement on best solution
    h_current = best_h.copy()
    c5_current = best_c5
    eta = 0.001
    eps = 1e-6
    max_iter = 10000
    patience = 50
    last_improvement = 0

            h_minus[j] -= eps

        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            last_improvement = i
        else:
            if i - last_improvement > patience:
                eta *= 0.95
                last_improvement = i
    
    # Final validation
    final_h = repair_h(h_current, n_points)
    final_c5 = evaluate_h(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
