import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # CCCP Phase with gradient ascent in convex approximations
    max_iter_cccp = 5000
    learning_rate_cccp = 0.01
    patience_cccp = 100
    last_improvement_cccp = 0
    worsened_count = 0
    
    for i in range(max_iter_cccp):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Identify k_max by correlation
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        
        # Gradient of the correlation at k_max for updates
        # Find the derivative of h[i] influence on overlap at k_max
        grad = np.zeros(n_points)
        # For each i, compute how much change in h[i] affects the correlation at k_max
        # For finite differences
        epsilon = 1e-6
        h_plus = h_current.copy()
        h_minus = h_current.copy()
        
        for j in range(n_points):
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            h_plus = project_h(h_plus, n_points)
            h_minus = project_h(h_minus, n_points)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[j] = (corr_plus[k_max] - corr_minus[k_max]) / (2 * epsilon)
            
            # Reset for next iteration
            h_plus[j] = h_current[j]
            h_minus[j] = h_current[j]
        
        # Update with gradient (descent to minimize correlation)
        h_new = h_current - learning_rate_cccp * grad
        h_new = project_h(h_new, n_points)
        
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
            worsened_count = 0
        else:
            worsened_count += 1
        
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement_cccp = i
        
        # Adaptive learning rate
        if worsened_count > patience_cccp:
            learning_rate_cccp *= 0.95
            worsened_count = 0
            
        # Perturb occasionally to escape local minima 
        if i % 500 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Coordinate Descent Phase 
    max_iter_cd = 10000
    learning_rate_cd = 0.02
    patience_cd = 100
    last_improvement_cd = 0
    worsened_count_cd = 0
    
    h_current = h_best.copy()
    c5_current = c5_best
    
    for i in range(max_iter_cd):
        if time.time() - start_time > budget_s - 5:
            break
            
        for j in range(n_points):
            # Compute gradient component for h[j]
            # Find the maximum overlap shift first
            corr = correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr)
            # Now compute derivative for the specific h[j]
            # We can compute this more efficiently using finite differences
            epsilon = 1e-6
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            h_plus = project_h(h_plus, n_points)
            h_minus = project_h(h_minus, n_points)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            grad_component = (corr_plus[k_max] - corr_minus[k_max]) / (2 * epsilon)
            
            # Update h[j] 
            h_current[j] = h_current[j] - learning_rate_cd * grad_component
            h_current[j] = max(0, min(1, h_current[j]))
        
        # Project to preserve sum = n_points / 2
        h_current = project_h(h_current, n_points)
        
        c5_new = evaluate_h(h_current, dx)
        
        # Track improvement  
        if c5_new < c5_current:
            c5_current = c5_new
            worsened_count_cd = 0
        else:
            worsened_count_cd += 1
            
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement_cd = i
            
        # Adaptive learning rate
        if worsened_count_cd > patience_cd:
            learning_rate_cd *= 0.95
            worsened_count_cd = 0
            
        # Perturb occasionally to escape local minima
        if i % 1000 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)

    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
