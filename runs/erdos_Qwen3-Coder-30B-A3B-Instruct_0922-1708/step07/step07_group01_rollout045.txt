import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Binary search on C
    C_low = 0.0
    C_high = 0.5
    tol = 1e-5
    max_iter = 50

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        C = (C_low + C_high) / 2.0
        
        # Formulate LP for feasibility check
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints for all k
        for k in range(n_points):
            if k == 0:
                continue
            # Sum h_i * (1 - h_{i+k}) over valid range
            expr = 0
            for i in range(n_points - k):
                expr += h_vars[i] * (1 - h_vars[i + k])
            constraints.append(expr <= C * dx)
        
        # Create and solve LP
        objective = cp.Minimize(0)  # dummy objective
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_feasible = h_vars.value
                if h_feasible is not None:
                    h_feasible = project_h(h_feasible, n_points)
                    c5_feasible = evaluate_h(h_feasible, dx)
                    if c5_feasible < c5_best:
                        h_best = h_feasible.copy()
                        c5_best = c5_feasible
                    C_high = C
                else:
                    C_low = C
            else:
                C_low = C
                
        except:
            C_low = C
            
        if C_high - C_low < tol:
            break
    
    return (h_best.tolist(), c5_best, n_points)
