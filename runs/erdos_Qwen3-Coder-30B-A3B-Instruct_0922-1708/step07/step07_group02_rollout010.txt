import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_population(n_points, n_pop, initial_h_values=None):
    """Initialize a diverse population."""
    population = []
    if initial_h_values is not None:
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, 2.0/n_points)
        population.append(h_init)
    
    # Fill remaining with random diversely
    for _ in range(n_pop - len(population)):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, 2.0/n_points)
        population.append(h)
    return population

def segment_blend(parent1, parent2):
    """Blend segments of two parents."""
    n_points = len(parent1)
    # Random crossover point
    crossover_point = np.random.randint(1, n_points)
    child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    return child

def mutate(h, mutation_rate=0.3, noise_scale=0.1):
    """Apply mutation to h."""
    h_mut = h.copy()
    n_points = len(h)
    # Segment flips
    if np.random.rand() < mutation_rate:
        # Determine number of segments to flip
        n_segments = np.random.randint(1, max(1, n_points//10))
        for _ in range(n_segments):
            start = np.random.randint(0, n_points)
            end = min(start + np.random.randint(1, n_points//5), n_points)
            # Flip region (0 -> 1, 1 -> 0) in a manner that still respects bounds
            h_mut[start:end] = 1 - h_mut[start:end]
    
    # Add noise
    if np.random.rand() < mutation_rate:
        noise = np.random.normal(0, noise_scale, n_points)
        h_mut += noise
    return h_mut

def fast_local_refinement(h, dx, max_iter=50):
    """Fast local refinement using coordinate descent."""
    h_refined = h.copy()
    current_c5 = evaluate_h(h_refined, dx)
    eps = 1e-6
    eta = 0.01
    
    for _ in range(max_iter):
        improved = False
        indices = list(range(len(h_refined)))
        np.random.shuffle(indices)
        
        for j in indices:
            # Compute gradient using finite differences
            h_plus = h_refined.copy()
            h_minus = h_refined.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            # Ensure feasibility
            h_plus = project_h(h_plus, len(h_plus), dx)
            h_minus = project_h(h_minus, len(h_minus), dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_refined[j] -= eta * grad_j
            h_refined[j] = max(0, min(1, h_refined[j]))
        
        # Reproject
        h_refined = project_h(h_refined, len(h_refined), dx)
        new_c5 = evaluate_h(h_refined, dx)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            improved = True
        else:
            eta *= 0.999
        
        if not improved:
            break
    
    return h_refined

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Population-based memetic algorithm
    n_pop = 50
    n_selected = max(1, n_pop // 5)
    mutation_scale = 0.1
    population = initialize_population(n_points, n_pop, 
                                       initial_h_values if 'initial_h_values' in globals() else None)
    
    # Best solution tracking
    h_best = None
    c5_best = float('inf')
    
    iter_count = 0
    while time.time() - start_time < budget_s - 5:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < c5_best:
            c5_best = fitness[0]
            h_best = population[0].copy()
        
        # Selection
        selected = population[:n_selected]
        
        # Generate new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < n_pop:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = segment_blend(parent1, parent2)
            child = mutate(child, mutation_rate=0.3, noise_scale=mutation_scale)
            child = project_h(child, n_points, dx)
            child = fast_local_refinement(child, dx, max_iter=50)
            new_population.append(child)
        
        population = new_population[:n_pop]
        mutation_scale *= 0.999
        iter_count += 1
    
    if h_best is None:
        # Fallback to best from initial run
        h_best = population[0] if population else np.ones(n_points) * 0.5
        c5_best = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), c5_best, n_points)
