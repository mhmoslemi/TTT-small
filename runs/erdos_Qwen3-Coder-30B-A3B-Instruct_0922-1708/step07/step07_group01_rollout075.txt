import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a trapezoidal wave for good initial structure
        h_init = 0.5 * (1 + np.cos(np.pi * x))
        h_init = project_h(h_init, n_points)
    
    # Better initialization: try multiple random starts
    best_h = None
    best_c5 = float('inf')
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Track best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Symmetric optimization parameters
    learning_rate = 0.02
    max_iter = int(1e6)
    epsilon = 1e-6
    patience = 200
    last_improvement = 0
    step = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Identify the shift k with maximum overlap
        corr_full = correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr_full)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient only for the maximum overlap shift
        # For shift k, gradient wrt h[i] = (1 - h[i-k]) when k >= 0
        # or gradient wrt h[i] = (1 - h[i+k]) when k < 0
        grad = np.zeros(n_points)
        if max_k >= 0:
            # h shifted by max_k, so we look at h[i] and h[i+max_k], 
            # but we need to handle indices carefully
            for j in range(n_points):
                if j + max_k < n_points:
                    grad[j] = 1 - h_current[j + max_k]
        else:  # max_k < 0
            for j in range(n_points):
                if j - max_k < n_points:
                    grad[j] = 1 - h_current[j - max_k]
        
        # Update step
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new point
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            # Occasionally accept worse moves (simulated annealing)
            if np.random.rand() < 0.02:
                h_current = h_new
                c5_current = c5_new
                
        # Adapt learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = i
            
        # Periodic perturbations
        if i % 100 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
            
        step += 1
    
    # Final refinement

        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.003)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
