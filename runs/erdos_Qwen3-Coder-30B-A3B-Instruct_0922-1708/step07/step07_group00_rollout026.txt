import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Use Sobol sequence for diverse initializations
    sobol_sampler = qmc.Sobol(d=n_points, seed=seed)
    n_initializations = 500
    
    for i in range(n_initializations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new initialization using Sobol sequence
        h_candidate = sobol_sampler.random(1)[0]
        h_candidate = repair_h(h_candidate, n_points)
        
        # Optimize with Nelder-Mead
        try:
            result = minimize(
                lambda x: evaluate_h(repair_h(x, n_points), dx),
                h_candidate,
                method='Nelder-Mead',
                options={'maxiter': 1000, 'xtol': 1e-6, 'ftol': 1e-6}
            )
            if result.success:
                h_opt = repair_h(result.x, n_points)
                c5_opt = evaluate_h(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass
    
    # Final refinement with gradient descent from best
    h_refined = best_h.copy()
    learning_rate = 1e-5
    for i in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        base_corr = correlate(h_refined, 1 - h_refined, mode='full')
        max_k_idx = np.argmax(base_corr)
        
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[j] = (corr_plus[max_k_idx] - base_corr[max_k_idx]) / eps
        
        # Update step
        h_refined = h_refined - learning_rate * grad
        h_refined = repair_h(h_refined, n_points)
        c5_refined = evaluate_h(h_refined, dx)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
            
        # Adaptive learning rate
        if i % 100 == 0 and i > 0:
            learning_rate *= 0.95
    
    return (best_h.tolist(), best_c5, n_points)
