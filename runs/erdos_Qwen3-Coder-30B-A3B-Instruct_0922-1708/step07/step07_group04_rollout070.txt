import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Optimize using L-BFGS with box constraints
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = repair_h(h, n_points)
        c5_val = evaluate_h(h, dx)
        return c5_val
    
    def constraint_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1.0
    
    # Set up bounds (0 to 1 for each variable)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Set up constraint
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Initialize with the initial solution
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Use a few restarts with different initializations
    for restart in range(5):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Start with different random perturbations
        h_perturbed = h_current.copy()
        h_perturbed += 0.1 * (np.random.rand(n_points) - 0.5)
        h_perturbed = repair_h(h_perturbed, n_points)
        
        result = minimize(
            objective,
            h_perturbed.flatten(),
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        
        h_new = result.x.reshape(n_points)
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            h_current = h_new.copy()
    
    # Final coordinate-wise refinement
    h_final = h_best.copy()
    c5_final = c5_best
    
    # Coordinate descent on discrete points
    max_iter = 10000
    eta = 0.005
    eps = 1e-6
    patience = 50
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_final.copy()
        improved = False
        
        for j in range(n_points):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Forward difference
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            
            # Backward difference
            h_minus = h_final.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_final[j] - eta * grad_j
            
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_final:
            h_final = h_new
            c5_final = c5_new
            improved = True
            last_improvement = i
        
        if i - last_improvement > patience:
            eta *= 0.95
            last_improvement = i
            
        if not improved:
            break
    
    return (h_final.tolist(), c5_final, n_points)
