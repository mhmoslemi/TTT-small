import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    """Perform segment-based crossover between two parents."""
    n = len(parent1)
    cut1, cut2 = sorted(np.random.randint(0, n, 2))
    child = parent1.copy()
    child[cut1:cut2] = parent2[cut1:cut2]
    return child

def mutate(h, mutation_rate=0.1, delta=0.1):
    """Mutate h by randomly changing some entries."""
    h_mut = h.copy()
    n = len(h)
    indices = np.random.choice(n, int(n * mutation_rate), replace=False)
    for i in indices:
        h_mut[i] += np.random.uniform(-delta, delta)
    return h_mut

def tournament_select(population, fitnesses, k=3):
    """Select an individual using tournament selection."""
    idx = np.random.choice(len(population), k)
    best_idx = idx[np.argmin([fitnesses[i] for i in idx])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)

    # Genetic algorithm parameters
    population_size = 100
    n_generations = 100
    mutation_rate = 0.1
    elite_size = 10
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = h_init + np.random.normal(0, 0.02, n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_h(h, dx) for h in population]
    
    # Evolutionary optimization
    for gen in range(n_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep elite individuals
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_select(population, fitnesses)
            parent2 = tournament_select(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child, mutation_rate)
            child = repair_h(child, n_points)
            new_population.append(child)
        
        population = new_population
        fitnesses = [evaluate_h(h, dx) for h in population]

    # Find best solution from GA
    best_idx = np.argmin(fitnesses)
    h_best = population[best_idx]
    c5_best = fitnesses[best_idx]
    
    # Fine-tune with gradient-based method
    h_fine = h_best.copy()
    eta = 0.001
    eps = 1e-6
    max_iter = 5000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_fine.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_fine.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_fine.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_fine[j] - eta * grad_j
            
        # Apply repair step to maintain feasibility
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_fine = h_new
            c5_best = c5_new
        else:
            # Occasionally accept worse solutions to escape local minima
            if np.random.rand() < 0.01:
                h_fine = h_new
        
        # Store best
        if c5_new < evaluate_h(h_best, dx):
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
