import numpy as np
import time
import random
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with provided construction or create binary one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        # Set first half to 1, second half to 0 (better than random)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Temperature schedule and parameters
    temp = 0.1
    cooling_rate = 0.995
    max_iter = 100000
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Helper function to compute correlation and C5
    def compute_c5(current_h):
        corr = correlate(current_h, 1 - current_h, mode='full')
        return np.max(corr) * dx
    
    # Initial evaluation
    current_c5 = compute_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate candidate via perturbation
        candidate_h = h.copy()
        
        # Pick two random indices to modify
        i, j = random.sample(range(n_points), 2)
        
        # Compute change in sum caused by perturbations to i and j
        delta = random.uniform(-0.05, 0.05)  # Perturbation size
        
        # Adjust values with constraints
        old_i = candidate_h[i]
        old_j = candidate_h[j]
        candidate_h[i] += delta
        candidate_h[j] -= delta
        
        # Ensure they remain in [0,1]
        if candidate_h[i] < 0:
            candidate_h[j] += candidate_h[i]
            candidate_h[i] = 0
        elif candidate_h[i] > 1:
            candidate_h[j] -= (candidate_h[i] - 1)
            candidate_h[i] = 1
            
        if candidate_h[j] < 0:
            candidate_h[i] += candidate_h[j]
            candidate_h[j] = 0
        elif candidate_h[j] > 1:
            candidate_h[i] -= (candidate_h[j] - 1)
            candidate_h[j] = 1
            
        # Ensure sum constraint
        current_sum = np.sum(candidate_h)
        target_sum = n_points / 2
        diff = target_sum - current_sum
        # Add difference to first index
        candidate_h[0] += diff
        
        # Ensure it stays within bounds
        if candidate_h[0] < 0:
            candidate_h[0] = 0
        elif candidate_h[0] > 1:
            candidate_h[0] = 1
            
        # Ensure all values are still in valid range
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Compute new overlap
        candidate_c5 = compute_c5(candidate_h)
        
        # Accept or reject based on Metropolis criterion
        if candidate_c5 < current_c5 or random.random() < np.exp(-(candidate_c5 - current_c5) / temp):
            h = candidate_h
            current_c5 = candidate_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Cool temperature
        temp *= cooling_rate
        
        # Occasionally restart from best if stuck
        if iteration % 5000 == 0 and iteration > 0:
            if abs(best_c5 - current_c5) < 1e-6:
                h = best_h.copy()
                current_c5 = best_c5
                temp = max(0.001, temp * 0.5)  # Increase temp to escape local minimum
    
    # Final validation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
