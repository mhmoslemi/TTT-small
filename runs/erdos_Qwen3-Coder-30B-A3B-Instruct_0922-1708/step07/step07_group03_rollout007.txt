import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * target_sum / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * target_sum / np.sum(h)
    
    # Frank-Wolfe parameters
    max_iter = 5000
    backtracking_factor = 0.5
    initial_gamma = 0.5
    
    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute gradient direction (approximated via finite differences)
        grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum - target_sum) > 1e-8:
                h_plus = h_plus * target_sum / h_plus_sum
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - best_c5) / eps
        
        # Solve linear subproblem: find s that minimizes grad^T * s s.t. constraints
        # The minimum is at one of the extreme points of feasible region:
        # 1. s = 0 vector (all zeros) - but this is not feasible due to sum constraint
        # 2. s = 1 vector (all ones) - scaled to meet sum constraint
        # 3. s = one-hot vector (0 or 1) - also scaled to meet sum constraint
        
        # We solve s that minimizes grad*s subject to s[i] in [0,1] and sum(s)=target_sum
        # This is a linear program:
        # minimize sum(grad[i]*s[i])
        # subject to 0 <= s[i] <= 1 for all i
        #          sum(s) = target_sum
        
        # Simple approach: use projection of -grad onto feasible region
        # We compute s that minimizes <grad, s> s.t. sum(s) = target_sum, 0 <= s <= 1
        
        # Start with s = ones * (target_sum / n_points), then adjust based on gradient
        s = np.ones(n_points) * (target_sum / n_points)
        # Ensure it satisfies the constraint (sum matches)
        s = s * target_sum / np.sum(s)
        
        # But we prefer the approximate min in direction of -grad
        # So we take s = argmin{grad^T * s : 0 <= s <= 1, sum(s) = target_sum}
        
        # Simple heuristic: sort gradient, set s to be 1 at lowest gradient indices, 0 otherwise
        # But better: we can directly construct such s using least-gradient method
        # Get argmin over all s where some are 1 and others 0, _and_ sum(s) = target_sum
        # i.e., select min `target_sum` entries of h (instead of 1's from where h=1)
        
        # np.argsort(grad) gives indices in ascending order of gradient values (smaller better)
        # Make s = 1 where indices are among first target_sum indices (min gradient)
        s = np.zeros_like(h)
        # Sorted indices by increasing gradient value
        ordered_indices = np.argsort(grad)
        for i in range(int(target_sum)):
            s[ordered_indices[i]] = 1.0 
        # Scale to sum to target_sum
        if np.sum(s) > 0:
            s = s * target_sum / np.sum(s)
        
        # Now compute step size gamma using line search
        gamma = initial_gamma
        best_gamma = 0.0
        best_new_c5 = best_c5
        
        for _ in range(20):  # Max backtracking steps
            h_new = (1 - gamma) * h + gamma * s
            # Ensure h_new stays in [0,1] and has correct sum
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if abs(h_new_sum - target_sum) > 1e-8:
                h_new = h_new * target_sum / h_new_sum
            
            new_c5 = compute_c5(h_new)
            
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_gamma = gamma
            gamma *= backtracking_factor
            if gamma < 1e-10:
                break
                
        if best_gamma > 0:
            h = (1 - best_gamma) * h + best_gamma * s
            # Ensure h stays in [0,1] and sum matches
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * target_sum / h_sum
            
            if best_new_c5 < best_c5:
                best_c5 = best_new_c5
                best_h = h.copy()
        
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
