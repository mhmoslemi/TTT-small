import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    S = 50  # number of particles
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    T = 1.0
    cooling_rate = 0.999
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        particles.append(np.random.rand(n_points))
        particles[0] = particles[0] * (n_points / 2) / np.sum(particles[0])
        velocities.append(np.zeros(n_points))
    
    # Initialize remaining particles randomly
    for _ in range(S - len(particles)):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
    
    # Track personal bests
    personal_bests = []
    personal_best_fitness = []
    
    for i in range(S):
        h = particles[i]
        corr = correlate(h, 1 - h, mode='full')
        fitness = np.max(corr) * dx
        personal_bests.append(h.copy())
        personal_best_fitness.append(fitness)
    
    # Global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_bests[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        for i in range(S):
            h = particles[i]
            
            # Compute current fitness
            corr = correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_bests[i] = h.copy()
            
            # Update global best
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best = h.copy()
        
        # Update particles
        for i in range(S):
            if (time.time() - start_time) > budget_s - 5:
                break
                
            r1, r2 = np.random.rand(2)
            v = w * velocities[i] + c1 * r1 * (personal_bests[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            
            # Apply simulated annealing
            if np.random.rand() < T:
                v += np.random.uniform(-0.1, 0.1, n_points)
            
            particles[i] = particles[i] + v
            
            # Repair constraints
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to meet sum constraint
            h_sum = np.sum(particles[i])
            target_sum = n_points / 2
            if abs(h_sum - target_sum) > 1e-8:
                particles[i] = particles[i] * target_sum / h_sum
                
            velocities[i] = v
        
        # Decrease temperature
        T = max(T * cooling_rate, 0.01)
    
    # Final evaluation of global best
    corr = correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (global_best.tolist(), final_c5, n_points)
