import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidate(n_points, seed=0):
    np.random.seed(seed)
    # Generate a candidate using a sum of sinusoids
    h = np.zeros(n_points)
    # Use 3-5 sinusoidal modes with random coefficients
    num_modes = np.random.randint(3, 6)
    for i in range(num_modes):
        A = np.random.uniform(-0.5, 0.5)
        B = np.random.uniform(0.2, 0.8)
        k = np.random.uniform(1, 5)
        x = np.linspace(0, 2, n_points)
        h += A * np.cos(k * np.pi * x) + B * np.sin(k * np.pi * x)
    # Ensure values in [0,1] and correct sum
    h = repair_h(h, n_points)
    return h

    start_time = time.time()
    
    # Collect initial population

    # Use initial_h_values if available

        h_init = repair_h(h_init, n_points)
        candidates.append(h_init)
    
    # Generate diverse initial candidates
    num_candidates = 50
    for i in range(num_candidates):
        if time.time() - start_time > budget_s - 5:
            break
        h_candidate = generate_fourier_candidate(n_points, seed=i+1000)
        candidates.append(h_candidate)
    
    # Evaluate all candidates
    candidate_scores = []
    for h in candidates:
        c5_score = evaluate_h(h, dx)
        candidate_scores.append((h, c5_score))
    
    # Sort by score
    candidate_scores.sort(key=lambda x: x[1])
    top_candidates = candidate_scores[:10]
    
    best_h = top_candidates[0][0]
    best_c5 = top_candidates[0][1]
    
    # Refinement using gradient descent
    max_refinement_iterations = 2000
    learning_rate = 0.01
    patience = 200
    best_improvement_iter = 0
    
    for h_refine, c5_refine in top_candidates:
        h_current = h_refine.copy()
        c5_current = c5_refine
        
        for i in range(max_refinement_iterations):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros(n_points)
            epsilon = 1e-6
            
            for j in range(n_points):
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                h_plus[j] += epsilon
                h_minus[j] -= epsilon
                h_plus = repair_h(h_plus, n_points)
                h_minus = repair_h(h_minus, n_points)
                
                c5_plus = evaluate_h(h_plus, dx)
                c5_minus = evaluate_h(h_minus, dx)
                grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
            
            # Update using gradient descent
            h_new = h_current - learning_rate * grad
            h_new = repair_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            # Accept or reject
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new

                    h_current = h_new
                    c5_current = c5_new
            
            # Adapt learning rate
            if i - best_improvement_iter > patience:
                learning_rate *= 0.95
                if learning_rate < 1e-6:
                    break
    
    return (best_h.tolist(), best_c5, n_points)
