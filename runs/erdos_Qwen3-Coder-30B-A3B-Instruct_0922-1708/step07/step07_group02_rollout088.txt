import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def segment_blend(h1, h2, p=0.5):
    """Blend two step functions by swapping segments."""
    n = len(h1)
    split_point = int(p * n)
    h_new = np.concatenate([h1[:split_point], h2[split_point:]])
    return h_new

def mutate_h(h, mutation_rate=0.3, noise_scale=0.1):
    """Add mutation to a step function."""
    h_new = h.copy()
    if np.random.rand() < mutation_rate:
        # Random segment flip
        n = len(h_new)
        start = np.random.randint(0, n)
        end = np.random.randint(start + 1, n + 1)
        h_new[start:end] = 1 - h_new[start:end]
    if np.random.rand() < mutation_rate:
        # Add noise
        noise = np.random.normal(0, noise_scale, len(h_new))
        h_new += noise
    return h_new

def local_refine(h, dx, max_iter=50, eta=0.01, eps=1e-6):
    """Fast local refinement using coordinate descent."""
    h_refined = h.copy()
    for _ in range(max_iter):
        improved = False
        indices = list(range(len(h_refined)))
        np.random.shuffle(indices)
        for j in indices:
            h_plus = h_refined.copy()
            h_minus = h_refined.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = project_h(h_plus, len(h_plus), dx)
            h_minus = project_h(h_minus, len(h_minus), dx)
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            h_refined[j] -= eta * grad_j
            h_refined[j] = max(0, min(1, h_refined[j]))
        h_refined = project_h(h_refined, len(h_refined), dx)
    return h_refined

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = np.zeros(n_points)
        # Create a piecewise constant function
        mid = n_points // 2
        h_init[:mid] = 0.75
        h_init[mid:] = 0.25
        h_init = project_h(h_init, n_points, dx)
    
    # Memetic algorithm parameters
    population_size = 50
    elite_size = 10
    max_generations = 1500
    mutation_scale = 0.1
    noise_scale = 0.05
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h_candidate = h_init.copy()
        # Add some variation
        h_candidate = mutate_h(h_candidate, mutation_rate=0.4, noise_scale=noise_scale)
        h_candidate = project_h(h_candidate, n_points, dx)
        population.append(h_candidate)
    
    # Track best solution
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all candidates
        fitness = [evaluate_h(h, dx) for h in population]
        sorted_indices = np.argsort(fitness)
        
        # Update best solution
        if fitness[sorted_indices[0]] < c5_best:
            h_best = population[sorted_indices[0]].copy()
            c5_best = fitness[sorted_indices[0]]
            
        # Selection
        elite_indices = sorted_indices[:elite_size]
        parents = [population[i] for i in elite_indices]
        
        # Generate offspring
        offspring = []
        while len(offspring) < population_size - elite_size:
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            h_child = segment_blend(parent1, parent2, 0.5)
            h_child = mutate_h(h_child, mutation_rate=0.3, noise_scale=mutation_scale)
            h_child = project_h(h_child, n_points, dx)
            
            # Local refine offspring
            h_child = local_refine(h_child, dx, max_iter=50)
            
            offspring.append(h_child)
            if len(offspring) >= population_size - elite_size:
                break
                
        # Replace worst candidates
        worst_indices = sorted_indices[-(population_size - elite_size):]
        for i, idx in enumerate(worst_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
                
        # Adaptive mutation rate
        mutation_scale *= 0.999
        if mutation_scale < 0.01:
            mutation_scale = 0.01
            
    return (h_best.tolist(), c5_best, n_points)
