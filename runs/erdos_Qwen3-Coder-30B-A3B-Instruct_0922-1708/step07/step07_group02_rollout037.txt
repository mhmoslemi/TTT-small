import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def generate_candidate(n_points, dx, initial_h=None):
    """Generate a feasible candidate solution."""
    if initial_h is not None:
        h = initial_h.copy()
    else:
        h = np.random.rand(n_points)
    return project_h(h, n_points, dx)

def crossover(parent1, parent2, n_points):
    """Perform segment-wise crossover."""
    # Randomly select crossover points
    cut1 = np.random.randint(1, n_points-1)
    cut2 = np.random.randint(cut1+1, n_points)
    
    offspring = parent1.copy()
    offspring[cut1:cut2] = parent2[cut1:cut2]
    
    return offspring

def mutate(h, n_points, mutation_scale):
    """Apply mutation to a candidate."""
    h_mut = h.copy()
    # Random segment flip (30% chance)
    if np.random.rand() < 0.3:
        flip_start = np.random.randint(0, n_points)
        flip_len = np.random.randint(1, n_points//5)
        end = min(flip_start + flip_len, n_points)
        h_mut[flip_start:end] = 1 - h_mut[flip_start:end]
    
    # Gaussian noise addition
    noise = np.random.normal(0, mutation_scale, n_points)
    h_mut += noise
    
    return project_h(h_mut, n_points, 2.0/n_points)

def local_refinement(h, n_points, dx, max_iter=50):
    """Fast local refinement using coordinate descent."""
    h_refine = h.copy()
    eps = 1e-6
    eta = 0.01
    
    for _ in range(max_iter):
        improved = False
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices:
            h_plus = h_refine.copy()
            h_minus = h_refine.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            h_refine[j] -= eta * grad_j
            h_refine[j] = max(0, min(1, h_refine[j]))
            
        h_refine = project_h(h_refine, n_points, dx)
        
    return h_refine

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    initial_h = None
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        initial_h = h_init
    
    # Memetic algorithm parameters
    population_size = 50
    num_parents = max(1, population_size // 5)
    max_generations = 1000
    mutation_scale = 0.1
    
    # Initialize population
    population = [generate_candidate(n_points, dx, initial_h) for _ in range(population_size)]
    fitness = [evaluate_h(h, dx) for h in population]
    
    # Best solution tracking
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create offspring
        offspring = []
        for _ in range(population_size):
            # Select parents
            parent1 = population[np.random.randint(0, num_parents)]
            parent2 = population[np.random.randint(0, num_parents)]
            
            # Crossover
            child = crossover(parent1, parent2, n_points)
            
            # Mutation
            child = mutate(child, n_points, mutation_scale)
            
            # Local refinement
            child = local_refinement(child, n_points, dx, 50)
            
            offspring.append(child)
        
        # Replace population
        population = offspring
        fitness = [evaluate_h(h, dx) for h in population]
        
        # Adaptive mutation scaling
        mutation_scale *= 0.999
    
    return (best_h.tolist(), best_c5, n_points)
