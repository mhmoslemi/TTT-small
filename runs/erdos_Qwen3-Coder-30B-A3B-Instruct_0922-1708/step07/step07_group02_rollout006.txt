import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate a solution."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip (30% chance)
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    return mutated

    dx = 2.0 / n_points

    mutation_scale = 0.1
    mutation_decay = 0.999
    stagnation_counter = 0
    best_c5 = float('inf')

    # Initialize population

        h_init = project_h(h_init, n_points, dx)
        population.append(h_init)
    else:
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
        population.append(h_init)

    # Add diverse initial solutions
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        population.append(h)

    # Evolution loop
    generation = 0
    while time.time() - start_time < budget_s - 5:
        generation += 1
        
        # Evaluate all population members
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Track best
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Selection: keep top 20%
        top_count = max(1, population_size // 5)
        selected = population[:top_count]
        
        # Generate new population
        new_population = selected.copy()
        
        # Mutation-based evolution
        while len(new_population) < population_size:
            # Select two parents for crossover
            parent1 = selected[np.random.randint(len(selected))]
            parent2 = selected[np.random.randint(len(selected))]
            
            # Blend parents (simple averaging)
            child = (parent1 + parent2) / 2.0
            child = project_h(child, n_points, dx)
            
            # Mutate
            child = mutate(child, generation, mutation_scale)
            new_population.append(child)
            
        population = new_population
        
        # Adjust mutation scale
        if generation % 10 == 0:
            mutation_scale *= mutation_decay
            
        # Handle stagnation with perturbation
        if stagnation_counter > 100:
            # Inject some diversity
            for i in range(1, min(5, len(population))):
                if np.random.rand() < 0.5:
                    population[i] = mutate(population[i], generation, 0.05)

            best_c5 = final_c5

        best_h = population[0].copy()

    return (best_h.tolist(), best_c5, n_points)
