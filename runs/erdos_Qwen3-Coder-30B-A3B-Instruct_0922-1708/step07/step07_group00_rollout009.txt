import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of the maximum overlap with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Identifying the k that gives maximum overlap
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n - 1)
    
    # Computing gradient for this k
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    # Normalize the gradient
    grad = grad / (np.linalg.norm(grad) + 1e-12)
    
    return grad

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    swarm_size = 100
    inertia_start = 0.9
    inertia_end = 0.1
    cognitive = 2.0
    social = 2.0
    max_iterations = 1000
    local_search_interval = 100
    local_search_steps = 100
    
    # Initialize swarm
    particles = []
    velocities = []
    particle_bests = []
    particle_best_scores = []
    
    for _ in range(swarm_size):
        h = h_init.copy()
        # Add small random noise to initial solution
        h += np.random.normal(0, 0.01, n_points)
        h = repair_h(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        particle_bests.append(h.copy())
        particle_best_scores.append(evaluate_h(h, dx))
    
    # Global best
    global_best_idx = np.argmin(particle_best_scores)
    global_best = particles[global_best_idx].copy()
    global_best_score = particle_best_scores[global_best_idx]
    
    # Main loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * iteration / max_iterations
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (particle_bests[i] - particles[i]) +
                social * r2 * (global_best - particles[i])
            )
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate current particle
            current_score = evaluate_h(particles[i], dx)
            
            # Update particle best
            if current_score < particle_best_scores[i]:
                particle_best_scores[i] = current_score
                particle_bests[i] = particles[i].copy()
                
                # Update global best
                if current_score < global_best_score:
                    global_best_score = current_score
                    global_best = particles[i].copy()
        
        # Local search refinement
        if iteration % local_search_interval == 0 and iteration > 0:
            # Gradient-based refinement of global best
            h_refined = global_best.copy()
            for _ in range(local_search_steps):
                if time.time() - start_time > budget_s - 5:
                    break
                grad = compute_gradient(h_refined, dx, n_points)
                h_refined = h_refined - 1e-5 * grad
                h_refined = repair_h(h_refined, n_points)
                score_refined = evaluate_h(h_refined, dx)
                if score_refined < global_best_score:
                    global_best_score = score_refined
                    global_best = h_refined.copy()
    
    # Final evaluation
    final_c5 = evaluate_h(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
