import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    # PSO parameters
    S = 50  # number of particles
    max_iter = 5000
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    
    # SA parameters
    T = 1.0  # initial temperature
    cooling_rate = 0.999  # cooling rate
    
    # Initialize particles
    particles = np.random.rand(S, n_points)
    for i in range(S):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Personal best positions
    personal_best = particles.copy()
    personal_best_fitness = np.full(S, float('inf'))
    
    # Global best
    global_best_idx = 0
    global_best_fitness = float('inf')
    
    # Main optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate all particles
        for i in range(S):
            h_current = particles[i]
            
            # Compute correlation
            corr = correlate(h_current, 1 - h_current, mode='full')
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = h_current.copy()
                personal_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best_idx = i
                    global_best_fitness = fitness
                    
        # Update particles
        for i in range(S):
            if i == global_best_idx:
                continue
                
            # Velocity update
            r1, r2 = np.random.rand(2)
            v = (w * (particles[i] - particles[i]) + 
                 c1 * r1 * (personal_best[i] - particles[i]) + 
                 c2 * r2 * (particles[global_best_idx] - particles[i]))
            
            # Apply SA perturbation
            if np.random.rand() < T:
                v += np.random.uniform(-0.1, 0.1, n_points)
                
            # Update position
            particles[i] = particles[i] + v
            
            # Repair constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
        # Cool down
        T = max(T * cooling_rate, 0.01)
    
    # Final best solution
    final_h = particles[global_best_idx]
    corr = correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (final_h.tolist(), final_c5, n_points)
