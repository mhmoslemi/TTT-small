import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    initial_temp = 0.1
    cooling_rate = 0.995
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Binary initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        h = h * (n_points / 2) / np.sum(h)
    
    # Local search with simulated annealing
    best_h = h.copy()
    best_c5 = float('inf')
    
    current_h = h.copy()
    current_c5 = np.max(correlate(current_h, 1 - current_h, mode='full')) * dx
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    temp = initial_temp
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor by perturbing two random indices
        i, j = np.random.randint(0, n_points, 2)
        delta = np.random.uniform(-0.05, 0.05)
        
        # Ensure we don't exceed bounds
        if current_h[i] + delta < 0:
            delta = -current_h[i]
        elif current_h[i] + delta > 1:
            delta = 1 - current_h[i]
            
        if current_h[j] - delta < 0:
            delta = current_h[j]
        elif current_h[j] - delta > 1:
            delta = current_h[j] - 1
            
        # Apply changes
        new_h = current_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        
        # Project back to valid range
        new_h = np.clip(new_h, 0, 1)
        
        # Scale to meet constraint sum(h) = n_points / 2
        h_sum = np.sum(new_h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            new_h = new_h * target_sum / h_sum
        
        # Evaluate new solution
        new_c5 = np.max(correlate(new_h, 1 - new_h, mode='full')) * dx
        
        # Accept or reject based on simulated annealing
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
            current_h = new_h
            current_c5 = new_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Cool down temperature
        temp *= cooling_rate
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
