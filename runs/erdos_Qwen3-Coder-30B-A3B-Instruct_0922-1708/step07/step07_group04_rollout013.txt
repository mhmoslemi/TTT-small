import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_individual(h, mutation_rate=0.1, perturb_magnitude=0.1):
    """Mutate an individual by randomly perturbing some elements."""
    h_mut = h.copy()
    num_to_mutate = int(len(h) * mutation_rate)
    indices = np.random.choice(len(h), num_to_mutate, replace=False)
    for i in indices:
        h_mut[i] += np.random.uniform(-perturb_magnitude, perturb_magnitude)
    return repair_h(h_mut, len(h))

def crossover_parents(h1, h2, crossover_rate=0.5):
    """Crossover two parents to create offspring."""
    h_child = h1.copy()
    mask = np.random.rand(len(h1)) < crossover_rate
    h_child[mask] = h2[mask]
    return h_child

def tournament_selection(population, fitnesses, tournament_size=3):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    winners = [population[i] for i in indices]
    winner_fitness = [fitnesses[i] for i in indices]
    best_idx = np.argmin(winner_fitness)
    return winners[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Generate initial population
    population.append(h_init)
    for _ in range(population_size - 1):
        h = mutate_individual(h_init, mutation_rate=0.1, perturb_magnitude=0.1)
        population.append(h)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    # Evolution loop
    generations = 100
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection and reproduction
        new_population = []
        
        # Elite: keep top 20%
        elite_indices = np.argsort(fitnesses)[:int(population_size * 0.2)]
        for i in elite_indices:
            new_population.append(population[i])
        
        # Generate rest via crossover and mutation
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child = crossover_parents(parent1, parent2)
            child = mutate_individual(child, mutation_rate=0.1, perturb_magnitude=0.05)
            new_population.append(child)
        
        population = new_population[:population_size]
    
    # Final gradient refinement for best solution
    if best_h is not None:
        h_current = best_h.copy()
        c5_current = evaluate_h(h_current, dx)
        h_best = h_current.copy()
        c5_best = c5_current
        eta = 0.001
        eps = 1e-6
        max_iter = 5000
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            h_new = h_current.copy()
            for j in range(n_points):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h_current.copy()
                h_minus[j] -= eps
                h_minus = repair_h(h_minus, n_points)
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad_j = (c5_plus - c5_minus) / (2 * eps)
                
                h_new[j] = h_current[j] - eta * grad_j
            
            h_new = repair_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    
        return (h_best.tolist(), c5_best, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
