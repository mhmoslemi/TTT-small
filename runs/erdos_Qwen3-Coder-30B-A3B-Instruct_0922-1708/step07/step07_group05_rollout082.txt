import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size):
    population = []
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Fill the rest with triangular pulses
    while len(population) < pop_size:
        h = np.zeros(n_points)
        # Place triangular pulse
        peak_idx = np.random.randint(n_points)
        width = np.random.randint(10, n_points//2)
        left = max(0, peak_idx - width//2)
        right = min(n_points, peak_idx + width//2)
        for i in range(left, right):
            if i < peak_idx:
                h[i] = (i - left) / (peak_idx - left)
            else:
                h[i] = (right - i) / (right - peak_idx)
        h = repair_h(h, n_points)
        population.append(h)
    return population

def selection(population, fitness):
    sorted_idx = np.argsort(fitness)
    selected = [population[i] for i in sorted_idx[:len(population)//2]]
    return selected

def crossover(parent1, parent2):
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n_points = len(h)
    mask = np.random.rand(n_points) < mutation_rate
    noise = np.random.normal(0, 0.05, n_points)
    h_mut = h + noise * mask
    h_mut = np.clip(h_mut, 0, 1)
    return h_mut

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # GA parameters
    pop_size = 50
    max_gen = 1000
    patience = 200
    mutation_rate = 0.2
    crossover_rate = 0.5
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        f = evaluate_h(h, dx)
        fitness.append(f)
    fitness = np.array(fitness)
    
    best_h = population[np.argmin(fitness)].copy()
    best_fitness = np.min(fitness)
    
    last_improvement = 0
    
    for generation in range(max_gen):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection
        selected = selection(population, fitness)
        
        # Create new population
        new_population = [best_h.copy()]  # Elitism
        while len(new_population) < pop_size:
            if len(selected) < 2:
                # Randomly create a new individual
                h = np.zeros(n_points)
                peak_idx = np.random.randint(n_points)
                width = np.random.randint(10, n_points//2)
                left = max(0, peak_idx - width//2)
                right = min(n_points, peak_idx + width//2)
                for i in range(left, right):
                    if i < peak_idx:
                        h[i] = (i - left) / (peak_idx - left)
                    else:
                        h[i] = (right - i) / (right - peak_idx)
                h = repair_h(h, n_points)
                new_population.append(h)
                continue
                
            # Crossover
            parent1, parent2 = np.random.choice(selected, size=2, replace=False)
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
                child1 = repair_h(child1, n_points)
                child2 = repair_h(child2, n_points)
                new_population.append(child1)
                if len(new_population) < pop_size:
                    new_population.append(child2)
            else:
                # Just mutate
                child1 = mutate(parent1, mutation_rate)
                child1 = repair_h(child1, n_points)
                new_population.append(child1)
                
        # Replace population
        population = new_population
        
        # Evaluate new population
        fitness = []
        for h in population:
            f = evaluate_h(h, dx)
            fitness.append(f)
        fitness = np.array(fitness)
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_fitness:
            best_fitness = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            last_improvement = generation
            
        # Early termination
        if generation - last_improvement > patience:
            break
    
    return (best_h.tolist(), best_fitness, n_points)
