import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(params, n_points):
    """Parametrize h from piecewise constant segments."""
    K = len(params) // 2
    heights = params[:K]
    boundaries = np.concatenate([[0], np.cumsum(params[K:])])
    boundaries = np.clip(boundaries, 0, 2)
    boundaries = np.append(boundaries, 2)
    
    # Ensure boundaries are ordered
    boundaries = np.sort(boundaries)
    
    # Adjust boundary to match [0,2]
    boundaries[0] = 0
    boundaries[-1] = 2
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(K):
        start_idx = int(boundaries[i] / dx)
        end_idx = int(boundaries[i+1] / dx)
        if start_idx < n_points:
            h[start_idx:end_idx] = heights[i]
    
    # Fix any issues at boundary indices
    for i in range(n_points):
        if i * dx >= 2:
            break
        pos = i * dx
        for j in range(K):
            if boundaries[j] <= pos < boundaries[j+1]:
                h[i] = heights[j]
                break
    
    return h

def compute_gradient_h(params, dx, n_points, k_max):
    """Compute gradient of C5 with respect to params."""
    eps = 1e-6
    grad = np.zeros_like(params)
    
    # Reference C5 evaluation
    h_base = parameterize_h(params, n_points)
    h_base = repair_h(h_base, n_points)
    corr_base = correlate(h_base, 1 - h_base, mode='full')
    max_k_idx = np.argmax(corr_base)
    
    for i in range(len(params)):
        params_plus = params.copy()
        params_plus[i] += eps
        h_plus = parameterize_h(params_plus, n_points)
        h_plus = repair_h(h_plus, n_points)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr_base[max_k_idx]) / eps
    
    return grad
