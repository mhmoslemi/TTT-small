import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def fourier_basis(x, K):
    """Construct Fourier basis up to K harmonics."""
    basis = np.zeros((len(x), 2*K))
    for k in range(1, K+1):
        basis[:, 2*(k-1)] = np.cos(k * np.pi * x)
        basis[:, 2*(k-1)+1] = np.sin(k * np.pi * x)
    return basis

def sigmoid(x):
    """Numerically stable sigmoid function."""
    return 1.0 / (1.0 + np.exp(-x))
