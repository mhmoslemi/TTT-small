import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization
        h = np.ones(n_points) * 0.5
    
    # Frank-Wolfe optimization
    best_c5 = float('inf')
    best_h = h.copy()
    gamma = 0.5
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient (by finite differences)
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps

        # Solve linear subproblem: minimize grad^T * s subject to constraints
        # Simple greedy: set s to {-1, 1} such that s_i = -sign(grad_i) when possible
        s = np.zeros_like(h)
        for i in range(n_points):
            if grad[i] < 0:
                s[i] = 1
            else:
                s[i] = 0
                
        # Ensure s has correct sum
        s_sum = np.sum(s)
        if abs(s_sum - n_points / 2) > 1e-8:
            s = s * (n_points / 2) / s_sum
            
        # Ensure s is in [0,1]
        s = np.clip(s, 0, 1)
        
        # Backtracking line search for step size
        while True:
            h_new = (1 - gamma) * h + gamma * s
            # Rescale to keep sum constraint
            h_new_sum = np.sum(h_new)
            h_new = h_new * (n_points / 2) / h_new_sum
            
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < c5 or gamma < 1e-6:
                h = h_new
                break
            gamma *= 0.5
            
        # Update gamma
        gamma = min(0.99 * gamma, 0.9)
    
    # Final check
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
