import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with best known construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initial random construction with correct integral
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Parameters
    max_iter = 20000
    learning_rate = 1e-3
    barrier_param = 1e3
    barrier_decay = 0.99
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_h(h_vec):
        # Project onto [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Ensure integral constraint
        current_sum = np.sum(h_vec)
        if abs(current_sum - n_points/2) > 1e-8:
            h_vec = h_vec * (n_points/2) / current_sum
        return h_vec
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current C5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient via adjoint method
        # We need to find which lag achieves the max overlap
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_corr = corr[max_lag_idx]
        
        # Gradient computation
        grad = np.zeros_like(h)
        # Let's do this by computing the gradient of the correlation that achieved maximum
        # The derivative of correlation term ∫h(x)(1-h(x+k)) dx with respect to h_i
        # For each point i, we add contribution when (x+k) = i, so k = i - x
        # We go through the correlation terms and accumulate the gradient
        
        # Note: Efficient way to compute gradient
        # concat h with 1-h and use correlate
        # For a specific lag k, ∂/∂h_i of ∫h(x)(1-h(x+k)) dx:
        # if x = i, it's (1-h(i+k))
        # But we must be careful about retaining proper alignment
        # Let's just signal the positions that contributed to maximum
        
        # Build gradient more consistently via finite differences
        eps = 1e-6
        grad_fd = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad_fd[i] = (c5_plus - c5) / eps
        
        # Update with gradient
        h = h - learning_rate * grad_fd
        h = project_h(h)
        
        # Update barrier parameter
        barrier_param *= barrier_decay
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
