import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Convex optimization approach
    # Variables
    h_var = cp.Variable(n_points)
    C5_var = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1]
    constraints.append(cp.sum(h_var) == n_points / 2)
    
    # Add quadratic constraints for all lags k
    full_corr = correlate(np.ones(n_points), np.ones(n_points), mode='full')
    max_lag = len(full_corr) - 1
    
    # Build correlation matrix (for all lags)
    h_full = np.concatenate([h, np.zeros(n_points)])
    
    for k in range(max_lag + 1):
        # Calculate the actual correlation for this lag
        start = k if k < n_points else 0
        end = start + n_points
        if k >= n_points:
            end = 2 * n_points - k
        
        # This is effectively the i-th element of the full convolution
        # Here we make it specific to only the relevant part
        lag_corr = cp.sum(cp.multiply(h_var, cp.roll(1 - h_var, k)))
        constraints.append(lag_corr <= C5_var)
    
    # Objective
    objective = cp.Minimize(C5_var)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    # Solve with a solver that supports quadratic constraints
    solver = None
    if cp.MOSEK in cp.installed_solvers():
        solver = cp.MOSEK
    elif cp.GUROBI in cp.installed_solvers():
        solver = cp.GUROBI
    elif cp.CVXOPT in cp.installed_solvers():
        solver = cp.CVXOPT
    else:
        # Fallback to ECOS for simple constraints if no other is available
        solver = cp.ECOS
    
    try:
        prob.solve(solver=solver, verbose=False, time_limit=budget_s - 10)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            result_h = h_var.value
            result_c5 = C5_var.value
            h_opt = np.clip(result_h, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        else:
            # If CVXPY fails, use a simpler optimization approach
            h_opt = h.copy()
            result_c5 = float('inf')
        
    except Exception:
        # If CVXPY fails or times out, fallback to local optimization
        h_opt = h.copy()
        result_c5 = float('inf')
    
    # Final refinement with a local search if needed
    if result_c5 == float('inf') or (time.time() - start_time) < budget_s - 30:
        # Small local refinement
        learning_rate = 1e-3
        max_iter = 5000
        for iteration in range(max_iter):
            if (time.time() - start_time) > budget_s - 5:
                break
                
            # Compute current C5
            corr = correlate(h_opt, 1 - h_opt, mode='full')
            current_c5 = np.max(corr) * dx
            
            # For each point, try small adjustments
            grad = np.zeros_like(h_opt)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h_opt.copy()
                h_plus[i] += eps
                if h_plus[i] > 1:
                    h_plus[i] = 1
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - current_c5) / eps
            
            # Update
            h_opt = h_opt - learning_rate * grad
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            if current_c5 < result_c5:
                result_c5 = current_c5

    # Final evaluation
    corr = correlate(h_opt, 1 - h_opt, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_opt.tolist(), final_c5, n_points)
