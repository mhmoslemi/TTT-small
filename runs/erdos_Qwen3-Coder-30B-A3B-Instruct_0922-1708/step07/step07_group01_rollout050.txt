import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a simple step function
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 1.0
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient descent parameters
    max_iter = int(1e6)
    epsilon = 1e-6
    patience = 100
    last_improvement = 0
    learning_rate = 0.05
    
    # Multiple random restarts
    num_restarts = 5
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_start = h_init.copy()
        # Add some noise for variety
        noise = np.random.normal(0, 0.01, n_points)
        h_start = project_h(h_start + noise, n_points)
        
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current, dx)
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Find the index of maximum overlap
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_idx = np.argmax(corr)
            # The shift k that gives maximum overlap
            k_max = max_idx - (len(h_current) - 1)
            
            # Compute exact gradient based on the k_max shift
            grad = np.zeros(n_points)
            # Create shifted version of h_current
            h_shifted = np.roll(h_current, k_max)
            if k_max >= 0:
                grad = 1 - h_shifted
            else:
                grad = 1 - np.roll(h_current, k_max)
                
            # Apply gradient to update h
            h_new = h_current - learning_rate * grad
            
            # Project back to feasible set
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept move if improvement
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
            else:
                # Occasionally accept worse moves
                if np.random.rand() < 0.01:
                    h_current = h_new.copy()
                    c5_current = c5_new
            
            # Update best
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
            
            # Adaptive learning rate
            if i - last_improvement > patience:
                learning_rate *= 0.95
                last_improvement = i
            
            # Occasionally jitter to escape local minima
            if i % 200 == 0 and i > 0:
                noise = np.random.normal(0, 0.005, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
    
    # Fine-tune with local search
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 20), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
