import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    mutation_scale = 0.1
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    # Add initial solution and generate random solutions
    population.append(h_init)
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        population.append(h)
    
    best_solution = None
    best_c5 = float('inf')
    
    generation = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 5:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_solution = h.copy()
        
        # Selection: keep top 20%
        sorted_indices = np.argsort(fitness)
        elite_count = max(1, population_size // 5)
        elite_indices = sorted_indices[:elite_count]
        
        # Generate new population
        new_population = [population[i] for i in elite_indices]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(elite_indices, 2, replace=False)
            parent1, parent2 = population[idx1], population[idx2]
            
            # Crossover (averaging)
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutate
            child = mutate(child, generation, mutation_scale)
            new_population.append(child)
        
        population = new_population[:population_size]
        
        # Adjust mutation scale
        if generation % 10 == 0:
            # Decrease mutation rate
            mutation_scale *= 0.999
            
        # Check for stagnation
        if generation % 50 == 0:
            if len(new_population) > 0:
                best_in_gen = min(evaluate_h(h, dx) for h in new_population)
                if best_in_gen >= best_c5 - 1e-6:
                    stagnation_count += 1
                else:
                    stagnation_count = 0
            
            # Inject random perturbations if stuck
            if stagnation_count > 3:
                for i in range(0, min(5, len(population))):
                    h = population[i]
                    h = mutate(h, generation, 0.1)
                    population[i] = h
                stagnation_count = 0
        
        # Keep mutation scale bounded
        mutation_scale = max(0.001, mutation_scale)
    
    if best_solution is None:
        # Fallback to the best in final population
        fitness = [evaluate_h(h, dx) for h in population]
        best_idx = np.argmin(fitness)
        best_solution = population[best_idx]
        best_c5 = fitness[best_idx]
    
    # Final evaluation
    final_c5 = evaluate_h(best_solution, dx)
    
    return (best_solution.tolist(), final_c5, n_points)
