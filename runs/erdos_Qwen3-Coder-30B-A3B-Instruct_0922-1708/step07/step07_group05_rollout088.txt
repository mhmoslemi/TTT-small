import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Create a structured construction
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)

    # PSO parameters
    num_particles = 100
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    max_iter = 400
    patience = 100
    
    # Initialize particles
    particles = []
    velocities = []
    for _ in range(num_particles):
        particle = h_init + np.random.normal(0, 0.01, n_points)
        particle = repair_h(particle, n_points)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
    
    # Track personal and global best
    pbest = [particles[i].copy() for i in range(num_particles)]
    pbest_values = [evaluate_h(particles[i], dx) for i in range(num_particles)]
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO loop
    last_improvement = 0
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        # Update each particle
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            current_value = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
                    last_improvement = iter_num
        
        # Mutation
        if iter_num % 25 == 0:
            num_mutate = int(mutation_rate * num_particles)
            mutate_indices = np.random.choice(num_particles, num_mutate, replace=False)
            for i in mutate_indices:
                mask = np.random.rand(n_points) < 0.2
                particles[i][mask] += np.random.normal(0, mutation_sigma, np.sum(mask))
                particles[i] = repair_h(particles[i], n_points)
    
    return (gbest.tolist(), gbest_value, n_points)
