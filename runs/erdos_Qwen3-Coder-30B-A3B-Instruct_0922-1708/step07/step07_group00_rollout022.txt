import numpy as np
import time
from scipy.signal import correlate

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with the given construction or a random one
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Use a piecewise constant representation with 5 segments
    K = 5
    # Initialize segment boundaries (excluding 0 and 2)
    boundaries = np.linspace(0.2, 1.8, K-1)
    # Initialize heights
    heights = np.random.rand(K)
    heights = heights * (n_points / 2) / np.sum(heights * (np.diff(np.concatenate([[0], boundaries, [2]]))))
    
    def reconstruct_h(boundaries, heights):
        """Reconstruct h from boundaries and heights."""
        segments = np.concatenate([[0], boundaries, [2]])
        h = np.zeros(n_points)
        for i in range(K):
            start_idx = int(segments[i] / dx)
            end_idx = int(segments[i+1] / dx)
            if end_idx > start_idx:
                h[start_idx:end_idx] = heights[i]
        return h
    
    def obj_fun(boundaries, heights):
        """Objective function to minimize."""
        h = reconstruct_h(boundaries, heights)
        h = repair_h(h, n_points)
        return evaluate_h(h, dx)
    
    def compute_grad(boundaries, heights, eps=1e-6):
        """Compute gradient using finite differences."""
        grad_b = np.zeros_like(boundaries)
        grad_h = np.zeros_like(heights)
        
        orig_val = obj_fun(boundaries, heights)
        
        # Compute gradient for boundaries
        for i in range(len(boundaries)):
            boundaries_plus = boundaries.copy()
            boundaries_plus[i] += eps
            val_plus = obj_fun(boundaries_plus, heights)
            grad_b[i] = (val_plus - orig_val) / eps
            
        # Compute gradient for heights
        for i in range(len(heights)):
            heights_plus = heights.copy()
            heights_plus[i] += eps
            val_plus = obj_fun(boundaries, heights_plus)
            grad_h[i] = (val_plus - orig_val) / eps
            
        return grad_b, grad_h
    
    # Optimization loop
    current_boundaries = boundaries.copy()
    current_heights = heights.copy()
    current_c5 = obj_fun(current_boundaries, current_heights)
    best_c5 = current_c5
    best_boundaries = current_boundaries.copy()
    best_heights = current_heights.copy()
    
    max_iter = 5000
    learning_rate = 1e-3
    decay_rate = 0.995
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break

        # Compute gradients
        grad_b, grad_h = compute_grad(current_boundaries, current_heights)
        
        # Update boundaries and heights (with projections)
        current_boundaries -= learning_rate * grad_b
        current_heights -= learning_rate * grad_h
        
        # Ensure boundaries are in valid order and range
        current_boundaries = np.clip(current_boundaries, 0.01, 1.99)
        current_boundaries = np.sort(current_boundaries)

        h_temp = reconstruct_h(current_boundaries, current_heights)

        current_c5 = obj_fun(current_boundaries, current_heights)
        
        # Track best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_boundaries = current_boundaries.copy()
            best_heights = current_heights.copy()
        
        # Adaptive learning rate
        if i % 100 == 0:
            learning_rate *= decay_rate

    # Final reconstruction
    final_h = reconstruct_h(best_boundaries, best_heights)
    final_h = repair_h(final_h, n_points)
    
    return (final_h.tolist(), best_c5, n_points)
