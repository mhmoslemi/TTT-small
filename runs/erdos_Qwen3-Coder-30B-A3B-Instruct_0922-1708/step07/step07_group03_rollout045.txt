import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    # Define variables
    h_var = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Get correlation indices
    max_lag = 2 * n_points - 1
    corr_indices = []
    
    # Generate all correlation constraints
    for k in range(max_lag):
        # For correlation at lag k
        corr = 0
        for i in range(n_points):
            j = i + k - (n_points - 1)  # lag offset
            if 0 <= j < n_points:
                corr += h_var[i] * (1 - h_var[j])
        corr *= dx
        constraints.append(corr <= C5)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    solvers_to_try = ['MOSEK', 'GUROBI', 'ECOS']
    solver = None
    for s in solvers_to_try:
        try:
            if cp.installed_solvers() >= {s}:
                solver = s
                break
        except:
            continue
    
    if solver is None:
        solver = 'ECOS'  # fallback
    
    best_h = h
    best_c5 = float('inf')
    
    try:
        # Solve once to get a baseline
        prob.solve(solver=solver, verbose=False)
        if prob.status == 'optimal':
            solution = h_var.value
            if solution is not None:
                h_opt = np.clip(solution, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                corr = correlate(h_opt, 1 - h_opt, mode='full')
                c5_val = np.max(corr) * dx
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
    except:
        pass  # fall back to simpler approach
    
    # Gradient-based optimization with restarts
    learning_rate = 1e-3
    max_iter = 10000
    perturb_every = 1000
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Sample from current solution to potentially escape local minima
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        if iteration % perturb_every == 0 and iteration > 0:
            # Small random perturbation to escape local minima
            h = np.clip(h + np.random.normal(0, 1e-3, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Estimate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps

        # Update step
        h = h - learning_rate * grad
        
        # Projection to [0,1]  
        h = np.clip(h, 0, 1)
        
        # Rescale to meet constraint
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
