import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, population_size):
    population = []
    # If initial_h_values exists, use it
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Add random or structured initial individuals
    while len(population) < population_size:
        h = np.zeros(n_points)
        # Create a triangular pulse
        peak_idx = np.random.randint(0, n_points)
        width = np.random.randint(5, n_points // 4)
        start_idx = max(0, peak_idx - width // 2)
        end_idx = min(n_points, peak_idx + width // 2)
        for i in range(start_idx, end_idx):
            h[i] = max(0, 1 - abs(i - peak_idx) / (width / 2))
        
        h = repair_h(h, n_points)
        if len(population) < population_size:
            population.append(h)
    
    return population

def selection(population, fitnesses):
    # Tournament selection
    selected = []
    for _ in range(len(population)):
        idx1, idx2 = np.random.choice(len(population), 2, replace=False)
        if fitnesses[idx1] < fitnesses[idx2]:
            selected.append(population[idx1])
        else:
            selected.append(population[idx2])
    return selected

def crossover(parent1, parent2):
    # Single-point crossover
    n_points = len(parent1)
    point = np.random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n_points = len(h)
    num_mutation = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, num_mutation, replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.05)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # GA parameters
    population_size = 50
    generations = 1000
    mutation_rate = 0.2
    elite_size = 1
    patience = 200
    
    # Initialize population
    population = initialize_population(n_points, population_size)
    
    # Evaluation
    fitnesses = [evaluate_h(h, dx) for h in population]
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    last_improvement = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection
        selected_population = selection(population, fitnesses)
        
        # Crossover and Mutation
        new_population = [best_h]  # Elitism
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected_population, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = repair_h(mutate(child1, mutation_rate), n_points)
            child2 = repair_h(mutate(child2, mutation_rate), n_points)
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        new_population = new_population[:population_size]
        
        # Evaluation
        new_fitnesses = [evaluate_h(h, dx) for h in new_population]
        
        # Replacement
        population = new_population
        fitnesses = new_fitnesses
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            last_improvement = gen
            
        # Check for early stopping
        if gen - last_improvement > patience:
            break
    
    return (best_h.tolist(), best_fitness, n_points)
