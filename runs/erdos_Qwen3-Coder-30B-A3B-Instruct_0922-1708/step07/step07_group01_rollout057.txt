import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Use a more structured search approach: varying binary configurations
    # Start with current solutions
    for iter_num in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Construct a new candidate
        h_new = h_current.copy()
        # Permute a few random locations
        indices = np.random.choice(n_points, max(3, n_points // 10), replace=False)
        
        for idx in indices:
            # Try randomized changes
            step_size = 0.05
            h_new[idx] += np.random.normal(0, step_size)
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Project back to feasible set
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Accept new solution if better or sometimes randomly
        if c5_new < c5_current or np.random.rand() < 0.05:
            h_current = h_new
            c5_current = c5_new
            
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            
    # Final polishing by local neighborhood search
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        # Small perturbations
        indices = np.random.choice(n_points, max(2, n_points // 20), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.01)
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
