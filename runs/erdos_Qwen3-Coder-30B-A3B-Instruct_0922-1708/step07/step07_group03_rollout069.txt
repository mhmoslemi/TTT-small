import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize particles
    S = 50  # number of particles
    inertia = 0.7
    cognitiveness = 2.0
    socialness = 2.0
    temperature = 1.0
    cooling_rate = 0.999
    
    # Initialize swarm
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        particles.append(np.ones(n_points) * 0.5)
        velocities.append(np.zeros(n_points))
        
    # Fill remaining particles with random values
    for i in range(S - len(particles)):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.rand(n_points) - 0.5)
    
    # Track best values
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [float('inf')] * S
    global_best = particles[0].copy()
    global_best_fitness = float('inf')
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate all particles
        for i in range(S):
            h = particles[i]
            
            # Compute overlap integral
            corr = correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best[i] = h.copy()
            
            # Update global best
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best = h.copy()
                
        # Update particles
        for i in range(S):
            h = particles[i]
            v = velocities[i]
            
            # PSO velocity update
            rp = np.random.rand(n_points)
            rg = np.random.rand(n_points)
            
            v = (inertia * v + 
                 cognitiveness * rp * (personal_best[i] - h) + 
                 socialness * rg * (global_best - h))
            
            # Simulated Annealing perturbation
            if np.random.rand() < temperature:
                v += (np.random.rand(n_points) - 0.5) * 0.2
            
            # Update position
            h = h + v
            
            # Repair constraints
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-8:
                h = h * (n_points / 2) / h_sum
                
            # Update particles and velocities
            particles[i] = h
            velocities[i] = v
            
        # Decrease temperature for simulated annealing
        temperature = max(temperature * cooling_rate, 0.01)
    
    # Final evaluation
    corr = correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (global_best.tolist(), final_c5, n_points)
