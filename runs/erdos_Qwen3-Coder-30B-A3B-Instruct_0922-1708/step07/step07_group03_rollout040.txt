import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize particles
    S = 50  # number of particles
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    
    # PSO particles and velocities
    particles = np.random.rand(S, n_points)
    velocities = np.random.randn(S, n_points) * 0.1
    
    # Scale particles to meet constraint
    for i in range(S):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Initialize personal bests
    pbest = particles.copy()
    pbest_fitness = np.full(S, float('inf'))
    
    # Initialize global best
    gbest_idx = 0
    gbest_fitness = float('inf')
    
    # If initial_h_values available, use it for one particle
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles[0] = h_init
        pbest[0] = h_init
    
    # Evaluate initial particles
    for i in range(S):
        h = particles[i]
        corr = correlate(h, 1 - h, mode='full')
        fitness = np.max(corr) * dx
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest_idx = i
    
    # Main PSO loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        for i in range(S):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (particles[gbest_idx] - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            h = particles[i]
            corr = correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest[i] = h.copy()
                
                # Update global best if needed
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_idx = i
    
    # Return best solution found
    best_h = particles[gbest_idx]
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
