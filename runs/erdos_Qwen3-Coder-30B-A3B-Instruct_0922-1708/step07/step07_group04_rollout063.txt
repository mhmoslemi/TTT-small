import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, rate=0.1, step=0.1):
    """Mutate h by randomly perturbing some entries."""
    h_new = h.copy()
    mask = np.random.rand(len(h_new)) < rate
    h_new[mask] += np.random.uniform(-step, step, np.sum(mask))
    return repair_h(h_new, len(h))

def crossover_h(h1, h2):
    """Crossover two individuals by segment shuffling."""
    n = len(h1)
    cut1, cut2 = sorted(np.random.choice(n, 2, replace=False))
    h_new = h1.copy()
    h_new[cut1:cut2] = h2[cut1:cut2]
    return repair_h(h_new, n)

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 100
    elite_size = 10
    mutation_rate = 0.1
    mutation_step = 0.1
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = mutate_h(h_init, mutation_rate, mutation_step)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fitness.append(-evaluate_h(h, dx))
    
    # Evolution loop
    best_fitness = min(fitness)
    best_individual = population[np.argmin(fitness)].copy()
    patience_counter = 0
    best_gen = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Sort by fitness (lower is better)
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Update best solution
        if fitness[0] < best_fitness:
            best_fitness = fitness[0]
            best_individual = population[0].copy()
            best_gen = gen
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Selection - keep top 50% as parents
        parents = population[:population_size//2]
        
        # Create new generation
        new_population = parents[:elite_size]  # Elitism
        
        # Generate offspring through crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(len(parents), 2, replace=False)
            parent1 = parents[idx1] if fitness[idx1] < fitness[idx2] else parents[idx2]
            idx3, idx4 = np.random.choice(len(parents), 2, replace=False)
            parent2 = parents[idx3] if fitness[idx3] < fitness[idx4] else parents[idx4]
            
            # Crossover
            child = crossover_h(parent1, parent2)
            
            # Mutation
            child = mutate_h(child, mutation_rate, mutation_step)
            
            new_population.append(child)
        
        population = new_population
        
        # Evaluate new population
        fitness = []

    # Final gradient refinement on best individual
    h_refined = best_individual.copy()
    c5_refined = -best_fitness
    eta = 0.001

    last_improvement = 0

            h_plus = h_refined.copy()

            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_refined.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_refined[j] - eta * grad_j
        
        # Apply repair step to maintain feasibility
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_refined:
            h_refined = h_new
            c5_refined = c5_new
            last_improvement = i
        else:
            # Occasionally accept worse solutions to escape local minima
            if np.random.rand() < 0.01:
                h_refined = h_new
                c5_refined = c5_new
                last_improvement = i
                
        # Decrease learning rate
        if i - last_improvement > 100:
            eta *= 0.95
            last_improvement = i
    
    return (h_refined.tolist(), c5_refined, n_points)
