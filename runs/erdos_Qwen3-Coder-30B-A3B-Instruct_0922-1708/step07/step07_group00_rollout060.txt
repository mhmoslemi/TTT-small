import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Optimization using scipy minimize (better than gradient descent)
    def objective(h):
        return evaluate_h(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try optimization using different methods
    methods = ['SLSQP', 'trust-constr']
    for method in methods:
        if time.time() - start_time > budget_s - 10:
            break
            
        try:
            result = minimize(objective, current_h, method=method, 
                            bounds=bounds, constraints=cons, 
                            options={'maxiter': 1000, 'ftol': 1e-8})
            if result.success:
                optimized_h = repair_h(result.x, n_points)
                optimized_c5 = evaluate_h(optimized_h, dx)
                if optimized_c5 < best_c5:
                    best_h = optimized_h
                    best_c5 = optimized_c5
        except Exception:
            pass
    
    # Final refinement with local search around best solution
    if time.time() - start_time < budget_s - 10:
        # Perturb best solution and try again
        for _ in range(50):
            if time.time() - start_time > budget_s - 10:
                break
                
            h_pert = best_h + np.random.normal(0, 0.001, n_points)
            h_pert = repair_h(h_pert, n_points)
            c5_pert = evaluate_h(h_pert, dx)
            if c5_pert < best_c5:
                best_h = h_pert
                best_c5 = c5_pert
    
    return (best_h.tolist(), best_c5, n_points)
