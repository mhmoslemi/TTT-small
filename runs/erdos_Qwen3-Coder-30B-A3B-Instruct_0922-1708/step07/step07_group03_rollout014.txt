import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Start with a binary construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        h = h * (n_points / 2) / np.sum(h)
    
    # Annealing parameters
    temp = 0.1
    cooling_rate = 0.995
    min_temp = 1e-6
    
    # Evaluate initial solution
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Generate neighbor solution
        h_new = h.copy()
        
        # Pick two random indices to perturb
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Random change within feasible region
        delta = np.random.uniform(-0.05, 0.05)
        delta_i = min(delta, 1.0 - h_new[i])
        delta_j = max(-delta_i, -h_new[j])
        
        h_new[i] += delta_i
        h_new[j] += delta_j
        
        # Maintain sum constraint
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            # Adjust all points to maintain sum
            h_new = h_new * target_sum / h_sum
        
        # Compute new overlap
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject with simulated annealing
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
            h = h_new
            current_c5 = new_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Cooling
        temp = max(temp * cooling_rate, min_temp)
        
    return (best_h.tolist(), best_c5, n_points)
