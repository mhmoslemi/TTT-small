import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize particles
    S = 50  # swarm size
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    T = 1.0   # initial temperature
    cooling_rate = 0.999
    
    # Initialize positions and velocities
    particles = np.random.rand(S, n_points)
    velocities = np.random.randn(S, n_points) * 0.1
    
    # Scale particles to meet constraint
    for i in range(S):
        particles[i] *= (n_points / 2) / np.sum(particles[i])
    
    # Initialize personal bests
    personal_best_positions = particles.copy()
    personal_best_values = np.full(S, np.inf)
    
    # Initialize global best
    global_best_position = None
    global_best_value = np.inf
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        # Set first particle to this initial solution
        particles[0] = h_init * (n_points / 2) / np.sum(h_init)
        personal_best_positions[0] = particles[0].copy()
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate all particles
        for i in range(S):
            h = particles[i]
            
            # Compute correlation
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < personal_best_values[i]:
                personal_best_values[i] = c5
                personal_best_positions[i] = h.copy()
            
            # Update global best
            if c5 < global_best_value:
                global_best_value = c5
                global_best_position = h.copy()
        
        # Update velocities and positions
        for i in range(S):
            r1, r2 = np.random.rand(2)
            
            # Velocity update
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best_positions[i] - particles[i]) + 
                           c2 * r2 * (global_best_position - particles[i]))
            
            # Apply simulated annealing: perturb with probability T
            if np.random.rand() < T:
                velocities[i] += np.random.uniform(-0.1, 0.1, n_points)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair constraints: clip values and rescale
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (n_points / 2) / np.sum(particles[i])
            
        # Update temperature
        T = max(T * cooling_rate, 0.01)
    
    # Final evaluation of global best
    if global_best_position is None:
        # Fallback to random initialization if nothing was found
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1 - h, mode='full')
        final_c5 = np.max(corr) * dx
        best_h = h
    else:
        corr = correlate(global_best_position, 1 - global_best_position, mode='full')
        final_c5 = np.max(corr) * dx
        best_h = global_best_position
    
    return (best_h.tolist(), final_c5, n_points)
