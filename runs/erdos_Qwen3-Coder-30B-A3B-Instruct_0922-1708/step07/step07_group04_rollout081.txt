import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Phase 1: Convex optimization
    # Define the optimization problem using CVXPy
    h_var = cp.Variable(n_points)
    h_lb = np.zeros(n_points)
    h_ub = np.ones(n_points)
    
    # Constraints
    constraints = [
        h_var >= h_lb,
        h_var <= h_ub,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Quadratic approximation of max overlap using squared norm
    # For given h, the correlation matrix is standardized and we minimize 
    # squared l2 norm to avoid non-convex max
    corr_expr = cp.Variable(2 * n_points - 1)
    # Add linear constraints relating h_var and corr_expr to make it
    # a QP, by using approximate of max term.
    
    # Construct approximate convex problem using auxiliary variables
    # We use a grouped approach, using max of correlation elements
    # by introducing auxiliary variables for each correlation value
    corr_values = [cp.Variable() for _ in range(2 * n_points - 1)]
    
    # Build simplified 2nd order relaxation by using proximal steps around convexification
    # This approach alternates between convex QP and local refinement
    
    # Convex part with weighted penalty on correlations
    # Convex optimization approach with CVXPy with a fixed number of iterations
    quad_approx = 0
    for i in range(2 * n_points - 1):
        corr_values[i] = cp.Variable()
        quad_approx += corr_values[i]**2
    
    # Objectives are modeled via constraint
    # This is non-convex in general, so we'll use 
    # an approximate convex problem using the method of
    # majorization/minimization
    
    # For now, let's just perform a hybrid of proximal steps and oscilation
    # throughout the 2nd phase: getting constructive ultra-optimization
    
    # Begin hybrid optimization
    # Use CVX optimization for a few steps to get close
    
    max_iter = 5000
    patience = 100
    last_improvement = 0
    eta = 0.01
    eps = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Do gradient descent with finite difference step for order-2 analyzer
        h_new = h_current.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_current[j] - eta * grad_j
            
        # Apply repair step to maintain feasibility
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if better or at random with small probability
        if c5_new < c5_current or np.random.rand() < 0.01:
            h_current = h_new
            c5_current = c5_new
            
            # Update best if improved
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        
        # Adjust learning rate
        if i - last_improvement > patience:
            eta *= 0.95
            last_improvement = i
    
    # Final evaluation and return
    final_c5 = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), final_c5, n_points)
