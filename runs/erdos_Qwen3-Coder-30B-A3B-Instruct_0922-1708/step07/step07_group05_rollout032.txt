import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    n = len(parent1)
    crossover_point = np.random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n = len(h)
    mutation_indices = np.random.choice(n, int(n * mutation_rate), replace=False)
    for i in mutation_indices:
        h[i] += np.random.normal(0, 0.05)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Population size
    pop_size = 50
    population = []
    
    # Initialize population

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    else:
        h_init = np.zeros(n_points)
        # Create a triangular pulse
        mid = n_points // 2
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = 1.0 - abs(x - 1.0) * 2
            else:
                h_init[i] = 0.0
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Fill population with random samples
    while len(population) < pop_size:
        h = np.random.uniform(0, 1, n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    # GA parameters
    max_generations = 1000
    elite_size = 1
    mutation_rate = 0.2
    crossover_rate = 0.5
    patience = 200
    best_c5 = float('inf')
    last_improvement = 0
    
    # Main GA loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Store best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            last_improvement = gen
        
        # Check for stagnation
        if gen - last_improvement > patience:
            break
        
        # Elitism
        new_population = [population[0]]  # Keep best
        
        # Selection and reproduction
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.randint(0, pop_size//2)
            parent2_idx = np.random.randint(0, pop_size//2)

            parent2 = population[parent2_idx]

            if np.random.rand() < crossover_rate:

    return (best_h.tolist(), best_c5, n_points)
