import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Phase 1: Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Phase 2: CVX optimization
    h = cp.Variable(n_points)
    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
    
    # Objective: minimize the maximum overlap term via quadratic approximation
    # We do not solve for the true max directly but approximate with squared l2 norm
    # to maintain convexity
    # Formulate a convex approximation
    corr = correlate(h.value, 1 - h.value, mode='full')
    # Use a weighted sum to approximate max value
    # Constant weights to keep the problem convex
    weights = np.ones_like(corr)
    objective = cp.Minimize(cp.sum_squares(corr) * dx)
    
    # Use CVXOPT solver with strict iteration limit
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=5000)
    
    h_current = np.array(h.value).flatten()
    h_current = repair_h(h_current, n_points)
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Phase 3: Post-optimization with proximal gradient descent
    eta = 0.01
    eps = 1e-6
    max_iter = 2000
    patience = 100
    last_improvement = 0
    prev_c5 = c5_current
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Proximal gradient descent step
        h_new = h_current.copy()
        
        # Approximate gradient w.r.t. each element using finite differences
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Proximal update
            h_new[j] = h_current[j] - eta * grad_j
            
        # Apply repair
        h_new = repair_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if better
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        elif c5_new < prev_c5:
            # Accept with small probability
            if np.random.rand() < 0.001:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
        
        # Update learning rate if no improvement for a while
        if i - last_improvement > patience:
            eta *= 0.95
            last_improvement = i

        prev_c5 = c5_current

    return (h_best.tolist(), c5_best, n_points)
