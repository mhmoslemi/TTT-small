import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a piecewise constant construction
        h_init = np.zeros(n_points)
        # Create a two-part construction for sparsity and structure
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.5
        h_init = project_h(h_init, n_points)
    
    # Multi-start optimization
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Try multiple starting points
    initial_configs = []
    
    # Configuration 1: Truncated cosine
    x = np.linspace(0, 2, n_points)
    h1 = 0.5 * (1 + np.cos(np.pi * x))  
    h1 = project_h(h1, n_points)
    initial_configs.append(h1)
    
    # Configuration 2: Piecewise with step
    h2 = np.zeros(n_points)
    h2[:n_points//3] = 0.6
    h2[n_points//3:2*n_points//3] = 0.4
    h2[2*n_points//3:] = 0.3
    h2 = project_h(h2, n_points)
    initial_configs.append(h2)
    
    # Configuration 3: Random perturbation of initial
    h3 = h_init + np.random.normal(0, 0.05, n_points)
    h3 = project_h(h3, n_points)
    initial_configs.append(h3)

    # Configuration 4: Constant plus noise
    h4 = 0.5 * np.ones(n_points) + np.random.normal(0, 0.05, n_points)
    h4 = project_h(h4, n_points)
    initial_configs.append(h4)
    
    # Optimization loop with multiple starts
    max_iter_per_start = 5000
    patience = 200
    for it, h_init in enumerate(initial_configs):
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        # Exact gradient computation
        learning_rate = 0.02
        last_improvement = 0
        
        for i in range(max_iter_per_start):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Find the shift that maximizes overlap
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr)
            max_shift = max_shift_idx - (n_points - 1)  # Center of the full correlation
            
            # Compute exact gradient with respect to the maximum shift
            # This is the gradient for the shift that boosts the maximum overlap
            h_shifted = np.roll(h_current, -max_shift)
            grad = 1 - h_shifted
            
            # Update with gradient step
            h_new = h_current + learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept improvement or occasionally accept worse moves
            if c5_new < c5_current or np.random.rand() < 0.01:
                h_current = h_new.copy()
                c5_current = c5_new
            else:
                # Reduce learning rate if no progress
                if i - last_improvement > patience:
                    learning_rate *= 0.95
                    last_improvement = i
            
            # Track best
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                last_improvement = i
                
            # Heavy perturbation occasionally to escape local minima
            if i % 100 == 0:
                if np.random.rand() < 0.3:
                    delta = np.random.normal(0, 0.03, n_points)
                    h_current = project_h(h_current + delta, n_points)
                    c5_current = evaluate_h(h_current, dx)
            
    # Final intensification with local search
    h_best = best_h.copy()
    c5_best = best_c5
    
    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_current = h_best.copy()
        # Perturb about 10% of values
        indices = np.random.choice(n_points, size=n_points//10, replace=False)
        for idx in indices:
            h_current[idx] += np.random.normal(0, 0.01)
        
        h_current = project_h(h_current, n_points)
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < c5_best:
            h_best = h_current
            c5_best = c5_current
    
    return (h_best.tolist(), c5_best, n_points)
