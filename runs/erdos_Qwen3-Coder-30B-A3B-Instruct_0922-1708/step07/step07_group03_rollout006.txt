import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization
        h = np.ones(n_points) * 0.5
    
    # Frank-Wolfe algorithm
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate current C5
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient approximation via finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Find descent direction by solving linear subproblem
        # Minimize grad^T * s subject to 0 <= s <= 1, sum(s) = n_points/2
        # This is equivalent to selecting s with minimal gradient value
        
        # Simple approach: find s = argmin_i grad_i, subject to constraints
        # For a simpler direction, we'll set s to 1 at index that minimizes grad
        # and 0 elsewhere, then scale to meet constraint
        s = np.zeros(n_points)
        idx = np.argmin(grad)
        s[idx] = 1.0
        
        # Scale to meet sum constraint
        current_sum = np.sum(s)
        if current_sum > 0:
            s = s * (n_points / 2) / current_sum
        else:
            s = np.ones(n_points) * (n_points / 2) / n_points
            
        # Ensure s stays within bounds
        s = np.clip(s, 0, 1)
        
        # Line search for step size gamma
        gamma = 0.5
        max_backtrack = 10
        for _ in range(max_backtrack):
            new_h = (1 - gamma) * h + gamma * s
            # Normalize to sum constraint
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_h = np.clip(new_h, 0, 1)
            
            corr_new = correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(corr_new) * dx
            
            if new_c5 < current_c5:
                h = new_h
                break
            gamma *= 0.5
            
        # Maintain feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
