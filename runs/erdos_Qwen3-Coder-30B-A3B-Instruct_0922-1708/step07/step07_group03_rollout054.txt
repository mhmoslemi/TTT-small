import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter_barrier = 1000
    max_iter_gd = 10000
    learning_rate = 1e-3
    mu_initial = 1e3
    barrier_decrease_factor = 0.99
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
        
    # Helper function to compute gradient via adjoint method
    def compute_gradient(h, max_lags):
        grad = np.zeros_like(h)
        for lag in max_lags:
            if lag >= 0:
                # Shift 1-h backward
                shifted = np.concatenate([np.zeros(lag), 1 - h[:-lag]])
            else:
                # Shift 1-h forward
                shifted = np.concatenate([1 - h[-lag:], np.zeros(-lag)])
            grad += shifted
        return grad * dx
    
    # Helper function for projection
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2) > 1e-10:
            h = h * (n_points / 2) / h_sum
        return h
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Barrier method phase
    mu = mu_initial
    for iter_barrier in range(max_iter_barrier):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_lags = np.where(corr == max_corr)[0]
        # Convert indices to actual lag values
        max_lags = max_lags - (len(h) - 1)
        
        c5 = max_corr * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient
        grad = compute_gradient(h, max_lags)
        
        # Update step with barrier
        h = h - learning_rate * grad - mu * (h - np.clip(h, 0, 1))
        
        # Project
        h = project_h(h)
        
        # Update barrier parameter
        mu *= barrier_decrease_factor
    
    # Gradient descent phase
    for iter_gd in range(max_iter_gd):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        max_lags = np.where(corr == max_corr)[0]
        # Convert indices to actual lag values
        max_lags = max_lags - (len(h) - 1)
        
        c5 = max_corr * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute exact gradient via adjoint
        grad = compute_gradient(h, max_lags)
        
        # Update step
        h = h - learning_rate * grad
        
        # Project to feasible set
        h = project_h(h)
    
    # Final check
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
