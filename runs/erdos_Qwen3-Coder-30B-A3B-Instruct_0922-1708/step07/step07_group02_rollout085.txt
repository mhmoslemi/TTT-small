import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_population(n_points, n_pop, initial_h=None):
    """Initialize a diverse population of feasible solutions."""
    population = []
    if initial_h is not None:
        population.append(initial_h.copy())
        n_pop -= 1
    
    # Generate diverse candidates
    for _ in range(n_pop):
        # Random walk from a base solution
        h = np.random.rand(n_points)
        h = project_h(h, n_points, 2.0 / n_points)
        population.append(h)
    
    return population

def segment_blend(parent1, parent2, alpha=0.5):
    """Blend two parents using segment-wise crossover."""
    n = len(parent1)
    # Cut at random points
    cuts = sorted(np.random.choice(n-2, 2, replace=False) + 1)
    cut1, cut2 = cuts[0], cuts[1]
    
    child = parent1.copy()
    # Swap segments
    child[cut1:cut2] = parent2[cut1:cut2]
    
    return child

def mutate_individual(h, mutation_rate=0.3, noise_scale=0.1):
    """Apply mutation to an individual."""
    n = len(h)
    h_mut = h.copy()
    
    # Apply segment flip mutations
    if np.random.rand() < mutation_rate:
        cut1 = np.random.randint(0, n)
        cut2 = np.random.randint(cut1, n)
        h_mut[cut1:cut2] = 1 - h_mut[cut1:cut2]
    
    # Add Gaussian noise
    if np.random.rand() < mutation_rate:
        noise = np.random.normal(0, noise_scale, n)
        h_mut += noise
        h_mut = np.clip(h_mut, 0, 1)
    
    return h_mut

def local_refinement(h, dx, max_iter=50, lr=0.01, eps=1e-6):
    """Fast local refinement using coordinate descent."""
    h_refined = h.copy()
    for _ in range(max_iter):
        h_new = h_refined.copy()
        improved = False
        
        # Randomize order
        indices = list(range(len(h_refined)))
        np.random.shuffle(indices)
        
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            # Ensure feasibility
            h_plus = project_h(h_plus, len(h_plus), dx)
            h_minus = project_h(h_minus, len(h_minus), dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            h_new[j] -= lr * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
        
        h_new = project_h(h_new, len(h_new), dx)
        c5_new = evaluate_h(h_new, dx)
        c5_old = evaluate_h(h_refined, dx)
        
        if c5_new < c5_old:
            h_refined = h_new
            improved = True
        else:
            lr *= 0.995
            if lr < 1e-6:
                break
                
    return h_refined

    dx = 2.0 / n_points
    start_time = time.time()

        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    # Memetic algorithm parameters
    n_pop = 50
    n_select = n_pop // 5
    mutation_rate = 0.3
    noise_scale = 0.1
    max_generations = 1500
    
    # Initialize population
    population = initialize_population(n_points, n_pop, h_init)
    
    # Best solution tracking
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Main memetic loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate population
        fitness_scores = [evaluate_h(h, dx) for h in population]
        
        # Track best
        best_idx = np.argmin(fitness_scores)
        if fitness_scores[best_idx] < c5_best:
            h_best = population[best_idx].copy()
            c5_best = fitness_scores[best_idx]
        
        # Selection: keep top individuals
        selected_indices = np.argsort(fitness_scores)[:n_select]
        selected_pop = [population[i] for i in selected_indices]
        
        # Generate offspring
        new_population = selected_pop[:]  # Keep top individuals
        
        # Create offspring
        while len(new_population) < n_pop:
            parent1 = np.random.choice(selected_pop)
            parent2 = np.random.choice(selected_pop)
            child = segment_blend(parent1, parent2)
            child = mutate_individual(child, mutation_rate, noise_scale)
            child = project_h(child, n_points, dx)
            # Local refinement
            child_refined = local_refinement(child, dx)
            new_population.append(child_refined)
            
        population = new_population[:n_pop]
        
        # Adaptive mutation rate
        noise_scale *= 0.999
        if noise_scale < 1e-6:
            noise_scale = 1e-6
    
    # Final evaluation of best solution
    c5_best = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), c5_best, n_points)
