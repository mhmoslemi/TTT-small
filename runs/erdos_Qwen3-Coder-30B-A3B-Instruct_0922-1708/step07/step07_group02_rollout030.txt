import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate a solution with adaptive mutation and maintain feasibility."""
    mutated = h.copy()
    n = len(h)
    
    # Random segment flip (30% chance)
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
        
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)

    # ES parameters
    population_size = 50
    generation = 0
    mutation_scale = 0.1
    patience = 100
    last_improvement = 0
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = mutate(h_init, 0, 0.1)
        population.append(h)
    
    while time.time() - start_time < budget_s - 5:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort and select top 20%
        indices = np.argsort(fitness)
        top_indices = indices[:max(1, population_size // 5)]
        
        # Update best solution
        if fitness[indices[0]] < best_c5:
            best_c5 = fitness[indices[0]]
            best_h = population[indices[0]].copy()
            last_improvement = generation
        
        # Create new population through mutation and recombination
        new_population = []
        
        # Keep top performers
        for i in top_indices:
            new_population.append(population[i].copy())
        
        # Add new candidates
        while len(new_population) < population_size:
            # Select two parents
            parent1 = population[np.random.choice(top_indices)]
            parent2 = population[np.random.choice(top_indices)]
            
            # Recombine (uniform crossover)
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            
            # Mutate
            child = mutate(child, generation, mutation_scale)
            new_population.append(child)
        
        # Update population
        population = new_population
        
        # Adaptive mutation
        if generation - last_improvement > patience:
            mutation_scale *= 0.9
            # Inject random variation
            for i in range(len(population)):
                if np.random.rand() < 0.1:
                    population[i] = mutate(population[i], generation, 0.05)
        else:
            mutation_scale *= 0.999
        
        mutation_scale = max(0.001, mutation_scale)
    
    return (best_h.tolist(), best_c5, n_points)
