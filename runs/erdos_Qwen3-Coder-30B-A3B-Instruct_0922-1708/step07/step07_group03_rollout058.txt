import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random binary initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe algorithm
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient (directional derivative approximation)
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
            
        # Solve linear subproblem: minimize grad^T s subject to constraints
        # Simple greedy approach: set s to be 1 in positions where grad is most negative
        # and 0 elsewhere, then normalize
        s = np.zeros(n_points)
        sorted_indices = np.argsort(grad)
        # Select points to set to 1, ensuring sum constraint
        target_sum = n_points / 2
        s[sorted_indices[:int(target_sum)]] = 1.0
        
        # Normalize s to meet the sum constraint
        s_sum = np.sum(s)
        if s_sum > 0:
            s = s * target_sum / s_sum
            
        # Ensure s is in [0,1]
        s = np.clip(s, 0, 1)
        
        # Line search for step size
        gamma = 0.5
        max_backtrack = 20
        for _ in range(max_backtrack):
            h_new = (1 - gamma) * h + gamma * s
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new <= current_c5:
                h = h_new
                break
            gamma *= 0.5
        
        # Ensure h is feasible
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
