import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize swarm
    S = 50
    swarm = []
    velocities = []
    
    # Initialize particles
    for _ in range(S):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h)
        else:
            h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)  # Small random velocities
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # SA parameters
    T = 1.0
    cooling_rate = 0.999
    
    # Personal bests
    personal_best = [swarm[i].copy() for i in range(S)]
    personal_best_fitness = [float('inf')] * S
    
    # Global best
    global_best = swarm[0].copy()
    global_best_fitness = float('inf')
    
    # Track best overall
    best_h = global_best.copy()
    best_c5 = global_best_fitness
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Update each particle
        for i in range(S):
            h = swarm[i]
            v = velocities[i]
            
            # Compute fitness
            corr = correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = h.copy()
                personal_best_fitness[i] = fitness
                
            # Update global best
            if fitness < global_best_fitness:
                global_best = h.copy()
                global_best_fitness = fitness
                
                # Update best overall
                if fitness < best_c5:
                    best_h = h.copy()
                    best_c5 = fitness
            
            # PSO velocity update
            r1, r2 = np.random.rand(2)
            v = w * v + c1 * r1 * (personal_best[i] - h) + c2 * r2 * (global_best - h)
            
            # Apply SA with some probability
            if np.random.rand() < T:
                v += np.random.rand(n_points) * 0.1 - 0.05
                
            # Update position
            h_new = h + v
            
            # Repair constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update swarm and velocity
            swarm[i] = h_new
            velocities[i] = v
            
        # Decrease temperature for SA
        T = max(T * cooling_rate, 0.01)
    
    # Final correction to make sure constraint is satisfied
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
