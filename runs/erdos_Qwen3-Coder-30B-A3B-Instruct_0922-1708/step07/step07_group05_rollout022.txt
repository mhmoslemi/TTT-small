import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidate(n_points, seed=None):
    if seed is not None:
        np.random.seed(seed)
    # Generate a candidate using sum of cos/sin with random coefficients
    # to create diverse function shapes that satisfy constraints
    
    # Generate random coefficients for basis functions
    # We'll use 3-5 cosine/sine basis functions 
    n_basis = np.random.randint(3, 6)
    
    # Create candidate with random phases and amplitudes
    h = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    
    # Add several basis functions to create rich shape
    for _ in range(n_basis):
        k = np.random.uniform(1, 10)  # frequency
        a = np.random.uniform(0, 0.5)  # amplitude  
        b = np.random.uniform(-0.5, 0.5)  # phase
        h += a * np.cos(k * x + b)
    
    # Add a constant bias term to make it fit constraints
    h += 0.5
    h = repair_h(h, n_points)
    return h

    start_time = time.time()

        h_init = repair_h(h_init, n_points)
    else:
        h_init = np.full(n_points, 0.5)
        
    h_best = h_init.copy()
    c5_best = evaluate_h(h_init, dx)
    
    # Generate diverse candidates using structured sampling
    num_candidates = 50
    candidates = []
    
    candidates.append(h_init)
    for i in range(num_candidates - 1):
        h_cand = generate_fourier_candidate(n_points, seed=seed + i)
        candidates.append(h_cand)
    
    # Evaluate all candidates first
    candidate_scores = []
    for h_cand in candidates:
        c5 = evaluate_h(h_cand, dx)
        candidate_scores.append((c5, h_cand))
    
    # Sort with best first
    candidate_scores.sort(key=lambda x: x[0])
    
    # Refine top candidates via gradient descent
    patience = 200
    for i, (c5_score, h_curr) in enumerate(candidate_scores[:10]):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_current = h_curr.copy()
        c5_current = c5_score
        
        # Refinement loop
        for step in range(2000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Gradient estimation
            epsilon = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_current.copy()

                h_plus[j] += epsilon
