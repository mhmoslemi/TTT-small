import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize swarm
    S = 50
    particles = np.random.rand(S, n_points)
    velocities = np.random.randn(S, n_points) * 0.1
    
    # Scale initial particles
    for i in range(S):
        particles[i] *= (n_points / 2) / np.sum(particles[i])
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        particles[0] = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # SA parameters
    T = 1.0
    cooling_rate = 0.999
    
    # Initialize personal best and global best
    personal_best = particles.copy()
    personal_best_fitness = np.full(S, float('inf'))
    global_best_idx = 0
    global_best_fitness = float('inf')
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate all particles
        for i in range(S):
            h = particles[i]
            corr = correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best[i] = h.copy()
                
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best_idx = i
                
        # Update velocities and positions
        for i in range(S):
            r1, r2 = np.random.rand(2)
            pbest = personal_best[i]
            gbest = personal_best[global_best_idx]
            
            # Velocity update
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Apply Simulated Annealing with small probability
            if np.random.rand() < T:
                velocities[i] += np.random.rand(n_points) * 0.2 - 0.1
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (n_points / 2) / np.sum(particles[i])
            
        # Cool temperature
        T = max(T * cooling_rate, 0.01)
    
    # Return best result
    best_h = personal_best[global_best_idx]
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
