import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # PSO parameters
    swarm_size = 50
    max_iter = 100000
    w_start = 0.9
    w_end = 0.4
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # One particle from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        # Initialize with structured feasible solution 
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    
    # Fill remaining particles
    while len(particles) < swarm_size:
        # Random initialized particles with diversity
        h_random = np.random.rand(n_points)
        h_random = project_h(h_random, n_points, dx)
        particles.append(h_random.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Convert to arrays for easier computation
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # Track best particles
    personal_best = particles.copy()
    personal_best_fitness = np.array([evaluate_h(p, dx) for p in particles])
    
    # Find global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO main loop
    for iter_idx in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iter_idx / max_iter)
        
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = project_h(particles[i], n_points, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best if needed
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
        
        # Occasionally, add stochastic perturbation around best to escape local optima
        if iter_idx % 1000 == 0 and iter_idx > 0:
            perturb_idx = np.argmin(personal_best_fitness)
            perturb = personal_best[perturb_idx].copy()
            noise = np.random.normal(0, 0.01, n_points)
            perturb += noise
            perturb = project_h(perturb, n_points, dx)
            perturb_fitness = evaluate_h(perturb, dx)
            if perturb_fitness < global_best_fitness:
                global_best = perturb
                global_best_fitness = perturb_fitness
    
    # Return best solution found
    return (global_best.tolist(), global_best_fitness, n_points)
