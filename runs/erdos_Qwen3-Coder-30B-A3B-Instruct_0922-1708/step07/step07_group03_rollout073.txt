import time
import cvxpy as cp
from scipy.signal import correlate

        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create a simple initial guess
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Prepare for solving the optimization problem
    h_vars = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints for feasible h
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
    
    # Add quadratic constraints
    for lag in range(n_points):
        # We compute correlation based on discrete convolution
        # To handle this as quadratic constraints, we'll proceed with 
        # computing at each lag separately
        corr_lag = cp.Variable(1)
        # Use a trick to convert quadratic constraints to form acceptable by CVXPY
        # For large dimension, it might be easier to use a different approach
        # Let's try solving at a few lags as approximation
        
    # Simplified approach: use non-convex iterative optimization to find a better solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Start with a better search strategy
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Try different starting points and local optimization
    for trial in range(10):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Initialize with random
        h_trial = np.random.rand(n_points)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Some local optimization with gradient descent
        lr = 1e-3
        for iter in range(2000):
            if (time.time() - start_time) > budget_s - 2:
                break
                
            c5_current = compute_c5(h_trial)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_trial.copy()
            
            # Compute gradient via finite differences
            grad = np.zeros(n_points)
            eps = 1e-6
            for i in range(n_points):
                h_plus = h_trial.copy()
                h_plus[i] += eps
                if h_plus[i] > 1:
                    h_plus[i] = 1
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)

                grad[i] = (c5_plus - c5_current) / eps
            
            # Update step with clipping and normalization

        # End trial
    
    # Final check
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
