import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, rate=0.1, magnitude=0.1):
    """Mutate h by perturbing a fraction of entries."""
    h_new = h.copy()
    n_points = len(h)
    n_mutate = int(n_points * rate)
    indices = np.random.choice(n_points, n_mutate, replace=False)
    for i in indices:
        h_new[i] += np.random.normal(0, magnitude)
    return repair_h(h_new, n_points)

def crossover_h(h1, h2):
    """Perform segment crossover between two h vectors."""
    n_points = len(h1)
    split = np.random.randint(1, n_points)
    h_new = np.concatenate([h1[:split], h2[split:]])
    return repair_h(h_new, n_points)

def genetic_algorithm(seed, budget_s, n_points=100):
    np.random.seed(seed)
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Create initial population
    population = [h_init]
    for _ in range(population_size - 1):
        h = mutate_h(h_init, rate=0.1, magnitude=0.1)
        population.append(h)
    
    best_individual = h_init
    best_fitness = evaluate_h(h_init, dx)
    
    # Evolution loop
    generation = 0
    max_generations = 100
    patience = 20
    last_improvement = 0
    
    while generation < max_generations and time.time() - start_time < budget_s - 5:
        generation += 1
        
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fitness = -evaluate_h(h, dx)  # Negative because we minimize
            fitnesses.append(fitness)
        
        # Selection: Tournament selection
        new_population = []
        for _ in range(population_size):
            # Tournament
            tournament_size = 3
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmax(tournament_fitnesses)]
            new_population.append(population[winner_index].copy())
        
        # Crossover and mutation
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = new_population[i]
                parent2 = new_population[i+1]
                child1 = crossover_h(parent1, parent2)
                child2 = crossover_h(parent2, parent1)
                child1 = mutate_h(child1, rate=0.1, magnitude=0.05)
                child2 = mutate_h(child2, rate=0.1, magnitude=0.05)
                new_population[i] = child1
                new_population[i+1] = child2
            else:
                # Odd population size, mutate the last one
                new_population[i] = mutate_h(new_population[i], rate=0.1, magnitude=0.05)
        
        population = new_population
        
        # Track best individual
        for h in population:
            fitness = -evaluate_h(h, dx)
            if fitness < best_fitness:
                best_fitness = fitness
                best_individual = h.copy()
                last_improvement = generation
        
        # Early stopping if no improvement
        if generation - last_improvement > patience:
            break
    
    # Gradient refinement on best individual
    h_refined = best_individual.copy()
    c5_refined = evaluate_h(h_refined, dx)
    max_iter_refine = 5000
    eta = 0.001
    eps = 1e-6
    
    for i in range(max_iter_refine):
        if time.time() - start_time > budget_s - 2:
            break
        h_new = h_refined.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_refined.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_refined[j] - eta * grad_j
        
        # Apply repair step
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_refined:
            h_refined = h_new
            c5_refined = c5_new
        else:
            eta *= 0.99  # Decrease learning rate

    return (h_refined.tolist(), c5_refined, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return genetic_algorithm(seed, budget_s)
