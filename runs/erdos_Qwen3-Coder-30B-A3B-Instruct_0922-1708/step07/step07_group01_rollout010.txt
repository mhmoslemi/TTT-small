import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Generate diverse initializations
    initializations = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
        initializations.append(h_init)
    
    # Add structured waveforms
    x = np.linspace(0, 2, n_points)
    
    # Sine wave
    h_sine = 0.5 * (1 + np.sin(np.pi * x))
    h_sine = project_h(h_sine, n_points)
    initializations.append(h_sine)
    
    # Trapezoidal
    h_trap = np.zeros(n_points)
    mid = n_points // 2
    h_trap[mid-10:mid+10] = 1.0
    h_trap = project_h(h_trap, n_points)
    initializations.append(h_trap)
    
    # Cosine taper
    h_cos = 0.5 * (1 + np.cos(np.pi * x))
    h_cos = project_h(h_cos, n_points)
    initializations.append(h_cos)
    
    # Random initialization
    h_rand = np.random.rand(n_points)
    h_rand = project_h(h_rand, n_points)
    initializations.append(h_rand)
    
    # Bump function
    h_bump = np.zeros(n_points)
    center = n_points // 2
    width = n_points // 10
    for i in range(max(0, center-width), min(n_points, center+width)):
        h_bump[i] = 1.0 - abs(i - center) / width
    h_bump = project_h(h_bump, n_points)
    initializations.append(h_bump)
    
    # Try a few more different random patterns
    for _ in range(3):
        h = np.random.rand(n_points)
        # Add some structure to avoid degenerate cases
        h += 0.1 * np.sin(2*np.pi*x)
        h = project_h(h, n_points)
        initializations.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Optimization with multiple restarts
    max_iter_per_restart = int(5e4)
    patience = 500
    num_restarts = len(initializations) * 20
    restarts_done = 0
    
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Select good initialization
        init_idx = restart % len(initializations)
        h_current = initializations[init_idx].copy()
        
        # Local optimization with gradient method
        c5_current = evaluate_h(h_current, dx)
        
        # Adaptive learning rate
        learning_rate = 0.01
        
        last_improvement = 0
        for it in range(max_iter_per_restart):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient at the shift that maximizes the correlation
            corr_full = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr_full)
            # The corresponding correlation value
            max_corr = corr_full[max_shift_idx]
            
            # We're minimizing max correlation, so the gradient direction is 
            # in the direction that decreases it
            # Approximate gradient: use the shift making max overlap to get gradient
            # but we can't sample exact gradients easily. Use finite differences on 
            # a subset of coordinates.
            
            # Approximate gradient by perturbing each coordinate and 
            # measuring change in overlap at max_k
            epsilon = 1e-4
            grad = np.zeros_like(h_current)
            
            # Sample approximately 1/4 of coordinates
            indices = np.random.choice(n_points, size=max(5, n_points // 4), replace=False)
            
            for idx in indices:
                h_plus = h_current.copy()
                h_minus = h_current.copy()
                
                h_plus[idx] += epsilon
                h_minus[idx] -= epsilon
                
                h_plus = project_h(h_plus, n_points)
                h_minus = project_h(h_minus, n_points)
                
                # Evaluate correlation enhancement at max shift
                c_plus = correlate(h_plus, 1 - h_plus, mode='full')[max_shift_idx]
                c_minus = correlate(h_minus, 1 - h_minus, mode='full')[max_shift_idx]
                
                grad[idx] = (c_plus - c_minus) / (2 * epsilon)
            
            # Update using gradient
            h_new = h_current - learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept better solutions or occasionally worse ones
            if c5_new <= c5_current or np.random.rand() < 0.01:
                h_current = h_new
                c5_current = c5_new
            
            # Track best
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
                last_improvement = it
            else:
                if it - last_improvement > patience:
                    # Reduce learning rate if no progress
                    learning_rate *= 0.95
                    last_improvement = it
                    
            # Sometimes jitter to escape plateaus
            if it % 100 == 0 and it > 0:
                noise = np.random.normal(0, 0.005, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
                
        restarts_done += 1
        
        # Random restart
        h_restart = np.random.rand(n_points)
        h_restart = project_h(h_restart, n_points)
        initializations.append(h_restart)
    
    if best_h is None:
        # fallback
        h = np.ones(n_points) * 0.5
        best_h = project_h(h, n_points)
        best_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
