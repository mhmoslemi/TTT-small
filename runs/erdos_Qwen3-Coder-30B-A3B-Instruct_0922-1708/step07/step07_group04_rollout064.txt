import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, mutation_rate=0.1, perturbation=0.1):
    """Mutate h by randomly perturbing some entries."""
    h_new = h.copy()
    n_points = len(h)
    indices_to_mutate = np.random.choice(n_points, 
                                         int(mutation_rate * n_points), 
                                         replace=False)
    for i in indices_to_mutate:
        h_new[i] += np.random.uniform(-perturbation, perturbation)
    return repair_h(h_new, n_points)

def crossover_h(parent1, parent2):
    """Crossover two parents by swapping segments."""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def tournament_selection(population, fitnesses, k=3):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), k, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Genetic algorithm parameters
    population_size = 100
    n_generations = 100
    mutation_rate = 0.1
    perturbation = 0.1
    elite_size = 10
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = mutate_h(h_init, mutation_rate, perturbation)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_h(h, dx) for h in population]
    
    # Track best solution
    best_idx = np.argmin(fitnesses)
    h_best = population[best_idx].copy()
    c5_best = fitnesses[best_idx]
    
    # Evolution loop
    for gen in range(n_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        top_indices = sorted_indices[:population_size // 2]
        bottom_indices = sorted_indices[population_size // 2:]
        
        # Create new population
        new_population = [population[i] for i in top_indices]
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[top_indices[i]])

            c5_current = c5_new

                c5_best = c5_current
                last_improvement = i

            h_current = h_new
            
        # Adjust learning rate based on progress
        if i - last_improvement > patience:
            eta *= 0.95
            last_improvement = i
    
    return (h_best.tolist(), c5_best, n_points)
