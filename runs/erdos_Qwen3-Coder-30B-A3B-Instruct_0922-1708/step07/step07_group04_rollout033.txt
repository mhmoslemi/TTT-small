import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Phase 1: Convex formulation to find a good starting point
    h_vars = cp.Variable(n_points)
    
    # Constraints: box constraints and sum constraint
    constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
    
    # Objective: maximize the maximum overlap (approximated with quadratic)
    h = h_init.copy()
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    
    # Approximate max with squared l2 norm of correlations
    # This is a surrogate objective for the convex solver
    h_ref = h_vars
    correlations = []
    for i in range(2 * n_points - 1):
        corr_sum = 0
        for j in range(n_points):
            if 0 <= i - j < n_points:
                corr_sum += h_ref[j] * (1 - h_ref[i - j])
        correlations.append(corr_sum)
    
    objective = cp.Minimize(cp.sum_squares(correlations) / 1e6)
    
    prob = cp.Problem(objective, constraints)
    
    # Solve with warm start from h_init
    try:
        prob.solve(solver=cp.CVXOPT, warm_start=True, verbose=False, max_iters=5000)
        if prob.status == cp.OPTIMAL:
            h_current = np.array(h_vars.value).flatten()
        else:
            h_current = h_init.copy()
    except:
        h_current = h_init.copy()
    
    h_current = repair_h(h_current, n_points)
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Phase 2: Refinement with gradient-based method
    patience = 1000
    last_improvement = 0
    eta = 0.01
    eps = 1e-6
    max_iter = 2000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Estimate gradient via finite differences
        h_new = h_current.copy()
        for j in range(n_points):
            # Perturb j-th coordinate
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_current[j] - eta * grad_j
            
        # Apply repair to restore constraints
        h_new = repair_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            # With small probability, still accept
            if np.random.rand() < 0.01:
                h_current = h_new
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
        
        # Adjust learning rate if no improvement
        if i - last_improvement > patience:
            eta *= 0.95
            last_improvement = i
    
    # Final check
    h_best = repair_h(h_best, n_points)
    c5_final = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), c5_final, n_points)
