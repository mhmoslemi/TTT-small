import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a structured construction
        h_init = np.zeros(n_points)
        # Create a piecewise linear function that is 0 on [0,0.5) and 1 on (1.5,2]
        # and slopes linearly in between
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0

        # Repair to sum to 1
        h_init = repair_h(h_init, n_points)

    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    patience = 200
    last_improvement = 0
    
    for i in range(50000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find shift k that maximizes overlap
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute gradient only for max overlap k
        grad = np.zeros(n_points)

            # Compute gradient contribution of h[j]

            shifted_h = np.roll(h_current, j - max_k)

            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[j] += 1e-6

            c5_plus = evaluate_h(h_plus, dx)
