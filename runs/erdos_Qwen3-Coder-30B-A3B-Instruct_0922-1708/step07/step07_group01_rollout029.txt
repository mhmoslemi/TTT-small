import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def get_max_shift(h):
    """Find the shift that maximizes overlap."""
    corr = correlate(h, 1 - h, mode='full')
    max_shift = np.argmax(corr)
    return max_shift

def compute_gradient(h, max_shift, n_points):
    """Compute exact gradient at max_shift."""
    grad = np.zeros(n_points)
    # Shift h by max_shift and compute gradient
    shifted = np.roll(h, max_shift)
    grad = (1 - shifted)
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient descent parameters
    learning_rate = 0.05
    max_iter = int(1e6)
    epsilon = 1e-6
    patience = 50
    last_improvement = 0
    warmup = 200
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find max overlap shift
        max_shift = get_max_shift(h_current)
        
        # Compute gradient at max_shift
        grad = compute_gradient(h_current, max_shift, n_points)
        
        # Stochastic gradient descent step
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Update if improvement or occasionally accept worse moves
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse moves to escape local minima
            if i > warmup and np.random.rand() < 0.01:
                h_current = h_new.copy()
                c5_current = c5_new
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = i
            
        # Occasionally jitter to escape local minima
        if i % 100 == 0 and i > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
            
        # Occasionally restore best solution
        if i % 500 == 0:
            h_current = h_best.copy()
            c5_current = c5_best
    
    # Final refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break

        # Perturb a small number of coordinates
        indices = np.random.choice(n_points, max(1, n_points // 30), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.002)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
