import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Create initial structure
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Smooth trapezoidal
        h_init = 0.5 * (1 + np.cos(np.pi * x)) 
        h_init = project_h(h_init, n_points)
    
    # Multi-start approach
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Collect multiple starting points
    initializations = []
    # Start from initial solution
    initializations.append(h_init.copy())
    
    # Different structured initializations
    x = np.linspace(0, 2, n_points)
    for i in range(3):
        h = np.zeros(n_points)
        h += 0.5 * np.sin(np.pi * (x + i * 0.2))
        h += 0.3 * np.cos(2 * np.pi * (x + i * 0.1))
        h = project_h(h, n_points)
        initializations.append(h)
    
    # Add perturbed versions
    for i in range(2):
        h = h_init + np.random.normal(0, 0.05, n_points)
        h = project_h(h, n_points)
        initializations.append(h)
    
    # Optimization loop for each initialization
    for h_start in initializations:
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current, dx)
        
        # Gradient descent with adaptive learning rate
        lr = 0.01
        patience = 50
        no_improve = 0
        
        for iteration in range(10000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient with respect to maximum overlap
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr)
            # The gradient is the convolution with 1-h, at the shift that gives max
            h_shifted = np.roll(1 - h_current, max_shift_idx - (n_points - 1))
            grad = h_shifted
            
            # Apply gradient descent step
            h_new = h_current - lr * grad
            h_new = project_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > patience:
                    lr *= 0.95
                    no_improve = 0
            
            # Update best solution
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                
    # Final local search
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = best_h.copy()
        # Small random perturbations
        indices = np.random.choice(n_points, max(2, n_points // 20), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return (best_h.tolist(), best_c5, n_points)
