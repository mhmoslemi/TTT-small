import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter_barrier = 1000
    max_iter_gd = 10000
    learning_rate = 1e-3
    mu_initial = 1e3
    barrier_decrease_factor = 0.99
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Barrier method
    mu = mu_initial
    for barrier_iter in range(max_iter_barrier):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation and maximum lag
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient via adjoint method
        # For maximum lag, compute gradient of the corresponding correlation
        max_lag = max_lag_idx - (n_points - 1)
        grad = np.zeros_like(h)
        
        # Gradient is 1-h with appropriate shift
        if max_lag >= 0:
            grad[max_lag:] += (1 - h[:-max_lag])
        else:
            grad[:max_lag] += (1 - h[-max_lag:])
        
        # Update step
        h = h - learning_rate * grad
        
        # Projection to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to meet constraint sum(h) = n_points / 2
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
            
        # Update barrier parameter
        mu *= barrier_decrease_factor
    
    # Gradient descent phase
    for gd_iter in range(max_iter_gd):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation and maximum lag
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient via adjoint method
        max_lag = max_lag_idx - (n_points - 1)
        grad = np.zeros_like(h)
        
        if max_lag >= 0:
            grad[max_lag:] += (1 - h[:-max_lag])
        else:
            grad[:max_lag] += (1 - h[-max_lag:])
        
        # Update step
        h = h - learning_rate * grad
        
        # Projection to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to meet constraint sum(h) = n_points / 2
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
