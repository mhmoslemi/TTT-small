import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # PSO parameters
    swarm_size = 50
    inertia_weight_start = 0.9
    inertia_weight_end = 0.4
    cognitive_coef = 1.5
    social_coef = 1.5
    velocity_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_fitness = []
    
    # First particle from initial solution
    swarm.append(h_init.copy())
    velocities.append(np.zeros(n_points))
    personal_best.append(h_init.copy())
    personal_best_fitness.append(evaluate_h(h_init, dx))
    
    # Remaining particles with different strategies
    for i in range(1, swarm_size):
        if i % 3 == 0:
            # Sine wave pattern
            h = 0.5 + 0.3 * np.sin(np.linspace(0, 4*np.pi, n_points))
        elif i % 3 == 1:
            # Sawtooth pattern
            h = np.linspace(0, 1, n_points)
            h = h[::-1]  # reverse
        else:
            # Random
            h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        personal_best.append(h.copy())
        personal_best_fitness.append(evaluate_h(h, dx))
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]
    
    # PSO main loop
    max_iter = 100000
    stagnation_counter = 0
    stagnation_threshold = 500
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        inertia_weight = inertia_weight_start - (inertia_weight_start - inertia_weight_end) * (i / max_iter)
        
        # Update each particle
        for j in range(swarm_size):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = cognitive_coef * r1 * (personal_best[j] - swarm[j])
            social = social_coef * r2 * (global_best - swarm[j])
            velocities[j] = inertia_weight * velocities[j] + cognitive + social
            
            # Clip velocity
            velocities[j] = np.clip(velocities[j], -velocity_max, velocity_max)
            
            # Update position
            swarm[j] = swarm[j] + velocities[j]
            
            # Project to feasible region
            swarm[j] = project_h(swarm[j], n_points, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(swarm[j], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[j]:
                personal_best[j] = swarm[j].copy()
                personal_best_fitness[j] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = swarm[j].copy()
                    global_best_fitness = fitness
                    stagnation_counter = 0
                else:
                    stagnation_counter += 1
            
            # Check for stagnation
            if stagnation_counter > stagnation_threshold and i > 1000:
                # Perturb the worst particle
                worst_idx = np.argmax(personal_best_fitness)
                swarm[worst_idx] = personal_best[worst_idx] + np.random.normal(0, 0.02, n_points)
                swarm[worst_idx] = project_h(swarm[worst_idx], n_points, dx)
                personal_best_fitness[worst_idx] = evaluate_h(swarm[worst_idx], dx)
                stagnation_counter = 0
                
    return (global_best.tolist(), global_best_fitness, n_points)
