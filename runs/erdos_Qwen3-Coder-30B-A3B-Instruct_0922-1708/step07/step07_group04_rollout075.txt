import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Simulated Annealing with restarts
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Annealing parameters
    temp = 0.1
    cooling_rate = 0.95
    min_temp = 1e-6
    max_iter = 1000
    patience = 500
    
    # Track the last improvement
    last_improvement = 0
    
    # Simulated Annealing iterations
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Try a random perturbation
        h_new = h_current.copy()
        idx = np.random.randint(n_points)
        delta = np.random.normal(0, 0.02)
        h_new[idx] += delta
        h_new = repair_h(h_new, n_points)
        
        # Evaluate the change
        c5_new = evaluate_h(h_new, dx)
        delta_e = c5_new - c5_current
        
        # Accept or reject based on temperature
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / temp):
            h_current = h_new
            c5_current = c5_new
            
            # Update best if improved
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
        # Cool down temperature
        temp = max(temp * cooling_rate, min_temp)
        
        # Restart if no improvement
        if i - last_improvement > patience:
            h_current = h_best.copy()
            temp = 0.1
            last_improvement = i
            
    # Now refine with gradient descent
    eta = 0.01
    eps = 1e-6
    max_iter = 5000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Update each coordinate using gradient approximation
        h_new = h_current.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')

            h_minus = h_current.copy()

            h_current = h_new
            c5_current = c5_new
            
            # Update best if improved
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                
        # Reduce learning rate over time
        eta *= 0.9995
    
    return (h_best.tolist(), c5_best, n_points)
