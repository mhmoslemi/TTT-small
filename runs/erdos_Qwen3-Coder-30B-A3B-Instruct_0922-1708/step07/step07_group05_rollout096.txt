import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def generate_fourier_candidates(n_points, num_candidates=50):
    candidates = []
    for _ in range(num_candidates):
        # Generate Fourier-like basis functions
        h = np.zeros(n_points)
        # Use 3-5 Fourier modes with random coefficients
        num_modes = np.random.randint(3, 6)
        for _ in range(num_modes):
            k = np.random.randint(1, 10)
            a = np.random.uniform(-0.5, 0.5)
            b = np.random.uniform(-0.5, 0.5)
            x = np.linspace(0, 2, n_points)
            h += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
        
        # Ensure boundedness and sum constraint
        h = np.clip(h, 0, 1)
        h = repair_h(h, n_points)
        candidates.append(h)
    
    return candidates

        h_minus[i] -= epsilon
        h_plus = repair_h(h_plus, n_points)
        h_minus = repair_h(h_minus, n_points)
        
        c5_plus = evaluate_h(h_plus, dx)
        c5_minus = evaluate_h(h_minus, dx)
        grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
    
    return grad

    candidate_scores = []
