import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def create_segmented_h(n_points, dx):
    """Create a good starting configuration."""
    h = np.zeros(n_points)
    # Create a piecewise constant function with two regions
    mid = n_points // 2
    h[:mid] = 0.8
    h[mid:] = 0.2
    h = project_h(h, n_points, dx)
    return h

def crossover(parent1, parent2, dx, n_points):
    """Perform segment-wise crossover."""
    # Create a crossover mask
    mask = np.random.rand(n_points) > 0.5
    offspring = np.where(mask, parent1, parent2)
    return project_h(offspring, n_points, dx)

def mutate(h, dx, n_points, mutation_scale):
    """Mutate h with Gaussian noise and segment flips."""
    h = h.copy()
    # Add Gaussian noise
    noise = np.random.normal(0, mutation_scale, n_points)
    h += noise
    
    # Coarse segment flips
    if np.random.rand() < 0.3:
        flip_idx = np.random.randint(0, n_points)
        h[flip_idx] = 1 - h[flip_idx]
        
    # Project to ensure constraints
    h = project_h(h, n_points, dx)
    
    return h

def local_refinement(h, dx, n_points, max_iter=50):
    """Fast coordinate descent local refinement."""
    h_current = h.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Simple local search with fixed steps
    for _ in range(max_iter):
        improved = False
        indices = np.random.permutation(n_points)
        for j in indices:
            # Try small perturbations
            eps = 1e-5
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            h_new = h_current.copy()
            h_new[j] -= 0.01 * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
            
            h_new = project_h(h_new, n_points, dx)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                improved = True
                
        if not improved:
            break
            
    return h_current, c5_current

    num_parents = 10
    max_generations = 2000

        h_init = project_h(h_init, n_points, dx)
    else:
        h_init = create_segmented_h(n_points, dx)
        
    # Add the initial solution and its variants to the population
    population.append(h_init)
    for _ in range(population_size - 1):
        h = h_init.copy()
        h = mutate(h, dx, n_points, 0.05)
        population.append(h)
    
    # Best solution tracking
    best_h = None
    best_c5 = float('inf')
    
    # Evolve population over generations
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update global best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        # Select best parents
        parents = population[:num_parents]
        
        # Generate new population
        new_population = parents.copy()
        
        # Crossover and mutation
        for _ in range(population_size - num_parents):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            offspring = crossover(parent1, parent2, dx, n_points)
            
            # Add mutation
            offspring = mutate(offspring, dx, n_points, 0.1 * (0.99 ** gen))

            new_population = new_population[:population_size]
            
        population = new_population
        
    # Final check and return
    if best_h is None:
        # If no improvement, use the last generation's best
        last_fitness = [evaluate_h(h, dx) for h in population]
        best_idx = np.argmin(last_fitness)
        best_h = population[best_idx]
        best_c5 = last_fitness[best_idx]
        
    return (best_h.tolist(), best_c5, n_points)
