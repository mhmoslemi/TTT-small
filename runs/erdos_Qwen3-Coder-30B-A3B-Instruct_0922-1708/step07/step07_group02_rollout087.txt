import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate_individual(h, n_points, mutation_scale):
    """Apply mutation to a single individual."""
    h_mut = h.copy()
    # Flip random segments
    if np.random.random() < 0.3:
        idx1, idx2 = np.random.randint(0, n_points, 2)
        if idx1 > idx2:
            idx1, idx2 = idx2, idx1
        h_mut[idx1:idx2] = 1 - h_mut[idx1:idx2]
    # Add Gaussian noise
    noise = np.random.normal(0, mutation_scale, n_points)
    h_mut += noise
    h_mut = project_h(h_mut, n_points, 2.0 / n_points)
    return h_mut

def crossover_individuals(h1, h2, n_points):
    """Perform crossover between two individuals."""
    h_cross = h1.copy()
    # Blend crossover
    blend = np.random.random()
    for i in range(n_points):
        if np.random.random() < 0.5:
            h_cross[i] = h2[i]
    return h_cross

def local_optimize(h, n_points, dx, max_iter=50):
    """Fast local optimization with coordinate descent."""
    h_opt = h.copy()
    eta = 0.01
    eps = 1e-6
    for _ in range(max_iter):
        h_new = h_opt.copy()
        indices = list(range(n_points))
        np.random.shuffle(indices)
        improved = False
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < evaluate_h(h_opt, dx):
            h_opt = h_new
            improved = True
        if not improved:
            eta *= 0.999
        else:
            eta = min(0.01, eta * 1.001)
    return h_opt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Create a structured feasible solution
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        h_init = project_h(h_init, n_points, dx)
    
    # Memetic algorithm parameters
    population_size = 50
    selection_ratio = 0.2
    mutation_scale = 0.1
    max_generations = 1500
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = mutate_individual(h_init, n_points, mutation_scale)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    h_best = population[best_idx].copy()
    c5_best = fitness[best_idx]
    
    # Memetic loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness and select elites
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:int(population_size * selection_ratio)]]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size - len(elites)):
            parent1 = elites[np.random.randint(0, len(elites))]
            parent2 = elites[np.random.randint(0, len(elites))]
            child = crossover_individuals(parent1, parent2, n_points)
            child = mutate_individual(child, n_points, mutation_scale)
            child = local_optimize(child, n_points, dx)
            offspring.append(child)
        
        # Combine elites and offspring
        population = elites + offspring
        fitness = [evaluate_h(h, dx) for h in population]
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < c5_best:
            h_best = population[best_idx].copy()
            c5_best = fitness[best_idx]
        
        # Reduce mutation scale
        mutation_scale *= 0.99
        
        # Occasionally inject random candidate
        if np.random.random() < 0.05 or generation % 100 == 0:
            h_random = np.random.rand(n_points)
            h_random = project_h(h_random, n_points, dx)
            population.append(h_random)
            fitness.append(evaluate_h(h_random, dx))
            if fitness[-1] < c5_best:
                h_best = h_random.copy()
                c5_best = fitness[-1]
    
    return (h_best.tolist(), c5_best, n_points)
