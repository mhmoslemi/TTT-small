import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a simple construction
        h = np.ones(n_points) * 0.5
        
    # Frank-Wolfe optimization
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe step size and parameters
    gamma = 0.5
    backtracking_factor = 0.5
    min_improvement = 1e-8
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break

        # Compute current overlap integral
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient direction
        # We want to minimize max_k int h(x)(1-h(x+k)) dx
        # So we compute the gradient w.r.t h
        # This is done via finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum) > 1e-10:
                h_plus = h_plus * (n_points / 2) / h_plus_sum
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps

        # Compute the direction s for Frank-Wolfe
        # This is the solution to min_s <grad, s> subject to 0 <= s <= 1, sum s = 1
        # This is a simple linear assignment problem
        s = np.zeros(n_points)
        sorted_indices = np.argsort(-grad)  # We want to minimize the gradient, so positive values should be minimized
        
        remaining_sum = n_points / 2
        remaining_grad = remaining_sum * 2  # dummy initialization
        s_copy = np.zeros(n_points)
        
        # Simple greedy approach for the direction
        for i in sorted_indices:
            if remaining_sum <= 0:
                break
            # Assign as much as possible to s[i]
            # But this s.requires a constraint: it must be between 0 and 1 
            # and also sum to n_points / 2.
            # So we increase s[i] gradually up to min(1, remaining_sum)
            take_amount = min(1.0, remaining_sum)
            s[i] = take_amount
            remaining_sum -= take_amount
            
        # Rescale s to have correct sum
        total_s = np.sum(s)
        if abs(total_s) > 1e-10:
            s = s * (n_points / 2) / total_s

        # Line search for step size
        alpha = gamma
        best_alpha = alpha
        
        # Do a simple backtracking line search
        while alpha > 1e-10:  # Earliest stopping condition
            h_candidate = (1 - alpha) * h + alpha * s
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate_sum = np.sum(h_candidate)
            if abs(h_candidate_sum) > 1e-10:
                h_candidate = h_candidate * (n_points / 2) / h_candidate_sum
            corr_candidate = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr_candidate) * dx
            if c5_candidate < current_c5 - min_improvement:
                best_alpha = alpha
                break
            alpha *= backtracking_factor
            
        # Update step
        h = (1 - best_alpha) * h + best_alpha * s
        
        # Project to valid region (maintain feasibility)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-10:
            h = h * (n_points / 2) / h_sum

    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
