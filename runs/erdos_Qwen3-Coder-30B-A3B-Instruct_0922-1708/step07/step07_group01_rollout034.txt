import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # DP-based optimization
    max_iter = int(1e6)
    patience = 100
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Try to improve h by modifying one element at a time
        best_change = None
        best_c5 = c5_current
        
        # Sample a few positions to try modifications
        candidates = np.random.choice(n_points, min(20, n_points), replace=False)
        
        for idx in candidates:
            # Try small perturbations
            for perturbation in [-0.02, -0.01, 0.01, 0.02]:
                h_new = h_current.copy()
                h_new[idx] += perturbation
                h_new = project_h(h_new, n_points)

            h_current = project_h(h_current, n_points)

        else:

                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
        
        # Update best solution
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
        
        # Adaptive refinement every 500 iterations
        if i % 500 == 0 and i > 0:
            # Refine by neighborhood search
            for _ in range(100):
                if time.time() - start_time > budget_s - 5:

            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
