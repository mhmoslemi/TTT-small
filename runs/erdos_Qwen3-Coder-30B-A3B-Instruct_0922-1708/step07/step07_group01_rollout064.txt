import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Generate a diverse set of initializations
        h_init = np.zeros(n_points)
        # Create different waveforms
        x = np.linspace(0, 2, n_points)
        waveforms = [
            0.5 * (1 + np.cos(np.pi * x)),  # Cosine
            np.maximum(0, 1 - np.abs(x - 1) * 2),  # Triangular
            np.where((x > 0.5) & (x < 1.5), 1.0, 0.0),  # Rectangular
            0.5 * (1 + np.sin(2 * np.pi * x)),  # Sine
            np.exp(-10 * (x - 1)**2),  # Gaussian
        ]
        h_init = waveforms[np.random.randint(len(waveforms))]
        h_init = project_h(h_init, n_points)
    
    # Create multiple initializations
    initializations = []
    for i in range(10):
        h = h_init.copy()
        # Add small random perturbation
        noise = np.random.normal(0, 0.01, n_points)
        h = project_h(h + noise, n_points)
        initializations.append(h)
    
    h_best = None
    c5_best = float('inf')
    
    for i, h_init in enumerate(initializations):
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
        
        # Gradient descent with adaptive learning rate
        learning_rate = 0.01
        patience = 50
        no_improvement = 0
        last_improvement = 0
        max_iter = 5000
        
        for j in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Find the shift that maximizes overlap
            corr = correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr)
            
            # Compute gradient for this k_max
            # Create shifted version of h_current for gradient
            pad = len(h_current) - 1
            h_padded = np.pad(h_current, pad, mode='constant')
            h_shifted = np.roll(h_padded, k_max)
            h_shifted = h_shifted[pad:-pad]  # Remove padding
            grad = (1 - h_shifted) * dx
            
            # Step with gradient ascent (maximize overlap)
            h_new = h_current + learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            # Adaptive update
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                last_improvement = j
            else:
                no_improvement += 1
                if no_improvement > patience:
                    learning_rate *= 0.9
                    no_improvement = 0
                    # Add some noise to escape
                    h_current = project_h(h_current + np.random.normal(0, 0.005, n_points), n_points)
                    c5_current = evaluate_h(h_current, dx)
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = j
    
    # Final local refinement
    if h_best is not None:
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            h_new = h_best.copy()
            indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.005)
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
