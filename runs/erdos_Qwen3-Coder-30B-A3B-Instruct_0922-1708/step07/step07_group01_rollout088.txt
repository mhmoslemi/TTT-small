import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Multiple initializations from different patterns
    initializations = []
    
    # Original initialization
    initializations.append(h_init.copy())
    
    # Random initialization
    h_random = np.random.uniform(0, 1, n_points)
    h_random = project_h(h_random, n_points)
    initializations.append(h_random)
    
    # Sine-based initialization
    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 * (1 + 0.8 * np.sin(np.pi * x))
    h_sine = project_h(h_sine, n_points)
    initializations.append(h_sine)
    
    # Step function initialization
    h_step = np.zeros(n_points)
    mid = n_points // 2
    h_step[:mid] = 1
    h_step = project_h(h_step, n_points)
    initializations.append(h_step)
    
    # Gaussian bump initialization
    x = np.linspace(0, 2, n_points)
    h_gauss = np.exp(-10 * (x - 1)**2)
    h_gauss = project_h(h_gauss, n_points)
    initializations.append(h_gauss)
    
    # Initialize best solution
    h_best = None
    c5_best = float('inf')
    
    # Optimize from each initialization
    for i, h_start in enumerate(initializations):
        h_current = h_start.copy()
        c5_current = evaluate_h(h_current, dx)
        
        # Optimization parameters
        learning_rate = 0.02
        max_iter = int(1e6)
        patience = 500
        last_improvement = 0
        stagnation_count = 0
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Get the shift that produces maximum overlap
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_shift_idx = np.argmax(corr)
            max_shift = max_shift_idx - (n_points - 1)  # Convert to actual shift
            
            # Compute gradient for the max shift
            h_shifted = np.roll(1 - h_current, -max_shift)
            gradient = h_shifted
            
            # Update with gradient
            h_new = h_current - learning_rate * gradient
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept improvement or occasionally accept worse moves
            if c5_new < c5_current or np.random.rand() < 0.02:
                h_current = h_new
                c5_current = c5_new
                stagnation_count = 0
            else:
                stagnation_count += 1
                
            # Update best solution found
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = iteration
                
            # Adaptive learning rate
            if stagnation_count > patience:
                learning_rate *= 0.95
                stagnation_count = 0
                if learning_rate < 1e-6:
                    break
                    
            # Occasionally jitter to escape local minima
            if iteration % 200 == 0 and iteration > 0:
                noise = np.random.normal(0, 0.01, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
    
    # Final refinement through local search
    if h_best is not None:
        for _ in range(500):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_new = h_best.copy()
            # Perturb small portion of coordinates
            indices = np.random.choice(n_points, max(2, n_points // 20), replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.002)
            
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
    
    if h_best is None:
        h_best = h_init
        c5_best = evaluate_h(h_init, dx)
        
    return (h_best.tolist(), c5_best, n_points)
