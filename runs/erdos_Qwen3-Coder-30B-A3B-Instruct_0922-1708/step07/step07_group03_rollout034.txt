import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe algorithm
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation to get current C5 value
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient
        # Approximate gradient by finite differences 
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            # Normalize
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Solve linear subproblem to find descent direction
        # Minimize <grad, s> subject to s in [0,1], sum(s) = n_points/2
        # We use a greedy approach to find a candidate descent direction
        s = np.zeros(n_points)
        indices = np.argsort(-grad)  # descending order
        
        # Start with zero vector and greedily assign values
        remaining_sum = n_points / 2
        for i in indices:
            if remaining_sum <= 0:
                break
            take = min(1.0, remaining_sum)
            s[i] = take
            remaining_sum -= take
        
        # If we didn't meet the sum constraint, adjust
        if abs(np.sum(s) - n_points / 2) > 1e-8:
            factor = (n_points / 2) / np.sum(s)
            s = s * factor
            
        # Ensure s remains in [0,1]
        s = np.clip(s, 0, 1)
        
        # Line search for optimal step size γ
        gamma = 0.5
        step_size = gamma
        best_gamma = gamma
        best_new_c5 = current_c5
        for _ in range(10):
            h_new = (1 - step_size) * h + step_size * s
            # Normalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            new_c5 = np.max(corr_new) * dx
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_gamma = step_size
                
            step_size *= 0.5
            if step_size < 1e-6:
                break
                
        gamma = best_gamma
        h = (1 - gamma) * h + gamma * s
        
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
