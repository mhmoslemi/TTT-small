import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size, initial_h_values=None):
    pop = []
    if initial_h_values is not None:
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        pop.append(h_init)
    
    # Fill remaining population with triangular pulses
    while len(pop) < pop_size:
        h = np.zeros(n_points)
        peak_idx = np.random.randint(0, n_points)
        width = np.random.randint(5, n_points//4)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if x >= 0.5 and x <= 1.5:
                h[i] = 1.0 - abs(x - 1.0) * 2.0
        h = repair_h(h, n_points)
        pop.append(h)
    
    return pop

def selection(pop, fitnesses):
    sorted_indices = np.argsort(fitnesses)
    selected = [pop[i] for i in sorted_indices[:len(pop)//2]]
    return selected

def crossover(parent1, parent2):
    n = len(parent1)
    crossover_point = np.random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n = len(h)
    indices = np.random.choice(n, int(n * mutation_rate), replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.05)
    return h

def genetic_algorithm(seed, budget_s, n_points=100):
    np.random.seed(seed)
    pop_size = 50
    max_generations = 1000
    patience = 200
    start_time = time.time()
    
    # Initialize population
    pop = initialize_population(n_points, pop_size, 
                                globals().get('initial_h_values', None))
    
    dx = 2.0 / n_points
    best_fitness = float('inf')
    best_h = None
    
    last_improvement = 0
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_h(h, dx) for h in pop]
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_h = pop[current_best_idx].copy()
            last_improvement = generation
            
        # Selection
        selected = selection(pop, fitnesses)
        
        # Create new population
        new_pop = [selected[0]]  # Elitism
        while len(new_pop) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = repair_h(mutate(child1), n_points)
            child2 = repair_h(mutate(child2), n_points)
            new_pop.extend([child1, child2])
            
        pop = new_pop[:pop_size]
        
        # Early stopping
        if generation - last_improvement > patience:
            break
            
    return best_h, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    h_best, c5_best = genetic_algorithm(seed, budget_s)
    n_points = 100
    return (h_best.tolist(), c5_best, n_points)
