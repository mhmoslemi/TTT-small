import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular pulse centered at x=1
        h_init = np.zeros(n_points)
        x_vals = np.linspace(0, 2, n_points)
        for i in range(n_points):
            x = x_vals[i]
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = max(0, 1 - abs(x - 1.0))
            else:
                h_init[i] = 0.0
        h_init = repair_h(h_init, n_points)

    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    learning_rate = 0.01
    epsilon = 1e-6
    max_iter = 50000
    patience = 100
    last_improvement = 0
    temperature = 0.1
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradients using finite differences
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            h_plus = repair_h(h_plus, n_points)
            h_minus = repair_h(h_minus, n_points)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update with gradient descent
        h_new = h_current - learning_rate * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject step
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
        else:
            # Simulated annealing
            if np.random.rand() < np.exp(-(c5_new - c5_current) / temperature):
                h_current = h_new
                c5_current = c5_new
        
        # Adapt learning rate and temperature
        if i - last_improvement > patience:
            learning_rate *= 0.95
            temperature *= 0.99
            if learning_rate < 1e-6:
                break
    
    return (h_best.tolist(), c5_best, n_points)
