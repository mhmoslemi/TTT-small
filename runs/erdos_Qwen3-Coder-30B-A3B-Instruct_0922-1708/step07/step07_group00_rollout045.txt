import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(theta, n_points):
    """Convert parameters to step function h."""
    K = len(theta) // 2
    a = theta[:K]
    x = np.concatenate([[0], theta[K:]])
    x = np.sort(x)
    x = np.clip(x, 0, 2)
    if x[-1] != 2:
        x[-1] = 2
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(K):
        start_idx = int(x[i] / dx)
        end_idx = int(x[i+1] / dx)
        h[start_idx:end_idx] = a[i]
        
    return h

def repair_parameters(theta):
    """Repair parameters to satisfy constraints."""
    K = len(theta) // 2
    a = np.clip(theta[:K], 0, 1)
    x = np.concatenate([[0], theta[K:]])
    x = np.sort(x)
    x = np.clip(x, 0, 2)
    if x[-1] != 2:
        x[-1] = 2
    return np.concatenate([a, x[1:]])

    eps = 1e-6

    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n_points - 1)
    
    for i in range(len(theta)):

        theta_plus[i] += eps
        theta_plus = repair_parameters(theta_plus)
        h_plus = parameterize_h(theta_plus, n_points)

        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    return grad
