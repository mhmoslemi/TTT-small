import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Start with feasible initial solution
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Convex relaxation approach
    # Using CVXPY for second-order cone programming relaxation
    # Instead of gradient descent, we use a convex approximation
    
    # Define optimization variables
    h_vars = cp.Variable(n_points, name="h")
    
    # Constraint for integral constraint
    integral_constraint = cp.Equality(cp.sum(h_vars), n_points / 2)
    
    # Bound constraints
    bound_constraints = [cp.Ge(h_vars, 0), cp.Le(h_vars, 1)]

    # Objective function

    # This is a relaxed version; regularization to make it convex

        # Every 100 iterations, re-evaluate what we have
        if i % 100 == 0:
            if time.time() - start_time > budget_s - 5:
                break
        
        # Convex optimization using CVXPY for this small problem    
        # Direct approach - define a projective gradient or proximal method
        # Try alternate optimization for neighbors
        
        # In each iteration, we use a small randomized movement
        h_new = h_current.copy()

        num_perturb = max(1, int(0.05 * n_points))

            h_new[idx] += np.random.uniform(-0.02, 0.02)
        
        # Project back onto constraints
        h_new = project_h(h_new, n_points, dx)
