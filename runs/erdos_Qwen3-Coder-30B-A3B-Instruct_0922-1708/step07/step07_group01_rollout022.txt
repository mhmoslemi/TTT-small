import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Cosine taper that starts and ends at 0
        h_init = 0.5 * (1 + np.cos(np.pi * x))
        h_init = project_h(h_init, n_points)
    
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Binary search for minimal C
    C_low = 0.0
    C_high = 0.5
    tolerance = 1e-5
    max_iterations = 50
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        C_mid = (C_low + C_high) / 2.0
        
        # Try to find a feasible solution for this C
        h_vars = cp.Variable(n_points)
        constraints = []
        
        # Add integral constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Add bound constraints
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Add overlap constraints
        for k in range(n_points):
            if k == 0:
                continue
            # For the overlap integral with shift k
            # Evaluate: ∫ h(x)(1-h(x+k)) dx
            # In discrete form: sum_{i=0}^{n-k} h[i] * (1 - h[i+k]) <= C * dx
            # We manually build the constraint
            overlap_sum = 0
            for i in range(n_points - k):
                overlap_sum += h_vars[i] * (1 - h_vars[i+k])
            constraints.append(overlap_sum <= C_mid * dx)
        
        # This optimization problem is not convex. Let's just solve a simpler LP-like problem
        # using a projection approach
        h_test = h_best.copy()
        h_test = project_h(h_test, n_points)
        
        # Use a modification of the previous approach but with more efficient updates
        # Perturb to get a better solution
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_new = h_test.copy()
            # Random perturbation
            rand_indices = np.random.choice(n_points, max(1, n_points // 10), replace=False)
            for i in rand_indices:
                h_new[i] += np.random.normal(0, 0.005)
            
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
            
            h_test = h_new
        
        # Check if this solution works for C_mid
        max_overlap = evaluate_h(h_best, dx)
        
        if max_overlap <= C_mid:
            C_high = C_mid
        else:
            C_low = C_mid
            
        if C_high - C_low < tolerance:
            break
    
    # Final refinement to handle the actual minimal C value
    # Our best h so far is valid. Return it directly
    return (h_best.tolist(), c5_best, n_points)
