import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Binary search for optimal C5
    C_low = 0.0
    C_high = 0.5
    epsilon = 1e-5
    best_h = h_init.copy()
    best_C = evaluate_h(h_init, dx)
    
    # Because we're bound by time, make a reasonable number of iterations
    max_iterations = 50
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        C_mid = (C_low + C_high) / 2.0
        
        # We solve a feasibility problem: does there exist h such that all overlaps are at most C_mid*dx?
        h_vars = cp.Variable(n_points)
        
        # Constraints:
        # Each overlap constraint
        constraints = []
        
        # Add sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Add box constraints
        constraints.append(h_vars >= 0)
        constraints.append(h_vars <= 1)
        
        # Add overlap constraints (until we figure out how to make them)
        # Instead of the full matrix, estimate some key constraints
        max_k = n_points // 2
        for k in range(1, max_k):
            # Find indices that can contribute to overlap term k
            overlap_sum = 0
            # We compute the circular convolution implicitly by slicing
            for i in range(n_points - k):
                overlap_sum += h_vars[i] * (1.0 - h_vars[i+k])
            # Add constraint: overlap_sum <= C_mid * dx
            constraints.append(overlap_sum <= C_mid * dx)
        
        # Minimize with a dummy objective (we only care about feasibility)
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                # Feasible solution
                h_feasible = h_vars.value
                if h_feasible is not None:
                    # This solution is feasible with C = C_mid, so try lower C
                    C_high = C_mid
                    best_h = h_feasible
                    best_C = C_mid
                else:
                    # This should not happen in practice...
                    C_low = C_mid
            else:
                # Infeasible solution
                C_low = C_mid
        except:
            # If solving fails, just do binary search
            C_low = C_mid
            
        if C_high - C_low < epsilon:
            break

    # Final refinement with local search around best solution
    h_final = project_h(best_h, n_points)
    c5_final = evaluate_h(h_final, dx)
    
    # Run a bit more gradient descent if time allows
    if time.time() - start_time < budget_s - 10:
        h_current = h_final.copy()
        c5_current = c5_final
        learning_rate = 0.01
        for _ in range(10000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros(n_points)
            epsilon_fd = 1e-6
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += epsilon_fd
                h_plus = project_h(h_plus, n_points)
                c5_plus = evaluate_h(h_plus, dx)
                
                h_minus = h_current.copy()
                h_minus[i] -= epsilon_fd
                h_minus = project_h(h_minus, n_points)
                c5_minus = evaluate_h(h_minus, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * epsilon_fd)
            
            # Update step
            h_new = h_current - learning_rate * grad
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            # Accept improvement
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < best_C:
                    best_h = h_current
                    best_C = c5_current
            else:
                # Reduce step size occasionally
                learning_rate *= 0.999

    c5_final = evaluate_h(best_h, dx)

    return (best_h.tolist(), c5_final, n_points)
