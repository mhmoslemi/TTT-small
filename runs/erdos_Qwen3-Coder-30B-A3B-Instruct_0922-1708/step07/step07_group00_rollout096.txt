import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def piecewise_h(theta, n_points):
    """Create a piecewise constant function from parameters."""
    K = len(theta) // 2
    a = theta[:K]
    x = np.concatenate([[0], theta[K:], [2]])
    x = np.sort(x)
    x = np.clip(x, 0, 2)
    dx_segments = np.diff(x)
    
    h = np.zeros(n_points)
    for i in range(K):
        start_idx = int(x[i] / 2.0 * n_points)
        end_idx = int(x[i+1] / 2.0 * n_points)
        if end_idx > start_idx:
            h[start_idx:end_idx] = a[i]
    return h

def evaluate_piecewise(theta, n_points, dx):
    """Evaluate C5 for piecewise function defined by parameters."""
    h = piecewise_h(theta, n_points)
    return evaluate_h(h, dx)

def compute_gradient(theta, n_points, dx, eps=1e-6):
    """Compute numerical gradient of piecewise function with respect to parameters."""
    base_value = evaluate_piecewise(theta, n_points, dx)
    grad = np.zeros_like(theta)
    
    for i in range(len(theta)):
        theta_plus = theta.copy()
        theta_plus[i] += eps
        val_plus = evaluate_piecewise(theta_plus, n_points, dx)
        grad[i] = (val_plus - base_value) / eps
    return grad

    current_c5 = evaluate_piecewise(current_theta, n_points, dx)
    best_theta = current_theta.copy()
    best_c5 = current_c5
    
    # Optimization parameters
    max_iter = 5000
    learning_rate = 1e-3
    
    for i in range(max_iter):
