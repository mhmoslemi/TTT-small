import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Sobol sequences for multiple initializations
    sampler = qmc.Sobol(d=n_points, seed=seed)
    initial_points = sampler.random(n=500)
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    for i in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Initialize h from Sobol point
        h = initial_points[i]
        h = h * (n_points / 2) / np.sum(h)
        h = repair_h(h, n_points)
        
        # Nelder-Mead optimization
        try:
            res = minimize(
                lambda x: evaluate_h(repair_h(x, n_points), dx),
                h,
                method='Nelder-Mead',
                options={'maxiter': 1000, 'xatol': 1e-6, 'fatol': 1e-6}
            )
            h = res.x
            h = repair_h(h, n_points)
        except:
            pass
        
        # Gradient-based refinement
        h_refined = h.copy()
        learning_rate = 1e-5
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            # Compute gradient using finite differences
            eps = 1e-6
            grad = np.zeros(n_points)
            corr_base = evaluate_h(h_refined, dx)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = evaluate_h(h_plus, dx)
                grad[j] = (corr_plus - corr_base) / eps
            
            # Update
            h_refined = h_refined - learning_rate * grad
            h_refined = repair_h(h_refined, n_points)
            corr = evaluate_h(h_refined, dx)
            
            if corr < best_c5:
                best_h = h_refined.copy()
                best_c5 = corr
            
            # Adaptive learning rate
            if _ % 100 == 0:
                learning_rate *= 0.99
        
        # Check if current is better
        curr_c5 = evaluate_h(h_refined, dx)
        if curr_c5 < best_c5:
            best_h = h_refined.copy()
            best_c5 = curr_c5
    
    return (best_h.tolist(), best_c5, n_points)
