import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def blend_segments(h1, h2, n_points):
    """Blend segments between two solutions."""
    # Take 50% from each solution
    blend = np.zeros(n_points)
    mid = n_points // 2
    blend[:mid] = h1[:mid]
    blend[mid:] = h2[mid:]
    return blend

def local_refinement(h, n_points, dx, max_iter=50, eta=0.01):
    """Fast coordinate descent refinement."""
    h_current = h.copy()
    c5_current = evaluate_h(h_current, dx)
    
    eps = 1e-6
    for _ in range(max_iter):
        h_new = h_current.copy()
        improved = False
        
        # Random order
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
            
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            improved = True
        else:
            eta *= 0.99
            
    return h_current, c5_current

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Create a structured feasible solution
        h_init = np.zeros(n_points)
        # Create piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Memetic algorithm parameters
    population_size = 50
    elite_size = 10
    mutation_scale = 0.1
    generations = 0
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = project_h(h_rand, n_points, dx)
        population.append(h_rand)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_h(h, dx)
        fitness.append(c5)
    
    # Keep track of the best
    best_idx = np.argmin(fitness)
    h_best = population[best_idx].copy()
    c5_best = fitness[best_idx]
    
    # Main memetic loop
    while time.time() - start_time < budget_s - 5:
        generations += 1
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:elite_size]]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size - elite_size):
            # Select two random parents
            idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
            parent1, parent2 = elites[idx1], elites[idx2]
            
            # Blend segments
            child = blend_segments(parent1, parent2, n_points)
            
            # Add Gaussian noise
            noise = np.random.normal(0, mutation_scale, n_points)
            child += noise
            
            # Project
            child = project_h(child, n_points, dx)
            
            # Local refinement
            child, _ = local_refinement(child, n_points, dx, max_iter=30)
            
            offspring.append(child)
            
        # Replace worst population members with offspring
        worst_indices = sorted_indices[-(population_size - elite_size):]
        for i, idx in enumerate(worst_indices):
            population[idx] = offspring[i]
            fitness[idx] = evaluate_h(offspring[i], dx)
            
        # Update best solution
        new_best_idx = np.argmin(fitness)
        if fitness[new_best_idx] < c5_best:
            h_best = population[new_best_idx].copy()
            c5_best = fitness[new_best_idx]
            
        # Mutation scale decrease
        mutation_scale *= 0.99
        
        # Occasionally introduce diversity
        if generations % 50 == 0:
            for i in range(int(0.1 * population_size)):
                idx = np.random.randint(0, population_size)
                h_new = np.random.rand(n_points)
                h_new = project_h(h_new, n_points, dx)
                population[idx] = h_new
                fitness[idx] = evaluate_h(h_new, dx)
    
    return (h_best.tolist(), c5_best, n_points)
