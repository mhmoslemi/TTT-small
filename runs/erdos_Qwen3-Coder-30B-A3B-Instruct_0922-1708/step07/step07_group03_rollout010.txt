import numpy as np
import time
import math
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create binary initialization with exact sum
        h = np.zeros(n_points)
        ones_count = n_points // 2
        h[:ones_count] = 1.0
        np.random.shuffle(h)
    
    # Temperature for simulated annealing
    temperature = 0.1
    cooling_rate = 0.995
    
    # Current and best solutions
    current_h = h.copy()
    best_h = h.copy()
    
    # Compute initial C5
    corr = correlate(current_h, 1 - current_h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    
    iter_count = 0
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 5:
        # Generate candidate by perturbing current solution
        candidate_h = current_h.copy()
        
        # Randomly select two indices to perturb
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Try different delta values
        delta_choices = [0.01, 0.02, 0.05, 0.1]
        delta = np.random.choice(delta_choices)
        
        # Apply symmetric adjustment
        if np.random.rand() > 0.5:
            delta = -delta
        
        # Ensure the adjustment maintains feasibility
        old_i = candidate_h[i]
        old_j = candidate_h[j]
        
        candidate_h[i] = np.clip(candidate_h[i] + delta, 0, 1)
        candidate_h[j] = np.clip(candidate_h[j] - delta, 0, 1)
        
        # Correct for sum constraint
        current_sum = np.sum(candidate_h)
        target_sum = n_points / 2
        
        if abs(current_sum - target_sum) > 1e-8:
            scale_factor = target_sum / current_sum
            candidate_h *= scale_factor
            
        # Compute new C5
        corr = correlate(candidate_h, 1 - candidate_h, mode='full')
        candidate_c5 = np.max(corr) * dx
        
        # Accept or reject based on simulated annealing
        if candidate_c5 < current_c5 or np.random.rand() < math.exp((current_c5 - candidate_c5) / (temperature + 1e-8)):
            current_h = candidate_h
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Update temperature
        temperature *= cooling_rate
        
        iter_count += 1
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
