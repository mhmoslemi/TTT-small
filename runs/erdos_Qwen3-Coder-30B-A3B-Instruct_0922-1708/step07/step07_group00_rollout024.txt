import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterized_h(params, n_points):
    """Convert parameters to h(x)"""
    K = len(params) // 2
    heights = params[:K]
    boundaries = np.concatenate([[0], params[K:]])
    boundaries = np.sort(np.clip(boundaries, 0, 2))
    boundaries = np.append(boundaries, 2.0)
    
    # Ensure proper boundaries and heights
    heights = np.clip(heights, 0, 1)
    
    # Build h step function
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(K):
        start_idx = int(boundaries[i] / dx)
        end_idx = int(boundaries[i+1] / dx)
        h[start_idx:end_idx] = heights[i]
    
    return h

def compute_gradient_fd(h, params, dx, k_max, eps=1e-6):
    """Finite difference gradient computation"""
    grad = np.zeros_like(params)
    n_points = len(h)
    
    corr_base = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr_base)
    
    for i in range(len(params)):
        params_plus = params.copy()
        params_plus[i] += eps
        h_plus = parameterized_h(params_plus, n_points)
        h_plus = repair_h(h_plus, n_points)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr_base[max_k_idx]) / eps
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Start with initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    # Initial parameterization: 5 segments (K=5)
    K = 5
    # Initialize parameters: heights + boundaries
    heights = [0.5] * K
    boundaries = [0.4, 0.8, 1.2, 1.6]
    params = np.array(heights + boundaries)
    params = np.clip(params, 0, 1)
    
    # Optimize with gradient descent
    learning_rate = 1e-3
    max_iter = 5000
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h = parameterized_h(params, n_points)
        h = repair_h(h, n_points)
        current_c5 = evaluate_h(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        grad = compute_gradient_fd(h, params, dx, n_points)
        params = params - learning_rate * grad
        
        # Constraints repair for parameters
        K = len(params) // 2
        params[:K] = np.clip(params[:K], 0, 1)
        params[K:] = np.sort(np.clip(params[K:], 0, 2))
        
        # Rescale to maintain integral constraint
        h_new = parameterized_h(params, n_points)
