import time
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.fft import fft, ifft

def repair_h(h, n_points):
    """Repair h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
