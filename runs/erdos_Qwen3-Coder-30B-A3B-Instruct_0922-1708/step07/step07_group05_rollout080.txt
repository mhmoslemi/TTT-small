import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, population_size):
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Fill the rest with triangular pulses
    while len(population) < population_size:
        h = np.zeros(n_points)
        peak_pos = np.random.uniform(0.5, 1.5)
        width = np.random.uniform(0.2, 1.0)
        left = max(0, peak_pos - width/2)
        right = min(2, peak_pos + width/2)
        for i in range(n_points):
            x = i * (2.0 / n_points)
            if left <= x <= right:
                h[i] = max(0, 1 - abs(x - peak_pos) / (width/2))
        h = repair_h(h, n_points)
        population.append(h)
    
    return population

def crossover(parent1, parent2):
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    n_points = len(h)
    h_mut = h.copy()
    num_mutations = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, num_mutations, replace=False)
    for i in indices:
        h_mut[i] += np.random.normal(0, 0.05)
    return h_mut

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # GA parameters
    population_size = 50
    generations = 1000
    mutation_rate = 0.2
    elite_size = 1
    
    # Initialize population
    population = initialize_population(n_points, population_size)
    fitness = [evaluate_h(h, dx) for h in population]
    
    # Find best
    best_idx = np.argmin(fitness)
    h_best = population[best_idx].copy()
    c5_best = fitness[best_idx]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        elite = population[:elite_size]
        
        # Selection: Top 50%
        selection_size = population_size // 2
        selected = population[:selection_size]
        selected_fitness = fitness[:selection_size]
        
        # Create new population
        new_population = elite[:]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = repair_h(mutate(child1, mutation_rate), n_points)
            child2 = repair_h(mutate(child2, mutation_rate), n_points)
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        fitness = [evaluate_h(h, dx) for h in population]
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < c5_best:
            h_best = population[best_idx].copy()
            c5_best = fitness[best_idx]
            
    return (h_best.tolist(), c5_best, n_points)
