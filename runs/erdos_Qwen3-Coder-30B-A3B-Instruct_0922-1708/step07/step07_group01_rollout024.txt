import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Start with current best
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Binary search for better C5
    c_low = 0.0
    c_high = 0.5
    tolerance = 1e-5
    max_iter = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Try to find a feasible h for this C
        # Set up the LP problem
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1]
        
        # Add the integral constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Add the overlap constraints:
        # For each k (offset), we have constraint sum(h[i] * (1 - h[i+k])) <= C * dx
        # We check all valid indices for each k
        for k in range(n_points):
            if k == 0:
                continue # This is just h[i]*h[i] = h[i]^2, not needed
            # Sum over valid i values
            expr = 0.0
            for i in range(n_points - k):
                expr += h_vars[i] * (1 - h_vars[i + k])
            constraints.append(expr <= c_mid * dx)
        
        objective = cp.Minimize(0)  # We just need a feasible solution
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                # Solution is feasible
                h_sol = h_vars.value
                if h_sol is not None:
                    h_feasible = project_h(h_sol, n_points)
                    c5_feasible = evaluate_h(h_feasible, dx)
                    if c5_feasible < c5_best:
                        h_best = h_feasible
                        c5_best = c5_feasible
                    c_high = c_mid
                else:
                    c_low = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
    
    # Final optimization using gradient descent on the best we have
    h_current = h_best.copy()
    c5_current = c5_best
    
    # Simple gradient descent loop
    for i in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute finite differences for gradient
        grad = np.zeros(n_points)
        epsilon = 1e-6
        for j in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[j] += epsilon
            h_minus[j] -= epsilon
            h_plus = project_h(h_plus, n_points)
            h_minus = project_h(h_minus, n_points)
            
            c_plus = evaluate_h(h_plus, dx)
            c_minus = evaluate_h(h_minus, dx)
            grad[j] = (c_plus - c_minus) / (2 * epsilon)
        
        # Update
        step_size = 0.005
        h_new = h_current - step_size * grad
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
    
    return (h_best.tolist(), c5_best, n_points)
