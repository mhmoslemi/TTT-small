import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Create a piecewise linear function that is 0 on [0,0.5) and 1 on (1.5,2]
        # and slopes linearly in between
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)

    # PSO parameters
    n_particles = 100
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    
    # Initialize particles and velocities
    particles = np.array([h_init + np.random.normal(0, 0.01, n_points) for _ in range(n_particles)])
    particles = np.array([repair_h(p, n_points) for p in particles])
    velocities = np.zeros((n_particles, n_points))
    
    # Track best solutions
    pbest = particles.copy()
    pbest_values = np.array([evaluate_h(p, dx) for p in particles])
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO loop
    max_iter = 400
    no_improvement_count = 0
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter / max_iter
        
        # Update particles
        for i in range(n_particles):
            # PSO update
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate new position
            new_value = evaluate_h(particles[i], dx)
            
            # Update personal best
            if new_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = new_value
                
                # Update global best
                if new_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = new_value
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
        
        # Mutation
        if iter % 25 == 0 and iter > 0:
            mutation_indices = np.random.choice(n_particles, int(n_particles * mutation_rate), replace=False)
            for i in mutation_indices:
                # Mutate 20% of the particle
                mutate_indices = np.random.choice(n_points, int(n_points * 0.2), replace=False)
                particles[i][mutate_indices] += np.random.normal(0, mutation_sigma, len(mutate_indices))
                particles[i] = repair_h(particles[i], n_points)
        
        # Early stopping
        if no_improvement_count > 100:
            break
    
    # Final evaluation
    final_c5 = evaluate_h(gbest, dx)
    return (gbest.tolist(), final_c5, n_points)
