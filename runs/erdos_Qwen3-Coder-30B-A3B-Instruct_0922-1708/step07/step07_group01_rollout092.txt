import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Structured initialization
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Phase 1: CCCP-like approach
    learning_rate = 0.01
    max_iter_phase1 = 5000
    patience = 500
    last_improvement = 0
    
    for i in range(max_iter_phase1):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find optimal shift for current h
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient w.r.t. h at shift k_max
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Gradient of (h ⋆ (1-h))[k_max] w.r.t. h[j]
            if j + k_max < n_points:
                grad[j] = 1 - h_current[j + k_max]
            else:
                # wrap-around case is handled by correlate
                pass
        
        # Updated h using gradient
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Update if improved
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
        else:
            # Occasionally accept worse moves (simulated annealing)
            if np.random.rand() < 0.01:
                h_current = h_new.copy()
                c5_current = c5_new
                
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Reduce learning rate with patience
        if i - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = i
            
        # Periodic perturbation
        if i % 500 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Phase 2: Cyclic Coordinate Descent
    learning_rate = 0.01
    max_iter_phase2 = 10000
    patience = 1000
    last_improvement = 0
    noise_std = 0.005
    
    for i in range(max_iter_phase2):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic order
        for j in range(n_points):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Current h without j
            h_current_no_j = h_current.copy()
            corr = correlate(h_current_no_j, 1 - h_current_no_j, mode='full')
            k_max = np.argmax(corr)
            
            # Gradient at shift k_max, for index j
            # Calculate effect of changing h[j]
            old_hj = h_current[j]
            
            # Approximate gradient w.r.t. h[j] by slight change
            h_temp_plus = h_current.copy()
            h_temp_plus[j] += 0.001
            h_temp_plus = project_h(h_temp_plus, n_points)
            corr_plus = correlate(h_temp_plus, 1 - h_temp_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            h_temp_minus = h_current.copy()
            h_temp_minus[j] -= 0.001
            h_temp_minus = project_h(h_temp_minus, n_points)
            corr_minus = correlate(h_temp_minus, 1 - h_temp_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad_j = (c5_plus - c5_minus) / 0.002
            
            # Update h[j]
            h_current[j] = h_current[j] - learning_rate * grad_j
            h_current[j] = np.clip(h_current[j], 0, 1)
            
            # Renormalize
            h_current = project_h(h_current, n_points)
            
            # Check for improvement
            c5_new = evaluate_h(h_current, dx)
            if c5_new < c5_current:
                c5_current = c5_new
            else:
                # Restore if not improving
                h_current[j] = old_hj
                
        # Update best found
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Periodic perturbation
        if i % 1000 == 0 and i > 0:
            for _ in range(10):
                idx = np.random.randint(0, n_points)
                h_current[idx] += np.random.normal(0, noise_std)
            h_current = project_h(h_current, n_points)
            c5_current = evaluate_h(h_current, dx)
            # If not improving, increase noise
            if i - last_improvement > patience:
                noise_std *= 1.1
                last_improvement = i
                
        # Reduce learning rate if stuck
        if i - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = i
    
    # Final refinement with local search
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
            
    return (h_best.tolist(), c5_best, n_points)
