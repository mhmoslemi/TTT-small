import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx):
    """Compute gradient of the maximum overlap with respect to h."""
    n = len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n - 1)
    
    # Compute finite difference gradient
    grad = np.zeros(n)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    # Normalize the gradient
    norm = np.linalg.norm(grad) + 1e-12
    grad = grad / norm
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    swarm_size = 100
    max_iter = 10000
    c1, c2 = 2.0, 2.0
    w_min, w_max = 0.1, 0.9
    inertia_decay = (w_max - w_min) / max_iter
    
    # Initialize swarm
    swarm = []
    velocities = []
    particle_bests = []
    particle_best_scores = []
    
    for _ in range(swarm_size):
        if time.time() - start_time > budget_s - 5:
            break
        h = h_init + np.random.normal(0, 0.01, n_points)
        h = repair_h(h, n_points)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_h(h, dx)
        particle_bests.append(h.copy())
        particle_best_scores.append(score)
    
    # Initialize global best
    best_idx = np.argmin(particle_best_scores)
    global_best = particle_bests[best_idx].copy()
    global_best_score = particle_best_scores[best_idx]
    
    # PSO optimization
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        w = w_max - iter_num * inertia_decay
        
        for i in range(swarm_size):
            if time.time() - start_time > budget_s - 5:
                break
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (particle_bests[i] - swarm[i]) +
                           c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            swarm[i] = repair_h(swarm[i], n_points)
            
            # Evaluate
            score = evaluate_h(swarm[i], dx)
            
            # Update particle best
            if score < particle_best_scores[i]:
                particle_best_scores[i] = score
                particle_bests[i] = swarm[i].copy()
                if score < global_best_score:
                    global_best_score = score
                    global_best = swarm[i].copy()
    
    # Local refinement with gradient descent
    final_h = global_best.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        grad = compute_gradient(final_h, dx)
        final_h = final_h - 1e-5 * grad
        final_h = repair_h(final_h, n_points)
        score = evaluate_h(final_h, dx)
        if score < global_best_score:
            global_best_score = score
            global_best = final_h.copy()
    
    return (global_best.tolist(), global_best_score, n_points)
