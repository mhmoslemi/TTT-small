import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
        
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Binary search on C
    c_low = 0.0
    c_high = 0.5
    tolerance = 1e-5
    max_iter = 50
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        c = (c_low + c_high) / 2.0  
        
        # Create optimization variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        
        # Integral constraint
        constraints.append(cp.sum(h) == n_points/2)
        
        # Overlap constraints for all time shifts
        # For each valid k, we have constraint: sum h_i * (1 - h_{i+k}) <= c * dx
        for k in range(1, n_points):
            # This is a sum over valid i: h_i * (1 - h_{i+k})
            # Compute this as a dot product with a shifted version of h
            # Special handling for different shift values
            if k < n_points:
                # Compute sum_{i=0}^{n-k-1} h_i * (1 - h_{i+k})
                # The range of i for which h_{i+k} is valid is [0,n-k-1]
                expr = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                constraints.append(expr <= c * dx)
            else:
                # For very large k, this dot product is empty 
                expr = cp.sum(cp.multiply(h[:0], 1 - h[:0]))  # empty
                constraints.append(expr <= c * dx)
        
        # Objective is just feasibility, so minimize a constant
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                h_sol = project_h(h_sol, n_points)  # Ensure sum = 1
                c5 = evaluate_h(h_sol, dx)
                if c5 < c5_best:
                    h_best = h_sol.copy()
                    c5_best = c5
                c_high = c
            else:
                c_low = c
        except Exception:
            # If optimization fails, try a feasible solution approach
            # A heuristic might be to compute the Lipschitz constant  
            # and adjust C to be conservative
            c_low = c

    # Refine with gradient-based local search for the final best solution
    h_final = h_best.copy()
    c5_final = c5_best
    
    # Simple local search to refine final result
    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Small random perturbations
        h_trial = h_final.copy()
        idxs = np.random.choice(n_points, size=max(2, n_points//10), replace=False)
        for idx in idxs:
            h_trial[idx] += np.random.normal(0, 0.005)
            
        h_trial = project_h(h_trial, n_points)
        c5_trial = evaluate_h(h_trial, dx)
        
        if c5_trial < c5_final:
            h_final = h_trial.copy()
            c5_final = c5_trial
    
    return (h_final.tolist(), c5_final, n_points)
