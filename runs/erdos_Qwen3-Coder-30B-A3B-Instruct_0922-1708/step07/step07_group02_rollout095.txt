import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, k_max, n_points, dx):
    """Compute analytical gradient for the k_max correlation."""
    grad = np.zeros_like(h)
    # For h[i] in the convolution with (1 - h) at lag k_max
    for i in range(n_points):
        j = i + k_max
        if j < n_points:
            grad[i] = (1 - h[j]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Gradient-based optimization with analytical gradients
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Optimization parameters
    eta = 0.01
    eps = 1e-6
    patience = 100
    last_improvement = 0
    max_iter = 100000
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all correlations and find maximum
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        c5_current = corr[k_max] * dx
        
        # Compute analytical gradient for this k_max
        grad = compute_gradient(h_current, k_max, n_points, dx)
        
        # Update h using gradient descent
        h_new = h_current - eta * grad
        
        # Project h to [0,1] and re-scale
        h_new = project_h(h_new, n_points, dx)
        
        # Evaluate new C5
        c5_new = evaluate_h(h_new, dx)
        
        # Check for improvement
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            
            # Update best if improved
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
        # Adaptive learning rate
        if c5_new < c5_current:
            eta = min(0.01, eta * 1.001)
        else:
            eta *= 0.999
            
        # Perturbation to escape local minima
        if i - last_improvement > patience:
            h_perturb = h_current.copy()
            num_perturb = max(1, int(0.01 * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)

            c5_perturb = evaluate_h(h_perturb, dx)
