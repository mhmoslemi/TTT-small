import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, mutation_rate=0.1, delta=0.1):
    """Mutate a step function by perturbing some entries."""
    h_new = h.copy()
    n_points = len(h)
    indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
    for i in indices:
        h_new[i] += np.random.uniform(-delta, delta)
    return repair_h(h_new, n_points)

def crossover_h(h1, h2):
    """Perform segment crossover between two step functions."""
    n_points = len(h1)
    split_point = np.random.randint(1, n_points)
    h_new = np.concatenate([h1[:split_point], h2[split_point:]])
    return h_new

def selection_tournament(population, fitnesses, k=3):
    """Select individuals using tournament selection."""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), k, replace=False)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)

    # Genetic Algorithm Setup
    population_size = 100
    generations = 100
    mutation_rate = 0.1
    elite_size = 10
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        population.append(mutate_h(h_init, mutation_rate))
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(h_init, dx)
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Track best
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitnesses[best_idx]
        
        # Select parents
        parents = selection_tournament(population, fitnesses)
        
        # Create new population
        new_population = []
        
        # Elite preservation
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            p1, p2 = np.random.choice(parents, 2, replace=False)
            child = crossover_h(p1, p2)
            child = mutate_h(child, mutation_rate)
            new_population.append(child)
        
        population = new_population[:population_size]
    
    # Refinement with coordinate-wise gradient descent
    h_refine = best_h.copy()
    c5_refine = best_c5
    learning_rate = 0.001
    eps = 1e-6
    patience = 20
    last_improvement = 0
    
    for i in range(10000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_refine.copy()
        for j in range(n_points):
            # Forward difference
            h_plus = h_refine.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            # Backward difference
            h_minus = h_refine.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            # Gradient estimate
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            h_new[j] = h_refine[j] - learning_rate * grad_j
        
        # Apply repair step
        h_new = repair_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement
        if c5_new < c5_refine:
            h_refine = h_new
            c5_refine = c5_new
            last_improvement = i
        elif i - last_improvement > patience:
            # If no significant improvement, try mutation
            h_refine = mutate_h(h_refine, 0.05, 0.05)
            c5_refine = evaluate_h(h_refine, dx)
            last_improvement = i
    
    # Final check
    c5_final = evaluate_h(h_refine, dx)
    
    return (h_refine.tolist(), c5_final, n_points)
