import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    if 'initial_h_values' in globals():

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)

    # Try to solve with CVXPY
    try:

        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add quadratic constraints for each lag
        full_corr = correlate(np.ones(n_points), np.ones(n_points), mode='full')
        max_lag = len(full_corr) - 1
        
        # For each lag, add constraint: integral <= C5
        for k in range(max_lag):
            # This represents the k-th lag in the correlation
            # For simplicity, we handle this as a linear approximation due to
            # computational complexity; however, we'll try a hybrid approach
            # where we use exact computation for a few iterations
            
        # Simple iterative improvement using gradient descent
        learning_rate = 1e-3
        max_iter = 20000
        best_c5 = float('inf')
        best_h = h.copy()
        
        # Check progress every 500 iterations
        for iteration in range(max_iter):
            if (time.time() - start_time) > budget_s - 5:
                break
                
            # Compute correlation
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

            grad = np.zeros_like(h)

                    h_plus[idx] = 1
                elif h_plus[idx] < 0:
                    h_plus[idx] = 0

                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[idx] = (c5_plus - c5) / eps
                
            # Update step with momentum
            h = h - learning_rate * grad
            
            # Projection to [0,1]
            h = np.clip(h, 0, 1)

            h_sum = np.sum(h)
            target_sum = n_points / 2
            if abs(h_sum - target_sum) > 1e-8:
                h = h * target_sum / h_sum
