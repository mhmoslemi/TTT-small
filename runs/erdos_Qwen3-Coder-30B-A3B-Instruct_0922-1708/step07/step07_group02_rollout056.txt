import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale, n_points):
    """Mutation operator for evolution strategy."""
    mutated = h.copy()
    # Random segment flip
    if np.random.rand() < 0.3:
        start = np.random.randint(n_points)
        end = start + np.random.randint(1, min(n_points//2, n_points-start))
        mutated[start:end] = 1 - mutated[start:end]
    # Add gaussian noise
    noise = np.random.randn(n_points) * mutation_scale
    mutated += noise
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n_points / 2) / current_sum
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # ES parameters
    population_size = 50
    elite_size = population_size // 5
    max_generations = 1000
    initial_mutation_scale = 0.1
    decay_rate = 0.999
    stagnation_limit = 100
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = mutate(h_init, 0, initial_mutation_scale, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_h(h, dx)
        fitness.append(fit)
    
    # Find best solution
    best_idx = np.argmin(fitness)
    h_best = population[best_idx].copy()
    c5_best = fitness[best_idx]
    
    # Optimization loop
    mutation_scale = initial_mutation_scale
    no_improvement_count = 0
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:elite_size]
        non_elite_indices = sorted_indices[elite_size:]
        
        # Create new population
        new_population = [population[i].copy() for i in elite_indices]
        
        # Generate offspring from elite
        for _ in range(population_size - elite_size):
            parent_idx = np.random.choice(elite_indices)
            child = mutate(population[parent_idx], generation, mutation_scale, n_points)
            new_population.append(child)
        
        # Evaluate new population
        fitness = []
        for h in new_population:
            fit = evaluate_h(h, dx)
            fitness.append(fit)
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < c5_best:
            h_best = new_population[best_idx].copy()
            c5_best = fitness[best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Update population and mutation scale
        population = new_population
        if no_improvement_count > stagnation_limit:
            mutation_scale *= 0.7
            no_improvement_count = 0
        else:
            mutation_scale *= decay_rate
        
        # Inject random perturbations occasionally
        if no_improvement_count > stagnation_limit:
            for i in range(min(5, population_size // 10)):
                idx = np.random.randint(population_size)
                h_perturb = mutate(h_best, generation, 0.1, n_points)
                population[idx] = h_perturb
            no_improvement_count = 0
    
    return (h_best.tolist(), c5_best, n_points)
