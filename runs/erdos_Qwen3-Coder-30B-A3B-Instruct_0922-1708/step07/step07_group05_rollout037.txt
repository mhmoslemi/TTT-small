import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular pulse
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)

    # PSO parameters
    pop_size = 100
    max_iter = 400
    w_init = 0.9
    w_min = 0.4
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    inertia_reduce = (w_init - w_min) / max_iter
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for i in range(pop_size):
        # Initialize around the starting point
        if i == 0:
            h = h_init.copy()
        else:
            # Add small random perturbations to the initial solution
            h = h_init + np.random.normal(0, 0.01, n_points)
        h = repair_h(h, n_points)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate_h(h, dx))
    
    # Global best
    best_idx = np.argmin(pbest_values)
    gbest = pbest[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    # PSO iteration
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Update inertia weight
        w = w_init - iter_count * inertia_reduce
        
        # Update each particle
        for i in range(pop_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair the particle
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            value = evaluate_h(particles[i], dx)
            
            # Update personal best
            if value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = value
                
                # Update global best
                if value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = value
        
        # Mutation every 25 iterations
        if iter_count % 25 == 0 and iter_count > 0:
            for i in range(int(pop_size * mutation_rate)):
                idx = np.random.randint(0, pop_size)
                # Mutate 20% of the values
                mutate_indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
                particles[idx][mutate_indices] += np.random.normal(0, mutation_sigma, len(mutate_indices))
                particles[idx] = repair_h(particles[idx], n_points)
                
                # Update personal best if improved
                value = evaluate_h(particles[idx], dx)
                if value < pbest_values[idx]:
                    pbest[idx] = particles[idx].copy()
                    pbest_values[idx] = value
                    
                    if value < gbest_value:
                        gbest = particles[idx].copy()
                        gbest_value = value
    
    return (gbest.tolist(), gbest_value, n_points)
