import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # PSO parameters
    swarm_size = 50
    max_iter = 10000
    inertia_start = 0.9
    inertia_end = 0.4
    cognitive_coeff = 1.5
    social_coeff = 1.5
    velocity_limit = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # One particle from initial_h_values (if available)
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
        swarm.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    else:
        # Create a structured initial solution
        h_init = np.zeros(n_points)
        # Build a multi-peak structure
        peaks = [0.3, 0.7, 0.5, 0.6, 0.4]
        for i, peak in enumerate(peaks):
            if (i * n_points // len(peaks) < n_points):
                start_idx = i * n_points // len(peaks)
                end_idx = (i + 1) * n_points // len(peaks) if i < len(peaks)-1 else n_points
                h_init[start_idx:end_idx] = peak
        h_init = project_h(h_init, n_points, dx)
        swarm.append(h_init.copy())
        velocities.append(np.zeros(n_points))
    
    # Fill the rest of swarm with random/structured particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        swarm.append(h)
        velocities.append(np.zeros(n_points))
        
    # Initialize personal bests and global best
    personal_bests = [evaluate_h(h, dx) for h in swarm]
    personal_best_h = [h.copy() for h in swarm]
    
    global_best_idx = np.argmin(personal_bests)
    global_best = personal_bests[global_best_idx]
    global_best_h = swarm[global_best_idx].copy()
    
    # PSO iteration
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        inertia = inertia_start - (inertia_start - inertia_end) * iter_num / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_update = (cognitive_coeff * r1 * (personal_best_h[i] - swarm[i]) + 
                         social_coeff * r2 * (global_best_h - swarm[i]))
            velocities[i] = inertia * velocities[i] + vel_update
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project onto feasible region
            swarm[i] = project_h(swarm[i], n_points, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if fitness < personal_bests[i]:
                personal_bests[i] = fitness
                personal_best_h[i] = swarm[i].copy()
                
                # Update global best
                if fitness < global_best:
                    global_best = fitness
                    global_best_h = swarm[i].copy()
                    
    return (global_best_h.tolist(), global_best, n_points)
