import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    S = 50  # number of particles
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    T = 1.0  # initial temperature
    cooling_rate = 0.999  # cooling rate for simulated annealing
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_fitness = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        personal_best.append(h_init.copy())
        personal_best_fitness.append(float('inf'))
    else:
        # Initialize with random distribution
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        personal_best.append(h)
        personal_best_fitness.append(float('inf'))
    
    # Initialize remaining particles
    for i in range(S - len(particles)):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        personal_best_fitness.append(float('inf'))
    
    global_best = None
    global_best_fitness = float('inf')
    
    # PSO optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Update each particle
        for i in range(S):
            h = particles[i]
            
            # Compute fitness
            corr = correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = h.copy()
                personal_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = h.copy()
                    global_best_fitness = fitness
            
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            if global_best is not None:
                cognitive = c1 * r1 * (personal_best[i] - h)
                social = c2 * r2 * (global_best - h)
                velocities[i] = w * velocities[i] + cognitive + social
                
                # Apply simulated annealing perturbation with probability T
                if np.random.rand() < T:
                    velocities[i] += np.random.uniform(-0.1, 0.1, n_points)
                
                particles[i] = h + velocities[i]
                
                # Repair constraints
                particles[i] = np.clip(particles[i], 0, 1)
                
                # Rescale to satisfy sum constraint
                h_sum = np.sum(particles[i])
                if abs(h_sum - n_points / 2) > 1e-8:
                    particles[i] = particles[i] * (n_points / 2) / h_sum
            else:
                # If no global best, just use random search
                particles[i] = np.random.rand(n_points)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        
        # Decrease temperature for simulated annealing
        T = max(T * cooling_rate, 0.01)
    
    # Final evaluation
    if global_best is not None:
        corr = correlate(global_best, 1 - global_best, mode='full')
        final_c5 = np.max(corr) * dx
        return (global_best.tolist(), final_c5, n_points)
    else:
        # If no solution found, return fallback
        h_fallback = np.ones(n_points) * 0.5
        corr = correlate(h_fallback, 1 - h_fallback, mode='full')
        final_c5 = np.max(corr) * dx
        return (h_fallback.tolist(), final_c5, n_points)
