import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def create_initial_population(n_points, pop_size):
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        population.append(h_init)
    
    # Create triangular pulses
    while len(population) < pop_size:
        h = np.zeros(n_points)
        peak_idx = np.random.randint(0, n_points)
        peak_x = peak_idx * (2.0 / n_points)
        # Create triangle from 0 to 2 with triangular pulse
        for i in range(n_points):
            x = i * (2.0 / n_points)
            # Triangle centered at peak_x with width 0.5
            width = 0.5
            if abs(x - peak_x) <= width/2:
                h[i] = 1 - 2 * abs(x - peak_x) / width
            elif x < peak_x:
                h[i] = 1 - 2 * (peak_x - x) / width
            else:
                h[i] = 1 - 2 * (x - peak_x) / width
        h = repair_h(h, n_points)
        population.append(h)
    
    return population

def selection(population, fitness):
    # Tournament selection
    selected = []
    pop_size = len(population)
    tournament_size = 3
    for _ in range(pop_size // 2):
        tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
        tournament_fitness = [fitness[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx])
    return selected

def crossover(parent1, parent2):
    # Single point crossover with equal weight to preserve properties
    point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2):
    h = h.copy()
    n_points = len(h)
    num_mutation = int(n_points * mutation_rate)
    mutation_indices = np.random.choice(n_points, num_mutation, replace=False)
    for i in mutation_indices:
        h[i] += np.random.normal(0, 0.05)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # GA parameters
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.5
    patience = 200
    
    # Initialize population
    population = create_initial_population(n_points, pop_size)
    fitness = [evaluate_h(h, dx) for h in population]
    
    # Track the best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_fitness = fitness[best_idx]
    
    last_improvement = 0
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection
        selected = selection(population, fitness)
        
        # Create new population
        new_population = [best_h]  # Elitism
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate and len(selected) >= 2:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                child1, child2 = crossover(parent1, parent2)
                child1 = repair_h(mutate(child1, mutation_rate), n_points)
                child2 = repair_h(mutate(child2, mutation_rate), n_points)
                new_population.append(child1)
                if len(new_population) < pop_size:
                    new_population.append(child2)
            else:
                # Just mutate a random selected individual
                parent = np.random.choice(selected)
                child = repair_h(mutate(parent, mutation_rate), n_points)
                new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitness = [evaluate_h(h, dx) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_fitness:
            best_h = population[current_best_idx]
            best_fitness = fitness[current_best_idx]
            last_improvement = generation
        
    return (best_h.tolist(), best_fitness, n_points)
