import numpy as np
import time
import math
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a more structured initial guess
        h = np.ones(n_points) * 0.5
        # Add some variation
        for i in range(n_points):
            if i < n_points // 4:
                h[i] = 0.2
            elif i < n_points // 2:
                h[i] = 0.8
            elif i < 3 * n_points // 4:
                h[i] = 0.3
            else:
                h[i] = 0.7
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)

    # Annealing parameters
    temperature = 0.1
    cooling_rate = 0.999
    min_temp = 1e-6
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        # Perturb h
        h_new = h.copy()
        
        # Select 2 random indices to modify
        i, j = np.random.randint(0, n_points, 2)
        
        # Compute change
        delta = np.random.uniform(-0.05, 0.05)
        
        # Apply changes ensuring constraints
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain sum constraint
        h_new_sum = np.sum(h_new)
        target_sum = n_points / 2
        h_new = h_new * target_sum / h_new_sum
        
        # Compute overlap integral for both old and new
        corr_old = correlate(h, 1 - h, mode='full')
        c5_old = np.max(corr_old) * dx
        
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept or reject based on Metropolis criterion
        if c5_new < c5_old or np.random.rand() < np.exp(-(c5_new - c5_old) / temperature):
            h = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Update temperature
        temperature = max(min_temp, temperature * cooling_rate)
        
        iteration += 1

    # Final evaluation
    corr_best = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_best) * dx
    
    return (best_h.tolist(), final_c5, n_points)
