import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        # Create a trapezoidal wave that is 0 at edges and peaks at center
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)

    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Phase 1: CCCP for good local optimum
    learning_rate = 0.01
    max_iter1 = 5000
    patience = 500
    last_improvement = 0
    
    for i in range(max_iter1):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Find shift k where overlap is maximum
        corr = correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        
        # Gradient of the overlap function at k_max
        h_shifted = np.roll(h_current, k_max - n_points + 1)
        grad = 1 - h_shifted
        
        # Update with gradient descent on the convex approximation
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept if improvement
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience:
            learning_rate *= 0.95
            last_improvement = i
            
        # Occasionally perturb
        if i % 500 == 0 and i > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
    
    # Phase 2: Cyclic coordinate descent
    max_iter2 = 10000
    learning_rate_cd = 0.1
    patience_cd = 1000
    
    for i in range(max_iter2):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Cyclic coordinate descent
        for j in range(n_points):
            # Find k where overlap is maximum
            corr = correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr)
            
            # Calculate gradient component for h[j]
            h_shifted = np.roll(h_current, k_max - n_points + 1)
            grad_component = 1 - h_shifted[j]
            # More accurate gradient for full convolution
            offset = k_max - n_points + 1
            # Gradient w.r.t. h[i] from shift k_max
            shifted = np.roll(h_current, offset)
            grad_component = 1 - shifted[j]
            
            # Update
            h_new = h_current.copy()
            h_new[j] -= learning_rate_cd * grad_component
            h_new = project_h(h_new, n_points)
            
            # Evaluate and accept
            c5_new = evaluate_h(h_new, dx)
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Adaptive learning rate
        if i - last_improvement > patience_cd:
            learning_rate_cd *= 0.99
            last_improvement = i
            
        # Occasionally perturb to escape local optima
        if i % 1000 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)

    return (h_best.tolist(), c5_best, n_points)
