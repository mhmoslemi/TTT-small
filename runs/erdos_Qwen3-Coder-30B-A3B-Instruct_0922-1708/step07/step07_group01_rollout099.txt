import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Binary search on the bound C
    c_low = 0.0
    c_high = 0.5
    tolerance = 1e-5
    max_iter = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Create a feasibility LP for the current C
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [0 <= h_vars, h_vars <= 1]
        
        # Sum constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Overlap constraints
        for k in range(1, n_points):
            # Sum over valid indices
            valid_indices = range(n_points - k)
            sum_expr = cp.sum([h_vars[i] * (1 - h_vars[i + k]) for i in valid_indices])
            constraints.append(sum_expr <= c_mid * dx)
        
        # Create and solve the problem
        objective = cp.Minimize(0)  # dummy objective
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                # Feasible, try lower C
                c_high = c_mid
                h_feasible = h_vars.value
                if h_feasible is not None:
                    h_feasible = project_h(h_feasible, n_points)
                    c5_feasible = evaluate_h(h_feasible, dx)
                    if c5_feasible < c5_best:
                        h_best = h_feasible
                        c5_best = c5_feasible
            else:
                # Infeasible, try higher C
                c_low = c_mid
        except:
            # In case of solver failure, try a more direct approach
            c_low = c_mid
            
        # Occasionally refine a better solution using local search
        if iteration % 5 == 0:
            h_refined = h_best.copy()
            # Local perturbation with small random steps
            for _ in range(10):
                if time.time() - start_time > budget_s - 5:
                    break
                idx = np.random.randint(n_points)
                h_refined[idx] += np.random.normal(0, 0.01)
                h_refined = project_h(h_refined, n_points)
                c5_refined = evaluate_h(h_refined, dx)
                if c5_refined < c5_best:
                    h_best = h_refined
                    c5_best = c5_refined
    
    # Final local search to refine best vector
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = h_best.copy()
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
