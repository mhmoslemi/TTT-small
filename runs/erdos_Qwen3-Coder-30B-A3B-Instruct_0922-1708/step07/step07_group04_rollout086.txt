import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate(h, rate=0.1, delta=0.1):
    """Apply mutation to a solution."""
    h_new = h.copy()
    mask = np.random.rand(len(h_new)) < rate
    h_new[mask] += np.random.uniform(-delta, delta, np.sum(mask))
    return repair_h(h_new, len(h_new))

def crossover(h1, h2):
    """Perform crossover between two solutions."""
    n = len(h1)
    point = np.random.randint(1, n-1)
    h_new = np.concatenate([h1[:point], h2[point:]])
    return repair_h(h_new, n)

def tournament_selection(population, fitnesses, k=3):
    """Select an individual using tournament selection."""
    idxs = np.random.choice(len(population), size=k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    population_size = 100
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = mutate(h_init, rate=0.1, delta=0.05)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_h(h, dx) for h in population]
    
    best_idx = np.argmin(fitnesses)
    h_best = population[best_idx].copy()
    c5_best = fitnesses[best_idx]
    
    # Evolutionary loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        top_half = sorted_indices[:population_size//2]
        
        # Create new population
        new_population = [population[i] for i in top_half]
        
        # Fill rest of population
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            child = mutate(child, rate=mutation_rate, delta=0.05)
            new_population.append(child)
        
        population = new_population
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Update best
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < c5_best:
            c5_best = fitnesses[best_idx]
            h_best = population[best_idx].copy()
    
    # Final gradient refinement on the best solution
    h_refined = h_best.copy()
    c5_refined = c5_best
    eta = 0.001
    eps = 1e-6
    max_iter = 5000
    patience = 50
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        h_new = h_refined.copy()
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            h_minus = h_refined.copy()
            h_minus[j] -= eps
            h_minus = repair_h(h_minus, n_points)
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            h_new[j] = h_refined[j] - eta * grad_j
            
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_refined:
            h_refined = h_new
            c5_refined = c5_new
            last_improvement = i
        else:
            if i - last_improvement > patience:
                eta *= 0.95
                last_improvement = i
    
    return (h_refined.tolist(), c5_refined, n_points)
