import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize particles for PSO
    S = 50  # number of particles
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
    else:
        particles.append(np.ones(n_points) * 0.5)
        velocities.append(np.random.rand(n_points) * 0.1)
        
    # Fill remaining particles
    for i in range(S - 1):
        h = np.random.rand(n_points) * 0.8 + 0.1
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
    
    # Personal best and global best
    pbest = [p.copy() for p in particles]
    pbest_scores = [float('inf')] * S
    gbest = particles[0].copy()
    gbest_score = float('inf')
    
    # PSO main loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate each particle
        for i in range(S):
            h = particles[i]
            corr = correlate(h, 1 - h, mode='full')
            score = np.max(corr) * dx
            
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = h.copy()
                
            if score < gbest_score:
                gbest_score = score
                gbest = h.copy()
        
        # Update particles
        for i in range(S):
            r1, r2 = np.random.rand(2)
            v = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            
            # Apply bounds to velocity
            v = np.clip(v, -0.2, 0.2)
            
            # Update position
            new_pos = particles[i] + v
            
            # Repair constraints
            new_pos = np.clip(new_pos, 0, 1)
            new_sum = np.sum(new_pos)
            target_sum = n_points / 2
            if abs(new_sum - target_sum) > 1e-8:
                new_pos = new_pos * target_sum / new_sum
                
            particles[i] = new_pos
            velocities[i] = v
            
    # Final evaluation
    corr = correlate(gbest, 1 - gbest, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (gbest.tolist(), final_c5, n_points)
