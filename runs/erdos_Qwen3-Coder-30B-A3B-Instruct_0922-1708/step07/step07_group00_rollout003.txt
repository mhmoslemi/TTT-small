import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.stats import qmc

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Generate initial points using Sobol sequences
    sampler = qmc.Sobol(d=n_points, seed=seed)
    n_initials = 500
    initial_points = sampler.random(n=n_initials)
    
    # Convert to feasible starting points
    for i in range(n_initials):
        if time.time() - start_time > budget_s - 5:
            break
        h_start = initial_points[i] 
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Nelder-Mead optimization
        result = minimize(evaluate_h, h_start, args=(dx,), method='Nelder-Mead',
                          options={'maxiter': 1000, 'disp': False, 'xtol': 1e-6})
        refined_h = result.x
        
        # Gradient-based refinement
        h_refined = refined_h.copy()
        learning_rate = 1e-5
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            # Compute gradient (finite differences)
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                grad[j] = (np.max(corr_plus) - np.max(correlate(h_refined, 1 - h_refined, mode='full'))) / eps
            
            # Update step
            h_refined = h_refined - learning_rate * grad
            h_refined = repair_h(h_refined, n_points)
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            c5_refined = np.max(corr) * dx
            
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
            learning_rate *= 0.995  # Adaptive decay
    
    return (best_h.tolist(), best_c5, n_points)
