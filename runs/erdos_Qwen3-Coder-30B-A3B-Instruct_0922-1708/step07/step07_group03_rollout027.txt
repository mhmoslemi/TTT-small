import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter_barrier = 1000
    max_iter_gd = 10000
    learning_rate = 1e-3
    mu_initial = 1e3
    barrier_decrease_factor = 0.99
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Barrier method phase
    mu = mu_initial
    for barrier_iter in range(max_iter_barrier):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlations and find maximum
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_indices = np.where(corr == np.max(corr))[0]
        max_lag = max_lag_indices[len(max_lag_indices)//2] - (len(h) - 1)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient using adjoint method for the maximum lag
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + max_lag >= 0 and i + max_lag < len(h):
                grad[i] += (1 - h[i + max_lag]) * dx
        
        # Update step with barrier penalty
        h_new = h - learning_rate * grad
        
        # Apply barrier penalty to keep h in [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet constraint
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h_new = h_new * target_sum / h_sum
            
        h = h_new
        
        # Decrease barrier parameter
        mu *= barrier_decrease_factor
    
    # Gradient descent phase
    for gd_iter in range(max_iter_gd):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlations and find maximum
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_indices = np.where(corr == np.max(corr))[0]
        max_lag = max_lag_indices[len(max_lag_indices)//2] - (len(h) - 1)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient using adjoint method
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + max_lag >= 0 and i + max_lag < len(h):
                grad[i] += (1 - h[i + max_lag]) * dx
        
        # Update step
        h = h - learning_rate * grad
        
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to meet constraint
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
