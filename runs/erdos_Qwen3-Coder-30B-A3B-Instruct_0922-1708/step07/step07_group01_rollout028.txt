import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def get_max_shift(h):
    """Get the shift that maximizes the overlap"""
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    # Convert from full correlation index to shift
    n = len(h)
    shift = max_idx - (n - 1)
    return shift

def compute_gradient(h, shift):
    """Compute gradient for the case of max overlap"""
    n = len(h)
    grad = np.zeros(n)
    # For shift k, d/dh_i of the overlap integral is (1-h_{i+k}) 
    # (accounting for cyclic boundary conditions)
    # We only need to compute for the actual shift, so we shift the (1-h) vector
    h_reflected = 1 - h
    # Apply the shift
    shifted_reflected = np.roll(h_reflected, shift)
    # For each position i, the gradient is (1 - h_{i+shift})
    grad = shifted_reflected
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        #Initialize with a simple smooth waveform
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Try multiple initializations
    for trial in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        h_current = h_init.copy()
        
        # Add noise to break symmetry
        noise_magnitude = 0.05
        h_current += np.random.normal(0, noise_magnitude, n_points)
        h_current = project_h(h_current, n_points)
        
        c5_current = evaluate_h(h_current, dx)
        
        # Gradient descent parameters
        lr = 0.02
        patience = 100
        last_improvement = 0
        max_iter = 10000
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Find shift that gives maximum overlap
            max_shift = get_max_shift(h_current)
            
            # Compute exact gradient for that shift
            gradient = compute_gradient(h_current, max_shift)
            
            # Update with gradient
            h_new = h_current - lr * gradient
            h_new = project_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                if c5_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_current
                    last_improvement = i
            else:
                # Occasionally accept worse moves
                if np.random.rand() < 0.05:
                    h_current = h_new
                    c5_current = c5_new
            
            # Adaptive learning rate
            if i - last_improvement > patience:
                lr *= 0.95
                last_improvement = i
            
            # Periodic perturbations
            if i % 500 == 0 and i > 0:
                noise = np.random.normal(0, 0.01, n_points)
                h_current = project_h(h_current + noise, n_points)
                c5_current = evaluate_h(h_current, dx)
                
        # Additional local search
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_new = best_h.copy()
            # Perturb small portion 
            indices = np.random.choice(n_points, max(2, n_points // 30), replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.005)
            
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    return (best_h.tolist(), best_c5, n_points)
