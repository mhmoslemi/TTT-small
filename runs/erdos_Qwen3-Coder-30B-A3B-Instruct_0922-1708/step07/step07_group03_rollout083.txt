import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        # Rescale to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Frank-Wolfe Frank-Wolfe algorithm
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current overlap integral
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient (approximated by finite differences)
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            # Rescale to maintain sum
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
            
        # Solve linear subproblem: minimize grad^T * s s.t. 0<=s<=1, sum(s)=1
        # This is equivalent to setting s_i = 0 if grad_i > 0, 1 if grad_i < 0
        # But we must maintain sum(s) = n_points/2 and 0<=s<=1
        s = np.zeros_like(h)
        sorted_indices = np.argsort(grad)
        remaining_sum = n_points / 2
        for i in sorted_indices:
            if remaining_sum <= 0:
                break
            if grad[i] > 0:
                s[i] = 0
            else:
                s[i] = min(1, remaining_sum)
                remaining_sum -= s[i]
        
        # If remaining_sum > 0 and all s[i] are 1, adjust
        if remaining_sum > 0:
            # Set all s[i] = 1 and rescale
            s = np.ones_like(h)
            s = s * (n_points / 2) / np.sum(s)
            
        # Line search for step size
        # Use backtracking
        gamma = 0.5
        max_backtracks = 20
        for _ in range(max_backtracks):
            h_new = (1 - gamma) * h + gamma * s
            # Ensure it's in [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to correct sum
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < c5 or gamma < 1e-5:
                h = h_new
                break
            gamma *= 0.5
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
