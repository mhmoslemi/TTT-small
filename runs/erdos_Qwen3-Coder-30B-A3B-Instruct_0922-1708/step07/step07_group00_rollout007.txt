import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # PSO parameters
    swarm_size = 100
    max_iter = 10000
    w_start, w_end = 0.9, 0.1
    c1, c2 = 2.0, 2.0
    v_max = 0.1
    
    # Initialize swarm
    particles = np.array([h_init.copy() for _ in range(swarm_size)])
    velocities = np.random.uniform(-v_max, v_max, (swarm_size, n_points))
    personal_best_positions = particles.copy()
    personal_best_scores = np.array([evaluate_h(p, dx) for p in particles])
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # Optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best_positions[i] - particles[i]) +
                           c2 * r2 * (global_best_position - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair position
            particles[i] = repair_h(particles[i], n_points)
            
            # Evaluate
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = particles[i].copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = particles[i].copy()
        
        # Local search every 50 iterations
        if iter_num % 50 == 0 and iter_num > 0:
            # Gradient-based refinement for global best
            refined_pos = global_best_position.copy()
            for _ in range(200):
                if time.time() - start_time > budget_s - 5:
                    break
                # Compute gradient via finite differences
                eps = 1e-6
                grad = np.zeros(n_points)
                base_corr = correlate(refined_pos, 1 - refined_pos, mode='full')
                max_k_idx = np.argmax(base_corr)
                
                for j in range(n_points):
                    pert_pos = refined_pos.copy()
                    pert_pos[j] += eps
                    pert_pos = repair_h(pert_pos, n_points)
                    pert_corr = correlate(pert_pos, 1 - pert_pos, mode='full')
                    grad[j] = (pert_corr[max_k_idx] - base_corr[max_k_idx]) / eps
                
                # Update using gradient
                refined_pos = refined_pos - 1e-5 * grad
                refined_pos = repair_h(refined_pos, n_points)
                corr = correlate(refined_pos, 1 - refined_pos, mode='full')
                new_score = np.max(corr) * dx
                
                if new_score < global_best_score:
                    global_best_score = new_score
                    global_best_position = refined_pos.copy()
    
    # Final check of global best
    final_corr = correlate(global_best_position, 1 - global_best_position, mode='full')
    final_score = np.max(final_corr) * dx
    
    return (global_best_position.tolist(), final_score, n_points)
