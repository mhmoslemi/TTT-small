import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Create a piecewise linear shape localized near center
        h_init = np.where((x >= 0.5) & (x <= 1.5), 
                         1.0 - np.abs(x - 1.0), 0.0)
        h_init = project_h(h_init, n_points)

    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Main optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Try different perturbations and local searches
        if iteration % 50 == 0:  # Periodic restart to avoid local minima
            h_current = h_init.copy()
            h_current = project_h(h_current, n_points)
            h_best = h_current.copy()
            c5_best = evaluate_h(h_best, dx)
            
        # Small random perturbations
        h_current = h_best.copy()
        for i in range(n_points):
            # Perturb small portion of coordinates
            if np.random.rand() < 0.2:
                h_current[i] += np.random.normal(0, 0.01)
                
        h_current = project_h(h_current, n_points)
        c5_current = evaluate_h(h_current, dx)
        
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current

        # Occasionally apply local search to improve further
        if iteration % 100 == 0:
            h_search = h_best.copy()
            # Make several small changes and see if we improve
            for _ in range(10):
                idx = np.random.randint(0, n_points)
                perturbation = np.random.normal(0, 0.01)
                h_search[idx] = max(0, min(1, h_search[idx] + perturbation))
            h_search = project_h(h_search, n_points)
            c5_search = evaluate_h(h_search, dx)
            if c5_search < c5_best:
                h_best = h_search
                c5_best = c5_search
                
    # Final check of the best solution
    c5_final = evaluate_h(h_best, dx)
    h_final = project_h(h_best, n_points)
    c5_final = evaluate_h(h_final, dx)
    
    return (h_final.tolist(), c5_final, n_points)
