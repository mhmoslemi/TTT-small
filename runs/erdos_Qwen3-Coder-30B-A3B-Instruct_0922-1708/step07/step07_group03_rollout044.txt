import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.random.rand(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)

    # PSO parameters
    S = 50  # swarm size
    max_iter = 5000
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    T = 1.0  # initial temperature for SA
    cooling_rate = 0.999
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_best = []
    personal_best_fitness = []
    
    for i in range(S):
        if i == 0 and 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_target = np.linspace(0, 2, n_points)
                h_init = np.interp(x_target, x_orig, h_init)
            particle = h_init * (n_points / 2) / np.sum(h_init)
        else:
            particle = np.random.rand(n_points) * 0.5
            particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
        personal_best.append(particle.copy())
        
        # Compute initial fitness
        corr = correlate(particle, 1 - particle, mode='full')
        fitness = np.max(corr) * dx
        personal_best_fitness.append(fitness)

    # Find global best
    global_best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[global_best_idx].copy()
    global_best_fitness = personal_best_fitness[global_best_idx]

    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Update particles
        for i in range(S):
            particle = particles[i]
            velocity = velocities[i]
            pbest = personal_best[i]
            
            # PSO velocity update
            r1, r2 = np.random.rand(2)
            velocity = (w * velocity + 
                       c1 * r1 * (pbest - particle) + 
                       c2 * r2 * (global_best - particle))
            
            # SA adjustment
            if np.random.rand() < T:
                velocity += np.random.rand(n_points) * 0.2 - 0.1
                
            # Update position
            particle += velocity
            
            # Apply constraints
            particle = np.clip(particle, 0, 1)
            particle_sum = np.sum(particle)
            particle = particle * (n_points / 2) / particle_sum
            
            # Store updated particle
            particles[i] = particle
            velocities[i] = velocity
            
            # Compute fitness
            corr = correlate(particle, 1 - particle, mode='full')
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particle.copy()
                personal_best_fitness[i] = fitness
                
                # Update global best if needed
                if fitness < global_best_fitness:
                    global_best = particle.copy()
                    global_best_fitness = fitness
                    
        # Cooling schedule
        T = max(T * cooling_rate, 0.01)
        
    # Final evaluation of global best
    corr = correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (global_best.tolist(), final_c5, n_points)
