import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def parameterize_h(params, n_points):
    """Convert parameters to h function."""
    K = len(params) // 2
    heights = params[:K]
    boundaries = np.concatenate([[0], params[K:]])
    boundaries = np.cumsum(boundaries)
    boundaries = np.clip(boundaries, 0, 2)
    
    # Ensure boundaries are in order and within [0, 2]
    boundaries = np.sort(boundaries)
    boundaries = np.clip(boundaries, 0, 2)
    
    # Add end boundary
    boundaries = np.append(boundaries, 2)
    
    # Create h values
    h_vals = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(K):
        start_idx = int(boundaries[i] / dx)
        end_idx = int(boundaries[i+1] / dx)
        if start_idx < end_idx:
            h_vals[start_idx:end_idx] = heights[i]
    
    return h_vals

def compute_gradient(h, dx):
    """Compute gradient of the maximum overlap with respect to h."""
    n = len(h)
    corr = correlate(h, 1 - h, mode='full')
    max_k_idx = np.argmax(corr)
    max_k = max_k_idx - (n - 1)
    
    # Approximate gradient
    grad = np.zeros(n)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_h(h_plus, n)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (corr_plus[max_k_idx] - corr[max_k_idx]) / eps
    
    # Normalize the gradient
    grad_norm = np.linalg.norm(grad) + 1e-12
    grad = grad / grad_norm
    
    return grad
