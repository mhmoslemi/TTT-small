import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def mutate(h, generation, mutation_scale):
    """Mutate a solution with adaptive scale."""
    mutated = h.copy()
    n = len(h)
    # Random segment flip
    if np.random.rand() < 0.3:
        start = np.random.randint(n)
        end = start + np.random.randint(1, min(n//2, n-start))
        mutated[start:end] = 1 - mutated[start:end]
    # Add gaussian noise
    noise = np.random.randn(n) * mutation_scale
    mutated += noise
    # Project to feasible region
    mutated = np.clip(mutated, 0, 1)
    current_sum = mutated.sum()
    if current_sum != 0:
        mutated *= (n / 2) / current_sum
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Evolution strategy parameters
    pop_size = 50
    elite_size = max(1, pop_size // 5)
    mutation_scale = 0.1
    decay_rate = 0.999
    
    # Initialize population
    population = [h_init]
    for _ in range(pop_size - 1):
        h_mut = mutate(h_init, 0, mutation_scale)
        population.append(h_mut)
    
    # Best solution tracking
    h_best = h_init.copy()
    c5_best = evaluate_h(h_best, dx)
    
    # Evolution loop
    generation = 0
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 5:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < c5_best:
            h_best = population[0].copy()
            c5_best = fitness[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Selection: keep top elite
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        # Mutation of elite members
        for i in range(len(elite)):
            h_mut = mutate(elite[i], generation, mutation_scale)
            new_population.append(h_mut)
            
        # Fill remaining with random mutations
        while len(new_population) < pop_size:
            parent = elite[np.random.randint(len(elite))]
            h_mut = mutate(parent, generation, mutation_scale)
            new_population.append(h_mut)
        
        # Update population
        population = new_population
        
        # Adaptive mutation
        mutation_scale *= decay_rate
        
        # Reintroduce diversity if stuck
        if stagnation_count > max_stagnation:
            for i in range(len(population)):
                if np.random.rand() < 0.1:
                    population[i] = mutate(h_best, generation, mutation_scale * 10)
            stagnation_count = 0
    
    return (h_best.tolist(), c5_best, n_points)
