import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Create a piecewise linear function with a triangular pulse
        h_init = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h_init[i] = 0.0
            elif x < 1.5:
                h_init[i] = (x - 0.5) / 1.0
            else:
                h_init[i] = 1.0
        h_init = repair_h(h_init, n_points)
    
    # PSO parameters
    n_particles = 100
    inertia_w = 0.9
    c1 = 2.0
    c2 = 2.0
    mutation_rate = 0.1
    mutation_sigma = 0.1
    
    # Initialize particles
    particles = np.tile(h_init, (n_particles, 1))
    for i in range(n_particles):
        for j in range(n_points):
            if np.random.rand() < 0.1:
                particles[i, j] += np.random.normal(0, mutation_sigma)
    particles = np.array([repair_h(p, n_points) for p in particles])
    
    # Initialize velocities
    velocities = np.zeros((n_particles, n_points))
    
    # Initialize personal best
    pbest = particles.copy()
    pbest_scores = np.array([evaluate_h(p, dx) for p in particles])
    
    # Initialize global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    max_iter = 400
    patience = 100
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Update inertia weight
        inertia_w = 0.9 - (0.5 * i / max_iter)
        
        for j in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[j] = (inertia_w * velocities[j] + 
                             c1 * r1 * (pbest[j] - particles[j]) + 
                             c2 * r2 * (gbest - particles[j]))
            
            # Add mutation noise
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, mutation_sigma, n_points)
                velocities[j] += mutation
            
            # Update position
            particles[j] += velocities[j]
            particles[j] = repair_h(particles[j], n_points)
            
            # Evaluate
            score = evaluate_h(particles[j], dx)
            
            # Update personal best
            if score < pbest_scores[j]:
                pbest[j] = particles[j].copy()
                pbest_scores[j] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = particles[j].copy()
                    gbest_score = score
                    last_improvement = i
        
        # Mutation every 25 iterations for diversity
        if i % 25 == 0 and i > 0:
            for j in range(n_particles):
                if np.random.rand() < mutation_rate:
                    indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
                    particles[j, indices] += np.random.normal(0, mutation_sigma, len(indices))
                    particles[j] = repair_h(particles[j], n_points)
    
    return (gbest.tolist(), gbest_score, n_points)
