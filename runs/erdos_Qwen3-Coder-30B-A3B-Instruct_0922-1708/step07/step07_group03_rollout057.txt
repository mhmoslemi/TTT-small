import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Simple initialization
        h = np.ones(n_points) * 0.5
    
    # Frank-Wolfe optimization
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current overlap
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Compute gradient direction
        # This is the derivative of overlap with respect to h
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Compute finite difference gradient approximation
            eps = 1e-6
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            h_plus = h_plus * (n_points / 2) / h_plus_sum
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Solve linear subproblem: minimize grad^T * s subject to constraints
        # s in [0,1], sum(s) = n_points/2
        # This can be solved directly by setting s = 0 for negative gradient,
        # s = 1 for positive gradient, and distributing the rest
        s = np.zeros(n_points)
        idx_sorted = np.argsort(-grad)  # descending order
        
        # Initialize with s = 1 for positions with largest positive gradient
        remaining_sum = n_points / 2
        for i in idx_sorted:
            if grad[i] <= 0:
                s[i] = 0
                continue
            if remaining_sum <= 0:
                s[i] = 0
                continue
            s[i] = min(1, remaining_sum)
            remaining_sum -= s[i]
            
        # If sum != n_points/2, adjust
        s_sum = np.sum(s)
        if abs(s_sum - n_points / 2) > 1e-8:
            # scale to target sum
            s = s * (n_points / 2) / s_sum
            
        # Line search
        alpha = 0.5
        max_alpha = 1.0
        min_alpha = 1e-5
        best_alpha = 0.0
        
        for _ in range(10):  # max line search steps
            h_new = (1 - alpha) * h + alpha * s
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            h_new = h_new * (n_points / 2) / h_new_sum
            
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < c5:
                best_alpha = alpha
                break
            else:
                alpha /= 2.0
                if alpha < min_alpha:
                    break

        # Update
        alpha = best_alpha if best_alpha > 0 else 0.1
        h = (1 - alpha) * h + alpha * s
        
        # Project to maintain sum
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum   
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
