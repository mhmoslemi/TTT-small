import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

        x = np.linspace(0, 2, n_points)

    # Let's instead use a simple convex relaxation approach.
    
    # Define the objective as minimizing the sum of squares of correlations
    # subject to etching in equally spaced steps the max constraint
    # For now, let's go with a simple approach: minimize sum of squared overlaps
    
    h_current = h_init.copy()
    
    # Solve the optimization problem using CVXPy (this can be fast or slow, we want fast)
    # For now, skip the QP approach and proceed to a gradient descent approach that works better for this problem
    
    # Phase 2: Gradient and proximal descent
    h_current = h_init.copy()

        for i in range(n_points):

            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            
            h_minus = h_current.copy()
            h_minus[i] -= eps
            h_minus = repair_h(h_minus, n_points)
            
            c5_plus = evaluate_h(h_plus, dx)

            grad[i] = (c5_plus - c5_minus) / (2 * eps)

        h_new = h_current - learning_rate * grad
