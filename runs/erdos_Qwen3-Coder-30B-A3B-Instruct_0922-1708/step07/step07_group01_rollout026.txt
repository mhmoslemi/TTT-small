import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, k_max, n_points):
    """Compute gradient of the overlap at shift k_max"""
    # Create shifted version of h
    h_shifted = np.zeros(2 * n_points - 1)
    h_shifted[n_points-1-k_max:n_points-1-k_max+n_points] = h
    # Gradient is (1 - h_shifted) for the current h positions
    grad = 1 - h_shifted[n_points-1:n_points-1+n_points]
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 * (1 + np.cos(np.pi * x))  # Cosine taper
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = int(1e6)
    patience = 100
    last_improvement = 0
    
    # Multiple restart strategy
    num_restarts = 5
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        lr = learning_rate
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute all correlations and find max overlap shift
            corr = correlate(h_current, 1 - h_current, mode='full')
            k_max = np.argmax(corr)
            
            # Compute exact gradient
            grad = compute_gradient(h_current, k_max, n_points)
            
            # Update with gradient
            h_new = h_current - lr * grad
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            # Accept move if improvement
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                if c5_current < c5_best:
                    h_best = h_current.copy()
                    c5_best = c5_current
                    last_improvement = i
            else:
                # Occasionally accept worse moves (simulated annealing)
                if np.random.rand() < 0.001:
                    h_current = h_new.copy()
                    c5_current = c5_new
            
            # Adaptive learning rate
            if i - last_improvement > patience:
                lr *= 0.95
                last_improvement = i
            
            # Periodic perturbations to escape local minima
            if i % 500 == 0 and i > 0:
                h_current = project_h(h_current + np.random.normal(0, 0.005, n_points), n_points)
                c5_current = evaluate_h(h_current, dx)
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_best.copy()
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
