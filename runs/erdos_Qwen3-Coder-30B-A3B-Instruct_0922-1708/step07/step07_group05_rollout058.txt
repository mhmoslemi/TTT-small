import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def create_individual(n_points, initial_h=None):
    if initial_h is not None and len(initial_h) == n_points:
        return repair_h(np.array(initial_h), n_points)
    else:
        # Create a triangular pulse centered at 1.0
        h = np.zeros(n_points)
        center = n_points // 2
        half_width = n_points // 4
        for i in range(max(0, center - half_width), min(n_points, center + half_width)):
            h[i] = 1.0 - abs(i - center) / half_width
        return repair_h(h, n_points)

def crossover(parent1, parent2, crossover_rate=0.5):
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points - 1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(individual, mutation_rate=0.2, mutation_strength=0.1):
    n_points = len(individual)
    mutated = individual.copy()
    
    num_mutations = max(1, int(n_points * mutation_rate))
    indices = np.random.choice(n_points, num_mutations, replace=False)
    
    for idx in indices:
        mutated[idx] += np.random.normal(0, mutation_strength)
    
    return repair_h(mutated, len(mutated))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Population size
    pop_size = 50
    max_generations = 1000
    patience = 200
    
    # Create initial population
    population = []
    initial_h = globals().get('initial_h_values', None)
    
    # Add the initial solution if available
    if initial_h is not None:
        population.append(create_individual(n_points, initial_h))
    
    # Fill remaining population with random individuals
    while len(population) < pop_size:
        population.append(create_individual(n_points))
    
    # Evaluate initial population
    fitness_scores = [evaluate_h(ind, dx) for ind in population]
    best_idx = np.argmin(fitness_scores)
    best_individual = population[best_idx].copy()
    best_fitness = fitness_scores[best_idx]
    
    # Track best so far
    best_found = best_individual.copy()
    best_c5 = best_fitness
    generation_no_improvement = 0
    
    # Main GA loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Elitism: keep top 10% of population
        new_population = population[:pop_size // 10]
        
        # Selection (tournament)
        def tournament_select():
            k = 3
            tournament_indices = np.random.choice(pop_size, k, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            return population[winner_idx]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            parent1 = tournament_select()
            parent2 = tournament_select()
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        new_population = new_population[:pop_size]
        
        # Evaluate new population
        fitness_scores = [evaluate_h(ind, dx) for ind in new_population]
        
        # Update best solution
        min_idx = np.argmin(fitness_scores)
        if fitness_scores[min_idx] < best_fitness:
            best_fitness = fitness_scores[min_idx]
            best_individual = new_population[min_idx].copy()
            generation_no_improvement = 0
        else:
            generation_no_improvement += 1
            
        # Check for early termination
        if generation_no_improvement > patience:
            break
            
        population = new_population
        
        # Update global best
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_found = best_individual.copy()
    
    return (best_found.tolist(), best_c5, n_points)
