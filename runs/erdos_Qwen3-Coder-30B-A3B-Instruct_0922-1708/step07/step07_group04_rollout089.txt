import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def crossover(parent1, parent2):
    """Perform segment-based crossover."""
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, delta=0.1):
    """Perform mutation by perturbing some entries."""
    h_new = h.copy()
    indices = np.random.choice(len(h), int(len(h) * mutation_rate), replace=False)
    h_new[indices] += np.random.uniform(-delta, delta, len(indices))
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a triangular waveform
        x = np.linspace(0, 2, n_points)
        h_init = 1.0 - np.abs(x - 1.0)
        h_init = repair_h(h_init, n_points)
    
    # Create initial population
    population.append(h_init)
    for _ in range(population_size - 1):
        h = mutate(h_init, mutation_rate=0.1, delta=0.1)
        h = repair_h(h, n_points)
        population.append(h)
    
    # GA parameters
    generations = 100
    elite_size = 10
    mutation_rate = 0.1
    tournament_size = 5
    
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(-c5)  # Negative because we minimize C5
        
        # Tournament selection
        new_population = []
        
        # Elitism
        sorted_indices = np.argsort(fitnesses)[::-1][:elite_size]
        for i in sorted_indices:
            new_population.append(population[i])
            
        # Create offspring
        while len(new_population) < population_size:
            # Tournament selection for parents
            parent1_idx = np.random.choice(len(population), tournament_size)
            parent2_idx = np.random.choice(len(population), tournament_size)
            parent1 = population[np.argmax([fitnesses[i] for i in parent1_idx])]
            parent2 = population[np.argmax([fitnesses[i] for i in parent2_idx])]
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, mutation_rate, 0.1)
            child2 = mutate(child2, mutation_rate, 0.1)
            
            # Repair
            child1 = repair_h(child1, n_points)
            child2 = repair_h(child2, n_points)
            
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Track best solution
        current_best_idx = np.argmax(fitnesses)
        current_best_h = population[current_best_idx]
        current_best_c5 = -fitnesses[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
    
    # Final refinement using gradient-based method on best solution
    if best_h is not None:
        h_current = best_h.copy()
        c5_current = evaluate_h(h_current, dx)
        
        # Refinement using gradient descent
        eta = 0.001
        eps = 1e-6
        max_iter = 10000
        patience = 50
        last_improvement = 0
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_new = h_current.copy()
            improved = False
            for j in range(n_points):
                # Forward difference
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                # Backward difference
                h_minus = h_current.copy()
                h_minus[j] -= eps
                h_minus = repair_h(h_minus, n_points)
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                # Gradient estimate
                grad_j = (c5_plus - c5_minus) / (2 * eps)
                
                # Update
                h_new[j] = h_current[j] - eta * grad_j
                
            # Apply repair step
            h_new = repair_h(h_new, n_points)
            
            # Evaluate new solution
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                improved = True
                
            # Update best if improved
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                last_improvement = i
                
            # Adjust learning rate if no improvement
            if not improved and i - last_improvement > patience:
                eta *= 0.95
                last_improvement = i
                
    return (best_h.tolist(), best_c5, n_points)
