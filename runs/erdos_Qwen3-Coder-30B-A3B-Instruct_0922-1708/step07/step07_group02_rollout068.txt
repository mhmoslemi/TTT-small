import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points, dx)
    else:
        # Initialize with a structured feasible solution
        h_init = np.zeros(n_points)
        # Create a piecewise constant function with two regions
        mid = n_points // 2
        h_init[:mid] = 0.5
        h_init[mid:] = 0.0
        # Adjust to sum to n_points/2
        h_init = project_h(h_init, n_points, dx)
    
    # Try a mixed approach: start with simple solution,
    # then try a convex relaxation approach to warm-start

    # Resolve using SOCP relaxation 
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    
    # Best solution
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Try warm-starting with convex relaxation approach
    try:
        # Create the convex problem with CVXPY
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Objective: minimize the maximum overlap
        # We solve the relaxed problem where we constrain the correlation norm
        # Use a direct approach: minimize sum of variables while trying to assist
        # the max (which we'll compute via the actual probe later)
        
        # Put constraints on the convex problem
        prob = cp.Problem(cp.Minimize(cp.sum(h_var)), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_convex = h_var.value
            h_convex = project_h(h_convex, n_points, dx)
            c5_convex = evaluate_h(h_convex, dx)
            if c5_convex < c5_current:
                h_current = h_convex
                c5_current = c5_convex
                h_best = h_current.copy()
                c5_best = c5_current
    except Exception as e:
        pass  # Skip convex relaxation if it fails
    
    # Then perform local search refinements on best solution
    # Preliminary improvement with gradient estimation
    eta = 0.01
    eps = 1e-6
    patience = 50
    last_improvement = 0
    max_iter = 50000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_new = h_current.copy()
        improved = False
        
        # Randomize order for exploration
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for j in indices:
            h_plus = h_new.copy()
            h_minus = h_new.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = project_h(h_plus, n_points, dx)
            h_minus = project_h(h_minus, n_points, dx)
            
            c5_plus = evaluate_h(h_plus, dx)
            c5_minus = evaluate_h(h_minus, dx)
            
            grad_j = (c5_plus - c5_minus) / (2 * eps)
            
            h_new[j] -= eta * grad_j
            h_new[j] = max(0, min(1, h_new[j]))
            
        h_new = project_h(h_new, n_points, dx)
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            improved = True
            
            if c5_current < c5_best:
                h_best = h_current.copy()
                c5_best = c5_current
                last_improvement = i
                
        if not improved:
            eta *= 0.999
        else:
            eta = min(0.01, eta * 1.001)
            
        # Early escape
        if i - last_improvement > patience:
            break
                
    return (h_best.tolist(), c5_best, n_points)
