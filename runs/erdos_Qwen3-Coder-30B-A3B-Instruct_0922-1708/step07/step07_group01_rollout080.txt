import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
    else:
        # Initialize with a structured waveform to get a good starting point
        h_init = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Try a double-cosine construction
        h_init = 0.5 * (1 + np.cos(np.pi * x)) * (1 + np.cos(np.pi * x / 2))
        h_init = project_h(h_init, n_points)
    
    # Initialize
    h_current = h_init.copy()
    c5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    c5_best = c5_current
    
    # Simple local search with different strategies
    max_iter = int(1e6)
    patience = 500
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Random perturbation to escape local minima
        if i % 100 == 0:
            h_perturbed = h_current.copy()
            # Add small noise
            for j in range(n_points):
                if np.random.rand() < 0.3:
                    delta = np.random.normal(0, 0.01)
                    h_perturbed[j] += delta
            h_perturbed = project_h(h_perturbed, n_points)
            c5_pert = evaluate_h(h_perturbed, dx)
            if c5_pert < c5_current:
                h_current = h_perturbed
                c5_current = c5_pert
        
        # Local search via coordinate-wise perturbations
        h_new = h_current.copy()
        # Perturb a few coordinates
        indices = np.random.choice(n_points, max(2, n_points // 20), replace=False)
        eps = 0.005
        for idx in indices:
            # Try both positive and negative small steps
            h_plus = h_new.copy()
            h_plus[idx] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            
            h_minus = h_new.copy()
            h_minus[idx] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_h(h_minus, dx)
            
            # Choose the better one
            if c5_plus < c5_minus and c5_plus < c5_current:
                h_new = h_plus
                c5_current = c5_plus
            elif c5_minus < c5_current:
                h_new = h_minus
                c5_current = c5_minus
                
        # Accept the change
        h_current = h_new
        c5_current = evaluate_h(h_new, dx)
        
        # Update best
        if c5_current < c5_best:
            h_best = h_current.copy()
            c5_best = c5_current
            last_improvement = i
            
        # Early stopping if no improvement for a while
        if i - last_improvement > patience:
            if np.random.rand() < 0.05: # occasionally restart
                h_current = h_best.copy()
                c5_current = c5_best
                last_improvement = i
            else:
                break
    
    # Final local refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        h_new = h_best.copy()
        # Perturb small portion of coordinates
        indices = np.random.choice(n_points, max(2, n_points // 50), replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.005)
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
    
    return (h_best.tolist(), c5_best, n_points)
