import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible set with ∫h dx = 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-12:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Multiple initializations to find better solutions
    initializations = []
    
    # Use existing initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = project_h(h_init, n_points)
        initializations.append(h_init)
    
    # Add some structured patterns
    # Sine pattern
    x = np.linspace(0, 2, n_points)
    h_sine = 0.5 * (1 + np.sin(np.pi * x - np.pi/2))
    h_sine = project_h(h_sine, n_points)
    initializations.append(h_sine)
    
    # Trapezoidal pattern
    h_trap = np.zeros(n_points)
    mid = n_points // 2
    h_trap[mid-10:mid+10] = 1.0
    h_trap = project_h(h_trap, n_points)
    initializations.append(h_trap)
    
    # Random pattern
    h_rand = np.random.rand(n_points)
    h_rand = project_h(h_rand, n_points)
    initializations.append(h_rand)
    
    # Piecewise constant
    h_const = np.zeros(n_points)
    h_const[:n_points//2] = 1.0
    h_const = project_h(h_const, n_points)
    initializations.append(h_const)
    
    # Run optimization from each initialization
    best_h = None
    best_c5 = float('inf')
    
    for i, h_init in enumerate(initializations):
        h_current = h_init.copy()
        c5_current = evaluate_h(h_current, dx)
        
        # Slightly perturb for diversity
        if i > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_current = project_h(h_current + noise, n_points)
            c5_current = evaluate_h(h_current, dx)
            
        # Simple gradient descent
        learning_rate = 0.02
        patience = 200
        best_iter = 0
        for iter_count in range(10000):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Find the shift that maximizes overlap
            corr = correlate(h_current, 1 - h_current, mode='full')
            max_k = np.argmax(corr)
            max_corr = corr[max_k]
            
            # Compute exact gradient at the position of max overlap
            # Gradient is direction that decreases the maximum overlap
            shift = max_k - (2 * n_points - 1) // 2
            if shift >= 0:
                grad = 1 - np.concatenate([h_current[shift:], np.zeros(shift)])
            else:
                grad = np.concatenate([np.zeros(-shift), h_current[:shift]])
                
            # Update using gradient descent
            h_new = h_current - learning_rate * grad
            h_new = project_h(h_new, n_points)
            
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new.copy()
                c5_current = c5_new
                best_iter = iter_count
            else:
                # Go back to best solution if no improvement
                if iter_count - best_iter > patience:
                    break
                
            # Adaptive learning rate
            if iter_count % 500 == 0 and iter_count > 0:
                learning_rate *= 0.95
                
        # Update global best
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
    
    # Final local search
    if best_h is not None:
        h_current = best_h.copy()
        for _ in range(2000):
            if time.time() - start_time > budget_s - 5:
                break
                
            h_new = h_current.copy()
            # Random perturbations
            indices = np.random.choice(n_points, max(2, n_points // 20), replace=False)
            for idx in indices:
                h_new[idx] += np.random.normal(0, 0.002)
            
            h_new = project_h(h_new, n_points)
            c5_new = evaluate_h(h_new, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
