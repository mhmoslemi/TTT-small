import numpy as np, time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def shape_h(n, a):
    dx = 2.0 / n
    p = 1.0 / (2.0 - a) if a > 0 else 0.5
    # compute x positions at midpoints maybe
    x = (np.arange(n) + 0.5) * dx
    h = np.empty(n)
    left_end = a
    right_start = 2 - a
    # ramp condition
    ramp_left = (x < left_end)
    ramp_right = (x > right_start)
    plateau = (~ramp_left) & (~ramp_right)
    # left ramp: linear: h = p * (x / a)
    h[ramp_left] = p * (x[ramp_left] / a)
    # right ramp: h = p * ((2 - x) / a)
    h[ramp_right] = p * ((2 - x[ramp_right]) / a)
    # plateau
    h[plateau] = p
    # handle a=0 separately: uniform
    if a == 0:
        h[:] = 0.5
    # ensure constraints: clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust sum to n/2
    total = h.sum()
    target = n / 2
    if total != 0:
        factor = target / total
        h *= factor
        h = np.clip(h, 0.0, 1.0)
        # after clipping, sum may deviate; adjust again:
        total2 = h.sum()
        if abs(total2 - target) > 1e-8:

            # distribute evenly to entries that are not at bounds

                h = np.clip(h, 0.0, 1.0)

def hill_climb_continuous(h, steps, rng):
    n = len(h)
    best = h.copy()
    best_val = evaluate_c5(best)
    for _ in range(steps):
        i, j = rng.integers(0, n, size=2, endpoint=False)
        # generate delta small relative to value range
        delta = (rng.random() - 0.5) * 0.1  # [-0.05,0.05]
        hi, hj = best[i], best[j]
        # propose new values while maintaining sum constraint
        hi_new = hi + delta
        hj_new = hj - delta
        # check bounds
        if hi_new < 0 or hi_new > 1 or hj_new < 0 or hj_new > 1:
            continue
        candidate = best.copy()
        candidate[i] = hi_new
        candidate[j] = hj_new
        val = evaluate_c5(candidate)
        if val < best_val:
            best = candidate
            best_val = val
        # Optionally break if time out.
    return best, best_val
```

def subgradient_descent(n, iters, step):
    h = np.full(n, 0.5)   # satisfies sum = n/2
    target = n/2
    for it in range(iters):
        corr = np.correlate(h, 1 - h, mode='full')
        s_idx = np.argmax(corr)  # index in correlation array, shift = s_idx - (n-1)
        s = s_idx - (n - 1)
        # compute gradient:
        g = np.ones_like(h)
        # h_{i+s}: shift h by s (right shift). For indices out-of-range set 0.
        if s >= 0:
            # h_{i+s} corresponds to h shifted left
            g[:n - s] -= h[s:]   # i from 0 to n-1-s: subtract h[i+s]
            # h_{i-s} corresponds to h shifted right
            g[s:] -= h[:n - s]   # i from s to n-1: subtract h[i-s]
        else:
            s_abs = -s
            g[s_abs:] -= h[:n - s_abs]  # h_{i+s}
            g[:n - s_abs] -= h[s_abs:]  # h_{i-s}
        # gradient step
        h = h - step * g
        # project onto [0,1]
        h = np.clip(h, 0.0, 1.0)
        # enforce sum constraint:
        cur_sum = h.sum()
        diff = target - cur_sum
        # Distribute diff uniformly among free variables:
        # find indices not at bound
        mask = (h > 1e-12) & (h < 1 - 1e-12)
        if mask.any():
            # distribute diff equally
            incr = diff / mask.sum()
            h[mask] += incr
            # after this may violate bounds; clip again and repeat
            h = np.clip(h, 0.0, 1.0)
        else:
            # all at bounds; maybe adjust all uniformly:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        # optionally reduce step size
    return h, evaluate_c5(h)

from typing import Tuple

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_bounds_and_sum(v, target):
    # binary search for tau
    lo = -1.0
    hi = 1.0
    for _ in range(40):
        tau = (lo + hi) / 2.0

        s = w.sum()
        if s < target:
            lo = tau
        else:
            hi = tau
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    return w

def gradient_max_shift(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    g = np.ones_like(h)
    if shift >= 0:

        if s < n:
            g[:n - s] -= h[s:]
            g[s:] -= h[:n - s]

        s = -shift

            g[s:] -= h[:n - s]
            g[:n - s] -= h[s:]
    return g, shift, corr[idx] * (2.0 / n)

    for it in range(max_iters):

def generate_edge_constant_shape(n, edge_frac):
    # edge_frac fraction of total length (2) for each edge zero region
    total_length = 2.0
    a = edge_frac * total_length  # actual length
    dx = total_length / n
    e = int(round(a / dx))
    if e * 2 > n:
        e = n // 2
    interior = n - 2 * e
    if interior <= 0:

        return np.full(n, 0.5)
    p = (n / 2.0) / interior  # target sum = n/2

    h = project_to_bounds_and_sum(h, n/2.0)

def generate_ramp_shape(n, a):

        return np.full(n, 0.5)
    if a >= 2:
        a = 2 - 1e-12

    x = (np.arange(n) + 0.5) * dx
    left = x < a
    right = x > (2.0 - a)

    h = np.empty(n, dtype=float)

    h[right] = p * ((2.0 - x[right]) / a)
    h[middle] = p

    h = project_to_bounds_and_sum(h, n/2.0)

import time
import numpy as np

def _evaluate_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] while enforcing sum(v)=target."""
    # binary search for the shift τ such that sum(clamp(v+τ,0,1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(40):
        tau = (lo + hi) * 0.5
        w = np.clip(v + tau, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = tau
        else:
            hi = tau
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_of_max_shift(h: np.ndarray):
    """Gradient of the currently maximal overlap term."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # actual integer shift
    g = np.ones_like(h)
    if shift >= 0:
        s = shift
        if s < n:
            g[: n - s] -= h[s:]
            g[s:] -= h[: n - s]
    else:
        s = -shift
        if s < n:
            g[s:] -= h[: n - s]
            g[: n - s] -= h[s:]
    return g, shift


def _subgradient_descent(h0: np.ndarray, max_iters: int, step0: float,
                         rng: np.random.Generator, start_time: float,
                         budget: float):
    """Projected sub‑gradient descent for the min‑max problem."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _evaluate_c5(best_h)

    step = step0
    for it in range(max_iters):
        # time guard
        if time.time() - start_time > budget * 0.98:
            break

        g, _ = _gradient_of_max_shift(h)
        if np.linalg.norm(g) < 1e-12:
            # inject a tiny random direction when the gradient vanishes
            g = rng.normal(scale=0.01, size=n)

        h = h - step * g
        h = _project_to_bounds_and_sum(h, target)

        val = _evaluate_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        step *= 0.9995          # slow decay
    return best_h, best_val


def _shape_edge_constant(n: int, edge_frac: float):
    """Zero on each side (fraction of total length), constant inside."""
    total_len = 2.0
    a = edge_frac * total_len                 # actual edge length
    dx = total_len / n
    e = int(round(a / dx))                    # number of points on each edge
    e = min(e, n // 2)
    interior = n - 2 * e
    if interior <= 0:
        return np.full(n, 0.5)
    p = (n / 2.0) / interior                   # height of interior plateau
    h = np.zeros(n)
    h[e : n - e] = p
    return _project_to_bounds_and_sum(h, n / 2.0)


def _shape_ramp(n: int, a: float):
    """Linear ramps of length a on each side, flat plateau in the middle."""
    if a <= 0.0:
        return np.full(n, 0.5)
    a = min(a, 1.999999)                       # avoid division by zero
    dx = 2.0 / n
    p = 1.0 / (2.0 - a)                        # peak value (≤1 for a≤1)
    x = (np.arange(n) + 0.5) * dx
    left = x < a
    right = x > (2.0 - a)
    middle = ~left & ~right
    h = np.empty(n, dtype=float)
    h[left] = p * (x[left] / a)
    h[right] = p * ((2.0 - x[right]) / a)
    h[middle] = p
    h = np.clip(h, 0.0, 1.0)
    return _project_to_bounds_and_sum(h, n / 2.0)
