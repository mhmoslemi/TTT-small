def gradient_step(vec):
    n = len(vec)
    val, corr = compute_overlap(vec)
    idx_max = int(np.argmax(corr))
    shift = idx_max - (n - 1)  # k in indices
    # compute forward/backward contributions
    forward = np.zeros(n)
    backward = np.zeros(n)
    if shift > 0:
        forward[:n - shift] = vec[shift:]
        backward[shift:] = vec[:n - shift]
    elif shift < 0:
        s = -shift
        forward[s:] = vec[:n - s]
        backward[:n - s] = vec[s:]
    else:
        # shift == 0
        forward = vec
        backward = vec
    grad = 1.0 - forward - backward
    # i (max grad) will decrease; j (min grad) increase
    i = np.argmax(grad)
    j = np.argmin(grad)
    if grad[i] <= grad[j] + 1e-12:
        return None, val  # cannot improve
    max_delta = min(vec[i], 1.0 - vec[j])
    if max_delta <= 1e-12:
        return None, val
    # choose delta proportionally to grad difference
    delta = max_delta * 0.5  # maybe constant fraction
    # apply
    vec_new = vec.copy()
    vec_new[i] -= delta
    vec_new[j] += delta
    new_val, _ = compute_overlap(vec_new)

import time
import numpy as np
import random

def _evaluate(h):
    """Return the C5 value for a step vector h."""
    n = h.size
    dx = 2.0 / n
    # correlation with zero padding (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def _gradient_step(h):
    """Perform a single gradient‐based transfer move on h.
    Returns (new_h, new_val) if an improving move is found, else (None, cur_val)."""
    n = h.size
    cur_val, corr = _evaluate(h)
    # index of the worst shift (largest overlap)
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # integer shift k (in units of dx)
    # forward/backward contributions of the shift
    forward = np.zeros(n)
    backward = np.zeros(n)
    if shift > 0:
        forward[: n - shift] = h[shift:]
        backward[shift:] = h[: n - shift]
    elif shift < 0:
        s = -shift
        forward[s:] = h[: n - s]
        backward[: n - s] = h[s:]
    else:  # shift == 0
        forward = h
        backward = h
    # gradient of the active term (1 - forward - backward)
    grad = 1.0 - forward - backward
    i = int(np.argmax(grad))      # position we would like to decrease
    j = int(np.argmin(grad))      # position we would like to increase
    if grad[i] <= grad[j] + 1e-12:
        return None, cur_val
    # maximum admissible transfer keeping [0,1] bounds
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 1e-14:
        return None, cur_val
    # try a moderate step
    delta = 0.5 * max_delta
    h2 = h.copy()
    h2[i] -= delta
    h2[j] += delta
    new_val, _ = _evaluate(h2)
    if new_val < cur_val:
        return h2, new_val
    # fallback: try a smaller step
    delta = 0.1 * max_delta
    h2[i] = h[i] - delta
    h2[j] = h[j] + delta
    new_val, _ = _evaluate(h2)
    if new_val < cur_val:
        return h2, new_val
    return None, cur_val

def _random_transfer(h):
    """Randomly move mass between two coordinates while preserving the sum."""
    n = h.size
    i, j = random.sample(range(n), 2)
    low = max(-h[i], h[j] - 1.0)
    high = min(1.0 - h[i], h[j])
    if low >= high:
        return None, None
    delta = random.uniform(low, high)
    h2 = h.copy()
    h2[i] += delta
    h2[j] -= delta
    val, _ = _evaluate(h2)
    return h2, val

def _make_feasible(h):
    """Clip to [0,1] and enforce the exact sum = n/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # Simple redistribution of the residual on free variables
    for _ in range(1000):
        if abs(diff) < 1e-12:
            break
        # pick a random index that can absorb the needed change
        idx = random.randrange(n)
        if diff > 0:  # need to decrease sum
            dec = min(h[idx], diff)
            if dec > 0:
                h[idx] -= dec
                diff -= dec
        else:         # need to increase sum
            inc = min(1.0 - h[idx], -diff)
            if inc > 0:
                h[idx] += inc
                diff += inc
    # final safety clip
    h = np.clip(h, 0.0, 1.0)
    return h

def _optimize(start_h, time_limit):
    """Return the best vector found starting from start_h within time_limit seconds."""
    h = start_h.copy()
    best_h = h.copy()
    best_val, _ = _evaluate(best_h)
    cur_val = best_val
    no_improve_steps = 0
    end_time = time.time() + time_limit
    while time.time() < end_time:
        # try deterministic gradient step
        step = _gradient_step(h)
        if step[0] is not None:
            h, cur_val = step
            if cur_val < best_val - 1e-12:
                best_val, best_h = cur_val, h.copy()
                no_improve_steps = 0
            else:
                no_improve_steps += 1
            continue
        # fall back to a small random transfer
        h2, val2 = _random_transfer(h)
        if h2 is not None and val2 < cur_val:
            h, cur_val = h2, val2
            if cur_val < best_val - 1e-12:
                best_val, best_h = cur_val, h.copy()
                no_improve_steps = 0
            else:
                no_improve_steps += 1
            continue
        # occasional random restart if stagnating
        no_improve_steps += 1
        if no_improve_steps >= 800:
            # generate a fresh feasible point (binary half‑filled)
            n = h.size
            half = n // 2
            fresh = np.zeros(n, dtype=float)
            fresh[:half] = 1.0
            if n % 2:
                fresh[half] = 0.5
            np.random.shuffle(fresh)
            fresh = _make_feasible(fresh)
            h, cur_val = fresh, _evaluate(fresh)[0]
            no_improve_steps = 0
    # final feasibility pass
    best_h = _make_feasible(best_h)
    best_val, _ = _evaluate(best_h)
    return best_h, float(best_val)

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # ----- initial construction -------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: a simple cosine shape (average 0.5, integral 1)
        n0 = 45
        xs = np.linspace(0.0, 2.0, n0, endpoint=False)
        base = 0.5 + 0.5 * np.cos(np.pi * xs)
        base = _make_feasible(base)

    # we will try two resolutions: the original length and a doubled one
    candidates = []

    # 1) original length
    candidates.append(base)

    # 2) doubled resolution (repeat each entry)
    if base.size * 2 <= 1000:
        doubled = np.repeat(base, 2)
        doubled = _make_feasible(doubled)
        candidates.append(doubled)

    # allocate time proportionally
    per_run = budget_s / (len(candidates) + 0.5)  # a little extra for safety
    best_h, best_val = None, float('inf')
    for vec in candidates:
        h_opt, val_opt = _optimize(vec, per_run * 0.95)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt

    # final sanity check
    best_h = _make_feasible(best_h)
    best_val, _ = _evaluate(best_h)

    return (best_h.tolist(), float(best_val), int(best_h.size))
