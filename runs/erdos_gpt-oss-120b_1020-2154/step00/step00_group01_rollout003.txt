import time
import math
import numpy as np

def _project_onto_feasible(v: np.ndarray, total: float) -> np.ndarray:
    """Project vector v onto the set {0<=x<=1, sum x = total}."""
    lo = -np.max(v) - 1.0          # guarantees sum=0 after clipping
    hi = 1.0 - np.min(v) + 1.0     # guarantees sum=n after clipping
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < total:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x) (1‑h(x+k)) dx  (discrete version)."""
    dx = 2.0 / h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random point in the feasible polytope."""
    v = rng.random(n)
    return _project_onto_feasible(v, n / 2.0)


def _gradient_worst_shift(h: np.ndarray) -> tuple:
    """
    Return the gradient of the worst (max) shift term.
    The worst shift is the one where the correlation h·h is minimal,
    i.e. where the C5 value is attained.
    """
    n = h.size
    corr = np.correlate(h, h, mode="full")
    s_idx = np.argmin(corr)                 # shift giving minimal overlap
    s = s_idx - (n - 1)                     # actual integer shift
    grad = np.ones_like(h)                  # d/dh_i of Σ h_i (1‑h_{i‑s})
    if s >= 0:
        grad[s:] -= h[: n - s]              # h_{i‑s}
        grad[: n - s] -= h[s:]              # h_{i+s}
    else:
        t = -s
        grad[: n - t] -= h[t:]              # h_{i+s}
        grad[t:] -= h[: n - t]              # h_{i‑s}
    return grad, s_idx


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] (discretised) to minimise C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # initialise
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 48                       # even size – convenient for binary‑like patterns
        h0 = _random_feasible(n, rng)

    total = n / 2.0                 # required sum of h_i
    best_h = h0.copy()
    best_val = _c5_value(best_h)

    # ------------------------------------------------------------------
    # main optimisation loop
    while time.time() - start_time < budget_s * 0.94:
        # random restart with some probability
        if rng.random() < 0.25:
            cur = _random_feasible(n, rng)
        else:
            cur = best_h.copy()

        cur_val = _c5_value(cur)

        # inner gradient descent
        for _ in range(2500):
            if time.time() - start_time >= budget_s * 0.94:
                break

            grad, _ = _gradient_worst_shift(cur)

            # line search
            step = 0.8
            improved = False
            while step > 1e-8:
                trial = cur - step * grad
                trial = _project_onto_feasible(trial, total)
                trial_val = _c5_value(trial)
                if trial_val < cur_val - 1e-12:
                    cur, cur_val = trial, trial_val
                    improved = True
                    break
                step *= 0.5

            if not improved:
                # occasional random swap‑like perturbation
                i, j = rng.choice(n, 2, replace=False)
                eps = min(0.1, cur[i], 1.0 - cur[j])
                if eps > 1e-8:
                    trial = cur.copy()
                    trial[i] -= eps
                    trial[j] += eps
                    trial_val = _c5_value(trial)
                    if trial_val < cur_val:

            # keep global best
            if cur_val < best_val:

                # stop early if we already beat the target record
                if best_val <= 0.3808:

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_onto_feasible(best_h, total)
    best_val = _c5_value(best_h)

    return best_h, float(best_val), int(n)
