def improve_worst_shift(h, step):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1-h, mode='full') * dx
    max_idx = np.argmax(corr)
    shift = max_idx - (N - 1)
    k = abs(shift)
    if k == 0:
        return h, corr[max_idx]  # cannot improve
    # overlapping region
    if shift >= 0:
        h_i = h[:N - k]
        h_ipk = h[k:]
    else:
        # negative shift; i+k means j = i+shift negative, but we just use absolute.
        h_i = h[k:]
        h_ipk = h[:N - k]
    # produce bad/good idx
    bad = np.where((h_i > 1e-8) & (h_ipk < 1e-8))[0]
    good = np.where((h_i < 1 - 1e-8) & (h_ipk > 1e-8))[0]
    if len(bad) == 0 or len(good) == 0:
        return h, corr[max_idx]
    i = bad[np.random.randint(len(bad))]
    j = good[np.random.randint(len(good))]
    # idx i corresponds to position in h_i (depends on shift sign)
    if shift >= 0:
        i_h = i
        j_h = j
    else:
        i_h = i + k
        j_h = j + k
    max_delta = min(step, h[i_h], 1 - h[j_h])
    if max_delta <= 1e-12:
        return h, corr[max_idx]
    delta = np.random.rand() * max_delta
    h_new = h.copy()
    h_new[i_h] -= delta
    h_new[j_h] += delta
    # ensure sum unchanged: delta transferred same amount
    # compute new corr
    new_corr = np.correlate(h_new, 1 - h_new, mode='full') * dx
    new_c5 = np.max(new_corr)
    if new_c5 <= corr[max_idx]:
        return h_new, new_c5
    else:
        # reject move
        return h, corr[max_idx]
