import time
import math
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function given by h."""
    # np.correlate implements linear cross‑correlation with zero padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / h.size))

def repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h so that sum(h) == target while keeping 0 ≤ h ≤ 1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute the deficit / surplus greedily
    idxs = np.arange(h.size)
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 5000:
        attempts += 1
        if diff > 0:
            # need to increase some entry
            cand = idxs[h < 1.0 - 1e-12]
            if cand.size == 0:
                break
            i = int(np.random.choice(cand))
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            # need to decrease some entry
            cand = idxs[h > 1e-12]
            if cand.size == 0:
                break
            i = int(np.random.choice(cand))
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    # final clip for safety
    np.clip(h, 0.0, 1.0, out=h)
    return h

def random_binary_vector(n: int) -> np.ndarray:
    """Binary vector with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def simulated_annealing(
    h0: np.ndarray,
    budget_s: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """Simple SA that respects the box and sum constraints."""
    n = h0.size
    target_sum = n / 2.0
    h = h0.copy()
    cur_score = compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    T0 = 0.02               # initial temperature
    Tmin = 1e-6
    it = 0
    while True:
        if time.time() - start_time > budget_s - 0.5:
            break
        it += 1
        # linear temperature schedule
        elapsed = time.time() - start_time
        T = max(Tmin, T0 * (1.0 - elapsed / budget_s))

        # propose a transfer of mass between two positions
        i = random.randrange(n)
        while h[i] <= 1e-12:
            i = random.randrange(n)
        j = random.randrange(n)
        while h[j] >= 1.0 - 1e-12:
            j = random.randrange(n)

        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0.0:
            continue
        # step size scales with temperature
        step_factor = max(0.01, T / T0)
        delta = delta_max * step_factor * random.random()

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        new_score = compute_c5(h_new)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / (T + 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            h = h_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = h_new.copy()

        # occasional full‑re‑mutation (helps escape deep traps)
        if random.random() < 0.0005:
            cand = random_binary_vector(n)
            cand_score = compute_c5(cand)
            if cand_score < cur_score:
                h = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_h = cand.copy()

        # periodic sum‑re‑projection to eliminate drift
        if it % 3000 == 0:
            s = h.sum()
            if abs(s - target_sum) > 1e-9:
                h = repair_sum(h, target_sum)
                cur_score = compute_c5(h)
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()

    return best_h, best_score

def local_refine(h: np.ndarray, cur_score: float, steps: int = 20000) -> tuple[np.ndarray, float]:
    """Greedy descent using tiny transfers."""
    n = h.size
    target_sum = n / 2.0
    for _ in range(steps):
        i = random.randrange(n)
        if h[i] <= 1e-12:
            continue
        j = random.randrange(n)
        if h[j] >= 1.0 - 1e-12:
            continue
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0.0:
            continue
        delta = delta_max * 0.05  # small deterministic step
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < cur_score - 1e-12:
            h = h_new
            cur_score = new_score
    # final repair (should already be exact)
    h = repair_sum(h, target_sum)
    cur_score = compute_c5(h)
    return h, cur_score

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) that beats the current record."""
    random.seed(seed)
    np.random.seed(seed)

    # Desired resolution – 200 points gives a good trade‑off between quality and speed.
    n_points = 200
    target_sum = n_points / 2.0

    # Attempt to seed from the provided construction if it exists.
    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        try:
            base = np.asarray(init_vec, dtype=np.float64)
            if base.size > 0:
                # repeat / truncate to required length
                reps = int(np.ceil(n_points / base.size))
                h0 = np.tile(base, reps)[:n_points].copy()
                h0 = np.clip(h0, 0.0, 1.0)
                h0 = repair_sum(h0, target_sum)
            else:
                raise ValueError
        except Exception:
            h0 = random_binary_vector(n_points)
    else:
        h0 = random_binary_vector(n_points)

    start = time.time()
    best_h, best_score = simulated_annealing(h0, budget_s, start)

    # Final greedy fine‑tuning
    best_h, best_score = local_refine(best_h, best_score, steps=50000)

    # Safety final projection (guarantees feasibility)
    best_h = repair_sum(best_h, target_sum)
    best_score = compute_c5(best_h)

    # Return values in the expected format
    return (best_h.tolist(), best_score, n_points)
