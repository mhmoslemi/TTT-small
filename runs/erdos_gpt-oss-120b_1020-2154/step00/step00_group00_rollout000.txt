import time
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] (discretised) that
    minimizes C5 = max_k ∫ h(x)*(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise a feasible h (0≤h≤1, sum(h)=n/2)
    # ------------------------------------------------------------------
    # use provided initial_h_values if available, otherwise start from scratch
    try:
        init = np.asarray(initial_h_values, dtype=float)  # type: ignore[name-defined]
        have_init = True
    except Exception:
        have_init = False

    # target number of points (even for exact sum constraint)
    target_n = 200
    if target_n % 2:
        target_n += 1

    if have_init:
        # upscale / downscale the initial array to target_n by linear interpolation
        n0 = init.shape[0]
        if n0 != target_n:
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, target_n, endpoint=False)
            h = np.interp(x_new, x_old, init)
        else:
            h = init.copy()
    else:
        # simple binary start: half ones, half zeros, shuffled
        h = np.zeros(target_n, dtype=float)
        h[: target_n // 2] = 1.0
        rng.shuffle(h)

    n = target_n
    dx = 2.0 / n
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # 2. Project onto the feasible set (bounds + exact sum)
    # ------------------------------------------------------------------
    def project_to_feasible(v):
        v = np.clip(v, 0.0, 1.0)
        cur = v.sum()
        diff = target_sum - cur
        # distribute the remaining mass greedily
        # iterate over random indices until diff is exhausted
        idxs = rng.permutation(n)
        for idx in idxs:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                allowed = 1.0 - v[idx]
                step = min(diff, allowed, 0.1)
                v[idx] += step
                diff -= step
            else:
                allowed = v[idx]
                step = min(-diff, allowed, 0.1)
                v[idx] -= step
                diff += step
        # tiny correction if needed
        if abs(v.sum() - target_sum) > 1e-9:
            v += (target_sum - v.sum()) / n
            v = np.clip(v, 0.0, 1.0)
        return v

    h = project_to_feasible(h)

    # ------------------------------------------------------------------
    # 3. Helper to compute C5 for a given h
    # ------------------------------------------------------------------
    def compute_c5(arr):
        conv = np.correlate(arr, 1.0 - arr, mode="full") * dx
        return conv.max()

    best_h = h.copy()
    best_val = compute_c5(best_h)

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    iter_count = 0
    # parameters for the annealing schedule
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s * 0.99:
            break

        # linear cooling of the maximal allowed transfer size
        progress = elapsed / budget_s
        max_transfer = 0.2 * (1.0 - progress) + 0.001 * progress

        # propose a transfer of mass δ from i to j (i≠j)
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # feasible maximal δ given bounds
        cap = min(max_transfer, h[i], 1.0 - h[j])
        if cap <= 1e-12:
            continue
        delta = rng.random() * cap
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        val = compute_c5(h_new)

        # acceptance rule
        if val < best_val - 1e-12:
            h = h_new
            best_h = h_new.copy()
            best_val = val
        else:
            # temperature that decays to a very small value
            T = 0.05 * (1.0 - progress) + 0.001 * progress
            if rng.random() < np.exp(-(val - best_val) / T):
                h = h_new

        # occasional random restart if we stagnate too long
        iter_count += 1
        if iter_count % 20000 == 0:
            # keep the best found so far, but re‑seed the current point randomly
            h = best_h.copy()
            # apply a few random swaps to escape flat regions
            for _ in range(50):
                a, b = rng.integers(n, size=2)
                if a == b:
                    continue
                cap = min(0.05, h[a], 1.0 - h[b])
                if cap > 0:
                    d = rng.random() * cap
                    h[a] -= d
                    h[b] += d

    # ------------------------------------------------------------------
    # 5. Final projection and evaluation
    # ------------------------------------------------------------------
    best_h = project_to_feasible(best_h)
    best_val = compute_c5(best_h)

    return best_h, float(best_val), n
