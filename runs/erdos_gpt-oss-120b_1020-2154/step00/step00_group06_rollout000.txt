import numpy as np
import time
import math


def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def _project_feasible(h: np.ndarray, target_sum: float) -> None:
    """
    Fast alternating–projection onto the box [0,1] and the affine hyperplane
    sum(h)=target_sum. Modifies h in place.
    """
    np.clip(h, 0.0, 1.0, out=h)
    n = h.shape[0]
    for _ in range(10):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)


def _project_exact(h: np.ndarray, target_sum: float) -> None:
    """
    Exact Euclidean projection of h onto {x: 0<=x_i<=1, sum x_i = target_sum}
    using a simple bisection on the Lagrange multiplier λ.
    Modifies h in place.
    """
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(h - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    np.copyto(h, np.clip(h - lam, 0.0, 1.0))


def _subgradient_optimize(
    n: int,
    init_h: np.ndarray,
    rng: np.random.Generator,
    start_time: float,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the maximal correlation.
    Returns the best vector seen and its C5 value.
    """
    target = n / 2.0
    h = init_h.copy()
    _project_feasible(h, target)

    best_h = h.copy()
    best_val = compute_c5(h)

    it = 0
    while True:
        if time.time() - start_time > time_limit - 0.5:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        max_corr = float(corr[idx_max])
        val = max_corr * (2.0 / n)

        if val < best_val:
            best_val = val
            best_h = h.copy()

        # shift producing the maximum
        s = idx_max - (n - 1)

        grad = np.zeros_like(h)
        if s >= 0:
            L = n - s
            grad[:L] += (1.0 - h[s:])
            grad[s:] -= h[:L]
        else:
            t = -s
            L = n - t
            grad[t:] += (1.0 - h[:L])
            grad[:L] -= h[t:]

        # diminishing step size
        alpha = 0.1 / math.sqrt(it + 1.0)
        h -= alpha * grad
        _project_feasible(h, target)

        it += 1

    return best_h, best_val


def _local_improve(
    h: np.ndarray,
    cur_val: float,
    rng: np.random.Generator,
    start_time: float,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """
    Greedy random pairwise mass transfers that keep the sum unchanged.
    Accepts only improving moves.
    """
    n = h.shape[0]
    best_h = h.copy()
    best_val = cur_val

    while True:
        if time.time() - start_time > time_limit - 0.2:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # try moving mass from i to j
        max_delta = min(best_h[i], 1.0 - best_h[j])
        if max_delta <= 0.0:
            # try opposite direction
            max_delta = min(best_h[j], 1.0 - best_h[i])
            if max_delta <= 0.0:
                continue
            i, j = j, i  # swap so we always move from i to j

        delta = max_delta * rng.random() * 0.5  # up to half of the feasible amount
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        val_new = compute_c5(h_new)
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_h = h_new

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimises a step function h on [0,2] to obtain a low upper bound for C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    time_limit = float(budget_s)

    # ------------------------------------------------------------------
    # 1.  Baseline – use parent solution if it exists
    # ------------------------------------------------------------------
    best_h = None
    best_val = float("inf")
    best_n = None

    try:
        parent = np.array(initial_h_values, copy=True)  # type: ignore[name-defined]
        n_parent = parent.shape[0]
        _project_exact(parent, n_parent / 2.0)
        val_parent = compute_c5(parent)
        best_h, best_val, best_n = parent, val_parent, n_parent
    except Exception:
        pass  # no parent available

    # ------------------------------------------------------------------
    # 2.  Search over several discretisations
    # ------------------------------------------------------------------
    candidate_ns = [80, 96, 108, 120, 144, 168, 192, 210, 240, 280]
    # ensure the parent size (if any) is tried first
    if best_n is not None and best_n not in candidate_ns:
        candidate_ns = [best_n] + candidate_ns

    for n in candidate_ns:
        if time.time() - start > time_limit - 2.0:
            break

        target_sum = n / 2.0
        restarts = 3 if n != best_n else 5

                h0 = best_h.copy()

            h_opt, val_opt = _subgradient_optimize(n, h0, rng, start, time_limit)

            h_opt, val_opt = _local_improve(h_opt, val_opt, rng, start, time_limit)

            # final exact projection (ensures exact feasibility)
            _project_exact(h_opt, target_sum)
            val_opt = compute_c5(h_opt)

            if val_opt < best_val:
                best_h, best_val, best_n = h_opt, val_opt, n

    # ------------------------------------------------------------------
    # 3.  Fallback in the extremely unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_h is None:

    # final sanity projection and value recomputation
    _project_exact(best_h, best_n / 2.0)
    best_val = compute_c5(best_h)

    return best_h, best_val, best_n
