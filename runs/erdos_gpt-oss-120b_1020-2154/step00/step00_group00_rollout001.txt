def project_capped_simplex(v, S, max_iter=100):
    lo = np.min(v) - 1
    hi = np.max(v) + 1
    for _ in range(max_iter):
        lam = (lo + hi) / 2
        h = np.clip(v - lam, 0, 1)
        s = h.sum()
        if abs(s - S) < 1e-12:
            return h
        if s > S:
            lo = lam
        else:
            hi = lam
    # after loop, compute final h
    h = np.clip(v - lam, 0, 1)
    # adjust diff
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        # adjust on any free variable
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            h[free[0]] += diff
          # ensure within bounds
          h = np.clip(h,0,1)
    return h

import time, numpy as np
from scipy.optimize import differential_evolution

def project_capped_simplex(v, S, tol=1e-12, max_iter=60):
    # solve min ||h-v||^2 s.t. 0<=h_i<=1, sum h_i = S
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if abs(s - S) <= tol:
            return h
        if s > S:
            lo = lam
        else:
            hi = lam
    h = np.clip(v - lam, 0.0, 1.0)
    # small adjustment if needed
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size > 0:
            h[free[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_of_h(h, dx):
    # return max overlap integral
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    end_time = start + budget_s - 0.1
    rng = np.random.default_rng(seed)
    # Determine dimensionality
    try:
        init_h = np.asarray(initial_h_values, dtype=float)
        n = init_h.size
    except NameError:
        n = 44
        init_h = rng.random(n)
    dx = 2.0 / n
    S = n / 2.0  # required sum of h

    # Ensure initial feasible
    init_h = project_capped_simplex(init_h, S)

    # Objective for DE (scaled C5)
    def obj(x):
        h = project_capped_simplex(x, S)
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    # Build initial population
    popsize = 15
    dim = n
    pop = rng.random((popsize, dim))
    for i in range(popsize):
        pop[i] = project_capped_simplex(pop[i], S)
    # Replace first individual with provided initial
    pop[0] = init_h.copy()

    # Callback to enforce time limit
    def de_callback(xk, convergence=None):
        return time.time() >= end_time

    bounds = [(0.0, 1.0)] * n
    # Run DE with generous iteration count; stop early via callback
    result = differential_evolution(
        obj,
        bounds,
        init=pop,
        maxiter=2000,
        popsize=popsize,
        polish=False,
        updating='deferred',
        callback=de_callback,
        seed=seed,
        disp=False,
    )

    best_x = result.x
    best_h = project_capped_simplex(best_x, S)
    best_val = c5_of_h(best_h, dx)

    # If initial point already better, keep it
    init_val = c5_of_h(init_h, dx)
    if init_val < best_val:
        best_h, best_val = init_h, init_val

    # Local refinement using random perturbations
    step = 0.4

import time
import numpy as np
from scipy.optimize import differential_evolution

def _project_capped_simplex(v, S, tol=1e-12, max_iter=60):
    """Euclidean projection of v onto {0<=h_i<=1, sum(h)=S}."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if abs(s - S) <= tol:
            return h
        if s > S:
            lo = lam
        else:
            hi = lam
    h = np.clip(v - lam, 0.0, 1.0)
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            h[free[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h


def _c5_of_h(h, dx):
    """Maximum overlap integral for a step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.2  # safety margin
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine dimension, sum constraint and initial point
    # ------------------------------------------------------------------
    try:
        init_h = np.asarray(initial_h_values, dtype=float)
        n = init_h.size
    except NameError:
        n = 44
        init_h = rng.random(n)
    dx = 2.0 / n
    S = n / 2.0                      # required sum of h

    init_h = _project_capped_simplex(init_h, S)   # guarantee feasibility
    best_h = init_h.copy()
    best_val = _c5_of_h(best_h, dx)

    # ------------------------------------------------------------------
    # DE objective (scaled C5)
    # ------------------------------------------------------------------
    def _obj(x):
        h = _project_capped_simplex(x, S)
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

    # initial population: random feasible vectors + the provided one
    popsize = 15
    pop = rng.random((popsize, n))
    for i in range(popsize):
        pop[i] = _project_capped_simplex(pop[i], S)
    pop[0] = best_h.copy()          # seed with the known construction

    bounds = [(0.0, 1.0)] * n

    def _de_stop(_xk, _conv=None):
        return time.time() >= deadline

    # ------------------------------------------------------------------
    # Run Differential Evolution (will stop early if time runs out)
    # ------------------------------------------------------------------
    result = differential_evolution(
        _obj,
        bounds,
        init=pop,
        maxiter=2000,
        popsize=popsize,
        polish=False,
        updating='deferred',
        callback=_de_stop,
        seed=seed,
        disp=False,
    )
    # update best solution from DE
    h_de = _project_capped_simplex(result.x, S)
    val_de = _c5_of_h(h_de, dx)
    if val_de < best_val:
        best_h, best_val = h_de, val_de

    # ------------------------------------------------------------------
    # Simple local refinement: pairwise swaps with decreasing step size
    # ------------------------------------------------------------------
    step = 0.4
    while time.time() < deadline - 0.5:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 2.0 * step
        trial = best_h.copy()
        trial[i] += delta
        trial[j] -= delta
        trial = _project_capped_simplex(trial, S)
        val_trial = _c5_of_h(trial, dx)
        if val_trial + 1e-12 < best_val:
            best_h, best_val = trial, val_trial
            step = min(step * 1.1, 1.0)
        else:
            step *= 0.995

    # final feasibility enforcement
    best_h = _project_capped_simplex(best_h, S)

    return (best_h, float(best_val), n)
