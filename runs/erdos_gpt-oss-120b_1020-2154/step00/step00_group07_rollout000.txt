import time
import math
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Return the C5 value (max overlap) for a binary vector h."""
    n = h.shape[0]
    # correlation of h with (1-h); np.correlate is fast in C.
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function (binary vector) that yields a small C5 value.

    Returns
    -------
    h_values : np.ndarray
        Array of values in [0,1] (here binary) satisfying the constraints.
    c5_bound : float
        The achieved C5 bound for the returned vector.
    n_points : int
        Number of discretisation points (length of the array).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1.  Prepare an initial feasible binary vector.
    # ------------------------------------------------------------------
    best_h = None
    best_c5 = float("inf")

    try:
        init = initial_h_values   # pre‑imported by the harness
        n_init = len(init)
        # If it already is a binary vector with the correct number of ones, use it.
        if np.all((init == 0) | (init == 1)) and init.sum() == n_init // 2:
            cur = init.astype(np.int8).copy()
            cur_c5 = _c5_of(cur)
            best_h, best_c5 = cur, cur_c5
        else:
            init = None
    except Exception:     # initial_h_values not supplied
        init = None

    # Choose a working length. 200 gives a good trade‑off between flexibility
    # and evaluation speed.  If a binary initial vector of another length exists,
    # we keep that length.
    if best_h is not None:
        n = best_h.shape[0]
    else:
        n = 200
    m = n // 2

    # If we did not obtain a usable start, build a random binary vector.
    if best_h is None:
        h = np.zeros(n, dtype=np.int8)
        h[:m] = 1
        rng.shuffle(h)
        cur_c5 = _c5_of(h)
        best_h, best_c5 = h.copy(), cur_c5
    else:
        h = best_h.copy()
        cur_c5 = best_c5

    # ------------------------------------------------------------------
    # 2.  Simulated‑annealing / hill‑climbing loop.
    # ------------------------------------------------------------------
    # Temperature schedule (starts relatively high, cools exponentially).
    T = 0.12
    cooling = 0.9997

    # Main loop runs until the time budget is almost exhausted.
    while time.time() - start_time < budget_s * 0.96:
        # pick a random 0 and a random 1 and swap them.
        zeros = np.flatnonzero(h == 0)
        ones = np.flatnonzero(h == 1)

        i = rng.choice(zeros)
        j = rng.choice(ones)

        # perform the swap
        h[i] = 1
        h[j] = 0

        new_c5 = _c5_of(h)
        delta = new_c5 - cur_c5

        accept = False
        if delta <= 0:
            accept = True
        else:
            if rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # revert the swap
            h[i] = 0
            h[j] = 1

        # geometric cooling
        T = max(T * cooling, 5e-7)

        # occasional random restart if we have been stuck for a while
        if (time.time() - start_time) % 30 < 0.01:   # approx. every 30 s
            if rng.random() < 0.15:  # 15 % chance to restart
                h = np.zeros(n, dtype=np.int8)
                h[:m] = 1
                rng.shuffle(h)
                cur_c5 = _c5_of(h)

    # ------------------------------------------------------------------
    # 3.  Return best found solution.

    best_h = best_h.astype(float)            # required output type
    final_c5 = _c5_of(best_h.astype(np.int8))
