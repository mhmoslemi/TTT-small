    """ Perform Bayesian optimization using Gaussian Process """
    initial_points = [space.sample() for _ in range(n_initial)]
    initial_vals = [objective_func(x) for x in initial_points]

    optimizer = Optimizer(
        dimensions=space.dimensions,
        base_estimator=GaussianProcessRegressor,
        n_initial_points=n_initial,
        acq_func='expected_improvement',
        random_state=seed
    )
    optimizer.tell(initial_points, initial_vals)

    results = []
    for _ in range(n_calls - n_initial):
        x = optimizer.ask()
        val = objective_func(x)
        results.append((val, x))
        optimizer.tell(x, val)

    # Extract best point
    if results:
        best = min(results, key=lambda x: x[0])
        return best[1]
    return None
