import time, math
import numpy as np

def _project_to_sum(v, target):
    """Project v onto the hyperplane sum(v)=target with box [0,1] bounds."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + lo, 0.0, 1.0)


def _eval_corr(h, dx):
    """Return (max value, full correlation array, index of max)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx] * dx, corr, idx


def _propose_move(cur, i, j, step, rng):
    """Return a new vector where an amount delta is moved from i to j."""
    # delta will be subtracted from i and added to j (i -> j)
    lower = max(-step, cur[i] - 1.0, -cur[j])
    upper = min(step, cur[i], 1.0 - cur[j])
    if lower >= upper:
        return None
    delta = rng.uniform(lower, upper)
    new = cur.copy()
    new[i] -= delta
    new[j] += delta
    return new


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ----- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
        target_sum = n / 2.0
        h0 = np.clip(h0, 0.0, 1.0)
        if not np.isclose(h0.sum(), target_sum):
            h0 = _project_to_sum(h0, target_sum)
    else:
        n = 70
        target_sum = n / 2.0
        h0 = rng.random(n)
        h0 = _project_to_sum(h0, target_sum)

    dx = 2.0 / n

    # current state and best ever
    cur_h = h0.copy()
    cur_score, cur_corr, cur_max_idx = _eval_corr(cur_h, dx)
    best_h = cur_h.copy()
    best_score = cur_score

    # simulated‑annealing schedule
    init_temp = 0.02
    final_temp = 1e-6
    max_step = 0.12
    min_step = 0.001

    # bookkeeping
    it = 0
    last_best_time = start
    no_improve_iters = 0
    TARGET = 0.3808  # stop early if we beat this

    while True:
        now = time.time()
        if now - start >= budget_s:
            break

        # linear schedules
        frac = (now - start) / budget_s
        temp = init_temp * (1.0 - frac) + final_temp * frac
        step = max_step * (1.0 - frac) + min_step * frac

        # ------- propose a move --------------------------------------
        if rng.random() < 0.7:                     # targeted move
            lag = cur_max_idx - (n - 1)            # lag with current worst overlap
            if lag >= 0:
                i = rng.integers(0, n - lag)
                j = i + lag
            else:
                i = rng.integers(-lag, n)
                j = i + lag
        else:                                      # random pair
            i, j = rng.choice(n, size=2, replace=False)

        cand = _propose_move(cur_h, i, j, step, rng)
        if cand is None:

        cand_score, cand_corr, cand_max_idx = _eval_corr(cand, dx)

        # ------- Metropolis acceptance --------------------------------
        accept = False
        if cand_score <= cur_score:
            accept = True
        else:
            if temp > 0.0:
                prob = math.exp(-(cand_score - cur_score) / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_h = cand
            cur_score = cand_score
            cur_corr = cand_corr
            cur_max_idx = cand_max_idx

            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
                last_best_time = now
                no_improve_iters = 0
                if best_score < TARGET:

            else:

        else:

            cur_score, cur_corr, cur_max_idx = _eval_corr(cur_h, dx)
            no_improve_iters = 0

        it += 1

    # final projection to guarantee constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_sum(best_h, target_sum)

    # final accurate score using the provided helper if present
    if "evaluate_erdos_solution" in globals():
        final_score = float(evaluate_erdos_solution(best_h))
    else:

    return best_h.tolist(), final_score, n
