import math
import time
import numpy as np


def project_to_constraints(arr: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy the sum constraint exactly."""
    arr = np.clip(arr, 0.0, 1.0)
    deficit = target_sum - arr.sum()
    # redistribute the deficit over free (non‑boundary) entries
    for _ in range(10):
        if abs(deficit) < 1e-12:
            break
        mask = (arr > 0.0) & (arr < 1.0)
        free = mask.sum()
        if free == 0:
            mask = np.ones_like(arr, dtype=bool)
            free = arr.size
        delta = deficit / free
        arr[mask] += delta
        arr = np.clip(arr, 0.0, 1.0)
        deficit = target_sum - arr.sum()
    return arr


def c5_value(h: np.ndarray) -> float:
    """Compute the C₅ value of a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
