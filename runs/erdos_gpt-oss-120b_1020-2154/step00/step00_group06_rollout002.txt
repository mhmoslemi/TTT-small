        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)

        new_h[i1] = 0.0

import time, math
import numpy as np

def _evaluate_h(h: np.ndarray) -> float:
    """Return C₅ for a step function given by the sample vector h."""
    # full linear correlation of h with (1‑h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / h.shape[0]))


def _sturmian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Initialise a binary vector of length n with exactly n/2 ones
    using a Sturmian (beat‑frequency) construction.
    """
    alpha = (math.sqrt(5.0) - 1.0) / 2.0          # ~0.618, irrational
    frac = np.mod(np.arange(n, dtype=np.float64) * alpha, 1.0)
    h = (frac < 0.5).astype(np.float64)

    target = n // 2
    cur = int(h.sum())
    if cur > target:
        idx = np.where(h == 1)[0]
        flip = rng.choice(idx, size=cur - target, replace=False)
        h[flip] = 0.0
    elif cur < target:
        idx = np.where(h == 0)[0]
        flip = rng.choice(idx, size=target - cur, replace=False)
        h[flip] = 1.0
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] (sampled uniformly) to minimise
    C₅ = max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------
    # Choose problem size and initialise h
    # ------------------------------------------------------------
    if "initial_h_values" in globals():
        # use the provided construction if it exists
        h = np.array(initial_h_values, dtype=np.float64)
        n = h.shape[0]
        target_sum = n / 2.0
        # tiny projection onto the sum constraint (normally already satisfied)
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            # distribute the tiny residual to the first feasible coordinate
            for i in range(n):
                lo = max(-diff, -h[i])
                hi = min(diff, 1.0 - h[i])
                if lo <= diff <= hi:
                    h[i] += diff
                    break
    else:
        n = 200                     # even size gives exact half‑mass
        h = _sturmian_init(n, rng)

    # ensure the exact integral constraint (sum = n/2)
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        diff = target_sum - cur_sum
        for i in range(n):
            if 0.0 <= h[i] + diff <= 1.0:
                h[i] += diff
                break

    cur_h = h.copy()
    cur_c5 = _evaluate_h(cur_h)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------
    # Simulated‑annealing like search
    # ------------------------------------------------------------
    max_step = 1.0          # maximal amount transferred in one move
    min_step = 0.005        # minimal step near the end of the run
    T0 = 0.02               # start temperature for Metropolis acceptance

    it = 0
    last_improve = 0

    while True:
        if time.time() - start > budget_s - 0.05:
            break

        elapsed = time.time() - start
        prog = elapsed / budget_s
        step = max_step * (1.0 - prog) + min_step * prog
        T = T0 * (1.0 - prog) + 1e-6

        # pick a donor (h[i] > 0) and a receiver (h[j] < 1)
        i = rng.integers(n)
        for _ in range(5):
            if cur_h[i] > 1e-12:
                break
            i = rng.integers(n)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        for _ in range(5):
            if cur_h[j] < 1.0 - 1e-12:
                break
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

        max_up = 1.0 - cur_h[j]   # how much we can add to j
        max_down = cur_h[i]       # how much we can subtract from i
        delta_max = min(step, max_up, max_down)
        if delta_max <= 0:
            continue

        d = rng.random() * delta_max
        new_h = cur_h.copy()
        new_h[i] -= d
        new_h[j] += d

        new_c5 = _evaluate_h(new_h)
        delta = new_c5 - cur_c5

        if delta <= 0.0 or rng.random() < math.exp(-delta / T):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                last_improve = it

        it += 1

        if it - last_improve > 200_000:
            cur_h = _sturmian_init(n, rng)
            cur_c5 = _evaluate_h(cur_h)
            last_improve = it

    return (best_h.tolist(), float(best_c5), n)
