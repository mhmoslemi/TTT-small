def project_onto_box_hyperplane(v, target):
    # Find λ such that sum(clip(v - λ,0,1)) == target.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    return w

   # choose delta within allowed range
   # max delta amount we can move from i to j while staying [0,1]:
   max_delta = min(h[i], 1 - h[j])
   min_delta = -min(h[j], 1 - h[i]) # negative move
   if max_delta == 0 and min_delta == 0:
       return h
   delta = np.random.uniform(min_delta, max_delta)
   new = h.copy()
   new[i] -= delta
   new[j] += delta
   return new

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _project_box_hyperplane(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the intersection of the box [0,1]^n and the hyperplane sum(v)=target."""
    # binary search for lambda such that sum(clip(v - lambda,0,1)) == target
    lo, hi = -2.0, 2.0  # wide enough for any reasonable v
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the active correlation term with respect to h."""
    n = h.shape[0]
    g = np.zeros_like(h)

    if k >= 0:
        r = n - 1 - k
        if r >= 0:
            idx = slice(0, r + 1)            # i where i and i+k are valid
            g[idx] += 1.0
            g[idx] -= h[idx + k]
        if k < n:
            idx2 = slice(k, n)                # i where i-k is valid
            g[idx2] -= h[idx2 - k]
    else:
        ka = -k
        # i where i and i+k are valid  (i >= ka)
        idx = slice(ka, n)
        g[idx] += 1.0
        g[idx] -= h[idx - ka]
        # i where i-k is valid  (i <= n-ka-1)
        idx2 = slice(0, n - ka)
        g[idx2] -= h[idx2 + ka]
    return g


def _max_corr_shift(h: np.ndarray):
    """Return the shift (integer) attaining the maximal correlation."""
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (h.shape[0] - 1)
    return shift, corr[idx] * (2.0 / h.shape[0])


def _c5_value(h: np.ndarray) -> float:
    """Exact C5 value for a discretised step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * (2.0 / h.shape[0]))


def _random_feasible(n: int, target: float) -> np.ndarray:
    """Generate a random feasible vector (box + sum constraint)."""
    v = np.random.rand(n)                  # uniform in [0,1]
    return _project_box_hyperplane(v, target)


def _random_perturb(h: np.ndarray) -> np.ndarray:
    """Make a small feasible move by transferring mass between two coordinates."""
    n = h.shape[0]
    # choose source with positive mass and destination with room
    src = np.where(h > 1e-9)[0]
    dst = np.where(h < 1 - 1e-9)[0]
    if src.size == 0 or dst.size == 0:
        return h
    i = int(np.random.choice(src))
    j = int(np.random.choice(dst))
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return h
    delta = np.random.uniform(0.0, max_delta)
    new = h.copy()
    new[i] -= delta
    new[j] += delta
    return new

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # Determine dimensionality
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = int(h0.shape[0])
    else:
        n = 45
        h0 = _random_feasible(n, n / 2.0)

    target_sum = n / 2.0               # ∫ h = 1  ⇒ Σ h * (2/n) = 1  ⇒ Σ h = n/2
    h0 = _project_box_hyperplane(h0, target_sum)

    best_h = h0.copy()
    best_score = _c5_value(best_h)

    start_time = time.time()
    max_iters = 3000000                # generous upper bound; actual stop by time
    it = 0
    # schedule parameters
    step_initial = 0.6
    prob_random = 0.15                 # chance of performing a random perturbation

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.96:
            break
        if it >= max_iters:
            break

        # Choose move type
        if random.random() < prob_random:
            cand = _random_perturb(best_h)
        else:
            # gradient step from current best
            shift, _ = _max_corr_shift(best_h)
            grad = _gradient_for_shift(best_h, shift)
            # decreasing step size
            step = step_initial * (1.0 - it / max_iters)
            if step < 5e-4:
                step = 5e-4
            cand = best_h - step * grad
            cand = _project_box_hyperplane(cand, target_sum)

        cand_score = _c5_value(cand)

        # Accept if improvement
        if cand_score < best_score:
            best_h = cand
            best_score = cand_score

        # Occasionally replace current base with a fresh random point to explore
        if it % 5000 == 0 and it > 0:

                best_h = _random_feasible(n, target_sum)
                best_score = _c5_value(best_h)
