def project_to_constraints(v, target_sum, max_iter=100, tol=1e-9):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = v.sum()
        diff = target_sum - cur_sum
        if abs(diff) <= tol:
            break
        if diff > 0:
            # need to increase sum
            free = v < 1.0
        else:
            free = v > 0.0
        if not np.any(free):
            break
        add = diff / free.sum()
        v[free] += add
        v = np.clip(v, 0.0, 1.0)
    # After loop, may still have diff due to rounding; optionally adjust smallest values.
    v = np.clip(v, 0.0, 1.0)
    # final scaling (if still off) we can slight adjust might break bounds but diff small.
    cur_sum = v.sum()
    if abs(cur_sum - target_sum) > 1e-6:
        # Force scaling of interior entries proportionally (avoid boundaries)
        # find indices not at 0/1 for adjustments
        interior = (v > 1e-12) & (v < 1 - 1e-12)
        if interior.any():
            v[interior] *= target_sum / cur_sum
        else:
            # else spread diff uniformly and clip again
            v += (target_sum - cur_sum) / len(v)
            v = np.clip(v, 0.0, 1.0)
    return v

import time
import numpy as np

def project_to_constraints(v: np.ndarray, target_sum: float, tol: float = 1e-12) -> np.ndarray:
    """Euclidean projection of v onto the box [0,1]^n intersected with sum(v)=target_sum."""
    lo = -np.max(v) - 1.0          # low enough => all zeros
    hi = 1.0 - np.min(v) + 1.0     # high enough => all ones
    for _ in range(100):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    # final clip (should already satisfy)
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def evaluate_c5(h: np.ndarray) -> float:
    """Discrete C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def upsample_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function to a finer grid."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    new_h = project_to_constraints(new_h, new_n / 2.0)
    return new_h

def propose_candidate(base: np.ndarray, rng: np.random.Generator,
                     max_delta: float = 0.2, change_frac: float = 0.03) -> np.ndarray:
    """Create a new candidate by random noise + a small mass transfer."""
    n = base.shape[0]
    cand = base.copy()
    # random additive noise on a subset
    k = max(1, int(change_frac * n))
    idx = rng.choice(n, size=k, replace=False)
    cand[idx] += rng.normal(scale=max_delta, size=k)
    # random transfer between a pair
    i, j = rng.integers(0, n, size=2)
    if i != j:
        delta_max = min(cand[i], 1.0 - cand[j])
        if delta_max > 0:
            delta = rng.uniform(0.0, max_delta * delta_max)
            cand[i] -= delta
            cand[j] += delta
        else:
            # try opposite direction
            delta_max = min(cand[j], 1.0 - cand[i])
            if delta_max > 0:
                delta = rng.uniform(0.0, max_delta * delta_max)
                cand[j] -= delta
                cand[i] += delta
    # enforce constraints
    cand = project_to_constraints(cand, n / 2.0)
    return cand

def hill_climb(start_h: np.ndarray, budget: float, rng: np.random.Generator) -> (np.ndarray, float):
    """Simple stochastic hill‑climbing within a time budget."""
    best_h = start_h.copy()
    best_val = evaluate_c5(best_h)
    n = best_h.shape[0]
    start = time.time()
    # schedule for noise magnitude (decays linearly)
    while time.time() - start < budget:
        elapsed = time.time() - start
        frac = 1.0 - elapsed / budget
        delta_scale = 0.2 * frac + 0.01   # keep a tiny floor
        cand = propose_candidate(best_h, rng, max_delta=delta_scale)
        val = evaluate_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand
        # occasional random restart to escape plateaus
        if rng.random() < 0.003:
            rnd = rng.random(n)
            rnd = project_to_constraints(rnd, n / 2.0)
            val_r = evaluate_c5(rnd)
            if val_r < best_val:
                best_val, best_h = val_r, rnd
    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_total = time.time()

    # --- choose grid size -------------------------------------------------
    n_points = 336                     # good compromise between flexibility and speed
    target_sum = n_points / 2.0

    # --- initial construction ---------------------------------------------
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        h0 = upsample_h(init_arr, n_points)
    else:
        # sin^2 shape as a neutral start
        x = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.sin(np.pi * x / 2.0) ** 2
        h0 = project_to_constraints(h0, target_sum)

    # --- stochastic optimization -------------------------------------------
    # reserve 90 % of the budget for the local search
    search_budget = budget_s * 0.9
    best_h, best_val = hill_climb(h0, search_budget, rng)

    # --- optional gradient refinement --------------------------------------
    try:
        import scipy.optimize as opt
        # smooth penalty formulation
        def obj_raw(x):
            y = np.clip(x, 0.0, 1.0)
            y = project_to_constraints(y, target_sum)
            return evaluate_c5(y) + 1e3 * abs(y.sum() - target_sum)

        res = opt.minimize(
            obj_raw,
            best_h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n_points,
            options={'maxfun': 2500, 'ftol': 1e-12}
        )
        refined = project_to_constraints(res.x, target_sum)
        refined_val = evaluate_c5(refined)
        if refined_val < best_val:
            best_h, best_val = refined, refined_val
    except Exception:
        # optimizer not available or failed – ignore
        pass

    # --- final evaluation --------------------------------------------------
    c5_bound = float(evaluate_erdos_solution(best_h))

    # make sure we exit before the allotted budget
    elapsed = time.time() - start_total
    if elapsed > budget_s:
        # fallback: return the best found so far (already stored)
        pass

    return (best_h.tolist(), c5_bound, n_points)
