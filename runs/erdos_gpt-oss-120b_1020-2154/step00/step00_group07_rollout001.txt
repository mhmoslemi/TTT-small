def optimize_for_n(n, time_budget, rng, start_h=None):

    # initialise

        cur_h = start_h.copy()
    else:

    cur_c5 = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    # maintain lists of indices
    ones = list(np.flatnonzero(cur_h))
    zeros = list(np.flatnonzero(1 - cur_h))
    # parameters
    max_no_improve = max(2000, n)  # stoppage threshold
    no_improve = 0
    # time management
    start_time = time.time()
    # main loop
    while time.time() - start_time < time_budget:
        # random swap between a 1 and a 0
        if not ones or not zeros:
            break
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]
        # swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        # update lists locally
        ones[i_pos] = j
        zeros[j_pos] = i
        new_c5 = compute_c5(cur_h, dx)
        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            # revert lists
            ones[i_pos] = i
            zeros[j_pos] = j
            no_improve += 1
        if no_improve >= max_no_improve:
            # random restart

            cur_c5 = compute_c5(cur_h, dx)
            ones = list(np.flatnonzero(cur_h))
            zeros = list(np.flatnonzero(1 - cur_h))
            no_improve = 0

def _c5_of_h(h, dx):
    """Return max overlap integral for binary step h."""
    n = h.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    o = 1.0 - h                     # complement
    ha = np.fft.rfft(h, n=size)
    ho = np.fft.rfft(o[::-1], n=size)   # reversed complement
    conv = np.fft.irfft(ha * ho, n=size)[: 2 * n - 1]
    return float(conv.max() * dx)


def _init_h(n, rng):
    """Create a balanced binary vector of length n."""
    # try to reuse the provided construction
    if 'initial_h_values' in globals() and len(initial_h_values) == n:
        h = np.array(initial_h_values, dtype=np.float64)
        target = n // 2
        cur = int(round(h.sum()))
        if cur != target:
            diff = cur - target
            if diff > 0:
                ones = np.flatnonzero(h == 1)
                flip = rng.choice(ones, size=diff, replace=False)
                h[flip] = 0.0

                zeros = np.flatnonzero(h == 0)
                flip = rng.choice(zeros, size=-diff, replace=False)
                h[flip] = 1.0

    # otherwise use a Sturmian (Beatty) construction, which is already well balanced

            d = np.abs(frac[zeros] - 0.5)
            order = np.argsort(d)[::-1]
            h[zeros[order[: -diff]]] = 1.0
    return h


def _optimize_n(n, budget, rng, start_h=None):
    """Hill‑climbing with random restarts for a fixed resolution."""
    dx = 2.0 / n
    if start_h is not None:
        cur = start_h.copy()
    else:
        cur = _init_h(n, rng)

    cur_c5 = _c5_of_h(cur, dx)
    best_h = cur.copy()
    best_c5 = cur_c5

    # maintain mutable index lists for O(1) swaps
    ones = list(np.flatnonzero(cur))
    zeros = list(np.flatnonzero(1 - cur))

    max_stall = max(2000, n)
    stall = 0
    start = time.time()
    while time.time() - start < budget:
        if not ones or not zeros:
            break
        i_pos = rng.integers(len(ones))
        j_pos = rng.integers(len(zeros))
        i = ones[i_pos]
        j = zeros[j_pos]

        # perform swap
        cur[i] = 0.0
        cur[j] = 1.0
        ones[i_pos] = j
        zeros[j_pos] = i

        new_c5 = _c5_of_h(cur, dx)
        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            stall = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur.copy()
                if best_c5 <= 0.3808:               # early exit if target reached
                    break
        else:
            # revert
            cur[i] = 1.0
            cur[j] = 0.0
            ones[i_pos] = i
            zeros[j_pos] = j
            stall += 1

        if stall >= max_stall:
            cur = _init_h(n, rng)
            cur_c5 = _c5_of_h(cur, dx)
            ones = list(np.flatnonzero(cur))
            zeros = list(np.flatnonzero(1 - cur))
            stall = 0
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a small C5 value."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # Choose a resolution that balances quality and speed.
    # 200 points give a fine approximation while still cheap.
    n = 200
    remaining = budget_s - (time.time() - start)
    # leave a tiny margin for final evaluation
    best_h, best_c5 = _optimize_n(n, remaining * 0.97, rng)

    # final recomputation for safety
    best_c5 = _c5_of_h(best_h, 2.0 / n)
    return best_h, float(best_c5), int(n)
