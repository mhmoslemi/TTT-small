def compute_c5(h):
    n = len(h)
    dx = 2.0/n
    # Use FFT zero padded length L = 1 << (2*n). Actually to compute linear conv length 2*n-1
    L = 1
    while L < 2*n:
        L <<= 1
    F1 = np.fft.rfft(h, L)
    F2 = np.fft.rfft(1 - h, L)
    conv = np.fft.irfft(F1 * F2, L)[:2*n-1]  # linear conv.
    maxc = conv.max()
    return maxc * dx
```

def project_to_sum_box(v, target):
    # water-filling
    lo = -2.0
    hi = 2.0
    for _ in range(40):
        mid = (lo+hi)/2
        x = np.clip(v + mid, 0, 1)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo+hi)/2, 0, 1)
    # adjust any rounding error
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff equally among non-boundary entries
        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0, 1)
        else:
            # all at bounds, just adjust first element
            x[0] = np.clip(x[0] + diff, 0, 1)
    return x
```

import time
import numpy as np

def _project_to_sum_box(v, target):
    """Project vector v onto the box [0,1]^n with sum(v)=target."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # very small residual correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # all at bounds – adjust a single entry safely
            idx = np.argmin(np.abs(v - 0.5))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def _c5_of(h):
    """C5 value for a given step vector (using the same metric as the judge)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _worst_shift_and_val(h):
    """Return (shift, raw_correlation_max) for the current h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)                # integer shift (negative allowed)
    return shift, corr[idx]

def _grad_for_shift(h, shift):
    """Gradient of the worst‑shift correlation w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        i = slice(0, n - shift)
        j = slice(shift, n)
        grad[i] += 1.0 - h[j]
        grad[j] -= h[i]
    else:
        s = -shift
        i = slice(s, n)
        j = slice(0, n - s)
        grad[i] += 1.0 - h[j]
        grad[j] -= h[i]
    return grad

def _random_feasible(n):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    v = np.random.rand(n)
    return _project_to_sum_box(v, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points) where
    h_values is a feasible step function on [0,2],
    c5_bound = evaluate_erdos_solution(h_values) and
    n_points = len(h_values).
    """
    np.random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------------ #
    # Determine problem size and initialise the vector.
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        h = init.copy()
        # ensure feasibility (tiny adjustments may be needed)
        h = _project_to_sum_box(h, n / 2.0)
    else:
        n = 200                     # a comfortable even size when no starter is present
        h = _random_feasible(n)

    best_h = h.copy()
    best_score = _c5_of(best_h)

    # parameters for the local search
    init_step = 0.12               # initial step length in gradient direction
    step = init_step
    max_iters_per_restart = 4000
    no_improve_limit = 300
    restarts = 0

    while time.time() - start < budget_s * 0.94:
        # ------------------------------------------------------------------
        # Gradient‑descent on the active (worst) shift.
        cur_h = h.copy()
        cur_score = _c5_of(cur_h)
        target_sum = n / 2.0
        no_improve = 0

        for _ in range(max_iters_per_restart):
            if time.time() - start >= budget_s * 0.94:
                break

            shift, _ = _worst_shift_and_val(cur_h)
            grad = _grad_for_shift(cur_h, shift)

            improved = False
            # try a few step‑size candidates
            for factor in (1.0, 0.5, 0.25):
                cand = _project_to_sum_box(cur_h - step * factor * grad, target_sum)
                cand_score = _c5_of(cand)
                if cand_score < cur_score - 1e-12:
                    cur_h, cur_score = cand, cand_score
                    improved = True
                    break

            if improved:
                no_improve = 0
                # slowly enlarge step if we are consistently improving
                step = min(step * 1.07, init_step)
                if cur_score < best_score:
                    best_h, best_score = cur_h.copy(), cur_score
            else:
                no_improve += 1
                step *= 0.55
                if step < 1e-6 or no_improve >= no_improve_limit:
                    # give up this trajectory, restart later
                    break

        # ------------------------------------------------------------------
        # Occasionally inject a random perturbation to escape flat regions.
        if np.random.rand() < 0.15:
            h = _random_feasible(n)
        else:
            h = best_h.copy()  # continue from the best found so far

        restarts += 1
        # modest step size reset for the next run
        step = init_step

    # final evaluation using the supplied helper (ensures identical metric)
    c5_bound = evaluate_erdos_solution(best_h)

    return best_h, float(c5_bound), n
