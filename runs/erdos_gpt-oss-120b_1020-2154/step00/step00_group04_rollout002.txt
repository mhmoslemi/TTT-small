import time, math
import numpy as np

def _project_to_feasible(v, target, tol=1e-10, max_iter=20):
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum(x) = target}.
    Simple water-filling iterative scheme.
    """
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        excess = np.sum(x) - target
        if abs(excess) <= tol:
            break
        free = (x > 0.0) & (x < 1.0)
        nfree = np.count_nonzero(free)
        if nfree == 0:          # all at bounds – cannot adjust further
            break
        delta = excess / nfree
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
    return x

def _shift_zero_pad(arr, shift):
    """Return arr shifted by `shift` positions with zero padding outside."""
    n = arr.size
    out = np.zeros_like(arr)
    if shift >= 0:
        out[:n - shift] = arr[shift:]
    else:
        s = -shift
        out[s:] = arr[:n - s]
    return out

def _gradient_for_shift(h, shift):
    """Gradient of the worst‑case overlap term with respect to h."""
    forward = _shift_zero_pad(h, shift)     # h_{i+shift}
    backward = _shift_zero_pad(h, -shift)   # h_{i-shift}
    return 1.0 - forward - backward

def _max_corr_raw(h):
    """Raw maximum of the overlap sum (no dx scaling)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr), corr

def _triangle_h(n):
    """Triangular step function on [0,2] with height 1."""
    dx = 2.0 / n
    x = np.arange(n) * dx
    h = np.minimum(x, 2.0 - x)
    # the triangle already has integral ≈1, but enforce exactness
    target = n / 2.0
    h = _project_to_feasible(h, target)
    return h

    start = time.time()

    # ----- problem size -------------------------------------------------
    n = 200                      # number of sample points (must be even)
    target = n / 2.0             # required sum of h
    dx = 2.0 / n

    # ----- initial candidate --------------------------------------------
    # start from a deterministic triangle; keep the best discovered
    best_h = _triangle_h(n)
    best_score = evaluate_erdos_solution(best_h)

    # also try a random feasible start
    h = rng.random(n)
    h = _project_to_feasible(h, target)
    cur_score_raw, cur_corr = _max_corr_raw(h)
    cur_score = cur_score_raw * dx
    if cur_score < best_score:
        best_score = cur_score
        best_h = h.copy()

    # ----- parameters for the stochastic descent ------------------------
    temperature = 0.08          # initial step magnitude (also acts as SA T)
    decay = 0.9999
    it = 0

    # main optimisation loop
    while time.time() - start < budget_s * 0.95:
        # locate worst shift for the current iterate
        idx = np.argmax(cur_corr)
        shift = idx - (n - 1)

        # gradient step towards reducing the worst term
        grad = _gradient_for_shift(h, shift)
        alpha = temperature
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_to_feasible(h_new, target)

        # evaluate
        new_score_raw, new_corr = _max_corr_raw(h_new)
        new_score = new_score_raw * dx

        # Metropolis acceptance
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / (temperature + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            h = h_new
            cur_corr = new_corr
            cur_score = new_score
            if new_score < best_score:

                best_h = h.copy()

    final_score = evaluate_erdos_solution(best_h)
