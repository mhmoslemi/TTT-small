def _evaluate_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _shift_array(arr: np.ndarray, k: int) -> np.ndarray:
    """Return arr shifted by k with zero padding (h_{i‑k})."""
    n = arr.size
    if k >= 0:
        out = np.empty_like(arr)
        out[:k] = 0.0
        out[k:] = arr[: n - k]
    else:
        k = -k
        out = np.empty_like(arr)
        out[: n - k] = arr[k:]
        out[n - k :] = 0.0
    return out
