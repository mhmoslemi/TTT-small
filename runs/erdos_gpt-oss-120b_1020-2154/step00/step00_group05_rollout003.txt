import numpy as np, time, math

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project(h, total):
    h = np.clip(h, 0.0, 1.0)
    diff = total - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Determine capacity
    if diff > 0:
        cap = 1.0 - h
    else:
        cap = h
    cap_sum = cap.sum()
    if cap_sum == 0:
        return h
    # distribute proportionally
    h = h + diff * cap / cap_sum
    # Clip again
    h = np.clip(h, 0.0, 1.0)
    return h
