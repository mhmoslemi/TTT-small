def compute_c5(h):
    N = len(h)
    dx = 2.0/N
    # correlation (full)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def random_binary_h(N):
    # ensure N even else include fractional entry
    half = N//2
    h = np.zeros(N)
    idx = np.random.choice(N, half, replace=False)
    h[idx] = 1.0
    if N % 2 == 1:
        # need a fractional entry to sum to N/2 half-integer
        # sum currently = half, target = N/2 => half+0.5
        # add 0.5 to a random zero entry
        zero_idx = np.where(h == 0)[0]
        choice = np.random.choice(zero_idx)
        h[choice] = 0.5
    return h

def project_onto_constraints(h, target_sum):

    # adjust sum to target by adding/subtracting uniformly on interior?

    # We'll use water-filling algorithm.

    # Since diff magnitude is small (due to previous operations), we can just distribute diff equally across all entries not at bound, then clip again iteratively.
    diff = target_sum - h.sum()
    # iterative distribution
    eps = 1e-12
    max_iter=1000
    for _ in range(max_iter):
        if abs(diff) < eps:

        # find indices with room to adjust in needed direction

        # distribute equally

        # limit to not exceed bounds

        # clip

        h = h_new

    target=N/2.0
    best_h = h.copy()
    best_c5=compute_c5(best_h)

        return compute_c5(x)

import numpy as np, time, random
from scipy.optimize import minimize

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    # compute full correlation
    # using np.correlate (fast enough)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_constraints(v, target):
    # simple projection: clip then adjust by adding diff to random non-bound entries
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    # If diff ~0, done
    if abs(diff) < 1e-12:
        return v
    # Try to fix using single entry
    if diff > 0:
        idx = np.where(v < 1 - 1e-12)[0]
    else:
        idx = np.where(v > 1e-12)[0]
    if idx.size > 0:
        i = np.random.choice(idx)
        delta = np.clip(diff, -v[i], 1.0 - v[i])
        v[i] += delta
        # Recursively adjust remainder if still diff remains
        return project_onto_constraints(v, target)
    else:
        # fallback scaling
        if v.sum() == 0:
            v += target / len(v)
        else:
            v = v * (target / v.sum())
        v = np.clip(v, 0.0, 1.0)
        return v

def construct_three_piece(N, a, f):
    # a is length of full block where h=1
    # f is intermediate value
    dx = 2.0 / N
    xs = (np.arange(N) + 0.5) * dx
    L1 = a
    # Avoid division by zero if f is zero
    if f <= 0:
        Lf = 0.0
    else:
        Lf = (1.0 - a) / f
    # Ensure Lf non-negative
    if Lf < 0:
        Lf = 0.0
    h = np.zeros(N)
    h[xs < L1] = 1.0
    mask_mid = (xs >= L1) & (xs < L1 + Lf)
    h[mask_mid] = f
    # other remain zero
    # adjust sum constraint
    target = N / 2.0
    h = project_onto_constraints(h, target)
    return h

def random_three_piece_search(N, iterations, time_limit, start_time):
    best_h = None
    best_c5 = np.inf
    # compute target for sum
    for _ in range(iterations):
        # time check
        if time.time() - start_time > time_limit:
            break
        # sample Lf in [0,2]
        Lf = 2.0 * np.random.rand()
        f = np.random.rand()  # [0,1)
        # Avoid f=0 exactly -> leads to infinite Lf; ensure f > small
        if f < 1e-6:
            continue
        a = 1.0 - f * Lf
        # check feasibility
        if a < 0.0 or a > 1.0:
            continue
        if a + Lf > 2.0 + 1e-9:
            continue
        # construct
        h = construct_three_piece(N, a, f)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    return best_h, best_c5

def hill_climb_refine(h, time_limit, start_time):
    N = len(h)
    target = N / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    # schedule of delta magnitudes; we can adjust based on N
    deltas = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for delta_max in deltas:
        while True:
            now = time.time()
            if now - start_time > time_limit:
                return best_h, best_c5
            # pick two indices i, j
            i, j = np.random.randint(N, size=2)
            if i == j:
                continue
            delta = (np.random.rand()*2 - 1) * delta_max
            hi = best_h[i] + delta
            hj = best_h[j] - delta
            if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
                continue
            h_new = best_h.copy()
            h_new[i] = hi
            h_new[j] = hj
            # sum unchanged
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                # continue with same delta_max
    # Should break out when time over
    return best_h, best_c5

def slsqp_optimize(h0, time_limit, start_time):
    N = len(h0)
    target = N / 2.0
    # objective wrapper
    def obj(x):
        return compute_c5(x)
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target})
    bounds = [(0.0, 1.0)] * N
    # call minimize with limited maxiter
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
        if not res.success:
            # use original
            h_opt = h0
        else:
            h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = project_onto_constraints(h_opt, target)
        c5_opt = compute_c5(h_opt)
        return h_opt, c5_opt
    except Exception:
        return h0, compute_c5(h0)

def optimize_for_N(N, total_budget, start_time):
    # allocate half for random search, half for refinement
    elapsed = time.time() - start_time
    remaining = total_budget - elapsed
    if remaining <= 0:
        return None, np.inf
    # allocate 70% for random search, 30% for refinement
    random_budget = remaining * 0.65
    refine_budget = remaining - random_budget
    # random search iterations estimate: try as many as possible within random_budget
    # we can break when time exceeded
    # Let's compute a reasonable iteration count: 20000 maybe
    max_iter = 20000
    best_h, best_c5 = random_three_piece_search(N, max_iter, start_time + random_budget, start_time)
    if best_h is None:
        # fallback: binary random
        best_h = random_binary_h(N)
        best_c5 = compute_c5(best_h)
    # refine using hill climbing and maybe SLSQP within remaining time
    # We'll try SLSQP first if time permits
    now = time.time()
    time_left = total_budget - (now - start_time)
    if time_left > 0.5:
        # run slsqp from best_h
        sl_time = time_left * 0.5
        h_slsqp, c5_slsqp = slsqp_optimize(best_h, sl_time, start_time)
        if c5_slsqp < best_c5:
            best_h, best_c5 = h_slsqp, c5_slsqp
        # update time left
        now = time.time()
        time_left = total_budget - (now - start_time)
    # Hill climb
    if time_left > 0.1:
        hc_h, hc_c5 = hill_climb_refine(best_h, start_time + time_left, start_time)
        if hc_c5 < best_c5:
            best_h, best_c5 = hc_h, hc_c5
    return best_h, best_c5

def random_binary_h(N):
    # generate binary vector with sum = N/2
    half = N // 2
    h = np.zeros(N, dtype=float)
    idx = np.random.choice(N, half, replace=False)
    h[idx] = 1.0
    if N % 2 == 1:
        # need an extra 0.5 sum
        remaining = N / 2.0 - half
        # set one zero to 0.5
        zeros = np.where(h == 0)[0]
        if zeros.size > 0:
            i = np.random.choice(zeros)
            h[i] = remaining
    # ensure sum exactly target
    target = N / 2.0
    h = project_onto_constraints(h, target)
    return h

def improve_step(h):
    N = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # find max index
    idx_max = np.argmax(corr)
    # compute lag
    k = idx_max - (N - 1)  # shift positive if idx > N-1
    # if k>=0:
    if k >= 0:
        # overlapping i from 0..N-1-k
        # compute contributions: w_i = h[i] * (1 - h[i+k])
        # we want to reduce w_i, we can move mass from i to i+k.
        # compute potential delta_i = min(h[i], 1 - h[i+k])
        # choose i with largest w_i (or some random among high ones)
        # implement.
        i_candidates = np.arange(N - k)
        w = h[i_candidates] * (1 - h[i_candidates + k])
        # choose i with max w, also where h[i] > 0 and h[i+k] < 1
        mask = (h[i_candidates] > 1e-6) & (h[i_candidates + k] < 1 - 1e-6)
        if np.any(mask):
            idx = np.argmax(w * mask)  # mask will zero out others
            i = i_candidates[idx]
            # compute delta = min(h[i], 1 - h[i+k]) * alpha (e.g., 0.5)
            delta = min(h[i], 1 - h[i + k]) * 0.5
            if delta > 1e-12:
                h[i] -= delta
                h[i + k] += delta
    else:
        # negative lag: treat similarly swapping opposite direction
        k_abs = -k
        i_candidates = np.arange(N - k_abs)
        w = h[i_candidates + k_abs] * (1 - h[i_candidates])
        mask = (h[i_candidates + k_abs] > 1e-6) & (h[i_candidates] < 1 - 1e-6)

            i = i_candidates[idx]
            delta = min(h[i + k_abs], 1 - h[i]) * 0.5
            if delta > 1e-12:
                h[i + k_abs] -= delta
                h[i] += delta
    return h

import numpy as np
import time
import random
from scipy.optimize import minimize

# ----- basic utilities -------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """max_k ∫ h(x)(1‑h(x+k)) dx approximated on a uniform grid."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_constraints(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to the required target (simple fix)."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # try to fix the whole difference on a single free entry
    if diff > 0:
        free = np.where(v < 1.0 - 1e-12)[0]
    else:
        free = np.where(v > 1e-12)[0]
    if free.size:
        i = np.random.choice(free)
        delta = np.clip(diff, -v[i], 1.0 - v[i])
        v[i] += delta
        return project_onto_constraints(v, target)
    # fallback: scale uniformly (should be a tiny correction)
    scale = target / v.sum() if v.sum() > 0 else target / len(v)
    v = np.clip(v * scale, 0.0, 1.0)
    return v


def random_binary_h(N: int) -> np.ndarray:
    """Binary (or half‑binary) vector with exact sum N/2."""
    half = N // 2
    h = np.zeros(N, dtype=float)
    idx = np.random.choice(N, half, replace=False)
    h[idx] = 1.0
    if N % 2:
        # need an extra 0.5 (because target sum is N/2 = half+0.5)
        missing = N / 2.0 - half
        zeros = np.where(h == 0)[0]
        if zeros.size:
            h[np.random.choice(zeros)] = missing
    return project_onto_constraints(h, N / 2.0)


# ----- improvement routine ---------------------------------------------
def improve_h(h: np.ndarray, deadline: float, max_iter: int = 5000) -> (np.ndarray, float):
    """
    Greedy reduction of the maximal correlation.
    Moves preserve the total integral exactly.
    """
    N = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    for _ in range(max_iter):
        if time.time() >= deadline:
            break

        # correlation and worst lag
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        idx_max = int(np.argmax(corr))
        k = idx_max - (N - 1)          # lag (may be negative)

        if k >= 0:
            cand = np.arange(N - k)
            eligible = (best_h[cand] > 1e-8) & (best_h[cand + k] < 1 - 1e-8)
            if eligible.any():
                w = best_h[cand] * (1.0 - best_h[cand + k])
                w = w * eligible
                i = cand[int(np.argmax(w))]
                delta = min(best_h[i], 1.0 - best_h[i + k]) * 0.5
                if delta > 1e-12:
                    best_h[i] -= delta
                    best_h[i + k] += delta
            else:
                i, j = np.random.randint(N, size=2)
                if i != j:
                    delta = (np.random.rand() * 2 - 1) * 0.02
                    hi, hj = best_h[i] + delta, best_h[j] - delta
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        best_h[i], best_h[j] = hi, hj
        else:
            kp = -k
            cand = np.arange(N - kp)
            eligible = (best_h[cand + kp] > 1e-8) & (best_h[cand] < 1 - 1e-8)
            if eligible.any():
                w = best_h[cand + kp] * (1.0 - best_h[cand])
                w = w * eligible
                i = cand[int(np.argmax(w))]
                delta = min(best_h[i + kp], 1.0 - best_h[i]) * 0.5
                if delta > 1e-12:
                    best_h[i + kp] -= delta
                    best_h[i] += delta
            else:
                i, j = np.random.randint(N, size=2)
                if i != j:
                    delta = (np.random.rand() * 2 - 1) * 0.02
                    hi, hj = best_h[i] + delta, best_h[j] - delta
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        best_h[i], best_h[j] = hi, hj

        # evaluate the new max correlation
        c5_now = float(np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * (2.0 / N))
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now

    return best_h, best_c5


# ----- main driver -----------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) for a constructed step function.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    total_budget = float(budget_s)

    # use a reasonably fine grid; 200 points gives dx = 0.01
    N = 200
    target_sum = N / 2.0

    best_h = None
    best_c5 = np.inf

    # main stochastic search loop
    while True:
        now = time.time()
        remaining = total_budget - (now - start)
        if remaining < 0.8:          # leave a small margin for final tidy‑up
            break

        # allocate a chunk of time for one improvement run
        chunk = min(8.0, remaining * 0.7)          # at most 8 s per restart
        deadline = now + chunk

        # initialise – either a fresh random pattern or a jittered copy of the current best
        if best_h is None or random.random() < 0.3:
            h0 = random_binary_h(N)
        else:
            # small random perturbation of the current best
            h0 = best_h.copy()
            i, j = np.random.randint(N, size=2)
            delta = (np.random.rand() * 2 - 1) * 0.05
            hi, hj = h0[i] + delta, h0[j] - delta
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                h0[i], h0[j] = hi, hj

        # greedy improvement on this start
        h_opt, c5_opt = improve_h(h0, deadline, max_iter=5000)

        # keep the best found so far
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt.copy(), c5_opt

    # final clean‑up: enforce constraints exactly and recompute the bound
    if best_h is None:                     # very unlikely, fallback to uniform 0.5
        best_h = np.full(N, 0.5)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraints(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, N
