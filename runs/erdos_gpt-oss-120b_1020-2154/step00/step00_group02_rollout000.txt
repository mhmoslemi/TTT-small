def hill_climb(h, max_iter, start_step, time_limit):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    step = start_step
    n = len(h)
    start_time = time.time()
    i = 0
    while i < max_iter:
        if time.time() - start_time > time_limit:
            break
        # random source and target
        # efficient: precompute list of indices where h>1e-12 and h<1-1e-12
        src_candidates = np.where(best_h > 1e-12)[0]
        tgt_candidates = np.where(best_h < 1 - 1e-12)[0]
        if len(src_candidates) == 0 or len(tgt_candidates) == 0:
            break
        i_src = np.random.choice(src_candidates)
        i_tgt = np.random.choice(tgt_candidates)
        # ensure src != tgt
        if i_src == i_tgt:
            continue
        max_delta = min(step, best_h[i_src], 1 - best_h[i_tgt])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand()
        # new candidate
        h_new = best_h.copy()
        h_new[i_src] -= delta
        h_new[i_tgt] += delta
        # compute score
        new_score = compute_c5(h_new)
        if new_score < best_score - 1e-8:
            best_h = h_new
            best_score = new_score
            # maybe increase step slightly to accelerate? but we can keep.
        else:
            # optional: accept if new_score <= best_score + tolerance? Not necessary.
            pass
        # gradually reduce step
        step *= 0.9995  # small decay
        if step < 1e-4:
            step = 1e-4
        i += 1
    return best_h, best_score

def continuous_refine(h, time_limit):
    # uses random move transferring small delta between indices
    cur = h.copy()
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score
    t0 = time.time()
    # step initial is maybe 0.01 (i.e., 1% of 1)
    step = 0.05  # up to 5% per move
    while time.time() - t0 < time_limit:
        # Choose source with cur[i] > 1e-8, target with cur[j] < 1-1e-8
        src_candidates = np.where(cur > 1e-8)[0]
        tgt_candidates = np.where(cur < 1 - 1e-8)[0]
        if len(src_candidates) == 0 or len(tgt_candidates) == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.choice(tgt_candidates)
        max_delta = min(step, cur[i], 1 - cur[j])
        if max_delta <= 1e-12:
            continue
        delta = max_delta * np.random.rand()
        # apply
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_score = compute_c5(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
        # shrink step gradually
        step *= 0.9995
        if step < 1e-5:
            step = 1e-5
    # after loop ensure sum exactly.
    diff = (len(cur) / 2) - np.sum(best)
    if abs(diff) > 1e-8:
        # adjust first element
        best[0] += diff
        best = np.clip(best, 0, 1)
    return best, best_score

import time
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Discrete evaluation of the Erdős overlap constant for a sampled step function."""
    n = h.size
    dx = 2.0 / n
    # full correlation between h and (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def random_binary_h(n: int) -> np.ndarray:
    """Return a binary vector of length n with exactly n/2 ones (n must be even)."""
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def simulated_annealing_binary(n: int, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Simple annealing on binary vectors.
    Returns the best vector found and its C5 value.
    """
    start = time.time()
    cur = random_binary_h(n)
    cur_score = compute_c5(cur)
    best = cur.copy()
    best_score = cur_score

    # initial temperature (a fraction of the current score)
    T0 = max(0.05, cur_score * 0.3)
    while time.time() - start < time_limit:
        # pick a 1 and a 0 to swap
        ones_idx = np.where(cur > 0.5)[0]
        zeros_idx = np.where(cur < 0.5)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        new = cur.copy()
        new[i] = 0.0
        new[j] = 1.0
        new_score = compute_c5(new)
        delta = new_score - cur_score

        elapsed = time.time() - start
        T = T0 * (1.0 - elapsed / time_limit)  # linear cooling
        if delta < 0 or np.random.rand() < np.exp(-delta / (T + 1e-12)):
            cur = new
            cur_score = new_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # occasional random restart to diversify
        if np.random.rand() < 0.001:
            cur = random_binary_h(n)
            cur_score = compute_c5(cur)
    return best, best_score

    step = 0.05  # initial transfer magnitude
    n = cur.size
    while time.time() - start < time_limit:
        src = np.where(cur > 1e-8)[0]
        tgt = np.where(cur < 1 - 1e-8)[0]
        if src.size == 0 or tgt.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.choice(tgt)
        max_delta = min(step, cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand()
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_score = compute_c5(new)
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
        # slowly shrink the step size
        step *= 0.9995
        if step < 1e-5:
            step = 1e-5
    # final feasibility repair (tiny numerical drift)
    target_sum = n / 2.0
    diff = target_sum - np.sum(best)
    if abs(diff) > 1e-10:
        # distribute the diff to the first element that can absorb it
        for idx in range(n):
            cand = best[idx] + diff
            if 0.0 <= cand <= 1.0:
                best[idx] = cand
                break
        # fallback: clamp globally and renormalise
        best = np.clip(best, 0.0, 1.0)
        best *= target_sum / np.sum(best)
    return best, compute_c5(best)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function that yields a low Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = None

    # If a parent solution is supplied, evaluate it first.
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        parent_score = compute_c5(parent)
        if parent_score < best_c5:
            best_h, best_c5, best_n = parent.copy(), parent_score, parent.size
    except Exception:
        pass

    # Candidate sample sizes (must be even)
    candidate_ns = [44, 48, 52, 56, 60, 64, 68, 72, 80, 90, 100]

    # Main search loop – runs until the budget is nearly exhausted.
    while time.time() - start_time < budget_s * 0.94:
        n = random.choice(candidate_ns)

        # Split the remaining budget: most for the binary annealing,
        # the rest for fine continuous tweaks.
        remaining = budget_s - (time.time() - start_time)
        # guarantee at least a couple of seconds for each sub‑stage
        sub_budget = max(5.0, remaining * 0.15)
        half = sub_budget * 0.6
        third = sub_budget * 0.4

        # 1) binary annealing
        h_bin, _ = simulated_annealing_binary(n, half)

        # 2) continuous refinement
        h_ref, _ = continuous_refine(h_bin, third)

        # final exact score
        final_score = compute_c5(h_ref)

        # keep the best overall
        if final_score < best_c5:
            best_h, best_c5, best_n = h_ref.copy(), final_score, n

    # Safety fallback – if nothing was found (unlikely)
    if best_h is None:
        n = 45
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final tiny correction to guarantee the integral constraint exactly.
    target_sum = best_n / 2.0
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-12:
        # adjust the first coordinate that can absorb the difference
        for idx in range(best_n):
            cand = best_h[idx] + diff
            if 0.0 <= cand <= 1.0:
                best_h[idx] = cand
                break
        best_h = np.clip(best_h, 0.0, 1.0)
        # Renormalise just in case
        best_h *= target_sum / np.sum(best_h)

    best_c5 = compute_c5(best_h)

    # Return a plain Python list for compatibility with the harness.
    return best_h.tolist(), float(best_c5), int(best_n)
