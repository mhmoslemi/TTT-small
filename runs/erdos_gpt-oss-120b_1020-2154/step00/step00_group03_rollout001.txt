def project_to_sum_and_bounds(h, target):
    h = np.clip(h, 0, 1)
    tol = 1e-9
    while True:
        current_sum = h.sum()
        diff = target - current_sum
        if abs(diff) < tol:
            break
        free = (h > 0 + 1e-12) & (h < 1 - 1e-12)
        free_idx = np.where(free)[0]
        if len(free_idx) == 0:
            # cannot adjust further; distribute to nearest bound
            # find indices at bounds can be nudged tiny
            free_idx = np.arange(len(h))
        delta = diff / len(free_idx)
        h[free_idx] += delta
        # clip again
        h = np.clip(h, 0, 1)
    return h
```

def project_to_sum_and_bounds(x, target):
    # Clip to [0,1] first.
    x = np.clip(x, 0.0, 1.0)
    # Use lambda shift method.
    # Find lambda such that sum(clipped(x + lambda)) = target.
    # Since sum is monotonic increasing in lambda, we can binary search.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum_and_bounds(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the intersection of the box [0,1]^n and the hyperplane sum=x."""
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)


# ----------------------------------------------------------------------
# Smooth approximation of the max objective and its gradient
# ----------------------------------------------------------------------
class SmoothObj:
    __slots__ = ("beta",)

    def __init__(self, beta: float):
        self.beta = beta

    def __call__(self, x: np.ndarray) -> float:
        beta = self.beta
        f = np.correlate(x, 1.0 - x, mode="full")
        fb = beta * f
        m = np.max(fb)
        scaled = fb - m
        exp_scaled = np.exp(scaled)
        sum_exp = np.sum(exp_scaled)
        # (max + log sum)/beta  ==  log-sum-exp / beta
        return (m + np.log(sum_exp)) / beta

    def gradient(self, x: np.ndarray) -> np.ndarray:
        beta = self.beta
        n = x.shape[0]
        f = np.correlate(x, 1.0 - x, mode="full")
        fb = beta * f
        m = np.max(fb)
        scaled = fb - m
        exp_scaled = np.exp(scaled)
        sum_exp = np.sum(exp_scaled)
        soft = exp_scaled / sum_exp            # softmax, sums to 1
        grad = np.ones(n)                     # base term Σ soft = 1
        # contributions from shifted copies of x
        for shift in range(-n + 1, n):
            wk = soft[shift + n - 1]
            if wk == 0.0:
                continue
            if shift >= 0:
                grad[: n - shift] -= wk * x[shift:]
            else:
                grad[-shift:] -= wk * x[: n + shift]
            inv = -shift
            if inv >= 0:
                grad[: n - inv] -= wk * x[inv:]
            else:
                grad[-inv:] -= wk * x[: n + inv]
        return grad


class SumConstraint:
    __slots__ = ("target",)

    def __init__(self, target: float):
        self.target = target

    def __call__(self, x: np.ndarray) -> float:
        return float(np.sum(x) - self.target)

    def jac(self, x: np.ndarray) -> np.ndarray:
        return np.ones_like(x)


def optimize_h(start_h: np.ndarray, betas, time_limit: float) -> np.ndarray:
    """Run a sequence of SLSQP optimisations with increasing smoothing."""
    n = start_h.shape[0]
    target = n / 2.0
    bounds = [(0.0, 1.0) for _ in range(n)]
    eq_con = SumConstraint(target)
    h = start_h.copy()
    t0 = time.time()
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        obj = SmoothObj(beta)
        res = minimize(
            obj,
            h,
            method="SLSQP",
            jac=obj.gradient,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": eq_con, "jac": eq_con.jac}],
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
        # take whatever the optimizer returned
        h = np.clip(res.x, 0.0, 1.0)
        # enforce exact sum (tiny drift may appear)
        if abs(h.sum() - target) > 1e-7:
            h = project_to_sum_and_bounds(h, target)
    return h


def hill_climb(h: np.ndarray, cur_val: float, rng: np.random.RandomState,
               time_limit: float) -> (np.ndarray, float):
    """Simple pairwise move hill‑climb for the true max objective."""
    start = time.time()
    n = h.shape[0]
    best_h = h.copy()
    best_val = cur_val
    while time.time() - start < time_limit:
        i, j = rng.randint(0, n, 2)
        if i == j:
            continue
        max_eps = min(best_h[i], 1.0 - best_h[j])
        if max_eps <= 0.0:
            continue
        eps = rng.uniform(0.0, max_eps)
        cand = best_h.copy()
        cand[i] -= eps
        cand[j] += eps
        val = compute_c5(cand)
        if val < best_val - 1e-12:
            best_h, best_val = cand, val
    return best_h, best_val


def random_feasible_h(n: int, rng: np.random.RandomState) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum n/2."""
    x = rng.rand(n)
    return project_to_sum_and_bounds(x, n / 2.0)
