def project_sum(h, target_sum):
    # ensure h in [0,1] and sum = target_sum within tolerance.
    h = np.clip(h, 0, 1)
    # simple iterative approach
    for _ in range(10):
        s = np.sum(h)
        if abs(s - target_sum) < 1e-8:
            break
        # compute scaling factor, but ensure we don't scale too extreme
        factor = target_sum / s
        h = h * factor
        # clip again
        h = np.clip(h, 0, 1)
    # final adjustment: distribute remaining diff across non-saturated entries
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size > 0:
            h[free] += diff / free.size
            h = np.clip(h, 0, 1)
    # final check
    # If still not exact, scale again without bounds maybe.
    # Ensure sum ~ target_sum within tolerance:
    s = np.sum(h)
    if abs(s - target_sum) > 1e-6:
        # last resort: distribute diff proportionally across all entries not at bound

def gradient_step(h, dx, step):
    N = h.size
    # compute correlation full
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    # shift s = max_idx - (N-1) because correlation vector length 2N-1 => index shift relative to zero shift at N-1.
    s = max_idx - (N - 1)
    # compute gradient g sized N.
    g = np.zeros_like(h)
    # For each j: if j+s in range, add (1 - h[j+s])
    idx1 = np.arange(N)
    idx2 = idx1 + s
    mask1 = (idx2 >= 0) & (idx2 < N)
    g[mask1] += (1.0 - h[idx2[mask1]])
    # For each j: if j-s in range, subtract h[j-s]
    idx3 = idx1 - s
    mask2 = (idx3 >= 0) & (idx3 < N)
    g[mask2] -= h[idx3[mask2]]
    # Now g is derivative of correlation but not multiplied by dx; but we can incorporate step size accordingly.
    # Step: h_new = h - step * g
    h_new = h - step * g
    # clip and project sum.
    # Clip to [0,1]
    h_new = np.clip(h_new, 0, 1)
    # Project sum to target: target_sum = N/2.
    target_sum = N/2
    h_new = project_sum(h_new, target_sum)
    return h_new, corr[max_idx] * dx

def project_sum(h, target):
    # Clip to [0,1] then adjust using a water-filling algorithm.
    h = np.clip(h, 0.0, 1.0)
    # Use simple iterative adjustment as described.
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Determine free indices where moving won't exceed bounds.
        free = (h > 0) & (h < 1)
        if not np.any(free):
            # No free slots; just distribute diff evenly ignoring bounds (will be corrected next iteration)
            delta = diff / len(h)
            h += delta
            h = np.clip(h, 0, 1)
            continue
        # distribute diff equally among free entries
        delta = diff / free.sum()
        h[free] += delta
        h = np.clip(h, 0, 1)

    s = h.sum()
    if abs(s - target) > 1e-8:
        # Scale all non-zero entries accordingly (may violate bounds), do final scaling
        # but trust earlier loops.
        h = h * (target / s)
        h = np.clip(h, 0, 1)
    # Final ensure sum close to target; if not maybe adjust smallest diff
    # but accept.
    return h

import time

# Helper functions

def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_sum(h, target):
    """Project h onto [0,1] and enforce sum(h)=target (within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    # water‑filling style adjustment
    for _ in range(12):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            # no free variables – distribute uniformly and re‑clip
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / free.sum()
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final scaling if still off (rare)
    s = h.sum()
    if abs(s - target) > 1e-9:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        # one more water‑fill pass
        h = project_sum(h, target)
    return h


def upsample_to_n(src, n_target):
    """Linearly up‑sample src (array) to length n_target keeping the integral."""
    n_src = src.size
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_target, endpoint=False)
    dst = np.interp(x_dst, x_src, src.astype(np.float64))
    target_sum = n_target / 2.0
    dst = dst * (target_sum / dst.sum())
    dst = np.clip(dst, 0.0, 1.0)
    dst = project_sum(dst, target_sum)
    return dst
