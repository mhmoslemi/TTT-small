import time
import numpy as np

# ----- helper functions ----------------------------------------------------

def c5_value(h):
    """Return the maximum overlap integral for a step function given by array h."""
    # correlation of h with (1-h); multiply by dx = 2/n
    n = h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def random_binary(n):
    """Return a binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def improve_once(cur, cur_val):
    """Propose a swap of a 1 and a 0, keep it iff it does not increase the objective."""
    # locate positions of 1s and 0s
    ones = np.where(cur == 1.0)[0]
    zeros = np.where(cur == 0.0)[0]
    # random opposite pair
    i1 = np.random.choice(ones)
    i0 = np.random.choice(zeros)
    # perform swap
    cur[i1] = 0.0
    cur[i0] = 1.0
    new_val = c5_value(cur)
    if new_val <= cur_val:
        return cur, new_val, True
    # revert
    cur[i1] = 1.0
    cur[i0] = 0.0
    return cur, cur_val, False


def anneal_optimize(n, time_allow, start_vec=None):
    """Simulated‑annealing style search for a good binary vector of length n."""
    start = time.time()
    # initialise current solution
    if start_vec is not None:
        cur = start_vec.astype(np.float64).copy()
    else:
        cur = random_binary(n)
    cur_val = c5_value(cur)
    best_h = cur.copy()
    best_val = cur_val

    # temperature schedule
    T = 0.02
    decay = 0.9995
    it = 0
    # counters for stagnation
    no_improve = 0

    while time.time() - start < time_allow:
        # propose a swap
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)

        cur[i1] = 0.0
        cur[i0] = 1.0
        new_val = c5_value(cur)

        # acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert swap
            cur[i1] = 1.0
            cur[i0] = 0.0
            no_improve += 1

        # temperature decay
        it += 1
        if it % 500 == 0:
            T *= decay

        # occasional restart if stuck too long
        if no_improve > 5000:
            cur = random_binary(n)
            cur_val = c5_value(cur)
            no_improve = 0

        # early exit if we already beat the target
        if best_val < 0.3808:
            break

    return best_h, best_val


# ----- main entry point ----------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a low Erdős C₅ bound."""
    np.random.seed(seed)
    start_total = time.time()

    # gather candidate discretisation sizes
    candidates = []
    try:
        init = initial_h_values  # provided by the harness, may be None
        n0 = len(init)
        candidates.append(n0)
    except NameError:
        init = None
        n0 = None

    # add a few larger even sizes (the search is cheap for these lengths)
    extra = [96, 120, 144, 168, 192, 224, 256]
    for n in extra:
        if n != n0:
            candidates.append(n)

    best_h = None
    best_val = np.inf
    best_n = None

    # simple time division – give each size a fair share of the budget
    remaining = budget_s - 2.0  # small safety margin
    per_n = max(5.0, remaining / max(len(candidates), 1))

    for n in candidates:
        # adjust allowed time to stay within global budget
        elapsed = time.time() - start_total
        if elapsed + per_n > budget_s:
            per_n = max(0.0, budget_s - elapsed - 0.5)
        if per_n <= 0:
            break

        # initial vector for this size (use provided parent if sizes match)
        start_vec = None
        if init is not None and n == n0:
            start_vec = init

        # run the optimiser
        h_opt, val_opt = anneal_optimize(n, per_n, start_vec)

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            best_n = n

        # stop early if we have already passed the required bound
        if best_val < 0.3808:
            break

    # final safety check – ensure constraints are met
    if best_h is None:
        # fallback: a simple balanced random vector
        best_n = 80
        best_h = random_binary(best_n)
        best_val = c5_value(best_h)

    return best_h, float(best_val), int(best_n)
