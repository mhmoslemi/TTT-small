def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')

def project_to_sum(h, target):

def smooth_max_neg_Q(h, tau):
    Q = np.correlate(h, h, mode='full')
    # compute softmax weights for -tau*Q
    # Use logsumexp for stability
    logs = -tau * Q
    max_log = np.max(logs)
    w = np.exp(logs - max_log)
    w = w / w.sum()
    # compute gradient
    n = len(h)
    grad = np.zeros_like(h)
    # from l=0..n-1
    w0 = w[n-1]
    grad -= 2 * w0 * h
    for l in range(1, n):
        w_l = w[n-1 + l]  # same as w[n-1 - l] due to symmetry
        coeff = 2 * w_l
        # contributions
        grad[:n - l] -= coeff * h[l:]
        grad[l:] -= coeff * h[:n - l]
    # gradient of smooth max = - sum w_l * ∂Q/∂h , already negative sign inc.
    return grad, Q, w  # maybe not needed
```

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper: projection onto the box [0,1] with a prescribed sum.
def _project_sum_box(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Return the projection of x onto {0<=y<=1, sum(y)=target_sum}."""
    lo, hi = -1.0, 1.0
    # binary search for the shift λ such that sum(clamp(x+λ)) = target_sum
    for _ in range(50):
        lam = (lo + hi) * 0.5
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


# ----------------------------------------------------------------------
# Compute the actual C5 value for a discretised step function.
def _c5_of_h(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


# ----------------------------------------------------------------------
# Core optimisation: projected gradient descent on a smooth approximation
# of max_k ( h·(1-h shifted) ).
def _optimize_h(
    n: int,
    x0: np.ndarray,
    max_iters: int,
    tau_start: float,
    tau_end: float,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    target = n / 2.0
    h = _project_sum_box(x0, target)

    best_h = h.copy()
    best_val = _c5_of_h(h)

    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        tau = tau_start + (tau_end - tau_start) * it / max_iters

        # Autocorrelation Q_k = Σ_i h_i h_{i+k}
        Q = np.correlate(h, h, mode="full")          # length 2n‑1, lag = -(n‑1)…+(n‑1)

        # Soft‑max weights for -τ·Q   (focus on the smallest Q → largest overlap)
        logw = -tau * Q
        w = np.exp(logw - logsumexp(logw))

        # Gradient of the smooth max (negative of weighted sum of ∂Q/∂h)
        grad = np.empty_like(h)
        grad.fill(0.0)

        # lag 0 contribution
        w0 = w[n - 1]
        grad -= 2.0 * w0 * h

        # positive lags (negative are symmetric because w is symmetric)
        for lag in range(1, n):
            wl = w[n - 1 + lag]          # weight for +lag (same as for –lag)
            coeff = 2.0 * wl
            # contribution of h_{i+lag}
            grad[: n - lag] -= coeff * h[lag:]
            # contribution of h_{i‑lag}
            grad[lag:]     -= coeff * h[: n - lag]

        # Simple diminishing step size
        step = 0.5 / np.sqrt(it + 1.0)
        h = h - step * grad
        h = _project_sum_box(h, target)

        val = _c5_of_h(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    return best_h, best_val


# ----------------------------------------------------------------------

    """
    Return a candidate step function for the Erdős C_5 problem.

    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Keep the best solution seen so far.
    best_h = None
    best_c5 = np.inf
    best_n = None

    # Use the supplied parent construction if it exists.

    # Candidate sample sizes (keep them modest for speed).
