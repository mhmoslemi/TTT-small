import numpy as np
import random
import time
from math import exp

def _max_corr(h):
    """Maximum of h * (1-h) correlation (full mode)."""
    # np.correlate returns a float array; max is the objective value.
    return np.max(np.correlate(h, 1.0 - h, mode="full"))

def _max_corr_and_lag(h):
    """Return (max value, lag) of the correlation."""
    c = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(c))
    maxv = float(c[idx])
    lag = idx - (len(h) - 1)
    return maxv, lag

def _initial_binary(n, seed):
    """Generate a binary start vector of length n with exactly n/2 ones."""
    np.random.seed(seed)
    rand = np.random.choice(n, n // 2, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[rand] = 1.0
    return h

def _upgrade_initial(h_init, target_n, seed):
    """Upsample/adjust a given initial array to length target_n and make it binary."""
    n0 = h_init.shape[0]
    if n0 == target_n:
        h = (h_init >= 0.5).astype(np.float64)
    else:
        # linear interpolation onto the new grid
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        interp = np.interp(x_new, x_old, h_init)
        h = (interp >= 0.5).astype(np.float64)

    # enforce exactly half ones
    need = target_n // 2 - int(h.sum())
    if need > 0:
        zeros = np.where(h == 0)[0]
        flip = np.random.choice(zeros, need, replace=False)
        h[flip] = 1.0
    elif need < 0:
        ones = np.where(h == 1)[0]
        flip = np.random.choice(ones, -need, replace=False)
        h[flip] = 0.0
    return h

def _propose_targeted_swap(h, ones, zeros, rng):
    """Pick a swap that directly attacks the current worst lag."""
    _, lag = _max_corr_and_lag(h)
    n = len(h)
    if lag >= 0:
        left = h[: n - lag]
        right = h[lag:]
        mism = np.where((left == 1) & (right == 0))[0]
    else:
        d = -lag
        left = h[d:]
        right = h[: n - d]
        mism = np.where((left == 1) & (right == 0))[0] + d
    if mism.size == 0:
        # fallback to random
        i = rng.choice(ones)
        j = rng.choice(zeros)
        return i, j
    i = int(rng.choice(mism))
    # try to swap with a zero that is not i (any zero works)
    j = rng.choice(zeros)
    while j == i:
        j = rng.choice(zeros)
    return i, j

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a low Erdős C₅ bound."""
    start_time = time.time()
    rng = np.random.RandomState(seed)
    random.seed(seed)

    # ---------------------------------------------------------------
    # Choose discretisation size (even, <1000). 200 gives a comfortable
    # trade‑off between resolution and speed.
    # ---------------------------------------------------------------
    N = 200

    # ---------------------------------------------------------------
    # Initialise h either from the supplied construction or randomly.
    # ---------------------------------------------------------------
    if "initial_h_values" in globals():
        h = _upgrade_initial(initial_h_values, N, seed)
    else:
        h = _initial_binary(N, seed)

    # Ensure feasibility (should already hold).
    assert h.shape[0] == N
    assert np.all((h == 0) | (h == 1))
    assert int(h.sum()) == N // 2

    # ---------------------------------------------------------------
    # Book‑keeping structures.
    # ---------------------------------------------------------------
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))

    cur_h = h.copy()
    cur_M = _max_corr(cur_h)
    best_h = cur_h.copy()
    best_M = cur_M

    # Target maximal count (integer) derived from the required bound:
    target_max = int(np.floor(0.3808 * (N / 2)))  # ≤ 38 for N=200

    # Simulated‑annealing schedule (linear cooling).
    T_start = 5.0
    T_end = 0.01
    # use 95 % of the allocated wall‑time for the search.
    time_limit = budget_s * 0.95

    # Parameters
    prob_targeted = 0.5   # fraction of moves that are targeted
    iter_count = 0

    while time.time() - start_time < time_limit:
        # temperature based on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / time_limit
        T = T_start * (1.0 - frac) + T_end * frac

        # propose a swap
        if rng.rand() < prob_targeted:
            i, j = _propose_targeted_swap(cur_h, ones, zeros, rng)
        else:
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # perform the swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0

        new_M = _max_corr(cur_h)

        # acceptance test
        accept = False
        if new_M <= cur_M:
            accept = True
        else:
            # Metropolis criterion
            delta = new_M - cur_M
            if T > 0 and rng.rand() < exp(-delta / T):
                accept = True

        if accept:
            # update lists
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)

            cur_M = new_M
            if cur_M < best_M:
                best_M = cur_M
                best_h = cur_h.copy()
                # early exit if we already beat the target
                if best_M <= target_max:
                    break
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0

        iter_count += 1

    # final bound
    dx = 2.0 / N
    c5_bound = best_M * dx

    return best_h, c5_bound, N
