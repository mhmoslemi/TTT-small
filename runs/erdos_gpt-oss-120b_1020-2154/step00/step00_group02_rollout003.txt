        # Clip to binary? Maybe it's binary? Might be fractional previously; we can round to binary while keeping sum constraint
        init_binary = (init >= 0.5).astype(np.float64)
        # adjust sum to exactly n/2 is needed
        # compute diff
        diff = int(n/2 - np.sum(init_binary))
        if diff > 0:
            # add ones at zeros nearest something
            zeros = np.where(init_binary == 0)[0]
            idx = rng.choice(zeros, size=diff, replace=False)
            init_binary[idx] = 1
        elif diff < 0:
            ones = np.where(init_binary == 1)[0]
            idx = rng.choice(ones, size=-diff, replace=False)
            init_binary[idx] = 0
        h = init_binary.copy()
    else:
        # random binary with exactly n/2 ones
        h = np.zeros(n, dtype=np.float64)
        ones_idx = rng.choice(n, size=n//2, replace=False)
        h[ones_idx] = 1.0

    best_h = h.copy()
    best_score = compute_c5(best_h)

    C_cur = best_score
    # Precompute ones/zeros list
    ones = list(np.where(h==1)[0])
    zeros = list(np.where(h==0)[0])

    # SA parameters
    Tmax = 0.2  # adjust maybe 0.2
    Tmin = 1e-4
    max_iters = 300000  # will be truncated by time
    for it in range(max_iters):
        if time.time() - start > time_limit * 0.99:
            break
        # temperature schedule
        T = Tmax * (1 - it / max_iters) + Tmin
        # pick swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i] # produce new array (swap in place)
        # evaluate
        C_new = compute_c5(h)
        # acceptance
        accept = False
        if C_new < C_cur:
            accept = True
        else:
            # metropolis
            if T > 0:
                prob = np.exp(-(C_new - C_cur) / T)
                if rng.random() < prob:
                    accept = True
        if accept:
            C_cur = C_new
            # update lists (swap indices)
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if C_new < best_score:
                best_score = C_new
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        # optionally restart if stuck? maybe after some iterations with no improvement
        # not needed initially
    return best_h, best_score

def greedy_optimize(h, rng, max_iters, target=0.3808):
    n=len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        max_idx = np.argmax(corr)
        Cmax = corr[max_idx]
        if Cmax <= target:
            best_h = h.copy()
            best_score = Cmax
            break
        if Cmax < best_score:
            best_score = Cmax
            best_h = h.copy()
        k = max_idx - (n - 1)
        # Determine overlapping indices
        if k >= 0:
            i_range = np.arange(0, n - k)
        else:
            i_range = np.arange(-k, n)
        idx_i = i_range
        idx_j = idx_i + k
        # compute masks
        h_i = h[idx_i]
        h_j = h[idx_j]
        maskA = (h_i == 1) & (h_j == 0)
        maskB = (h_i == 0) & (h_j == 1)
        idxA = idx_i[maskA]
        idxB = idx_i[maskB]
        if idxA.size > 0 and idxB.size > 0:
            i_swap = rng.choice(idxA)
            j_swap = rng.choice(idxB)
            h[i_swap], h[j_swap] = h[j_swap], h[i_swap]
            # continue to next iteration
            continue
        else:
            # cannot improve for this shift, perform random swap
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            i_swap = rng.choice(ones_idx)
            j_swap = rng.choice(zeros_idx)
            h[i_swap], h[j_swap] = h[j_swap], h[i_swap]
            continue
    return best_h, best_score

def triangle_h(n):
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx  # point at center of each bin
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    # sum needed:
    target_sum = n / 2
    current_sum = h.sum()
    # scaling
    scale = target_sum / current_sum
    h = h * scale
    # clip to [0,1] (should be ok)
    h = np.clip(h, 0.0, 1.0)
    # fix sum again (since clipping may change sum)
    current_sum = h.sum()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_to_bounds_and_sum(v, target):
    h = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        residual = target - h.sum()
        if abs(residual) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        nfree = np.count_nonzero(free)
        if nfree == 0:
            # cannot adjust via free variables, distribute to any feasible
            # Choose first element not at bound
            if residual > 0:
                idx = np.where(h < 1)[0]
            else:
                idx = np.where(h > 0)[0]
            if idx.size == 0:
                break
            i = idx[0]
            h[i] = min(max(h[i] + residual, 0.0), 1.0)
            break
        delta = residual / nfree
        h[free] += delta

        h = np.clip(h, 0.0, 1.0)

import time
import numpy as np


def compute_c5(h: np.ndarray) -> float:
    """Discrete evaluation of the Erdős overlap functional."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def project_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the box [0,1]^n
    together with the affine constraint sum(v) = target.
    """
    h = np.clip(v, 0.0, 1.0)
    for _ in range(20):
        residual = target - h.sum()
        if abs(residual) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        nfree = int(free.sum())
        if nfree == 0:
            # cannot distribute freely – adjust a single feasible entry
            if residual > 0:
                cand = np.where(h < 1)[0]
            else:
                cand = np.where(h > 0)[0]
            if cand.size == 0:
                break
            i = cand[0]
            h[i] = min(max(h[i] + residual, 0.0), 1.0)
            break
        delta = residual / nfree
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)

    # final tiny correction if needed
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        free = (h > 0) & (h < 1)
        if free.any():
            i = np.flatnonzero(free)[0]
            h[i] = min(max(h[i] + residual, 0.0), 1.0)
    return h


def triangle_step(n: int) -> np.ndarray:
    """
    Build a symmetric tent (triangular) step function on [0,2]
    with the required integral 1.
    """
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx               # mid‑point of each interval
    h_raw = np.maximum(0.0, 1.0 - np.abs(x - 1.0))   # exact triangle shape
    target = n / 2.0
    return project_bounds_sum(h_raw, target)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C₅ bound and the number of points."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # resolution – enough points to approximate the continuous optimum well
    n = 200
    h = triangle_step(n)
    best_score = compute_c5(h)

    # optional cheap local refinement (random mass swaps)
    max_iter = 2000
    target_sum = n / 2.0
    for _ in range(max_iter):
        if time.time() - start > budget_s * 0.95:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a small amount from i to j while staying in [0,1]
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0.0:
            continue
        delta = delta_max * rng.random()
        if delta == 0.0:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        new_score = compute_c5(h_new)
        if new_score < best_score - 1e-12:
            h = h_new
            best_score = new_score

    # final projection to guarantee feasibility
    h = project_bounds_sum(h, target_sum)
    best_score = compute_c5(h)

    return (h.tolist(), float(best_score), int(n))
