    c = np.correlate(h, 1 - h, mode='full')
    # softmax approximation: g = (1/beta) * log sum exp(beta*c)
    max_c = np.max(c)   # for stability
    expb = np.exp(beta * (c - max_c))
    soft = (np.log(np.sum(expb)))/beta + max_c    # actual objective
    # probabilities p_i = expb_i / sumexpb
    p = expb / np.sum(expb)
    # Compute gradient G = sum_i p_i * dc_i/dh
    N = len(h)
    G = np.zeros_like(h)
    # compute for each shift i (index in c). For correlate, shift index i corresponds to lag = i - (N-1)
    for idx, pi in enumerate(p):
        lag = idx - (N - 1)

def project_box_simplex(v, target):
    # v: np array
    # returns x in [0,1] with sum = target closest to v.
    # Use water-filling: find lambda s.t. sum_i min(max(v_i - lambda, 0), 1) = target.
    lo = -np.max(v) - 1.0  # ensure all become >1 when subtracting lo? Actually if lambda very negative, v_i - lambda large => min(...,1) =1 => sum = N.
    hi = np.max(v) + 1.0   # ensure all become 0 when lambda large positive => sum =0.
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0, 1)
    # Due to rounding may not exactly meet target, we adjust small residual.
    residual = target - x.sum()
    if abs(residual) > 1e-8:
        # distribute residual among non-boundary entries
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            # Add residual equally
            x[free] += residual / free.sum()
            # Clip again
            x = np.clip(x, 0, 1)
        else:
            # If all at bounds, adjust smallest difference
            # Add to entries where x < 1
            idx = np.argsort(x)  # ascending
            for i in idx:
                add = min(1 - x[i], residual)
                x[i] += add
                residual -= add
                if residual <= 1e-12:
                    break
    return x

def gradient_shift(h, lag):
    N = h.shape[0]
    G = np.zeros_like(h)
    if lag >= 0:
        # For indices where both i and i+lag are valid
        # term (1 - h[i+lag]) for i from 0..N-lag-1 added to G[i]
        G[:N - lag] += 1 - h[lag:]
        # term -h[i] for i from lag..N-1 added to G[i]
        G[lag:] -= h[:N - lag]
    else:
        l = -lag
        # shift left
        # For i from l..N-1, we have (1 - h[i-l]) contribution for G[i]
        G[l:] += 1 - h[:N - l]
        # For i from 0..N-l-1, we have -h[i+l] contribution
        G[:N - l] -= h[l:]
    return G

    # Option: use bisection lambda such that X = clip(v - λ, 0,1) sum = target.

        X = np.clip(v - mid, 0.0, 1.0)

            lo = mid
        else:

    X = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

    residual = target - X.sum()

            if residual > 0:
                # add to max possible

                    if add <= 0:

import time, numpy as np

def project_box_simplex(v, target_sum):
    # Clip to [0,1] after shifting by lambda via bisection.
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust residual
    residual = target_sum - x.sum()
    if abs(residual) > 1e-12:
        free = (x > 1e-10) & (x < 1 - 1e-10)
        if free.any():
            add = residual / free.sum()
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback simple correction
            idx = np.arange(len(x))
            if residual > 0:
                for i in idx[x < 1 - 1e-12]:
                    inc = min(1 - x[i], residual)
                    x[i] += inc
                    residual -= inc
                    if residual <= 1e-12:
                        break
            else:
                for i in idx[x > 1e-12]:
                    dec = min(x[i], -residual)
                    x[i] -= dec
                    residual += dec
                    if residual >= -1e-12:
                        break
    return x

def gradient_shift(h, lag):
    N = h.shape[0]
    G = np.zeros_like(h)
    if lag >= 0:
        L = N - lag
        if L > 0:
            G[:L] += 1 - h[lag:]
            G[lag:] -= h[:L]
    else:
        l = -lag
        L = N - l
        if L > 0:
            G[l:] += 1 - h[:L]
            G[:L] -= h[l:]
    return G

def compute_c5(h):
    N = h.shape[0]
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * (2.0 / N)

import time

# ------------------------------------------------------------
# projection of a vector onto the box [0,1] together with the
# linear constraint sum(v) = target (the Euclidean projection)
# ------------------------------------------------------------
def _project_box_simplex(v, target):
    # Bisection on the Lagrange multiplier λ for the formula
    #   x_i = clip(v_i - λ, 0, 1)  and  Σ x_i = target
    lo, hi = -2.0, 2.0               # wide enough bounds
    for _ in range(60):              # enough for double precision
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny residual due to numerical errors – fix it quickly
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        if free.any():
            add = residual / free.sum()
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
        else:                           # fallback simple correction
            idx = np.arange(len(x))
            if residual > 0:            # need to add mass
                for i in idx[x < 1.0 - 1e-12]:
                    inc = min(1.0 - x[i], residual)
                    x[i] += inc
                    residual -= inc
                    if residual <= 1e-12:
                        break
            else:                       # need to remove mass
                for i in idx[x > 1e-12]:
                    dec = min(x[i], -residual)
                    x[i] -= dec
                    residual += dec
                    if residual >= -1e-12:
                        break
    return x


# ------------------------------------------------------------
# gradient of a single correlation term c_lag = Σ h_i (1-h_{i+lag})
# ------------------------------------------------------------
def _grad_one_shift(h, lag):
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        L = n - lag
        if L > 0:
            g[:L]      += 1.0 - h[lag:]      # (1 - h_{i+lag}) part
            g[lag:]    -= h[:L]             # -h_{i-lag} part
    else:                     # lag < 0
        l = -lag
        L = n - l
        if L > 0:
            g[l:]      += 1.0 - h[:L]
            g[:L]      -= h[l:]
    return g


# ------------------------------------------------------------
# exact C5 value for a given vector h
# ------------------------------------------------------------
def _c5_val(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)
