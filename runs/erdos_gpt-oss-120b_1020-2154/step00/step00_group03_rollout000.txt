def c5_score(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    dx = 2.0 / len(h)
    return float(max_val * dx)

def local_search_bin(h0, time_budget, rng):

    best_h = h0.copy()

    ones = np.flatnonzero(best_h == 1)

        else:
            # revert swap
            best_h[i] = 1
            best_h[j] = 0
        iter += 1
    return best_h, best_score

def c5_score(h):
    N = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    return float(max_val * (2.0 / N))

def binary_from_continuous(arr):
    # arr as float; choose top half values as ones
    N = len(arr)
    half = N // 2
    idx = np.argsort(-arr)  # descending
    binarr = np.zeros(N, dtype=int)
    binarr[idx[:half]] = 1
    return binarr

def local_search_bin(init_h, time_budget, rng):
    cur_h = init_h.copy()
    cur_score = c5_score(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score
    N = len(cur_h)
    start = time.time()
    # initial temperature...
    T0 = 0.02  # this is small relative to typical increments (maybe 0.01)
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # compute temperature linearly decaying to near zero
        T = T0 * max(0.0, 1 - elapsed / time_budget)
        # propose swap
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_score = c5_score(cur_h)
        # decide acceptance
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            # probability depends on temperature
            # scale delta by typical values (maybe 0.001)
            delta = new_score - cur_score
            # If T is zero, reject
            if T > 0:
                if rng.random() < np.exp(-delta / T):
                    accept = True
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0

import time
import numpy as np

def _c5_score(h: np.ndarray) -> float:
    """Return the C5 value for a binary (or [0,1]) vector h."""
    n = h.shape[0]
    # correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def _binarize_from_continuous(arr: np.ndarray) -> np.ndarray:
    """Turn a real vector into a binary one with exactly half the entries =1."""
    n = arr.shape[0]
    half = n // 2
    idx = np.argsort(-arr)               # descending order
    bin_arr = np.zeros(n, dtype=int)
    bin_arr[idx[:half]] = 1
    return bin_arr


def _local_search_bin(init_h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """
    Simulated‑annealing hill climbing on a binary vector with exactly n/2 ones.
    Returns the best vector found and its C5 score.
    """
    cur_h = init_h.copy()
    cur_score = _c5_score(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    n = cur_h.shape[0]
    start = time.time()
    # initial temperature – roughly the typical improvement step
    T0 = 0.02

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # linear cooling
        T = max(T0 * (1.0 - elapsed / time_budget), 1e-12)

        # pick a random 1 and a random 0 and swap them
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_score = _c5_score(cur_h)

        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            # Metropolis acceptance
            delta = new_score - cur_score
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0

    return best_h, best_score

    """
    rng = np.random.default_rng(seed)
    start_all = time.time()

    # ----------------------------------------------------------------------
    # 1. Obtain a feasible starting vector.
    # ----------------------------------------------------------------------
    try:
        raw = initial_h_values          # pre‑imported by the harness
    except NameError:
        raw = None

    if raw is None:
        # no parent supplied – build a random binary vector of length 70
        n0 = 70
        h0 = np.zeros(n0, dtype=int)
        h0[rng.choice(n0, size=n0 // 2, replace=False)] = 1
    else:
        arr = np.asarray(raw)
        n0 = arr.shape[0]
        # turn the possibly‑continuous parent into a proper binary vector
        if not np.all(np.logical_or(arr == 0, arr == 1)):
            h0 = _binarize_from_continuous(arr.astype(float))
        else:
            # already binary but the count might be off by a few entries
            half = n0 // 2
            if arr.sum() != half:
                h0 = _binarize_from_continuous(arr.astype(float))
            else:
                h0 = arr.astype(int)

    # sanity checks
    n0 = h0.shape[0]
    assert n0 % 2 == 0, "problem size must be even for a 0/1 solution"
    assert h0.sum() == n0 // 2, "initial vector does not satisfy the measure constraint"

    # ----------------------------------------------------------------------
    # 2. Base optimisation at the given resolution.
    # ----------------------------------------------------------------------
    time_used = time.time() - start_all
    time_left = max(0.0, budget_s - time_used)

    # allocate roughly 35 % of the remaining budget to the base size
    base_budget = min(time_left * 0.35, 300.0)
    best_h, best_score = _local_search_bin(h0, base_budget, rng)

    # ----------------------------------------------------------------------
    # 3. Progressive up‑sampling and refinement.
    #    Each up‑sampling doubles the resolution; we stop when we either run
    #    out of time or exceed 800 points (still far below the 1 k limit).
    # ----------------------------------------------------------------------
    current_h = best_h
    current_n = n0
    while True:
        time_used = time.time() - start_all
        time_left = max(0.0, budget_s - time_used)
        if time_left < 5.0:
            break
        # try to double the resolution
        new_n = current_n * 2
        if new_n > 800:          # keep the size modest for speed
            break

        # up‑sample by simple repetition (preserves the exact half‑measure)
        up_h = np.repeat(current_h, 2)

        # add a small random perturbation to break long runs which often
        # appear after a plain repeat
        # (flip a few randomly chosen positions while keeping the count)
        perturb = min(5, new_n // 20)          # ≤5 flips
        for _ in range(perturb):
            ones_idx = np.where(up_h == 1)[0]
            zeros_idx = np.where(up_h == 0)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            up_h[i] = 0
            up_h[j] = 1

        # allocate a chunk of the remaining time to the finer grid
        refine_budget = min(time_left * 0.45, 500.0)
        refined_h, refined_score = _local_search_bin(up_h, refine_budget, rng)

        # keep the best among the previous and the refined solution
        if refined_score < best_score:
            best_h, best_score = refined_h, refined_score
            current_h, current_n = refined_h, new_n
        else:
            # still keep the finer grid for a possible next scaling step
            current_h, current_n = up_h, new_n

        # loop again if time permits

    # ----------------------------------------------------------------------
    # 4. Return the best solution discovered.
    # ----------------------------------------------------------------------
    final_h = best_h.astype(float)          # ensure floating point output
    final_score = _c5_score(final_h)
    final_n = final_h.shape[0]

    return final_h, final_score, final_n
