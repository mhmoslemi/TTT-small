def anneal(n, time_limit):

    h = random binary with sum n/2.
    best_h = h.copy()

    cur_h = h.copy()

    start = time.time()
    iteration = 0
    while time.time() - start < time_limit:
        # random move: swap random 1-index and 0-index
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        cur_h[j] = 1
        new_val = evaluate(cur_h)
        # acceptance criteria
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val)/T):
            cur_val = new_val
            # update best
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
        iteration += 1
        # schedule temperature decreasing
        progress = (time.time() - start) / time_limit
        T = T0 * (1 - progress)

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def hill_climb_fixed(start_h, t_end, rng):
    h = start_h.astype(float).copy()
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_val = cur_c5
    # indexes of 1s and 0s
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    # iteration counters
    no_improve = 0
    # define thresholds
    RESTART_AFTER = 10000  # attempts without improvement
    while time.time() < t_end:
        # random swap indices
        idx_one = rng.integers(len(ones))
        idx_zero = rng.integers(len(zeros))
        i = ones[idx_one]
        j = zeros[idx_zero]
        # swap
        h[i] = 0.0
        h[j] = 1.0
        new_val = compute_c5(h)
        if new_val <= cur_c5:
            # accept
            cur_c5 = new_val
            # update index lists
            ones[idx_one] = j
            zeros[idx_zero] = i
            # record best
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # reject
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        if no_improve >= RESTART_AFTER:
            # random shuffle again
            h = np.concatenate([np.ones(n//2), np.zeros(n - n//2)])
            rng.shuffle(h)
            cur_c5 = compute_c5(h)
            best_val = cur_c5 if cur_c5 < best_val else best_val
            best_h = h.copy() if cur_c5 < best_val else best_h
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_improve = 0
    return best_h, best_val

import time
import math
import numpy as np

# ------------------------------------------------------------
# Core evaluation
# ------------------------------------------------------------
def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a binary step function h."""
    n = h.shape[0]
    # correlation of h with its complement (1-h) on the line, no wrap‑around
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * 2.0 / n)


# ------------------------------------------------------------
# Simple generators for a starting vector
# ------------------------------------------------------------
def _make_binary(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Force a {0,1} vector with exactly n/2 ones."""
    n = vec.shape[0]
    target = n // 2
    vec = (vec >= 0.5).astype(float)
    cur = int(vec.sum())
    if cur != target:
        diff = target - cur
        if diff > 0:
            zeros = np.where(vec == 0)[0]
            flip = rng.choice(zeros, size=diff, replace=False)
            vec[flip] = 1.0
        else:
            ones = np.where(vec == 1)[0]
            flip = rng.choice(ones, size=-diff, replace=False)
            vec[flip] = 0.0
    return vec


def _sturmian(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Balanced Sturmian word: h[i]=1 iff {i·α} < 0.5."""
    frac = (np.arange(n, dtype=float) * alpha) % 1.0
    h = (frac < 0.5).astype(float)
    return _make_binary(h, rng)


def _thue_morse(n: int, rng: np.random.Generator) -> np.ndarray:
    """Thue–Morse word (parity of popcount) truncated to length n."""
    h = np.empty(n, dtype=float)
    for i in range(n):
        h[i] = 1.0 if (i.bit_count() % 2 == 0) else 0.0
    return _make_binary(h, rng)


def _random_half(n: int, rng: np.random.Generator) -> np.ndarray:
    h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
    rng.shuffle(h)
    return h


# ------------------------------------------------------------
# Local optimisation – greedy hill climbing with restarts
# ------------------------------------------------------------
def _hill_climb(start: np.ndarray, t_end: float, rng: np.random.Generator):
    """Greedy hill climb on a binary vector until time runs out."""
    h = start.astype(float).copy()
    n = h.shape[0]
    cur_val = _c5_value(h)
    best_h = h.copy()
    best_val = cur_val

    # positions of 1s and 0s – maintained for O(1) swaps
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    restart_after = 8000  # attempts without improvement before a random restart
    stagnation = 0

    while time.time() < t_end:
        # pick a random 1 and a random 0 and swap them
        i_one = rng.integers(len(ones))
        i_zero = rng.integers(len(zeros))
        i = ones[i_one]
        j = zeros[i_zero]

        h[i] = 0.0
        h[j] = 1.0
        new_val = _c5_value(h)

        if new_val <= cur_val:
            # keep the swap
            cur_val = new_val
            ones[i_one] = j
            zeros[i_zero] = i
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
            stagnation += 1

        if stagnation >= restart_after:
            # random restart respecting the half‑ones constraint
            h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)])
            rng.shuffle(h)
            cur_val = _c5_value(h)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stagnation = 0

    return best_h, best_val

# Main entry point
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] with ∫h=1 that minimises
    max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    best_h = None
    best_val = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Use the provided construction (if any) as a warm start
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = _make_binary(h0, rng)
        n0 = h0.shape[0]
        val0 = _c5_value(h0)
        if val0 < best_val:
            best_val, best_h, best_n = val0, h0.copy(), n0

        # a short hill‑climb on the given instance
        t_end = min(deadline, time.time() + budget_s * 0.25)
        h_loc, v_loc = _hill_climb(h0, t_end, rng)
        if v_loc < best_val:
            best_val, best_h, best_n = v_loc, h_loc.copy(), n0

    # ------------------------------------------------------------------
    # Random explorations with simple structured starts
    # ------------------------------------------------------------------
    sizes = [80, 120, 160, 200, 240]
    alphas = [math.sqrt(2) - 1, (math.sqrt(5) - 1) / 2, 0.6180339887498949]  # three irrationals
    generators = []

    # explicit Sturmian seeds
    for n in sizes:
        for a in alphas:
            generators.append(lambda n=n, a=a: _sturmian(n, a, rng))

    # Thue–Morse (powers of two up to 256)
    for n in [64, 128, 256]:
        generators.append(lambda n=n: _thue_morse(n, rng))

    # plain random half‑ones
    for n in sizes:
        generators.append(lambda n=n: _random_half(n, rng))

    # ------------------------------------------------------------------
    # Main loop – keep trying until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # choose a generator, build a start vector and launch a short climb
        start_vec = generators[rng.integers(len(generators))]()
        n = start_vec.shape[0]
        # allocate a small time slice (5 s or the remaining budget, whichever is smaller)
        remaining = deadline - time.time()
        slice_time = min(5.0, remaining)
        if slice_time < 0.5:
            break
        t_end = time.time() + slice_time
        cur_val = _c5_value(start_vec)
        if cur_val < best_val:
            best_val, best_h, best_n = cur_val, start_vec.copy(), n
        h_opt, v_opt = _hill_climb(start_vec, t_end, rng)
        if v_opt < best_val:
            best_val, best_h, best_n = v_opt, h_opt.copy(), n

    # ------------------------------------------------------------------
    # Safety fallback
    # ------------------------------------------------------------------
    if best_h is None:
        best_n = 80
        best_h = _random_half(best_n, rng)
        best_val = _c5_value(best_h)

    best_h = np.array(best_h, dtype=float)
    return best_h, float(best_val), int(best_n)
