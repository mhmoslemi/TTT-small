def project(h, C, n_points):
    h_clipped = np.clip(h, 0, 1)
    S = np.sum(h_clipped)
    if S == C:
        return h_clipped
    elif S < C:
        # Need to add D
        D = C - S
        # Sort h_clipped in ascending order
        sorted_indices = np.argsort(h_clipped)
        # Distribute D to increase the smallest elements first, up to 1
        remaining = D
        for i in sorted_indices:
            if remaining <= 0:
                break
            max_increase = 1 - h_clipped[i]
            if max_increase <= 0:
                continue
            if remaining <= max_increase:
                h_clipped[i] += remaining
                remaining = 0
            else:
                h_clipped[i] += max_increase
                remaining -= max_increase
        # After this, sum should be C

        D = S - C
        # Sort h_clipped in descending order
        sorted_indices = np.argsort(-h_clipped)

            if remaining <= max_decrease:

                h_clipped[i] -= max_decrease
