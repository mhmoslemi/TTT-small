import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    n = 26

    rows = 5
    cols = 5
    grid = np.zeros((n, 2))
    radius = np.zeros(n)

    for i in range(n):
        row = i // cols

        radius[i] = 0.04  # Initial guess for radius

    def objective_function(params):

        return -np.sum(radii)

    constraints = []

    # Constraint: radius >= 0
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2 * n + i]})

    for i in range(n):
        def in_square(x, i=i):
            x_i, y_i = x[:2 * n][2 * i:2 * (i + 1)]

            return (x_i - r_i >= -1e-12) and (x_i + r_i <= 1 + 1e-12) and \

        constraints.append({'type': 'ineq', 'fun': in_square})
