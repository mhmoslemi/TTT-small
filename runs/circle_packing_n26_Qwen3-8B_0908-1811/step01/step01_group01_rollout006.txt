import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def generate_hexagonal_positions(n, square_size=1.0):
    """
    Generate initial positions for a hexagonal close packing layout.
    """
    positions = []
    for i in range(n):
        row = i // 6
        col = i % 6
        x = (col + 0.5 * (row % 2)) / 5.0
        y = row / 5.0
        if row % 2 == 1:
            y += 0.5 / 5.0
        x = square_size * x
        y = square_size * y
        positions.append([x, y])
    return np.array(positions)

def compute_distances(positions):
    """
    Precompute distances between all pairs of positions.
    """
    n = len(positions)
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            dists[i, j] = np.linalg.norm(positions[i] - positions[j])
    return dists

def objective_function(radii, positions):
    """
    Objective function: maximize the sum of radii.
    """
    return -np.sum(radii)  # Minimize negative sum

def constraint_no_overlap(radii, positions, dists):
    """
    Constraint: non-overlapping circles.
    """
    return np.min(dists - (radii[:, np.newaxis] + radii[np.newaxis, :]))  # >= 0

def constraint_boundary(radii, positions):
    """
    Constraint: each circle must be within the square.
    """
    x, y = positions.T
    return np.min(x - radii) >= -1e-12 and np.min(y - radii) >= -1e-12 and np.max(x + radii) <= 1 + 1e-12 and np.max(y + radii) <= 1 + 1e-12

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
