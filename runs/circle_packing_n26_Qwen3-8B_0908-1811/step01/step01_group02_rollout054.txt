import numpy as np
import scipy.optimize as opt
from typing import Tuple
from itertools import combinations

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial placement using hexagonal packing
    def initial_hexagonal_packing(num_circles: int) -> np.ndarray:
        # Estimate the number of rows and columns for hexagonal packing
        rows = int(np.sqrt(num_circles)) + 1
        cols = int(np.ceil(num_circles / rows))
        
        # Hexagonal lattice spacing (this is an approximate setup)
        spacing = 0.35  # This can be adjusted
        centers = np.zeros((num_circles, 2))
        index = 0
        
        for row in range(rows):
            for col in range(cols):
                if index >= num_circles:
                    break
                x = col * spacing
                y = row * (np.sqrt(3) / 2) * spacing
                if row % 2 == 1:
                    x += spacing / 2  # offset for odd rows
                centers[index] = np.array([x, y])
                index += 1
        
        # Normalize to fit within the unit square
        centers = centers / np.max(centers)
        return centers

    def objective_function(radii: np.ndarray, centers: np.ndarray) -> float:
        return -np.sum(radii)  # Minimize negative sum of radii

    def constraint_centers(centers: np.ndarray) -> np.ndarray:
        # Ensure centers are inside the square [0, 1] x [0, 1]
        return np.hstack((centers[:, 0], centers[:, 1]))

    def constraint_radii(radii: np.ndarray) -> np.ndarray:
        # Ensure radii are non-negative
        return radii

    def constraint_overlap(centers: np.ndarray, radii: np.ndarray) -> np.ndarray:
        # Calculate pairwise distances and ensure they are at least the sum of radii
        n = centers.shape[0]
        dists = np.zeros(n * (n - 1) // 2)
        for i, j in combinations(range(n), 2):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            dists[i * (n - 1) // 2 + j] = dist - (radii[i] + radii[j])
        return dists

    # Setup initial configuration
    initial_centers = initial_hexagonal_packing(26)
    initial_radii = np.full(26, 0.05)  # Initial small radii

    # Define the optimization problem with constraints
    def objective(x: np.ndarray) -> float:
        return objective_function(x[26:], x[:26])

    def constraint1(x: np.ndarray) -> np.ndarray:
        return constraint_centers(x[:26])

    def constraint2(x: np.ndarray) -> np.ndarray:
        return constraint_radii(x[26:])

    def constraint3(x: np.ndarray) -> np.ndarray:
        return constraint_overlap(x[:26], x[26:])

    bounds = [(0, 1) for _ in range(52)]  # 26 centers, 26 radii
    constraints = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2},
        {'type': 'ineq', 'fun': constraint3}
    ]

    result = opt.minimize(
        fun=objective,
        x0=np.concatenate((initial_centers.flatten(), initial_radii)),
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    # Extract results
    centers = result.x[:52].reshape((26, 2))
    radii = result.x[26:]
    sum_radii = -result.fun

    # Validate the packing
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {message}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
