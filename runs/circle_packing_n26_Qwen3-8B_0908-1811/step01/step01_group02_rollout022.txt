import numpy as np
import scipy.optimize as optimize
import math

def run_packing():
    # Initial grid-based placement: Create a hexagonal grid with 5 rows and 6 columns
    # Adjusted to fit within the unit square, with some spacing between rows and columns
    num_circles = 26
    rows = 5
    cols = 6
    spacing = 0.3
    x_offsets = np.linspace(0.1, 0.9, cols)
    y_offsets = np.linspace(0.1, 0.9, rows)
    
    # Create initial centers
    centers = np.zeros((num_circles, 2))
    idx = 0
    for i in range(rows):
        for j in range(cols):
            x = x_offsets[j]
            y = y_offsets[i]
            centers[idx] = [x, y]
            idx += 1
    
    # Initial radii: All start with a small radius (e.g., 0.1)
    radii = np.full(num_circles, 0.1)
    
    # Optimization bounds
    bounds = []
    for i in range(num_circles):
        bounds.append((0.0, 0.5))  # Radius bounds
        bounds.append((0.0, 1.0))  # X-position bound
        bounds.append((0.0, 1.0))  # Y-position bound
    
    # Objective function: Maximize sum of radii
    def objective(x):
        radii_opt = x[::3]
        centers_opt_x = x[1::3]
        centers_opt_y = x[2::3]
        centers_opt = np.column_stack((centers_opt_x, centers_opt_y))
        return -np.sum(radii_opt)  # Negative for minimization

    # Constraints: Circle must be within the unit square
    def constraint_square(x, idx):
        r = x[3*idx]
        x_pos = x[3*idx + 1]
        y_pos = x[3*idx + 2]
        return x_pos - r >= -1e-12
    def constraint_square_right(x, idx):
        r = x[3*idx]
        x_pos = x[3*idx + 1]
        y_pos = x[3*idx + 2]
        return x_pos + r <= 1 + 1e-12
    def constraint_square_top(x, idx):
        r = x[3*idx]
        x_pos = x[3*idx + 1]
        y_pos = x[3*idx + 2]
        return y_pos - r >= -1e-12
    def constraint_square_bottom(x, idx):
        r = x[3*idx]
        x_pos = x[3*idx + 1]
        y_pos = x[3*idx + 2]
        return y_pos + r <= 1 + 1e-12

    # Constraints for non-overlapping
    def constraint_overlap(x, i, j):
        r_i = x[3*i]
        r_j = x[3*j]
        x_i = x[3*i + 1]
        y_i = x[3*i + 2]
        x_j = x[3*j + 1]
        y_j = x[3*j + 2]
        dx = x_i - x_j
        dy = y_i - y_j
        dist = math.sqrt(dx*dx + dy*dy)
        return dist - r_i - r_j + 1e-12

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
