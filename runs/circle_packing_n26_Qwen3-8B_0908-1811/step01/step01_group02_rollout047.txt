import numpy as np
import scipy.optimize as optimize
import scipy.spatial.distance as dist
from typing import Tuple

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            if dist_sq < (radii[i] + radii[j])**2 - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective_function(params, n_circles):
    centers = params[:2 * n_circles].reshape((n_circles, 2))
    radii = params[2 * n_circles:]
    return -np.sum(radii)

def constraints_function(params, n_circles):
    centers = params[:2 * n_circles].reshape((n_circles, 2))
    radii = params[2 * n_circles:]
    
    constraints = []
    
    # Boundary constraints
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        constraints.append((x - r + 1e-12, 0.0))
        constraints.append((1.0 - x - r + 1e-12, 0.0))
        constraints.append((y - r + 1e-12, 0.0))
        constraints.append((1.0 - y - r + 1e-12, 0.0))
    
    # Non-overlap constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            dist_min = radii[i] + radii[j]
            constraints.append((dist_sq - dist_min**2 + 1e-12, 0.0))
    
    return constraints

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    initial_centers = np.random.rand(n_circles, 2)
    initial_radii = np.random.rand(n_circles) * 0.1  # Small initial radii

    params = np.concatenate((initial_centers.flatten(), initial_radii))
    
    # Objective is to maximize the sum of radii, so minimize the negative sum
    result = optimize.minimize(
        fun=objective_function,
        x0=params,
        args=(n_circles,),
        method='SLSQP',
        constraints=[{'type': 'ineq', 'fun': lambda x: constraint} for constraint in constraints_function(params, n_circles)],
        bounds=[(0.0, 1.0) for _ in range(2 * n_circles)] + [(0.0, 1.0) for _ in range(n_circles)]
    )
    
    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")
    
    centers = result.x[:2 * n_circles].reshape((n_circles, 2))
    radii = result.x[2 * n_circles:]
    sum_radii = np.sum(radii)
    
    validation_result, message = validate_packing(centers, radii)
    if not validation_result:
        raise RuntimeError(f"Packing invalid: {message}")
    
    return centers, radii, sum_radii
