import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Total number of circles
    n_circles = 26

    # Initial hexagonal grid layout
    rows = 6
    cols = 5  # 5 columns can fit in a square with some spacing

    # Grid spacing for hexagonal packing
    cell_width = 1.0 / (cols + 1)
    cell_height = np.sqrt(3) / (rows + 1)

    # Shift for staggered rows
    shift = cell_width / 2

    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)

    # Initialize circles in a hexagonal grid
    for i in range(rows):
        for j in range(cols):
            col = j * cell_width + cell_width / 2
            row = i * cell_height + cell_height / 2
            if i % 2 == 1:
                col += shift
            centers[i * cols + j] = [col, row]
            radii[i * cols + j] = 0.05  # Initial radius (will be optimized)

    # Objective function to maximize total radii (minimize negative sum)
    def objective(radii_vec):
        return -np.sum(radii_vec)

    # Constraints: non-overlapping, within square
    def constraints(radii_vec):
        centers_ = centers
        for i in range(n_circles):
            x, y = centers_[i]
            if x - radii_vec[i] < -1e-12 or x + radii_vec[i] > 1 + 1e-12 or \
               y - radii_vec[i] < -1e-12 or y + radii_vec[i] > 1 + 1e-12:
                return np.inf  # Violates boundary constraints
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers_[i][0] - centers_[j][0]
                dy = centers_[i][1] - centers_[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_vec[i] + radii_vec[j] - 1e-12:
                    return np.inf  # Violates non-overlap constraint
        return 0  # All constraints satisfied

    # Add constraints to the optimization
    cons = {'type': 'ineq', 'fun': constraints}

    # Initial guess: radii
    radii_initial = radii.copy()

    # Run the optimization
    result = opt.minimize(objective, radii_initial, constraints=cons, method='SLSQP')

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
