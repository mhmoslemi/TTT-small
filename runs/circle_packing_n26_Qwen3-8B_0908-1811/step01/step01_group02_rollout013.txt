import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal packing with 26 circles
    num_circles = 26
    radius_initial = 0.05  # Initial guess for small radii

    # Generate initial hexagonal grid
    def generate_hexagonal_grid(num_circles):
        def hexagonal_positions(n):
            # Generate hexagonal grid of size n x n with spacing 1
            if n == 0:
                return np.array([])
            grid = []
            for i in range(n):
                for j in range(n):
                    x = i + 0.5 * j
                    y = j * np.sqrt(3) / 2
                    grid.append([x, y])
            return np.array(grid)

        # Find the smallest n where hexagonal grid has >= num_circles
        n = 1
        while True:
            grid = hexagonal_positions(n)
            if len(grid) >= num_circles:
                break
            n += 1

        # Trim the grid to the first num_circles
        grid = grid[:num_circles]
        return grid

    initial_centers = generate_hexagonal_grid(num_circles)

    # Initial radii (same for all circles for simplicity)
    initial_radii = np.full(num_circles, radius_initial)

    # Define the objective function (maximize the sum of radii)
    def objective(x):
        # x has shape (2 * num_circles, )
        centers = x[:2 * num_circles].reshape((num_circles, 2))
        radii = x[2 * num_circles:]
        sum_radii = np.sum(radii)
        return -sum_radii  # Minimize the negative of the sum

    # Define the constraints
    def constraint_boundary(x):
        centers = x[:2 * num_circles].reshape((num_circles, 2))
        radii = x[2 * num_circles:]
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:

        return 0

    def constraint_overlap(x):

        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-12:

        return 0

        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
    for _ in range(num_circles):
        bounds.append((0.0, 1.0))  # radius

    result = optimize.minimize(

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
