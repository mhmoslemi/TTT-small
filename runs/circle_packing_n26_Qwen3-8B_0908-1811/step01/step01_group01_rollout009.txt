def validate_packing(centers, radii):
    """
    Validates that the packing is feasible (non-overlapping, within bounds).

    Args:
        centers: np.ndarray of shape (n, 2), containing circle centers.
        radii: np.ndarray of shape (n,), containing circle radii.

    Returns:
        (bool, str): True if valid, False if invalid with error message.
    """
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
