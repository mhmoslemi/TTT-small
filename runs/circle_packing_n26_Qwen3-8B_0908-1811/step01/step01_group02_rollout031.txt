import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Start with a grid-based packing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Define a hybrid layout with a base grid and an extra circle
    # Start with 5x5 grid (25 circles) and place the 26th circle strategically
    grid_rows, grid_cols = 5, 5
    grid_gap = 0.25
    for i in range(grid_rows):
        for j in range(grid_cols):
            centers[i * grid_cols + j] = [j * grid_gap, i * grid_gap]
            radii[i * grid_cols + j] = 0.12

    # Place the extra circle in the center to potentially increase the total sum
    centers[-1] = [0.5, 0.5]
    radii[-1] = 0.12

    # Define the objective function: maximize total radius
    def objective(radii):
        return -np.sum(radii)

    # Define the constraints
    def bounds(radii):
        return [(0.0, 1.0) for _ in range(num_circles)]

    def inequality_constraints(radii):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]) >= -1e-12)
        return constraints

    # Add constraints that circles must be within the unit square
    def square_constraints(radii):
        constraints = []
        for i in range(num_circles):
            x, y = centers[i]
            constraints.append(x - radii[i] >= -1e-12)
            constraints.append(x + radii[i] <= 1.0 + 1e-12)
            constraints.append(y - radii[i] >= -1e-12)
            constraints.append(y + radii[i] <= 1.0 + 1e-12)
        return constraints

    # Combine constraints
    all_constraints = inequality_constraints(radii) + square_constraints(radii)

    # Add bounds
    bounds = Bounds([0.0] * num_circles, [1.0] * num_circles)

    # Initialize the solver
    result = minimize(
        objective,
        radii,
        method="SLSQP",
        bounds=bounds,
        constraints=[{"type": "ineq", "fun": lambda r: c} for c in all_constraints]

    # Extract the optimized results

        raise ValueError(f"Invalid packing: {msg}")

    return final_centers, optimized_radii, final_sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
