import numpy as np
import scipy.optimize as opt
import math
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Run a circle packing algorithm to pack 26 circles in a unit square [0,1]×[0,1]
    to maximize the sum of radii. Returns (centers, radii, sum_radii).
    """

    # Initial guess: place 26 circles in a hexagonal grid near the center
    # with some spacing to prevent overlap
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    radius_guess = 0.05

    # Initial placement: hexagonal grid with some spacing
    rows = 5
    cols = 6
    spacing = 0.25  # initial spacing between centers
    for i in range(rows):
        for j in range(cols):
            x = 0.5 + (j - cols / 2) * spacing
            y = 0.5 + (i - rows / 2) * spacing * np.sqrt(3) / 2
            idx = i * cols + j
            centers[idx] = [x, y]
            radii[idx] = radius_guess

    # Optimize using Scipy's constrained optimization
    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(x, radii):
        # x: flattened array of centers
        # radii: array of radii
        centers = x.reshape((num_circles, 2))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return 1.0  # violation
        return 0.0  # no violation

    def constraint_boundary(x, radii):
        centers = x.reshape((num_circles, 2))
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return 1.0  # violation
        return 0.0  # no violation

    # Initial guess for centers and radii
    x0 = centers.flatten()
    r0 = radii

    # Set up constraints
    cons = [
        {'type': 'ineq', 'fun': lambda x: -constraint_overlap(x, r0)},
        {'type': 'ineq', 'fun': lambda x: -constraint_boundary(x, r0)}
    ]

    # Bounds for radii (non-negative)
    bounds = [(0, None) for _ in range(num_circles)]

    # Optimization
    result = opt.minimize(
        objective, x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8
    )

    # Extract results
    centers = result.x.reshape((num_circles, 2))
    radii = r0  # radii not optimized in this setup (can be extended)

    # Re-check for any violations to ensure consistency
    is_valid, message = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(message)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
