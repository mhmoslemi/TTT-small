import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def run_packing():
    # Initial grid-based hexagonal packing
    def initial_centers(n_circles):
        spacing = 1.0 / np.sqrt(3)
        rows = int(np.sqrt(n_circles / 2))
        cols = int(np.sqrt(2 * n_circles))
        centers = np.zeros((n_circles, 2))
        idx = 0
        for row in range(rows):
            for col in range(cols):
                if row % 2 == 0:
                    centers[idx, 0] = col * spacing
                    centers[idx, 1] = row * spacing * np.sqrt(3) / 2
                else:
                    centers[idx, 0] = (col + 0.5) * spacing
                    centers[idx, 1] = row * spacing * np.sqrt(3) / 2
                idx += 1
        # Normalize to fit in [0, 1] x [0, 1]
        centers = centers / np.max(centers)
        return centers

    n_circles = 26
    centers = initial_centers(n_circles)
    radii = np.ones(n_circles) * 0.1  # Initial radius guess

    def objective(radii, centers):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_center_bounds(radii, centers):
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return np.inf
        return 0

    def constraint_overlap(radii, centers):
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        return 0

    def constraint_non_negative(radii):
        for r in radii:
            if r < 0:
                return np.inf
        return 0

    # Create optimization problem
    bounds = Bounds(0, 1.0)  # Radii must be between 0 and 1.0
    cons = [
        LinearConstraint(np.ones(n_circles), 0, 1),  # Radii must be non-negative
        NonlinearConstraint(constraint_center_bounds, 0, 0),
        NonlinearConstraint(constraint_overlap, 0, 0),
        NonlinearConstraint(constraint_non_negative, 0, 0)
    ]

    # Optimize
    result = minimize(
        fun=objective,
        x0=radii,
        args=(centers,),
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        tol=1e-6,
        options={'maxiter': 500}
    )

    # Extract solution
    optimized_radii = result.x
    optimized_centers = centers.copy()

    # Validate and return
    is_valid, message = validate_packing(optimized_centers, optimized_radii)
    if not is_valid:
        raise ValueError(f"Validation failed: {message}")

    return optimized_centers, optimized_radii, np.sum(optimized_radii)

# Helper function to validate packing
def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"
