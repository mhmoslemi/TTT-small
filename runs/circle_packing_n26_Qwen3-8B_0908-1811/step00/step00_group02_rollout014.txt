import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Number of circles
    n = 26

    # Hexagonal packing setup
    # Calculate number of rows and columns
    rows = int(np.ceil(np.sqrt(n)))  # Number of rows
    cols = int(np.ceil(n / rows))    # Number of columns

    # Initial radius based on hexagonal grid spacing
    # Use a horizontal spacing of 2r and vertical spacing of sqrt(3)r
    r0 = 1.0 / (2 * (cols - 1))  # Adjust to fit in unit square

    # Generate initial center coordinates
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * 2 * r0
            else:
                x = (j * 2 + 1) * r0
            y = i * np.sqrt(3) * r0
            if x > 1 - 1e-6 or y > 1 - 1e-6:
                continue  # Skip if outside the unit square
            centers[idx] = [x, y]
            idx += 1
            if idx == n:
                break
        if idx == n:
            break

    # Initial estimate for radii
    radii = np.full(n, r0)

    # Objective function to maximize the sum of radii
    def objective(x, centers, radii):
        return -np.sum(x)  # Minimize negative sum to maximize

    # Constraints:
    # - Circles must be within the unit square
    # - Circles must not overlap
    # - Radii must be non-negative

    # Bounds on radii
    radii_bounds = Bounds(0, 1.0)

    # Optimization variables: radii
    x0 = radii.copy()

    # Constraints for the optimization
    def constraint1(x, centers):
        # Ensure all circles are within the unit square
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = x[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return False
        return True

    def constraint2(x, centers):
        # Ensure no two circles overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < x[i] + x[j] - 1e-6:
                    return False
        return True

    # Run the optimization
    res = opt.minimize(
        lambda x: objective(x, centers, radii),
        x0,
        bounds=radii_bounds,
        constraints=[
            {"type": "ineq", "fun": constraint1, "args": (centers,)},
            {"type": "ineq", "fun": constraint2, "args": (centers,)},
        ],
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 1000}
    )

    # Extract results
    optimized_radii = res.x
    sum_radii = np.sum(optimized_radii)

    # Validate the packing
    valid, message = validate_packing(centers, optimized_radii)
    if not valid:
        raise ValueError(f"Packing validation failed: {message}")

    return centers, optimized_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
