import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    n_circles = 26

    ])
    
    centers = cp.Variable((n_circles, 2))
    radii = cp.Variable(n_circles, nonneg=True)

    # Objective: Maximize the total sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints for boundaries
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        constraints = [
            x - r >= 0,
            x + r <= 1,
            y - r >= 0,
            y + r <= 1
        ]
        for c in constraints:
            objective += c

    # Constraints for non-overlap
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = cp.norm(centers[i] - centers[j])
            constraints.append(dist >= radii[i] + radii[j])

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve()

    # Convert to numpy arrays
    centers_np = centers.value
    radii_np = radii.value
    sum_radii = np.sum(radii_np)

    return (centers_np, radii_np, sum_radii)
