    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Generate hexagonal grid with some padding
    rows = 5
    cols = 6
    spacing = 1.0 / (cols + 1)
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + spacing / 2
            y = i * (np.sqrt(3) * spacing) + spacing / 2
            centers[i * cols + j] = [x, y]

    # Initial radii estimate (close to the minimum possible)

    radii[-1] = 0.05  # Adjust one circle for better spacing

    # Objective function: maximize sum of radii

        radii = radii_vec
        return -np.sum(radii)  # Negative for minimization

    # Equality constraints: none; all constraints are inequality
    constraints = []

    def non_overlap_constraint(radii_vec):
        centers_current = centers
        radii_current = radii_vec
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_current[i, 0] - centers_current[j, 0]
                dy = centers_current[i, 1] - centers_current[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_current[i] + radii_current[j] - 1e-12:
                    return dist - (radii_current[i] + radii_current[j])
        return 0.0

    constraints.append({'type': 'ineq', 'fun': non_overlap_constraint})

    def boundary_constraint(radii_vec):
        centers_current = centers
        radii_current = radii_vec
        for i in range(num_circles):
            x, y = centers_current[i]
            r = radii_current[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1.0
        return 0.0

    constraints.append({'type': 'ineq', 'fun': boundary_constraint})

    # Bounds: radii must be non-negative
    bounds = [(0.0, 1.0) for _ in range(num_circles)]

    # Run optimization
    result = optimize.minimize(

        x0=radii,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
