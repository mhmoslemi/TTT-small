import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():

    n_circles = 26

    def distance(p1, p2):
        return np.linalg.norm(p1 - p2)

    def is_valid(centers, radii):
        for i in range(n_circles):
            x, y = centers[i]
