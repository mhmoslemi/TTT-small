import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import constraints

def run_packing():
    # Define the total number of circles
    num_circles = 26

    # Calculate initial hexagonal grid layout inside unit square
    def hexagonal_grid(num_circles):
        # Calculate number of rows (approximate hexagonal layer)
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols_per_row = np.ceil(num_circles / rows)

        # Generate hexagonal grid positions
        centers = np.zeros((num_circles, 2))

        for i in range(rows):
            for j in range(cols_per_row):
                index = i * cols_per_row + j
                if index >= num_circles:
                    break
                # Hexagonal grid positions
                dx = 1.0 / (cols_per_row - 1) if cols_per_row > 1 else 0
                dy = np.sqrt(3) / 2 / (rows - 1) if rows > 1 else 0
                x = j * dx
                y = i * dy
                centers[index] = [x, y]

        return centers

    # Initial grid positions
    initial_centers = hexagonal_grid(num_circles)

    # Initial radii: Make small initial radii to avoid collisions
    initial_radii = np.full(num_circles, 0.05)

    # Objective function to minimize (negative sum of radii)
    def objective(x):
        return -np.sum(x[26:])

    # Flatten centers and radii for optimization
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])

    # Bounds: centers are within [0, 1], radii are non-negative
    bounds = Bounds([0, 0] * 2 * num_circles, [1, 1] * 2 * num_circles + [np.inf] * num_circles)

    # Constraints: no two circles overlap and within square
    constraints = []

    def constraint_overlap(x):
        centers = x[:2 * num_circles].reshape(num_circles, 2)
        radii = x[2 * num_circles:]
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return dist - (radii[i] + radii[j] - 1e-12)
        return 0  # No overlap violations

    constraints.append({'type': 'ineq', 'fun': constraint_overlap})

    # Constraint: circle within square
    def constraint_boundary(x):
        centers = x[:2 * num_circles].reshape(num_circles, 2)
        radii = x[2 * num_circles:]
        for i in range(num_circles):
            x_center, y_center = centers[i]
            r = radii[i]
            if x_center - r < 0 or x_center + r > 1 or y_center - r < 0 or y_center + r > 1:
                return 0
        return 1

    constraints.append({'type': 'ineq', 'fun': constraint_boundary})

    # Optimization
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-6
    )

    if not result.success:
        raise ValueError("Optimization failed.")

    # Extract the final centers and radii
    final_centers = result.x[:2 * num_circles].reshape(num_circles, 2)
    final_radii = result.x[2 * num_circles:]

    # Assert valid packing
    valid, message = validate_packing(final_centers, final_radii)
    if not valid:
        raise ValueError(f"Packing failed due to {message}")

    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
