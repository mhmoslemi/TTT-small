import numpy as np
import scipy.optimize as opt
from typing import Tuple

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: grid-based packing
    n_circles = 26
    rows = int(np.sqrt(n_circles))
    cols = (n_circles + rows - 1) // rows

    radii = np.zeros(n_circles)
    spacing = 1.0 / (cols + 1)
    for i in range(n_circles):
        row = i // cols
        col = i % cols

        centers[i, 1] = (row + 0.5) * spacing
        radii[i] = 0.05  # Initial small radius

    # Optimization function to maximize sum of radii
    def objective(x):
        centers_flat = x[:2*n_circles].reshape((n_circles, 2))
        radii_flat = x[2*n_circles:]
        return -np.sum(radii_flat)  # Negative for minimization

    # Constraints: boundary and non-overlapping
    def constraint_boundary(x):
        centers_flat = x[:2*n_circles].reshape((n_circles, 2))
        radii_flat = x[2*n_circles:]
        for i in range(n_circles):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if (x_i - r_i < 0 - 1e-12 or x_i + r_i > 1 + 1e-12
                or y_i - r_i < 0 - 1e-12 or y_i + r_i > 1 + 1e-12):
                return 1.0
        return 0.0

    def constraint_overlap(x):
        centers_flat = x[:2*n_circles].reshape((n_circles, 2))

                    return 1.0
        return 0.0

    # Optimization
    bounds = []
    for _ in range(n_circles):
        bounds.append((0.0, 1.0))  # x-coordinate
        bounds.append((0.0, 1.0))  # y-coordinate
    for _ in range(n_circles):
        bounds.append((0.0, 0.5))  # radius (max radius is 0.5)

    # Constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},

    ]
