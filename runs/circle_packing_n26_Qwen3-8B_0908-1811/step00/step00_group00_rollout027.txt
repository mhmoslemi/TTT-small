import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Initial guess: grid-based layout
    n = 26
    grid_points = np.linspace(0, 1, 5)
    centers = np.array([np.meshgrid(grid_points, grid_points)][0].flatten(), dtype=float)
    centers = centers.reshape(-1, 2)
    radii = np.full(n, 0.15)  # initial radius

    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii by minimizing -sum(radii)

    def constraint_overlap(i, j):
        def constraint(x):
            # x is a flattened array of centers and radii
            centers_flat = x[:2*n].reshape(n, 2)
            radii_flat = x[2*n:]
            r_i = radii_flat[i]
            r_j = radii_flat[j]
            dx = centers_flat[i, 0] - centers_flat[j, 0]
            dy = centers_flat[i, 1] - centers_flat[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            return dist - (r_i + r_j)
        return constraint

    def constraint_boundary(i):
        def constraint(x):
            centers_flat = x[:2*n].reshape(n, 2)
            radii_flat = x[2*n:]
            x_c, y_c = centers_flat[i]
            r = radii_flat[i]
            return np.array([x_c - r, x_c + r, y_c - r, y_c + r]) - np.array([0, 1, 0, 1])
        return constraint

    # Define all constraints
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})

    for i in range(n):
        for j in range(i+1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j)})

    # Set bounds for centers (0 to 1) and radii (0 to 1)
    bounds = Bounds([0, 0] * n, [1, 1] * n)

    # Optimize
    x0 = np.concatenate([centers.flatten(), radii])
    result = opt.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8
    )

    # Extract results
    centers_opt = result.x[:2*n].reshape(n, 2)
    radii_opt = result.x[2*n:]
    sum_radii = np.sum(radii_opt)
    
    # Validate the packing
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
