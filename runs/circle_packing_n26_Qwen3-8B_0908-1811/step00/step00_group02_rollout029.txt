import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial guess: hexagonal packing (shifted grid)
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Create initial hexagonal grid
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            if idx >= n:
                break
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers[idx] = [x, y]
            # Set initial radius as distance to the nearest boundary or other circle
            min_dist = np.inf
            for k in range(n):
                if k == idx:
                    continue
                dx = centers[k][0] - centers[idx][0]
                dy = centers[k][1] - centers[idx][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < min_dist:
                    min_dist = dist
            min_dist = min(min_dist, 1 - centers[idx][0], centers[idx][0], 1 - centers[idx][1], centers[idx][1])
            radii[idx] = min_dist * 0.9  # reduce slightly to avoid overlap

    # Define constraints for optimization
    def objective(radii):
        return -np.sum(radii)  # maximize sum of radii

    def constraint_center_in_boundary(radii):
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8:
                return -1
            if y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return -1
        return 0

    def constraint_no_overlap(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Set bounds for radii (non-negative)
    bounds = [(0, 1) for _ in range(n)]

    # Optimization
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        method='L-BFGS-B',
        constraints=[
            {'type': 'ineq', 'fun': constraint_center_in_boundary},
            {'type': 'ineq', 'fun': constraint_no_overlap}
        ],
        options={'ftol': 1e-12, 'maxiter': 1000}
    )

    # Final centers and radii
    centers = np.array(result.x)
    radii = np.array(result.x)

    # Final validation
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(message)

    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
