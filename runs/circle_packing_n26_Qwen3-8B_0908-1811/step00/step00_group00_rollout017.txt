from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    n = 26

    # Initial guess for circle centers (randomly placed within [0,1]^2)
    centers = np.random.rand(n, 2)
    # Initialize radii to small positive values
    radii = np.full(n, 0.02)

    # Function to compute the negative of the sum of radii (to maximize)
    def objective(radii_flat):
        return -np.sum(radii_flat)

    # Function to compute constraint penalties
    def constraint_penalty(radii_flat, centers):
        # Reshape radii_flat into a 1D array
        radii = radii_flat.reshape(-1, 1)
        # Ensure centers are within [0,1]^2 and radii are non-negative
        penalty = 0
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                penalty += 1e6 * (max(0, x - r - 1e-12) + max(0, x + r - 1 - 1e-12) + max(0, y - r - 1e-12) + max(0, y + r - 1 - 1e-12))
            if r < 0:
                penalty += 1e6 * (-r)
        # Ensure no two circles overlap
        dist_matrix = cdist(centers, centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                    penalty += 1e6 * (radii[i] + radii[j] - dist_matrix[i, j])
        return penalty

    def total_objective(radii_flat):
        obj = objective(radii_flat)
        penalty = constraint_penalty(radii_flat, centers)
        return obj + penalty

    bounds = [(0, 1.0) for _ in range(n)]

    # Run the optimization
    result = minimize(
        total_objective,
        radii.flatten(),
        method='L-BFGS-B',
        bounds=bounds,
        jac=None,
        options={'maxiter': 100, 'disp': False}
    )

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
