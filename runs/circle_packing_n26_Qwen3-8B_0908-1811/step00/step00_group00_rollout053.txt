import numpy as np
import scipy.optimize as optimize
import cvxpy as cp

def run_packing():
    # Initial hexagonal grid packing
    n = 26
    num_rows = int(np.sqrt(n)) + 1
    num_cols = int(np.ceil(n / num_rows))

    # Calculate the initial circle radius
    # The hexagonal grid inner spacing
    dx = 1.0 / (num_cols - 1)
    dy = np.sqrt(3) / (num_rows - 1)

    # Initialize centers and radii
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    radius = 1.0 / (2 * (num_cols + num_rows))

    idx = 0
    for i in range(num_rows):
        for j in range(num_cols):
            if idx >= n:
                break
            x = j * dx
            y = i * dy
            centers[idx] = [x, y]
            radii[idx] = radius
            idx += 1

    # Optimization objective: maximize sum of radii
    # Constraints: non-overlap and boundary constraints

    # Reformulate the problem for optimization
    # Define variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n, non_negative=True)

    # Objective function
    objective = cp.Maximize(cp.sum(r))

    # Create constraints
    constraints = []

    # Boundary constraint
    for i in range(n):
        x_i, y_i = x[i]
        r_i = r[i]
        constraints.append(x_i >= r_i)
        constraints.append(x_i <= 1 - r_i)
        constraints.append(y_i >= r_i)
        constraints.append(y_i <= 1 - r_i)

    # Non-overlap constraint

            dx_ij = x[i][0] - x[j][0]

            constraints.append(dist >= r[i] + r[j])

    # Define the problem
    prob = cp.Problem(objective, constraints)

    try:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
