import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial random positions and radii
    np.random.seed(42)  # For reproducibility
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.random.rand(n) * 0.1  # Initial small radii

    # Function to check the validity of the packing
    def validate_packing(centers, radii):
        n = centers.shape[0]
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    # Optimization variables
    # Flatten centers and radii for optimization
    x = np.concatenate((centers.flatten(), radii))

    # Objective function to maximize sum of radii
    def objective(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        return -np.sum(radii_flat)  # Minimize negative sum (maximize sum)

    # Constraint functions
    def constraint_overlap(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return dist - (radii_flat[i] + radii_flat[j] - 1e-12)
        return 0.0

    def constraint_boundary(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        for i in range(n):
            x_center, y_center = centers_flat[i]
            r = radii_flat[i]
            if x_center - r < -1e-12 or x_center + r > 1 + 1e-12:
                return x_center - r + 1e-12
            if y_center - r < -1e-12 or y_center + r > 1 + 1e-12:
                return y_center - r + 1e-12
        return 0.0

    # Bounds for positions (must be within [0,1])
    bounds = [(0.0, 1.0) for _ in range(n*2)] + [(0.0, 1.0) for _ in range(n)]

    # Constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    # Perform optimization
    result = opt.minimize(objective, x, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)

    # Extract results
    optimal_x = result.x
    centers_opt = optimal_x[:n*2].reshape(n, 2)
    radii_opt = optimal_x[n*2:]
    sum_radii = np.sum(radii_opt)

    # Validate the packing
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    return centers_opt, radii_opt, sum_radii
