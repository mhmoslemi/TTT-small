import numpy as np
import scipy.optimize as optimize
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial estimate of centers using a hexagonal grid
    def generate_initial_centers(n_circles: int = 26) -> np.ndarray:
        # Approximate hexagonal grid in a square
        rows = int(np.sqrt(n_circles))
        cols = (n_circles + rows - 1) // rows
        centers = np.zeros((n_circles, 2))
        for i in range(n_circles):
            row = i // cols
            col = i % cols
            x = col / (cols - 1) if cols > 1 else 0.5
            y = row / (rows - 1) if rows > 1 else 0.5
            centers[i] = [x, y]
        return centers

    # Objective function to maximize the sum of radii
    def objective(radii: np.ndarray) -> float:
        return -np.sum(radii)  # Minimize negative of sum to maximize

    # Constraints for optimization
    def constraints(centers: np.ndarray, radii: np.ndarray) -> np.ndarray:
        # Boundary constraint
        boundary_violations = np.zeros(n_circles)
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                boundary_violations[i] = 1e6  # Penalize violation

        # Non-overlap constraint
        overlap_violations = np.zeros(n_circles)
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-12:
                    overlap_violations[i] += 1e6
                    overlap_violations[j] += 1e6

        total_violation = np.sum(boundary_violations) + np.sum(overlap_violations)
        return total_violation

    # Helper function to evaluate constraints
    def constraint_func(x: np.ndarray) -> np.ndarray:
        # x is a flattened array of [centers, radii]
        centers = x[:n_circles * 2].reshape(n_circles, 2)
        radii = x[n_circles * 2:].reshape(n_circles)
        return constraints(centers, radii)

    # Main optimization
    n_circles = 26
    centers = generate_initial_centers(n_circles)
    radii = np.ones(n_circles) * 0.05  # Initial radii

    # Flatten the variables for optimization
    x0 = np.concatenate([centers.flatten(), radii])

    # Bounds: centers in [0,1] and radii non-negative
    bounds = []
    for i in range(n_circles):
        bounds.append((0.0, 1.0))  # x-coordinate
        bounds.append((0.0, 1.0))  # y-coordinate
        bounds.append((0.0, 1.0))  # radius

    # Optimization
    result = optimize.minimize(

    sum_radii = np.sum(radii_opt)

    return centers_opt, radii_opt, sum_radii
