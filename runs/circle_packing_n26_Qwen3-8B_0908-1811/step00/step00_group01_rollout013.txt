import cvxpy as cp
import itertools

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26

    radii = np.full(n, 0.05)

    # Use a hexagonal grid to start with
    # We will place circles in a hexagonal pattern

    # Grid parameters
    grid_cols = 6  # number of columns
    grid_rows = 5  # number of rows

    centers = np.zeros((n, 2))
    idx = 0

    for row in range(grid_rows):
        for col in range(grid_cols):
            x = col / (grid_cols - 1)

            # Adjust for hexagonal packing
            if row % 2 == 1:
                x += 0.5 / (grid_cols - 1)
            centers[idx] = [x, y]
            idx += 1

    # Define variables for optimization
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n, nonneg=True)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []
    for i in range(n):
        # Ensure the circle is inside the unit square
        constraints.append(centers[i, 0] >= radii[i])
        constraints.append(centers[i, 0] <= 1 - radii[i])
        constraints.append(centers[i, 1] >= radii[i])
        constraints.append(centers[i, 1] <= 1 - radii[i])

            dist = cp.norm(centers[i] - centers[j])

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-6)

    if prob.status != cp.OPTIMAL:
