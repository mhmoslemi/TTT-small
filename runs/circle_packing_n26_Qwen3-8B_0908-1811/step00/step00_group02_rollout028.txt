from scipy.optimize import minimize
from scipy.optimize import Bounds

def compute_hex_grid(num_circles, square_size=1.0):
    """Generate a hexagonal grid of circles within a square."""
    # Calculate number of rows and columns for hexagonal packing
    num_rows = int(np.ceil(np.sqrt(num_circles)))
    num_cols = int(np.ceil(num_circles / num_rows))
    
    # Calculate the horizontal and vertical spacing between centers
    h_spacing = square_size / (num_cols - 1) if num_cols > 1 else square_size
    v_spacing = (np.sqrt(3) / 2) * h_spacing
    
    # Generate initial circle centers
    centers = []
    for i in range(num_rows):
        for j in range(num_cols):
            x = j * h_spacing
            y = i * v_spacing
            if i % 2 == 1:
                x += h_spacing / 2
            centers.append((x, y))
    
    # Convert to numpy array and scale to [0,1]^2
    centers = np.array(centers) / square_size
    return centers

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
