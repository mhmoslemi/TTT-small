import numpy as np
import scipy.optimize as optimize

# Function to compute centers in a hexagonal grid
def hexagonal_grid(num_circles, square_size=1):
    # Compute the number of rows
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    # Calculate spacing
    dx = square_size / (cols - 1)
    dy = np.sqrt(3) / 2 * square_size / (rows - 1)
    
    # Generate grid points
    centers = np.zeros((num_circles, 2))
    idx = 0
    for i in range(rows):
        for j in range(cols):
            x = j * dx
            y = i * dy
            # Offset even rows
            if i % 2 == 1:
                x += dx / 2
            centers[idx] = [x, y]
            idx += 1
    return centers

# Objective function for optimization: maximize sum of radii
def objective_function(x, num_circles, square_size=1, centers=None):

    centers_new = x[num_circles:]
    # Compute the maximum radius such that all circles fit inside the square
    max_r = np.min([np.min(centers_new[:, 0]), np.min(centers_new[:, 1]),
                   square_size - np.max(centers_new[:, 0]),

def validate_packing(centers, radii):
