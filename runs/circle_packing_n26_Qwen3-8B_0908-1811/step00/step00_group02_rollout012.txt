from scipy.optimize import Bounds, LinearConstraint, NonlinearConstraint

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    num_circles = 26

    # Using a hexagonal grid for initial placement
    rows = int(np.sqrt(num_circles))
    cols = num_circles // rows
    remainder = num_circles % rows

        centers[i, 1] = y

    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers /= np.max([max_x, max_y])

    # Initial guess for radii
    radii = np.ones(num_circles) * 0.05

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize the negative sum

    # Define the constraints
    def constraint_overlap(radii, centers):

        dist = np.zeros(n * (n - 1) // 2)
        for i in range(n):

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist[(i * (i - 1) // 2) + (j - i - 1)] = np.sqrt(dx*dx + dy*dy)
        return dist - (radii[:n] + radii[n:]) + 1e-12

        y = centers[:, 1] - radii[:n]
        x_neg = x < -1e-12
        y_neg = y < -1e-12
        x_pos = x > 1 + 1e-12
        y_pos = y > 1 + 1e-12
        return np.concatenate([x_neg, y_neg, x_pos, y_pos])

    # Define the bounds for radii (non-negative)
    bounds = Bounds(np.zeros(num_circles), np.inf)

    # Convert the constraints to scipy format
    nonlin_con = NonlinearConstraint(

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
