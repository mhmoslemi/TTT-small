import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: place circles in a hexagonal grid inside the unit square
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initialize in a hexagonal grid with some spacing
    rows = int(np.sqrt(n))
    cols = int(np.ceil(n / rows))
    
    spacing = 0.4  # Initial spacing to allow some room for expansion
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 1:
                x = (j + 0.5) * spacing
            else:
                x = j * spacing
            y = i * spacing * np.sqrt(3) / 2
            if i * cols + j < n:
                centers[i * cols + j] = np.array([x, y])
                radii[i * cols + j] = 0.1  # Initial guess for radius

    # Bound for radius (non-negative)
    bounds = Bounds(0, 10)

    # Objective function: maximize sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Negative for minimization

    # Constraints: circles must not overlap and must be inside the unit square

    # Constraint 1: Each circle must be inside the unit square
    def square_constraint(radii_vec, centers):
        for i in range(n):
            x, y = centers[i]
            r = radii_vec[i]
            if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
                return -1
        return 0

    # Constraint 2: Distance between centers must be at least sum of radii
    def non_overlap_constraint(radii_vec, centers):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_vec[i] + radii_vec[j] - 1e-8:
                    return -1
        return 0

    # Set up constraints
    cons = [
        LinearConstraint(np.ones(n), 0, 1),  # Radii must be in [0, 1]
        NonlinearConstraint(lambda r: square_constraint(r, centers), -np.inf, 0),
        NonlinearConstraint(lambda r: non_overlap_constraint(r, centers), -np.inf, 0)
    ]

    # Initial guess
    initial_guess = radii.copy()

    # Optimize
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons
    )

    # Extract result
    radii_opt = result.x
    centers_opt = centers.copy()
    sum_radii = np.sum(radii_opt)

    # Validate the result
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(msg)

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
