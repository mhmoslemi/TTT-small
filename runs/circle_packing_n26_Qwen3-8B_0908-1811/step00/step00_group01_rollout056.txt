import scipy.optimize as opt
from scipy.optimize import Bounds
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess for positions (rough hexagonal grid)
    n = 26
    rows = 5
    cols = 6
    spacing = 1.0 / (cols - 1)  # Spacing between columns
    spacing_y = 1.0 / (rows - 1)  # Spacing between rows
    initial_positions = np.zeros((n, 2))
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if idx >= n:
                break
            x = col * spacing
            y = row * spacing_y
            initial_positions[idx] = [x, y]
            idx += 1

    # Initial guess for radii (equal for all)
    initial_radii = np.full(n, 0.1)

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative because we minimize

    def constraint_overlap(x, radii):

        radii = radii
        dist_matrix = cdist(centers, centers)

            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1.0  # Infeasible
        return 0.0

    def constraint_boundary(x, radii):
        centers = x[:n, :]
        radii = radii

            x_i, y_i = centers[i]
            r_i = radii[i]

                return -1.0
            if y_i - r_i < -1e-8 or y_i + r_i > 1.0 + 1e-8:
                return -1.0
        return 0.0

        # Ensure the radii are non-negative
        return np.maximum(radii, 0.0)

    # Optimization bounds for positions and radii
    bounds_positions = Bounds(np.zeros((n, 2)), np.ones((n, 2)))
    bounds_radii = Bounds(np.zeros(n), np.inf)

    # Initial guess for positions and radii
    x0 = np.vstack((initial_positions, initial_radii))

    # Optimization options
    options = {

        'maxiter': 500,
        'ftol': 1e-8,
        'eps': 1e-8

    ]

        x0=x0,

        method='SLSQP',

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
