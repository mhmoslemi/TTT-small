```python
import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal grid
    num_circles = 26
    num_rows = 5
    num_cols = 5
    radius_initial = 0.10
    spacing = 2 * radius_initial

    # Initial grid
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles) + radius_initial

    for i in range(num_rows):
        for j in range(num_cols):
            row = i
            col = j
            if row % 2 == 1:
                col += 0.5
            x = col * spacing
            y = row * spacing * np.sqrt(3) / 2
            centers[i * num_cols + j] = [x, y]
            radii[i * num_cols + j] = radius_initial

    def objective_function(radii, centers, num_circles):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(radii, centers):
        constraints = []
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            constraints.append({'type': 'ineq', 'fun': lambda r: r})
            constraints.append({'type': 'ineq', 'fun': lambda r: 1 - (x - r)})
            constraints.append({'type': 'ineq', 'fun': lambda r: 1 - (x + r)})
            constraints.append({'type': 'ineq', 'fun': lambda r: 1 - (y - r)})
            constraints.append({'type': 'ineq', 'fun': lambda r: 1 - (y + r)})
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraints.append({'type': 'ineq', 'fun': lambda r: dist - (radii[i] + radii[j])})
        return constraints

    # Convert centers to a list for optimization
    centers_list = centers.tolist()
    radii_list = radii.tolist()

    # Optimization
    result = optimize.minimize(
        objective_function,
        radii_list,
        args=(np.array(centers_list), num_circles),
        method='SLSQP',
        constraints=constraints(radii_list, np.array(centers_list))
    )

    if not result.success:
        raise RuntimeError('Optimization failed.')

    centers_opt = np.array(centers_list)
    radii_opt = np.array(result.x)
    sum_radii = np.sum(radii_opt)

    # Validate the packing
    is_valid, msg = validate_packing(centers_opt, radii_opt)
    if not is_valid:
        raise RuntimeError(f'Packing validation failed: {msg}')

    return centers_opt, radii_opt, sum_radii
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
