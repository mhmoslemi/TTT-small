import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid of 26 circles in a 5x6 grid (leaving one slot empty)
    # Positions are adjusted to cover the unit square [0,1]^2
    n = 26
    grid = np.zeros((n, 2))
    count = 0
    for row in range(5):
        for col in range(6):
            if count >= n:
                break
            x = col / 5.0 + 1e-3
            y = row / 5.0 + 1e-3
            grid[count] = [x, y]
            count += 1

    # Initial guess for radii: same for all circles
    radii_initial = np.full(n, 0.05)

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negate for minimization

    # Constraints
    def constraint_bounds(radii):
        return np.all(radii >= 0)

    def constraint_boundary(radii):
        # Check if all circles are within the square
        centers = np.array([grid[i] for i in range(n)])
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 - 1e-12 or x + r > 1 + 1e-12 or y - r < 0 - 1e-12 or y + r > 1 + 1e-12:
                return False
        return True

    def constraint_overlap(radii):
        # Check non-overlapping between any pair of circles
        centers = np.array([grid[i] for i in range(n)])
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # We define constraints in a way that allows scpy.optimize to use them
    def constraint_function(radii):
        return np.array([constraint_bounds(radii), constraint_boundary(radii), constraint_overlap(radii)])

    # Use SLSQP with bounds and constraints
    bounds = [(0, 1) for _ in range(n)]  # Radii are in [0, 1]

    # Initial guess for optimization
    radii_guess = np.full(n, 0.05)

    # Set up optimization
    result = opt.minimize(
        objective,
        radii_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraint_function}
    )

    # Extract final result
    final_radii = result.x
    final_centers = grid
    total_sum = np.sum(final_radii)

    # Validate if the final solution is within acceptable tolerances
    is_valid, msg = validate_packing(final_centers, final_radii)
    if not is_valid:
        raise RuntimeError(f"Invalid packing: {msg}")

    return final_centers, final_radii, total_sum

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
