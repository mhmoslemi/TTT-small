import matplotlib.pyplot as plt
from scipy.optimize import minimize
from itertools import combinations

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define initial layout: Hexagonal packing within unit square
    def initialize_hexagonal_grid(num_circles):
        rows = int(np.ceil(np.sqrt(num_circles)))
        cols = int(np.ceil(num_circles / rows))
        radius = 0.15  # Initial guess for radius
        centers = np.zeros((num_circles, 2))
        for i in range(rows):
            for j in range(cols):
                x = j * (2 * radius * np.sqrt(3)) / (cols - 1) + 0.5 * radius
                y = i * (2 * radius) / (rows - 1) + 0.5 * radius
                if i % 2 == 1:
                    x += radius * np.sqrt(3)
                idx = i * cols + j
                if idx < num_circles:
                    centers[idx] = [x, y]
        return centers, np.full(num_circles, radius)

    # Objective function to minimize (negative sum of radii)
    def objective(radii, centers, num_circles):
        return -np.sum(radii)

    # Constraint functions
    def constraint_inside_box(radii, centers):
        return np.all(centers[:, 0] - radii >= -1e-12) and np.all(centers[:, 0] + radii <= 1 + 1e-12) and \
               np.all(centers[:, 1] - radii >= -1e-12) and np.all(centers[:, 1] + radii <= 1 + 1e-12)

    def constraint_no_overlap(radii, centers):
        for i, j in combinations(range(num_circles), 2):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii[i] + radii[j] - 1e-12:
                return False
        return True

    # Main optimization function
    def optimize_packing(num_circles, centers, initial_radii):
        bounds = [(0, 1.0) for _ in range(num_circles)]
        constraints = [
            {'type': 'ineq', 'fun': lambda r: constraint_inside_box(r, centers)},
            {'type': 'ineq', 'fun': lambda r: constraint_no_overlap(r, centers)}
        ]
        result = minimize(
            fun=lambda r: -np.sum(r),
            x0=initial_radii,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'maxiter': 500, 'ftol': 1e-10}
        )
        return result.x

    # Define the initial layout
    centers, initial_radii = initialize_hexagonal_grid(26)
    radii = optimize_packing(26, centers, initial_radii)

    # Final validation check
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {message}")

    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
