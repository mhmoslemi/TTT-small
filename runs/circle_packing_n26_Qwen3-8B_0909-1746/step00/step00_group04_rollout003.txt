import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial hexagonal packing inside the unit square
    # We will create a hexagonal grid of 26 circles
    # The hexagonal grid has a top row with 6, then 7, 8, 7, 6 circles
    # We'll place the hexagonal grid in the unit square

    # For simplicity, use a hexagonal grid with a vertical offset between rows
    rows = 5
    cols = [6, 7, 8, 7, 6]
    num_circles = sum(cols)
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Place circles in a hexagonal grid
    offset_y = np.sqrt(3) / 2  # Vertical spacing for hexagonal grid
    offset_x = 1.0  # Horizontal spacing
    y = 0.0
    for i in range(rows):
        row_idx = 0
        for j in range(cols[i]):
            x = j * offset_x
            if i % 2 == 1:
                x += offset_x / 2  # Offset alternate rows
            centers[row_idx, 0] = x
            centers[row_idx, 1] = y
            row_idx += 1
        y += offset_y

    # Normalize the positions to fit in the unit square
    x_min = np.min(centers[:, 0])
    x_max = np.max(centers[:, 0])
    y_min = np.min(centers[:, 1])
    y_max = np.max(centers[:, 1])
    centers[:, 0] = (centers[:, 0] - x_min) / (x_max - x_min)
    centers[:, 1] = (centers[:, 1] - y_min) / (y_max - y_min)

    # Initial radii: assign a base radius
    base_radius = 0.05
    radii[:] = base_radius

    # Create a function to compute total radius sum
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints: non-overlapping and within unit square
    def constraint_overlap(x):
        centers_flat = x[:num_circles, 0]
        centers_flat2 = x[:num_circles, 1]
        radii_flat = x[num_circles:]

        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i] - centers_flat[j]
                dy = centers_flat2[i] - centers_flat2[j]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-9:
                    return -1  # Constraint violated
        return 0

    def constraint_boundary(x):
        centers_flat = x[:num_circles, 0]
        centers_flat2 = x[:num_circles, 1]
        radii_flat = x[num_circles:]

        for i in range(num_circles):
            x_i, y_i = centers_flat[i], centers_flat2[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -1  # Constraint violated
        return 0

    # Flatten the centers and radii for optimization
    x0 = np.vstack((centers, radii)).flatten()

    # Bounds: positions in [0, 1], radii >= 0
    bounds = []
    for i in range(num_circles):
        bounds.extend([(0, 1), (0, 1)])  # X, Y
        bounds.append((0, 1))  # Radius for each circle
    bounds = np.array(bounds)

    # Set up constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    # Run optimization
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-5,
        options={'maxiter': 1000}
    )

    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)

    # Extract the optimized centers and radii
    centers_opt = result.x[:num_circles * 2].reshape((num_circles, 2))

    if not is_valid:
        raise RuntimeError("Invalid packing: " + message)

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
