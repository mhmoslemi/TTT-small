import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based arrangement with circles evenly spaced
    num_circles = 26
    grid_size = int(np.ceil(np.sqrt(num_circles)))
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Start with a simple grid layout
    for i in range(num_circles):
        row = i // grid_size
        col = i % grid_size
        x = col / (grid_size - 1)
        y = row / (grid_size - 1)
        centers[i] = [x, y]
        radii[i] = 0.05  # Initial small radius to avoid overlap

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Define the constraints
    def constraints_func(x):
        centers_flat = x[:num_circles*2].reshape((num_circles, 2))
        radii_flat = x[num_circles*2:]
        # Boundary constraints
        for i in range(num_circles):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if not (0 - 1e-12 <= x_i - r_i and x_i + r_i <= 1 + 1e-12 and
                    0 - 1e-12 <= y_i - r_i and y_i + r_i <= 1 + 1e-12):
                return np.inf
        # Non-overlap constraints
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers_flat[i] - centers_flat[j])**2))
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return np.inf
        return 0

    # Flatten centers and radii for optimization
    x0 = np.concatenate([centers.flatten(), radii])

    # Define bounds for centers (0 to 1) and radii (0 to max possible)
    bounds = Bounds(np.zeros(num_circles*2), np.ones(num_circles*2))
    bounds = Bounds(
        np.zeros(num_circles*2 + num_circles),
        np.ones(num_circles*2 + num_circles)
    )

    # Run optimization
    result = minimize(
        objective,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints_func},
        options={'maxiter': 1000, 'ftol': 1e-8, 'disp': False}
    )

    # Extract the solution
    centers_opt = result.x[:num_circles*2].reshape((num_circles, 2))
    radii_opt = result.x[num_circles*2:]
    sum_radii = np.sum(radii_opt)

    # Validate the packing
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Packing is invalid: {message}")

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
