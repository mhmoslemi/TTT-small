import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial guess: hexagonal grid packing
    n = 26
    rows = int(np.sqrt(n))
    cols = n // rows + 1
    centers = []
    radii = []

    # Initial radius for all circles
    initial_radius = 0.1

    for i in range(rows):
        for j in range(cols):
            x = j * (1.0 / (cols - 1)) if cols > 1 else 0.5
            y = i * (1.0 / (rows - 1)) if rows > 1 else 0.5
            centers.append([x, y])
            radii.append(initial_radius)

    centers = np.array(centers)
    radii = np.array(radii)

    # Objective function: maximize sum of radii (minimize negative sum)
    def objective(x):
        return -np.sum(x)

    # Bounds: radii must be non-negative
    bounds = Bounds(lb=0, ub=np.inf)

    # Constraints: no circle can go out of bounds
    def constraint_out_of_bounds(x, i):
        r = x[i]
        x_center, y_center = centers[i]
        if x_center - r < 0 or x_center + r > 1 or y_center - r < 0 or y_center + r > 1:
            return 1.0
        return 0.0

    # Constraints: no overlap between any two circles
    def constraint_overlap(x, i, j):
        r_i = x[i]
        r_j = x[j]
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = np.sqrt(dx * dx + dy * dy)
        if dist < r_i + r_j - 1e-12:
            return 1.0
        return 0.0

    # Prepare constraint functions for optimization
    def constraint_out_of_bounds_wrapper(x):
        for i in range(n):
            if constraint_out_of_bounds(x, i) > 0:
                return np.inf
        return 0.0

    def constraint_overlap_wrapper(x):
        for i in range(n):
            for j in range(i + 1, n):
                if constraint_overlap(x, i, j) > 0:
                    return np.inf
        return 0.0

    # Combine all constraints
    def constraint_wrapper(x):
        return constraint_out_of_bounds_wrapper(x) + constraint_overlap_wrapper(x)

    # Solve the optimization problem
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint_wrapper},
        method="SLSQP"
    )

    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)

    optimized_radii = result.x
    sum_radii = np.sum(optimized_radii)

    # Validate the packing
    is_valid, message = validate_packing(centers, optimized_radii)
    if not is_valid:
        raise RuntimeError("Invalid packing: " + message)

    return centers, optimized_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
