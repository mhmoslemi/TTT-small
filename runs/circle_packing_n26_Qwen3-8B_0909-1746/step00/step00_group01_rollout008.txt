import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial grid-based packing for 25 circles (5x5 grid)
    num_circles = 26
    grid_size = int(np.ceil(np.sqrt(num_circles)))
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Place 25 circles in a 5x5 grid
    for i in range(grid_size):
        for j in range(grid_size):
            if i * grid_size + j < num_circles - 1:
                x = j / (grid_size - 1)
                y = i / (grid_size - 1)
                centers[i * grid_size + j] = [x, y]
                radii[i * grid_size + j] = 1 / (grid_size + 1)  # Initial estimate for radius
    
    # Place the 26th circle

    centers[-1] = [0.5, 0.5]
    radii[-1] = 0.02  # Begin with a small radius to avoid overlap

    # Define objective function (maximize sum of radii)
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize

    # Define constraints
    def constraints(radii):
        # Constraint 1: Radii are non-negative
        if np.any(radii < 0):
            return np.inf
        # Constraint 2: Circles are within the square
        for i in range(num_circles):
            x, y = centers[i]

            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

        return 0

    # Define bounds for radii (0 to some upper limit)
    bounds = [(0, 0.5) for _ in range(num_circles)]

    # Optimization
    result = minimize(

        bounds=bounds,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints},

    )

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
