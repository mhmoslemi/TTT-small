import numpy as np
import scipy.optimize as opt

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Define initial hexagonal packing
    # Adapted from https://en.wikipedia.org/wiki/Circle_packing_in_a_circle
    # A hexagonal packing of 26 circles of equal radius r
    # We'll use larger circles and adjust them within the unit square

    def compute_hex_packing(n: int) -> np.ndarray:
        # Generate initial hexagonal grid points
        # It's a good starting point for circle packing
        # We will scale it to fit within [0,1]^2
        # Arrange circles in a hexagonal grid

        r = 0.13  # Initial radius estimate
        grid_size = int(np.sqrt(n))
        grid = np.zeros((n, 2))
        idx = 0
        for i in range(grid_size):
            for j in range(grid_size):
                x = j + 0.5 * (i % 2)
                y = i
                grid[idx] = [x, y]
                idx += 1
        # Scale to unit square
        scale = 1.0 / (grid.max() - grid.min())
        return grid * scale

    def objective_function(params: np.ndarray, centers: np.ndarray, n: int) -> float:
        # Objective function to minimize the total area (equivalent to maximizing sum of radii)
        # params = [r1, r2, ..., rn]
        # Returns total area of circles
        return np.sum(np.pi * params**2)

    def constraint_overlap(i: int, j: int, centers: np.ndarray, radii: np.ndarray) -> float:
        # Constraint function to enforce non-overlapping
        # 2 * radii[i] + radii[j] - distance >= 0
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = np.sqrt(dx**2 + dy**2)
        return dist - radii[i] - radii[j]

    def constraint_boundary(i: int, center: np.ndarray, radius: float) -> float:
        # Constraint function to enforce circle stays within unit square
        # x - radius >= 0, x + radius <= 1, y - radius >= 0, y + radius <= 1
        x, y = center
        return np.minimum(x - radius, x + radius - 1, y - radius, y + radius - 1)

    def compute_overlap_constraints(centers: np.ndarray, constraints: list) -> list:
        # Generate non-overlapping constraints
        n = centers.shape[0]
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append(lambda x, i=i, j=j: constraint_overlap(i, j, centers, x))
        return constraints

        n = centers.shape[0]

            x, y = centers[i]

    centers = compute_hex_packing(26)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
