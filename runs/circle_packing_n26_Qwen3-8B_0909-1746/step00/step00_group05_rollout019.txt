from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal packing: 5 rows with 5, 6, 6, 6, 5 circles
    # Assuming circles are equally spaced, with equal radii
    n_circles = 26
    radii = np.full(n_circles, 0.08)  # Initial guess
    centers = np.zeros((n_circles, 2))

    # Initial hexagonal arrangement with equal radius
    radius = 0.08
    row_height = 2 * radius * np.sqrt(3)
    col_width = 2 * radius
    for i in range(n_circles):
        row = i // 6
        col = i % 6
        x = col * col_width - 0.5 * col_width + radius
        y = row * row_height - 0.5 * row_height + radius
        centers[i, 0] = x
        centers[i, 1] = y

    # Refine with optimization
    def objective(radii):
        return -np.sum(radii)

    # Bounds for radii
    bounds = Bounds(0, 1.0)

    # Nonlinear constraints: circle within square and no overlap
    def constraints(radii, centers):
        # Check boundary conditions
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]

                return np.inf

                if dist < radii[i] + radii[j] - 1e-8:

        return 0

    res = minimize(

        radii,
        method='SLSQP',
        bounds=bounds,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
