import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal packing for 26 circles
    def initial_hexagonal_packing(n):
        radius = 0.15  # initial radius
        centers = np.array([
            [0.5 + i * (2 * radius), 0.5] for i in range(n)
        ])
        return centers, np.full(n, radius)

    n = 26
    centers, radii = initial_hexagonal_packing(n)

    # Objective function: minimize total repulsion energy
    def objective(x):
        # x has shape (26 * 3) => [x1, y1, r1, x2, y2, r2, ...]
        x_flat = x.reshape((n, 3))  # [x, y, r]
        centers = x_flat[:, :2]
        radii = x_flat[:, 2]
        
        energy = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                repulsion = np.log(1 + (radii[i] + radii[j]) / dist)
                energy += repulsion

        return energy

    # Constraint 1: Circle stays within [0,1]^2
    def constraint_circle_in_square(x):
        x_flat = x.reshape((n, 3))
        centers = x_flat[:, :2]
        radii = x_flat[:, 2]
        for i in range(n):
            x, y, r = centers[i], radii[i]
            if not (0 <= x - r and x + r <= 1 and 0 <= y - r and y + r <= 1):
                return 1.0
        return 0.0

    # Constraint 2: No overlap between circles
    def constraint_no_overlap(x):
        x_flat = x.reshape((n, 3))
        centers = x_flat[:, :2]
        radii = x_flat[:, 2]
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j]:
                    return 1.0
        return 0.0

    # Bounds: Each circle has x, y in [0, 1], r >= 0
    bounds = [(0, 1) for _ in range(n * 2)] + [(0, 1) for _ in range(n)]

    # Initial guess: flatten centers and radii
    initial_guess = np.concatenate([centers.flatten(), radii])

    # Optimization
    result = opt.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_circle_in_square},
            {'type': 'ineq', 'fun': constraint_no_overlap}
        ],
        options={'maxiter': 500}
    )

    # Extract the centers and radii
    centers_opt = result.x[:n*2].reshape((n, 2))
    radii_opt = result.x[n*2:].reshape((n,))

    # Return the result
    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
