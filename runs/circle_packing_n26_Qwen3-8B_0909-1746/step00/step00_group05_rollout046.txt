import numpy as np
import scipy.optimize as optimize
from typing import Tuple

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based positions
    num_circles = 26
    grid_size = int(np.sqrt(num_circles))  # For a square grid
    grid = np.mgrid[0.1:0.9:grid_size*1j, 0.1:0.9:grid_size*1j].transpose(1, 0, 2).reshape(-1, 2)
    
    # Initial equal radius (for optimization starting point)
    initial_radius = 0.05
    
    # Define objective function to minimize (negative of total radii)
    def objective(x):
        centers = x[:2*num_circles].reshape(num_circles, 2)
        radii = x[2*num_circles:]
        return -np.sum(radii)
    
    # Define constraints
    def constraints(x):
        centers = x[:2*num_circles].reshape(num_circles, 2)

        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:

        # Ensure circles do not overlap
        for i in range(num_circles):

        return np.array([])
    
    # Initial guess: grid positions and equal radii
    initial_guess = np.vstack((grid, np.full(num_circles, initial_radius))).ravel()

    result = optimize.minimize(
        objective,
        initial_guess,
        method='SLSQP',

    if not is_valid:

    # Compute sum of radii
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
