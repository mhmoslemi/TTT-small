import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run_packing():
    # Step 1: Initial placement in a hexagonal grid
    n = 26
    # Arrange 26 circles in a hexagonal pattern
    # First, compute the number of rows (approximate sqrt(n) = 5.1)
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # We use a grid-like pattern with some hexagonal spacing
    centers = []
    for r in range(rows):
        for c in range(cols):
            # Each row alternates the x-position
            if r % 2 == 0:
                x = c / (cols - 1) if cols > 1 else 0.5
            else:
                x = (c + 0.5) / (cols - 1) if cols > 1 else 0.5
            y = r / (rows - 1) if rows > 1 else 0.5
            centers.append([x, y])
    
    centers = np.array(centers)
    
    # Step 2: Optimization using Nelder-Mead simplex
    # Objective function: maximize the sum of radii
    def objective(params):
        centers_new = params[:2*n].reshape(n, 2)
        radii_new = params[2*n:]
        # Ensure non-negative radii
        radii_new = np.maximum(radii_new, 1e-8)
        # Enforce boundary constraints
        for i in range(n):
            x, y = centers_new[i]
            r = radii_new[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return -np.inf  # Invalid packing
        # Enforce non-overlapping constraints
        for i in range(n):

                dx = centers_new[i, 0] - centers_new[j, 0]
                dy = centers_new[i, 1] - centers_new[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_new[i] + radii_new[j] - 1e-12:
                    return -np.inf  # Invalid packing
        # Return the negative of sum for maximization
        return -np.sum(radii_new)

    # Initial guess: equal radii and positions from the initial grid
    # We assume initial radii are small to avoid overlaps
    initial_radii = np.full(n, 0.05)
    initial_params = np.concatenate([centers.flatten(), initial_radii])
    bounds = Bounds(np.zeros(2*n + n), np.full(2*n + n, np.inf))

    # Set optimization options
    options = {
        'maxiter': 500,
        'xatol': 1e-8,
        'fatol': 1e-8,
    }

    # Run optimization
    result = opt.minimize(
        objective,
        initial_params,
        method='Nelder-Mead',
        bounds=bounds,
        options=options
    )

    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)

    centers_opt = result.x[:2*n].reshape(n, 2)

    if not valid:

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
