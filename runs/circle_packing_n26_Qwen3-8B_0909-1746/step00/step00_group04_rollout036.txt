import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initialize with a hexagonal packing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    rows = int(np.sqrt(num_circles * 2 / 3))
    cols = int(np.sqrt(num_circles * 2 / 3))
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * (1 + np.sqrt(3)/2) / rows
            else:
                x = (j + 0.5) * (1 + np.sqrt(3)/2) / rows
            y = i * np.sqrt(3)/2 / rows
            if i * cols + j < num_circles:
                centers[i * cols + j] = [x, y]
                radii[i * cols + j] = 1 / rows

    def objective(radii_vec):
        return -np.sum(radii_vec)  # Minimize negative sum to maximize sum

    # Define constraints
    def constraint_center_in_square(x):
        centers = x[:2*num_circles].reshape(num_circles, 2)
        radii = x[2*num_circles:]
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if not (0 - 1e-12 <= x_i - r_i <= 1 + 1e-12 and 0 - 1e-12 <= y_i - r_i <= 1 + 1e-12):
                return -1
        return 0

    def constraint_overlap(x):
        centers = x[:2*num_circles].reshape(num_circles, 2)
        radii = x[2*num_circles:]
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Initial guess with centers and radii
    x0 = np.zeros(2*num_circles + num_circles)
    x0[:2*num_circles] = centers.ravel()
    x0[2*num_circles:] = radii

    # Bounds: centers are in [0, 1], radii are non-negative
    bounds = [(0, 1) for _ in range(2*num_circles)] + [(0, 1) for _ in range(num_circles)]

    # Constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_center_in_square},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Perform optimization
    result = opt.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6
    )

    # Extract results
    centers_opt = result.x[:2*num_circles].reshape(num_circles, 2)
    radii_opt = result.x[2*num_circles:]
    sum_radii = np.sum(radii_opt)

    # Validate packing

    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
