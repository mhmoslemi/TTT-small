import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt

def run_packing():

    # Hexagonal packing: each row has (2n + 1) circles in the first row, (2n) circles in the second, etc.
    # We'll start with 5 rows for 26 circles

        max_circles_per_row = 5
        circles_per_row = [max_circles_per_row - i for i in range(n_rows)]

        for row in range(n_rows):
            row_centers = []
            for col in range(circles_per_row[row]):
                x = col + 0.5

                row_centers.append([x, y])
            centers.extend(row_centers)

    centers = initial_hexagonal_packing()
    radii = np.full(centers.shape[0], 0.1)

    def constraints(radii):

        for i in range(n):
            r = radii[i]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
