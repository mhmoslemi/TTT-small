import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initialize centers in a grid pattern, perturbed slightly
    n = 26
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 5 + np.random.uniform(-0.1, 0.1)
        y = row / 5 + np.random.uniform(-0.1, 0.1)
        centers[i] = [x, y]

    # Initialize radii with a uniform value based on the grid
    # Start with a baseline radius that should be feasible
    base_radius = 0.05
    radii = np.full(n, base_radius)

    # Objective function: maximize the sum of radii with penalties for overlap and boundary violations
    def objective(x: np.ndarray) -> float:
        # x is a 2D array of [centers, radii]
        # Extract centers and radii
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:].reshape(n)
        
        # Calculate sum of radii
        sum_radii = np.sum(radii)
        
        # Penalties for overlap and boundary violations
        penalties = 0.0
        
        # Penalties for boundary violation
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or
                y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                penalties += 1e4 * (max(0, (x_i - r_i + 1e-12)) + max(0, (x_i + r_i - 1 - 1e-12)) +
                                    max(0, (y_i - r_i + 1e-12)) + max(0, (y_i + r_i - 1 - 1e-12)))
        
        # Penalties for overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    penalties += 1e4 * (radii[i] + radii[j] - dist)
        
        return -sum_radii + penalties

    # Bounds for the optimization: centers must be in [0, 1]^2, radii must be non-negative
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])  # x and y coordinates for each center
        bounds.extend([(0.0, 1.0)])              # radius for each circle

    # Optimization with L-BFGS-B method
    result = opt.minimize(
        fun=objective,
        x0=np.concatenate([centers.flatten(), radii]),
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )

    # Extract the final centers and radii

    return final_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
