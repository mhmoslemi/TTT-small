import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial hexagonal grid configuration (positions for 26 circles)
    # We use rows and columns to generate a hexagonal grid and adjust for the unit square
    grid = []
    for row in range(5):
        for col in range(5 + (row % 2)):
            x = col + (row % 2) * 0.5
            y = row + 0.5
            grid.append([x, y])

    # Convert to numpy arrays
    centers = np.array(grid)
    radii = np.full(26, 0.05)  # Initial small radii

    # Define constraint for circle not to go outside the unit square
    def square_boundaries(x):
        return np.array([x[0] - x[1] - 1e-12, x[0] + x[1] - 1.0 + 1e-12,
                         x[2] - x[3] - 1e-12, x[2] + x[3] - 1.0 + 1e-12])
    
    # Define constraint for no overlap between circles
    def no_overlap(x):
        n = int(len(x) / 4)
        overlaps = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((x[2*i] - x[2*j])**2 + (x[2*i+1] - x[2*j+1])**2)
                overlap = dist - (x[2*n + i] + x[2*n + j]) + 1e-12
                overlaps.append(overlap)
        return np.array(overlaps)

    # Objective function: minimize the negative sum of radii (to maximize it)
    def objective(x):
        n = int(len(x) / 4)
        return -np.sum(x[2*n:])

    # Initial guess for centers and radii
    x0 = np.concatenate([centers.flatten(), radii])

    # Define bounds for centers and radii
    bounds = []
    for i in range(26):
        bounds += [(0.0, 1.0), (0.0, 1.0)]  # x and y for centers
        bounds += [(0.0, 1.0)]              # radius for each circle

    # Set up constraints
    constraints = [
        {'type': 'ineq', 'fun': square_boundaries},
        {'type': 'ineq', 'fun': no_overlap}
    ]

    # Run optimization
    result = opt.minimize(
        fun=objective,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 100, 'disp': False}
    )

    # Extract centers and radii
    centers_opt = result.x[:26*2].reshape(26, 2)
    radii_opt = result.x[26*2:].reshape(26)
    sum_radii = np.sum(radii_opt)

    return centers_opt, radii_opt, sum_radii
