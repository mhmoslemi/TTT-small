import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial placement based on a hexagonal grid
    # Number of rows and columns
    num_rows = 5
    num_cols = 6
    
    # Calculate grid spacing
    # For the unit square, we want to fit the circles such that their centers stay within [0, 1]^2
    # We assume square packing for simplicity initially
    dx = 1.0 / (num_cols - 1)
    dy = 1.0 / (num_rows - 1)
    
    # Start the centers
    centers = np.zeros((num_rows * num_cols, 2))
    for i in range(num_rows):
        for j in range(num_cols):
            centers[i * num_cols + j] = [j * dx, i * dy]
    
    # Initial guess for radii
    radii = np.ones(num_rows * num_cols) * 0.1  # start with small radii
    
    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # negative for minimization

    # Constraints:
    # 1. Each circle must be inside the square
    def constraint_boundary(radii):
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return np.inf  # invalidate if any circle goes out of bounds
        return 0

    # 2. Each pair of circles must not overlap
    def constraint_overlap(radii):

                dy = centers[i, 1] - centers[j, 1]

                    return np.inf  # invalidate if circles overlap
        return 0

    # Combine constraints

    ]

    # Run optimization
    result = optimize.minimize(
        objective,

        method='SLSQP',
        bounds=bounds,
        constraints=cons

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
