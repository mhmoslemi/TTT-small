import numpy as np
from scipy.optimize import minimize
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26

    # Compute the side length of the hexagon for hexagonal packing
    # We use a hexagonal grid within the unit square
    # We use a hexagonal grid width that fits within [0,1]

    # Initial hexagonal grid configuration
    def hexagonal_grid(n):
        # Generate a hexagonal grid with n points
        # This is a rough approximation, can be refined
        points = []
        for i in range(n):
            row = i // 5
            col = i % 5
            x = col / 5
            y = (row + (col % 2)) / 5
            points.append((x, y))
        return np.array(points)

    # Initial centers
    centers = hexagonal_grid(n)
    radii = np.full(n, 0.05)  # Initial small radius

    # Optimization variables
    x = cp.Variable(n, 2)
    r = cp.Variable(n)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []
    for i in range(n):
        # Boundary constraints
        constraints.append(0 <= x[i, 0] - r[i])
        constraints.append(x[i, 0] + r[i] <= 1)
        constraints.append(0 <= x[i, 1] - r[i])
        constraints.append(x[i, 1] + r[i] <= 1)

    for i in range(n):
        for j in range(i + 1, n):
            # Non-overlap constraint
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-6)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, max_iter=1000)

    # Extract solution
    centers = np.reshape(x.value, (n, 2))
    radii = r.value

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
