import scipy.optimize as opt
import scipy.spatial.distance as dist
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based layout for 26 circles in a 5x5 grid
    n = 26
    grid_size = int(np.ceil(np.sqrt(n)))
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(grid_size):
        for j in range(grid_size):
            if idx < n:
                centers[idx, 0] = i / (grid_size - 1)
                centers[idx, 1] = j / (grid_size - 1)
                idx += 1
            else:
                break

    radii = np.full(n, 1.0 / grid_size)

    def objective(radii_vec):

                constraints.append(dist_ij - (radii_vec[i] + radii_vec[j]) >= 1e-8)
        return constraints

        constraints = []
        for i in range(n):
            x, y = centers[i]

            constraints.append(x - r >= 0 - 1e-8)
            constraints.append(x + r <= 1 + 1e-8)
            constraints.append(y - r >= 0 - 1e-8)
            constraints.append(y + r <= 1 + 1e-8)
        return constraints

    # Initial guess
    initial_radii = radii.copy()

    # Optimization setup
    bounds = [(0, 1.0) for _ in range(n)]

    # Optimize

        method='SLSQP',
        bounds=bounds,
        constraints=cons,

    )
