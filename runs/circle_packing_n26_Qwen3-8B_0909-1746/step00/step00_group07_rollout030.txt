import numpy as np
import scipy.optimize as opt
import math

def run_packing():
    """
    Returns a packing of 26 circles in a unit square with maximum total radius sum.
    Uses a gradient-based optimization approach with initial guesses based on hexagonal packing.
    """

    # Section 1: Initial guess based on hexagonal packing
    def hexagonal_grid_positions(n, side_length):
        """
        Generate positions for a hexagonal grid with n circles, where side_length is the distance between centers.
        """
        rows = int(math.ceil(math.sqrt(n)))
        cols = int(math.ceil(n / rows))
        positions = []
        for i in range(rows):
            for j in range(cols):
                x = j * side_length * np.sqrt(3) / 2
                y = i * side_length * 3 / 2
                if i % 2 == 1:
                    x += side_length * np.sqrt(3) / 4
                positions.append(np.array([x, y]))
        return np.array(positions)

    # Section 2: Objective function to maximize
    def objective(radii):
        return -np.sum(radii)  # We minimize the negative sum

    # Section 3: Constraints
    def constraints(centers, radii):
        # Overall constraints
        constraints = []

        # Circle radius non-negativity
        for i in range(len(radii)):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i]})

        # Circle boundary constraints
        for i in range(len(radii)):
            def boundary_constraint(r):
                x, y = centers[i]
                r_i = r[i]
                return [x - r_i + 1e-12, x + r_i - 1 - 1e-12,
                        y - r_i + 1e-12, y + r_i - 1 - 1e-12]
            constraints.extend([
                {'type': 'ineq', 'fun': lambda r, i=i: boundary_constraint(r)[0]},
                {'type': 'ineq', 'fun': lambda r, i=i: boundary_constraint(r)[1]},
                {'type': 'ineq', 'fun': lambda r, i=i: boundary_constraint(r)[2]},
                {'type': 'ineq', 'fun': lambda r, i=i: boundary_constraint(r)[3]}
            ])

        # Circle-circle non-overlap constraints
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                def circle_circle_constraint(r):
                    dx = centers[i][0] - centers[j][0]
                    dy = centers[i][1] - centers[j][1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    return dist - r[i] - r[j] + 1e-12
                constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: circle_circle_constraint(r)})

        return constraints

    # Section 4: Optimization
    def run_optimization():
        # Initial guess
        side_length = 0.3
        num_circles = 26
        initial_positions = hexagonal_grid_positions(num_circles, side_length)
        initial_radii = np.ones(num_circles) * 0.05

        # Setup optimization problem
        bounds = [(0, 1) for _ in range(2 * num_circles)]  # Bounds for x and y coordinates
        bounds += [(0, 0.5) for _ in range(num_circles)]  # Bounds for radii (limiting to prevent overlap)

        # Optimize
        result = opt.minimize(
            objective,
            np.concatenate((initial_positions.flatten(), initial_radii)),
            method='SLSQP',
            constraints=constraints(initial_positions, initial_radii),
            bounds=bounds,
            tol=1e-6,
            options={'maxiter': 500, 'disp': False}
        )

        # Extract results
        centers = result.x[:2*num_circles].reshape(-1, 2)
        radii = result.x[2*num_circles:]
        sum_radii = np.sum(radii)

        # Final validation
        is_valid, message = validate_packing(centers, radii)
        if not is_valid:
            raise ValueError(message)

        return centers, radii, sum_radii

    # Section 5: Final execution
    try:
        centers, radii, sum_radii = run_optimization()
        return centers, radii, sum_radii
    except Exception as e:
        raise RuntimeError(f"Optimization failed: {e}")

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
