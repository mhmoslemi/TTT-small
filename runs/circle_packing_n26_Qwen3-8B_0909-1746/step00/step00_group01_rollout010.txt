import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Number of circles
    n = 26

    # Initial radii
    initial_radii = np.ones(n) * 0.05

    # Initial centers - based on a hexagonal grid
    # We try to fit as many rows as we can within the square
    # Max 6 rows (each row height = 2r * sqrt(3)/2 = r * sqrt(3))
    max_rows = int(1 / (np.sqrt(3) * initial_radii[0]))
    max_cols = int(1 / (2 * initial_radii[0]))
    if max_rows * max_cols < n:
        max_rows = int(n / max_cols)

    centers = np.zeros((n, 2))
    for i in range(n):

        y = row * (np.sqrt(3) * initial_radii[0])
        centers[i] = [x, y]

    # Maximize the total radii
    # Define the objective function to minimize (negative of sum of radii)
    # Let's use a hybrid approach: row-wise optimization and then full optimization

    # Objective function
    def objective(x):
        # x is a flat array: [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
        flat_centers = x[:2 * n].reshape((n, 2))
        flat_radii = x[2 * n:]

        # Check if any radius is negative
        if np.any(flat_radii < 0):
            return np.inf

        # Compute total radii
        total_radii = np.sum(flat_radii)

            x, y = flat_centers[i]
            r = flat_radii[i]

                return np.inf

        for i in range(n):

                dx = flat_centers[i, 0] - flat_centers[j, 0]
                dy = flat_centers[i, 1] - flat_centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < flat_radii[i] + flat_radii[j] - 1e-12:
                    return np.inf

        return -total_radii

    bounds = []

    return centers_result, radii_result, total_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
