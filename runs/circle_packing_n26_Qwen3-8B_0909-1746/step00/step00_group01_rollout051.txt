import scipy.optimize as opt
import numba as nb

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal grid: 26 circles
    num_circles = 26
    grid = np.array([
        # Hexagonal grid in unit square
        [0.25, 0.25],
        [0.5, 0.25],
        [0.75, 0.25],
        [0.25, 0.45],
        [0.5, 0.45],
        [0.75, 0.45],
        [0.25, 0.65],
        [0.5, 0.65],
        [0.75, 0.65],
        [0.25, 0.85],
        [0.5, 0.85],
        [0.75, 0.85],
        [0.125, 0.125],
        [0.375, 0.125],
        [0.625, 0.125],
        [0.875, 0.125],
        [0.125, 0.325],
        [0.375, 0.325],
        [0.625, 0.325],
        [0.875, 0.325],
        [0.125, 0.525],
        [0.375, 0.525],
        [0.625, 0.525],
        [0.875, 0.525],
        [0.125, 0.725],
        [0.375, 0.725],
        [0.625, 0.725],
        [0.875, 0.725]
    ])

    # Initial estimate for radii
    radii_init = np.ones(num_circles) * 0.05

    @nb.njit
    def objective(radii):
        return -np.sum(radii)

    @nb.njit
    def constraint_overlap(radii):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = grid[i, 0] - grid[j, 0]
                dy = grid[i, 1] - grid[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-6:
                    return -1.0
        return 0.0

    @nb.njit
    def constraint_boundary(radii):
        for i in range(len(radii)):
            x, y = grid[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1.0
        return 0.0

    bounds = [(0.0, 0.1) for _ in range(num_circles)]

    # Optimization with constraints
    result = opt.minimize(
        fun=objective,
        x0=radii_init,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        method='SLSQP'
    )

    # Final output
    centers = grid
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
