import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt
from scipy.spatial.distance import cdist

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    n = 26

    # Initial guess: Hexagonal packing in a unit square
    # We'll use a more efficient starting point for this specific task
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial guess for radii
    radii = np.ones(n) * 0.05  # Start with small radii, will optimize

    # Define the objective function
    def objective(x):
        return -np.sum(x)  # We maximize the sum of radii by minimizing the negative sum

    # Define the constraints
    def constraint_boundary(x):
        x_centers = x[:n * 2].reshape(n, 2)
        r = x[n * 2:]
        for i in range(n):
            x_i, y_i = x_centers[i]
            r_i = r[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or
                y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                return np.inf  # Violation: large penalty
        return 0  # No violation

    def constraint_overlap(x):
        x_centers = x[:n * 2].reshape(n, 2)
        r = x[n * 2:]
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((x_centers[i] - x_centers[j]) ** 2))
                if dist < r[i] + r[j] - 1e-12:
                    return np.inf  # Violation: large penalty
        return 0  # No violation

    # Prepare initial guess for optimization
    initial_guess = np.concatenate((centers.reshape(n * 2), radii))

    # Set up bounds: All centers must be within [0,1]^2, radii must be >= 0
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x-coordinate
        bounds.append((0.0, 1.0))  # y-coordinate
        bounds.append((0.0, np.inf))  # radius

    # Set up constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Run optimization
    result = opt.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}
    )

    # Extract and validate result
    centers_opt = result.x[:n * 2].reshape(n, 2)
    radii_opt = result.x[n * 2:]
    sum_radii = np.sum(radii_opt)

    # Ensure non-negative radii and valid centers
    radii_opt[radii_opt < 0] = 0
    centers_opt = np.clip(centers_opt, 0, 1)

    # Final validation
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    return centers_opt, radii_opt, sum_radii


def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
