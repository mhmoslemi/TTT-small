from scipy.optimize import minimize
from numpy import sqrt, pi

def run_packing():
    # Initial approximate arrangement of 26 circles (hexagonal packing)
    num_circles = 26
    circles_per_layer = [6, 10, 10]
    radii_initial = np.full(num_circles, 0.06)  # Initial radius guess
    centers = []

    # Hexagonal grid parameters
    row_spacing = sqrt(3) / 2
    col_spacing = 1

    for layer, count in enumerate(circles_per_layer):
        for i in range(count):
            x = i * col_spacing
            y = layer * row_spacing
            centers.append([x, y])
    
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii_initial)

    # Objective function: maximize sum of radii
    def objective(x):
        # Reshape x to (n, 2) for centers and (n,) for radii
        x_centers = x[:2*num_circles].reshape(num_circles, 2)
        x_radii = x[2*num_circles:]
        
        # Compute the sum of radii
        sum_radii = np.sum(x_radii)
        return -sum_radii  # Minimize negative sum to maximize sum

    # Constraints
    def constraints(x):
        x_centers = x[:2*num_circles].reshape(num_circles, 2)
        x_radii = x[2*num_circles:]
        
        # Boundary constraint: all circles must be inside unit square
        boundary_constraints = []
        for i in range(num_circles):
            x_i, y_i = x_centers[i]
            r_i = x_radii[i]
            boundary_constraints.append((x_i - r_i >= -1e-12) & (x_i + r_i <= 1 + 1e-12) &
                                        (y_i - r_i >= -1e-12) & (y_i + r_i <= 1 + 1e-12))
        boundary_constraints = np.prod(boundary_constraints)

        # Non-overlapping constraint
        non_overlapping = []

                non_overlapping.append(dist >= x_radii[i] + x_radii[j] - 1e-12)
        non_overlapping = np.prod(non_overlapping)

        return (boundary_constraints, non_overlapping)

    # Initial guess: centered around initial arrangement with small perturbations
    x0 = np.concatenate([centers.flatten(), radii])

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
