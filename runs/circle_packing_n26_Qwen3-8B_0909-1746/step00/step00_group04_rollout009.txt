import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal-like arrangement with 26 circles
    # We'll fit 26 circles in a hexagonal pattern with rows.
    # We'll fit 5 rows: 6, 6, 6, 6, 6 circles (total 30)
    # We'll reduce to 26 by removing some circles later
    
    num_circles = 26
    rows = 5
    cols_per_row = 6
    spacing = 1.0 / (cols_per_row - 1)
    
    # Generate hexagonal grid
    centers = []
    for i in range(rows):
        for j in range(cols_per_row):
            x = j * spacing
            y = i * (np.sqrt(3) / 2) * spacing
            centers.append([x, y])
    
    # Convert to numpy array
    centers = np.array(centers)
    
    # Remove 4 circles to fit 26
    # Remove outer ones to optimize
    centers = centers[:26]
    
    # Initial guess for radii (based on hexagonal packing)
    # The maximum possible radius in this configuration is ~0.16
    radii = np.full(num_circles, 0.16)
    
    # Optimization bounds for centers
    center_bounds = Bounds([0, 0], [1, 1])
    
    # Optimization bounds for radii
    radius_bounds = Bounds([0], [1])
    
    # Optimization objective: maximize sum of radii
    def objective(x):
        return -np.sum(x[26:])

    # Optimization constraints: non-overlapping and within square
    def constraint_center(x):
        return np.concatenate([x[:26], x[26:]])

    def constraint_overlap(x):
        # x[:26] are centers, x[26:] are radii
        centers = x[:26].reshape(num_circles, 2)
        radii = x[26:]
        dist_matrix = pdist(centers)
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = dist_matrix[i * num_circles + j]
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - (radii[i] + radii[j]) + 1e-8
        return 0.0

    def constraint_boundary(x):
        centers = x[:26].reshape(num_circles, 2)
        radii = x[26:]
        for i in range(num_circles):
            x_center, y_center = centers[i]
            r = radii[i]
            if x_center - r < 0 or x_center + r > 1 or y_center - r < 0 or y_center + r > 1:
                return 1e-8 - (x_center - r) if x_center - r < 0 else 1e-8 - (x_center + r - 1) if x_center + r > 1 else 1e-8 - (y_center - r) if y_center - r < 0 else 1e-8 - (y_center + r - 1)
        return 0.0
    
    # Perform optimization
    x0 = np.concatenate([centers.flatten(), radii])
    result = opt.minimize(
        objective,
        x0,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}

        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 1000}
    )

    radii_opt = result.x[26:]
    sum_radii = np.sum(radii_opt)
    
    # Validate packing
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")
    
    return (centers_opt, radii_opt, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
