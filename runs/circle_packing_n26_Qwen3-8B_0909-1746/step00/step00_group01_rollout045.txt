import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initialize centers with a hexagonal packing
    num_circles = 26
    radius = 0.05  # Initial guess for radius
    centers = np.zeros((num_circles, 2))
    
    # Generate hexagonal grid
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        x = col + (0.5 if row % 2 else 0)
        y = row
        centers[i] = [x, y]
    
    # Scale grid to fit in unit square
    centers /= np.max(centers, axis=0)  # Normalize to [0,1] x [0,1]
    
    # Initial guess for radii
    radii = np.full(num_circles, radius)
    
    # Total packing sum
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraint: each circle must lie within unit square
    def constraint_in_square(radii, centers):
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        return True

    # Constraint: no overlapping between circles
    def constraint_no_overlap(radii, centers):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Bounds: radii must be non-negative
    bounds = [(0, 1) for _ in range(num_circles)]

    # Optimization with Nelder-Mead
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda r: constraint_in_square(r, centers)},
            {'type': 'ineq', 'fun': lambda r: constraint_no_overlap(r, centers)}
        ],
        method='Nelder-Mead',
        options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 1000}
    )

    # Extract final centers and radii
    final_centers = centers
    final_radii = result.x

    # Validate the packing

    if not is_valid:
        raise ValueError(msg)

    # Compute sum of radii
    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
