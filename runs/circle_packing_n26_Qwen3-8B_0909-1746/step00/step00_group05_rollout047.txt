import scipy.optimize as opt

def run_packing():

    num_circles = 26
    grid_rows = 5
    grid_cols = 6
    dx = 1.0 / grid_cols
    dy = np.sqrt(3)/2 / grid_rows
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

            x = j * dx
            y = i * dy
            idx = i * grid_cols + j
            centers[idx, 0] = x
            centers[idx, 1] = y
            radii[idx] = 0.05  # Initial radius
