import numpy as np
import scipy.optimize as optimize
from typing import Tuple
import math

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def initial_placement(n_circles: int) -> Tuple[np.ndarray, np.ndarray]:
    # Initial hexagonal grid layout
    rows = int(math.ceil(math.sqrt(n_circles)))
    cols = int(math.ceil(n_circles / rows))
    
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j

                break

            centers[idx] = [x, y]
            radii[idx] = 0.05  # Initial radius

    return centers, radii

def objective_function(radii, centers):
    return -np.sum(radii)  # Minimize negative of sum to maximize sum

def constraints(centers, radii, n_circles):
    constraints = []

    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        constraints.append({

        })

            'type': 'ineq',
            'fun': lambda x, i=i, r=r: 1.0 - x[0] - r,
            'args': (i, r)
        })

            'type': 'ineq',
            'fun': lambda x, i=i, r=r: x[1] - r,
            'args': (i, r)
        })
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i, r=r: 1.0 - x[1] - r,
            'args': (i, r)
        })

    # Non-overlapping constraints
    for i in range(n_circles):
        for j in range(i + 1, n_circles):

                dx = x[2 * i] - x[2 * j]
                dy = x[2 * i + 1] - x[2 * j + 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                return dist - x[2 * n_circles + i] - x[2 * n_circles + j]
            constraints.append({
                'type': 'ineq',
                'fun': constraint
            })

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    n_circles = 26
    centers, radii = initial_placement(n_circles)
