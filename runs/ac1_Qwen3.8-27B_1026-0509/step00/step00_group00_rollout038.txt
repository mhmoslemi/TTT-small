    """Project x onto {a >= 0, sum(a) = 1}."""
    n = len(x)
    # Binary search for tau
    lo, hi = 0.0, np.max(x)
    for _ in range(100):
        tau = (lo + hi) / 2
        s = np.sum(np.maximum(x - tau, 0))
        if s > 1:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2
