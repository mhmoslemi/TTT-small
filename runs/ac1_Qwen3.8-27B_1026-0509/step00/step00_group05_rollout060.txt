        accepted = False
        for mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            e = eta * mult / d_norm
            x_try = x * np.exp(-e * d)
            x_try = np.maximum(x_try, 1e-30)
            s = x_try.sum()
            if s < 1e-10:
                continue
            x_try /= s
            M_try = self_conv(x_try).max()
            if M_try < M:
                x = x_try
                accepted = True
                break

        if not accepted:
            x = x * np.exp(rng.normal(0, 0.03, n))
            x = np.maximum(x, 1e-30)
            x /= x.sum()

        if (time.monotonic() - last_improve) > 3.0:
            c_check = self_conv(x)
            sc = 2.0 * n * c_check.max()
            if sc < best_score - 1e-10:
                best_score = sc
                best_work = x.copy()
                best = np.clip(best_work * n, 0, 1000)
                last_improve = time.monotonic()
                eta = min(eta * 1.15, 1.5)
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 20:
                    sigma = min(0.05 + 0.02 * stall_count, 0.8)
                    x = x * np.exp(rng.normal(0, sigma, n))
                    x = np.maximum(x, 1e-30)
                    x /= x.sum()
                    eta = 0.3
                    stall_count = 0

        eta *= 0.9998
        if eta < 0.02:
            eta = 0.02

    final = np.clip(best, 0, 1000)
    if final.sum() >= 0.01:
        sc = evaluate_sequence(final.tolist())
        if not np.isfinite(sc):
            final = np.clip(x0, 0, 1000)
            sc = evaluate_sequence(final.tolist())
    else:
        final = np.clip(x0, 0, 1000)

    return [float(v) for v in final]
