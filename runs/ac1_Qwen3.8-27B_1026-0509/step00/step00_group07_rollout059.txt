def lp_mirroring(a, num_iters, eta):
    n = len(a)
    b = a.copy()  # or b = ones
    for _ in range(num_iters):
        conv = np.convolve(a, b, 'full')  # length 2n-1
        M = max(np.convolve(a, a, 'full'))  # the threshold from a*a
        # Actually, M should be fixed from a*a, not recomputed each iter

        if not V_mask.any():

        # Compute penalty
        a_rev = a[::-1]

        # Multiplicative update
        b = b * np.exp(-eta * penalty / M)
        b = np.maximum(b, 1e-10)  # keep non-negative
    return b

    shifted = conv - max_conv
    logsumexp = max_conv + np.log(np.sum(np.exp(beta * shifted)))
    smooth_max = logsumexp - np.log(len(conv))  # average in log space... 
    # Actually, smoothed max = (1/m) * logsumexp(β*x) is not quite right

For n points, sum_i (2i/(n-1)-1)² = (n-1)²/n * sum_{k=0}^{n-1} (2k/(n-1)-1)²/(n-1)² * n... 

Let me just compute: let x_i = 2i/(n-1) - 1 for i=0,...,n-1.
These are equally spaced from -1 to 1.
sum x_i² = (n/(n-1)²) * sum_{i=0}^{n-1} (2i-(n-1))²
= (n/(n-1)²) * (4*sum i² - 4(n-1)*sum i + n*(n-1)²)
= (n/(n-1)²) * (4*(n-1)n(2n-1)/6 - 4(n-1)²n/2 + n(n-1)²)
= (n/(n-1)²) * (n(n-1)(2n-1)*2/3 - 2n(n-1)² + n(n-1)²)
= (n/(n-1)²) * n(n-1) * [(2n-1)*2/3 - 2(n-1) + (n-1)]
= (n/(n-1)) * [(4n-2)/3 - (n-1)]
= (n/(n-1)) * (4n-2-3n+3)/3
= (n/(n-1)) * (n+1)/3
= n(n+1)/(3(n-1))

So center = n - t² * n(n+1)/(3(n-1)) ≈ n - t²*n/3 for large n.
