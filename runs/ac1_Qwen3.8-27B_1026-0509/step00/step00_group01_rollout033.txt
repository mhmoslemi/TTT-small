import numpy as np
import time

def simplex_project(b):
    """Project b onto {x >= 0, sum x = 1}."""
    n = len(b)
    # Sort in descending order
    u = np.sort(b)[::-1]
    # Find the threshold
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(b - theta, 0)

def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum x = 1}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find rho such that u[rho] > (cssv[rho] - 1) / (rho+1)
    cond = u * np.arange(1, n+1) - (cssv - 1) > 0
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
