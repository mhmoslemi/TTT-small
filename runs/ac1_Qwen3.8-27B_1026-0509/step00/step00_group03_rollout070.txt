import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    try:
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()
    except Exception:
        a = np.ones(512, dtype=np.float64)
    if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
        a = np.ones(512, dtype=np.float64)
    n = int(a.size)

    def fast_score(x):
        Sx = x.sum()
        if Sx < 0.01:
            return np.inf
        return 2.0 * n * float(np.convolve(x, x).max()) / (Sx * Sx)

    best = a.copy()
    best_score = fast_score(a)

    # Precompute LP matrix structure once (A[k, j] = a[k-j] for valid (k,j)).
    row_counts = np.array([min(k, n - 1) - max(0, k - n + 1) + 1 for k in range(2 * n - 1)], dtype=np.int64)
    row_starts = np.zeros(2 * n, dtype=np.int64)
    row_starts[1:] = np.cumsum(row_counts)[:-1]
    total_nnz = int(row_counts.sum())
    all_cols = np.concatenate([np.arange(max(0, k - n + 1), min(n - 1, k) + 1) for k in range(2 * n - 1)])
    all_rows = np.repeat(np.arange(2 * n - 1), row_counts)

    def solve_lp(cur, M, tlimit):
        data = np.empty(total_nnz)
        for k in range(2 * n - 1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            data[row_starts[k]:row_starts[k] + row_counts[k]] = cur[lo:hi + 1][::-1]
        A = csr_matrix((data, (all_rows, all_cols)), shape=(2 * n - 1, n))
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': max(0.2, tlimit)})
        if res.status == 0 and res.x is not None:
            return np.asarray(res.x, dtype=np.float64)
        return None

    lp_calls = 0
    no_improve = 0

    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        M = float(b.max())
        k_star = int(np.argmax(b))
        S = float(a.sum())
        improved = False

        remaining = deadline - time.monotonic()
        if remaining > 4.0 and (lp_calls < 50 or lp_calls % 4 == 0):
            g = solve_lp(a, M, max(0.5, remaining - 10.0))
            lp_calls += 1
            if g is not None:
                gs = float(g.sum())
                if gs > 1e-10:
                    g = g * (S / gs)
                    for t in (1.0, 0.7, 0.5, 0.3, 0.1):
                        cand = (1.0 - t) * a + t * g
                        cand = np.clip(cand, 0.0, 1000.0)
                        sc = fast_score(cand)
                        if sc < best_score:
                            best_score = sc
                            best = cand.copy()
                            a = cand.copy()
                            improved = True
                            break

        if not improved:
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            jj = k_star - np.arange(lo, hi + 1)
            grad = np.zeros(n)
            grad[lo:hi + 1] = a[jj] * S - M
            gnorm = float(np.linalg.norm(grad))
            if gnorm > 1e-12:
                eta = 2.0 * S / gnorm
                for f in (0.05, 0.02, 0.005, 0.2, 0.001):
                    cand = a - (eta * f) * grad
                    cand = np.clip(cand, 0.0, 1000.0)
                    if cand.sum() > 0.01:
                        sc = fast_score(cand)
                        if sc < best_score:
                            best_score = sc
                            best = cand.copy()
                            a = cand.copy()
                            improved = True
                            break

        if not improved:
            no_improve += 1
            if no_improve >= 6:
                width = max(2, n // 20)
                start = int(rng.integers(0, n - width + 1))
                a[start:start + width] *= np.exp(rng.normal(0.0, 0.35, size=width))
                if rng.random() < 0.3:
                    a = 0.5 * (a + a[::-1])
                a = np.clip(a, 0.0, 1000.0)
                sc = fast_score(a)
                if sc < best_score:
                    best_score = sc
                    best = a.copy()
                no_improve = 0
        else:
            no_improve = 0

    final = float(evaluate_sequence([float(x) for x in best]))
    if not np.isfinite(final):
        return [1.0] * 128
    return [float(x) for x in best]
