def proj(x):
    x = np.asarray(x, dtype=np.float64)
    u = np.sort(x)[::-1]
    css = np.cumsum(u)
    ar = np.arange(1, n+1, dtype=np.float64)

    r = idx[-1]
    theta = (css[r] - 1.0)/(r + 1.0)
    out = x - theta
    out[out < 0] = 0.0

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    def norm1(x):
        a = np.asarray(x, dtype=np.float64).copy()
        a[~np.isfinite(a)] = 0.0
        a[a < 0.0] = 0.0
        s = float(a.sum())
        if not np.isfinite(s) or s <= 1e-300:
            return np.full(n, 1.0 / float(n), dtype=np.float64)
        a /= s
        return a

    def proj_simplex(x):
        a = np.asarray(x, dtype=np.float64).copy()
        a[~np.isfinite(a)] = 0.0
        u = np.sort(a)[::-1]
        css = np.cumsum(u)
        idx = np.nonzero(u * ar > css - 1.0)[0]
        if idx.size == 0:
            return np.full(n, 1.0 / float(n), dtype=np.float64)
        r = idx[-1]
        theta = (css[r] - 1.0) / float(r + 1)
        out = a - theta
        out[out < 0.0] = 0.0
        return out

    def fft_conv(v):
        F = np.fft.fft(v, N)
        c = np.fft.ifft(F * F).real[:L]
        return F, c

    def quick_acc(v):
        _, c = fft_conv(v)
        m = float(np.max(c))

            return 0.0
        return float(2.0 * n * m)

        x[x < 0.0] = 0.0
        s = float(x.sum())

        cap = max(float(cap), 1.0 / float(L))

            s = float(x.sum())
            deficit = 1.0 - s
            if abs(deficit) < 1e-14:
                break
            if deficit > 0.0:
                mask = x < cap - 1e-15
                if not mask.any():
                    return np.full(L, 1.0 / float(L), dtype=np.float64)
                add = deficit / float(mask.sum())
                x[mask] += add
                x[x > cap] = cap
            else:

                if not mask.any():
                    return np.full(L, 1.0 / float(L), dtype=np.float64)
                red = -deficit / float(mask.sum())
                x[mask] -= red
                x[x < 0.0] = 0.0
        s = float(x.sum())
        if s > 0.0:

        return x
