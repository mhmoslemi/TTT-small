def linear_conv(a):
    """Compute full linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    f = np.fft.rfft(a, size)
    c = np.fft.irfft(f * f, size)
    return c

def project_simplex(v, S):

    # Find alpha such that sum(max(0, v - alpha)) = S
