        dur = time.monotonic() - ta
        exact_time += dur
        next_eval = now + max(4.0, min(25.0, 2.0 * dur))
        arr = np.asarray(a, dtype=np.float64).copy()
        if best["a"] is None:
            best["a"] = arr
        if np.isfinite(sc) and float(sc) < best["score"]:
            best["score"] = float(sc)
            best["a"] = arr

    def normalize(a):
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if not np.isfinite(s) or s <= 1e-12:
            return np.full(n, 1.0 / n)
        a = a / s
        if not np.all(np.isfinite(a)):
            return np.full(n, 1.0 / n)
        return a

    def logJ(b, alpha):
        if alpha <= 1e-12:
            s = float(b.sum())
            return math.log(max(s, 1e-300))
        p = alpha + 1.0
        if p < 32.0:
            with np.errstate(over="ignore", under="ignore", invalid="ignore"):
                val = float(np.sum(np.power(b, p)))
            if val <= 0.0 or not np.isfinite(val):
                return -1e300
            return math.log(val) - math.log(p)
        maxb = float(b.max())
        if maxb <= 0.0 or not np.isfinite(maxb):
            return -1e300
        q = b / maxb
        tiny = np.finfo(np.float64).tiny
        with np.errstate(divide="ignore", over="ignore", under="ignore", invalid="ignore"):
            logq = np.log(np.maximum(q, tiny))
            shifted = p * logq
            m = float(shifted.max())
            s = float(np.sum(np.exp(shifted - m)))
            if s <= 0.0 or not np.isfinite(s):
                return p * math.log(maxb) - math.log(p)
            return float(p * math.log(maxb) + m + math.log(s) - math.log(p))

    alpha0_step = max(3.0, min(20.0, total / 75.0))

    def optimize(a0, end):
        a = normalize(a0)
        vel = np.zeros(n, dtype=np.float64)
        step = 0.15
        mom = 0.70
        alpha = 1.0
        last_alpha = time.monotonic()
        alpha_span = alpha0_step
        best_surr = conv_max(a)
        best_local = a.copy()
        no_accept = 0
        worse = 0
        it = 0

        while time.monotonic() < end:
            ss = float(a.sum())
            if not np.isfinite(ss) or ss <= 1e-12:
                a = normalize(best_local)
                vel *= 0.0
                continue

            A = _fft.rfft(a, N)
            b = _fft.irfft(A * A, N)[:L]
            np.maximum(b, 0.0, out=b)
            cur = float(b.max())
            if not np.isfinite(cur):
                a = normalize(best_local)
                break

            if cur < best_surr - 1e-16:
                best_surr = cur
                best_local = a.copy()
                no_accept = 0

            if cur <= best_surr * (1.0 + 1e-6):
                worse = 0
            else:
                worse += 1
                if worse > 30:
                    a = best_local.copy()
                    vel *= 0.1
                    step = max(0.02, 0.5 * step)
                    worse = 0
                    no_accept = 0

            try_update(a, False)

            now = time.monotonic()
            if now - last_alpha > alpha_span and alpha < 1024.0:
                alpha = min(2.0 * alpha, 1024.0)
                last_alpha = now
                alpha_span = min(80.0, 1.25 * alpha_span)
                step = max(0.02, 0.75 * step)
                vel *= 0.3

            if alpha < 1.0:
                w = np.full(L, 1.0 / L, dtype=np.float64)
            else:
                w = b / cur
                with np.errstate(over="ignore", under="ignore", invalid="ignore"):
                    w = np.power(w, alpha)
                s = float(w.sum())
                if not np.isfinite(s) or s <= 0.0:
                    idx = int(np.argmax(b))
                    w = np.zeros(L, dtype=np.float64)
                    w[idx] = 1.0
                else:
                    w /= s

            Ar = _fft.rfft(a[::-1], N)
            W = _fft.rfft(w, N)
            gfull = _fft.irfft(Ar * W, N)
            g = gfull[n - 1:n - 1 + n].copy()
            g -= g.mean()
            denom = float(np.mean(np.abs(g)))
            if not np.isfinite(denom) or denom <= 1e-300:
                base = best_local if best_local is not None else a
                a = normalize(base * np.exp(rng.normal(0.0, 0.25, n)))
                vel = np.zeros(n, dtype=np.float64)
                continue

            mean_a = 1.0 / n
            direction = g * (mean_a / denom)
            vel = mom * vel + (1.0 - mom) * direction
            vdenom = float(np.mean(np.abs(vel)))
            if not np.isfinite(vdenom) or vdenom <= 1e-300:
                search_dir = direction
            else:
                search_dir = vel * (mean_a / vdenom)

            accepted = False
            cur_logJ = None
            for mult in (1.0, 0.5, 0.25):
                trial = a - step * mult * search_dir
                np.clip(trial, 0.0, 1000.0, out=trial)
                s = float(trial.sum())
                if not np.isfinite(s) or s <= 1e-12:
                    continue
                trial = trial / s

                At = _fft.rfft(trial, N)
                bt = _fft.irfft(At * At, N)[:L]
                np.maximum(bt, 0.0, out=bt)
                tmax = float(bt.max())
                if not np.isfinite(tmax):
                    continue

                if tmax <= cur * (1.0 + 1e-10) + 1e-16:
                    accept = True
                elif tmax <= cur * 1.05:
                    if cur_logJ is None:
                        cur_logJ = logJ(b, alpha)
                    tlog = logJ(bt, alpha)
                    accept = np.isfinite(tlog) and tlog <= cur_logJ + 1e-9
                else:
                    accept = False

                if accept:
                    a = trial
                    step = min(0.5, step * (1.15 if mult == 1.0 else 1.0))
                    accepted = True
                    break

            if not accepted:
                step = max(0.005, 0.5 * step)
                vel *= 0.5
                no_accept += 1
                if no_accept >= 25:
                    sigma = 0.12 + 0.05 * min(4, int(no_accept // 25))
                    base = best_local if best_local is not None else a
                    a = normalize(base * np.exp(rng.normal(0.0, sigma, n)))
                    vel = np.zeros(n, dtype=np.float64)
                    step = max(0.05, step)
                    no_accept = 0

            it += 1
            if it % 2048 == 0:
                sym = 0.5 * (a + a[::-1])
                s = float(sym.sum())
                if np.isfinite(s) and s > 1e-12:
                    sym = sym / s
                    St = _fft.rfft(sym, N)
                    bt = _fft.irfft(St * St, N)[:L]
                    np.maximum(bt, 0.0, out=bt)
                    tmax = float(bt.max())
                    if np.isfinite(tmax) and tmax < cur * 0.999:
                        a = sym
                        vel *= 0.0
                        step = max(0.05, 0.8 * step)

        try_update(best_local, True)
        return best_local, best_surr

    try_update(inits[0], True)

    if time.monotonic() > deadline - 5.0:
        pass
    else:
        rem = len(inits)
        for z in inits:
            rem -= 1
            if time.monotonic() >= deadline - 5.0:
                break
            span = max(8.0, (deadline - time.monotonic()) / max(1, rem + 1))
            end = min(deadline - 2.0, time.monotonic() + span)
            optimize(z, end)

    if time.monotonic() < deadline - 25.0:
        start = best["a"] if best["a"] is not None else inits[0]
        optimize(start, deadline - 5.0)

    def clean(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size == 0:
            a = np.ones(16, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        if not np.all(np.isfinite(a)):
            a = np.full(16, 1.0 / 16, dtype=np.float64)
        s = a.sum()
        if not np.isfinite(s) or s < 0.01:
            a = np.full(16, 1.0 / 16, dtype=np.float64)
        else:
            a = a / s
        return a

    candidate = clean(best["a"] if best["a"] is not None else inits[0])
    if time.monotonic() < deadline - 0.5:
        try_update(candidate, True)
    else:
        if best["a"] is None:
            best["a"] = candidate
            best["score"] = math.inf

    out = clean(best["a"] if best["a"] is not None else inits[0])
    return [float(v) for v in out]
