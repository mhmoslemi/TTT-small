    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256)
    n = parent.size
    S = np.sqrt(2.0 * n)
    a = parent / parent.sum() * S if parent.sum() > 1e-10 else np.full(n, S / n)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    bounds = [(0.0, 1000.0)] * n

    # Phase 1: L-BFGS on scale-invariant p-power surrogate
    for p in (8, 16, 32):
        if time.monotonic() > deadline - 120:
            break
        def f_obj(x, p=p):
            c = np.convolve(x, x)
            sn = x.sum()
            if sn < 1e-10:
                return 1e10
            return np.sum(c ** p) ** (1.0 / p) / sn ** 2
        def f_grad(x, p=p):
            c = np.convolve(x, x)
            sn = x.sum()
            if sn < 1e-10:
                return np.ones_like(x)
            corr = np.correlate(c ** (p - 1), x, mode='valid')
            P = np.sum(c ** p)
            if P < 1e-30:
                return np.zeros_like(x)
            Q = P ** (1.0 / p)
            return (2.0 * Q / sn ** 2) * (corr / P - 1.0 / sn)
        try:
            r = minimize(f_obj, a, jac=f_grad, method='L-BFGS-B', bounds=bounds,
                        options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-8, 'maxls': 50})
            a_try = np.clip(r.x, 0, 1000)
            sc = float(evaluate_sequence(a_try.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best, best_score, a = a_try.copy(), sc, a_try
        except Exception:
            pass

    # Phase 2: Frank-Wolfe iteration with LP subproblem
    if time.monotonic() < deadline - 60:
        off = list(range(n))
        for it in range(120):
            if time.monotonic() > deadline - 60:
                break
            c_ = np.convolve(a, a)
            M = float(c_.max())
            try:
                A = sparse.diags([np.full(n, a[d]) for d in range(n)], off,
                                shape=(2 * n - 1, n), format='csc')
                r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                           bounds=[(0, None)] * n, method='highs',
                           options={'time_limit': 25})
            except Exception:
                break
            if not r.success or r.x is None:
                break
            g = r.x
            sg = g.sum()
            if sg < 1e-10:
                break
            g_sc = g / sg * S
            improved = False
            for t in (0.4, 0.25, 0.15, 0.08, 0.04):
                at = np.clip((1.0 - t) * a + t * g_sc, 0, 1000)
                sc = float(evaluate_sequence(at.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best, best_score, a = at.copy(), sc, at
                    improved = True
                    break
            if not improved and it > 15:
                break

    # Phase 3: Stochastic local refinement
    rng = np.random.default_rng(seed)
    ts = 0.04
    while time.monotonic() < deadline - 3:
        at = np.clip(best * np.exp(rng.normal(0.0, ts, n)), 0, 1000)
        sc = float(evaluate_sequence(at.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best, best_score = at.copy(), sc
            ts = max(0.003, ts * 0.97)
        else:
            ts = max(0.003, ts * 0.998)

    # Final L-BFGS polish if time remains
    if time.monotonic() < deadline - 15:
        p = 32
        def f_obj2(x):
            c = np.convolve(x, x)
            sn = x.sum()
            if sn < 1e-10: return 1e10
            return np.sum(c ** p) ** (1.0 / p) / sn ** 2
        def f_grad2(x):
            c = np.convolve(x, x)
            sn = x.sum()
            corr = np.correlate(c ** (p - 1), x, mode='valid')
            P = np.sum(c ** p)
            if sn < 1e-10 or P < 1e-30: return np.zeros_like(x)
            Q = P ** (1.0 / p)
            return (2.0 * Q / sn ** 2) * (corr / P - 1.0 / sn)
        try:
            r = minimize(f_obj2, best, jac=f_grad2, method='L-BFGS-B', bounds=bounds,
                        options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-8})
            at = np.clip(r.x, 0, 1000)
            sc = float(evaluate_sequence(at.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = at.copy()
                best_score = sc
        except Exception:
            pass

    out = [float(v) for v in best]
    if any(not np.isfinite(v) for v in out):
        out = [1.0] * n
    return out
