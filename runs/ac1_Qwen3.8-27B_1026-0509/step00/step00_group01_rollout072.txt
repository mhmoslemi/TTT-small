import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
from scipy.signal import fftconvolve


def _build_A(f, S_idx, n):
    rows, cols, vals = [], [], []
    for k in S_idx:
        k = int(k)
        j0, j1 = max(0, k - n + 1), min(n - 1, k)
        if j0 > j1:
            continue
        js = np.arange(j0, j1 + 1)
        rows.append(np.full(js.shape, k))
        cols.append(js)
        vals.append(f[k - js])
    if rows:
        rows = np.concatenate(rows)
        cols = np.concatenate(cols)
        vals = np.concatenate(vals)
        return csr_matrix((vals, (rows, cols)), shape=(len(S_idx), n))
    return csr_matrix((len(S_idx), n))


def _lp_solve(f, M, S_idx, deadline):
    n = f.size
    A = _build_A(f, S_idx, n)
    c = -np.ones(n)
    b = np.full(A.shape[0], M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, 1000.0)] * n,
                      method='highs', options={'presolve': True})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    g = np.asarray(res.x, dtype=float)
    gs = g.sum()
    if gs < 1e-9:
        return None
    return g / gs


def _cplp(f, M, deadline, cap=1200):
    n = f.size
    ctr = n - 1
    w = max(1, n // 250)
    S_idx = np.unique(np.concatenate([
        np.arange(max(0, ctr - w), min(2 * n - 1, ctr + w + 1)),
        np.array([0, 2 * n - 2])
    ]))
    g = None
    for _ in range(30):
        if time.monotonic() > deadline:
            break
        g = _lp_solve(f, M, S_idx, deadline)
        if g is None:
            break
        conv = fftconvolve(f, g)
        viol = np.where(conv > M + 1e-9 * max(M, 1.0))[0]
        if viol.size == 0:
            break
        S_idx = np.unique(np.concatenate([S_idx, viol]))
        if S_idx.size >= cap:
            break
    return g
