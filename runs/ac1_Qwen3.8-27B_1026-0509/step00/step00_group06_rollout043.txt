            g[lo:hi+1] = a[k_star - hi : k_star - lo + 1][::-1]
            # Actually: g[i] = a[k_star - i] for valid i

            direction = -g
            direction -= direction.mean()
            
            # Learning rate (decreasing)
            lr = 0.05 / (1.0 + it * 0.001)
            
            candidate = a + lr * direction
            np.clip(candidate, 0, None, out=candidate)
            cs = candidate.sum()
            if cs > 0:
                candidate /= cs
            
            c_score = score(candidate)
            if c_score < a_best_score:
                a_best = candidate.copy()
                a_best_score = c_score
            
            a = candidate  # move regardless (to explore)
        
        # Check if this initialization beat the global best
        if a_best_score < best_score:
            best = a_best
            best_score = a_best_score
    
    # Final validation
    best_score_final = float(evaluate_sequence(best.tolist()))
    if np.isfinite(best_score_final):
        best_score = best_score_final
    
    return [float(x) for x in best]
