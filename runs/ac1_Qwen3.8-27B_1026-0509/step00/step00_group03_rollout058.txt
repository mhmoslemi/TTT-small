            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_min <= i_max:
                idx = np.arange(i_min, i_max + 1)
                row[idx] = a[k - idx]
            A_list.append(row)
        A_lp = np.array(A_list)
        
        # Solve LP
        try:
            c_obj = -np.ones(n)
            res = linprog(c_obj, A_ub=A_lp, b_ub=np.full(len(active), M),
                         bounds=[(0, 1000)] * n, method='highs')
        except Exception:
            break
        
        if not res.success:
            # Try smaller active set
            active = np.argsort(-c_conv)[:5]
            A_list = []
            for k in active:
                row = np.zeros(n)
                i_min = max(0, k - n + 1)
                i_max = min(k, n - 1)
                if i_min <= i_max:
                    idx = np.arange(i_min, i_max + 1)
                    row[idx] = a[k - idx]
                A_list.append(row)
            A_lp = np.array(A_list)
            try:
                res = linprog(-np.ones(n), A_ub=A_lp, b_ub=np.full(len(active), M),
                             bounds=[(0, 1000)] * n, method='highs')
            except:
                break
            if not res.success:
                break
        
        b_lp = res.x
        b_sum = b_lp.sum()
        if b_sum <= 0:
            break
        
        # Normalize b_lp to same sum as a
        b_lp = b_lp * (s / b_sum)
        b_lp = np.clip(b_lp, 0, 1000)
        
        # Update
        a_new = (1 - t_step) * a + t_step * b_lp
        
        new_score = get_score(a_new)
        if np.isfinite(new_score) and new_score < best_score:
            a = a_new
            best_a = a_new.copy()
            best_score = new_score
            if time.monotonic() > deadline - 1:
                break
        else:
            t_step = max(0.05, t_step * 0.6)
            # Occasionally randomize
            if rng.random() < 0.1:
                t_step = float(rng.uniform(0.1, 0.5))
    
    # Final check
    final = [float(x) for x in best_a]
    fs = float(evaluate_sequence(final))
    if not np.isfinite(fs):
        # Fallback
        final = [1.0] * n
        fs = evaluate_sequence(final)
    
    return final
