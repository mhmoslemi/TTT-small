import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    p = 2.0 * p / (n * (n + 1))

    scale = 50.0
    a = p * scale
    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    
    def score_normalized(p_vec):
        P = np.fft.rfft(p_vec, fft_size)
        c = np.fft.irfft(P * P, fft_size)[:2*n-1]
        return c
    
    # Project onto probability simplex
    def simplex_project(v):
        v = v.copy()
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)
        return np.maximum(v - theta, 0)
    
    # Gradient refinement
    alpha = 0.05
    for it in range(500000):
        if time.monotonic() > deadline - 10.0:
            break
        if time.monotonic() > deadline - 5.0 and it % 1000 != 0:
            break
        
        c = score_normalized(p)
        M = c.max()
        k_star = int(c.argmax())
        
        # Subgradient of max c_k w.r.t. p
        # d c[k_star] / d p[i] = 2 * p[k_star - i] (if 0 <= k_star - i < n)
        # We want to DECREASE M, so gradient step is p -= alpha * g
        g = np.zeros(n)
        # Efficient: g[i] = 2*p[k_star-i] for valid i
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            g[i] = 2.0 * p[k_star - i]

        top3 = np.argsort(c)[-3:]
        g2 = np.zeros(n)
        for k_top in top3:

            lo2 = max(0, k_top - (n - 1))
            hi2 = min(n - 1, k_top)

            else:
                alpha *= 0.98

            alpha *= 0.9
        
        # Periodic exact evaluation
