    # Returns sparse matrix where A[k,j] = f[j] if valid
    rows = np.zeros(n*n, dtype=np.int32)
    cols = np.zeros(n*n, dtype=np.int32)

    def conv(a, b):
        # Use FFT for speed
        n_a, n_b = len(a), len(b)
        n_fft = 1
        while n_fft < n_a + n_b - 1:
            n_fft *= 2
        fa = np.fft.rfft(a, n_fft)
        fb = np.fft.rfft(b, n_fft)
        conv = np.fft.irfft(fa * fb, n_fft)
        return conv[:n_a + n_b - 1]
