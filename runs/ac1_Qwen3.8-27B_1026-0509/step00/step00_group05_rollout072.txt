import time
import numpy as np


def _proj_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    if v.min() < 0.0:
        v = v - v.min()
    u = np.sort(v)[::-1]
    cs = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    rho = np.nonzero(u * idx > (cs - S))[0]
    if rho.size == 0:
        theta = 0.0
        m = 0
    else:
        rho = rho[-1]
        m = rho + 1
        theta = (cs[rho] - S) / m
    out = np.maximum(v - theta, 0.0)
    out *= (S / np.sum(out)) if np.sum(out) > 0 else 1.0
    return out
