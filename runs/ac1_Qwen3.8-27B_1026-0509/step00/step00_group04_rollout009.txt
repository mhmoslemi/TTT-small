```python
def peak_reduction(h, n, alpha=0.1):
    conv = np.convolve(h, h, mode='full')
    t_star = int(np.argmax(conv))
    peak_val = conv[t_star]
    avg_val = conv.mean()
    
    if peak_val <= avg_val * 1.01:
        return h  # Already flat
    
    # Positions contributing to the peak: j where h[j]*h[t_star-j] is large
    contrib = np.zeros(n)
    j_lo = max(0, t_star - n + 1)
    j_hi = min(n - 1, t_star)
    for j in range(j_lo, j_hi + 1):
        partner = t_star - j
        if 0 <= partner < n:
            contrib[j] += h[j] * h[partner]
    
    total_contrib = contrib.sum()
    if total_contrib < 1e-15:
        return h
    
    # Reduce contributions proportionally
    reduction = alpha * (peak_val - avg_val) / peak_val
    h_red = h * (1 - reduction * contrib / (h * contrib).where(contrib > 0, 1))
    # This is getting messy. Let me just do:
    h_red = h.copy()
    mask = contrib > 0
    h_red[mask] *= (1 - alpha * 0.5)
    h_red[~mask] *= (1 + alpha * 0.5 * mask.sum() / max(1, (~mask).sum()))
    h_red = np.maximum(h_red, 0)
    h_red /= h_red.sum()
    return h_red
```
