import numpy as np
import time
from scipy.fft import fft, ifft

def _convolve_fft(a, b):
    """Full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()  # Next power of 2
    A = fft(a, size)
    B = fft(b, size)
    return ifft(A * B).real[:2*len(a)-1]

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000).copy()
    n = len(a)
    S = a.sum()
    a = a / S  # Normalize so sum = 1
    
    best = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    
    # Break symmetry with perturbation
    pert = rng.normal(0, 0.1, n)
    a = np.clip(a + pert, 0, None)
    a /= a.sum()
    
    # Subgradient descent
    lr = 0.1
    no_improve = 0
    max_no_improve = 200
    
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        b = np.convolve(a, a)  # Use direct for correctness
        M = b.max()
        active = np.where(b >= M - 1e-12)[0]
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            idx = np.arange(lo, hi)
            g[lo:hi] += a[k - idx]
        g /= len(active)
        
        # Center (constraint sum(a)=1)
        g -= g.mean()
        
        # Multiplicative update
        a_new = a * np.exp(-lr * g)
        a_new = np.clip(a_new, 1e-15, None)
        a_new /= a_new.sum()
        
        # Evaluate
        score = float(evaluate_sequence(a_new.tolist()))
        if score < best_score:
            best = a_new.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        
        # Re-randomize if stuck
        if no_improve > max_no_improve:
            a = best + rng.normal(0, 0.05, n)
            a = np.clip(a, 0, None)
            a /= a.sum()
            lr = min(lr * 1.5, 1.0)
            no_improve = 0
        else:
            a = a_new
        
        # Decay learning rate
        if it % 100 == 0:
            lr *= 0.95
    
    return [float(x) for x in best]
