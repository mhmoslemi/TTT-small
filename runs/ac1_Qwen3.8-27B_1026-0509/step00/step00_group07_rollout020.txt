```python
import numpy as np
import time

def fw_step(a, n):
    """One Frank-Wolfe step. Returns new a."""
    # Compute convolution
    conv = np.convolve(a, a, mode='full')
    j_star = np.argmax(conv)
    t_current = conv[j_star]
    
    # Linear objective: min sum_i g_i * (S_{j_star} a)_i subject to g in simplex
    # The FW direction is: min_{e_i} (S_{j*} a)_i
    # (S_j a)_i = a[j-i] if 0 <= j-i < n, else 0
    
    v = np.zeros(n)
    i_min = max(0, j_star - n + 1)
    i_max = min(n - 1, j_star)
    for i in range(i_min, i_max + 1):
        v[i] = a[j_star - i]
    
    # The FW vertex is e_{argmin(v)}
    idx = np.argmin(v)
    
    # Line search: minimize f(alpha) = max_j [(1-alpha)^2 C_j + 2*alpha*(1-alpha) D_j + alpha^2 E_j]
    # where C = conv(a,a), D = conv(a, e_idx), E = conv(e_idx, e_idx)
    # D_j = a[j - idx] if 0 <= j-idx < n, else 0
    # E is 1 at j = 2*idx, 0 elsewhere
    
    D = np.zeros(2*n - 1)
    D[idx:idx+n] = a  # D[j] = a[j-idx] for j in [idx, idx+n-1]
    E = np.zeros(2*n - 1)
    E[2*idx] = 1.0
    
    # f(alpha) = max_j [(1-alpha)^2 C_j + 2*alpha*(1-alpha) D_j + alpha^2 E_j]
    # For line search, evaluate at a few points
    
    def obj(alpha):
        return np.max((1-alpha)**2 * conv + 2*alpha*(1-alpha) * D + alpha**2 * E)
    
    # Golden section or just try a few points
    best_alpha = 0.0
    best_val = obj(0.0)
    for alpha in np.linspace(0, 1, 20):
        val = obj(alpha)
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    return (1 - best_alpha) * a + best_alpha * np.eye(1, n, idx)[0]  # (1-alpha)*a + alpha*e_idx
```
