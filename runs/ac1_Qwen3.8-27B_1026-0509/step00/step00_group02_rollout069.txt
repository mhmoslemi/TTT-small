        norm = float(f.sum())
        f = f / norm * np.sqrt(2.0 * n)

    # FFT size: small power of two above 2n-1, avoiding huge memory spikes.
    size = 1
    while size < 2 * n:
        size <<= 1
    if size > 1 << 22 and size < 1 << 23:
        size = 1 << 21

    def fft_conv(a, b=None):
        if b is None:
            Fa = np.fft.rfft(a, size)
            return np.fft.irfft(Fa * Fa, size)[:2 * n - 1]
        return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:2 * n - 1]

    def quick_score(x):
        C = fft_conv(x)
        s = x.sum()
        if s <= 0.01:
            return np.inf
        return float(2.0 * n * C.max() / (s * s))

    def exact_score(x):
        try:
            return float(evaluate_sequence(x.tolist()))
        except Exception:
            return np.inf

    def grad_peak(x, k):
        g = np.zeros(n)
        a = max(0, k - n + 1)
        b = min(n - 1, k)
        if a <= b:
            g[a:b + 1] += x[k - b:k - a + 1][::-1]
        a2 = max(0, k)
        b2 = min(n - 1, k + n - 1)
        if a2 <= b2:
            g[a2:b2 + 1] += x[a2 - k:b2 - k + 1]
        return g

    def feasible_save(x):
        nonlocal best, best_score
        s = exact_score(x)
        if np.isfinite(s) and s < best_score:
            best = x.copy()
            best_score = s if np.isfinite(s) else best_score
