        X = np.fft.rfft(x, size)
        B = np.fft.irfft(X * X, size)[:2*n-1]
        
        # Compute softmax
        maxB = B.max()
        exp_aB = np.exp(alpha * (B - maxB))
        p = exp_aB / exp_aB.sum()
        
        # Compute gradient of log-sum-exp
        p_rev = p[::-1]
        PX = np.fft.rfft(p_rev, size) * np.fft.rfft(x, size)
        px = np.fft.irfft(PX, size)
        cross_corr = px[n-1:2*n-1][::-1]
        
        S = x.sum()
        G = 2 * alpha * cross_corr  # gradient of log-sum-exp w.r.t. x
        
        # Gradient of F = 2n * L / S^2 where L = (1/alpha) * log_sum_exp
        L = (1.0/alpha) * (np.log(exp_aB.sum()) + alpha * maxB)
        # dF/dx_m = (2n/S^2) * (G_m - 2L/S)
        grad_F = (2.0 * n / S**2) * (G - 2.0 * L / S)
        
        # Gradient step
        x_new = x - lr * grad_F
        x_new = np.clip(x_new, 0.0, 1000.0)
        
        # Evaluate
        score = evaluate_sequence(x_new.tolist())
        if np.isfinite(score) and score < best_score:

            best_score = score
            lr = min(lr * 1.05, 10.0)  # increase step size on success
        else:
            lr = max(lr * 0.7, 0.01)  # decrease on failure
            x_new = x  # revert
        
        x = x_new
    
    return best.tolist()
