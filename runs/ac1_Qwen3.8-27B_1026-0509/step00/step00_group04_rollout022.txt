import numpy as np
import time
from scipy.sparse import coo_matrix
import cvxpy as cp

def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)(t) = (f*g)(t) for g in R^n."""
    n = len(f)
    rows, cols, data = [], [], []
    for t in range(2 * n - 1):
        k_min = max(0, t - n + 1)
        k_max = min(t, n - 1)
        for k in range(k_min, k_max + 1):
            rows.append(t)
            cols.append(k)
            data.append(f[t - k])
    A = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_lp(f, M, solver=None):
    """Solve max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    g = cp.Variable(n)
    constraints = [A @ g <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    for s in [solver, 'CLARABEL', 'SCS', 'GLPK', 'CBC', 'GLPK_MI']:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate') and g.value is not None:
                return np.maximum(np.asarray(g.value, dtype=np.float64), 0)
        except Exception:
            continue
    return None
