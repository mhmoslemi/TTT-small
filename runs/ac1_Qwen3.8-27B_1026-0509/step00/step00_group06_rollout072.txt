def _score_internal(arr):
    a = np.clip(np.maximum(arr, 0.0), 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * a.size * np.convolve(a, a).max() / (s * s))
