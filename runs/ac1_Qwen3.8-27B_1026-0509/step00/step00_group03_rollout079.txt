import numpy as np
import time

def _proj_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Find tau such that sum(max(v - tau, 0)) = 1
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u - (cssv - 1) / rho > 0
    if not cond.any():
        return np.ones(n) / n
    rho_hat = rho[cond][-1]
    tau = (cssv[cond][-1] - 1) / rho_hat
    return np.maximum(v - tau, 0)

def _convolve_max(a):
    """Compute max of full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)[:2*n-1]
    return np.max(b), np.argmax(b)
