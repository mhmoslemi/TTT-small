def smooth_grad(a, b, t=10.0):
    """Compute gradient of softmax-weighted objective."""
    # b = convolve(a, a), length conv_len
    # w = softmax(t*b), length conv_len
    log_w = t * b - t * b.max()  # numerical stability
    w = np.exp(log_w)
    w /= w.sum()

    # This is a cross-correlation
    # Let's compute it via FFT
    # g[i] = 2 * sum_k w[k] * a[k-i]
    # = 2 * (w * a)[i] where * is correlation
    # In terms of convolution: g = 2 * convolve(w, a[::-1]) with proper alignment

    # of the generating functions... Let me just use direct FFT.
    
    # Define: g[i] = 2 * sum_{k} w[k] * a[k-i] for 0<=k<len(w), 0<=k-i<n
    # This is 2 * (w ⊛ a)[i] where ⊛ is cross-correlation
    # Cross-correlation: (w ⊛ a)[i] = sum_k w[k] * a[k-i]
    # = convolve(w, a_flipped) at appropriate index
    # a_flipped = a[::-1]
    # (w * a_flipped)[n-1+i] = sum_k w[k] * a_flipped[n-1+i-k] = sum_k w[k] * a[k-i]
    # So g/2 = convolve(w, a_flipped)[n-1 : n-1+n]

import time

    def do_conv(a):
        F = np.fft.rfft(a, n=fft_len)
        return np.fft.irfft(F * F, n=fft_len)[:conv_len]

    def project(a):
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return np.full(n, 1.0 / n)
        return a * (1.0 / s)

    best_a = None
    best_score = np.inf

        nonlocal best_a, best_score
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:

        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Initializations
    inits = []
    try:
        p = np.clip(parent, 0, 1000).astype(np.float64)
        if p.sum() > 0:
            inits.append(p / p.sum())
    except Exception:
        pass
    j = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)
    hanning = np.sin(np.pi * j / (n - 1)) ** 2
    inits.append(hanning / hanning.sum())
    mid = (n - 1) / 2.0
    for sig in (0.2, 0.3, 0.4, 0.5):
        g = np.exp(-0.5 * ((j - mid) / (sig * n)) ** 2)
        inits.append(g / g.sum())
    # Skewed (exponential decay from one end)
    for base in (0.015, 0.02, 0.03):
        sk = base ** j
        inits.append(sk / sk.sum())

        # Top-K peaks with softmax weighting
        top_idx = np.argpartition(b, -K)[-K:]
        top_vals = b[top_idx].copy()
        order = np.argsort(top_vals)[::-1]
