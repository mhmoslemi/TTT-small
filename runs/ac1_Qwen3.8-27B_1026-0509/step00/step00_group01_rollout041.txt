```python
import numpy as np
import scipy.sparse as sp
import cvxpy as cp
import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)) - 15.0
    n = 1024

    best_f = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best_f.tolist()))

    # Precompute Toeplitz sparsity pattern
    rows_list, cols_list = [], []
    for k in range(2 * n - 1):
        lo, hi = max(0, k - n + 1), min(n - 1, k)
        rows_list.append(np.full(hi - lo + 1, k, dtype=np.int32))
        cols_list.append(np.arange(lo, hi + 1, dtype=np.int32))
    T_rows = np.concatenate(rows_list)
    T_cols = np.concatenate(cols_list)

    def build_T(f):
        return sp.csr_matrix((f[T_rows - T_cols], (T_rows, T_cols)), shape=(2 * n - 1, n))

    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))

    for restart in range(300):
        if time.monotonic() > deadline:
            break

        # Non-symmetric random initialization
        f = 1.0 + 0.5 * rng.random(n)
        score = float(evaluate_sequence(f.tolist()))
        if score < best_score:
            best_score, best_f = score, f.copy()

        stall = 0
        for it in range(200):
            if time.monotonic() > deadline - 5:
                break

            M = float(np.convolve(f, f).max())
            T = build_T(f)
            prob = cp.Problem(objective, [T @ g <= M, g >= 0])
            try:
                prob.solve(solver=cp.SCS, max_iters=10000)
            except Exception:
                break

            if g.value is None:
                break

            gv = np.maximum(g.value, 0)

            # Check if LP gives a non-trivial direction
            if np.linalg.norm(gv - f) < 1e-6 * np.linalg.norm(f):
                stall += 1
                if stall > 5:
                    break
                f = np.maximum(f + 0.1 * rng.standard_normal(n), 0.01)
                continue

            stall = 0
            improved = False
            for t in [0.5, 0.25, 0.1, 0.05, 0.01]:
                fn = (1 - t) * f + t * gv
                fn = np.maximum(fn, 0.0)
                s = float(evaluate_sequence(fn.tolist()))
                if np.isfinite(s) and s < score:
                    f, score = fn, s
                    if score < best_score:
                        best_score, best_f = score, f.copy()
                    improved = True
                    break

            if not improved:
                stall += 1
                if stall > 10:
                    break
                f = np.maximum(f * (1.0 + 0.08 * rng.standard_normal(n)), 0.01)

        # Periodic local refinement
        if time.monotonic() < deadline - 30 and best_score < 1.95:
            f = best_f.copy()
            score = best_score
            for _ in range(2000):
                if time.monotonic() > deadline - 10:
                    break
                kstar = int(np.convolve(f, f).argmax())
                lo, hi = max(0, kstar - n + 1), min(n - 1, kstar)
                grad = np.zeros(n)
                grad[lo:hi + 1] = f[kstar - np.arange(lo, hi + 1)]
                for step in [0.05, 0.02, 0.01, 0.005]:
                    fn = f - step * grad
                    fn = np.maximum(fn, 0.0)
                    if fn.sum() < 1e-6:
                        continue
                    s = float(evaluate_sequence(fn.tolist()))
                    if np.isfinite(s) and s < score:
                        f, score = fn, s
                        if score < best_score:
                            best_score, best_f = score, f.copy()
                        break

    return [float(x) for x in best_f]
```
