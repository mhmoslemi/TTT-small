```python
def solve_lp_cutting_plane(a, M, max_iter=50):
    n = len(a)
    # Start with center constraints
    active = list(range(n//2 - 5, n//2 + 5))  # around center
    
    for _ in range(max_iter):
        # Build reduced constraint matrix
        rows = []
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            rows.append((k, j_start, j_end))
        
        # Use linprog
        from scipy.optimize import linprog
        c_vec = -np.ones(n)
        A_ub = np.zeros((len(active), n))
        for idx, (k, js, je) in enumerate(rows):
            A_ub[idx, js:je+1] = a[js:je+1]
        b_ub = np.full(len(active), M)
        bounds = [(0, 1000)] * n
        
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        
        # Check all constraints
        conv = np.convolve(a, g)
        violations = conv - M
        if np.max(violations) < 1e-6:
            return g  # All constraints satisfied
        
        # Add the most violated constraint
        worst = np.argmax(violations)
        if worst not in active:
            active.append(int(worst))
            if len(active) > n:  # Cap active set
                # Remove some constraints that are least active
                pass
    
    return g
```
