import time
from scipy.optimize import linprog
from scipy import sparse

def solve_lp(a, M):
    """Solve: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    # (a*g)_k = sum_i a[i] * g[k-i] (valid indices)
    # This is a Toeplitz matrix T where T[k,i] = a[i] if 0 <= k-i < n, else 0
    # Actually T[k,i] = a[max(0, min(n-1, k-i))]... no.
    
    # (a*g)_k = sum_{i=0}^{n-1} a[i] * g[k-i] where g[j]=0 for j<0 or j>=n
    # = sum_{i: max(0,k-n+1) <= i <= min(n-1,k)} a[i] * g[k-i]
    
    # Let's build the sparse matrix row by row
    n_con = 2*n - 1
    # Row k: (a*g)_k = sum_i T[k,i] * g[i]
    # T[k,i] = a[k-i] if 0 <= k-i <= n-1, else 0
    # i.e., T[k,i] = a[k-i] for max(0, k-(n-1)) <= i <= min(n-1, k)
    
    rows = []
    cols = []
    vals = []
    for k in range(n_con):
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k-i])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(n_con, n))
    c = -np.ones(n)  # maximize sum = minimize -sum
    res = linprog(c, A_ub=A, b_ub=np.full(n_con, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return np.ones(n) / n

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def build_conv_matrix(a):
    """Build sparse matrix T where T[k,i] gives (a*g)_k = sum_i T[k,i] g[i]."""
    n = len(a)
    n_con = 2 * n - 1
    row_idx = []
    col_idx = []
    vals = []
    for k in range(n_con):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            row_idx.append(k)
            col_idx.append(i)
            vals.append(a[k - i])
    return sparse.csr_matrix((vals, (row_idx, col_idx)), shape=(n_con, n))

    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, 1000)] * n, method='highs')
    if res.success:
        return res.x
    return None

Actually, let me try to think about this more carefully. The problem is fundamentally about making the convolution flat. For the convolution to be flat (all $f_k \approx 1/(2n-1)$), we need the polynomial $P(x) = \sum a_i x^i$ to satisfy $|P(e^{i\omega})|^2 \approx$ constant. This is the "minimum variance" or "flat top" problem.

With the constraint $a_i \geq 0$, this is very restrictive. The coefficients of $P(x)$ must all be nonnegative, which means $P$ is a "positive polynomial" on the unit circle (in the sense that all Taylor coefficients are nonnegative).

For a positive polynomial $P$ of degree $n-1$ with $P(1) = S$, what's the minimum of $\max_{\omega} |P(e^{i\omega})|^2$?

By the maximum principle for analytic functions... $P$ is a polynomial, so it's entire. On the unit circle, $|P(e^{i\omega})|$ has a maximum. By Parseval: $\frac{1}{2\pi}\int_0^{2\pi} |P(e^{i\omega})|^2 d\omega = \sum a_i^2$.

So $\max |P|^2 \geq \frac{1}{2\pi}\int |P|^2 = \sum a_i^2 \geq S^2/n$.

And $\max (a*a)_k$ is related to the peak of the IFT of $|P|^2 / (2\pi)$. Specifically:
$(a*a)_k = \frac{1}{2\pi}\int_0^{2\pi} |P(e^{i\omega})|^2 e^{ik\omega} d\omega$.

So $\max_k (a*a)_k \leq \frac{1}{2\pi}\int |P|^2 = \sum a_i^2$ (trivial bound, as before).

And $\max_k (a*a)_k \geq \max_k \text{Re}\left[\frac{1}{2\pi}\int |P|^2 e^{ik\omega} d\omega\right]$.

The average of $(a*a)_k$ over $k$ is $\frac{1}{2n-1} \sum (a*a)_k = S^2/(2n-1)$.

So $\max_k (a*a)_k \geq S^2/(2n-1)$, giving score $\geq 2n/(2n-1)$.
