def _simplex_project(v):
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (sv - 1))[0][-1]
    theta = (sv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
