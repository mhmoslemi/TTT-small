import numpy as np
import time
from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 5.0
    
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    m = 2 * n - 1
    
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    def build_conv_matrix(x):
        A = np.zeros((m, n))
        for j in range(n):
            end = min(j + n, m)
            A[j:end, j] = x[j]
        return A
    
    def solve_lp(x):
        M = np.max(np.convolve(x, x))
        A = build_conv_matrix(x)
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(m, M), 
                     bounds=[(0.0, 1000.0)]*n, method='highs')
        if res.success:
            return res.x
        return None
    
    # Main loop
    t_mix = 0.3
    no_improve = 0
    while time.monotonic() < deadline:
        g = solve_lp(best)
        if g is not None and np.sum(g) > 0:
            g = g / np.sum(g) * np.sum(best)  # rescale
            x_new = (1 - t_mix) * best + t_mix * g
            x_new = np.clip(x_new, 0.0, 1000.0)
            score = float(evaluate_sequence(x_new.tolist()))
            if np.isfinite(score) and score < best_score:
                best = x_new.copy()
                best_score = score
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 20:
            # Perturbation
            pert = 1.0 + 0.05 * rng.standard_normal(n)
            best = np.clip(best * pert, 0.0, 1000.0)
            best_score = float(evaluate_sequence(best.tolist()))
            no_improve = 0
        
        if no_improve > 5:
            t_mix = max(0.1, t_mix * 0.95)
    
    return [float(x) for x in best]

        x += 0.01 * S0 * rng.standard_normal(n)
        x = np.clip(x, 0.0, 1000.0)
        if np.sum(x) > 0:
            x *= S0 / np.sum(x)
    
    # Phase 2: Local perturbations (fast)
    while time.monotonic() < deadline:
        x = best.copy()
        # Random block perturbation
        width = int(rng.integers(n//100, n//10))
        start = int(rng.integers(0, n - width + 1))
        x[start:start+width] *= np.exp(rng.normal(0, 0.1))
        x = np.clip(x, 0, 1000)
        sc = float(evaluate_sequence(x.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = x.copy()
            best_score = sc
    
    return [float(v) for v in best]

Let me also think about the LP approach more carefully. The LP:
max sum(g)
s.t. (x*g)[k] <= M for all k
     g >= 0

where M = max(x*x).

The dual of this LP:
min M * sum(y)
s.t. A^T y >= 1 (where A is the conv matrix)
     y >= 0

The dual has 2n-1 variables and n constraints. The constraint A^T y >= 1 means: for each j, sum_k A[k,j]*y[k] >= 1, i.e., sum_k x[j]*y[k] (for valid k) >= 1, i.e., x[j] * (sum of y[k] for k in [j, j+n-1]) >= 1.

So the dual is:
min M * sum(y)
s.t. x[j] * (y[j] + y[j+1] + ... + y[j+n-1]) >= 1 for all j
     y >= 0
