def simplex_project(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)

    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    # Find rho such that sum_{i<=rho} (v_i - theta) = target_sum
    # where theta = (sum_{i<=rho} v_i - target_sum) / rho
    rho = np.arange(1, n+1)

    valid = theta < sorted_v

    theta_val = (cumsum[idx-1] - target_sum) / idx

def _simplex_project(v, target):
    n = v.size
    if target <= 0:
        return np.zeros(n)
    sv = np.sort(v)[::-1]
    cs = np.cumsum(sv)
    rho = np.arange(1, n + 1)
    theta = (cs - target) / rho
    mask = theta < sv
    if not mask.any():
        mask[-1] = True
    idx = np.max(np.where(mask)[0]) + 1
    t = (cs[idx - 1] - target) / idx
    return np.maximum(v - t, 0)
