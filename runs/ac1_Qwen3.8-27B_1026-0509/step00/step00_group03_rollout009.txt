```python
import time

        x = np.ones(n)

        x = np.clip(x, 0, 1000.0)
        s = x.sum()
        if s < 1e-10:
            return np.ones(n)
        return x  # Don't normalize during LP, just ensure nonneg
    
    best_x = make_valid(x.copy())
    best_score = float(evaluate_sequence(best_x.tolist()))
    
    # Build sparse convolution matrix F (2n-1 x n)
    # F[k,j] = 1 if max(0,k-n+1) <= j <= min(k,n-1)
    rows, cols = [], []
    for k in range(2*n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
    F = sparse.csr_matrix((np.ones(len(rows)), (rows, cols)), shape=(2*n-1, n))
    
    # LP iterations
    t = 0.3  # mixing parameter
    for it in range(500):
        if time.monotonic() > deadline:
            break
        
        # Compute current M
        C = np.convolve(x, x)
        M = C.max()
        S = x.sum()
        if S < 0.01:
            x = np.ones(n)
            continue

        c_obj = -np.ones(n)  # minimize -sum(b)
        res = optimize.linprog(c_obj, A_ub=F, b_ub=np.full(2*n-1, M),
                               bounds=[(0, None)]*n, method='highs',
                               options={'time_limit': 30, 'dual_feasibility_tol': 1e-8})
        
        if not res.success:
            break
        
        b = res.x
        S_b = b.sum()
        if S_b < 1e-10:
            break
        
        # Rescale b to match sum of x
        b = b * S / S_b
        
        # Update
        x_new = (1 - t) * x + t * b
        x_new = make_valid(x_new)
        
        # Check
        sc = float(evaluate_sequence(x_new.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_x = x_new.copy()
        
        x = x_new

    fs = float(evaluate_sequence(best_x.tolist()))
    if not np.isfinite(fs):
        return [1.0] * n
    return [float(v) for v in best_x]
```

```python
def build_F(n):
    # F is (2n-1) x n
    # F[k,j] = 1 iff max(0, k-n+1) <= j <= min(k, n-1)
    # Equivalent: 0 <= k-j <= n-1
    k_idx = np.arange(2*n-1)[:, None]  # (2n-1, 1)
    j_idx = np.arange(n)[None, :]  # (1, n)
    F_dense = ((k_idx - j_idx) >= 0) & ((k_idx - j_idx) <= n-1)
    F = sparse.csr_matrix(F_dense)
    return F
```
