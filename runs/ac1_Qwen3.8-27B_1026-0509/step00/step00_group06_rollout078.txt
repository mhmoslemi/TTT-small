import numpy as np
import time

try:
    from scipy.linalg import toeplitz
except Exception:
    toeplitz = None
try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _pow2(n):
    p = 1
    while p < n:
        p <<= 1
    return p


def _fft_conv(x):
    n = x.size
    if n == 0:
        return np.array([])
    N = _pow2(2 * n - 1)
    F = np.fft.rfft(x, N)
    c = np.fft.irfft(F * F, N)[:2 * n - 1]
    return c


def _fft_max(x):
    c = _fft_conv(x)
    if c.size == 0:
        return 0.0
    return float(np.max(c))


def _corr_q(p, f):
    n = f.size
    L = p.size
    total = n + L - 1
    N = _pow2(total)
    F = np.fft.rfft(f, N)
    P = np.fft.rfft(p[::-1], N)
    r = np.fft.irfft(F * P, N)[:total]
    if n == 1:
        return np.maximum(r[0:1], 0.0)
    q = r[2 * n - 2:n - 2:-1]
    return np.maximum(np.asarray(q, dtype=np.float64), 0.0)


def _as_nonneg_array(x):
    a = np.asarray(x, dtype=np.float64).ravel()
    if a.size == 0 or not np.all(np.isfinite(a)):
        return None
    a = np.clip(a, 0.0, 1000.0)
    return a


def _normalize(x, target=1.0):
    a = _as_nonneg_array(x)
    if a is None:
        return None
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return a * (target / s)


def _resample(a, m):
    arr = _as_nonneg_array(a)
    if arr is None or m <= 0:
        return None
    if arr.size == 1:
        return np.full(m, float(arr[0]), dtype=np.float64)
    if arr.size == m:
        return arr.copy()
    xa = np.linspace(0.0, 1.0, arr.size)
    xb = np.linspace(0.0, 1.0, m)
    return np.interp(xb, xa, arr)


def _consider_exact(f, gb):
    a = _normalize(f, 1.0)
    if a is None:

    try:

def _step(f, beta, vertex, etas):
    n = f.size
    c = _fft_conv(f)
    if c.size == 0:
        return None, float("inf")
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return None, float("inf")

    z = beta * (c / M)
    z -= z.max()
    p = np.exp(z)
    ps = p.sum()
    if ps <= 0.0:
        return None, float("inf")
    p = p / ps

    q = _corr_q(p, f)
    qmax = float(np.max(q)) if q.size else 0.0

    best = None
    best_sc = float("inf")

    def consider(cand):
        nonlocal best, best_sc
        cand = _normalize(cand, 1.0)
        if cand is None:
            return
        sc = 2.0 * n * _fft_max(cand)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand

    if vertex:
        if qmax > 0.0:
            h = 1.0 - q / qmax
        else:
            h = np.ones_like(f)
        h = np.maximum(h, 0.0)
        hs = float(h.sum())
        if hs > 0.0:
            hh = h / hs
            for a in (0.3, 0.7, 1.0):
                consider((1.0 - a) * f + a * hh)
    else:
        if qmax > 0.0:
            qn = q / qmax
        else:
            qn = np.zeros_like(f)
        base = -qn
        for eta in etas:
            w = np.exp(eta * base)
            consider(f * w)

    return best, best_sc

def _struct_inits(n, seed):
    rng = np.random.default_rng(seed)
    i = np.arange(n)
    out = [np.ones(n, dtype=np.float64)]
    for a in (0.5, 0.8):
        out.append(np.linspace(1.0 - a, 1.0 + a, n))
        out.append(np.linspace(1.0 + a, 1.0 - a, n))
    for L in (max(1, n // 20), max(1, n // 10), max(1, n // 5)):
        out.append(np.exp(-i / float(L)))
    for freq in (1, 2):
        for ph in (0.0, 0.25, 0.5):
            out.append(1.0 + 0.6 * np.sin(2.0 * np.pi * freq * i / n + 2.0 * np.pi * ph))
    out.append(np.exp(rng.normal(0.0, 0.4, n)))
    r = 2.0 * (rng.random(n) > 0.5).astype(np.float64) - 1.0
    out.append(1.0 + 0.4 * r)
    return [x for x in out if np.all(np.isfinite(x))]


def _run_eg(n, seed, deadline, gb, parent):

def _build_A_scaled(f, M):
    n = f.size
    a = f / M
    if toeplitz is not None:
        firstcol = np.zeros(2 * n - 1, dtype=np.float64)
        firstcol[:n] = a
        firstrow = np.zeros(n, dtype=np.float64)
        firstrow[0] = a[0]
        try:
            return firstcol.__class__(toeplitz(firstcol, firstrow)).astype(np.float64)
        except Exception:
            return None

    A = np.empty((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=np.float64)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if hi > lo:
            if hi - lo == 1:
                row[lo] = a[k - lo]
            else:
                row[lo:hi] = a[k - lo:k - hi:-1]
        A[k] = row
    return A


def _run_lp(m, seed, deadline, gb, init):

            break

        try:
            res = linprog(
                cobj,
                A_ub=A,
                b_ub=b,
                bounds=bounds,

        except Exception:

        g = np.asarray(g, dtype=np.float64)

        for a in (0.1, 0.3, 0.6, 1.0):
