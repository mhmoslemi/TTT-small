import numpy as _np
import time as _time

def _fft_conv(a):
    n = a.size
    L = 1
    need = 2 * n - 1
    while L < need:
        L <<= 1
    A = _np.fft.rfft(a, L)
    return _np.fft.irfft(A * A, L)[:need]

def _score(a):
    s = a.sum()
    if s <= 1e-12:
        return _np.inf
    b = _fft_conv(a)
    return float(2 * a.size * b.max() / (s * s))

                a = y
                base = min(base * 1.05, 1.5)
                break
            a = y  # keep last for next shrink
        else:

            a = best.copy()
            if base < 1e-5:

def propose_candidate(seed=42, budget_s=1500, **kwargs):
