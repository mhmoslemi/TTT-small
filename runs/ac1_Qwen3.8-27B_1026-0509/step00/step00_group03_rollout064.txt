import numpy as np
import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 5.0

    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000).copy()
    n = a.size
    if n == 0:
        a = np.ones(64)
        n = 64
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s

    best = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()

    def score_fn(a):
        return 2.0 * len(a) * np.convolve(a, a).max()

    def grad_step(a, alpha, p):
        c = np.convolve(a, a)
        c_shifted = c - c.max() + 1e-15
        w = c_shifted ** p
        ws = w.sum()
        if ws < 1e-30:
            return a.copy()
        w /= ws
        r = a[::-1]
        N = 1
        while N < 3 * n + 10:
            N *= 2
        W = np.fft.rfft(w, N)
        R = np.fft.rfft(r, N)
        conv_wr = np.fft.irfft(W * R, N)
        grad = 2.0 * conv_wr[n - 1:2 * n - 1]
        a_new = a * np.exp(-alpha * grad)
        a_new = np.maximum(a_new, 0.0)
        s2 = a_new.sum()
        if s2 < 1e-10:
            return a.copy()
        a_new /= s2
        return a_new

    alpha = 0.5
    p = 4

    while time.monotonic() < deadline:
        for _ in range(500):
            if time.monotonic() > deadline:
                break
            a = grad_step(a, alpha, p)
            sc = score_fn(a)
            if sc < best_score:
                best_score = sc
                best = a.copy()
        alpha *= 0.9
        if alpha < 0.02:
            alpha = 0.5
            p = min(p * 1.5, 64)
            if rng.random() < 0.3:
                a = best.copy()
                noise = rng.normal(0, 0.05, n)
                a *= np.exp(noise)
                a = np.maximum(a, 0)
                a /= a.sum()

    return [float(x) for x in best]
