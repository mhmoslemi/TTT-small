import numpy as np
import time
from scipy.signal import correlate
from scipy.sparse import csr_matrix
import cvxpy as cp


def _eval(a):
    return float(evaluate_sequence([float(x) for x in a]))


def _block_perturb(a, rng, scale):
    a2 = np.array(a, dtype=np.float64).copy()
    n = a2.size
    w = int(rng.integers(max(2, n // 300), max(3, n // 8)))
    s = int(rng.integers(0, max(1, n - w + 1)))
    a2[s:s + w] *= float(np.exp(rng.normal(0.0, scale)))
    if rng.random() < 0.3:
        a2 = 0.5 * (a2 + a2[::-1])
    return np.clip(a2, 0.0, 1000.0)


def _grad_step(a, eta, rng):
    conv = np.convolve(a, a)
    m = conv.max()
    tau = max(1e-9, 0.03 * (m if m > 0 else 1.0))
    w = np.exp((conv - m) / tau)
    ws = w.sum()
    if ws <= 0 or not np.isfinite(ws):
        return None
    w = w / ws
    try:
        g = correlate(a, w, mode='valid', method='fft')
    except Exception:
        g = correlate(a, w, mode='valid')
    g = g - g.mean()
    g = g + 0.08 * (rng.random(a.size) - 0.5)
    sd = g.std()
    if sd <= 0:
        return None
    factor = 1.0 - eta * (g / sd)
    a2 = np.clip(a * factor, 0.0, None)
    s2 = a2.sum()
    if s2 < 1e-12:
        return None
    a2 = a2 * (a.sum() / s2)
    return np.clip(a2, 0.0, 1000.0)


def _lp_step(a, budget_ok):
    conv = np.convolve(a, a)
    M = conv.max()
    if M <= 0:
        return None
    n = a.size
    K = 12
    peak = np.argsort(conv)[-K:]
    rows, cols, vals = [], [], []
    for i, k in enumerate(peak):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if hi < lo:
            continue
        js = np.arange(lo, hi + 1)
        vs = a[k - hi: k - lo + 1]
        rows.append(np.repeat(i, js.size))
        cols.append(js)
        vals.append(np.asarray(vs, dtype=np.float64))
    if not rows:
        return None
    C = csr_matrix((np.concatenate(vals), (np.concatenate(rows), np.concatenate(cols))),
                   shape=(len(peak), n))
    b = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), [C @ b <= M])
    try:
        prob.solve(solver=cp.ECOS, max_iters=4000, verbose=False)
    except Exception:
        return None
    if not budget_ok() or b.value is None:
        return None
    b = np.maximum(b.value, 0.0)
    sb = b.sum()
    if sb < 1e-12 or not np.isfinite(sb):
        return None
    return b * (a.sum() / sb)
