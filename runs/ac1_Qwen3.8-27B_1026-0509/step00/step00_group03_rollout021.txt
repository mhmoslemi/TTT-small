import time
from scipy.optimize import linprog

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(2535)
    n = parent.size
    target_sum = np.sqrt(2.0 * n)

    def normalize(x):
        s = x.sum()
        if s < 1e-14:
            return np.full(n, target_sum / n)
        return x * (target_sum / s)

    def exact_score(x):
        return float(evaluate_sequence(x.tolist()))

    def build_conv_mat(a):
        C = np.zeros((2 * n - 1, n))
        for i in range(n):
            C[i:i + n, i] = a
        return C

    def lp_step(a):
        C = build_conv_mat(a)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.empty((2 * n - 1, n + 1))
        A_ub[:, :n] = C
        A_ub[:, n] = -1.0
        b_ub = np.zeros(2 * n - 1)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])
        bounds = [(0, None)] * (n + 1)
        rem = deadline - time.monotonic() - 10
        if rem < 1:
            return None
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': min(30.0, rem)})
        if res.success and res.x is not None:
            return np.maximum(res.x[:n], 0.0)
        return None

    a = normalize(parent.astype(np.float64).copy())
    best = a.copy()
    best_score = exact_score(best)

    # Phase 1: LP water-filling
    stalled = 0
    for it in range(5000):
        if time.monotonic() > deadline - 180:
            break
        b_new = lp_step(a)
        if b_new is None:
            stalled += 1
            if stalled > 5:
                break
            continue
        stalled = 0
        curr = exact_score(a)

            if best_s < best_score:

    return [float(x) for x in a]
