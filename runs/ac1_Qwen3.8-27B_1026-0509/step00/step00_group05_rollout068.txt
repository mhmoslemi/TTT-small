                    jlo = max(0, k - n + 1); jhi = min(k, n - 1)
                    if jhi >= jlo:
                        A[k, jlo:jhi+1] = a[k-jlo : k-jhi-1 : -1]
                g = cp.Variable(n)
                prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= delta, g >= 0])
                prob.solve(solver=cp.CLARABEL, verbose=False)
                if g.value is None: break
                gv = np.maximum(g.value, 0.0); gs = gv.sum()
                if gs < 1e-8: break
                gv /= gs
                ok = False
                for alpha in (0.5, 0.25, 0.1):
                    at = _psimplex((1-alpha)*a + alpha*gv)
                    sc = _escore(at, n)
                    if sc < best_s:
                        best_s = sc; best_a = at.copy(); a = at; ok = True; break
                    if sc < _escore(a, n):
                        a = at; break
                if not ok: break
        except Exception:
            pass

    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        best_a = np.ones(n)
    return [float(x) for x in best_a]
