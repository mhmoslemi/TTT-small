    n = len(height_sequence_1)
    target_sum = np.sqrt(2.0 * n)
    
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()
    a = np.maximum(a, 0) * (target_sum / a.sum())
    
    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))
    
    # Adam parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Temperature for log-sum-exp
    tau = 1.0
    
    while time.monotonic() < deadline:
        t += 1
        # Compute c and log-sum-exp gradient
        c = np.convolve(a, a)  # length 2n-1
        # Softmax with temperature
        c_scaled = c / tau
        c_max = c_scaled.max()
        exp_c = np.exp(c_scaled - c_max)
        p = exp_c / exp_c.sum()
        
        # Gradient of log-sum-exp w.r.t. a[i]:
        # sum_k p_k * 2*a[k-i] * 1[0<=k-i<=n-1]
        grad = np.zeros(n)
        # For efficiency, only use top K peaks
        top_K = min(50, len(c))
        top_idx = np.argpartition(c, -top_K)[-top_K:]
        p_top = p[top_idx]
        p_top = p_top / p_top.sum()
        
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n-1, k)
            grad[lo:hi+1] += 2.0 * p_top[list(top_idx).index(k)] * a[k-hi:k-lo+1][::-1]
        
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new *= target_sum / s
        
        score = float(evaluate_sequence(a_new.tolist()))
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
            a = a_new.copy()
            
        # Decrease temperature periodically

            lr *= 0.9
    
    return [float(x) for x in best_a]
