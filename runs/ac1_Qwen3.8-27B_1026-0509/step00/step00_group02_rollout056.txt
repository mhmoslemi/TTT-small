        P = np.fft.rfft(p, N)
        corr = np.fft.irfft(P * np.conj(Ff), N)
        grad = 2.0 * corr[:n]

        # Momentum
        momentum = beta * momentum + (1 - beta) * grad
        f_new = f - alpha * momentum
        f_new = np.maximum(f_new, 0.0)
        ns = f_new.sum()
        if ns < 1e-10:
            alpha *= 0.5
            momentum *= 0.5
            rejects += 1
            continue
        f_new = f_new / ns * T

        new_sc, _ = fast_eval(f_new)

        if new_sc < best_score:
            best = f_new.copy()
            best_score = new_sc
            rejects = 0
            alpha = min(alpha * 1.03, 0.8)
            tau = min(tau * 1.003, 35.0)
            beta = min(beta + 0.002, 0.9)
        elif new_sc < (2.0 * n * conv.max()) / (f.sum() ** 2):
            f = f_new
            rejects = 0
            alpha *= 0.97
        else:
            rejects += 1
            alpha *= 0.6
            momentum *= 0.3
            if rejects > 80:
                f = best + rng.normal(0, 0.04 * best.mean(), n)
                f = np.maximum(f, 0)
                f = f / f.sum() * T
                alpha = 0.05
                tau = max(tau * 0.7, 1.5)
                beta = 0.3
                momentum = np.zeros(n)
                rejects = 0

    final = float(evaluate_sequence(best.tolist()))
    if final < best_score:
        best_score = final

    return [float(x) for x in best]
