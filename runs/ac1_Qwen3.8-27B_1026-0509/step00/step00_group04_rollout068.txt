        return 2.0 * n * M / (S * S)

        M = float(b.max())

                lo = 0
            hi = k

                hi = n - 1
            if lo <= hi:
                v[lo:hi + 1] += 2.0 * a[(k - hi):(k - lo + 1)]
        v /= float(len(K))
        return v, M

        return 2.0 * n * (v * S - 2.0 * M) / (S ** 3)
