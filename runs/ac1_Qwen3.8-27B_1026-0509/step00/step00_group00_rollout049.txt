import time


def _fft_conv(a):
    n = len(a)
    m = 2 * n - 1
    sz = 1 << (m - 1).bit_length()
    F = np.fft.rfft(a, sz)
    return np.fft.irfft(F * F, sz)[:m]


def _grad_peak(a, k):
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n, k + 1)
    if lo < hi:
        g[lo:hi] = 2.0 * a[k - hi + 1 : k - lo + 1][::-1]
    return g
