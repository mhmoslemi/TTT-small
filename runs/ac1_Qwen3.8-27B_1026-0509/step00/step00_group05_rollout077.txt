def _make_conv(n):
    N = 1
    while N < 2*n-1: N <<=1
    Ncorr=1
    while Ncorr < 3*n-2: Ncorr <<=1
    def conv(x):
        X=np.fft.rfft(x, N)
        return np.fft.irfft(X*X, N)[:2*n-1]
    def corr_grad(x, w):
        X=np.fft.rfft(x, Ncorr)
        W=np.fft.rfft(w[::-1], Ncorr)
        C=np.fft.irfft(X*W, Ncorr)
        return 2.0*C[n-1:2*n-1][::-1]
    def score(x):
        S=x.sum();
        if S<0.01: return np.inf
        return 2.0*n*conv(x).max()/(S*S)
    return conv, corr_grad, score

        add_start(a * (1.0 - t) ** al)
    for al in (0.1, 0.2, 0.4):
        add_start((idx + 0.5) ** (-0.5) * (1.0 - t) ** al)
    for eps in (0.01, 0.02, 0.05):
        add_start(a + eps * unit)
        add_start(reverse_a := a[::-1] + eps * unit)
    for _ in range(6):
        y = np.cumsum(rng.normal(0.0, 0.08, n))
        y -= y.max()
        add_start(np.exp(0.25 * y) * (0.5 + 0.5 * (idx / float(max(1, n - 1)))))
    add_start(a[::-1])
    add_start(unit)

    if starts:
        starts.sort(key=lambda z: z[0])
        best_score, best_x = starts[0]
    else:
        best_x = unit / float(n)
        best_score = conv_score(best_x)

    if not math.isfinite(best_score):
        best_x = unit / float(n)
        best_score = conv_score(best_x)

    gold_x = best_x.copy()

    def softmax(y):
        y = np.asarray(y, dtype=np.float64)
        if y.size != n or not np.all(np.isfinite(y)):
            return None
        y = y - np.max(y)
        y = np.clip(y, -700.0, 0.0)
        e = np.exp(y)
        s = float(np.sum(e))
        if not math.isfinite(s) or s <= 0:
            return None
        x = e / s
        return np.clip(x, 0.0, 1000.0)

    def update_best(x):
        nonlocal best_x, best_score
        x = np.asarray(x, dtype=np.float64)
        if x.size != n or not np.all(np.isfinite(x)):
            return
        x = np.clip(x, 0.0, 1000.0)
        s = float(np.sum(x))
        if s < 1e-12:
            return
        x = x / s
        sc = conv_score(x)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_x = x.copy()

    if minimize is not None:
        lbfgs_x = best_x.copy()
        for beta_fac, maxfun in ((8, 900), (40, 750), (150, 600)):
            if time.monotonic() > deadline - 75.0:
                break
            beta = float(n) * float(beta_fac)
            y0 = np.log(np.maximum(lbfgs_x, 1e-300))
            y0 -= y0.max()

            def fun(y, beta=beta):
                x = softmax(y)
                if x is None:
                    return 1e100, np.zeros(n, dtype=np.float64)
                q = conv_full(x)
                q = np.maximum(q, 0.0)
                qmax = float(np.max(q))
                scaled = beta * (q - qmax)
                ue = np.exp(np.clip(scaled, -700.0, 0.0))
                Z = float(np.sum(ue))
                if not math.isfinite(Z) or Z <= 0:
                    return 1e100, np.zeros(n, dtype=np.float64)
                w = ue / Z
                G = corr_grad(x, w)
                gm = float(np.dot(G, x))
                gy = x * (G - gm)
                val = qmax + math.log(Z) / beta
                if not math.isfinite(val):
                    return 1e100, np.zeros(n, dtype=np.float64)
                gy = np.where(np.isfinite(gy), gy, 0.0)
                return float(val), gy

            cnt = [0]

            def callback(yk):
                cnt[0] += 1
                if cnt[0] % 60 == 0:
                    xk = softmax(yk)
                    if xk is not None:
                        update_best(xk)

            try:
                res = minimize(
                    fun,
                    y0,
                    method="L-BFGS-B",
                    jac=True,
                    options={
                        "maxiter": 2500,
                        "maxfun": int(maxfun),
                        "ftol": 1e-12,
                        "gtol": 1e-8,
                        "maxls": 50,
                        "disp": False,
                        "callback": callback,
                    },
                )
                xr = softmax(res.x)
                if xr is not None:
                    update_best(xr)
                    score_xr = conv_score(xr)
                    if math.isfinite(score_xr) and score_xr <= 1.1 * best_score:
                        lbfgs_x = xr.copy()
                    else:
                        lbfgs_x = best_x.copy()
                else:
                    lbfgs_x = best_x.copy()
            except Exception:
                lbfgs_x = best_x.copy()

    def polish(x0, min_slice_s=20.0):
        nonlocal best_x, best_score
        if time.monotonic() + min_slice_s > deadline:
            min_slice_s = max(2.0, deadline - time.monotonic() - 2.0)
        current = np.asarray(x0, dtype=np.float64)
        cur_score = conv_score(current)
        beta = float(n) * 30.0
        eta = 1.0
        stalls = 0
        improved_since_beta = 0

        while time.monotonic() < deadline - min_slice_s:
            q = conv_full(current)
            q = np.maximum(q, 0.0)
            qmax = float(np.max(q))
            scaled = beta * (q - qmax)
            ue = np.exp(np.clip(scaled, -700.0, 0.0))
            Z = float(np.sum(ue))
            if not math.isfinite(Z) or Z <= 0:
                break
            w = ue / Z
            G = corr_grad(current, w)
            gm = float(np.dot(G, current))
            h = G - gm
            hn = float(np.linalg.norm(h))
            if not math.isfinite(hn) or hn < 1e-14:
                beta *= 1.8
                if beta > 5e7:
                    break
                continue

            eta = max(0.05, min(5.0, 0.5 / max(1e-12, float(np.mean(np.abs(h)))) * max(1, n) / 1000.0))
            if not math.isfinite(eta) or eta <= 0:
                eta = 1.0

            improved = False
            e = eta
            for _ in range(24):
                z = -e * h
                z -= np.max(z)
                xn = current * np.exp(z)
                s = float(np.sum(xn))
                if not math.isfinite(s) or s <= 1e-12:
                    e *= 0.5
                    continue
                xn = np.clip(xn / s, 0.0, 1000.0)
                sc = conv_score(xn)
                if math.isfinite(sc) and sc < cur_score - 1e-12:
                    current = xn
                    cur_score = sc
                    improved = True
                    eta = e * 1.15
                    break
                e *= 0.5

            if improved:
                if cur_score < best_score:
                    best_score = cur_score
                    best_x = current.copy()
                stalls = 0
                improved_since_beta += 1
                if improved_since_beta % 120 == 0:
                    beta = min(beta * 1.7, 5e7)
                    improved_since_beta = 0
            else:
                stalls += 1
                eta *= 0.6
                if stalls % 80 == 0:
                    beta = min(beta * 1.8, 5e7)
                    pert = current * np.exp(rng.normal(0.0, 0.08, n))
                    ps = float(np.sum(pert))
                    if ps > 0:
                        pert = np.clip(pert / ps, 0.0, 1000.0)
                        psc = conv_score(pert)
                        if math.isfinite(psc) and psc < cur_score:
                            current = pert
                            cur_score = psc
                if stalls > 500 or beta > 5e7:
                    break

    polish(best_x, min_slice_s=max(10.0, 0.1 * float(budget_s)))

    candidate = [float(v) for v in np.clip(best_x, 0.0, 1000.0)]
    final_score = evaluate_sequence(candidate)
    if not math.isfinite(final_score):
        fallback = parent.copy() if parent is not None and parent.size == n else a.copy()
        s = float(np.sum(fallback))
        if s > 0:
            fallback = fallback / s
        candidate = [float(v) for v in np.clip(fallback, 0.0, 1000.0)]
        final_score = evaluate_sequence(candidate)
        if not math.isfinite(final_score):
            candidate = [1.0 / float(n)] * n

    return candidate
