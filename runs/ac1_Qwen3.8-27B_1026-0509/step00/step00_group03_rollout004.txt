    while time.monotonic() < deadline:
        t += 1
        c = np.convolve(a, a)
        cmax = c.max()

        k_active = np.where(c >= 0.999 * cmax)[0]

        # grad_k[l] = 2*a[k-l] if 0<=k-l<n
        g = np.zeros(n)

            hi = min(n - 1, k)
            # a[k-l] for l in [0, n-1] where 0<=k-l<n means l in [k-n+1, k]
            # So g[l] += 2*a[k-l] for l in [max(0,k-n+1), min(n-1,k)]

            idx = k - np.arange(lo, hi+1)  # these are the indices into a
            g[lo:hi+1] += 2.0 * a[idx]

        a_new = a - step * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Check improvement
        c_new = np.convolve(a_new, a_new)
        if c_new.max() < cmax:
            a = a_new
            score = 2.0 * n * c_new.max()
            if score < best_score:
                best = a.copy()
                best_score = score
        else:

            step *= 0.9
            # Or try perturb

                a = a + step * 0.1 * (rng.standard_normal(n) - 0.5)
                a = np.maximum(a, 0)
                a = a / a.sum()

            a_pert = a_pert / a_pert.sum()
            if get_score(a_pert) < get_score(a):
                a = a_pert
    
    # Final validation
    result = best.copy()
    if result.sum() < 0.01:
        result *= 0.01 / result.sum()
    return [float(x) for x in result]

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    n = len(a)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / S  # sum = 1
    
    best = a.copy()
    best_score = 2.0 * n * np.convolve(best, best).max()
    
    def subgradient(a, c, cmax, eps=0.01):
        """Compute a subgradient of max(conv(a,a)) at point a."""
        # Active set: k where c[k] >= (1-eps)*cmax
        active = np.where(c >= (1-eps) * cmax)[0]
        # Limit to top 50 for speed
        if len(active) > 50:
            # Keep top 50
            active_c = c[active]
            top_idx = np.argsort(-active_c)[:50]
            active = active[top_idx]
        
        g = np.zeros(n)
        for k in active:
            # g[l] += 2*a[k-l] for l where 0<=k-l<n
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k - np.arange(lo, hi+1)]
        g /= len(active)
        return g
    
    # Main optimization loop
    step = 0.005
    min_step = 1e-6
    patience = 0
    max_patience = 500
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        c = np.convolve(a, a)
        cmax = c.max()
        current_score = 2.0 * n * cmax
        
        if current_score < best_score:
            best = a.copy()
            best_score = current_score
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            # Random restart: perturb current best
            a = best * (1 + 0.1 * rng.standard_normal(n))
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0:
                a = a / s
            else:
                a = np.ones(n) / n
            patience = 0
            step = 0.005
            continue
        
        g = subgradient(a, c, cmax)
        
        # Projected gradient step
        a_new = a - step * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            step *= 0.5
            continue
        a_new = a_new / s
        
        # Evaluate
        c_new = np.convolve(a_new, a_new)
        if c_new.max() <= cmax + 1e-10:
            a = a_new
        else:

            step *= 0.7
            if step < min_step:
                # Try random perturbation

                a_pert = np.maximum(a_pert, 0)

                if sp > 0:

                    c_p = np.convolve(a_pert, a_pert).max()
                    if c_p < cmax:
                        a = a_pert
                step = 0.001
        # Slowly decrease step
        step = max(min_step, step * 0.9995)

        final_score = evaluate_sequence(result.tolist())
