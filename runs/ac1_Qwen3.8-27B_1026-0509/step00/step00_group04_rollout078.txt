import numpy as np
import time
from scipy.fft import fft, ifft


def _pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _score_fft(p, n, M, L):
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-300:
        return float("inf")
    try:
        F = fft(np.asarray(p, dtype=np.float64), M)
        b = ifft(F * F, M).real[:L]
        mb = float(np.max(b))
        if not np.isfinite(mb):
            return float("inf")
        return float(2.0 * n * mb / (s * s))
    except Exception:
        return float("inf")


def _add_candidate(cands, p, n, M, L, limit=260):
    try:
        p = np.asarray(p, dtype=np.float64).ravel()
    except Exception:
        return
    if p.size != n or n <= 0:
        return
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-120:
        return
    p = p / s
    sc = _score_fft(p, n, M, L)
    if not np.isfinite(sc):
        return
    if len(cands) >= limit:
        cands.sort(key=lambda z: z[0])
        if sc >= cands[-1][0]:
            return
        cands.pop()
    cands.append((sc, n, M, L, p))


def _build_fixed_candidates(n, M, L, rng):

            fallback = np.ones(nmain, dtype=np.float64)
            out = [float(x) for x in fallback]
            fscore = float(evaluate_sequence(out))
    except Exception:
        fallback = np.ones(nmain, dtype=np.float64)
        out = [float(x) for x in fallback]

    if time.monotonic() < deadline - 2.0:
        try:
            if len(out) != nmain:
                pass
            if not all(np.isfinite(x) and x >= 0.0 and x <= 1000.0 for x in out):
                out = [float(x) for x in np.ones(nmain, dtype=np.float64)]
        except Exception:
            pass

    return out
