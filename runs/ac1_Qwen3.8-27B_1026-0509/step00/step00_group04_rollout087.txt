def local_optimize(f, max_iter=100, deadline, best_score):
    n=len(f); f=f/ f.sum()
    cur = f.copy()
    cur_score = score_small(cur)
    no_improve=0
    for it in range(max_iter):
       if time > deadline-...: break
       if cur_score < best_score: update best
       M = max(conv(cur,cur))
       # solve LP for p
       p = solve_p(cur)
       if p is None:
          # multiplicative weights step
          p = mw_step(cur, steps=20)
       # line search
       improved=False
       for t in [1.0,0.7,0.5,0.3,0.2,0.1,0.05]:
           h=(1-t)*cur + t*p
           # maybe add tiny random to break symmetry if p~cur
           sc=score_small(h)
           if sc < cur_score*(1-1e-7): # improvement
              cur=h; cur_score=sc; improved=True; break
       if improved: no_improve=0
       else:
          no_improve +=1
          # perturb cur
          cur = cur * (1+0.1*rng.normal(0,1)); cur=np.clip(cur,0); cur/=sum
          # if many no improve, break
          if no_improve>5: break
    return cur,cur_score

import numpy as np
import time
from scipy.linalg import toeplitz
from scipy.optimize import linprog

_INF = float("inf")


def _fast_score(f):
    n = f.shape[0]
    if n == 0:
        return _INF
    s = float(np.sum(f))
    if not np.isfinite(s) or s < 0.01:
        return _INF
    b = np.convolve(f, f)
    m = float(np.max(b))
    if not np.isfinite(m):
        return _INF
    return float(2.0 * n * m / (s * s))


def _norm(f):
    f = np.asarray(f, dtype=float).ravel()
    if f.size == 0:
        return None
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.clip(f, 0.0, 1000.0)
    s = float(np.sum(f))
    if not np.isfinite(s) or s <= 0.0:
        return None
    return f / s


def _consider(arr, state):
    f = _norm(arr)
    if f is None:
        return
    sc = _fast_score(f)
    if not np.isfinite(sc) or sc + 1e-12 >= state["score"]:
        return
    try:
        esc = evaluate_sequence(f.tolist())
    except Exception:
        esc = _INF
    if np.isfinite(esc) and esc < state["score"]:
        state["score"] = float(esc)
        state["arr"] = f.copy()


_lp_ctx = {}


def _ctx(n):
    ctx = _lp_ctx.get(n)
    if ctx is None:
        m = 2 * n - 1
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        ctx = {
            "m": m,
            "c": c,
            "A_eq": A_eq,
            "b_eq": np.array([1.0]),
            "b_ub": np.zeros(m),
            "rc": -np.ones(m),
            "bounds": [(0.0, 1.0)] * n + [(0.0, None)],
        }
        _lp_ctx[n] = ctx
    return ctx


def _solve_p(f, end):
    n = f.shape[0]
    ctx = _ctx(n)
    m = ctx["m"]
    rem = float(end - time.monotonic())
    if rem < 2.0:
        return None
    first_col = np.zeros(m)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    A = toeplitz(first_col, first_row)
    A_ub = np.empty((m, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = ctx["rc"]
    tl = max(0.5, min(5.0, 0.25 * rem))
    try:
        res = linprog(
            ctx["c"],
            A_ub=A_ub,
            b_ub=ctx["b_ub"],
            A_eq=ctx["A_eq"],
            b_eq=ctx["b_eq"],
            bounds=ctx["bounds"],
            method="highs",
            options={"presolve": True, "time_limit": tl},
        )
        if res.status == 0 and res.x is not None:
            p = res.x[:n]
            p = np.clip(p, 0.0, None)
            s = float(np.sum(p))
            if np.isfinite(s) and s > 1e-12:
                return p / s
    except Exception:
        pass
    return None


def _mw_p(f, steps=12):
    n = f.shape[0]
    p = np.ones(n) / n
    for _ in range(steps):
        Ap = np.convolve(f, p)
        if Ap.size == 0:
            return None
        k = int(np.argmax(Ap))
        w = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            cols = np.arange(lo, hi + 1)
            w[cols] = f[k - cols]
        scale = 0.5 / max(1e-12, float(np.max(w)))
        p *= np.exp(-scale * w)
        s = float(np.sum(p))
        if not np.isfinite(s) or s <= 0.0:
            return None
        p /= s
    return p


def _make_init(n, kind, rng):
    x = np.linspace(0.0, 1.0, n)
    if kind == 0:
        alpha = float(rng.choice([0.2, 0.5, 1.0, 3.0]))
        f = rng.dirichlet(np.full(n, alpha))
    elif kind == 1:
        lam = float(rng.uniform(-6.0, 6.0))
        f = np.exp(lam * x)
    elif kind == 2:
        a = float(rng.uniform(0.15, 1.0))
        i = np.arange(1, n + 1)
        f = i ** (-a)
        if rng.random() < 0.5:
            f = f[::-1]
    elif kind == 3:
        base = float(rng.uniform(0.02, 0.6))
        slope = float(rng.uniform(0.0, 10.0))
        f = base + slope * x
    elif kind == 4:

        phi = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.2, 0.95))
        f = 1.0 + amp * np.cos(2.0 * np.pi * k * x / n + phi)
        f = np.maximum(f, 1e-4)
    else:
        split = float(rng.uniform(0.3, 0.7))
        high = float(rng.uniform(0.4, 2.5))
        f = np.where(x < split, high, 1.0 / high)
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 0.0)
    s = float(np.sum(f))
    if not np.isfinite(s) or s <= 0.0:

        return f
    f = f / s
    f *= 1.0 + 1e-5 * rng.normal(size=n)

    s = float(np.sum(f))
    if s <= 0.0:

    else:
        f /= s

            noise = 1.0 + 0.12 * rng.normal(size=n)

            s = float(np.sum(f))
            if s <= 0.0:
                f = best_f.copy()
                cur = best_sc
            else:
                f /= s
                cur = _fast_score(f)
        else:

                    h /= hs

                else:

                cur = _fast_score(f)
                if stall > 5:
                    break

        if cur < best_sc:
            best_sc = cur
            best_f = f.copy()

    return best_f, best_sc


def _top_inits(n, count=3):
    res = []

    def push(a):
        f = _norm(a)
        if f is None:
            return
        sc = _fast_score(f)
        if np.isfinite(sc):
            res.append((sc, f.copy()))

    i = np.arange(1, n + 1, dtype=float)
    x = np.linspace(0.0, 1.0, n)
    for a in (0.30, 0.40, 0.50, 0.60, 0.70, 0.80, 0.90):

        push(base + slope * x)

    res.sort(key=lambda z: z[0])
    return [f for _, f in res[:count]]
