def _norm1(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _surrogate(a, n):
    return float(2.0 * n * np.convolve(a, a, mode='full').max())


def _optimize(a, n, rng, deadline, best, best_score):
    T = 50.0

    eta_min = 1e-9
    stall = 0

    local_max = _surrogate(a_local, n)

        c = np.convolve(a_local, a_local, mode='full')
        max_c = float(c.max())

            local_max = max_c
            stall = 0
        else:
            stall += 1
            if stall > 4000:
                break

        p = np.exp(T * (c - max_c))
        ps = p.sum()
        if ps <= 0:
