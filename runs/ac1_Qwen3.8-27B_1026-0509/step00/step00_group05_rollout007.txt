```python
import numpy as np
import time

def _score(x):
    """Fast score computation using FFT."""
    n = len(x)
    S = x.sum()
    if S < 0.01:
        return np.inf
    # FFT-based full convolution
    f = np.fft.rfft(x, n=2*n)
    conv = np.fft.irfft(f*f, n=2*n-1)
    max_c = conv.max()
    return 2.0*n*max_c/(S*S)

def _gradient(x, k_star):
    """Gradient of score w.r.t. x, given argmax lag k_star."""
    n = len(x)
    S = x.sum()
    c_max = np.fft.irfft(np.fft.rfft(x, 2*n)**2, 2*n-1)[k_star]
    # actually let's just pass c_max
    # grad[i] = 2n/S^2 * (S * dc_dk[i] - 2*c_max*x[i])
    # dc_dk[i] = 2*x[k_star - i] if 0 <= k_star-i <= n-1 else 0
    dc = np.zeros(n)
    lo = max(0, k_star - (n-1))
    hi = min(n-1, k_star)
    for i in range(lo, hi+1):
        dc[i] = 2.0 * x[k_star - i]
    # Handle the case where k_star - i == i (i.e., i = k_star/2)
    # This is already handled correctly: dc[i] = 2*x[k_star-i] = 2*x[i] when k_star=2i
    
    grad = (2.0*n/(S*S)) * (S*dc - 2.0*c_max*x)
    return grad
```
