```python
def ac1_score(a):
    """Compute the AC1 score for a nonnegative sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * b.max() / (s * s)

def ac1_smoothed(a, beta=100.0):
    """Smoothed version using log-sum-exp."""
    n = len(a)
    s = a.sum()
    b = np.convolve(a, a)
    # normalize
    b_norm = b / (s * s)
    M = b_norm.max()
    logsum = np.log(np.sum(np.exp(beta * (b_norm - M)))) / beta + M
    return 2.0 * n * logsum
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    N_fft = 1 << (3 * n - 1).bit_length()

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    def fft_corr(w, x):
        """Compute g[j]=sum_m x[m]*w[j+m] for j=0..n-1 via FFT."""
        Xr = np.fft.rfft(x[::-1], N_fft)
        W = np.fft.rfft(w, N_fft)
        full = np.fft.irfft(Xr * W, N_fft)[:3 * n - 1]
        return full[n - 1:2 * n - 1]

    def smooth_obj_grad(x, beta):
        x = np.maximum(x, 0.0)
        b = np.convolve(x, x)
        M = b.max()
        sh = beta * (b - M)
        s = sh.max()
        e = np.exp(sh - s)
        se = e.sum()
        f = 2.0 * n * (M + (s + np.log(se)) / beta)
        w = e / se
        grad = 2.0 * fft_corr(w, x)
        return f, grad

    def local_opt(a_init, max_time):
        a = a_init.copy()
        s = a.sum()
        if s < 1e-15:
            return a
        a /= s
        t0 = time.monotonic()
        for beta in [30, 80, 200, 500, 1500, 5000]:
            if time.monotonic() - t0 > max_time:
                break
            if time.monotonic() > deadline - 30:
                break
            def fg(x):
                return smooth_obj_grad(x, beta)
            try:
                res = minimize(fg, a, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 120, 'maxfun': 300,
                                        'ftol': 1e-14, 'gtol': 1e-10})
                a = np.maximum(res.x, 0.0)
            except Exception:
                pass
            sn = a.sum()
            if sn > 1e-15:
                a /= sn
        return a

    def active_refine(a, max_time):
        a = a.copy()
        t0 = time.monotonic()
        for _ in range(500):
            if time.monotonic() - t0 > max_time or time.monotonic() > deadline - 5:
                break
            b = np.convolve(a, a)
            M = b.max()
            ks = int(b.argmax())
            g = np.zeros(n)
            for j in range(n):
                kj = ks - j
                if 0 <= kj < n:
                    g[j] = 2.0 * a[kj] if ks != 2 * j else a[j]
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            lr = 0.5 / gn
            for _ in range(20):
                a_new = a - lr * g
                a_new = np.maximum(a_new, 0.0)
                sn = a_new.sum()
                if sn < 1e-15:
                    break
                a_new /= sn
                b_new = np.convolve(a_new, a_new)
                if b_new.max() < M * (1 - 1e-7):
                    a = a_new
                    break
                lr *= 0.5
            if lr < 1e-12:
                break
        return a

    # Phase 1: optimize parent
    a1 = local_opt(parent, max_time=min(250, deadline - time.monotonic() - 30))
    sc1 = float(evaluate_sequence(a1.tolist()))
    if np.isfinite(sc1) and sc1 < best_score:
        best_score, best = sc1, a1.copy()

        sc2 = float(evaluate_sequence(a2.tolist()))
        if np.isfinite(sc2) and sc2 < best_score:
            best_score, best = sc2, a2.copy()

    # Phase 3: alternate initializations

    inits.append(np.exp(-2.0 * j / n))
    inits.append(np.exp(-2.0 * (n - 1 - j) / n))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * j / n) + 0.01)
    inits.append(np.exp(rng.normal(0, 0.6, n)))

    for k, ai in enumerate(inits):
        if time.monotonic() > deadline - 80:
            break
        ai = np.maximum(ai, 0.0)
        s = ai.sum()
        if s < 1e-10:
            continue
        ai /= s
        rem = deadline - time.monotonic() - 40
        if rem < 60:
            break

        sc = float(evaluate_sequence(ao.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score, best = sc, ao.copy()
            if time.monotonic() < deadline - 50:
                ao2 = active_refine(best, max_time=40)
                sc2 = float(evaluate_sequence(ao2.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score, best = sc2, ao2.copy()

        best = parent / max(parent.sum(), 1e-10)
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = np.clip(parent, 0, 1000.0)
    return [float(v) for v in best]
```
