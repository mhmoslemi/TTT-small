import numpy as np
import time

def propose_candidate(seed=42, budget_s=1500, **kwargs):

    Ng = 1 << (4 * n - 3).bit_length()

    def self_conv(f):
        F = np.fft.rfft(f, Nc)
        return np.fft.irfft(F * F, Nc)[:2 * n - 1]

    def compute_grad(f, C, beta):
        mc = C.max()
        p = np.exp(beta * (C - mc))
        p /= p.sum()
        h = np.zeros(2 * n - 1)
        h[:n] = f[::-1]
        P = np.fft.rfft(p, Ng)
        H = np.fft.rfft(h, Ng)
        conv = np.fft.irfft(P * H, Ng)
        return 2.0 * conv[n - 1:2 * n - 1]

    def proj(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-15:
            return np.full(n, 1.0 / n)
        return x / s

    inits = []
    inits.append(parent / parent.sum())
    rr = 0.99
    ef = rr ** np.arange(n)
    inits.append(ef / ef.sum())
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    two = np.concatenate([np.ones(n // 2), 0.4 * np.ones(n - n // 2)])
    inits.append(two / two.sum())
    inits.append(rng.random(n) + 0.01)
    inits[-1] /= inits[-1].sum()

    best_score = np.inf
    best_f = parent.copy()
