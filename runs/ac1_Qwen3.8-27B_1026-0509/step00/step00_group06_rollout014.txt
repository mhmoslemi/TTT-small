        # = 2 * (weights_shifted_correlation)[j]

        # Let W = weights (length 2n-1, zero-padded to 2n-1)
        # Let A = a (length n)

        # This is getting too complicated. Let me just use a direct loop for small n
        # or a smarter FFT approach.
        
        # Simpler: use the fact that for the single-peak case, grad = 2 * a[k*-j]
        # For the soft-max case, it's a weighted sum of such vectors.
        # Let me just compute it using matrix-free operations.
        
        # For now, let's use the single-peak gradient (simpler and often sufficient):
        peak_idx = np.argmax(b)
        g = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(peak_idx, n - 1)
        g[lo:hi+1] = 2 * a[peak_idx - hi : peak_idx - lo]
        
        # Project to maintain sum: subtract mean
        g = g - g.mean()
        
        # Step
        step = eta0 / (1 + t * eta0 / n)  # decreasing step size
        a_new = a - step * g
        a_new = np.clip(a_new, 0, 1e-10)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new / a_new.sum()
        
        # Accept if improved (on the soft-max surrogate or exact)
        b_new = fft_convolve(a_new, a_new)
        if np.max(b_new) <= np.max(b) + 1e-12:
            a = a_new
    
    return a
