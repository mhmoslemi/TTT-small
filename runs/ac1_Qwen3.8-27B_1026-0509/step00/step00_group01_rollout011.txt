    """Full convolution of x with itself using FFT."""
    n = len(x)
    size = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    fft_size = 1 << (size - 1).bit_length()
    X = fft(x, fft_size)
    c = np.real(ifft(X * X, fft_size))[:size]
    return c

def simplex_project(v):
    """Project v onto the simplex {x >= 0 : sum x = 1}."""
    n = len(v)

import time
from scipy.fft import fft, ifft

def _conv(x):
    """Full autocorrelation (conv with self) of x."""
    n = len(x)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    X = fft(x, fsize)
    return np.real(ifft(X * X, fsize))[:size]

def _smooth_grad(x, c, eps):
    """Gradient of eps*log(sum exp(c_k/eps)) w.r.t. x."""
    n = len(x)
    # Softmax weights
    c_shifted = c - c.max()
    w = np.exp(c_shifted / eps)
    w /= w.sum()
    # g[j] = 2 * sum_m x[m] * w[m+j]
    # = 2 * (x_reversed * w)[n-1+j] for j=0..n-1
    x_rev = x[::-1]
    fsize = 1 << (n + len(w) - 1 - 1).bit_length()
    XR = fft(x_rev, fsize)
    W = fft(w, fsize)
    conv = np.real(ifft(XR * W, fsize))
    g = 2.0 * conv[n-1:2*n-1]
    return g

    """Project onto {x >= 0, sum x = 1}."""
    n = len(v)
    u, idx = np.sort(v)[::-1], np.argsort(v)[::-1]

    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]

    k = rho[-1]
    theta = (cssv[k] - 1) / (k + 1)
    out = np.zeros(n)
    out[idx] = np.maximum(v[idx] - theta, 0)
    return out
