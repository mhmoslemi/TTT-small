            lr = max(lr * 0.95, 0.1)
            
            # Periodic perturbation to escape local minima
            if no_improve % 500 == 499:
                x = x * (1 + 0. * rng.standard_normal(n))  # no-op, just timing
                # Block perturbation
                w = int(rng.integers(10, max(20, n//50)))
                s = int(rng.integers(0, n - w))
                x[s:s+w] *= np.exp(rng.normal(0, 0.3, w))
                x = np.clip(x, 0.001, 1000.0)
                s2 = safe_score(x)
