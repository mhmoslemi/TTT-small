        return a / s if s > 1e-15 else np.ones(n) / n

    def optimize(a0, max_it=50000, beta_start=50.0, beta_end=20000.0):
        a = a0.copy()
        best_sc = score(a)
        ba = a.copy()
        a_rev = a[::-1].copy()
        stale = 0

        for it in range(max_it):
            if time.monotonic() > deadline:
                break
            beta = beta_start + (beta_end - beta_start) * min(1.0, (it + 1) / max_it)
            c = conv_rr(a, a, sz1)
            w = np.exp(beta * (c - c.max()))
            ws = w.sum()
            if ws < 1e-30 or ws == 1e308:
                w = np.zeros_like(c)
                w[np.argmax(c)] = 1.0
            else:
                w /= ws

            fa = np.fft.rfft(w, sz2)
            fp = np.fft.rfft(a_rev, sz2)
            d = np.fft.irfft(fa * fp, sz2)
            g = 2.0 * d[n - 1:2 * n - 1]

            lr = 0.005 * (1.0 + it / 5000.0) ** (-0.5)
            a = feasible(a - lr * g)
            a_rev = a[::-1].copy()

            if it % 300 == 0:
                sc = score(a)
                if sc < best_sc - 1e-8:
                    best_sc = sc
                    ba = a.copy()
                    stale = 0
                else:
                    stale += 300
                if stale > 8000:
                    a = feasible(a + rng.normal(0, 0.02, n))
                    a_rev = a[::-1].copy()
                    stale = 0
        sc = score(a)
        if sc < best_sc:
            best_sc = sc
            ba = a.copy()
        return ba, best_sc

    inits = [feasible(parent)]
    i = np.arange(n, dtype=np.float64)

    for amp in [0.2, 0.4, 0.6, 0.8]:
        inits.append(feasible(1.0 + amp * np.cos(2 * np.pi * i / n)))
    for amp in [0.3, 0.5, 0.7]:
        inits.append(feasible(1.0 + amp * (2.0 * i / (n - 1) - 1.0) ** 2))
    for amp in [0.3, 0.5]:
        inits.append(feasible(1.0 + amp * np.cos(4 * np.pi * i / n)))
    for amp in [0.3, 0.5]:
        inits.append(feasible(1.0 - amp * np.sin(np.pi * i / (n - 1)) ** 2))
    inits.append(feasible(1.0 + 0.5 * np.cos(2 * np.pi * i / n) + 0.2 * np.cos(4 * np.pi * i / n)))

    best_a = inits[0].copy()
    best_sc = score(best_a)

    for init in inits:
        if time.monotonic() > deadline:
            break
        a_opt, sc = optimize(init)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt

    if time.monotonic() < deadline:
        lr_local = 0.03
        for _ in range(50000):
            if time.monotonic() > deadline:
                break
            a_try = feasible(best_a * np.exp(rng.normal(0, lr_local, n)))
            sc = score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_a = a_try
            elif _ > 10000 and _ % 5000 == 0:
                lr_local *= 0.7

    fs = evaluate_sequence(best_a.tolist())
    if not np.isfinite(fs):
        best_a = inits[0]
        fs = evaluate_sequence(best_a.tolist())
    return [float(x) for x in best_a]
