Actually, let me use a simpler approach that's known to work well:

**Mirror descent / exponentiated gradients on the simplex:**
- Maintain a in the simplex (sum = 1, a >= 0).
- At each step, compute the subgradient of f(a) = max_k c[k] at a.
- Update: a_new = a * exp(-lr * grad) / sum(a * exp(-lr * grad)).

This is the "Hedge" algorithm applied to our problem. It's simple, O(n) per step, and works on the simplex.

import time

def conv(a):
    """Full autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    # FFT size should be >= size, use next power of 2 for efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)[:size]
    return C

    deadline = time.monotonic() + budget_s - 5.0

    best_c = conv(best_a).max()
    best_score = 2 * n * best_c
    
    # Exponentiated gradient descent on simplex
    lr = 0.1  # learning rate
    patience = 0
    
    while time.monotonic() < deadline:
        # Compute autocorrelation
        c = conv(a)
        max_c = c.max()
        
        # Score
        score = 2 * n * max_c  # since sum = 1
        
        if score < best_score:
            best_a = a.copy()
            best_score = score
            patience = 0
        else:
            patience += 1
        
        if patience > 5000:
            # Reset learning rate or perturb
            lr = max(0.001, lr * 0.5)
            patience = 0
        
        # Gradient of max_c w.r.t. a
        # dc[k*]/da[j] = a[k*-j] if 0 <= k*-j < n, with factor 2 if k*=2j
        k_star = c.argmax()
        grad = np.zeros(n)
        # For each j, the contribution is a[(k_star - j)] if 0 <= k_star - j < n
        # But we need to be careful about the factor of 2
        # c[k] = sum_i a[i] * a[k-i]
        # dc[k]/da[j] = a[k-j] + a[k-j] = 2*a[k-j] if j != k-j
        #              = a[k-j] * 2 = 2*a[j] if j = k-j (i.e., k = 2j)
        # Actually dc[k]/da[j] = a[k-j] if 0<=k-j<n (counting both i=j and k-i=j)
        # But if j = k-j (i.e., k=2j), the term a[j]*a[j] contributes 2*a[j] to dc/d a[j]
        # So: dc[k]/da[j] = a[k-j] if 0 <= k-j < n and j != k-j
        #                      2*a[j] if k = 2*j
        #                      0 otherwise
        
        # Actually, let me just compute it directly:
        # dc[k*]/da[j] for all j:
        low = max(0, k_star - (n-1))
        high = min(k_star, n-1)
        # j ranges over [low, high] for i=j, and over [k_star-(n-1), k_star] for k-i=j
        # Combined: j in [max(0, k_star-n+1), min(k_star, n-1)]
        # dc/d a[j] = a[k_star - j] (if 0 <= k_star - j < n)
        # But this counts the term once. The full derivative:
        # c[k] = sum_{i in valid range} a[i] * a[k-i]
        # d/d a[j]: if j is in the valid range as i, get a[k-j]
        #           if j is in the valid range as k-i (i.e., i = k-j in valid range), get a[i] = a[k-j]
        # These are the same when j = k-j.
        # So dc/da[j] = a[k-j] if 0<=k-j<n and j in [0,n-1]
        # But if j = k-j (k=2j), it's 2*a[j] because the same variable appears twice.
        
        # Simpler: create an array where grad[j] = a[k_star - j] for valid j
        valid_j = np.arange(max(0, k_star - (n-1)), min(k_star, n-1) + 1)
        # Wait, for c[k] to have a[j] as a factor, we need:
        # - j as i: need 0 <= k-j < n, i.e., k-n+1 <= j <= k
        # - j as (k-i): need 0 <= i < n where i = k-j, so 0 <= k-j < n, same condition
        # So j must satisfy: 0 <= j < n AND k-n+1 <= j <= k
        j_min = max(0, k_star - (n-1))
        j_max = min(n-1, k_star)
        if j_min <= j_max:
            idx = np.arange(j_min, j_max + 1)
            grad[idx] = a[k_star - idx]
            # Fix diagonal: where k_star - j == j, i.e., j == k_star/2
            if k_star % 2 == 0:
                mid = k_star // 2
                if j_min <= mid <= j_max:
                    grad[mid] = 2 * a[mid]
        
        # Exponentiated gradient update
        a_new = a * np.exp(-lr * grad)
        s = a_new.sum()
        if s > 0 and np.isfinite(s):
            a = a_new / s
        else:
            # Fallback: uniform perturbation
            a = a * np.exp(-lr * rng.normal(0, 0.01, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        # Occasionally add random perturbation to escape local minima
        if rng.random() < 0.01:
            a *= np.exp(rng.normal(0, 0.05, n))
            a = np.maximum(a, 1e-10)
            a /= a.sum()
        
        # Adaptive learning rate
        if score < best_score:
            lr = min(1.0, lr * 1.02)
        else:
            lr = max(0.001, lr * 0.999)
    
    # Convert back: scale so that evaluate_sequence gives the best score
    # Since score is scale-invariant, we can return a directly
    # But need to make sure sum > 0.01 for the evaluator
    best_a = np.maximum(best_a, 0)
    if best_a.sum() < 0.01:
        best_a = np.ones(n) / n
    # Scale to be in reasonable range
    best_a = best_a / best_a.sum() * np.sqrt(2 * n)
    
    return [float(x) for x in np.clip(best_a, 0, 1000)]

import numpy as np
import time

def autoref(a):
    """Full linear autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 5.0
    
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000).astype(np.float64)
    n = a.size
    S = a.sum()
    if S < 0.01:
        a = np.ones(n) / n
    else:
        a /= S  # normalize to sum = 1
    
    def score(a):
        c = autoref(a)
        return 2.0 * n * c.max()  # sum=1
    
    best_a = a.copy()
    best_score = score(best_a)
    
    # Soft parameter
    beta = 50.0
    lr = 0.05
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = autoref(a)
        smax = c.sum()
        if smax < 1e-30:
            break
        
        # Soft-max gradient
        # Use log-sum-exp for stability
        mc = c.max()
        exp_bc = np.exp(beta * (c - mc))
        w = exp_bc / exp_bc.sum()  # weights
        # Gradient of soft-max: sum_k w_k * dc[k]/da
        # For brevity, use the top-50 lags
        top_k = 50
        idx = np.argsort(c)[-top_k:]  # top 50 lags
        # Weight within top-k
        top_c = c[idx]
        top_mc = top_c.max()
        top_w = np.exp(beta * (top_c - top_mc))
        top_w /= top_w.sum()
        
        # Compute gradient: sum_k top_w[k] * dc[idx[k]]/da
        grad = np.zeros(n)
        for w_k, k in zip(top_w, idx):
            j_min = max(0, k - (n-1))
            j_max = min(n-1, k)
            if j_min <= j_max:
                jj = np.arange(j_min, j_max + 1)
                vals = a[k - jj]
                if k % 2 == 0:
                    mid = k // 2
                    if j_min <= mid <= j_max:
                        vals[mid - j_min] *= 2
                grad += w_k * vals
        
        # Projected gradient descent on simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new /= a_new_s
        
        # Check score
        new_score = 2.0 * n * autoref(a_new).max()
        
        if new_score < best_score:
            best_a = a_new.copy()
            best_score = new_score
            no_improve = 0
            lr = min(0.5, lr * 1.01)
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                lr = max(0.001, lr * 0.9)
                beta *= 1.1
            if no_improve > 2000:
                # Perturb
                a *= np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 1e-10)
                a /= a.sum()
                no_improve = 0
                lr = 0.05
                beta = 50.0
        
        a = a_new
    
    # Final: scale best_a to a reasonable range
    best_a = np.maximum(best_a, 0)
    bs = best_a.sum()
    if bs < 0.01:
        best_a = np.ones(n) / n
    else:
