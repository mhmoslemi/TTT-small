import numpy as np
import time


def _score_fast(x):
    """Quick score using FFT-based convolution for speed during search."""
    n = len(x)
    f = np.fft.rfft(x, 2 * n - 1)
    conv = np.fft.irfft(f * f)[:2 * n - 1]
    max_b = np.max(conv)
    s = np.sum(x)
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    if n == 0 or not np.isfinite(parent).all():

        n = 256

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    uni = np.ones(n, dtype=np.float64)
    s = float(evaluate_sequence(uni.tolist()))
    if s < best_score:
        best = uni.copy()
        best_score = s

    # Local search: multiplicative block perturbations + gradient step
    reached_target = best_score <= 1.50274
    stagnation = 0
    best_iter = 0

    while time.monotonic() < deadline and not reached_target:

            peak_lag = int(np.argmax(conv))
            # Gradient of conv[peak_lag] w.r.t. best
            grad = np.zeros(n, dtype=np.float64)
            lo = max(0, peak_lag - n + 1)
            hi = min(n - 1, peak_lag)
            for j in range(lo, hi + 1):
                idx = peak_lag - j

            alpha = 0.01 * best.sum() / (np.linalg.norm(grad) + 1e-12)
            c2 = best - alpha * grad
            c2 = np.clip(c2, 0.0, 1000.0)
