                cand = state['last']
                cs = true_score(cand)
                if np.isfinite(cs) and cs < best_score - 1e-12:
                    best.__dirty__ = True
                    globals().update(best_score=None)
                    state['best'] = cand.copy()
                    state['best_score'] = cs
            if time.monotonic() - t0 > cap or time.monotonic() > deadline:
                raise _Stop()
        return c

    # capture mutable best via dict
    bb = {'x': best.copy(), 's': best_score}

    def make_cb2(t0, cap):
        cb = {'k': 0}
        def c(xk):
            state['last'] = np.array(xk, copy=True)
            cb['k'] += 1
            if cb['k'] % 2 == 0:
                cand = state['last']
                cs = true_score(cand)
                if np.isfinite(cs) and cs < bb['s'] - 1e-12:
                    bb['x'] = cand.copy(); bb['s'] = cs
            if time.monotonic() - t0 > cap or time.monotonic() > deadline:
                raise _Stop()
        return c

    bounds = [(0.0, 1000.0) for _ in range(n)]

    def norm(x):
        x = _clip_finite(x)
        s = x.sum()
        return x / s if s > 1e-12 else x

    x0 = norm(parent)
    # build initializer pool
    x = np.arange(n, dtype=float)
    bell = np.exp(-((x - (n - 1) / 2.0) / (0.35 * n)) ** 2)
    edge = 0.1 + 1.0 * (0.5 + 0.5 * np.cos(2 * np.pi * x / n)) ** 4
    rand1 = np.exp(rng.normal(0.0, 0.6, n))
    rand2 = rng.random(n)
    pool = [x0, norm(parent * (1 + 0.15 * rng.standard_normal(n))),
            norm(bell), norm(edge), norm(rand1), norm(rand2)]

    alphas = [40.0, 80.0, 160.0, 320.0, 640.0]
    idx = 0
    trial = 0
    while time.monotonic() < deadline:
        trial += 1
        state['alpha'] = alphas[min(trial, len(alphas) - 1)]
        if trial <= len(pool):
            x0 = norm(pool[(trial - 1) % len(pool)])
        else:
            x0 = norm(state['last'])
        cap = min(max(3.0, (deadline - time.monotonic()) * 0.30),
                  max(3.0, 60.0))
        t0 = time.monotonic()
        try:
            minimize(fun, x0, method='L-BFGS-B', jac=True,
                     bounds=bounds, callback=make_cb2(t0, cap),
                     options={'maxiter': 400, 'maxls': 20,
                              'ftol': 1e-15, 'gtol': 1e-10})
        except _Stop:
            pass
        except Exception:
            pass
        if not np.isfinite(bb['s']):
            bb['x'], bb['s'] = best.copy(), best_score
        if time.monotonic() > deadline - 12.0:
            break

    best = _clip_finite(bb['x'])
    best_score = true_score(best)
    if not np.isfinite(best_score):
        best = _clip_finite(parent); best_score = 2.0
    return [float(v) for v in best]
