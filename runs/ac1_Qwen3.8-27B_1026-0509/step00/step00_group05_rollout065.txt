import numpy as np
import time


def _project_simplex(v):
    x = np.asarray(v, dtype=np.float64)
    if np.isfinite(x).all() and x.ndim == 1:
        n = x.size
        mu = np.sort(x)[::-1]
        sv = np.cumsum(mu)
        rho = np.nonzero(mu * np.arange(1, n + 1) > (sv - 1.0))[0][-1:]
        if rho.size:
            r = rho[0] + 1
            theta = (sv[r - 1] - 1.0) / r
            y = np.maximum(x - theta, 0.0)
            s = y.sum()
            if s > 0:
                return y / s
    return np.ones_like(x)
