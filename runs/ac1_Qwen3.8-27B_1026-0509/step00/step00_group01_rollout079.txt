    def aut(x):
        X = np.fft.fft(x, fsize)
        c = np.fft.ifft(X * X, fsize).real[:conv_len]
        np.maximum(c, 0.0, out=c)
        return c

    def fast_score(c):
        return 2.0 * n * float(c.max())

    def softmax(y):
        y = np.clip(y, -10.0, 10.0)
        m = y.max()
        e = np.exp(y - m)
        s = e.sum()
        return e / s

    starts = []

    def consider_x(x):
        x = np.asarray(x, dtype=np.float64).ravel()
        x = np.clip(x, 0.0, 1000.0)
        s = x.sum()
        if x.size != n or not np.all(np.isfinite(x)) or s <= 1e-12:
            return
        x = x / s
        c = aut(x)
        sc = fast_score(c)
        if np.isfinite(sc):
            starts.append((sc, x.copy()))

    def consider_y(y):
        y = np.asarray(y, dtype=np.float64).ravel()
        if y.size != n or not np.all(np.isfinite(y)):
            return
        consider_x(softmax(y))

    consider_x(x0)

    idx = np.arange(n, dtype=np.float64)
    t = idx / (n - 1) if n > 1 else np.zeros(n)
    u = 2.0 * t - 1.0
    q = u * (1.0 - u * u)
    s1 = np.sin(2.0 * np.pi * t)

    fixed = [
        (2.0, 1.0, 0.0, 0.0),
        (1.8, 1.0, 0.0, 0.0),
        (2.2, 1.2, 0.0, 0.0),
        (2.4, 1.4, 0.0, 0.0),
        (1.5, 0.8, 0.0, 0.0),
        (2.0, 0.8, 0.5, 0.3),
        (2.0, 0.8, -0.5, 0.3),
        (1.2, 0.6, 0.0, 0.0),
        (2.7, 1.6, 0.0, 0.0),
    ]
    for A, B, D, S in fixed:
        if time.monotonic() > deadline - 8.0:
            break
        consider_y(-A * u * u + B * u + D * q + S * s1)

    A_vals = (0.4, 0.8, 1.4, 2.0, 2.8, 3.8)
    B_vals = (-1.8, -1.0, -0.4, 0.0, 0.4, 1.0, 1.8, 2.6)
    D_vals = (-0.8, 0.0, 0.8)
    S_vals = (-0.6, 0.0, 0.6)
    maxc = 260
    cnt = 0
    stop = time.monotonic() > deadline - 10.0
    for A in A_vals:
        if stop:
            break
        aa = -A * u * u
        for B in B_vals:
            if stop:
                break
            ab = aa + B * u
            for D in D_vals:
                if stop:
                    break
                abq = ab + D * q
                for S in S_vals:
                    if cnt >= maxc or stop:
                        stop = True
                        break
                    cnt += 1
                    consider_y(abq + S * s1)

    if time.monotonic() < deadline - 12.0:
        for _ in range(80):
            if time.monotonic() > deadline - 10.0:
                break
            A = rng.uniform(0.6, 3.0)
            B = rng.uniform(-2.0, 2.0)
            D = rng.uniform(-1.0, 1.0)
            S = rng.uniform(-0.8, 0.8)
            y = -A * u * u + B * u + D * q + S * s1
            for m in (1, 2):
                y = y + rng.uniform(0.0, 1.0) * np.sin(2.0 * np.pi * m * t + rng.uniform(0.0, 2.0 * np.pi))
            consider_y(y)

    if not starts:
        starts = [(2.0, x0.copy())]
    starts.sort(key=lambda z: z[0])

    best_internal = float(starts[0][0])
    best_x = starts[0][1].copy()

    def optimize(x_init, limit):
        limit = min(limit, deadline - 6.0)
        y = np.log(np.maximum(x_init, 1e-300))
        y -= y.mean()
        y = np.clip(y, -10.0, 10.0)
        x = softmax(y)
        c = aut(x)
        cmax = float(c.max())
        bx = x.copy()
        bs = 2.0 * n * cmax

        lr = 0.2
        min_lr = 1e-3
        max_lr = 2.0
        ef = 0.8
        min_ef = 0.01
        noimp = 0
        rej = 0
        it = 0

        while time.monotonic() < limit:
            s = 2.0 * n * cmax
            if s < bs - 1e-12:
                bs = s
                bx = x.copy()
                noimp = 0
            else:
                noimp += 1

            if noimp > 600:
                y = y + rng.normal(0.0, 0.25, n)
                y -= y.mean()
                y = np.clip(y, -10.0, 10.0)
                x = softmax(y)
                c = aut(x)
                cmax = float(c.max())
                ef = 1.0
                lr = 0.2
                noimp = 100

            eps = max(1e-15, ef * cmax)
            sc = (c - cmax) / eps
            ex = np.exp(sc)
            es = float(ex.sum())
            if not np.isfinite(es) or es <= 0.0:
                break
            w = ex / es

            FR = np.fft.fft(x[::-1], gsize)
            FW = np.fft.fft(w, gsize)
            h = np.fft.ifft(FR * FW, gsize).real
            g = 4.0 * n * h[n - 1:2 * n - 1]
            g -= g.mean()
            gs = float(np.max(np.abs(g)))
            if gs <= 1e-14:
                break
            g /= gs

            curp = 2.0 * n * eps * (cmax + np.log(es))
            yt = np.clip(y - lr * g, -10.0, 10.0)
            xt = softmax(yt)
            ct = aut(xt)
            cmx = float(ct.max())
            sc_t = (ct - cmx) / eps
            es_t = float(np.exp(np.minimum(sc_t, 0.0)).sum())
            newp = 2.0 * n * eps * (cmx + np.log(max(es_t, 1e-300)))

            if newp < curp:
                y = yt
                x = xt
                c = ct
                cmax = cmx
                lr = min(lr * 1.12, max_lr)
                if (it & 15) == 15 and ef > min_ef:
                    ef *= 0.985
                rej = 0
            else:
                lr = max(lr * 0.55, min_lr)
                rej += 1
                if rej > 25:
                    y = y + rng.normal(0.0, 0.08, n)
                    y -= y.mean()
                    y = np.clip(y, -10.0, 10.0)
                    x = softmax(y)
                    c = aut(x)
                    cmax = float(c.max())
                    ef = max(ef * 1.3, 0.3)
                    lr = 0.2
                    rej = 0
            it += 1

        x = softmax(y)
        c = aut(x)
        s = 2.0 * n * float(c.max())
        if s < bs:
            bs = s
            bx = x.copy()
        return bx, float(bs)

    top = []
    for sc, x in starts:
        if len(top) >= 3:
            break
        if any(np.max(np.abs(x - y)) < 1e-3 for y in top):
            continue
        top.append(x)
    if not top:
        top = [starts[0][1].copy()]

    fracs = (0.6, 0.3, 0.1)
    for i, xs in enumerate(top):
        now = time.monotonic()
        rem = deadline - now - 6.0
        if rem < 15.0:
            break
        frac = fracs[i] if i < len(fracs) else 0.1
        lim = now + rem * frac
        if len(top) > 1 and i == len(top) - 1:
            lim = deadline - 8.0
        bx, bs = optimize(xs, lim)
        if bs < best_internal:
            best_internal = bs
            best_x = bx

    if time.monotonic() < deadline - 25.0 and best_internal > 1.30:
        ybest = np.log(np.maximum(best_x, 1e-300))
        ybest -= ybest.mean()
        ybest = np.clip(ybest, -10.0, 10.0)
        for _ in range(2000):
            if time.monotonic() > deadline - 12.0:
                break
            z = ybest + rng.normal(0.0, 0.08, n)
            z -= z.mean()
            z = np.clip(z, -10.0, 10.0)
            x = softmax(z)
            sc = fast_score(aut(x))
            if sc < best_internal:
                best_internal = sc
                best_x = x.copy()
                ybest = z

    final_candidates = [best_x, x0]
    final_candidates.extend(top[:3])

    final = x0.copy()
    fin = float("inf")
    for cand in final_candidates:
        if time.monotonic() > deadline - 1.0:
            break
        cand = np.asarray(cand, dtype=np.float64).ravel()
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.size == 0 or not np.all(np.isfinite(cand)) or cand.sum() < 0.01:
            continue
        try:
            sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            sc = float("inf")
        if np.isfinite(sc) and sc < fin:
            fin = sc
            final = cand

    if not np.isfinite(fin):
        final = x0
    final = np.array(final, dtype=np.float64, copy=True).ravel()
    final = np.clip(final, 0.0, 1000.0)
    if final.size == 0 or not np.all(np.isfinite(final)):
        final = np.ones(256, dtype=np.float64)
    return [float(v) for v in final]
