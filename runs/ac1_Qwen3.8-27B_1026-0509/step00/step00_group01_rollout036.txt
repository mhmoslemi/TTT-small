        a = a / a.sum()
        t0 = time.monotonic()
        for _ in range(1000):
            if time.monotonic() > deadline - 10 or time.monotonic() - t0 > max_time:
                break
            b = np.convolve(a, a)
            M = b.max()
            A = np.zeros((2 * n - 1, n))
            for j in range(n):
                A[j:j + n, j] = a[j]
            res = optimize.linprog(-np.ones(n), A_ub=A,
                b_ub=np.full(2 * n - 1, M), bounds=[(0, None)] * n,
                method='highs', options={'time_limit': min(30, max_time)})
            if not res.success:
                break

                break

            a = 0.5 * a + 0.5 * g

            if a.sum() < 1e-10:

            idx = np.arange(n)
