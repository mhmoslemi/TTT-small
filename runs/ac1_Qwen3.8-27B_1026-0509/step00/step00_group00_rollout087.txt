from scipy.fft import next_fast_len, fft, ifft

def _conv(a):
    a = np.asarray(a, dtype=np.float64)

    if n < 250:
        return np.convolve(a, a, mode='full')
    N = next_fast_len(2*n - 1)

    c = A * A

    return float(2.0 * conv.max() / n)

def _init_profile(nm, rng, center=0.43, width=0.30, tilt=0.5):

from scipy.fft import next_fast_len, fft, ifft


def _conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n < 512:
        return np.convolve(a, a, mode="full")
    N = next_fast_len(2 * n - 1)
    A = fft(a, N)
    return np.real(ifft(A * A))[:2 * n - 1]


def _score(conv, n):
    return float(2.0 * conv.max() / n)

    a = bell * ramp
    a = np.clip(a, 0.0, 1000.0)

        except Exception:
            s = np.inf
        if np.isfinite(s) and s < final_score:
            final_score = s
            final_best = A.copy()

    A = np.clip(np.asarray(final_best, dtype=np.float64).ravel(), 0.0, 1000.0)
    if A.size < 1 or A.sum() < 0.01 or not np.all(np.isfinite(A)):
        A = np.clip(par, 0.0, 1000.0)
        if A.sum() > 1e-9:
            A = A * (A.size / A.sum())
        else:
            A = np.ones(4806, dtype=np.float64)
    s = float(evaluate_sequence(A.tolist()))
    if not np.isfinite(s):
        A = np.clip(par, 0.0, 1000.0)
        if A.sum() > 1e-9:
            A = A * (A.size / A.sum())
        else:
            A = np.ones(4806, dtype=np.float64)
    return [float(x) for x in A]
