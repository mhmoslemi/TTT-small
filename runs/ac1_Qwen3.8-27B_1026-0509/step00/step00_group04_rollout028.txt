The constraint is: for each k = 0, ..., 2n-2, (A@g)[k] <= M where A[k,j] = x[k-j] if 0 <= k-j < n, else 0.

So A[k,j] = x[k-j] * I(0 <= k-j < n) * I(0 <= j < n).

For a fixed column j, the nonzeros are at rows k where 0 <= k-j < n, i.e., j <= k < j+n. Since k ranges from 0 to 2n-2, the valid rows are max(0,j) to min(2n-2, j+n-1) = j to min(2n-2, j+n-1).

For j in 0..n-1: j+n-1 <= 2n-2 (since j <= n-1), so rows are j to j+n-1, count = n.

The values in column j are: A[j,j] = x[0], A[j+1,j] = x[1], ..., A[j+n-1,j] = x[n-1].

So column j is [x[0], x[1], ..., x[n-1]] starting at row j. This confirms my earlier analysis.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).ravel()
    except Exception:
        parent = None
    if parent is None or parent.size == 0 or not np.all(np.isfinite(parent)):
        n = 3809
        parent = np.ones(n)
    else:
        n = parent.size

    best = parent.astype(np.float64).copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Precompute sparse structure indices (column j has x[0..n-1] at rows j..j+n-1)
    all_rows = np.empty(n * n, dtype=np.int32)
    all_cols = np.empty(n * n, dtype=np.int32)
    for j in range(n):
        all_rows[j*n:(j+1)*n] = np.arange(j, j + n, dtype=np.int32)
        all_cols[j*n:(j+1)*n] = j

    def solve_lp(x, M, tl):
        data = np.tile(x, n)
        A = csr_matrix((data, all_rows, all_cols), shape=(2*n-1, n))
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                      method='highs', options={'time_limit': tl})
        if res.success and res.x is not None and np.all(np.isfinite(res.x)):
            return np.asarray(res.x, dtype=np.float64)
        return None

    # Phase 1: diverse initializations
    i = np.arange(n, dtype=np.float64)
    inits = [parent.copy()]
    inits.append(0.5*(1 + np.cos(2*np.pi*(i-(n-1)/2)/n)) + 0.15*rng.standard_normal(n))
    inits.append(np.exp(-((i - n*0.5)**2) / (2*(n/5)**2)))
    inits.append(np.abs(rng.standard_normal(n)) + 0.1)
    two_lvl = np.where((i % 3) < 1, 1.0, 0.4)
    inits.append(two_lvl)
    gaps = np.ones(n); gaps[::4] = 0.3
    inits.append(gaps)
    inits.append(0.5*(1 + np.cos(4*np.pi*i/n)))
    for cand in inits:
        cand = np.clip(cand.astype(np.float64), 0, 1000)
        if np.sum(cand) > 0.01:
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = cand.copy()
                best_score = sc

    # Phase 2: LP-based refinement
    stall = 0
    for it in range(5000):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        x = best.copy()
        s = float(np.sum(x))
        if s < 0.01:
            break
        x_n = x / s
        b = np.convolve(x_n, x_n)
        M = float(np.max(b))

        tl = max(1.0, min(15.0, remaining / 4.0))
        try:
            g = solve_lp(x_n, M, tl)
        except Exception:
            g = None

        if g is not None:
            sg = float(np.sum(g))
            if sg > 0.01 and np.count_nonzero(g > 1e-10) > n // 4:
                g_n = g / sg
                improved = False
                for t in (0.5, 0.3, 0.7, 0.2, 0.8, 0.1):
                    x_new = np.clip((1-t)*best + t*s*g_n, 0, 1000)
                    sc = float(evaluate_sequence(x_new.tolist()))
                    if np.isfinite(sc) and sc < best_score - 1e-10:
                        best = x_new.copy()
                        best_score = sc
                        improved = True
                        break
                if improved:
                    stall = 0
                else:
                    stall += 1
        else:
            stall += 1

        if stall >= 25:
            best = np.clip(best * (1 + 0.15*rng.standard_normal(n)), 0, 1000)
            sc = float(evaluate_sequence(best.tolist()))
            if np.isfinite(sc):
                best_score = sc
            stall = 0

    # Phase 3: local perturbation
    while time.monotonic() < deadline - 5:
        x = best.copy()
        nb = int(rng.integers(1, 5))
        for _ in range(nb):
            w = int(rng.integers(1, max(2, n//40)))
            pos = int(rng.integers(0, max(1, n - w)))
            x[pos:pos+w] *= np.exp(float(rng.normal(0, 0.06)))
        x = np.clip(x, 0, 1000)
        sc = float(evaluate_sequence(x.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = x.copy()
            best_score = sc

    return [float(v) for v in best]
